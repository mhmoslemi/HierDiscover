import time, math, numpy as np

def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    try:
        init = np.array(height_sequence_1, dtype=float)
    except Exception:
        init = np.ones(2100, dtype=float)

    n = init.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # ensure nonnegative, clip
    a = np.maximum(init, 0.0)
    a = a / a.sum() * target_sum

    best = a.copy()
    best_score = evaluate_sequence(best.tolist())

    iteration = 0
    lr0 = 0.3
    lr_decay = 1e-5
    alpha_start = 0.2
    alpha_end = 15.0
    alpha_ramp = 200000  # number of iterations over which to increase

    while time.time() < deadline:
        # compute learning rate
        lr = lr0 / (1 + lr_decay * iteration)

        # compute alpha
        if iteration < alpha_ramp:
            alpha = alpha_start + (alpha_end - alpha_start) * (iteration / alpha_ramp)
        else:
            alpha = alpha_end

        # compute convolution
        b = np.convolve(a, a)
        max_b = b.max()
        # smoothed max
        shift = b - max_b  # <=0
        # compute softmax weights
        # use np.exp(alpha * shift) to avoid overflow, shift negative
        exp_w = np.exp(alpha * shift)
        p = exp_w / exp_w.sum()

        # gradient of smoothed max
        grad = 2.0 * np.correlate(p, a, mode='valid')  # shape n

        # gradient descent step
        a_new = a - lr * grad

        # project onto nonnegative orthant
        a_new = np.maximum(a_new, 0.0)

        # renormalize
        s = a_new.sum()
        if s > 0:
            a_new = a_new * (target_sum / s)
        else:
            a_new = a.copy()

        # Clip values (0 to 1000)
        np.clip(a_new, 0.0, 1000.0, out=a_new)

        # jitter
        if iteration % 5000 == 0 and iteration > 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a_new = np.maximum(a_new + jitter, 0.0)
            a_new = a_new * (target_sum / a_new.sum())

        # evaluate
        cur_score = evaluate_sequence(a_new.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best = a_new.copy()
            print(f"[iter {iteration}] new best {best_score:.6f}")

        # accept step (greedy)
        if cur_score <= best_score or np.random.rand() < 0.01:
            a = a_new

        iteration += 1

    print("[done] iterations", iteration, "best", best_score)
    return best.tolist()

"""
Propose a height sequence by minimizing the maximal self‑convolution value.

The evaluation routine returns
    E = 2 * n * max_k (a * a)[k] / (sum(a)^2).

If we keep sum(a) = sqrt(2 n) (the “standard” normalisation) then
    E = max_k (a * a)[k] .
Thus we only have to make the largest entry of the autocorrelation
as small as possible.

The algorithm works on the simplex (non‑negative entries with fixed sum)
and minimises a *smooth* approximation of the max using a soft‑max
( log‑sum‑exp ) surrogate.  The gradient of the surrogate is easy to
compute via a correlation (so each iteration is O(n log n) with NumPy).

We start from the best known construction (``height_sequence_1`` if it
exists) or a uniform vector, then perform a projected (additive) gradient
descent with an annealing schedule:

* the smoothing parameter `alpha` starts small (very smooth) and is
  increased slowly – this first spreads the mass uniformly,
  later focuses on the true maximum;
* the learning rate decays with the iteration counter;
* every few thousand steps a tiny random jitter is added to escape
  shallow plateaux;
* the best sequence seen so far (according to the exact
  ``evaluate_sequence``) is remembered and finally returned.

The whole search runs for the allotted wall‑clock time (default 1000 s)
and respects the limits on the individual heights (0 ≤ h ≤ 1000).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full discrete convolution a * a (length 2·len(a)-1)."""
    return np.convolve(a, a)


def _softmax_weights(b: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute the soft‑max weights for a vector `b`:

        w_k = exp(alpha * (b_k - max(b))) / sum_j exp(alpha * (b_j - max(b)))

    The subtraction of max(b) guarantees numerical stability.
    """
    shift = b - b.max()          # non‑positive numbers
    # For very large alpha the exponentials may under‑flow;
    # NumPy handles this safely (result becomes 0).
    exp_vals = np.exp(alpha * shift)
    total = exp_vals.sum()
    if total == 0.0:         # should not happen, but guard against division by zero
        return np.full_like(b, 1.0 / b.size)
    return exp_vals / total


def _smooth_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smooth maximum (log‑sum‑exp) of the convolution a*a
    with respect to `a`.  The result is a vector of length n (= len(a)).
    """
    b = _conv_full(a)                       # b has length 2n-1
    w = _softmax_weights(b, alpha)          # same length as b
    # gradient_i = 2 * sum_j a_j * w_{i+j}
    # This is exactly the (valid) cross‑correlation of w with a.
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad


def _project_and_rescale(a: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project onto the non‑negative orthant, clip to the limits used by
    ``evaluate_sequence`` and finally renormalise the vector so that
    sum(a) == target_sum.
    """
    a = np.maximum(a, 0.0)               # non‑negative
    np.clip(a, 0.0, 1000.0, out=a)       # evaluation caps
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list:
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation function.  Returns the best sequence found before the
    time limit expires.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # keep a safety margin

    # ------------------------------------------------------------------
    # Initialise the candidate vector.
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
    except Exception:
        # fallback: uniform vector of a moderately large size
        base = np.ones(2100, dtype=float)

    n = base.shape[0]
    target_sum = math.sqrt(2.0 * n)                # the normalisation used by the paper

    a = np.maximum(base, 0.0)
    a = a * (target_sum / a.sum())                # scale to the required L1 norm

    # Keep track of the best sequence seen so far.
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters (tuned empirically)
    iteration = 0
    lr0 = 0.8                # initial learning rate
    lr_decay = 5e-5          # slower decay than the previous code
    alpha_start = 0.05
    alpha_end = 12.0
    alpha_ramp_iters = 250000   # number of iterations over which alpha is increased

    # Main optimisation loop
    while time.time() < deadline:
        # ----- learning‑rate and smoothing schedule -----
        lr = lr0 / (1.0 + lr_decay * iteration)

        if iteration < alpha_ramp_iters:
            alpha = alpha_start + (alpha_end - alpha_start) * (iteration / alpha_ramp_iters)
        else:
            alpha = alpha_end

        # ----- compute a gradient of the smooth objective -----
        grad = _smooth_gradient(a, alpha)

        # ----- one projected gradient step -----
        a_new = a - lr * grad
        a_new = _project_and_rescale(a_new, target_sum)

        # ----- occasional random jitter to explore -----
        if iteration % 5000 == 0 and iteration > 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a_new = _project_and_rescale(a_new + jitter, target_sum)

        # ----- evaluate the (exact) objective -----
        try:
            cur_score = evaluate_sequence(a_new.tolist())
        except Exception:   # defensive programming – should never happen
            cur_score = float("inf")

        # ----- keep the best ever seen -----
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            print(f"[iter {iteration:07d}] new best = {best_score:.6f}")

        # ----- accept the step if it improves (or very rarely accept worse) -----
        if cur_score <= best_score or np.random.rand() < 0.003:
            a = a_new

        iteration += 1

    print(f"[done] iterations={iteration}, best score={best_score:.6f}")
    return best_seq.tolist()
