import time, math, numpy as np

def _conv_fft(a):
    n = a.shape[0]
    L = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    b = np.fft.irfft(fa * fa, n=L)
    return b[:2*n-1]

def _objective(a):
    n = a.shape[0]
    conv = _conv_fft(a)
    M = conv.max()
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * M / (S * S))

def _active_set(conv, eps=1e-12):
    M = conv.max()
    active = np.where(conv >= M - eps)[0]
    return M, active

def _rev_total(a, active):
    n = a.shape[0]
    rev = np.zeros_like(a)
    idx = np.arange(n)
    for k in active:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    return rev

def _project_normalize(a, target_sum):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    else:
        # should not happen; return uniform fallback
        a = np.full_like(a, target_sum / a.shape[0])
    np.clip(a, 0.0, 1000.0, out=a)
    return a

    The optimiser uses an Adam‑style projected sub‑gradient descent where the
    sub‑gradient is obtained from the full set of convolution entries attaining
    the current maximum.  Random jitter is added periodically to escape plateaus.
    The search runs until the allotted time budget expires and returns the best
    sequence encountered.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0   # keep a safety margin

    # ------------------------------------------------------------------
    # Initialise the problem size and a starting point
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]

    n = base.shape[0]

    a = base * (target_sum / base.sum())

    best_score = _objective(best_seq)

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast linear (non‑circular) convolution using FFT
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=L)
    b = np.fft.irfft(fa * fa, n=L)
    return b[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Compute the value of 2·n·max(conv(a,a)) / (sum(a)**2).
    This mirrors ``evaluate_sequence`` but is much faster because it
    uses the FFT based convolution and avoids the safety checks.
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    M = conv.max()
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * M / (S * S))


# ----------------------------------------------------------------------
def _active_set(conv: np.ndarray, eps: float = 1e-12):
    """Return (M, active_indices) where M is the maximum of conv and
    active_indices are the positions attaining (or almost attaining) it."""
    M = conv.max()
    active = np.where(conv >= M - eps)[0]
    return M, active


# ----------------------------------------------------------------------
def _rev_total(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """For a given active set of convolution indices compute
    rev[p] = Σ_{k∈active} a_{k‑p} (0 if the index is out of range)."""
    n = a.shape[0]
    rev = np.zeros_like(a)
    idx = np.arange(n)
    for k in active:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    return rev


# ----------------------------------------------------------------------
def _project_normalize(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Project onto the non‑negative orthant and renormalise to the given sum."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    else:                                   # should never happen, but keep things safe
        a = np.full_like(a, target_sum / a.shape[0])
    np.clip(a, 0.0, 1000.0, out=a)
    return a

    a = _project_normalize(a, target_sum)

    best_seq = a.copy()
    best_score = _objective(best_seq)
    print(f"[init] n={n}, initial objective={best_score:.6f}")

    # ------------------------------------------------------------------
    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05
    decay = 1e-5

    iteration = 0

    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1. current convolution and active set
        conv = _conv_fft(a)
        M, active = _active_set(conv)

        # --------------------------------------------------------------
        # 2. sub‑gradient of the objective (full active set)
        rev = _rev_total(a, active)
        S = a.sum()
        const = (4.0 * n * M * len(active)) / (S ** 3)
        grad = (4.0 * n / (S ** 2)) * rev - const

        # --------------------------------------------------------------
        # 3. Adam update with projection
        lr = lr0 / (1.0 + decay * iteration)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = _project_normalize(a_new, target_sum)

        # --------------------------------------------------------------
        # 4. occasional tiny random jitter to escape shallow plateaus
        if iteration % 800 == 0 and iteration > 0:
            jitter = np.random.randn(n) * 0.002 * target_sum / n
            a_new = _project_normalize(a_new + jitter, target_sum)

        # --------------------------------------------------------------
        # 5. evaluate the candidate (fast version)
        cur_score = _objective(a_new)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            print(f"[iter {iteration:06d}] new best = {best_score:.6f}")

        # --------------------------------------------------------------
        # 6. accept the step if it does not worsen the best known value
        #    (small probability to accept a worse move for exploration)
        if cur_score <= best_score or np.random.rand() < 0.003:
            a = a_new

        iteration += 1

    print(f"[done] iterations={iteration}, best objective={best_score:.6f}")
    return best_seq.tolist()
