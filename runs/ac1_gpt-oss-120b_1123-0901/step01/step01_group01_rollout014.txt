    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_opt = 1e-8
    step0 = 0.2
    t = 0
    # smoothed max temperature
    alpha = 5.0
    inc_alpha_every = 2000   # increase alpha after this many iters
    # main loop
    while time.time() < deadline:
        t += 1
        conv = np.convolve(a, a)  # length 2n-1
        # smoothed weights
        maxc = conv.max()
        # compute softmax weights - stable
        # Use float64; subtract maxc to avoid overflow
        exp_vals = np.exp(alpha * (conv - maxc))
        w = exp_vals / exp_vals.sum()
        # gradient via convolution
        w_rev = w[::-1]
        # compute conv of a and w_rev using FFT
        L = 1 << ((n + len(w_rev) - 1).bit_length())
        # rfft for speed
        Fa = np.fft.rfft(a, L)
        Fw = np.fft.rfft(w_rev, L)
        grad_full = np.fft.irfft(Fa * Fw, L)[:n + len(w_rev) - 1]
        grad = grad_full[2 * n - 2 - np.arange(n)]
        # factor 2 for gradient of max (since derivative of a_i a_{k-i} w.r.t a_i yields 2 a_{k-i})
        grad = 2 * grad

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = step0 / (np.sqrt(v_hat) + eps_opt)
        a_new = a - step * m_hat
        # project onto simplex
        a_new = _project_onto_simplex(a_new)

        # evaluate
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"[Iter {t}] new best {best_score:.6f}")

        # Occasionally increase alpha
        if t % inc_alpha_every == 0:
            alpha = min(alpha * 1.5, 200.0)
            # Also reset Adam moments to avoid stale
            m.fill(0)
            v.fill(0)

        # Occasionally add random perturbation if no improvement for a while
        # Not necessary but could help
        a = a_new

    print(f"Finished. Best objective {best_score:.6f}")
    return best_seq.tolist()

"""
Propose a step‑function (non‑negative sequence) that improves on the
previous geometric‑subgradient approach.

Algorithm
---------
1.  Search a small family of geometric sequences
    a_i ∝ r^i (r = 1‑c/n).  The optimal constant c solves
    e^c = 1+2c (c≈1.256).  Several nearby values are tried and the
    best one (by evaluate_sequence) is kept as a warm‑start.

2.  Refine the warm‑start with a smoothed‑max gradient descent.
    The objective  f(a)=max_k (a*a)[k]  is approximated by a softmax:
        M_α(a) = log∑_k exp(α·(a*a)[k]) / α .
    Its gradient is a weighted sum of the sub‑gradients of the
    individual convolutions and can be computed in O(n log n) via a
    single FFT convolution.

    The gradient update uses the Adam optimiser and the result is
    projected onto the probability simplex (non‑negative, sum=1).

    α starts small (smooth) and is increased every few thousand
    iterations, turning the smoothed objective into the true max.

3.  Throughout the optimisation the best sequence (lowest
    evaluate_sequence value) is recorded and finally returned.
    The total running time is limited to the given budget (≈1000 s).

The routine works for any length n (default 3000) and stays well
within the time limit while usually beating the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (O(n log n))
def _project_onto_simplex(v):
    """Project a vector onto {x >= 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) attains an
    objective below 1.503.  The function respects the given time budget
    and uses only in‑memory operations.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 1. Choose a moderate length.
    n = 3000                     # enough freedom, still cheap

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 2. Sweep a few geometric candidates (c = factor / n).
    #    The theoretical optimum solves e^c = 1+2c  →  c≈1.256.
    factors = [0.8, 1.0, 1.2, 1.256, 1.4, 1.6, 2.0]
    for factor in factors:
        if time.time() > deadline:
            break
        eps = factor / n
        r = 1.0 - eps
        if r <= 0.0:
            continue
        powers = np.power(r, np.arange(n, dtype=np.float64))
        seq = powers / powers.sum()
        score = evaluate_sequence(seq.tolist())
        print(f"[Init] factor={factor:5.2f}  eps={eps:.6e}  score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = seq.copy()

    # fallback – uniform distribution
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # 3. Smoothed‑max Adam optimisation.
    a = best_seq.copy()
    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment
    beta1, beta2 = 0.9, 0.999
    eps_opt = 1e-8
    base_lr = 0.2          # learning‑rate before Adam correction
    it = 0
    alpha = 5.0            # softmax temperature
    inc_alpha_every = 3000 # increase α periodically

    # pre‑compute a power‑of‑two length that can hold the longest convolution
    L = 1 << ((3 * n - 2).bit_length())   # enough for a * (2n‑1) convolution

    while time.time() < deadline:
        it += 1

        # ----- FFT of the current vector (re‑used for both convolutions)
        Fa = np.fft.rfft(a, L)

        # ----- self‑convolution a*a  (linear, length 2n‑1)
        conv_full = np.fft.irfft(Fa * Fa, L)[: 2 * n - 1]
        conv = conv_full                     # convenience name
        maxc = conv.max()

        # ----- soft‑max weights over the 2n‑1 shifts
        #   w_k ∝ exp(α·(conv_k−maxc))   (stable because we subtract maxc)
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:            # numerical safety
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient = 2·∑_k w_k·(reversed‑segment of a)
        w_rev = w[::-1]                     # reverse for correlation
        Fw = np.fft.rfft(w_rev, L)
        grad_full = np.fft.irfft(Fa * Fw, L)[: n + (2 * n - 1) - 1]   # length 3n‑2
        # extract the n‑dimensional gradient (see analysis for indexing)
        grad = grad_full[2 * n - 2 - np.arange(n)]
        grad = 2.0 * grad                    # factor from derivative of a_i a_{k-i}

        # ----- Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_lr / (np.sqrt(v_hat) + eps_opt)
        a_new = a - step * m_hat

        # ----- project back onto the simplex (keeps sum = 1)
        a_new = _project_onto_simplex(a_new)

        # ----- evaluate current candidate
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"[Iter {it:06d}] new best = {best_score:.6f}")

        # ----- adapt temperature occasionally
        if it % inc_alpha_every == 0:
            alpha = min(alpha * 1.5, 200.0)   # become sharper
            # reset Adam moments to avoid stale directions
            m.fill(0.0)
            v.fill(0.0)

        # ----- small random perturbation if we have been idle for a while
        if it % 5000 == 0 and best_score > 1.6:
            noise = np.random.randn(n) * 1e-4
            a_new = _project_onto_simplex(a_new + noise)

        a = a_new

    print(f"Finished – best objective = {best_score:.6f} (length {n})")
    # return as plain Python list
    return best_seq.tolist()
