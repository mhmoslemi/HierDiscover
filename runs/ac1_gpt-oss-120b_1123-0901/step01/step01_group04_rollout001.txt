def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """Find step function minimizing evaluate_sequence using an
    iterative linear‑programming refinement over the arcsine distribution.
    The algorithm works in two phases:
    1. start from the discrete arcsine law (optimal for the continuous case);
    2. repeatedly solve a linear program that finds a direction
       g with larger L1 norm while keeping the convolution f*g below the
       current maximum.  The direction is blended with the current f.
    The loop stops when the time budget is exhausted or when no
    improvement is observed for several iterations.
    """
    import time, numpy as np
    try:
        from scipy.optimize import linprog
    except Exception:
        linprog = None

    # Determine length n
    np.random.seed(seed)
    n = kwargs.get("length")
    if n is None:
        try:
            n = len(height_sequence_1)
        except Exception:
            n = 1200   # moderate length, fast LP

    # Build arcsine distribution
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * ((2*k-1)*(2*k)) / (4.0*k*k)
    a /= a.sum()   # Normalize to sum=1

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print("[init] n=", n, "score=", best_score)

    if linprog is None:
        # fallback simple hill climbing...
        # Return after some iterations
        # Use simple random moves similar to earlier but integrated
        deadline = time.time() + budget_s - 0.5
        it=0
        while time.time()<deadline and it<5000:
            i,j = np.random.choice(n,2,replace=False)
            src,dst = (i,j) if a[i]>a[j] else (j,i)
            delta = 0.01 * min(a[src], a[dst])
            if delta <=0: continue
            new_a = a.copy()
            new_a[src] -= delta
            new_a[dst] += delta
            new_score = evaluate_sequence(new_a.tolist())
            if new_score < best_score:
                a = new_a
                best_seq = new_a
                best_score = new_score
                print("[improve] it",it,"score",best_score)
            it+=1
        return best_seq.tolist()

    # else we have linprog.

    # Parameter for mixing step.
    step = 0.4   # start fairly aggressive
    no_improve = 0
    max_no_improve = 15
    deadline = time.time() + budget_s - 2.0   # leave margin

    while time.time() < deadline:
        # compute current M
        conv_f_f = np.convolve(best_seq, best_seq)
        M = conv_f_f.max()
        # Build constraint matrix A (size (2n-1) x n)
        # Use_dense for speed
        rows = 2*n-1
        A = np.zeros((rows, n), dtype=np.float64)
        # Fill columns: each column i has a shifted copy of best_seq
        # rows i .. i+n-1 get best_seq[0:n]
        # We can do using broadcasting if possible:
        # but simple loop:
        for i in range(n):
            A[i:i+n, i] = best_seq
        # Right-hand side vector
        b = np.full(rows, M, dtype=np.float64)

        # Linear program: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n, dtype=np.float64)

        # Solve
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs', options={'presolve': True})
        except Exception as e:
            print("[lp fail] exception", e)
            break

        if not res.success:
            print("[lp fail] status", res.message)
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            print("[lp warn] zero sum")
            break

        # Scale to same L1 norm as best_seq (which is 1.0)
        scale = 1.0 / sum_g
        g_scaled = g * scale

        # Mix
        new_f = (1 - step) * best_seq + step * g_scaled
        # Ensure non‑negative (should be)
        new_f = np.maximum(new_f, 0.0)

        new_score = evaluate_sequence(new_f.tolist())
        if new_score < best_score - 1e-9:
            best_seq = new_f
            best_score = new_score
            print(f"[improve] step={step:.3f} new_score={best_score:.6f}")
            # maybe increase step a bit
            step = min(step * 1.1, 0.9)
            no_improve = 0
        else:
            # no improvement, reduce step
            step *= 0.6
            no_improve += 1
            if step < 1e-4:
                # step too small, break
                print("[stop] step too small")
                break
            if no_improve >= max_no_improve:
                # Try a random perturbation to escape
                # Randomly move some mass
                i, j = np.random.choice(n,2,replace=False)
                src, dst = (i,j) if best_seq[i] > best_seq[j] else (j,i)
                delta = 0.01 * min(best_seq[src], best_seq[dst])
                if delta > 0:
                    pert = best_seq.copy()
                    pert[src] -= delta
                    pert[dst] += delta
                    pert_score = evaluate_sequence(pert.tolist())
                    if pert_score < best_score:
                        best_seq = pert
                        best_score = pert_score
                        print("[perturb] improved to", best_score)
                no_improve = 0
        # Check termination condition
        if time.time() > deadline:
            break

    print("[final] best_score", best_score, "iterations done")
    return best_seq.tolist()

    """
    Approximate optimum step function for the given evaluation.
    The algorithm starts from the discrete arcsine law (asymptotically optimal
    for the continuous problem) and iteratively improves it by solving a
    linear program which finds a direction that can increase the ℓ₁‑norm
    while keeping the convolution with the current iterate below the current
    worst‑case value.  The direction is scaled to the same total mass as the
    current sequence and mixed with a tunable step size.  The loop runs until
    the time budget expires or no improvement is observed.  If SciPy is not
    available a simple random hill‑climbing fallback is used.
    """

    import numpy as np

    try:
        from scipy.optimize import linprog
    except Exception:
        linprog = None

    a /= a.sum()                     # normalise to L1 = 1
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ----------- fallback if no LP ------------------------------------------------
    deadline = time.time() + budget_s - 0.5
    if linprog is None:
        print("[info] SciPy not available – using random hill climbing.")
        it = 0

            i, j = np.random.choice(n, 2, replace=False)
            src, dst = (i, j) if best_seq[i] > best_seq[j] else (j, i)
            delta = 0.01 * min(best_seq[src], best_seq[dst])

            candidate[dst] += delta

    max_no_improve = 12

        # build matrix for constraints f * g ≤ M
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        # each column i contains best_seq shifted by i positions
        for i in range(n):
            A[i:i + n, i] = best_seq

        b = np.full(rows, M, dtype=np.float64)

        # linear program: maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n

        # solve
        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                          method='highs', options={'presolve': True})
        except Exception as exc:
            print("[lp error] exception:", exc)
            break

        if not res.success:
            print("[lp] no success:", res.message)
            break

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Approximate the minimum of ``evaluate_sequence``.

    The algorithm proceeds in three stages:

    1. **Start point** – the discrete arcsine law
       a_k = C·(2k choose k)/4**k  (normalised to L¹‑norm = 1).  This
       construction attains the continuous optimum π/2 ≈ 1.571.

    2. **LP‑based refinement** – for the current sequence ``f`` we compute
       the worst convolution value ``M = max(f*f)`` and solve the linear
       program

           maximise   Σ g_i
           subject to f * g ≤ M          (pointwise convolution)
                      g_i ≥ 0 .

       The solution ``g`` is scaled to the same L¹‑norm as ``f`` and mixed
       with ``f`` using a tunable step size.  This operation never increases
       the worst convolution value and typically reduces the objective.
       The loop repeats until the time budget expires or no further
       improvement is observed.

    3. **Optional hill‑climbing** – if a few seconds remain we perform a
       very cheap random “move mass from a larger entry to a smaller one”
       hill‑climbing sweep to squeeze out any remaining slack.

    If SciPy is not available the function falls back to the simple
    hill‑climbing used in the previous run.

    The returned object is a plain ``list[float]`` as required by the
    evaluator.
    """
    import time
    import numpy as np

    # -------------------------------------------------------------
    # 0.  Optional SciPy import (needed for the LP step)
    # -------------------------------------------------------------
    try:
        from scipy.optimize import linprog
    except Exception:                     # pragma: no cover
        linprog = None

    # -------------------------------------------------------------
    # 1.  Choose a sequence length.
    #     Use the length of the global ``height_sequence_1`` when
    #     it exists, but never exceed ``max_n`` to keep the LP size
    #     tractable.
    # -------------------------------------------------------------
    np.random.seed(seed)
    max_n = 1500                          # safe upper bound for the LP
    n = kwargs.get("length")
    if n is None:
        try:
            n = len(height_sequence_1)    # type: ignore[name-defined]
        except Exception:
            n = 1200
    n = int(min(n, max_n))
    n = max(2, n)                         # we need at least two entries

    # -------------------------------------------------------------
    # 2.  Build the (unnormalised) arcsine coefficients.
    #     a_0 = 1,   a_{k+1} = a_k * (2k+1)(2k+2) / (4·(k+1)²)
    # -------------------------------------------------------------
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)

    # -------------------------------------------------------------
    # 3.  Normalise so that Σ a = 1 (the objective is scale‑invariant).
    # -------------------------------------------------------------
    a /= a.sum()
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # -------------------------------------------------------------
    # 4.  Simple hill‑climbing fallback if SciPy is unavailable.
    # -------------------------------------------------------------
    deadline = time.time() + budget_s - 0.5
    if linprog is None:                                 # pragma: no cover
        print("[info] SciPy not available – using random hill climbing.")
        it = 0
        while time.time() < deadline and it < 5000:
            i, j = np.random.choice(n, 2, replace=False)
            src, dst = (i, j) if best_seq[i] > best_seq[j] else (j, i)
            delta = 0.01 * min(best_seq[src], best_seq[dst])
            if delta <= 0.0:
                it += 1
                continue
            cand = best_seq.copy()
            cand[src] -= delta
            cand[dst] += delta
            sc = evaluate_sequence(cand.tolist())
            if sc < best_score:
                best_seq, best_score = cand, sc
                print(f"[improve] it={it}, score={best_score:.6f}")
            it += 1
        print(f"[final] best score={best_score:.6f}")
        return best_seq.tolist()

    # -------------------------------------------------------------
    # 5.  LP‑based improvement loop.
    # -------------------------------------------------------------
    step = 0.4                # start with a moderate mixing weight
    no_improve_cnt = 0
    max_no_improve = 12

    while time.time() < deadline:
        # (a) current worst convolution value
        conv_self = np.convolve(best_seq, best_seq)
        M = conv_self.max()

        # (b) build the constraint matrix for f * g ≤ M
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for i in range(n):
            A[i:i + n, i] = best_seq

        b = np.full(rows, M, dtype=np.float64)

        c = -np.ones(n, dtype=np.float64)

        except Exception as exc:                       # pragma: no cover
            print("[lp error] exception:", exc)
            break

            print("[lp] failure:", res.message)
            break

        sum_g = g.sum()
        if sum_g <= 0.0:                               # pragma: no cover

        g_scaled = g / sum_g

        # (e) try a full replacement first – it may already improve the bound
        score_full = evaluate_sequence(g_scaled.tolist())

                i, j = np.random.choice(n, 2, replace=False)
                src, dst = (i, j) if best_seq[i] > best_seq[j] else (j, i)
                delta = 0.01 * min(best_seq[src], best_seq[dst])
                if delta > 0.0:
                    pert = best_seq.copy()
                    pert[src] -= delta
                    pert[dst] += delta
                    pert_score = evaluate_sequence(pert.tolist())
                    if pert_score < best_score:
                        best_seq, best_score = pert, pert_score
                        print(f"[perturb] improved to {best_score:.6f}")
                no_improve_cnt = 0

        # safety check on the remaining time
        if time.time() > deadline:

    # -------------------------------------------------------------
    # 6.  Optional cheap fine‑tuning with random moves (if time remains)
    # -------------------------------------------------------------
    remaining = deadline - time.time()
