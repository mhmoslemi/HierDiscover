def gradient_descent(seq, n, deadline):
    step = 0.1
    gamma = 1.0
    best_local_seq = seq.copy()
    best_local_obj = compute_obj(seq, n)
    while time.time() < deadline:
        conv = np.convolve(seq, seq)
        obj = 2.0*n*conv.max()
        # compute softmax weights
        # compute stable exponentials
        shifted = conv - conv.max()
        exp_vals = np.exp(gamma * shifted)
        w = exp_vals / exp_vals.sum()
        # compute gradient via correlation
        corr = np.correlate(w, seq, mode='full')
        # length = len(w)+len(seq)-1
        L = len(conv)
        grad = 2.0 * corr[L-1:L-1+n]  # slice
        # update with step
        new_seq = seq - step * grad
        # projection
        new_seq = _project_onto_simplex(new_seq)
        new_conv = np.convolve(new_seq, new_seq)
        new_obj = 2.0*n*new_conv.max()
        # accept if better
        if new_obj <= obj:
            seq = new_seq
            if new_obj < best_local_obj:
                best_local_obj = new_obj
                best_local_seq = new_seq.copy()
        else:
            # reject; reduce step
            step *= 0.7
            # maybe also reduce gamma?
        # adjust step and gamma
        step = max(step*1.01, 1e-8)  # maybe increase a bit if accepted
        gamma = min(gamma*1.02, 50.0)  # gradually increase
        # termination condition
        if step < 1e-8:
            break
    return best_local_seq, best_local_obj

"""
Propose a step‑function (non‑negative sequence) that minimizes the supplied
evaluation function.

The search is performed in three stages within the 1000 s time budget:

1. **Initial seeds** – a collection of deterministic (uniform / geometric) and
   random Dirichlet vectors.

2. **Projected smoothed gradient descent** – we minimise a smooth surrogate of
   the “max‑convolution’’ term:
        fγ(a) = (1/γ)·log Σk exp(γ·(a*a)[k]) .
   For small γ the function is almost flat, for large γ it approaches the true
   max.  The gradient can be written with a single correlation
   `grad = 2·corr(w, a)`, where `w` are soft‑max weights.
   After each step we project the iterate onto the probability simplex
   (non‑negative entries that sum to one) using the O(n log n) routine
   `_project_onto_simplex`.

3. **Local stochastic refinement** – from the best candidate we repeatedly
   transfer a small amount of mass from indices that belong to the currently
   active convolution window (the shift where the maximum occurs) to a random
   other index.  Simple simulated‑annealing acceptance lets us escape shallow
   local minima.

The algorithm keeps the best sequence seen so far and returns it as a plain
Python list.  All operations respect the time limit; a ~5 000‑element
candidate can be obtained well within the budget and typically yields a
score below the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the simplex {x >= 0, sum(x)=1}
def _project_onto_simplex(v):
    """Project a (possibly negative) vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    # Find the last positive entry in (u - cssv / ind)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # All entries are non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _objective(seq, n):
    """Fast evaluation of 2 · n · max(conv) (sum(seq) is 1 after projection)."""
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Build a non‑negative step function using smoothed gradient descent and a
    local stochastic refinement.  The function returns the best sequence found
    before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a tiny safety margin

    # ------------------------------------------------------------------
    # 0. Basic parameters
    n = 5000                     # length of the candidate (adjustable)
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. Generate a few diverse starting points
    starts = []

    # uniform
    starts.append(np.full(n, 1.0 / n))

    # geometric families with several decay factors
    for factor in [1.2, 1.6, 2.0, 2.4, 3.0]:
        r = 1.0 - factor / n
        if r <= 0:
            continue
        geom = np.power(r, np.arange(n, dtype=np.float64))
        geom /= geom.sum()
        starts.append(geom)

    # a few random Dirichlet draws (moderate concentration)
    for _ in range(3):
        rnd = np.random.dirichlet(np.ones(n))
        starts.append(rnd)

    # ------------------------------------------------------------------
    # 2. Smoothed projected gradient descent on each start
    for idx, init in enumerate(starts):
        if time.time() > deadline:
            break

        seq = init.copy()
        # hyper‑parameters (they are adapted on‑the‑fly)
        step = 0.2          # initial gradient‑step length
        gamma = 1.0         # smoothing parameter for the soft‑max
        cur_score = _objective(seq, n)

        # keep a copy of the best sequence found in this run
        local_best_seq = seq.copy()
        local_best_score = cur_score

        it = 0
        while time.time() < deadline:
            # --------------------------------------------------------------
            # 2a – compute convolution and soft‑max weights
            conv = np.convolve(seq, seq)                 # length 2n‑1
            max_conv = conv.max()
            # (the exact objective is 2·n·max_conv)
            cur_score = 2.0 * n * max_conv

            # update global best
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print(f"[Init {idx}] new global best = {best_score:.6f}")

            # soft‑max weighting (numerically stable)
            shifted = conv - max_conv                     # max entry becomes 0
            exp_vals = np.exp(gamma * shifted)
            w = exp_vals / exp_vals.sum()                # shape (2n‑1,)

            # --------------------------------------------------------------
            # 2b – gradient of the smoothed objective:
            # grad = 2 * corr(w, seq)   (full correlation, then slice)
            corr_full = np.correlate(w, seq, mode='full')
            L = len(conv)                                 # = 2n‑1
            grad = 2.0 * corr_full[L - 1 : L - 1 + n]     # length n

            # --------------------------------------------------------------
            # 2c – take a step and project back onto the simplex
            new_seq = seq - step * grad
            new_seq = _project_onto_simplex(new_seq)      # guarantees sum=1

            # --------------------------------------------------------------
            # 2d – accept if we improved, otherwise shrink the step size
            new_conv = np.convolve(new_seq, new_seq)
            new_score = 2.0 * n * new_conv.max()

            if new_score <= cur_score:
                seq = new_seq
                # succeed → allow a slight increase of the step size
                step = min(step * 1.02, 1.0)
                # keep the best local solution
                if new_score < local_best_score:
                    local_best_score = new_score
                    local_best_seq = new_seq.copy()
            else:
                # failure → back‑track and reduce step
                step *= 0.7
                # if the step became tiny stop this run
                if step < 1e-9:
                    break

            # --------------------------------------------------------------
            # 2e – slowly increase the smoothing (annealing)
            gamma = min(gamma * 1.015, 50.0)   # cap to avoid overflow
            it += 1
            if it % 2000 == 0:
                print(f"[Init {idx}] it={it:05d}, cur={cur_score:.6f}, "
                      f"step={step:.3e}, gamma={gamma:.2f}")

        # after finishing this start keep its best
        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_seq.copy()
            print(f"[Init {idx}] end‑run best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Stochastic hill‑climbing refinement from the best candidate
    if best_seq is None:
        # safety fallback – uniform distribution
        best_seq = np.full(n, 1.0 / n)
    seq = best_seq.copy()
    cur_score = _objective(seq, n)

    # parameters for the hill‑climbing stage
    move_fraction = 0.05          # how much of the source weight we move
    iter_hc = 0

    while time.time() < deadline:
        # locate the active convolution shift (the one attaining the maximum)
        conv = np.convolve(seq, seq)
        k_max = int(conv.argmax())
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)

        # pick a source index inside the active window and a random destination
        src = np.random.randint(lo, hi + 1)
        dst = np.random.randint(0, n)
        if src == dst:
            continue

        # amount to transfer (a small fraction of the source mass)
        delta = move_fraction * seq[src] * np.random.rand()
        if delta <= 0.0:
            continue

        new_seq = seq.copy()
        new_seq[src] -= delta
        new_seq[dst] += delta

        # projection is not needed – the move preserves non‑negativity and sum,
        # but numerical noise may introduce tiny negatives; fix them.
        if new_seq[src] < 0:
            new_seq[dst] += new_seq[src]   # subtract the tiny negative overflow
            new_seq[src] = 0.0
        # small renormalisation to keep sum exactly 1 (helps numerical stability)
        new_seq /= new_seq.sum()

        new_score = _objective(new_seq, n)

        # accept if improvement, otherwise accept with a temperature‑dependent
        # probability (simple simulated annealing)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # linear temperature schedule: higher early on, zero at the end
            remaining = deadline - time.time()
            total = budget_s
            temp = max(0.0, remaining / total) * 0.02   # a tiny temperature
            if np.random.rand() < np.exp(-(new_score - cur_score) / (temp + 1e-12)):
                accept = True

        if accept:
            seq = new_seq
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print(f"[Hill {iter_hc:06d}] new global best = {best_score:.6f}")

        iter_hc += 1
        if iter_hc % 5000 == 0:
            print(f"[Hill] iter={iter_hc}, cur_score={cur_score:.6f}, "
                  f"best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best objective = {best_score:.6f} (length {n})")
    # Return a plain Python list as required by the evaluator
    return best_seq.tolist()
