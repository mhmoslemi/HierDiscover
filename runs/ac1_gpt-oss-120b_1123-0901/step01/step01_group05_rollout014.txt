def propose_candidate(seed=123, budget_s=1000, **kwargs):
    import time, numpy as np
    from scipy.optimize import linprog
    # Determine length.
    try:
        n = len(height_sequence_1)
        base = [float(x) for x in height_sequence_1]
    except Exception:
        n = 3000
        # generate inverse sqrt seq
        base = [1.0]
        for i in range(1,n):
            base.append(base[-1] * (2*i-1)/(2*i))
    base = np.array(base, dtype=np.float64)
    # normalize sum to sqrt(2n)
    target_sum = np.sqrt(2.0 * n)
    base = base * (target_sum / base.sum())
    f = base.copy()
    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")
    deadline = time.time() + budget_s - 0.5
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        conv_f = np.convolve(f, f)
        M = conv_f.max()
        # Because sum is fixed, score = M
        if M < best_score - 1e-9:
            best_score, best_f = M, f.copy()
            print(f"[iter {iteration}] improved max={M:.6f}")
        # Build lower triangular matrix
        A = np.zeros((n, n), dtype=np.float64)
        for k in range(n):
            A[k, :k+1] = f[k::-1]
        # Solve LP: max sum(g)
        c = -np.ones(n, dtype=np.float64)  # maximize sum -> minimize -sum
        b = np.full(n, M, dtype=np.float64)
        # bounds: g >= 0
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            print(f"[iter {iteration}] LP failed: {res.message}")
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break
        # Scale to target sum
        scale = target_sum / sum_g0
        g = g0 * scale
        # Try mixing with decreasing step sizes
        improved = False
        for t in (0.5, 0.3, 0.2, 0.1, 0.05):
            new_f = (1 - t) * f + t * g
            conv_new = np.convolve(new_f, new_f)
            M_new = conv_new.max()
            if M_new < M:
                f = new_f
                improved = True
                break
        if not improved:
            # No improvement found; stop early
            break
    # After iterations, maybe do final random local search for extra tweak
    # Use same cheap search as earlier but now on best_f
    if time.time() < deadline:
        rng = np.random.default_rng(seed + iteration)
        seq = best_f.copy()
        cur_score = best_score
        while time.time() < deadline:
            i = rng.integers(n)
            factor = np.exp(rng.normal(0,0.02))
            new_val = max(0.0, seq[i] * factor)
            cand = seq.copy()
            cand[i] = new_val
            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < cur_score:
                cur_score = cand_score
                seq = cand
                best_f = seq
                best_score = cur_score
                print(f"[random improve] score={cur_score:.6f} idx={i}")
        # final best_f = seq
    return best_f.tolist()

"""
propose_candidate
-----------------

Iterative improvement of a non‑negative step‑function using a linear‑programming
sub‑step.  Starting from the classic “inverse‑square‑root’’ construction we
normalise the sequence so that its ℓ¹‑norm equals √(2 n).  For a normalized
sequence the evaluation score coincides with the maximal entry of the
self‑convolution (the factor 2 n/(∑a)² reduces to 1).  The algorithm repeatedly:

1.  Computes the current worst convolution value M.
2.  Solves a linear programme  
        maximise   Σ g_i
        subject to  (f * g)_k ≤ M   (0 ≤ k < n) ,   g_i ≥ 0
    where (f * g)_k = Σ_j f_{k‑j} g_j .  The LP finds a vector g with the same
    bound M but a larger ℓ¹‑norm.
3.  Scales g back to the required normalisation and mixes it with the current
    function f using a convex combination (step‑size t).  Because the cross‑
    convolution f * g is bounded by M, a small enough t strictly reduces the
    maximal self‑convolution.
4.  Repeats the procedure until the time budget expires or no further
    improvement is observed.

A final cheap random‑local‑search (multiplicative perturbations of single
entries) is applied to polish the best sequence found.

The method stays within the 1000 s budget, uses only standard libraries
(Numpy, SciPy) and respects the security checks of `evaluate_sequence`.
"""

import time
import numpy as np

def _inverse_sqrt_seq(n: int) -> np.ndarray:
    """Return a_i = binom(2*i,i)/4**i for i=0..n‑1 (recurrence avoids overflow)."""
    seq = np.empty(n, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, n):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return seq


def _build_lower_toeplitz(f: np.ndarray) -> np.ndarray:
    """
    Build the (n × n) matrix A where A[k, j] = f[k‑j] for j ≤ k
    (the convolution constraints for k = 0 … n‑1).
    """
    n = f.size
    A = np.zeros((n, n), dtype=np.float64)
    for k in range(n):
        # fill the first (k+1) entries of row k with f[k], f[k‑1], …, f[0]
        A[k, : k + 1] = f[k::-1]
    return A


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Return a list of non‑negative floats that (hopefully) beats the current bound."""
    # ----------------------------------------------------------------------

    # ----------------------------------------------------------------------

    target_sum = np.sqrt(2.0 * n)
    f = base * (target_sum / base.sum())

    # Best sequence seen so far.
    best_f = f.copy()
    # The score equals the maximal convolution entry for a normalised sequence.
    best_score = evaluate_sequence(best_f.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # 2.  Main iterative LP / mixing loop.
    # ----------------------------------------------------------------------
    deadline = time.time() + budget_s - 0.4   # a small safety margin
    iteration = 0
    # step sizes tried in descending order; smaller steps are safer when close to optimum.
    step_sizes = (0.5, 0.3, 0.2, 0.1, 0.05)

    # Import linprog lazily – fall back to simple random search if unavailable.
    try:
        from scipy.optimize import linprog
    except Exception:
        linprog = None

    while time.time() < deadline:
        iteration += 1

        # Current maximal self‑convolution (score for a normalised sequence).
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # Record improvement.
        if M < best_score - 1e-9:
            best_score = M
            best_f = f.copy()
            print(f"[iter {iteration}] new best score={best_score:.6f}")

        # If SciPy is missing we break out and rely on the fallback local search.
        if linprog is None:
            break

        # ------------------------------------------------------------------
        # 2a.  Linear programme: maximise Σ g_i  subject to  (f * g)_k ≤ M.
        # ------------------------------------------------------------------
        A = _build_lower_toeplitz(f)                # shape (n, n)
        c = -np.ones(n, dtype=np.float64)           # minimise –Σ g_i  ⇔ maximise Σ g_i
        b = np.full(n, M, dtype=np.float64)         # right‑hand side
        bounds = [(0.0, None)] * n                   # g_i ≥ 0

        lp_res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                         method='highs', options={'disp': False})
        if not lp_res.success:
            # LP failure – exit the sophisticated loop.
            print(f"[iter {iteration}] LP failed: {lp_res.message}")
            break

        g0 = lp_res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break

        # 2b.  Rescale g0 to the target ℓ¹‑norm and mix with f.

                seq = cand
                best_f = seq
                best_score = cur_score
                print(f"[random] improved score={cur_score:.6f} (idx {i})")

    # ----------------------------------------------------------------------
    # 4.  Return the best sequence found (as a plain Python list).
    # ----------------------------------------------------------------------
    print(f"[final] best score={best_score:.6f}")
    return best_f.tolist()
