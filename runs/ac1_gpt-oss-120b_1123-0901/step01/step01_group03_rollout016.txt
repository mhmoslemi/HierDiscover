"""
Local‑search optimisation of a non‑negative step function.

The algorithm starts from a known good construction (the global variable
`height_sequence_1` if it exists, otherwise the central‑binomial series)
and repeatedly performs *mass‑transfer moves*:

    – pick a source index i (preferably one that contributes most to the
      current convolution peak) and a target index j,
    – move a small amount δ of mass from a[i] to a[j],
    – update the self‑convolution exactly by the analytic formula
      (no FFT required for the candidate).

The move size δ is annealed: early in the run a large fraction of a[i] can
be moved, later the fraction shrinks to a tiny value.  Any move that does
not increase the peak of the convolution (i.e. the evaluation metric)
is accepted, which guarantees monotonic improvement.  To escape plateaus
the routine occasionally adds a tiny Gaussian perturbation and periodically
re‑computes the convolution from scratch to avoid numerical drift.

The evaluation metric for a sequence with Σa_i = 1 is
    2·n·max(conv(a,a)),
so the code works directly with `max(conv)` and never calls the slower
`evaluate_sequence` inside the inner loop.  The best sequence found within
the allotted time (default 1000 s) is returned as a plain Python list.
"""

import numpy as np
import time
import math


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·len(a)-1)."""
    n = len(a)
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The function runs for at most `budget_s` seconds
    (a safety margin of a few seconds is subtracted) and returns the best
    found sequence as a list of Python floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Initialise a.
    # ------------------------------------------------------------------
    try:
        # Prefer the previously‑found construction if it exists.
        init = np.array(height_sequence_1, dtype=np.float64)
        init = np.maximum(init, 0.0)          # guard against accidental negatives
    except NameError:
        # Fallback: central‑binomial coefficients (the classic start).
        n = 3000
        init = np.empty(n, dtype=np.float64)
        init[0] = 1.0
        for i in range(1, n):
            init[i] = init[i - 1] * (2.0 * i - 1.0) / (2.0 * i)

    # Normalise to Σa_i = 1 (the metric is homogeneous, any positive sum works).
    a = init / init.sum()
    n = len(a)

    # Initial convolution and metric.
    conv = _full_convolution(a)
    current_score = 2.0 * n * conv.max()
    best_score = current_score
    best_a = a.copy()

    # ------------------------------------------------------------------
    # 2. Search parameters.
    # ------------------------------------------------------------------
    max_temp = 0.5          # initial maximal fraction of mass to move
    min_temp = 1e-4         # final fraction (near‑zero moves)
    iteration = 0

    # ------------------------------------------------------------------
    # 3. Main optimisation loop.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----------- annealing temperature (linear in remaining time) -------------
        remaining = max(0.0, deadline - time.time())
        total = max(1.0, budget_s - 5.0)
        frac_time = remaining / total
        temperature = min_temp + (max_temp - min_temp) * frac_time

        # ----------- locate the current peak of the convolution --------------------
        k_max = int(conv.argmax())                 # index where conv is maximal
        start = max(0, k_max - (n - 1))
        end = min(k_max, n - 1)

        # ----------- choose source index i (biased towards the peak) -------------
        idx_region = np.arange(start, end + 1)
        # contribution of each i to the peak is a[i] * a[k_max-i]
        w = a[idx_region] * a[k_max - idx_region]
        w_sum = w.sum()
        if w_sum > 1e-14:
            cw = np.cumsum(w)
            r = np.random.rand() * w_sum
            i = idx_region[np.searchsorted(cw, r)]
        else:
            # fallback: simple weighted by a (or uniform if all are zero)
            cum_a = np.cumsum(a)
            r = np.random.rand() * cum_a[-1]
            i = np.searchsorted(cum_a, r)

        if a[i] < 1e-12:          # not enough mass to move – skip this iteration
            continue

        # ----------- choose target index j (preferably outside the peak region) --
        if np.random.rand() < 0.7:
            # try to pick j outside the region that creates the peak
            while True:
                j = np.random.randint(0, n)
                if j < start or j > end:
                    break
        else:
            j = np.random.randint(0, n)
        if j == i:
            continue

        # ----------- propose a transfer of mass δ from i to j --------------------
        delta_frac = np.random.rand() * temperature       # 0 … temperature
        delta = a[i] * delta_frac
        if delta <= 0.0:
            continue

        # ----- build candidate convolution using the exact update formula ----------
        conv_cand = conv.copy()
        # linear part:  -2δ·a shifted by i  + 2δ·a shifted by j
        conv_cand[i:i + n] -= 2.0 * delta * a
        conv_cand[j:j + n] += 2.0 * delta * a
        # quadratic correction terms
        ii = 2 * i
        ij = i + j
        jj = 2 * j
        conv_cand[ii] += delta * delta
        conv_cand[ij] -= 2.0 * delta * delta
        conv_cand[jj] += delta * delta

        new_max = conv_cand.max()
        new_score = 2.0 * n * new_max

        # ----------- accept any non‑worsening move -------------------------------
        if new_score <= current_score + 1e-16:
            # apply the move permanently
            a[i] -= delta
            a[j] += delta
            conv = conv_cand
            current_score = new_score

            # update the best solution if an improvement occurred
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a.copy()
                print(f"Iter {iteration:8d}: new best score {best_score:.6f}")

        # ----------- occasional global perturbation to escape plateaus ----------
        if iteration % 5000 == 0 and iteration > 0:
            noise = np.random.randn(n) * 0.001
            a_pert = a + noise
            a_pert = np.maximum(a_pert, 0.0)
            s = a_pert.sum()
            if s > 0.0:
                a_pert /= s
                conv_pert = _full_convolution(a_pert)
                pert_score = 2.0 * n * conv_pert.max()
                if pert_score < best_score:
                    a = a_pert
                    conv = conv_pert
                    current_score = pert_score
                    best_score = pert_score
                    best_a = a.copy()
                    print(f"Iter {iteration:8d}: perturb improved to {best_score:.6f}")

        # ----------- periodic exact recomputation to avoid drift -----------------
        if iteration % 20000 == 0 and iteration > 0:
            conv = _full_convolution(a)
            current_score = 2.0 * n * conv.max()
            # (no print unless it improves the best)

        iteration += 1

        # early stop if we already beat the target bound
        if best_score <= 1.503:
            break

    print(f"Finished after {iteration} iterations. Best score = {best_score:.6f}")
    # Return as plain Python floats
    return best_a.tolist()
