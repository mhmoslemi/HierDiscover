"""
Optimise a non‑negative step function to minimise

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works in three stages:

1. **Initialisation** – start from the classic discrete arcsine distribution
   (the known asymptotically optimal shape) or from the global
   ``height_sequence_1`` if it exists.

2. **Flattening phase** – repeatedly locate the index where the
   convolution attains its maximum, compute the contribution of each
   coefficient to that peak and move a small fraction of mass from the
   “most responsible’’ coefficient to the “least responsible’’ one.
   By always decreasing the peak we drive the autocorrelation towards a
   flat (average‑valued) shape.  The convolution is evaluated by FFT
   (O(n log n)) so thousands of iterations fit comfortably into the
   1000 s budget.

3. **Fine‑tuning phase** – a light random hill‑climb (tiny mass swaps)
   cleans up any remaining slack.

Both phases respect the non‑negativity and the automatic clamping
performed by ``evaluate_sequence``.  The routine returns the best
sequence found as a plain ``list[float]``.
"""

import time
import numpy as np

# The evaluator is provided in the execution environment.
# from __main__ import evaluate_sequence   # not needed – it is already in globals()


def _conv_max_and_full(a: np.ndarray):
    """Return (max value, full linear convolution) of a with itself using FFT."""
    n = a.shape[0]
    # length of linear convolution = 2*n-1 → next power of two for efficient FFT
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    return conv.max(), conv


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum(a)^2 (identical to evaluate_sequence)."""
    s = a.sum()
    if s < 0.01:                         # same guard as evaluate_sequence
        return float("inf")
    n = a.shape[0]
    maxb, _ = _conv_max_and_full(a)
    return float(2.0 * n * maxb / (s * s))


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function with a low evaluation score.
    The function respects the global ``height_sequence_1`` if present,
    otherwise builds a fresh arcsine distribution.
    """
    np.random.seed(seed)

    # ----------------------------------------------------------------------
    # 1. Determine the length of the sequence.
    # ----------------------------------------------------------------------
    n = kwargs.get("length")
    if n is None:

            n = len(height_sequence_1)      # type: ignore[name-defined]
        except Exception:
            n = 3000                        # a moderate size that works fast

    # ----------------------------------------------------------------------
    # 2. Initialise the sequence.
    # ----------------------------------------------------------------------
    try:

        a = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if a.shape[0] != n:
            # truncate or pad with zeros
            if a.shape[0] > n:
                a = a[:n]
            else:
                a = np.pad(a, (0, n - a.shape[0]), mode="constant")
    except Exception:
        # Build the discrete arcsine distribution.
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for k in range(1, n):
            a[k] = a[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
        a = a / a.sum()                     # normalise to sum = 1

    a = np.maximum(a, 0.0)                  # safety – avoid accidental negatives
    a = a / a.sum()                         # keep sum exactly 1

    best_seq = a.copy()
    best_score = _objective(best_seq)
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # 3. Flattening phase – move mass from the strongest to the weakest
    #    contribution to the current maximal convolution peak.
    # ----------------------------------------------------------------------
    deadline = time.time() + budget_s - 0.5  # safety margin
    it = 0
    max_iters = 50000                       # hard stop to avoid endless loops

    while time.time() < deadline and it < max_iters:
        # compute current convolution and locate the peak
        max_val, conv = _conv_max_and_full(a)
        k_max = int(np.argmax(conv))

        # contribution of each coefficient to this peak (a[i] * a[k_max-i] if defined)
        contrib = np.zeros_like(a)
        i_min = max(0, k_max - (n - 1))
        i_max = min(n - 1, k_max)
        idxs = np.arange(i_min, i_max + 1, dtype=int)
        partners = k_max - idxs
        contrib[idxs] = a[idxs] * a[partners]

        src = int(np.argmax(contrib))               # most responsible coefficient
        dst = int(np.argmin(contrib))               # least responsible coefficient
        if src == dst:

    # ----------------------------------------------------------------------
