"""
Propose a candidate step‑function that beats the simple 1/√i construction.

Algorithm
---------
1. Build the classical “inverse‑square‑root’’ sequence
      a_i = C(2i,i) / 4**i   (coefficients of (1‑x)^‑½).
   The sequence is normalised so that its ℓ¹‑norm equals √(2 n) –
   this does not change the evaluation because the objective is scale‑
   invariant.

2. Iteratively improve the shape by solving a linear programme.
   For the current sequence f we compute M = max(f*f).  The LP
   maximises Σ g_i under the linear constraints
        (f * g)_k ≤ M   for all k,
        g_i ≥ 0.
   The solution g carries as much mass as possible in positions where
   f is small, without increasing the worst convolution value.
   After scaling g to the same ℓ¹‑norm as f we replace
        f ← (1‑t) f + t g
   with a modest step size t (≈0.2).  This convex combination keeps the
   normalisation fixed and provably does not increase the bound M.
   Repeating the step a few times quickly drives the max‑convolution
   down.

3. When the time budget is almost exhausted we fall back to a cheap
   random coordinate descent (multiplicative perturbations of a single
   entry) to polish the result.

The whole routine respects the 1000 s time budget, works with any
previously discovered sequence `height_sequence_1` and returns a list
of non‑negative floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# cvxpy is optional – it is used only for the LP refinement.
try:
    import cvxpy as cp
except Exception:  # pragma: no cover
    cp = None


def _inverse_sqrt_seq(n: int) -> np.ndarray:
    """
    Return the length‑n coefficients a_i = binom(2*i,i)/4**i.
    The recurrence a_{i+1}=a_i*(2*i+1)/(2*i+2) is used to avoid overflow.
    """
    seq = np.empty(n, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, n):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return seq

    """
    Build the matrix A (size (2n‑1)×n) such that A @ g == np.convolve(f, g).
    For a fixed f, the convolution is linear in g.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)

    # column i (corresponding to g_i) contains f shifted by i rows
    # i.e. A[i:i+n, i] = f[0:n]
    for i in range(n):
        A[i : i + n, i] = f
    return A


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Entry point required by the evaluation harness.
    Returns a list of non‑negative floats (the step heights).
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # 1. Length of the optimisation problem.
    # ----------------------------------------------------------------------
    try:
        n = len(height_sequence_1)      # type: ignore[name-defined]
    except Exception:
        n = 2000                         # a reasonable default

    # ----------------------------------------------------------------------
    # 2. Base sequence – inverse‑square‑root shape.
    # ----------------------------------------------------------------------
    f = _inverse_sqrt_seq(n)

    # Normalise to the standard ℓ¹‑norm √(2 n) (scale does not matter for the
    # objective, but it keeps the numbers moderate).
    target_sum = np.sqrt(2.0 * n)
    f = f * (target_sum / f.sum())

    # Initial evaluation (also triggers the internal security checks).
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # 3. LP based refinement (skip if cvxpy unavailable).
    # ----------------------------------------------------------------------
    t = 0.25               # step size for the convex update
    max_lp_iters = 6       # how many LP refinements we try
    lp_deadline = time.time() + budget_s - 30.0   # keep ~30 s for polishing

    if cp is not None:
        for it in range(max_lp_iters):
            if time.time() > lp_deadline:
                break

            # max convolution of the current candidate
            conv_f = np.convolve(best_seq, best_seq)
            M = conv_f.max()

            # Build the linear constraint matrix for (f * g) ≤ M
            A = _build_conv_matrix(best_seq)

            # Linear programme: maximise Σ g_i  subject to A g ≤ M , g ≥ 0
            g = cp.Variable(n, nonneg=True)
            constraints = [A @ g <= M]
            prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

            try:
                prob.solve(solver=cp.ECOS, warm_start=True, verbose=False,
                           max_iters=5000)
            except Exception:      # pragma: no cover
                print("[lp] solver exception – aborting LP phase")
                break

            if g.value is None:
                print("[lp] solver failed to return a solution")
                break

            g0 = np.maximum(g.value, 0.0)   # numerical safety
            sum_g0 = g0.sum()
            if sum_g0 == 0.0:
                print("[lp] trivial solution – aborting LP phase")
                break

            # Rescale g0 to the same ℓ¹‑norm as f (√(2 n))
            g_scaled = g0 * (target_sum / sum_g0)

            # Convex combination f ← (1‑t) f + t g
            new_f = (1.0 - t) * best_seq + t * g_scaled

            # Clip only to respect the security limits (values >1000 become 1000)
            new_f = np.clip(new_f, 0.0, 1000.0)

            new_score = evaluate_sequence(new_f.tolist())
            print(f"[lp iter {it}] score={new_score:.6f},  M={M:.6f}")

            if new_score < best_score:
                best_score = new_score
                best_seq = new_f
                print(f"[lp improve] new best score={best_score:.6f}")
                # keep the same step size
            else:
                # If no improvement we reduce the step size to be more cautious
                t = max(t * 0.6, 0.02)
                print(f"[lp no‑improve] reducing t → {t:.4f}")

    else:
        print("[info] cvxpy not available – skipping LP refinement phase")

    # ----------------------------------------------------------------------
    # 4. Cheap stochastic polishing (coordinate‑wise multiplicative tweaks).
    # ----------------------------------------------------------------------
    finish_deadline = time.time() + budget_s - 0.2
    n = best_seq.shape[0]            # might have been changed above
    while time.time() < finish_deadline:
        idx = rng.integers(n)
        old = best_seq[idx]

        # log‑normal proposal (keeps positivity)
        factor = np.exp(rng.normal(0.0, 0.018))
        cand_val = max(0.0, old * factor)

        cand = best_seq.copy()
        cand[idx] = cand_val

        # The security wrapper inside evaluate_sequence will clip any out‑of‑range
        # values, so we do not need explicit clipping here.
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[local improve] score={best_score:.6f} (idx {idx})")

    print(f"[final] best score achieved = {best_score:.6f}")
    return best_seq.tolist()
