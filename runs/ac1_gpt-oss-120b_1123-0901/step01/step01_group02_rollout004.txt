"""
Simulated‑annealing / hill‑climbing optimizer for the step‑function bound.

The objective is
    E(a) = 2 * n * max_k (a*a)[k] / (sum a)^2
with all a_i ≥ 0, sum a fixed to √(2n) (the problem is homogeneous, so this
normalisation does not affect the optimum).  The algorithm works as follows:

* initialise a from the best known construction `height_sequence_1`
  (or a uniform vector if that variable is unavailable);
* keep the current convolution `c = a * a` and its maximum `M`;
* at each iteration propose a transfer of a small amount of mass `δ` from an
  index that contributes heavily to the current maximal convolution peak to a
  random other index;
* evaluate the new sequence using the same formula (no call to
  `evaluate_sequence` is required inside the loop – the function is only used
  for the final verification);
* accept the move if it improves the score or with a probability
  `exp(-(new‑score‑old‑score)/T)` where the temperature `T` decays slowly;
* occasionally apply a tiny Gaussian jitter to all entries and periodically
  recompute the “peak contribution” distribution to keep the sampling focused
  on the most important coordinates;
* keep the best sequence ever seen and return it when the time budget expires.

The implementation uses only NumPy and pure Python, runs in O(n) per
iteration (the convolution is updated by `np.convolve` only for the proposed
candidate), and respects the limits enforced by `evaluate_sequence`
(non‑negative, ≤ 1000, sum ≫ 0.01).
"""

import time
import math
import numpy as np


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the provided
    ``evaluate_sequence`` function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Wall‑clock time budget (seconds) for the search.
    kwargs : dict
        Ignored – kept for API compatibility.

    Returns
    -------
    list[float]
        The best sequence found within the given budget.
    """
    rng = np.random.default_rng(seed)

    start = time.time()
    # keep a small safety margin for the harness
    deadline = start + budget_s - 5.0

    # --------------------------------------------------------------
    # 1.  Initialise a (use the known good construction if available)
    try:
        # height_sequence_1 is provided by the harness when a previous run
        # succeeded; it is a list of floats.
        base = np.asarray(height_sequence_1, dtype=float)
    except Exception:
        # fallback – a uniform vector of size 2100 (same length as the SOTA run)
        base = np.ones(2100, dtype=float)

    n = base.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # normalise to the fixed sum √(2 n)
    a = base * (target_sum / np.sum(base))

    # --------------------------------------------------------------
    # 2.  Helper that computes the score from a convolution array
    def score_from_conv(conv_arr: np.ndarray) -> float:
        """Return the objective value given the full convolution."""
        M = conv_arr.max()
        return 2.0 * n * M / (target_sum * target_sum)

    # initial convolution and score
    conv = np.convolve(a, a)
    cur_score = score_from_conv(conv)
    best_a = a.copy()
    best_score = cur_score

    print(f"[init] n={n}, initial score={cur_score:.6f}")

    # --------------------------------------------------------------
    # 3.  Simulated‑annealing parameters
    temperature0 = 0.2 * cur_score          # initial temperature
    temp_decay = 0.9999                      # exponential cooling factor
    temperature = temperature0

    # maximal transfer fraction (relative to the donor entry) – decays slowly
    step_frac0 = 0.5
    step_decay = 0.99995

    # how often to recompute the contribution distribution for the peak
    recompute_every = 200

    # store the contribution probabilities (updated every `recompute_every`
    # iterations)
    contrib_probs = None
    k_max = int(conv.argmax())

    iteration = 0
    while time.time() < deadline:
        # ----------------------------------------------------------
        # 3.1  (re)compute the index of the maximal convolution entry and
        #      a sampling distribution that favours donors contributing most
        #      to that peak.
        if iteration % recompute_every == 0:
            k_max = int(conv.argmax())
            # contributions to the peak: a[i] * a[k_max - i] (if the index is valid)
            lo = max(0, k_max - (n - 1))
            hi = min(n - 1, k_max)
            idx = np.arange(lo, hi + 1)
            contributions = np.zeros_like(a)
            contributions[idx] = a[idx] * a[k_max - idx]
            total_contrib = contributions.sum()
            if total_contrib > 0:
                contrib_probs = contributions / total_contrib
            else:
                contrib_probs = None

        # ----------------------------------------------------------
        # 3.2  Pick donor index i (preferentially from the peak contributors)
        if contrib_probs is not None and rng.random() < 0.8:
            i = int(rng.choice(n, p=contrib_probs))
        else:
            i = rng.integers(n)

        # ensure we can actually take some mass from i
        if a[i] <= 1e-12:
            iteration += 1
            continue

        # pick a distinct receiver index j uniformly
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # ----------------------------------------------------------
        # 3.3  Choose transfer amount δ
        step_frac = step_frac0 * (step_decay ** iteration)
        # never transfer more than what is available
        max_transfer = a[i] * step_frac
        # at least a tiny amount to keep progress
        delta = rng.random() * max_transfer
        if delta <= 0.0:
            iteration += 1
            continue

        # ----------------------------------------------------------
        # 3.4  Build the candidate vector
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        # ----------------------------------------------------------
        # 3.5  Compute its convolution and score (fast enough for n≈2100)
        conv_candidate = np.convolve(a_candidate, a_candidate)
        cand_score = score_from_conv(conv_candidate)

        # ----------------------------------------------------------
        # 3.6  Acceptance test (Metropolis criterion)
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # probabilistic acceptance of worsening moves
            prob = math.exp(-(cand_score - cur_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            conv = conv_candidate
            cur_score = cand_score

            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
                print(f"[iter {iteration:07d}] new best score = {best_score:.6f}")

        # ----------------------------------------------------------
        # 3.7  Temperature and step‑size cooling
        temperature = max(temperature * temp_decay, 1e-8)
        iteration += 1

        # ----------------------------------------------------------
        # 3.8  Occasionally add a tiny random jitter to all entries to
        #      escape very flat plateaus.  The jitter is re‑normalised to keep
        #      the total sum unchanged.
        if iteration % 5000 == 0:
            jitter = rng.normal(scale=1e-4 * target_sum, size=n)
            a += jitter
            a = np.clip(a, 0.0, 1000.0)
            # renormalise to the fixed sum
            a *= target_sum / a.sum()
            conv = np.convolve(a, a)
            cur_score = score_from_conv(conv)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[jitter {iteration:07d}] improved score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4.  Return the best sequence discovered.
    print(f"[done] iterations={iteration}, best score={best_score:.6f}")
    # The evaluation harness will call ``evaluate_sequence`` on the returned
    # list, so we simply return the raw numbers.
    return best_a.tolist()
