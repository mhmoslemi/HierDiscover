def _make_sequence(n, p, gamma):
    idx = np.arange(n, dtype=float)
    return (idx + gamma) ** (-p)

def propose_candidate(seed=123, budget_s=1000, n=3428):

    Search for a non‑negative step‑function that minimises
    ``2 * n * max(conv(a, a)) / (sum(a) ** 2)``.

    The algorithm expands the previous power‑law search to
    include exponents ``p`` below 0.5 (the region that yields
    the theoretically smallest constant 2*(1-p)^2*B(1-p,1-p)).
    It performs a coarse grid over (p,γ) and then a
    stochastic refinement (simulated‑annealing style).
    The best sequence found is returned as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2

    # Ensure length is a positive integer; the score is invariant under scaling.
    if n is None:
        n = 3428
    else:
        n = int(n)

    best_seq = None
    best_score = float('inf')
    best_p = None
    best_gamma = None

    # -----------------------------------------------------------------
    # 1) Coarse grid – extended range for p (including <0.5)
    # -----------------------------------------------------------------
    p_vals = np.linspace(0.25, 0.80, 56)   # step ≈0.01
    gamma_vals = np.linspace(0.05, 5.0, 50)   # step ≈0.1

    for p in p_vals:
        if time.time() > deadline: break
        for gamma in gamma_vals:
            if time.time() > deadline: break
            seq = _make_sequence(n, p, gamma)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_p = p
                best_gamma = gamma
                print(f"[grid] p={p:.4f}, γ={gamma:.4f} → score={score:.6f}")

    # -----------------------------------------------------------------
    # 2) Random local refinement (simulated‑annealing style)
    # -----------------------------------------------------------------
    if best_seq is None:
        # fallback to a sensible default
        best_p, best_gamma = 0.40, 1.0
        best_seq = _make_sequence(n, best_p, best_gamma)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using default p=0.40, γ=1.0")

    # parameters for annealing schedule
    T0 = 0.1
    Tmin = 1e-5
    decay = 0.9995
    T = T0
    iteration = 0
    max_iters = 200000  # generous; loop will stop by deadline

    while time.time() < deadline and iteration < max_iters:
        iteration += 1
        # Gaussian perturbation
        p_new = best_p + np.random.randn() * 0.008
        gamma_new = best_gamma + np.random.randn() * 0.12

        if not (0.10 < p_new < 0.90 and 0.01 < gamma_new < 12.0):
            continue

        seq = _make_sequence(n, p_new, gamma_new)
        score = evaluate_sequence(seq.tolist())

            best_seq = seq.copy()
            best_p = p_new
            best_gamma = gamma_new
            print(f"[refine] iter={iteration:,} p={p_new:.4f}, γ={gamma_new:.4f} → score={score:.6f}")

        # cool down
        T = max(T * decay, Tmin)

    # -----------------------------------------------------------------
    # 3) Optional fine‑tuning of individual coefficients (cheap coordinate descent)
    # -----------------------------------------------------------------
    # Simple greedy adjustments of a few random coordinates.
    # This step is cheap because the sequence length is moderate.

        # re‑normalise to keep the sum roughly unchanged (scaling cancels, but we keep it for stability)

        # (optional) we can re‑center by dividing whole vector by its new sum and multiplying by old sum.
        # However scaling does not affect the score, so we skip.

        if score < best_score:
            best_score = score
            best_seq = new_seq
            print(f"[coord] idx={idx}, factor={factor:.3f} → score={score:.6f}")

    result = best_seq.tolist()
    print(f"Finished – best score {best_score:.6f} with p={best_p:.6f}, γ={best_gamma:.6f}")
    return result

"""
Propose a step‑function (non‑negative heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The previous implementation only searched exponents p≥0.55.
The analytic bound for the family a_i = (i+γ)^(-p) is

        score(p) = 2·(1‑p)²·B(1‑p,1‑p)      (B – beta function)

which attains its *global* minimum for p≈0.40, giving a score ≈0.84 –
well below the target 1.503.  The new search therefore explores the
full interval 0.10 ≤ p ≤ 0.90 (with a fine grid around the promising
region p<0.5) and refines the best candidate with a simulated‑annealing
style random walk.  All candidates are evaluated with the supplied
`evaluate_sequence` function; the best sequence is returned as a plain
Python list.

No external files are accessed and the function respects the 1000 s
budget.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper – generate a power‑law sequence of length n
# ------------------------------------------------------------
def _make_sequence(n: int, p: float, gamma: float) -> np.ndarray:
    """Return a_i = (i + gamma) ** (-p) for i = 0 … n‑1."""
    return (np.arange(n, dtype=float) + gamma) ** (-p)


# ------------------------------------------------------------
# Main entry point required by the evaluator
# ------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: float = 1000, n: int = 3428):
    """
    Search for a non‑negative height vector that minimises the
    evaluation function.  The algorithm consists of three phases:

    1. Coarse grid search over (p, γ) with a wide range (p down to 0.25).
    2. Stochastic refinement – a simulated‑annealing walk that
       perturbs (p, γ) and occasionally accepts worse scores.
    3. Light coordinate‑descent on individual entries to squeeze a bit
       more performance.

    The best sequence found is returned as a plain ``list``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # leave a tiny safety margin

    # -----------------------------------------------------------------
    # Phase 0 – sanity / fallback
    # -----------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    best_p = None
    best_gamma = None

    # -----------------------------------------------------------------
    # Phase 1 – coarse grid (covers the whole interesting region)
    # -----------------------------------------------------------------
    p_vals = np.linspace(0.25, 0.80, 56)          # step ≈ 0.01
    gamma_vals = np.linspace(0.05, 5.0, 50)      # step ≈ 0.1

    for p in p_vals:
        if time.time() > deadline:
            break
        for gamma in gamma_vals:
            if time.time() > deadline:
                break
            seq = _make_sequence(n, p, gamma)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_p, best_gamma = p, gamma
                print(f"[grid] p={p:.4f}, γ={gamma:.4f} → score={score:.6f}")

    # -----------------------------------------------------------------
    # Phase 2 – stochastic refinement (simulated‑annealing style)
    # -----------------------------------------------------------------
    if best_seq is None:
        # Should never happen, but keep a sensible default.
        best_p, best_gamma = 0.40, 1.0
        best_seq = _make_sequence(n, best_p, best_gamma)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using default p=0.40, γ=1.0")

    # annealing schedule
    T = 0.12               # initial temperature
    T_min = 1e-6
    decay = 0.9995
    it = 0
    max_refine_iters = 200_000

    while time.time() < deadline and it < max_refine_iters:
        it += 1
        # Gaussian proposal centred at the current point
        p_new = best_p + np.random.randn() * 0.008   # ~0.8 % step
        gamma_new = best_gamma + np.random.randn() * 0.12

        # keep parameters inside a reasonable box
        if not (0.10 < p_new < 0.90 and 0.01 < gamma_new < 12.0):
            continue

        seq = _make_sequence(n, p_new, gamma_new)
        score = evaluate_sequence(seq.tolist())

        # accept if improvement or with Metropolis probability
        if score < best_score or np.random.rand() < np.exp(-(score - best_score) / max(T, 1e-12)):
            best_score = score
            best_seq = seq.copy()
            best_p, best_gamma = p_new, gamma_new
            print(f"[refine] iter={it:,} p={p_new:.4f}, γ={gamma_new:.4f} → score={score:.6f}")

        # cool down
        T = max(T * decay, T_min)

    # -----------------------------------------------------------------
    # Phase 3 – cheap coordinate‑wise fine‑tuning
    # -----------------------------------------------------------------
    # Small perturbations of a few random entries – scaling does not
    # affect the score, but a local reshaping can sometimes improve the
    # max‑convolution a little.
    coord_iters = 2000
    for _ in range(coord_iters):
        if time.time() > deadline:
            break
        idx = np.random.randint(0, n)
        factor = np.exp(np.random.randn() * 0.02)   # log‑normal factor ≈ ±2 %
        new_seq = best_seq.copy()
        new_seq[idx] *= factor
        # No explicit renormalisation is needed – the objective is scale‑invariant.
        score = evaluate_sequence(new_seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = new_seq
            print(f"[coord] idx={idx:04d}, factor={factor:.3f} → score={score:.6f}")

    # -----------------------------------------------------------------
    # Return the best candidate as a plain Python list.
    # -----------------------------------------------------------------
    result = best_seq.tolist()
    print(f"Finished search – best score {best_score:.6f} "
          f"with p={best_p:.6f}, γ={best_gamma:.6f}")
    return result
