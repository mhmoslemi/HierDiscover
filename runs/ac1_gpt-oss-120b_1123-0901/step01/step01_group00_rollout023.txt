"""
Propose a non‑negative step‑function sequence that minimizes

    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three stages:

1.  Quick parametric sweep (exponential‑type families) to obtain a decent
    initial candidate.
2.  Many short projected‑gradient‑descent runs on random starts.
    Convolutions are evaluated by FFT (O(n log n)) which allows large
    lengths (up to a few thousand) while keeping the runtime low.
3.  A simulated‑annealing refinement on the best sequence found so far.

All vectors are kept on the simplex  {x_i ≥ 0 , Σ x_i = √(2 n)} so that
the objective value equals the maximal autocorrelation entry.
The code respects the 1000 s time budget, prints intermediate progress
and finally returns the best sequence discovered.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x_i ≥ 0, Σ x_i = s }.
    Implementation of the algorithm by Wang & Carreira‑Perpiñán (2013).
    """
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def max_convolution_fft(a: np.ndarray):
    """
    Returns (max_value, argmax_index) of the linear convolution a*a
    using an FFT (zero‑padded to the next power of two).
    """
    n = a.shape[0]
    # next power of two >= 2*n-1
    m = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)
    conv = conv[: 2 * n - 1]          # discard circular tail
    k = int(np.argmax(conv))
    return float(conv[k]), k


def compute_evaluation(a: np.ndarray) -> float:
    """
    Exact evaluation of the objective for a given vector a.
    This matches the evaluate_sequence implementation (except the
    safety‑checks that are already satisfied by the search).
    """
    n = a.shape[0]
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return float("inf")
    max_val, _ = max_convolution_fft(a)
    return float(2.0 * n * max_val / (sum_a * sum_a))


def exponential_sequence(n: int, r: float) -> np.ndarray:
    """
    Geometric (exponential) sequence on the simplex.
    a_i = C * r**i  (r in (0,1]), C chosen so that Σ a_i = √(2 n).
    """
    if r >= 1.0 - 1e-12:                     # r == 1 → uniform sequence
        c = math.sqrt(2.0 * n) / n
        return np.full(n, c, dtype=float)

    idx = np.arange(n, dtype=float)
    # sum_{i=0}^{n-1} r**i = (1 - r**n) / (1 - r)
    geo_sum = (1.0 - r ** n) / (1.0 - r)
    c = math.sqrt(2.0 * n) / geo_sum
    return c * (r ** idx)

            max_val, k_max = max_convolution_fft(a)

            # compute sub‑gradient of the active convolution entry
            low = max(0, k_max - (n - 1))
            high = min(n - 1, k_max)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k_max - idx
            grad = np.zeros_like(a)
            grad[idx] = 2.0 * a[partner]

            max_grad = float(np.max(np.abs(grad)))
            if max_grad == 0.0:
                # all zero – random shake‑up
                a += np.random.randn(n) * (target_sum / n) * 0.01
                a = np.maximum(a, 0.0)
                a = project_onto_simplex(a, target_sum)
                continue

            # step size inversely proportional to the gradient magnitude
            eta = step_factor * max_val / max_grad
            a_new = a - eta * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = project_onto_simplex(a_new, target_sum)

            new_score = compute_evaluation(a_new)
            cur_score = compute_evaluation(a)

            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time():.2f}] New best (grad) = {best_score:.6f} (n={n})")

            # greedy acceptance with a tiny probability of uphill moves
            if new_score <= cur_score or np.random.rand() < 0.008:
                a = a_new
            else:
                # occasional random perturbation to escape plateaus
                if np.random.rand() < 0.015:
                    a = a_new

            # slowly damp the step factor – helps convergence on long runs
            step_factor = max(0.02, step_factor * 0.9995)

            # occasional global shake‑up
            if np.random.rand() < 0.02:
                a += np.random.randn(n) * (target_sum / n) * 0.02
                a = np.maximum(a, 0.0)
                a = project_onto_simplex(a, target_sum)

        # early stop if we already reached the target bound
        if best_score < 1.48:
            break

    # ------------------------------------------------------------------
    # 3) simulated annealing refinement on the current best sequence
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform sequence
        n = 1000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)
        best_score = compute_evaluation(best_seq)

    a = best_seq.copy()
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)

    anneal_start = time.time()
    anneal_deadline = deadline
    total_anneal = max(0.0, anneal_deadline - anneal_start)

    # annealing schedule (exponential cooling)
    T0 = 0.12
    T_end = 1e-6

    while time.time() < anneal_deadline:
        # pick two distinct indices, move a small amount from i→j
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        max_delta = a[i]
        delta = max_delta * np.random.rand() * 0.4          # up to 40 % of i's mass
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # no projection needed – sum stays constant and non‑negativity is kept

        new_score = compute_evaluation(a_new)
        cur_score = compute_evaluation(a)

        # temperature based on elapsed fraction
        elapsed = time.time() - anneal_start
        frac = elapsed / total_anneal if total_anneal > 0 else 1.0
        T = T0 * (T_end / T0) ** frac

        # Metropolis acceptance
        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time():.2f}] New best (anneal) = {best_score:.6f} (n={n})")

        # tiny random re‑perturbation to keep the walk ergodic
        if np.random.rand() < 0.003:
            a += np.random.randn(n) * (target_sum / n) * 0.008
            a = np.maximum(a, 0.0)
            a = project_onto_simplex(a, target_sum)

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain list)
    # ------------------------------------------------------------------
    if best_seq is None:
        # should never happen, but keep a safe uniform fallback
        n = 1000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    return best_seq.tolist()
