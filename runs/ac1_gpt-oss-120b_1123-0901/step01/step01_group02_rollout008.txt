"""
optimiser.py

Search algorithm for the step‑function problem.

The approach is a two‑phase optimiser:

1. **Soft‑max gradient descent** –  
   Instead of using the exact max of the auto‑convolution,
   we minimise a smooth surrogate
       L(a) = (1/β)·log(∑_k exp(β·b_k)),  b = a*a .
   Its gradient can be written analytically as a correlation
   between the current sequence and a weight vector that places
   most mass on the large convolution entries.  This gives a
   dense (global) descent direction and helps escape the
   plateaus that the naïve “max‑only’’ gradient gets stuck in.

2. **Linear‑programming boost** –  
   Every few thousand iterations we solve a small LP:

        maximise   ∑_j g_j
        subject to 0 ≤ g_j
                    (f * g)_k ≤ M   for all k

   where `f` is the current sequence and `M = max(f*f)`.
   The LP finds a complementary sequence `g` that can carry
   a lot of mass without increasing the current peak `M`.
   We then mix the current iterate with the (rescaled) LP
   solution.  Because the denominator of the evaluation
   function contains the square of the L¹‑norm, increasing
   that norm while keeping the peak unchanged strictly reduces
   the objective.

Both phases keep the iterates non‑negative, clip them to the
range required by `evaluate_sequence` and renormalise to the
fixed sum `√(2·n)`.  The best sequence seen so far is returned.

The optimiser runs for at most ``budget_s`` seconds (default 1000)
and prints progress information.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


def _conv_max(a: np.ndarray):
    """Return (max_value, index_of_max, full_convolution)."""
    b = np.convolve(a, a)
    max_val = b.max()
    max_idx = int(b.argmax())
    return max_val, max_idx, b


def _softmax_weights(b: np.ndarray, beta: float):
    """Soft‑max over the convolution vector b."""
    m = b.max()
    # subtraction of the max keeps the exponentials in a stable range
    exp_vals = np.exp(beta * (b - m))
    w = exp_vals / exp_vals.sum()
    return w


def _gradient_softmax(a: np.ndarray, beta: float):
    """
    Gradient of the soft‑max surrogate with respect to a.
    ∇L = 2 * corr(w, a)   (valid mode, length = len(a)).
    """
    # full convolution for the weights
    b = np.convolve(a, a)
    w = _softmax_weights(b, beta)
    # correlation gives exactly Σ_k w_k a_{k-i}
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad


def _build_conv_matrix(f: np.ndarray):
    """
    Build the dense matrix A (shape (2n‑1, n)) such that
    (A @ g) = f * g   (linear convolution) for any non‑negative g.
    """
    n = f.shape[0]
    m = 2 * n - 1
    # indices of the convolution matrix
    j = np.arange(n)                      # column indices
    k = np.arange(m)[:, None]             # row indices, shaped (m,1)
    idx = k - j                           # matrix of (row – col) indices
    mask = (idx >= 0) & (idx < n)         # valid positions where f contributes
    A = np.zeros((m, n), dtype=float)
    A[mask] = f[idx[mask]]
    return A

    Return a sequence of non‑negative heights that (hopefully) beats the
    current best bound.  The search consists of a soft‑max gradient descent

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Wall‑clock time budget in seconds.
    kwargs : dict
        Unused – kept for API compatibility.

    Returns
    -------
    list[float]
        The best sequence discovered within the allotted time.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin for wrap‑up

    # --------------------------------------------------------------
    # Initialise the sequence
    try:
        a = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
    except Exception:

    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)            # normalisation used by the paper
    a = a * (target_sum / a.sum())
    np.clip(a, 0.0, 1000.0, out=a)

    best_seq = a.copy()
    best_score = 2.0 * n * np.convolve(a, a).max() / (a.sum() ** 2)
    print(f"[init] n={n}, initial_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameters
    it = 0
