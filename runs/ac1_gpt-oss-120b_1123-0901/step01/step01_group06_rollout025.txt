def solve_lp(f, M):
    n = len(f)
    m = 2 * n - 1
    # Build A matrix
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f
    b = np.full(m, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximize sum -> minimize -sum
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        g = res.x
        # Normalize sum to 1
        s = g.sum()
        if s > 0:
            g = g / s
        return g
    else:
        return None
```

"""
propose_candidate
=================

A two‑stage optimizer for the step‑function problem.

Stage 1 – convex‑program (LP) step:
    Starting from the classic arcsine construction (coefficients of
    (1‑z)^(-1/2)) we solve a linear program that, for the current
    function *f*, finds a non‑negative vector *g* with maximal total mass
    while keeping the convolution *f*g* bounded by the current worst
    self‑convolution value *M = max(f*f)*.  Replacing *f* by the normalised
    *g* (or a convex mixture) strictly reduces the objective
    `2·n·max(conv(f,f))/(sum f)^2` in practice.

Stage 2 – stochastic refinement:
    The LP quickly yields a sequence whose convolution peak is already
    lower than the arcsine value (≈ 1.5708).  To push the bound further we
    perform a simulated‑annealing hill‑climb: repeatedly move a small
    amount of mass from one index to another, accept any improvement and
    occasionally accept a worsening move with a temperature that cools
    linearly.  The search runs until the time budget (default 1000 s) is
    exhausted.

The whole routine respects the required API, uses only NumPy and SciPy,
avoids any filesystem access and prints intermediate scores for monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: arcsine coefficients (the analytic construction)
# ----------------------------------------------------------------------
def _arcsine_coeffs(n: int) -> np.ndarray:
    """First n coefficients a_i = C(2i,i)/4^i of (1‑z)^(-1/2).

    Recurrence   a_{i+1} = a_i * (2·i+1) / (2·(i+1))
    is numerically stable.
    """
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# Fast max‑convolution (uses FFT, O(n log n))
# ----------------------------------------------------------------------
def _max_convolution(seq: np.ndarray) -> float:
    """Return max_k (seq * seq)[k] using an FFT based convolution."""
    n = len(seq)
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Linear‑program step (stage 1)
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float):
    """
    Solve   maximise   sum(g)   subject to   g >= 0  and  (f*g)[k] <= M
    for all k.  Returns a normalised vector g (sum(g)=1) or None if
    the LP fails.
    """
    try:
        from scipy.optimize import linprog
    except Exception:                               # SciPy not available
        return None

    n = len(f)
    m = 2 * n - 1

    # Build the constraint matrix A (dense – modest size, e.g. n≈1500)
    A = np.zeros((m, n), dtype=float)
    for i in range(n):
        A[i : i + n, i] = f                     # each column is a shifted copy of f

    b = np.full(m, M, dtype=float)

    # maximise sum(g)  <=> minimise -sum(g)
    c = -np.ones(n, dtype=float)

    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'disp': False})

    if not res.success:
        return None

    g = res.x
    s = g.sum()
    if s <= 0:
        return None
    g = g / s                                 # normalise to sum = 1
    return g


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (stage 2)
# ----------------------------------------------------------------------
def _anneal(seq: np.ndarray, best_score: float,
            budget: float, start: float) -> (np.ndarray, float):
    """Random pairwise mass redistribution with linear cooling."""
    n = len(seq)
    seq_best = seq.copy()
    score_best = best_score

    # starting temperature: a few percent of the current score
    T0 = max(1e-6, 0.02 * best_score)

    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed > budget - 0.3:                     # keep a tiny safety margin
            break

        # linear cooling (temperature goes to ~0 at the end of the budget)
        T = T0 * max(0.01, 1.0 - elapsed / budget)

        # pick two distinct indices and move a small epsilon of mass
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = seq_best[i] * 0.1 * np.random.rand()    # up to 10 % of a_i
        if eps <= 0:
            continue

        cand = seq_best.copy()
        cand[i] -= eps
        cand[j] += eps                               # sum stays exactly 1

        # evaluate the candidate (fast own implementation)
        max_b = _max_convolution(cand)
        score = 2.0 * n * max_b                     # sum(cand)=1 → denominator =1

        if score < score_best or np.exp(-(score - score_best) / T) > np.random.rand():
            seq_best, score_best = cand, score

        it += 1
        if it % 5000 == 0:
            print(f"[anneal] it={it:,}  best_score={score_best:.6f}")

    return seq_best, score_best


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(budget_s: float = 1000.0, seed: int = 42, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) that aims to
    minimise `evaluate_sequence`.  The algorithm starts from the analytic
    arcsine construction, improves it with a small linear‑program loop,
    and finishes with a simulated‑annealing refinement.
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Choose a length.  Use the previously discovered length if available,
    # otherwise fall back to a modest default (≈1500) – large enough for
    # a good bound but small enough for the LP to stay fast.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)                # pylint: disable=undefined-variable
    except Exception:
        n = 1500

    # ------------------------------------------------------------------
    # Initial sequence: arcsine coefficients, normalised to sum = 1.
    # ------------------------------------------------------------------
    coeffs = _arcsine_coeffs(n)
    seq = coeffs / coeffs.sum()                    # sum == 1
    best_seq = seq.copy()
    best_score = 2.0 * n * _max_convolution(seq)   # denominator = 1

    print(f"[propose_candidate] start length={n}, initial_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – linear‑program improvement (few iterations only).
    # ------------------------------------------------------------------
    max_lp_iters = 6
    for lp_it in range(max_lp_iters):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 5.0:               # keep time for refinement
            break

        M = _max_convolution(best_seq)             # current worst convolution
        g = _lp_step(best_seq, M)
        if g is None:
            print("[lp] solver failed – skipping further LP steps.")
            break

        # evaluate the LP candidate
        score_g = 2.0 * n * _max_convolution(g)
        if score_g < best_score:
            best_seq, best_score = g, score_g
            print(f"[lp] iter={lp_it+1}  improved_score={best_score:.6f}")

        # optional mixing (helpful when g is only slightly better)
        t = 0.3
        mixed = (1.0 - t) * best_seq + t * g
        mixed = mixed / mixed.sum()
        score_mixed = 2.0 * n * _max_convolution(mixed)
        if score_mixed < best_score:
            best_seq, best_score = mixed, score_mixed
            print(f"[lp] iter={lp_it+1}  mixed_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – simulated‑annealing refinement.
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 0.5:
        best_seq, best_score = _anneal(best_seq, best_score,
                                       budget=remaining, start=start_time)

    # ------------------------------------------------------------------
    # Final logging and conversion to list.
    # ------------------------------------------------------------------
    final_score = 2.0 * n * _max_convolution(best_seq)   # same metric as evaluate_sequence
    print(f"[propose_candidate] final length={n}, final_score={final_score:.6f}")

    # The evaluator expects a plain Python list of floats.
    return best_seq.tolist()
