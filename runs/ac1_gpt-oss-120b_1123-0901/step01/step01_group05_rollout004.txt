def fast_score(seq):
    n = len(seq)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    # use FFT
    m = 1
    # length for FFT: next power of 2 >= 2*n-1
    length = 1 << (2*n-1).bit_length()
    # but we could use np.fft.fft with size = 2*n-1
    # use real FFT? Using np.fft.rfft.
    # Implementation:
    size = 2*n-1
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)
    max_b = np.max(conv)
    return 2 * n * max_b / (sum_a * sum_a)

"""
Propose a candidate step‑function for the anti‑concentration task.

Algorithm
---------
1. Choose the length `n`.  If a previous sequence exists (`height_sequence_1`)
   we reuse its length, otherwise a default of 4000 is used.
2. Generate a few hand‑crafted shapes that are known to be good:
   * the “inverse‑sqrt’’ (central‑binomial) shape,
   * pure power‑law decays a_i ∝ (i+1)^{-p},
   * symmetric power‑law a_i ∝ ((i+1)*(n-i))^{-p},
   * a mixed two‑parameter family a_i ∝ (i+1)^{-p}·(n‑i)^{-q}.
   For each family a small grid of parameters is tried; the best shape so
   far (according to a fast FFT based score) is kept.
3. Starting from the best shape, a hill‑climbing / simulated‑annealing
   refinement is performed until the time budget expires.
   * The current convolution is computed with an FFT, the index `k_max`
     where the convolution attains its maximum is identified.
   * Indices that contribute most to this maximal entry are sampled
     with probability proportional to a_i·a_{k_max‑i}.
   * One (or the symmetric pair) of those entries is perturbed by a
     multiplicative log‑normal factor.  The size of the perturbation
     (the log‑normal σ) decays linearly with the remaining time
     (large steps early, fine tuning later).
   * If the new sequence improves the score it is accepted; the global
     best is updated whenever an improvement occurs.
4. The best sequence found is returned.  All entries are non‑negative
   and far below the 1000‑limit, so the security checks in
   `evaluate_sequence` do not alter the result.

The score being minimized is
    2·n·max(conv(a,a)) / (sum(a)^2)
which is scale‑invariant; therefore no explicit normalisation is needed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation (same objective as evaluate_sequence but much faster)
# ----------------------------------------------------------------------
def fast_score(seq: np.ndarray) -> float:
    """
    Return the objective 2*n*max(conv)/sum(seq)^2 using FFT.
    `seq` must be a 1‑D numpy array of non‑negative floats.
    """
    n = seq.shape[0]
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf

    # linear convolution via FFT (length 2*n-1)
    size = 2 * n - 1
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)
    max_b = np.max(conv.real)          # tiny imag parts are ignored
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Hand‑crafted families
# ----------------------------------------------------------------------
def inverse_sqrt_seq(n: int) -> np.ndarray:
    """a_i = binom(2i,i)/4**i  (same recurrence used in the reference)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def power_law_seq(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def symmetric_power_seq(n: int, p: float) -> np.ndarray:
    """a_i = ((i+1)*(n-i))^{-p}  (symmetric around the centre)."""
    i = np.arange(n, dtype=np.float64) + 1.0
    return (i * (n - i + 1.0)) ** (-p)


def mixed_power_seq(n: int, p: float, q: float) -> np.ndarray:
    """a_i = (i+1)^{-p} * (n-i)^{-q}."""
    i = np.arange(n, dtype=np.float64) + 1.0
    return (i ** (-p)) * ((n - i + 1.0) ** (-q))

# Main entry point required by the runner
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative floats that (hopefully) beats the current
    best bound.  The algorithm follows the outline described in the module
    docstring.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Determine the length of the construction
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        n = 4000                     # default length, large enough for a good shape

    # ------------------------------------------------------------------
    # 2. Scan a handful of analytic families
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    def try_candidate(arr: np.ndarray, label: str):
        nonlocal best_seq, best_val
        # clamp to the allowed range (unnecessary for our families but safe)
        arr = np.clip(arr, 0.0, 1000.0)
        val = fast_score(arr)
        if val < best_val:
            best_val = val
            best_seq = arr.copy()
            print(f"[family] {label:20s}  n={n:5d}  score={val:.6f}")

    # Inverse‑sqrt (central‑binomial) shape
    try_candidate(inverse_sqrt_seq(n), "inverse_sqrt")

    # Pure power‑law
    for p in np.linspace(0.45, 0.80, 8):
        try_candidate(power_law_seq(n, p), f"power_law p={p:.2f}")

    # Symmetric power‑law
    for p in np.linspace(0.20, 0.60, 7):
        try_candidate(symmetric_power_seq(n, p), f"symmetric p={p:.2f}")

    # Mixed two‑parameter family (small grid)
    for p in (0.30, 0.40):
        for q in (0.30, 0.40):
            try_candidate(mixed_power_seq(n, p, q), f"mixed p={p:.2f} q={q:.2f}")

    # ------------------------------------------------------------------
    # 3. Hill‑climbing / annealing refinement
    # ------------------------------------------------------------------
    # Ensure we have something to start from
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64)
        best_val = fast_score(best_seq)

    current_seq = best_seq.copy()
    current_val = best_val

    # Pre‑compute the FFT size (next power of two for a little speed)
    size_fft = 1 << ((2 * n - 1).bit_length())

    # Main optimisation loop
    while time.time() < deadline:
        # ----- recompute convolution of the current sequence -----
        fft_cur = np.fft.rfft(current_seq, n=size_fft)
        conv_cur = np.fft.irfft(fft_cur * fft_cur, n=size_fft).real
        k_max = int(np.argmax(conv_cur))
        max_conv = conv_cur[k_max]

        # ----- build a distribution that favours indices contributing to max -----
        idx = np.arange(n)
        partner = k_max - idx
        valid = (partner >= 0) & (partner < n)
        weights = np.zeros(n, dtype=np.float64)
        weights[valid] = current_seq[valid] * current_seq[partner[valid]]
        weight_sum = np.sum(weights)

        # With a small probability sample uniformly to keep exploration
        if weight_sum == 0.0 or rng.random() < 0.08:
            i = rng.integers(n)
        else:
            probs = weights / weight_sum
            i = rng.choice(n, p=probs)

        # Determine the symmetric partner (might be the same index)
        j = int(k_max - i)
        have_partner = (0 <= j < n) and (j != i)

        # ----- annealing schedule for the log‑normal perturbation -----
        time_left = deadline - time.time()
        frac = max(0.0, time_left / budget_s)
        sigma = 0.20 * frac + 0.005   # from ~0.20 down to ~0.005

        # Propose a multiplicative change (log‑normal)
        factor = np.exp(rng.normal(0.0, sigma))

        cand = current_seq.copy()
        cand[i] = max(0.0, cand[i] * factor)
        if have_partner and rng.random() < 0.5:
            cand[j] = max(0.0, cand[j] * factor)

        # Clip to the allowed range (well below 1000 anyway)
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_val = fast_score(cand)

        # ----- accept if better (simple hill‑climb) -----
        if cand_val < current_val:
            current_seq = cand
            current_val = cand_val
            # Update the global best as well
            if cand_val < best_val:
                best_seq = cand.copy()
                best_val = cand_val
                elapsed = time.time() - start_time
                print(
                    f"[improve] t={elapsed:6.2f}s  n={n:5d}  score={best_val:.6f}"
                )
        # Occasionally, even if not improving, replace the whole sequence
        # with the best known one to avoid drift (no‑op for us).

    # ------------------------------------------------------------------
    # 4. Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    return best_seq.tolist()
