def propose_candidate(seed=42, budget_s=1000, **kwargs):

    import numpy as np, math, time
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    # Determine length n
    try:
        n = len(height_sequence_1)
    except Exception:
        n = 2100
    target_sum = math.sqrt(2.0 * n)

    # Helper evaluate M
    def max_conv(a):
        return np.max(np.convolve(a, a))

    # Initialize best using provided if any
    best_seq = None
    best_score = float('inf')
    try:
        a0 = np.array(height_sequence_1, dtype=float)
        a0 = a0 * (target_sum / a0.sum())
        score0 = max_conv(a0)
        best_seq = a0.copy()
        best_score = score0
        print(f"[init] using height_sequence_1: score {best_score:.6f}")
    except Exception:
        pass

    # Determine list of initializers to try
    def generate_initializers():
        # uniform
        yield np.full(n, target_sum / n)
        # Dirichlet (random) a few
        for _ in range(4):
            x = np.random.rand(n)
            yield x / x.sum() * target_sum
        # Sparse binary uniform weight
        for _ in range(4):
            m = int(np.random.uniform(0.05, 0.3) * n)  # between 5% and 30% non-zero
            idx = np.random.choice(n, m, replace=False)
            a = np.zeros(n)
            a[idx] = target_sum / m
            yield a

    # Optimization routine
    def optimize(init_a, max_iters):
        nonlocal best_seq, best_score
        a = init_a.copy()
        # ensure sum correct
        if a.sum() == 0:
            a[:] = target_sum / n
        else:
            a = a * (target_sum / a.sum())
        # Adam moments
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        lr0 = 0.6
        decay = 5e-5
        # Softmax temperature schedule
        beta_temp0 = 0.5
        beta_inc = 0.0005
        for it in range(1, max_iters+1):
            # compute convolution
            conv = np.convolve(a, a)
            max_c = conv.max()
            # softmax over conv
            beta = beta_temp0 + beta_inc * it
            if beta > 30:
                beta = 30
            # stability: subtract max_c
            tmp = np.exp(beta * (conv - max_c))
            w = tmp / tmp.sum()
            # gradient: 2*correlation of w with a (valid)
            grad = 2 * np.correlate(w, a, mode='valid')
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1**it)
            v_hat = v / (1 - beta2**it)
            lr = lr0 / (1 + decay * it)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = a - step
            # projection
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 0:
                a_new = a_new * (target_sum / s)
            else:
                a_new = a  # fallback
            # jitter occasional
            if it % 500 == 0:
                jitter = np.random.randn(n) * target_sum * 0.0005 / n
                a_new = np.maximum(a_new + jitter, 0.0)
                a_new = a_new * (target_sum / a_new.sum())
            a = a_new
            # evaluate occasionally
            if it % 50 == 0:
                cur_M = max_conv(a)
                if cur_M < best_score:
                    best_score = cur_M
                    best_seq = a.copy()
                    print(f"[{time.time() - start:.1f}s] new best M={best_score:.6f} (it {it})")
            # check time
            if time.time() > deadline:
                break
        # final evaluation
        final_M = max_conv(a)
        if final_M < best_score:
            best_score = final_M
            best_seq = a.copy()
            print(f"[{time.time() - start:.1f}s] new best after run M={best_score:.6f}")
        return

    # Main loop: run optimizer on each init until time over
    for init in generate_initializers():
        if time.time() > deadline:
            break
        # allocate iteration budget based on remaining time: assume ~0.001s per iteration
        # Determine max_iters = int((deadline - time.time()) / 0.001)
        remaining = deadline - time.time()
        est_iter_time = 0.0008  # estimate (empirical)
        max_iters = max(1000, int(remaining / est_iter_time))
        # but cap to e.g. 200000 to avoid too many
        max_iters = min(max_iters, 200000)
        optimize(init, max_iters)

    # fallback: if best_seq still None, use uniform
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)
    # Final clamp to allowed range (0-1000)
    np.clip(best_seq, 0.0, 1000.0, out=best_seq)
    # Return as list
    print(f"[done] best M={best_score:.6f}")
    return best_seq.tolist()

"""
Propose a step‑height sequence by minimizing the peak of the
self‑convolution (i.e. 2·n·max(a*a)/(sum a)^2).  

The algorithm is a full‑gradient descent on a soft‑max approximation
of the max‑convolution, using an Adam‑style update, periodic jitter
and a simple projection onto the non‑negative simplex.
It starts from the best known construction (``height_sequence_1`` if
available) and also tries a few random initialisations.
The whole search stops after the allotted wall‑clock time and the
best sequence seen is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility: peak of the self‑convolution (the true objective value)
def max_convolution(a: np.ndarray) -> float:
    """Return max_k (a*a)[k] where * is the discrete convolution."""
    return float(np.max(np.convolve(a, a)))


# ----------------------------------------------------------------------
# Soft‑max gradient of the max‑convolution
def softmax_gradient(a: np.ndarray, beta: float):
    """
    Compute a smooth approximation of the gradient of
    max_k (a*a)[k] w.r.t. ``a`` using a temperature ``beta``.
    Returns (grad, max_val) where ``grad`` has the same shape as ``a``.
    """
    conv = np.convolve(a, a)                     # length 2n‑1
    max_val = conv.max()
    # stability: shift by the maximum so the largest exponent is 0
    w = np.exp(beta * (conv - max_val))
    w /= w.sum()
    # gradient = 2 * correlation(w, a)  (valid gives length n)
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad, max_val


# ----------------------------------------------------------------------
# Adam optimiser for one run starting from ``init_a``.
def run_optimization(init_a: np.ndarray,
                     n: int,
                     target_sum: float,
                     start_time: float,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float):
    """
    Perform a constrained Adam optimisation on ``init_a``.
    The sequence is kept non‑negative and re‑scaled after each step
    to keep its ℓ₁‑norm equal to ``target_sum``.
    The function returns the (possibly improved) best sequence and score.
    """
    a = init_a.copy()
    # enforce the normalisation immediately
    if a.sum() == 0.0:
        a[:] = target_sum / n
    else:
        a = a * (target_sum / a.sum())

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.6            # base learning rate
    decay = 5e-5         # simple 1/(1+decay·t) decay
    beta_temp0 = 0.5     # starting temperature for the soft‑max
    beta_inc = 0.0005    # linear increase of temperature

    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- gradient computation ------------------------------------
        beta = beta_temp0 + beta_inc * it
        if beta > 30.0:               # avoid overflow in exp
            beta = 30.0
        grad, cur_max = softmax_gradient(a, beta)

        # ---- Adam update --------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + decay * it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step

        # ---- projection onto the feasible set -----------------------
        a_new = np.maximum(a_new, 0.0)               # non‑negative
        s = a_new.sum()
        if s > 0.0:
            a_new = a_new * (target_sum / s)        # re‑scale ℓ₁‑norm
        else:                                        # should never happen
            a_new = a.copy()

        # ---- occasional jitter to escape shallow plateaus ----------
        if it % 500 == 0:
            jitter = np.random.randn(n) * target_sum * 5e-4 / n
            a_new = np.maximum(a_new + jitter, 0.0)
            a_new = a_new * (target_sum / a_new.sum())

        a = a_new

        # ---- cheap evaluation every few iterations ------------------
        if it % 50 == 0:
            cur_score = max_convolution(a)          # equals the true objective
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                elapsed = time.time() - start_time
                print(f"[{elapsed:6.1f}s] new global best = {best_score:.6f}")

        # optional early stop if we already beat the target
        if best_score < 1.503:
            break

    # final check for this run
    final_score = max_convolution(a)
    if final_score < best_score:
        best_score = final_score
        best_seq = a.copy()
        elapsed = time.time() - start_time
        print(f"[{elapsed:6.1f}s] final improvement of run = {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the
    evaluation function.  The routine runs a time‑bounded Adam
    optimisation starting from several initialisations (including the
    supplied ``height_sequence_1`` if present) and returns the best
    sequence found.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin so we never exceed the budget
    deadline = start + budget_s - 2.0

    # ------------------------------------------------------------------
    # Determine the problem size
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 2100                            # historic default
    target_sum = math.sqrt(2.0 * n)        # the normalisation used throughout

    # ------------------------------------------------------------------
    # Initialise the best solution with the provided construction, if any
    best_seq = None
    best_score = float("inf")
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        base = base * (target_sum / base.sum())
        best_seq = base.copy()
        best_score = max_convolution(best_seq)
        print(f"[init] starting from height_sequence_1, score = {best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Helper to generate a handful of diverse start vectors
    def generate_initialisers():
        # uniform start
        yield np.full(n, target_sum / n)

        # a few Dirichlet‑like random starts
        for _ in range(4):
            x = np.random.rand(n)
            yield x / x.sum() * target_sum

        # sparse uniform starts (5‑30 % non‑zero)
        for _ in range(4):
            density = np.random.uniform(0.05, 0.30)
            m = max(1, int(density * n))
            idx = np.random.choice(n, m, replace=False)
            a = np.zeros(n)
            a[idx] = target_sum / m
            yield a

        # a “Gaussian‑ish” start: sample a discrete normal then normalise
        x = np.arange(n)
        sigma = n / 6.0
        gauss = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
        gauss += 1e-12                     # avoid exact zeros
        yield gauss / gauss.sum() * target_sum

    # ------------------------------------------------------------------
    # Main optimisation loop – run as many independent starts as time permits
    for init_a in generate_initialisers():
        if time.time() >= deadline:
            break
        # Run a full optimisation from this start
        best_seq, best_score = run_optimization(
            init_a,
            n=n,
            target_sum=target_sum,
            start_time=start,
            deadline=deadline,
            best_seq=best_seq if best_seq is not None else init_a,
            best_score=best_score,
        )
        # Very early exit if we already beat the target bound
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)

    # Clamp to the limits enforced by ``evaluate_sequence`` (0 ≤ x ≤ 1000)
    np.clip(best_seq, 0.0, 1000.0, out=best_seq)

    elapsed_total = time.time() - start
    print(f"[done] total time = {elapsed_total:.1f}s, best score = {best_score:.6f}")
    return best_seq.tolist()
