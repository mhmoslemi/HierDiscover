"""
propose_candidate
=================

Search for a non‑negative step‑function that beats the simple arcsine
construction.  The algorithm starts from the (truncated) arcsine
distribution (which gives the asymptotic constant π/2) and then performs a
large simulated‑annealing / hill‑climbing loop.

* A transfer of a small amount δ of mass from a larger coefficient
  `src` to a smaller coefficient `dst` changes the auto‑convolution
  `conv = a * a` only on two shifted copies of `a`:
      conv_new = conv
                 + 2·δ·(shift(dst,a) – shift(src,a))
                 + δ²·e_{2·src} – 2·δ²·e_{src+dst} + δ²·e_{2·dst}
  (where `shift(i,a)` is `a` zero‑padded and shifted by `i`).

  Using NumPy slices this update costs O(n) instead of recomputing the
  whole convolution.

* The move size starts large (up to 20 % of the source entry) and is
  slowly reduced to a tiny value.  Early on we mostly take `src` as the
  current largest entry (biased search) and later switch to random
  indices, allowing both coarse and fine adjustments.

* The objective is `score = 2·n·max(conv) / (sum(a)²)`.  The sum stays
  (approximately) constant because every move conserves total mass, so
  we can compute the score from `conv.max()` without calling the heavy
  `evaluate_sequence` on each iteration.  The exact evaluator is used
  only for the final result and when an improvement is found.

The loop runs until the allocated time (default 1000 s) is exhausted,
printing occasional improvements.  The best sequence found is returned
as a plain Python list.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    # -------------------------------------------------------------
    # 1.  Length of the step‑function
    # -------------------------------------------------------------
    rng = np.random.default_rng(seed)
    n = kwargs.get("length")
    if n is None:
        try:
            n = len(height_sequence_1)       # type: ignore[name-defined]
        except Exception:
            n = 6027                         # fallback used previously

    # -------------------------------------------------------------
    # 2.  Initialise with the discrete arcsine distribution
    #     a_k ∝ (2k choose k) / 4**k  (normalised to sum = 1)
    # -------------------------------------------------------------
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        # recurrence for (2k choose k) / 4**k
        a[k] = a[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    a /= a.sum()                     # now Σ a = 1

    # -------------------------------------------------------------
    # 3.  First evaluation (exact, for logging)
    # -------------------------------------------------------------
    best_seq = a.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:               # safety – should never happen
        best_score = float(2 * n * np.convolve(a, a).max())
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # -------------------------------------------------------------
    # 4.  Pre‑compute the convolution and set up optimisation loop
    # -------------------------------------------------------------
    conv = np.convolve(a, a)                     # length 2·n‑1
    # parameters for the annealing schedule
    max_move_initial = 0.20    # up to 20 % of the source mass at start
    max_move_final   = 0.001   # 0.1 % at the end
    p_big_initial    = 0.70    # probability to pick the current biggest entry
    p_big_final      = 0.10    # become more random later

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a little safety margin
    it = 0
    # -------------------------------------------------------------
    # 5.  Main simulated‑annealing / hill‑climbing loop
    # -------------------------------------------------------------
    while time.time() < deadline:
        elapsed = time.time() - start_time
        t_frac = elapsed / (budget_s - 0.5)          # 0 … 1

        # linearly decay the move size and the bias towards the biggest entry
        max_move = max_move_initial * (1.0 - t_frac) + max_move_final * t_frac
        p_big    = p_big_initial    * (1.0 - t_frac) + p_big_final    * t_frac

        # ---- choose source and destination indices -----------------
        if rng.random() < p_big:
            src = int(np.argmax(a))                 # deterministic biggest
        else:
            src = rng.integers(n)
        # pick a destination different from src
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # always move from the larger entry to the smaller one
        if a[src] < a[dst]:
            src, dst = dst, src

        src_val = a[src]
        if src_val <= 0.0:
            continue                                 # nothing to move

        # ---- magnitude of the transfer ----------------------------
        delta = max_move * rng.random() * src_val    # 0 … max_move·src_val
        if delta <= 0.0:
            continue

        # ---- incremental update of the convolution ---------------
        conv_new = conv.copy()

        # shift contributions (vectorised slice updates)
        conv_new[src:src + n] -= 2.0 * delta * a
        conv_new[dst:dst + n] += 2.0 * delta * a

        # quadratic δ² terms at the three affected indices
        conv_new[2 * src]     += delta * delta
        conv_new[2 * dst]     += delta * delta
        conv_new[src + dst]   -= 2.0 * delta * delta

        # ---- compute the (approximate) score ----------------------
        # Σa is unchanged (≈1), so we can omit it in the ratio.
        cand_score = 2.0 * n * conv_new.max() / (a.sum() * a.sum())

        # ---- acceptance -------------------------------------------------
        improved = cand_score < best_score - 1e-12
        if improved:
            # accept the move
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            best_seq = a.copy()
            best_score = cand_score
            # verify with the official evaluator (cheap because we do it
            # only on improvements)
            try:
                exact = evaluate_sequence(best_seq.tolist())
                if exact < best_score:

            except Exception:

            if it % 1000 == 0:

            # temperature decays to a tiny value
            T = 0.001 * (1.0 - t_frac) + 1e-6
            if T > 0.0 and rng.random() < np.exp(-(cand_score - best_score) / T):
                a[src] -= delta
                a[dst] += delta
                conv = conv_new

        it += 1

        if it % 10000 == 0:
            total = a.sum()
            if abs(total - 1.0) > 1e-10:
                a /= total
                conv = np.convolve(a, a)

    # -------------------------------------------------------------
    # 6.  Final exact evaluation and return
    # -------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] iterations={it}, best score={final_score:.6f}")
    return best_seq.tolist()
