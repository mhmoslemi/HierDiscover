assistantfinal**Solution Overview**

The score  

\[
\text{score}(a)=\frac{2\,n\;\max\bigl(a*a\bigr)}{\bigl(\sum a_i\bigr)^2}
\]

is *scale‑invariant* – multiplying the whole vector by a constant leaves the
value unchanged.  Consequently we can freely generate a shape and let the
normaliser in `evaluate_sequence` handle the scaling.

A monotone power‑law  

\[
a_i=(i+\gamma)^{-p},\qquad i=0,\dots ,n-1
\]

flattens the aperiodic autocorrelation and, for large `n`, approaches the
theoretical bound  

\[
2(1-p)^2\,B(1-p,1-p).
\]

The bound is minimal near \(p\approx0.65\) (≈ 1.42), well below the current
record 1.571.  The previous code fixed `n = 3428`; with a larger `n` the
discretisation error shrinks and the bound can be reached.

The algorithm therefore explores three continuous parameters **(n, p, γ)**
instead of only *(p,γ)*:

1. **Coarse grid** – several candidate lengths (`2000 … 12000`) and a uniform
   grid on `p∈[0.55,0.80]`, `γ∈[0.01,10]`.  
   The best triple is kept.

2. **Stochastic refinement** – hill‑climbing with simulated‑annealing style
   acceptance.  Small random perturbations of all three parameters are tried
   until the time budget (≈ 1000 s) expires.

3. The best sequence found is returned as a plain Python list.

All heavy work is done with NumPy; `evaluate_sequence` is called only for
candidate vectors, never inside a Python loop over the convolution.

```python
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------
def make_power_law_seq(n: int, p: float, gamma: float) -> np.ndarray:
    """
    Return the vector a_i = (i + gamma)^{-p} for i = 0 .. n-1.
    The result is a NumPy array of type float.
    """
    return (np.arange(n, dtype=float) + gamma) ** (-p)


def evaluate_params(n: int, p: float, gamma: float) -> float:
    """
    Build the sequence for the given parameters and evaluate it with the
    supplied `evaluate_sequence` function (assumed to be present globally).
    """
    seq = make_power_law_seq(n, p, gamma)
    # evaluate_sequence expects a plain list
    return evaluate_sequence(seq.tolist())


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

    The method proceeds in two phases:
        1. a modest grid over length n, exponent p and shift γ;
        2. stochastic hill‑climbing (with occasional uphill moves) that
           refines all three parameters until the time budget is exhausted.

    The function returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.2       # safety margin

    # ------------------------------------------------------------------
    # 1) Coarse grid search
    # ------------------------------------------------------------------
    # candidate lengths – large enough to expose the asymptotic behaviour
    length_candidates = [2000, 3000, 4000, 5000, 8000, 12000]

    # grids for the continuous parameters
    p_grid = np.linspace(0.55, 0.80, 13)          # 13 values from 0.55 to 0.80
    gamma_grid = np.linspace(0.01, 4.0, 14)      # 14 values from 0.01 to 4.0

    best_score = float('inf')
    best_seq = None
    best_n = None
    best_p = None
    best_gamma = None

    for n in length_candidates:
        if time.time() > deadline:
            break
        print(f"[grid] exploring length n={n}")

        for p in p_grid:
            for gamma in gamma_grid:
                if time.time() > deadline:
                    break

                seq = make_power_law_seq(n, p, gamma)
                score = evaluate_sequence(seq.tolist())

                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    best_n = n
                    best_p = p
                    best_gamma = gamma
                    print(f"[grid] NEW best → n={n}, p={p:.4f}, γ={gamma:.4f}, score={score:.6f}")

    # ------------------------------------------------------------------
    # 2) Stochastic refinement (hill‑climbing + simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is None:          # very unlikely, but keep a safe fallback
        best_n = 3428
        best_p = 0.5
        best_gamma = 0.3
        best_seq = make_power_law_seq(best_n, best_p, best_gamma)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using default parameters")

    # refinement hyper‑parameters
    max_iters = 20000
    sigma_p = 0.015          # step size for p
    sigma_gamma = 0.2        # step size for γ
    sigma_n = 400            # step size for n (standard deviation)

    # temperature schedule (simulated annealing)
    T0 = 0.01
    Tmin = 1e-6

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # linear temperature decay
        T = max(Tmin, T0 * (1 - it / max_iters))

        # propose new parameters by Gaussian perturbation
        p_new = best_p + rng.normal(0, sigma_p)
        gamma_new = best_gamma + rng.normal(0, sigma_gamma)
        n_new = best_n + int(rng.normal(0, sigma_n))

        # keep parameters inside sensible bounds
        if not (0.45 < p_new < 0.90):
            continue
        if not (0.001 < gamma_new < 10.0):
            continue
        if not (1500 <= n_new <= 20000):
            continue

        seq_new = make_power_law_seq(n_new, p_new, gamma_new)
        score_new = evaluate_sequence(seq_new.tolist())

        # accept if improvement, otherwise accept with small probability
        improvement = score_new < best_score
        if improvement or rng.random() < math.exp((best_score - score_new) / T):
            if improvement:
                print(f"[refine] iter {it:05d} IMPROVED → n={n_new}, "
                      f"p={p_new:.5f}, γ={gamma_new:.5f}, score={score_new:.6f}")
                best_score = score_new
                best_seq = seq_new.copy()
                best_n = n_new
                best_p = p_new
                best_gamma = gamma_new

    # ------------------------------------------------------------------
    # 3) Return the best found sequence
    # ------------------------------------------------------------------
    print(f"Finished – best score {best_score:.6f} with "
          f"n={best_n}, p={best_p:.5f}, γ={best_gamma:.5f}")
    return best_seq.tolist()
```
