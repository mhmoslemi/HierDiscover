def _grad_softmax(a, conv, alpha):
    n = len(a)
    L = len(conv) # 2n-1
    max_conv = conv.max()
    # stable exponent
    exp_terms = np.exp(alpha * (conv - max_conv))
    w = exp_terms / exp_terms.sum()
    # reverse w
    w_rev = w[::-1]  # length L
    # compute convolution of a and w_rev via FFT
    size = 1 << ((n + L - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    fft_w_rev = np.fft.rfft(w_rev, size)
    conv_rev_full = np.fft.irfft(fft_a * fft_w_rev, size)[: n + L - 1]
    # grad_i = 2 * conv_rev_full[L-1 - i]
    grad = 2.0 * conv_rev_full[L-1 : L-n-1 : -1]  # slice descending
    return grad

def _euclidean_proj_simplex(v):
    """Return projection onto simplex {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.allclose(v.sum(), 1) and np.min(v) >= 0:
        return v.copy()
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho
    rho = np.nonzero(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def _grad_softmax_fft(a, conv, alpha):
    n = len(a)
    L = len(conv)
    max_conv = conv.max()
    # compute softmax weights, stable
    exp_terms = np.exp(alpha * (conv - max_conv))
    w = exp_terms / exp_terms.sum()

The evaluation function is  

\[
\text{score}(a)=\frac{2\,n\,
\displaystyle\max_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 ,
\]

where \((a*a)_k\) denotes the linear self‑convolution.  
Because the score is homogeneous of degree 0 we may restrict ourselves to
probability vectors \(\sum_i a_i=1\).  
The problem therefore becomes

*minimise*  \(\displaystyle M(a)=\max_k(a*a)_k\)  *subject to*  \(a_i\ge 0,\;\sum a_i =1\).

The classic sub‑gradient method (single active index) stops at the
central‑binomial distribution and yields the bound \(M\approx\pi/(4n)\)
(score ≈ π/2 ≈ 1.57086).  
To move past this stationary point we use a **smooth approximation of the
max‑operator** together with an Adam‑style optimiser on the simplex.

* **Smooth max** – replace the hard maximum by a soft‑max:
  \[
  \phi_\alpha(a)=\frac{1}{\alpha}\log\!\bigl(\sum_k e^{\alpha\,(a*a)_k}\bigr)
  \xrightarrow[\alpha\to\infty]{}\max_k (a*a)_k .
  \]
  The gradient of \(\phi_\alpha\) is a convex combination of the individual
  gradients of the convolution entries, which prevents the algorithm from
  getting stuck at a single active constraint.

* **Gradient on the simplex** – after a gradient step we project back onto
  the probability simplex using the Euclidean projection (the standard
  `v‑sorted` algorithm).

* **Adam learning‑rate adaptation** – per‑coordinate moments give a
  self‑scaling step size; an outer decay \(\eta_t\propto 1/\sqrt t\) keeps the
  updates diminishing with time.

* **Top‑K weighting** – instead of a full soft‑max we take the
  top‑\(k\) convolution entries (the ones that dominate the peak) and give
  them equal weight.  The value of \(k\) is reduced during the run, ending
  with a pure sub‑gradient (k = 1) for fine polishing.

* **Random perturbations** – occasional tiny Gaussian noise helps the
  optimiser escape shallow plateaus.

The whole routine runs until the allocated time expires (≈ 1000 s) and
returns the best vector found.  It works for any existing `height_sequence_1`
or builds a fresh central‑binomial start if none is present.

```python
"""
Propose a step‑function height sequence that (hopefully) improves the
known bound 1.570861 → something below 1.503.

The algorithm:
    * Fix the sum of the coefficients to 1 (probability simplex).
    * Initialise with the central‑binomial distribution (or the provided
      `height_sequence_1`).
    * Repeatedly minimise a smooth approximation of the max‑convolution:
        – compute the linear convolution a*a by FFT,
        – pick the top‑k largest convolution entries (k shrinks during the run) ,
        – build a weight vector w (equal weight on those top‑k entries),
        – obtain the gradient  g = 2 * (a * w_rev)   (cross‑correlation, via FFT),
        – perform an Adam update, project the result onto the simplex,
        – accept the step if the true peak does not increase,
          otherwise shrink the step size (back‑tracking) or apply a tiny random kick.
    * Keep the best vector seen so far and return it.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, sum(x) = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho index
    rho = np.nonzero(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)‑1) using a real FFT.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _grad_top_k(a: np.ndarray, conv: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of the (averaged) top‑k soft‑max of the convolution.
    The weight vector w has value 1/k on the k largest entries of conv,
    zero elsewhere.  The gradient is
        g_i = 2 * sum_{j} w_{i+j} * a_j,
    which is a cross‑correlation computed by a single FFT.
    """
    n = a.shape[0]
    L = conv.shape[0]          # = 2*n‑1
    # indices of the k largest convolution entries
    if k >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(-conv, k - 1)[:k]
    w = np.zeros(L, dtype=np.float64)
    w[idx] = 1.0 / k
    # cross‑correlation a * w_rev  (w reversed)
    w_rev = w[::-1]
    size = 1 << ((n + L - 1).bit_length())
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv_rev = np.fft.irfft(fa * fw, size)[: n + L - 1]
    # extract the required n entries (see analysis)
    grad = 2.0 * conv_rev[L - 1 : L - n - 1 : -1]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine – exposed as `propose_candidate`
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    evaluate_sequence(...).  The routine runs for at most `budget_s`
    seconds and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0    # a safety margin

    # --------------------------------------------------------------
    # 1. Choose the length and initialise the vector `a`.
    # --------------------------------------------------------------
    try:
        # use the previously discovered construction if it exists
        a0 = np.array(height_sequence_1, dtype=np.float64)
        n = a0.size
    except NameError:
        # fallback: central‑binomial coefficients, length ≈ 6000
        n = 6047
        a0 = np.empty(n, dtype=np.float64)
        a0[0] = 1.0
        for i in range(1, n):
            a0[i] = a0[i - 1] * (2.0 * i - 1.0) / (2.0 * i)

    # Normalise to a probability vector (sum = 1)
    a = a0.copy()
    a = _project_onto_simplex(a)

    # --------------------------------------------------------------
    # 2. Optimisation hyper‑parameters
    # --------------------------------------------------------------
    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # learning‑rate base (will be decayed with 1/√t)
    base_lr = 0.2

    # schedule for the number of active indices (top‑k)
    def current_k(t_iter: int) -> int:
        """Return the current k (size of active set) depending on iteration."""
        if t_iter < 2000:
            return max(1, int(0.05 * (2 * n - 1)))   # ≈5 % of the convolution length
        if t_iter < 15000:

        if t_iter < 50000:
            return max(1, int(0.005 * (2 * n - 1)))  # ≈0.5 %
        return 1                                   # pure sub‑gradient at the end

    # bookkeeping of the best solution seen so far
    conv = _full_convolution_fft(a)
    best_max = conv.max()
    best_a = a.copy()
    best_score = 2.0 * n * best_max

    iteration = 0
    print(f"Start optimisation, length = {n}, initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Main optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- compute convolution and the current peak -----
        conv = _full_convolution_fft(a)
        cur_max = conv.max()

        # ----- gradient (top‑k soft‑max) -----
        k = current_k(iteration)
        grad = _grad_top_k(a, conv, k)

        # ----- Adam update with a decaying outer step size -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # learning‑rate that decays like 1/√t
        lr = base_lr / math.sqrt(iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step
        a_candidate = _project_onto_simplex(a_candidate)

        # ----- evaluate candidate -----
        conv_candidate = _full_convolution_fft(a_candidate)
        cand_max = conv_candidate.max()

        # ----- back‑tracking line search if needed -----
        if cand_max > cur_max:
            # try a few reductions of the step size
            success = False
            for shrink in (0.5, 0.25, 0.125):
                step_shrunk = step * shrink
                a_tmp = a - step_shrunk
                a_tmp = _project_onto_simplex(a_tmp)
                conv_tmp = _full_convolution_fft(a_tmp)
                if conv_tmp.max() <= cur_max:
                    a_candidate = a_tmp
                    cand_max = conv_tmp.max()
                    success = True
                    break
            if not success:
                # small random kick – helps to escape plateaus
                a_candidate = a + np.random.randn(n) * 1e-5
                a_candidate = _project_onto_simplex(a_candidate)
                conv_candidate = _full_convolution_fft(a_candidate)
                cand_max = conv_candidate.max()

        # ----- accept the candidate if it does not worsen the true peak -----
        if cand_max <= cur_max:
            a = a_candidate
            cur_max = cand_max

        # ----- bookkeeping of the best solution -----
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            best_score = 2.0 * n * best_max
            print(f"Iter {iteration:7d}: new best score = {best_score:.6f} "
                  f"(peak={best_max:.8f}, k={k})")

        if iteration % 3000 == 0:
            a = _project_onto_simplex(a + np.random.randn(n) * 5e-4)

    # --------------------------------------------------------------
    # 4. Return the best vector found (as plain Python floats)
    # --------------------------------------------------------------
    final_score = 2.0 * n * best_max
    print(f"Finished after {iteration} iterations. Best score = {final_score:.6f}")
    return best_a.tolist()
```
