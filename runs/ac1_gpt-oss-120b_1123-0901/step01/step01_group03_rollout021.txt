    # conv: array of length 2n-1: conv_i = sum_j a_j a_{i-j}
    exp_vals = np.exp(lam * conv - lam * np.max(lam*conv))  # shift for stability
    w = exp_vals / exp_vals.sum()  # shape (2n-1,)
    # compute gradient: g = 2 * correlation(a, w)
    # correlation = np.convolve(a, w[::-1])  # length n+2n-2 = 3n-2
    # index offset: corr_offset = len(w)-1 = 2n-2 -> g_j = corr[2n-2 + j] 
    w_rev = w[::-1]
    size = 1 << ((len(a)+len(w_rev)-1).bit_length())
    # FFT convolution
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv_corr = np.fft.irfft(fa * fw, size)
    # conv_corr is length size, but we only need first len(a)+len(w_rev)-1 elements
    corr = conv_corr[:len(a)+len(w_rev)-1]  # length 3n-2
    offset = len(w)-1  # =2n-2
    g = 2.0 * corr[offset: offset + len(a)]  # slice length n
    return g

def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    # Choose length n
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 5000

    # Initialise a
    # central binomial coefficients

    a[0] = 1.0

        a[i] = a[i-1] * (2.0 * i - 1.0) / (2.0 * i)

    a = _simplex_projection(a)

    # Optionally take random mix of existing if available: use height_sequence_1 if sum>0
    try:
        old = np.array(height_sequence_1, dtype=np.float64)
        if len(old) == n:

            a = 0.5 * a + 0.5 * _simplex_projection(old)
    except NameError:
        pass

    best_a = a.copy()
    best_score = float('inf')
    # compute initial score
    conv = _full_convolution(a)
    score = 2.0 * n * np.max(conv) / (np.sum(a)**2)
    best_score = score
    best_a = a.copy()
    print(f"Initial score: {best_score:.6f}")

    # parameters
    eta0 = 0.1   # step size factor
    lam = 5.0    # initial temperature for softmax
    lambda_growth = 1.001  # growth per iteration
    max_lambda = 200.0
    iteration = 0

    while time.time() < deadline:
        # conv
        conv = _full_convolution(a)
        max_val = conv.max()
        # optional: compute exact score
        score = 2.0 * n * max_val / (np.sum(a)**2)  # sum=1 -> score = 2*n*max
        # track best
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"Iter {iteration:6d}: new best {best_score:.6f}, lambda {lam:.3f}")

        # compute softmax weights
        # shift for numerical stability
        scaled = lam * conv
        max_scaled = np.max(scaled)
        exp_vals = np.exp(scaled - max_scaled)
        w = exp_vals / exp_vals.sum()  # shape 2n-1

        # compute gradient
        # w_rev = w[::-1]
        # convolution via FFT
        size = 1 << ((n + len(w) - 1).bit_length())
        fa = np.fft.rfft(a, size)

        grad = 2.0 * corr_full[offset - np.arange(n)]

        # step size
        eta = eta0 / math.sqrt(iteration + 1.0)

        a_new = a - eta * grad
        a_new = _simplex_projection(a_new)

        # maybe accept if improvement
        conv_new = _full_convolution(a_new)

# Helper: projection onto the simplex {x ≥ 0, Σx = 1}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries will be zero -> uniform distribution
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: full linear convolution a*a using a power‑of‑two FFT
# ----------------------------------------------------------------------
def _self_convolution_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Return a*a (length 2·len(a)-1) using FFT of given size."""
    Fa = np.fft.rfft(a, size)
    conv_full = np.fft.irfft(Fa * Fa, size)
    conv = conv_full[: 2 * len(a) - 1]
    return conv.real

# Initialisation helpers
# ----------------------------------------------------------------------
def _central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4**i, i=0..n‑1, projected onto the simplex."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return _project_onto_simplex(a)


def _power_law(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-beta), projected onto the simplex."""
    a = np.arange(1, n + 1, dtype=np.float64) ** (-beta)
    return _project_onto_simplex(a)


def _geometric(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**i, projected onto the simplex (0<r<1)."""
    a = r ** np.arange(n, dtype=np.float64)
    return _project_onto_simplex(a)
