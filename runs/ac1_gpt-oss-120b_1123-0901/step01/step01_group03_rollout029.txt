def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time()+budget_s-5
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 2500
    # init a as central binomial
    a = np.empty(n)
    a[0]=1.0
    for i in range(1,n):
        a[i] = a[i-1] * (2*i-1)/(2*i)
    a /= a.sum()  # normalize to sum=1
    
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    print(f'Initial evaluation: {best_score}')
    
    gamma = 1.0
    gamma_max = 50.0
    eta0 = 0.5
    iteration = 0
    while time.time() < deadline:
        # compute convolution a*a
        conv = _full_convolution(a)  # length 2n-1
        max_c = conv.max()
        # compute smooth weights
        # compute softmax over conv scaled by gamma
        w = np.exp(gamma * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum
        # compute gradient via convolution w * a_rev
        a_rev = a[::-1]
        grad = _convolution_weighted(w, a_rev)  # returns length n
        grad = 2.0 * grad
        # step size schedule
        eta = eta0 / (np.sqrt(iteration+1) )
        # exponentiated update
        a = a * np.exp(-eta * grad)
        # renormalize to simplex (sum=1)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0/n)
        else:
            a = a / a_sum
        
        # evaluate
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f'Iter {iteration}: new best {best_score:.6f}')
        
        # update gamma gradually
        if iteration % 100 == 0 and gamma < gamma_max:
            gamma = min(gamma_max, gamma * 1.1)
        # occasional random perturb
        if iteration % 5000 == 0 and iteration>0:
            perturb = np.random.randn(n) * 1e-5
            a = np.maximum(a + perturb, 0)
            a = a / a.sum()
        iteration += 1
    print(f'Finished after {iteration} iters, best_score={best_score:.6f}')
    return best_a.tolist()

```
def _conv_weighted(w, a_rev):
    size = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(fw * fa, size)
    start = len(a_rev) - 1  # n-1
    return conv[start : start + len(a_rev)]

def _weighted_convolution(w, a_rev):
    m = len(w) + len(a_rev) - 1

    fw = np.fft.rfft(w, size)

def _project_onto_simplex(v):
    """

    (All entries are kept non‑negative.)
    """
    v = np.maximum(v, 0.0)

    if s == 0:

        return np.full_like(v, 1.0 / len(v))
    return v / s

    Returns an array of length 2·len(a)‑1.

    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv
