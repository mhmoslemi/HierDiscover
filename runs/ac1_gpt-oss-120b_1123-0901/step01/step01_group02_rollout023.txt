def build_Ab(f, M):
    n = len(f)

    cols = []

from scipy.optimize import linprog

def solve_lp_max_sum(f, M, ub=1000):
    n = len(f)
    # Build sparse matrix A_ub, shape (2n-1, n)
    # For each k row
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        # i_start..i_end inclusive
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    # bounds for each variable
    bounds = [(0, ub) for _ in range(n)]
    # objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        return None
    g = res.x
    return g
```

def solve_lp_max_sum(f, M, ub=1000.0):
    n = f.shape[0]
    rows = []
    cols = []
    data = []

    for k in range(2*n - 1):
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])

    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M, dtype=float)
    c = -np.ones(n, dtype=float)  # maximize sum => minimize -sum
    bounds = [(0, ub) for _ in range(n)]

    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None

"""
Sidon‑set based construction for the step–function heights.

The evaluation function simplifies to:
    E(a) = max_k (a * a)[k]         (since sum(a)=sqrt(2n) ⇒ 2·n·max/(sum²)=max)

Hence we want to keep every convolution entry small.  If all pairwise sums
i+j of the positions with a_i>0 are distinct, each convolution index receives
at most one unordered pair (i,j).  For i≠j the contribution is
2·a_i·a_j, for i=j it is a_i².  With equal weights w on m non‑zero positions
the worst value is 2·w² = 2·(S/m)² where S=√(2n).  Choosing the largest possible
m (a Sidon/B₂ set) makes this quantity ≈ 1.0, far below the target 1.503.

The algorithm:
* Build a large Sidon set of indices inside {0,…,n‑1} by a greedy procedure.
  The set is generated repeatedly with different random orderings and the
  largest one is kept.
* Set a_i = S/m for i in the set, 0 otherwise.  The vector already satisfies
  the sum constraint; a final renormalisation removes rounding error.
* Return the resulting sequence.

No iterative optimisation is required, the construction finishes in a few
milliseconds and yields a provably small upper bound.
"""

import time
import math
import numpy as np

def _max_convolution(a: np.ndarray) -> float:
    """Return the maximum value of the linear convolution a*a."""
    return np.convolve(a, a).max()

def _sidon_greedy(n: int, rng: np.random.Generator, attempts: int = 30) -> np.ndarray:
    """
    Greedy construction of a Sidon (B₂) set: a set of indices such that all
    pairwise sums are distinct.  Several random permutations are tried and the
    largest resulting set is kept.
    """
    best_set = []
    best_len = 0
    max_sum = 2 * n - 1                # number of possible convolution indices

    for _ in range(attempts):
        used = np.zeros(max_sum, dtype=bool)   # used[ i + j ]  = True  ⇔  sum already taken
        cur = []

        # random order improves the chance of reaching a larger set
        order = rng.permutation(n)
        for i in order:
            # quickly reject if i+i already used
            if used[i + i]:
                continue
            # check all sums i + j for already chosen j
            conflict = False
            for j in cur:
                if used[i + j]:
                    conflict = True
                    break
            if conflict:
                continue

            # accept i
            cur.append(i)
            # mark all new sums i + j (including i+i)
            for j in cur:
                used[i + j] = True

        if len(cur) > best_len:
            best_len = len(cur)
            best_set = cur
            # early exit if we already hit the theoretical limit
            # m(m+1)/2 <= 2n-1  ⇒  m ≈ 2*sqrt(n)
            if best_len * (best_len + 1) // 2 >= 2 * n - 1:
                break

    return np.array(sorted(best_set), dtype=int)

    start = time.time()
