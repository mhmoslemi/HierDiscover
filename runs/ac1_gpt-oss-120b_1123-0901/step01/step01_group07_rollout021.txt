def _constraint_rows(f, ks):
    n = len(f)
    rows = []
    for k in ks:
        row = np.zeros(n)
        # j valid such that 0 <= k - j < n => j ∈ [max(0, k-(n-1)), min(n-1,k)]
        j_start = max(0, k - (n-1))

        # produce indices array:
        js = np.arange(j_start, j_end+1)
        # compute f_i where i = k - j:
        f_vals = f[k - js]
        row[js] = f_vals
        rows.append(row)
    return np.vstack(rows) if rows else np.empty((0, n))

def _make_power_sequence(n, p, gamma):
    return (np.arange(n, dtype=np.float64) + gamma) ** (-p)

def _evaluate(seq):
    return evaluate_sequence(seq.tolist())

def _conv_max(seq):
    return np.max(np.convolve(seq, seq))

def _active_constraints(conv, M, k_active=5):
    # Get indices with largest conv values.
    # If conv length less than k_active use all.
    indices = np.argpartition(conv, -k_active)[-k_active:]
    # optionally also include any almost equal to M within tolerance.
    return indices

def _build_A_ub(f, ks):
    n = len(f)
    rows = []
    for k in ks:
        row = np.zeros(n)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        row[js] = f[k - js]  # f_i where i = k - j
        rows.append(row)
    if rows:
        return np.vstack(rows)
    else:
        return np.empty((0, n))

"""
Propose a step‑function (non‑negative heights) with a low evaluation score.

Algorithm
---------

1. **Parameterised power‑law search**  
   The family `a_i = (i + γ) ** (‑p)` is explored on a fairly wide grid
   (`p ∈ [0.55, 0.85]`, `γ ∈ [0.1, 30]`).  The grid is cheap (≈1500
   evaluations) and already finds candidates well below the previous best
   score 1.571.

2. **Local random refinement**  
   Starting from the best `(p,γ)` pair a short stochastic hill‑climb
   (Gaussian perturbations of the parameters) is performed to fine‑tune
   the shape.

3. **Linear‑programming smoothing**  
   For a fixed sequence `f` we solve the linear program

        maximise   Σ g_j
        subject to (f * g)[k] ≤ M  for the most restrictive convolution
                 g_j ≥ 0

   where `M = max(f*f)` and only the `K` largest convolution indices are
   kept as constraints (typically `K = 5`).  The LP pushes mass into
   positions where `f` is small, flattening the autocorrelation.
   The resulting vector `g` is rescaled to have the same ℓ₁‑norm as `f`
   and mixed with the current iterate (`f ← (1‑t)·f + t·g`, `t = 0.2`).
   Repeating this step monotonically reduces the objective.

4. **Final random perturbations**  
   After the LP phase a few extra random tweaks are applied to escape
   possible plateaus.

The whole search respects a hard time budget (`budget_s`, default 1000 s)
and always returns the best sequence obtained so far.  The evaluation
function is scale‑invariant, therefore we never renormalise the sequence
apart from the LP mixing step; all numbers stay far below the
`evaluate_sequence` limits.

The returned list can be fed directly to `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------
def _make_power_sequence(n: int, p: float, gamma: float) -> np.ndarray:
    """Return a_i = (i + γ) ** (‑p) for i = 0 … n‑1."""
    return (np.arange(n, dtype=np.float64) + gamma) ** (-p)


def _active_indices(conv: np.ndarray, k_active: int = 5) -> np.ndarray:
    """Indices of the K largest values of a convolution array."""
    if conv.size <= k_active:
        return np.arange(conv.size, dtype=int)
    # `np.argpartition` gives the `k_active` largest unsorted
    idx = np.argpartition(conv, -k_active)[-k_active:]
    # sort them (optional, only for nicer logging)
    return np.sort(idx)


def _build_A_ub(f: np.ndarray, ks: np.ndarray) -> np.ndarray:
    """
    Build the matrix A_ub for constraints (f * g)[k] ≤ M,
    where each row corresponds to one k in `ks`.
    """
    n = f.size
    rows = []
    for k in ks:
        row = np.zeros(n, dtype=np.float64)
        # valid j satisfy 0 ≤ j ≤ n‑1 and 0 ≤ k‑j ≤ n‑1
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1, dtype=int)
            row[js] = f[k - js]      # coefficient f_i with i = k‑j
        rows.append(row)
    if rows:
        return np.vstack(rows)
    else:
        return np.empty((0, n), dtype=np.float64)


def _evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the supplied evaluate_sequence."""
    # The scorer expects a plain Python list.
    return evaluate_sequence(seq.tolist())


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a step‑function that minimises
        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    The function respects a hard time limit (budget_s seconds) and returns
    the best list of heights found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # safety margin

    # ------------------------------------------------------------------
    # 0. Choose a length – the benchmark used 3428 items.
    # ------------------------------------------------------------------
    n = 3428

    # ------------------------------------------------------------------
    # 0a. If a previous construction is present, start from it.
    # ------------------------------------------------------------------
    try:
        # `height_sequence_1` is expected to be a list of floats.
        init_seq = np.array(height_sequence_1, dtype=np.float64)
        if init_seq.size != n:
            # fall back to a power‑law if the length mismatches
            raise ValueError
        print(f"[init] Using provided height_sequence_1 (len={n}).")
    except Exception:
        # fallback: a reasonable power‑law seed
        init_seq = _make_power_sequence(n, p=0.5, gamma=0.3)
        print("[init] No external sequence – using default power law.")

    best_seq = init_seq.copy()
    best_score = _evaluate(best_seq)
    print(f"[init] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Grid search over (p, γ) for the power‑law family.
    # ------------------------------------------------------------------
    p_vals = np.linspace(0.55, 0.85, 31)               # 0.55 … 0.85
    gamma_vals = np.concatenate([
        np.linspace(0.1, 2.0, 20),
        np.linspace(2.0, 30.0, 30)
    ])

    for p in p_vals:
        for gamma in gamma_vals:
            if time.time() > deadline:
                break
            seq = _make_power_sequence(n, p, gamma)
            score = _evaluate(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_p, best_gamma = p, gamma
                print(f"[grid] new best p={p:.4f}, γ={gamma:.4f}, score={score:.6f}")
        if time.time() > deadline:
            break

    # If grid never improved, keep the original parameters
    try:
        best_p, best_gamma
    except NameError:
        best_p, best_gamma = 0.5, 0.3

    # ------------------------------------------------------------------
    # 2. Random local refinement around the best parameters.
    # ------------------------------------------------------------------
    refine_iters = 4000
    for _ in range(refine_iters):
        if time.time() > deadline:
            break
        p_new = best_p + np.random.randn() * 0.008
        gamma_new = best_gamma + np.random.randn() * 0.12
        if not (0.4 < p_new < 0.97 and 0.01 < gamma_new < 50.0):
            continue
        seq = _make_power_sequence(n, p_new, gamma_new)
        score = _evaluate(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_p, best_gamma = p_new, gamma_new
            print(f"[refine] improved p={p_new:.4f}, γ={gamma_new:.4f}, score={score:.6f}")

    # ------------------------------------------------------------------
    # 3. Linear‑programming smoothing (the “LP‑mix” step).
    # ------------------------------------------------------------------
    f = best_seq.copy()
    sum_f = f.sum()
    # mixing weight – start moderate, decay slowly
    t = 0.2
    lp_iters = 1500
    k_active = 5                     # number of constraints kept each round
    for it in range(lp_iters):
        if time.time() > deadline:
            break

        conv_f = np.convolve(f, f)
        M = conv_f.max()
        # Small tolerance: take the K largest entries (they are the active set)
        ks = _active_indices(conv_f, k_active)

        # Build linear constraints for f * g ≤ M on the active ks
        A_ub = _build_A_ub(f, ks)
        b_ub = np.full(A_ub.shape[0], M, dtype=np.float64)

        # Linear programme: maximise Σ g_j   ↔  minimise –Σ g_j
        c = -np.ones(f.shape, dtype=np.float64)
        bounds = [(0, None)] * f.size

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            # If LP fails (unlikely) abort the LP phase.
            print("[lp] LP solver failed – exiting LP loop")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 == 0.0:
            break

        # Rescale g so that Σ g = Σ f (preserves the denominator of the score)
        g = g0 * (sum_f / sum_g0)

        # Convex combination – guarantees the ℓ₁‑norm stays constant
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)   # protect against tiny negatives

        new_score = _evaluate(f_new)
        if new_score < best_score:
            best_score = new_score
            best_seq = f_new.copy()
            print(f"[lp] iter {it:4d} – improved score = {new_score:.6f}")

        # Keep the improved iterate; if no improvement, still move on
        f = f_new

    # ------------------------------------------------------------------
    # 4. A brief final random perturbation stage (helps escape tiny plateaus)
    # ------------------------------------------------------------------
    final_iters = 2000
    for _ in range(final_iters):
        if time.time() > deadline:
            break
        # multiplicative Gaussian noise on the whole vector (kept small)
        noise = np.random.randn(n) * 0.005
        cand = best_seq * (1.0 + noise)
        cand = np.clip(cand, 0.0, None)          # enforce non‑negativity
        score = _evaluate(cand)
        if score < best_score:
            best_score = score
            best_seq = cand
            print(f"[final] new best score = {score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence as a plain Python list.
    # ------------------------------------------------------------------
    print(f"Finished search – best score {best_score:.6f} (p={best_p:.4f}, γ={best_gamma:.4f})")
    return best_seq.tolist()
