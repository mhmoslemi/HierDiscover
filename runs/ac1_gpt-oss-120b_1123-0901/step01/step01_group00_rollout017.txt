def _local_descent(n, init, target_sum, time_limit, base_step=0.1, max_iter=10000):
    a = init.astype(np.float64)
    a = _project_onto_simplex(a, target_sum)
    best_local_score = evaluate_sequence(a.tolist())  maybe store? Actually may evaluate after modifications.
    best_local_seq = a.copy()
    start = time.time()
    step = base_step
    it = 0
    while time.time() < time_limit and it < max_iter:
        # compute conv, gradient
        conv = np.convolve(a, a)
        k_max = int(np.argmax(conv))
        max_val = conv[k_max]
        # compute gradient
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        idx = np.arange(low, high+1)
        partner = k_max - idx
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[partner]

        # compute direction d
        mean_grad = np.mean(grad)
        d = -grad + mean_grad

        # adapt step size: scaling relative to max conv
        max_grad = np.max(np.abs(grad)) + 1e-12
        eta = step * max_val / max_grad  # similar to earlier but using d maybe scaling

        a_new = a + eta * d

        # project back onto simplex
        a_new = _project_onto_simplex(a_new, target_sum)

        # evaluate
        try:
            score = evaluate_sequence(a_new.tolist())
        except Exception:
            score = float('inf')
        if score < best_local_score:
            best_local_score = score
            best_local_seq = a_new.copy()
            # optionally print improvement
            print(f"[{time.time():.2f}] Local improve: {score:.6f} (n={n}, iter={it})")
        # accept move (always) maybe incorporate acceptance rule else revert?
        # simple accept always (makes search)
        a = a_new

        # optionally reduce step size gradually
        if it % 200 == 0 and it > 0:
            step *= 0.9

        it +=1
    return best_local_seq, best_local_score

def _mutate_sequence(seq, sigma):

    a = np.maximum(a, 0.0)

import cvxpy as cp

def solve_lp(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)  # automatically nonneg
    # constraints list
    constraints = []
    # compute convolution as linear expression
    # For each k, compute sum_i f[i] * g[k - i] where valid.
    # Use loop:
    for k in range(2*n - 1):
        # Determine indices
        low = max(0, k-(n-1))
        high = min(n-1, k)
        # Build expression
        expr = 0
        for i in range(low, high+1):
            expr += f[i] * g[k - i]
        constraints.append(expr <= M)
    # maximize sum g
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.GLPK, warm_start=True, verbose=False) # may need to set time limit?

    # Check status
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        # fallback
        return None
    g_val = g.value
    return g_val

    best_a = a.copy()

        best_score = evaluate_sequence(a.tolist())
    except Exception:
        best_score = float('inf')

    iteration = 0
    while time.time() < time_limit:
        max_val, grad, _, _ = _max_convolution_and_gradient(a)
        # compute direction d = -grad + mean(grad) * ones
        mean_grad = np.mean(grad)
        d = -grad + mean_grad
        # adapt step
        g_norm = np.max(np.abs(d)) if np.max(np.abs(d)) > 0 else 1.0

"""
Propose a sequence of non‑negative heights that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The routine is a hybrid optimiser:

*  A fast projected‑gradient descent on the sub‑gradient of the current
   maximal convolution entry (with a simple momentum‑like correction).
*  An optional LP‑refinement step (using cvxpy if it is available) that
   implements the “LP‑update” from the paper sketch.
*  A tiny evolutionary loop (mutation + short local descent) for global
   exploration.

All helpers are defined at the top level – no closures – and the function
``propose_candidate`` can be called with a time budget (default 1000 s).
It returns a list of non‑negative floats.  Progress is printed to stdout.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Utilities for the convolution‑gradient
# ----------------------------------------------------------------------
def _max_conv_and_grad(a: np.ndarray):
    """
    Compute max(conv(a, a)), its index and the gradient of that entry.
    Returns (max_val, grad, max_idx, full_conv).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_idx = int(np.argmax(conv))
    max_val = conv[max_idx]

    # gradient only touches the overlapping region
    low = max(0, max_idx - (n - 1))
    high = min(n - 1, max_idx)
    idx = np.arange(low, high + 1, dtype=int)
    partner = max_idx - idx
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[partner]
    return max_val, grad, max_idx, conv


# ----------------------------------------------------------------------
# Very small mutation helper (adds Gaussian noise and re‑projects)
# ----------------------------------------------------------------------
def _mutate(a: np.ndarray, target_sum: float, sigma: float) -> np.ndarray:
    """Return a mutated copy of a (Gaussian noise, then simplex projection)."""
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    b = _project_onto_simplex(b, target_sum)
    return b


# ----------------------------------------------------------------------
# Local projected‑gradient descent (sub‑gradient of the maximal entry)
# ----------------------------------------------------------------------
def _local_descent(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    base_step: float = 0.05,
    print_tag: str = "",
) -> tuple[np.ndarray, float]:
    """
    Run a short gradient descent starting from ``init``.
    Returns the best sequence seen and its evaluation score.
    """
    a = _project_onto_simplex(init.astype(float), target_sum)

    # evaluate the starting point
    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:  # safety
        best_score = math.inf
    best_seq = a.copy()

    step = base_step
    it = 0
    t0 = time.time()

    while time.time() < t0 + time_limit:
        max_val, grad, _, _ = _max_conv_and_grad(a)

        # direction that keeps the sum constant (mean‑zero adjustment)
        mean_g = np.mean(grad)
        d = -grad + mean_g  # sum(d) == 0 by construction

        # step size adapted to the magnitude of the gradient
        max_abs = np.max(np.abs(d)) if np.max(np.abs(d)) > 0 else 1.0
        eta = step * max_val / max_abs

        a_new = a + eta * d
        a_new = _project_onto_simplex(a_new, target_sum)

        # evaluate
        try:
            score = float(evaluate_sequence(a_new.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(
                f"[{time.time():.2f}] {print_tag} new local best = {score:.6f} (n={len(a)})"
            )

        # always move forward; occasional random shake‑up
        if np.random.rand() < 0.02:
            noise_sigma = 0.01 * target_sum / len(a)
            a = _mutate(a_new, target_sum, noise_sigma)
            step *= 0.9
        else:
            a = a_new

        # simple decay of the step size
        if it % 200 == 0 and it > 0:
            step *= 0.9
        it += 1

    return best_seq, best_score

def _solve_lp_for_g(f: np.ndarray, M: float):
    """Solve the LP: maximise Σ g_j subject to g ≥ 0 and f*g ≤ M."""
    if not _CVX_AVAILABLE:
        return None
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)

    constraints = []
    # Build each convolution inequality (linear in g)
    for k in range(2 * n - 1):
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        expr = cp.sum(cp.multiply(f[low : high + 1], g[partner]))
        constraints.append(expr <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.GLPK, warm_start=True, verbose=False, glpkk=1e-08)
    except Exception:  # fallback to ECOS if GLPK fails
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=2000)
        except Exception:
            return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    g_val = np.asarray(g.value).flatten()
    if g_val is None:
        return None
    # Numerical cleaning
    g_val = np.maximum(g_val, 0.0)
    return g_val


def _lp_refine(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    mix: float = 0.2,
    max_iter: int = 6,
) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑updates starting from ``init``.
    Returns the best sequence seen and its score.
    """
    a = _project_onto_simplex(init.astype(float), target_sum)
    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    t_start = time.time()
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        # current bound M
        conv = np.convolve(a, a)
        M = float(conv.max())

        # solve LP for a supporting direction g₀
        g0 = _solve_lp_for_g(a, M)
        if g0 is None or np.sum(g0) == 0:
            break

        # scale g₀ to the required simplex
        scale = target_sum / np.sum(g0)
        g = g0 * scale

        # convex combination a ← (1‑t) a + t g
        a = (1.0 - mix) * a + mix * g
        a = _project_onto_simplex(a, target_sum)

        # evaluate
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time():.2f}] LP‑refine iter {it+1} new best = {score:.6f} (n={len(a)})"
            )
    return best_seq, best_score

# Simple resize (interpolation / down‑sampling) to reuse a known good sequence
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize seq to length new_n preserving non‑negativity (linear interpolation)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        # pick equally spaced indices
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:
        # linear interpolation
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)
