def subgradient(a, b, M, tolerance):
    # a: vector len n
    # b: convolution a*a (length 2n-1)
    # M: max(b)
    active = np.where(b >= M - tolerance)[0]
    n = len(a)
    rev_sum = np.zeros_like(a)
    idx = np.arange(n)
    for k in active:
        j = k - idx
        mask = (j >=0) & (j < n)
        rev_sum[mask] += a[j[mask]]
    # gradient of M wrt a = 2 * rev_sum (for each a_i)
    # Then gradient of E:
    S = a.sum()
    gradM = 2 * rev_sum
    gradE = (2.0 * n / (S * S)) * gradM - (4.0 * n * M / (S ** 3))
    return gradE

    v = np.zeros_like(a)

    while time.time() < deadline:

        b = np.convolve(a, a)

            j = k - idx

            rev_sum[mask] += a[j[mask]]
        # gradient of M (subgradient)
        gradM = 2.0 * rev_sum
        # gradient of E

"""
Hybrid optimizer for the step‑function attack.

The algorithm mixes three ideas:

1. **Sub‑gradient descent** – the objective is
       E(a) = 2·n·max(a∗a)/(∑a)² .
   Keeping the total sum fixed (S = √(2·n)) makes the denominator constant.
   The gradient of `max(a∗a)` is non‑smooth; instead of using only the
   currently maximal index we collect *all* indices whose convolution value
   is within a tiny tolerance of the maximum and average their contributions.
   An Adam‑style update (first‑ and second‑moment adaptation) gives a
   well‑behaved step size.

2. **Random jitter** – every few thousand iterations a tiny Gaussian
   perturbation is added and the vector is re‑normalised.  This helps the
   search escape shallow plateaus.

3. **Occasional linear‑programming refinement** – after a fixed number of
   gradient steps we solve a small LP:

        maximise   Σ g_j
        subject to g_j ≥ 0,
                    (a ∗ g)_k ≤ M   (∀k),

   where `M = max(a∗a)`.  The LP finds a direction that can increase the
   total mass without worsening the current supremum.  The new candidate is
   obtained by a convex combination `a←(1‑t)·a + t·g` (t≈0.3).  The LP is
   solved with SciPy’s HIGH‑S solver; it is cheap for n≈2000 and yields a
   noticeable improvement.

All updates keep the entries non‑negative, cap them at the limits imposed
by `evaluate_sequence` (0 ≤ x ≤ 1000) and re‑scale to the fixed sum S.
The best sequence seen (according to the true `evaluate_sequence`) is
returned before the time budget expires.
"""

    # length needed for full convolution
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    b = np.fft.irfft(fa * fa, n=size)
    return b[: 2 * n - 1]


# ----------------------------------------------------------------------
# Helper: build the linear program matrix for the constraint (a ∗ g) ≤ M
def build_lp_matrix(a: np.ndarray, M: float):
    """
    Returns (A_ub, b_ub, bounds) suitable for scipy.optimize.linprog.
    The matrix encodes the linear constraints (a * g)[k] ≤ M for all k.
    """
    n = a.shape[0]
    # For offset d (= row–col) the diagonal entries are all a[d].
    # Using scipy.sparse.diags we can build the Toeplitz‑like matrix.
    diags = [np.full(n, a[d]) for d in range(n)]          # each diagonal length = n
    offsets = np.arange(n)                                # 0,1,…,n‑1
    A_ub = sp.diags(diags, offsets, shape=(2 * n - 1, n), format='csc')
    b_ub = np.full(2 * n - 1, M, dtype=float)
    # Upper bound 1000 follows the limits of evaluate_sequence.
    bounds = [(0.0, 1000.0) for _ in range(n)]
    return A_ub, b_ub, bounds


# ----------------------------------------------------------------------
# Helper: LP refinement – returns a new candidate or None on failure
def lp_refine(a: np.ndarray, S_target: float, t_mix: float = 0.3):
    """
    Solve the LP described in the docstring and return the mixed vector
    (1‑t)·a + t·g if the LP succeeds, otherwise None.
    """
    # current supremum of the auto‑convolution
    b_full = full_convolution_fft(a)
    M = b_full.max()

    # Build the LP
    A_ub, b_ub, bounds = build_lp_matrix(a, M)

    # Maximise Σ g_j  ==>  minimise -Σ g_j
    c = -np.ones_like(a)

    # Solve with HiGHS (fast for the sizes we use)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success or res.x is None:
        return None

    g = res.x
    total = g.sum()
    if total == 0.0:
        return None

    # re‑scale to the fixed sum S_target
    g = g * (S_target / total)
    # convex combination
    a_new = (1.0 - t_mix) * a + t_mix * g
    # re‑normalise (numerical drift)
    a_new = np.maximum(a_new, 0.0)
    a_new = a_new * (S_target / a_new.sum())
    np.clip(a_new, 0.0, 1000.0, out=a_new)
    return a_new


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises `evaluate_sequence`.
    The function runs for at most `budget_s` seconds and returns the best
    sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    safety = 5.0                     # leave a small margin for clean‑up
    deadline = start_time + budget_s - safety

    # ------------------------------------------------------------------
    # Initial sequence (use SOTA if available, otherwise uniform)
    try:
        base = np.array(height_sequence_1, dtype=float)
    except Exception:
        base = np.ones(2100, dtype=float)      # a reasonable fallback

    n = base.shape[0]
    S_target = math.sqrt(2.0 * n)               # fixed sum (normalisation)

    a = base * (S_target / base.sum())
    a = np.clip(a, 0.0, 1000.0)

    # Best solution so far
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Adam‑style optimiser state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.6                                   # initial learning‑rate

    # Current objective (approximation using the constant factor)
    const_factor = 2.0 * n / (S_target * S_target)   # 2·n / S²
    # Compute the first convolution and its maximum
    b_full = full_convolution_fft(a)
    M_cur = b_full.max()
    score_cur = const_factor * M_cur

    iteration = 0
    last_improve_iter = 0

    # Frequency of extra actions
    jitter_interval = 3000
    lp_interval = 12000
    exact_eval_interval = 5000

    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 1) Gradient (sub‑gradient) step
        # --------------------------------------------------------------
        # Full convolution (FFT based – fast enough for our n)
        b_full = full_convolution_fft(a)
        M = b_full.max()

        # indices that are (almost) maximal – tolerance chosen to capture
        # the whole plateau but stay small enough for a meaningful direction
        tol = max(1e-12, M * 1e-8)
        active = np.where(b_full >= M - tol)[0]

        # Compute the averaged reversed vector (∑_{k∈active} a_{k‑i})
        rev_sum = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            rev_sum[mask] += a[j[mask]]

        # Sub‑gradient of the max‑convolution term
        gradM = 2.0 * rev_sum                     # dM/da
        # Gradient of the whole objective (with fixed sum)
        grad = const_factor * (gradM - 2.0 * M / S_target)

        # Adam update
        lr = lr0 / np.sqrt(iteration)               # simple decay
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)              # projection onto ℝ₊ⁿ
        a_new_sum = a_new.sum()
        if a_new_sum > 0.0:
            a_new = a_new * (S_target / a_new_sum)  # keep the total sum fixed
        else:
            a_new = a.copy()

        # evaluate the new candidate (approximation, cheap)
        b_new = full_convolution_fft(a_new)
        M_new = b_new.max()
        score_new = const_factor * M_new

        # --------------------------------------------------------------
        # 2) Acceptance test (allow occasional uphill moves)
        # --------------------------------------------------------------
        # Simulated‑annealing style temperature – linearly decreasing
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s - safety)
        temperature = max(0.0, (1.0 - frac)) * const_factor * M

        accept = False
        if score_new <= score_cur:
            accept = True
        else:
            # Metropolis criterion
            if temperature > 0.0 and np.random.rand() < np.exp(-(score_new - score_cur) / temperature):
                accept = True

        if accept:
            a = a_new
            M_cur = M_new
            score_cur = score_new
            if iteration - last_improve_iter > 0 and score_cur < best_score:
                # exact evaluation to avoid drift caused by the approximation
                exact = evaluate_sequence(a.tolist())
                if exact < best_score:
                    best_score = exact
                    best_seq = a.copy()
                    last_improve_iter = iteration
                    print(f"[iter {iteration:07d}] new best exact score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 3) Periodic random jitter (helps escaping plateaus)
        # --------------------------------------------------------------
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * S_target / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (S_target / a.sum())

        # --------------------------------------------------------------
        # 4) Periodic exact evaluation (cheap bookkeeping)
        # --------------------------------------------------------------
        if iteration % exact_eval_interval == 0:
            exact = evaluate_sequence(a.tolist())
            if exact < best_score:
                best_score = exact
                best_seq = a.copy()
                last_improve_iter = iteration
                print(f"[iter {iteration:07d}] exact eval improvement → {best_score:.6f}")

        # --------------------------------------------------------------
        # 5) Occasional LP refinement (only if we still have time)
        # --------------------------------------------------------------
        if iteration % lp_interval == 0 and (deadline - time.time()) > 30.0:
            a_lp = lp_refine(a, S_target, t_mix=0.3)
            if a_lp is not None:
                # accept the LP step only if it does not increase the objective
                M_lp = full_convolution_fft(a_lp).max()
                score_lp = const_factor * M_lp
                if score_lp <= score_cur + 1e-12:
                    a = a_lp
                    M_cur = M_lp
                    score_cur = score_lp
                    # exact evaluation of the LP‑step result
                    exact = evaluate_sequence(a.tolist())
                    if exact < best_score:
                        best_score = exact
                        best_seq = a.copy()
                        last_improve_iter = iteration
                        print(f"[iter {iteration:07d}] LP refinement gave score {best_score:.6f}")

    # ------------------------------------------------------------------
    # End of search – return the best sequence found
    # ------------------------------------------------------------------
    print(f"[done] iterations={iteration}, best_score={best_score:.6f}")
    return best_seq.tolist()
