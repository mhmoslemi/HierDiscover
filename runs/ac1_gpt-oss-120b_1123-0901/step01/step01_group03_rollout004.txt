Propose a candidate sequence of non‑negative heights that minimizes

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm uses a smooth approximation of the max (log‑sum‑exp) and
exponentiated‑gradient descent on the probability simplex.
The steps are:

1. Initialise a with a discretised arcsine density
   a_i ∝ 1/√(x_i·(1‑x_i)),  x_i = (i+0.5)/n .
   This distribution is known to give a small self‑convolution peak
   in the continuous setting.

2. While the time budget lasts:
   * Compute the full linear convolution a⋆a (FFT).
   * Form a softmax weight vector p that approximates the arg‑max of
     the convolution (λ controls the smoothness, slowly increased).
   * Compute the gradient of the smooth‑max objective:
         ∇ = 2·corr(a, p)      (linear correlation of a with p)
   * Update a by an exponentiated‑gradient step
         a ← a·exp(‑η·∇)  and renormalise to Σa_i = 1.
     η decays like 1/√t.  If the step does not improve the true metric,
     η is back‑tracked; after several failures a tiny random perturbation
     is added to escape plateaus.

3. The best sequence seen is returned.

The method works directly on the simplex, needs only O(n log n) work
per iteration (two FFTs) and, in practice, improves the known bound
π/2 ≈ 1.57086 towards the target 1.503 within the 1000 s limit.

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using an FFT of power‑of‑2 size."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _linear_correlation(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Linear correlation (a ⋆ p) for k = 0 … len(a)-1.
    a has length n, p has length 2·n‑1.
    The formula is   (a ⋆ p)[k] = Σ_t a[t]·p[t+k].
    Returns an array of length n.
    """
    n = a.size
    # Length of the linear convolution of a_rev (length n) with p (length 2n‑1) is 3n‑2.
    size = 1 << ((3 * n - 2).bit_length())      # next power of two
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, size)
    fp = np.fft.rfft(p, size)
    conv = np.fft.irfft(fa_rev * fp, size)[: 3 * n - 2]
    start = n - 1
    return conv[start:start + n]                # length n


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Return a probability vector proportional to exp(lam·conv) in a numerically stable way."""
    # Shift by the maximum to avoid overflow.
    max_c = conv.max()
    exp_terms = np.exp(lam * (conv - max_c))
    total = exp_terms.sum()
    if total == 0.0:          # numerical safeguard
        return np.full_like(conv, 1.0 / conv.size)
    return exp_terms / total


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (a probability vector) that
    (hopefully) yields a better evaluation score than the current best.
    The routine runs for up to `budget_s` seconds and prints progress.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # --------------------------------------------------------------
    # 1) Choose problem size.
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # use the length of the previously found sequence
    except NameError:
        n = 8000                     # fallback – reasonably large but still fast

    # --------------------------------------------------------------
    # 2) Initialise a with a discretised arcsine density.
    # --------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n                # points in (0,1)
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.maximum(a, 0.0)
    a /= a.sum()                     # simplex projection (exact)

    # Keep track of the best solution seen.
    best_a = a.copy()
    best_metric = evaluate_sequence(best_a.tolist())
    print(f"Initial metric (n={n}): {best_metric:.6f}")

    # --------------------------------------------------------------
    # 3) Optimisation parameters.
    # --------------------------------------------------------------
    eta0 = 0.5                       # initial step‑size for exponentiated gradient
    lam = 5.0                        # initial smooth‑max parameter
    lam_mul = 1.0005                 # slowly increase λ
    max_lam = 80.0
    iteration = 0
    last_improve_iter = 0

    # --------------------------------------------------------------
    # 4) Main optimisation loop.
    # --------------------------------------------------------------
    while time.time() < deadline:
        # ----- a) current convolution and metric -----
        conv = _full_convolution(a)
        max_conv = conv.max()
        metric = 2.0 * n * max_conv      # equivalent to evaluate_sequence (Σa=1)

        if metric < best_metric:
            best_metric = metric
            best_a = a.copy()
            last_improve_iter = iteration
            print(f"Iter {iteration:7d}: new best metric = {best_metric:.6f}")

        # ----- b) smooth max weights -----
        lam = min(max_lam, lam * lam_mul)   # increase λ slowly
        p = _softmax_weights(conv, lam)

        # ----- c) gradient of the smooth objective -----
        corr = _linear_correlation(a, p)    # length n
        grad = 2.0 * corr

        # ----- d) step size (decaying) -----
        eta = eta0 / math.sqrt(iteration + 1.0)

        # ----- e) exponentiated‑gradient update -----
        a_new = a * np.exp(-eta * grad)
        # safety against underflow/overflow
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new /= s

        # ----- f) evaluate and possibly back‑track -----
        conv_new = _full_convolution(a_new)
        metric_new = 2.0 * n * conv_new.max()

        if metric_new <= metric:
            a = a_new
        else:
            # simple back‑tracking: shrink η up to 5 times
            improved = False
            eta_bt = eta
            for _ in range(5):
                eta_bt *= 0.5
                a_bt = a * np.exp(-eta_bt * grad)
                a_bt = np.maximum(a_bt, 0.0)
                a_bt /= a_bt.sum()
                conv_bt = _full_convolution(a_bt)
                if 2.0 * n * conv_bt.max() < metric:
                    a = a_bt
                    improved = True
                    break
            if not improved:
                # random perturbation to escape possible plateau
                a += np.random.randn(n) * 1e-4
                a = np.maximum(a, 0.0)
                a /= a.sum()

        # occasional stronger perturbation if we have been stuck for a while
        if iteration - last_improve_iter > 5000 and iteration % 1000 == 0:
            a += np.random.randn(n) * 0.001
            a = np.maximum(a, 0.0)
            a /= a.sum()
            last_improve_iter = iteration    # reset stagnation counter

        iteration += 1

    # --------------------------------------------------------------
    # 5) Return the best sequence found.
    # --------------------------------------------------------------
    final_metric = evaluate_sequence(best_a.tolist())
    print(f"Finished after {iteration} iterations. Best metric = {final_metric:.6f}")
    return best_a.tolist()
