    Otherwise blend a fraction of g into the current best (convex combination)

5.  Finish with a cheap stochastic local search (multiplicative perturbation

            print(f"[blend] improved to {best_score:.6f}")

        # Small random jitter to escape tiny plateaus.
        jitter = np.exp(rng.normal(0.0, 0.005, size=n))
        best_seq = np.clip(best_seq * jitter, 0.0, max_val)

    # --------------------------------------------------------------
    # 4. Fine‑grained random local search (single‑entry perturbations).
    # --------------------------------------------------------------
    while time.time() < deadline:
        i = rng.integers(n)
        old_val = best_seq[i]
        factor = np.exp(rng.normal(0.0, 0.02))
        new_val = max(0.0, old_val * factor)
        cand = best_seq.copy()
        cand[i] = new_val
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_seq = cand
            best_score = cand_score
            print(f"[improve] new best {best_score:.6f} (index {i})")

    print(f"[final] best score {best_score:.6f}")
    return best_seq.tolist()
