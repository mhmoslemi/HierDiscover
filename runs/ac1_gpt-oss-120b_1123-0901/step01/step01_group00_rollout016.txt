import scipy.optimize as opt

def lp_step(f, M):
    n = len(f)
    # build A
    A = np.zeros((2*n-1, n))
    for i in range(n):
        A[i:i+n, i] = f
    # objective: minimize -sum(b) => c = -np.ones(n)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    b_ub = np.full(2*n-1, M)
    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    b = res.x
    return b

"""
Propose a step‑function (non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The implementation follows the linear‑programming based scheme described in the
statement:

* Initialise a random vector on the simplex Σ a_i = 1.
* Repeatedly:
    – Compute M = max(conv(a, a)).
    – Solve the LP  maximise Σ b_i  subject to  b_i ≥ 0 and conv(a, b) ≤ M.
    – Normalise the LP solution b to Σ b_i = 1.
    – Either replace a by b or move a towards b (convex combination) and
      re‑project onto the simplex.
* Keep the best sequence ever seen (according to the exact ``evaluate_sequence``
  function).  The algorithm runs for at most ``budget_s`` seconds.

The LP uses ``scipy.optimize.linprog`` with the HiGHS interior‑point solver,
which is fast for the moderate sizes (n≈200‑1500) used here.
All heavy work (convolutions, projections) is done with NumPy/FFT, so the
search can perform many iterations within the time limit.
"""

import time
import numpy as np

# ---------------------------------------------------------------------------
# optional SciPy import – if unavailable we fall back to the previous method
# ---------------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    _HAS_SCIPY = True
except Exception:  # pragma: no cover
    _HAS_SCIPY = False

# ---------------------------------------------------------------------------
# 1.  Projection onto the probability simplex {x_i ≥ 0, Σ x_i = s}
# ---------------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex Σ x_i = s, x_i ≥ 0."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ---------------------------------------------------------------------------
# 2.  Fast evaluation – identical to ``evaluate_sequence`` (scale‑invariant)
# ---------------------------------------------------------------------------
def _evaluate_fast(seq: np.ndarray) -> float:
    """Return the exact score of ``seq`` (non‑negative, no clipping needed)."""
    seq = np.clip(seq, 0.0, 1000.0)
    total = float(seq.sum())
    if total < 0.01:
        return float("inf")
    n = seq.shape[0]
    m = 2 * n - 1
    # FFT size = next power of two (fast and exact for real data)
    size = 1 << (m - 1).bit_length()
    fft = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft * fft, n=size)[:m]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (total * total))


# ---------------------------------------------------------------------------
# 3.  Build the (2n‑1) × n convolution matrix C_f  where (C_f @ b)[k] = Σ_i f_i b_{k-i}
# ---------------------------------------------------------------------------
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Dense (2n‑1)×n matrix for the linear map b ↦ conv(f, b)."""
    n = f.shape[0]
    rows = 2 * n - 1
    C = np.zeros((rows, n), dtype=np.float64)
    # column j contains a shifted copy of f
    for j in range(n):
        C[j:j + n, j] = f
    return C


# ---------------------------------------------------------------------------
# 4.  One LP step – maximise Σ b_i subject to conv(f, b) ≤ M and b ≥ 0
# ---------------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float, time_limit: float = None) -> np.ndarray | None:
    """Return the LP solution (or None on failure)."""
    if not _HAS_SCIPY:
        return None
    n = f.shape[0]
    # Build constraint matrix
    A = _conv_matrix(f)
    # Objective: minimise –Σ b_i  (i.e. maximise Σ b_i)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n
    options = {}
    if time_limit is not None:
        # HiGHS respects a ``time_limit`` option (seconds)
        options["time_limit"] = max(0.0, time_limit)
    try:
        res = _opt.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options=options,
        )
    except Exception:  # pragma: no cover
        return None
    if res.success:
        return res.x
    return None
