    # Start with best known sequence if available else random
    try:
        init = np.array(height_sequence_1)

    # Use variable n: can try to keep best length found from height_sequence_1.

    # Use Adam optimizer
    a = init.copy()
    # project onto simplex to match sum
    a = project_simplex(a, target_sum) # ensures sum

    m = np.zeros_like(a) # first moment
    v = np.zeros_like(a) # second moment

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr_initial = 0.2 # adjust

    iteration = 0
    best_seq = a.copy()
    best_score = evaluate_sequence(a.tolist())
    start_time = time.time()
    deadline = start_time + budget - 5

    while time.time() < deadline:
        # compute gradient from earlier expression (depends on max index)
        M, k_max = _conv_max(a) # returns max and index
        # compute rev vector (a_{k_max - p}) only for valid indices
        rev = np.zeros_like(a)
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        rev[mask] = a[j[mask]]
        S = a.sum()
        grad = (4.0 * n / (S**2)) * rev - (4.0 * n * M / (S**3))

        # Adam update
        iteration += 1
        lr = lr_initial * np.sqrt(1 - beta2**iteration) / (1 - beta1**iteration)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        step = lr * m / (np.sqrt(v) + eps)
        a_new = a - step

        # project onto simplex with sum target_sum
        a_new = project_simplex(a_new, target_sum)

        # clip to [0,1000]
        np.clip(a_new, 0.0, 1000.0, out=a_new)

        # evaluate
        cur_score = evaluate_sequence(a_new.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            print(f"Iter {iteration}: new best {best_score}")

        # acceptance criteria: keep a_new if improved or small probability else revert
        if cur_score <= best_score + 1e-12:
            a = a_new
        else:
            if np.random.rand() < 0.01: # occasional acceptance.
                a = a_new

        # occasional random perturbation (jitter) to escape
        if iteration % 5000 == 0:
            jitter = np.random.randn(n) * (target_sum * 0.001 / n)
            a = np.maximum(a + jitter, 0.0)
            a = project_simplex(a, target_sum)
    return best_seq.tolist()

def simplex_projection(v, z=1.0):
    # ensure nonnegative after projection onto simplex {x>=0, sum x = z}
    # from https://github.com/erikvullum/fast_proj_simplex
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero? set uniform distribution
        theta = cssv[-1] / n
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def project_onto_simplex(v, z):
    """Return the Euclidean projection of v onto the simplex {x: x>=0, sum x = z}."""
    if v.sum() == z and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def conv_fft_max(a):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    M = conv.max()
    idxs = np.flatnonzero(conv >= M - 1e-12)
    return M, idxs, conv

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    # Determine sequence length
    try:
        base_seq = np.array(height_sequence_1, dtype=float)
        n = base_seq.shape[0]
    except Exception:
        n = 2100
        base_seq = None

    target_sum = math.sqrt(2.0 * n)

    # initial vector
    if base_seq is not None:
        # ensure non-negative, clip
        init = np.maximum(base_seq, 0.0)
        init = init.clip(0.0, 1000.0)
        init = project_onto_simplex(init, target_sum)
    else:
        init = np.full(n, target_sum / n)
    a = init.copy()
    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_seq = a.copy()
    best_val = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, best={best_val}")

    # Hyperparams
    lr0 = 0.5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    iter_count = 0

    # Phase 1: Adam based gradient descent (≈70% time)
    phase1_deadline = start + 0.7 * budget_s - 5.0
    while time.time() < phase1_deadline:
        iter_count += 1
        # compute convolution max and its indices
        M, max_idxs, conv = conv_fft_max(a)

        # compute rev (average over max indices)
        n = a.shape[0]
        idx = np.arange(n)
        rev = np.zeros_like(a)
        for k in max_idxs:
            j = k - idx
            mask = (j >= 0) & (j < n)
            rev[mask] += a[j[mask]]
        rev = rev / len(max_idxs)

        S = a.sum()
        # gradient derived for objective = max_b (since sum fixed)
        grad = (4.0 * n / (S * S)) * rev - (4.0 * n * M / (S ** 3))

        # Adam bias-corrected step
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        lr = lr0 * (np.sqrt(1 - beta2 ** iter_count) / (1 - beta1 ** iter_count))
        step = lr * m / (np.sqrt(v) + eps)

        a_new = a - step

            a = project_onto_simplex(a, target_sum)

    # Phase 2: Simulated annealing fine tuning
    cur_a = best_seq.copy()
    cur_M, _, _ = conv_fft_max(cur_a)
    best_seq2 = cur_a.copy()
    best_val2 = best_val

    # SA parameters
    T0 = max(0.1, best_val2 * 0.2)  # initial temperature
    Tmin = 1e-5
    total_sa_iters = 0
    # Determine when to stop: when time up

        T = max(Tmin, T0 * (1 - progress))

        i = np.random.randint(n)

            j += 1
        # delta = fraction of a[i]
        delta = cur_a[i] * np.random.rand() * 0.1  # up to 10%
        a_new = cur_a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # ensure non-negative (should be)
        # compute new max
        M_new, _, _ = conv_fft_max(a_new)
        # acceptance

        if delta_obj < 0 or np.random.rand() < np.exp(-delta_obj / (T + 1e-12)):

            cur_M = M_new
            # update best if better

"""
Advanced optimizer for the step‑function evaluation problem.

The algorithm consists of two stages:

1. **Adam‑based projected gradient descent**  
   – The objective  E(a)=2·n·max(a*a)/(∑a)² is homogeneous, therefore we
     keep the ℓ₁‑norm fixed to √(2n).  
   – The gradient of  max(a*a)  with respect to a is derived analytically
     (see the comment in the code).  
   – After a gradient step we **project onto the non‑negative simplex**
     {x ≥ 0,  ∑x = √(2n)} so the constraint is satisfied exactly.
   – The convolution and its maximum are obtained via an FFT, which makes
     each iteration O(n log n) and allows many thousands of updates.
   – Adam’s adaptive moments give a much more robust descent than the
     simple decaying step‑size used previously.

2. **Simulated‑annealing fine‑tuning**  
   – When the Adam phase stagnates we switch to a cheap stochastic local‑search:
     a random amount of mass is moved from one index to another (preserving the
     total sum).  
   – The new maximum convolution is computed with the same FFT routine.
   – A Metropolis criterion with a slowly decreasing temperature accepts
     occasional uphill moves, helping to escape shallow plateaus.
   – The best sequence found in either phase is returned.

Both phases run within the supplied time budget (≈1000 s).  The code uses only
NumPy and the provided `evaluate_sequence` function; no file or network I/O is
performed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x : x >= 0, sum(x) = z }.

    Implements the algorithm of Duchi et al. (2008).  Returns a new array.
    """
    if z < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # Fast path: already feasible
    if v.min() >= 0 and np.isclose(v.sum(), z):
        return v.copy()
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho_array = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_array > 0)[0]
    if rho.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_fft(a: np.ndarray, fft_size: int):
    """
    Compute the discrete convolution a*a via FFT, return
        M      – maximum value of the convolution,
        idxs   – all indices where the maximum is attained,
        conv   – the full convolution (length 2*len(a)‑1).
    """
    # FFT of a, zero‑padded to fft_size
    fa = np.fft.rfft(a, n=fft_size)
    # element‑wise product (a convolved with itself)
    conv_full = np.fft.irfft(fa * fa, n=fft_size)[: 2 * a.shape[0] - 1]
    M = conv_full.max()
    # Tolerance to collect all (near‑)maximal indices
    tol = 1e-12
    idxs = np.flatnonzero(conv_full >= M - tol)
    return M, idxs, conv_full


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  Returns the best sequence found before the time
    budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # --------------------------------------------------------------
    # Determine the problem size (use the known construction if present)
    try:
        base_seq = np.array(height_sequence_1, dtype=float)
        n = base_seq.shape[0]
    except Exception:
        n = 2100
        base_seq = None

    target_sum = math.sqrt(2.0 * n)                # normalisation used in the paper
    fft_size = 1 << ((2 * n - 1).bit_length())    # power‑of‑two size for the FFT

    # --------------------------------------------------------------
    # Initialise the current vector
    if base_seq is not None:
        a = np.clip(base_seq, 0.0, 1000.0)
        a = _project_onto_simplex(a, target_sum)
    else:
        a = np.full(n, target_sum / n, dtype=float)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Track the best solution overall
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam projected gradient descent (≈70 % of the budget)
    phase1_deadline = start_time + 0.70 * budget_s - 5.0
    lr0 = 0.5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    iteration = 0
    idx_array = np.arange(n)                       # reused index vector

    while time.time() < phase1_deadline:
        iteration += 1

        # -- convolution and maximal shift(s) ---------------------------------
        M, max_idxs, _ = _max_convolution_fft(a, fft_size)

        # -- build the (averaged) reversed‑vector needed for the gradient ------
        rev = np.zeros_like(a)
        for k in max_idxs:
            j = k - idx_array
            mask = (j >= 0) & (j < n)
            rev[mask] += a[j[mask]]
        rev /= len(max_idxs)

        S = a.sum()
        # Gradient of the objective (derived analytically)
        grad = (4.0 * n / (S * S)) * rev - (4.0 * n * M / (S ** 3))

        # -- Adam update ------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** iteration) /
                    (1.0 - beta1 ** iteration))
        step = lr * m / (np.sqrt(v) + eps)

        a_new = a - step
        a_new = _project_onto_simplex(a_new, target_sum)
        np.clip(a_new, 0.0, 1000.0, out=a_new)

        # -- evaluate ---------------------------------------------------------
        cur_score = evaluate_sequence(a_new.tolist())

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            print(f"[Adam iter {iteration:06d}] new best = {best_score:.6f}")

        # Accept strictly improving steps; occasionally accept deteriorations
        if cur_score <= best_score + 1e-12:
            a = a_new
        else:
            if np.random.rand() < 0.01:
                a = a_new

        # Tiny random jitter every few thousand steps to shake out of plateaus
        if iteration % 5000 == 0:
            jitter = np.random.randn(n) * (target_sum * 0.001 / n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_onto_simplex(a, target_sum)

    # --------------------------------------------------------------
    # Phase 2 – Simulated annealing for fine‑tuning (remaining budget)
    cur_a = best_seq.copy()
    cur_M, _, _ = _max_convolution_fft(cur_a, fft_size)
    best_seq_sa = cur_a.copy()
    best_score_sa = best_score

    # Simulated‑annealing schedule
    T0 = max(0.1, best_score_sa * 0.2)   # initial temperature
    Tmin = 1e-5
    sa_iters = 0

    while time.time() < deadline:
        sa_iters += 1
        # Linear temperature decay based on elapsed fraction of SA phase
        fraction = (time.time() - phase1_deadline) / (deadline - phase1_deadline)
        T = max(Tmin, T0 * (1.0 - fraction))

        # ---- propose a simple mass‑transfer move ----------------------------
        i = np.random.randint(n)
        if cur_a[i] == 0.0:
            continue
        # choose a distinct destination index
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction (up to 10 %) of the source mass
        delta = cur_a[i] * np.random.rand() * 0.10
        a_candidate = cur_a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        # ---- evaluate the candidate -----------------------------------------
        M_new, _, _ = _max_convolution_fft(a_candidate, fft_size)
        delta_obj = M_new - cur_M

        # Metropolis acceptance
        if delta_obj < 0 or np.random.rand() < math.exp(-delta_obj / (T + 1e-12)):
            cur_a = a_candidate
            cur_M = M_new
            # Keep the global best
            if M_new < best_score_sa:
                best_score_sa = M_new
                best_seq_sa = a_candidate.copy()
                print(f"[SA iter {sa_iters:06d}] new best = {best_score_sa:.6f}")

    # --------------------------------------------------------------
    # Choose the best of the two phases
    if best_score_sa < best_score:
        final_seq = best_seq_sa
        final_score = best_score_sa
    else:
        final_seq = best_seq
        final_score = best_score

    print(f"[done] Adam iters={iteration}, SA iters={sa_iters}, "
          f"best score={final_score:.6f}")
    return final_seq.tolist()
