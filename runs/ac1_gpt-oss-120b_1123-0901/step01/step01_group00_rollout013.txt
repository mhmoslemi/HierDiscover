def _conv_fft(a, b):
    n = len(a)+len(b)-1
    L = _next_pow_two(n)
    Fa = np.fft.rfft(a, n=L)
    Fb = np.fft.rfft(b, n=L)

def _grad_softmax(a, w):
    # a length n, w length 2n-1
    n = len(a)
    # reverse w
    w_rev = w[::-1]
    # convolution a * w_rev (full length n + (2n-1) -1 = 3n-2)
    grad_full = _conv_fft(a, w_rev)
    # gradient indices: k from 0..n-1 correspond to index offset = (2n-2) - k in grad_full
    start = 2*n-2
    grad = 2 * grad_full[start - np.arange(n)]
    return grad

"""
Heuristic optimizer for the “max‑convolution” objective.

The evaluation function (after the simplex normalisation Σa = √(2n))
is exactly  max(conv(a,a)).  We minimise this value by
* projecting onto the simplex after each step,
* using a smooth approximation of the max (soft‑max) to obtain a
  gradient that takes *all* convolution entries into account,
* computing convolutions and the gradient with FFTs (O(n log n)),
* gradually sharpening the soft‑max (annealing the temperature) so
  the algorithm first explores the landscape and finally behaves like
  the classic max‑gradient descent.
The routine performs many short restarts with different lengths
(500 … 3000) inside the allotted time budget (≈1000 s).  The best
sequence seen so far is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all defined at top level (no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i >= 0, Σx_i = s}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow_two(x: int) -> int:
    """Small utility – smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using an FFT of size
    next power‑of‑two (fast for the sizes we use, ≤ 3 000).
    """
    n = len(a) + len(b) - 1
    L = _next_pow_two(n)
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    C = np.fft.irfft(A * B, n=L)
    return C[:n]  # keep only the needed part


def _grad_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max of the auto‑convolution.
    a – the current sequence (length n)
    w – soft‑max weights (length 2n‑1, sum to 1)
    Returns a length‑n gradient.
    """
    n = a.shape[0]
    # reverse the weight vector
    w_rev = w[::-1]
    # convolution a * w_rev (full length = 3n‑2)
    conv_full = _conv_fft(a, w_rev)
    # indices that correspond to the gradient components
    start = 2 * n - 2
    grad = 2.0 * conv_full[start - np.arange(n)]
    return grad


def _run_opt(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    global_best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Perform a single optimisation run for a fixed length n.
    Returns the best sequence found in this run and its (max‑conv) score.
    """
    target_sum = math.sqrt(2.0 * n)          # simplex radius
    a = _project_onto_simplex(init.astype(float), target_sum)

    # initial data
    conv = _conv_fft(a, a)
    max_conv = float(np.max(conv))
    best_seq = a.copy()
    best_score = max_conv

    # hyper‑parameters
    step_factor = 0.5          # controls the raw step size
    alpha = 0.3                # soft‑max temperature (low = smooth)
    alpha_max = 25.0
    alpha_mul = 1.02           # annealing factor per iteration
    eps = 1e-12

    while time.time() < run_deadline:
        # ----- soft‑max weights -----
        w = np.exp(alpha * (conv - max_conv))   # shift by max to keep numbers ≤ 1
        w_sum = np.sum(w)
        if not np.isfinite(w_sum) or w_sum == 0.0:
            # fall back to a hard max (uniform on all max positions)
            mask = conv == max_conv
            w = np.where(mask, 1.0, 0.0)
            w_sum = np.sum(w)
        w /= w_sum

        # ----- gradient -----
        grad = _grad_softmax(a, w)
        max_abs_grad = float(np.max(np.abs(grad))) + eps

        # ----- step size (normalised) -----
        eta = step_factor * max_conv / max_abs_grad

        # ----- trial move -----
        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, target_sum)

        # recompute convolution for the trial point
        conv_new = _conv_fft(a_new, a_new)
        max_conv_new = float(np.max(conv_new))

        # ----- acceptance -----
        if max_conv_new < max_conv:
            # improvement – accept
            a, conv, max_conv = a_new, conv_new, max_conv_new
            # slightly increase the step factor to keep momentum
            step_factor = min(step_factor * 1.02, 2.0)
        else:
            # no improvement – shrink step size and possibly shake
            step_factor *= 0.7
            if np.random.rand() < 0.1:
                noise = np.random.randn(n) * (0.02 * target_sum / n)
                a = _project_onto_simplex(a + noise, target_sum)
                conv = _conv_fft(a, a)
                max_conv = float(np.max(conv))

        # keep the best seen during this run
        if max_conv < best_score:
            best_score = max_conv
            best_seq = a.copy()
            print(
                f"[{time.time():.2f}]  local best = {best_score:.6f}  (n={n})"
            )

        # anneal the temperature
        alpha = min(alpha_max, alpha * alpha_mul)

        # occasional random perturbation to escape plateaus
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a = _project_onto_simplex(a + noise, target_sum)
            conv = _conv_fft(a, a)
            max_conv = float(np.max(conv))

        # early exit if we already beat the global best
        if best_score < global_best_score:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the judging harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative sequence that minimises
    2·n·max(conv(a,a)) / (sum(a)**2).  The routine runs for at most
    ``budget_s`` seconds and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Main optimisation loop – many short restarts with different lengths
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a problem size (longer sequences give more freedom)
        n = np.random.randint(500, 3000)

        # initialise either from the supplied SOTA solution or randomly
        if np.random.rand() < 0.35 and "height_sequence_1" in globals():
            src = np.array(globals()["height_sequence_1"], dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-3
                init = np.concatenate([src, pad])
        else:
            init = np.random.rand(n)

        # give each run a modest time slice (≈ 5 s) – enough for many iterations
        run_deadline = min(deadline, time.time() + 5.0)

        seq, score = _run_opt(n, init, run_deadline, best_score)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f}  (n={n})"
            )
            # stop early if we have reached the target bound
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Fallback – if nothing was found (unlikely) return a safe uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    return best_seq.tolist()
