def propose_candidate(seed=42,budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    best_score = inf
    best_seq = None

    # define set of lengths to try
    lengths = [800,1200,2000,3000] maybe adapt based on remaining time.

    for n in lengths:
        if time.time() > deadline: break
        # initial geometric shape
        eps = 2.0 / n
        r = 1 - eps
        seq = np.power(r, np.arange(n))
        seq = seq / seq.sum()      # sum=1

        # optimization hyperparameters
        mu = 0.0005   # smoothing
        step = 0.1
        # use Adam parameters
        m = np.zeros_like(seq)
        v = np.zeros_like(seq)
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-8
        t = 0

        while time.time() < deadline:
            # compute conv
            conv = np.convolve(seq, seq)
            max_conv = conv.max()
            # compute softmax weights
            # stable exponentiation
            w_exp = np.exp((conv - max_conv) / mu)
            w = w_exp / w_exp.sum()
            # gradient via convolution of w with reversed seq
            grad = 2 * np.convolve(w, seq[::-1])[n-1 : 2*n-1]
            # Adam update
            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step_size = step / (np.sqrt(v_hat) + eps_adam)
            new_seq = seq - step_size * m_hat
            # projection onto simplex
            new_seq = _project_onto_simplex(new_seq)

            # evaluate
            score = evaluate_sequence(new_seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = new_seq.copy()
                print(f"[n={n} iter={t}] new best {score:.6f}")

            # update seq
            seq = new_seq

            # optional break after certain iterations or if improvement small
            # we check time

    # after loops
    print(f"Best objective: {best_score}")
    return best_seq.tolist()

"""
Propose a step‑function (non‑negative sequence) that beats the uniform construction.

The algorithm works in three stages:

1. **Initial candidates** – sweep a few geometric progressions
   a_i ∝ r^i (r = 1 – factor/n) on a chosen length n and keep the best
   one w.r.t. the objective (the objective for sum‑normalised vectors is
   2·n·max(conv)).

2. **Smooth max optimisation** – minimise the function
        F(a) = max_k (a*a)[k]
   on the probability simplex by a projected Adam optimiser.
   Instead of the non‑smooth max we use a temperature‑scaled soft‑max:
        w_k ∝ exp( (conv_k – max(conv))/μ )
   The gradient of the smoothed objective is
        ∇F ≈ 2·conv(w, a[::-1])[n‑1 : 2·n‑1] .
   After each Adam step we project back onto the simplex
   (non‑negative entries summing to 1).

3. **Budget aware loop** – the whole routine runs until the
   supplied time budget (≈ 1000 s) is exhausted, trying a few
   different lengths (800, 1200, 2000, 3000) and keeping the globally
   best sequence.

The code only uses NumPy and a standard O(n log n) simplex projection.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v):
    """Project a vector onto the probability simplex {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    # 1. sort v in descending order
    u = np.sort(v)[::-1]
    # 2. compute cumulative sum minus 1
    cssv = np.cumsum(u) - 1.0
    # 3. find the rho index
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:            # all entries non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _objective(seq):
    """Return the evaluation value for a sequence that already sums to 1."""
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    # sum(seq) is (numerically) 1, but keep the formula for safety
    s = seq.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """Return a non‑negative step‑function (list of floats) with low score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1.  Choose a few lengths to explore (larger n gives a better asymptotic constant
    #     but each iteration costs O(n²)).  The set can be adjusted depending on remaining time.
    lengths = [800, 1200, 2000, 3000]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # ------------------------------------------------------------------
        # 2.  Initial geometric sweep (as in the reference code)
        init_best_seq = None
        init_best_score = float("inf")
        for factor in [0.6, 0.8, 1.0, 1.5, 2.0, 2.5, 3.0]:
            if time.time() > deadline:
                break
            eps = factor / n
            r = 1.0 - eps
            if r <= 0.0:
                continue
            powers = np.power(r, np.arange(n, dtype=np.float64))
            seq = powers / powers.sum()                     # sum == 1
            sc = _objective(seq)
            if sc < init_best_score:
                init_best_score = sc
                init_best_seq = seq.copy()
            print(f"[init] factor={factor:4.2f}  eps={eps:.2e}  score={sc:.6f}")

        if init_best_seq is None:
            # fallback: uniform distribution
            init_best_seq = np.full(n, 1.0 / n)
            init_best_score = _objective(init_best_seq)

        # Keep the globally best candidate seen so far
        if init_best_score < best_score:
            best_score = init_best_score
            best_seq = init_best_seq.copy()
            print(f"[global] new best from init  score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 3.  Smooth‑max Adam optimisation on the simplex
        seq = init_best_seq.copy()

        # Adam hyper‑parameters
        lr = 0.15                     # base learning rate
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        m = np.zeros_like(seq)
        v = np.zeros_like(seq)

        # temperature schedule for the soft‑max (starts smoother, cools down)
        mu = 0.008                    # initial smoothing
        mu_decay = 0.9995

        it = 0
        while time.time() < deadline:
            it += 1

            # --- convolution and max ---
            conv = np.convolve(seq, seq)                # length 2n‑1
            max_conv = conv.max()

            # --- soft‑max weights ---
            # stability: subtract max before exponentiation
            exp_term = np.exp((conv - max_conv) / mu)
            w = exp_term / exp_term.sum()               # w ≥ 0, sum = 1

            # --- gradient of the smoothed objective (see analysis) ---
            # grad_i = 2 * sum_j w_{i+j} * seq_j
            # This is obtained via convolution of w with the reversed seq.
            grad_full = np.convolve(w, seq[::-1])        # length 3n‑2
            grad = 2.0 * grad_full[n - 1 : 2 * n - 1]    # slice length n

            # --- Adam update ---
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = lr / (np.sqrt(v_hat) + eps_adam)
            seq_new = seq - step * m_hat

            # --- projection onto the simplex ---
            seq_new = _project_onto_simplex(seq_new)

            # --- compute objective (fast, no extra evaluate_sequence call) ---
            cur_score = _objective(seq_new)

            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq_new.copy()
                print(f"[Iter n={n} it={it:05d}] new best score = {best_score:.6f}")

            # --- prepare next iteration ---
            seq = seq_new
            mu *= mu_decay                     # cool down the temperature

            # optional early‑stop if improvement stalls
            if it % 2000 == 0:
                # after a few thousand iterations we may prune tiny step sizes
                if lr * (1.0 - beta1) < 1e-7:
                    break

        print(f"[n={n}] finished after {it} iterations, best score so far = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Return the best sequence found (as a plain Python list)
    print(f"\nFinal best objective = {best_score:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
