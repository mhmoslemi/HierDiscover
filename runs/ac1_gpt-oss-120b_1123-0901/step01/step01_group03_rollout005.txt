def _project_onto_simplex(v):
    """Exact Euclidean projection onto simplex {x>=0, sum x = 1}."""
    # Based on algorithm from
    # https://doi.org/10.1007/s10107-005-0704-0
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative sequence) that minimises
    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm uses a smoothed version of the peak of the self‑convolution
and performs Adam optimisation on the probability simplex.

Key ideas
---------
* The objective is homogeneous, therefore we fix   Σ a_i = 1 .
* Instead of the non‑differentiable   max(conv)   we use a soft‑max:
      f_τ(a) = τ·log Σ_k exp(conv_k / τ)      (τ → 0 → max).
* For a given τ the gradient of f_τ is a convex combination of the
  gradients of the individual convolution entries and can be computed
  with one FFT‑based convolution.
* Adam (adaptive moment estimation) provides fast, noise‑robust updates.
* After every update the iterate is projected onto the simplex
  {x ≥ 0, Σx = 1} by the exact Euclidean projection of Duchi et al.
* τ is annealed (geometric decay) from a moderate start (≈0.1) down to

* Occasionally a small random kick is added to escape plateaus.
* The best vector seen during the run is returned.

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Exact Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}.
    Implementation of the algorithm by Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    # find the largest ρ satisfying u_ρ - (cumsum_ρ - 1)/ρ > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive -> return the uniform vector
        return np.full_like(v, 1.0 / n)

    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns the full convolution (len(x)+len(y)-1).
    """
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())      # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    return conv


def _full_self_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the linear convolution a * a (length 2·len(a) − 1) using FFT.
    Small negative values caused by numerical noise are clipped to 0.
    """
    conv = _fft_convolve(a, a)
    # eliminate round‑off negatives
    np.maximum(conv, 0.0, out=conv)
    return conv


def _gradient_softmax_max(a: np.ndarray, conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Gradient of the smoothed max   f_τ(a) = τ·log Σ exp(conv/τ)   with respect
    to the vector a (Σ a_i = 1, a_i ≥ 0).

    The gradient w.r.t. a_i is 2 * Σ_k w_k a_{k-i},
    where w_k = softmax(conv_k / τ).  The summation is exactly a
    cross‑correlation of a with the reversed weight vector.
    """
    # soft‑max weights (stable formulation)
    max_c = np.max(conv)
    w = np.exp((conv - max_c) / tau)
    w_sum = np.sum(w)
    if w_sum == 0.0:                # fallback (should not happen)
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # reverse the weight vector for the cross‑correlation
    w_rev = w[::-1]

    # convolution a * w_rev gives a vector of length 3·n−2
    grad_full = _fft_convolve(a, w_rev)

    n = a.shape[0]
    # extract the needed slice: grad_i = 2 * conv_rev[2n‑2‑i]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (Python floats) that aims to minimise
    the evaluation metric defined in ``evaluate_sequence``.
    The function runs for at most ``budget_s`` seconds and prints progress.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a length and an initial vector
    # ------------------------------------------------------------------
    try:
        # use the previously discovered sequence if available
        init_seq = np.asarray(height_sequence_1, dtype=np.float64)
        n = init_seq.size
    except NameError:
        # fall back to a moderate size (still large enough to see the constant)
        n = 5000
        # central‑binomial coefficients (C(2i,i)/4**i)
        init_seq = np.empty(n, dtype=np.float64)
        init_seq[0] = 1.0
        for i in range(1, n):
            init_seq[i] = init_seq[i - 1] * (2.0 * i - 1.0) / (2.0 * i)

    # normalise to the simplex Σa_i = 1
    a = _project_onto_simplex(init_seq)

    # ------------------------------------------------------------------
    # 2. Adam optimiser initialisation
    # ------------------------------------------------------------------
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.12                     # base learning rate (tuned empirically)

    beta1_pow = 1.0
    beta2_pow = 1.0

    # temperature schedule for the soft‑max
    tau = 0.12                    # start relatively smooth
    tau_min = 1e-4
    tau_decay = 0.9995            # geometric decay each iteration

    # ------------------------------------------------------------------
    # 3. Main optimisation loop
    # ------------------------------------------------------------------
    best_a = a.copy()
    conv = _full_self_convolution(a)
    best_score = 2.0 * n * np.max(conv)
    print(f"[init] n={n}, score={best_score:.6f}")

    iteration = 0
    no_improve_cnt = 0

    while time.time() < deadline:
        # ---- compute convolution and current score
        conv = _full_self_convolution(a)
        max_val = np.max(conv)
        score = 2.0 * n * max_val

        # keep the best ever seen
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve_cnt = 0
            print(f"[iter {iteration:6d}] new best score = {best_score:.6f}")
        else:
            no_improve_cnt += 1

        # ---- gradient of the smoothed max
        grad = _gradient_softmax_max(a, conv, tau)

        # ---- Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        beta1_pow *= beta1
        beta2_pow *= beta2

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # ---- project back onto the simplex
        a = _project_onto_simplex(a)

        # ---- temperature annealing
        tau = max(tau * tau_decay, tau_min)

        # ---- occasional random kick to escape flat plateaus
        if iteration % 2500 == 0 and iteration > 0:
            noise = np.random.randn(n) * 0.003
            a = _project_onto_simplex(a + noise)

        # ---- simple safeguard: reinitialise if stuck for a long time
        if no_improve_cnt >= 20000:
            # perturb heavily and keep moving
            a = _project_onto_simplex(a + np.random.rand(n) * 0.01)
            no_improve_cnt = 0
            print(f"[iter {iteration:6d}] random re‑initialisation triggered")

        iteration += 1

    # ------------------------------------------------------------------
    # 4. Return the best vector found (as plain Python floats)
    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations. Best score = {best_score:.6f}")
    # Convert to plain list of Python floats as required by the outer harness
    return [float(x) for x in best_a.tolist()]
