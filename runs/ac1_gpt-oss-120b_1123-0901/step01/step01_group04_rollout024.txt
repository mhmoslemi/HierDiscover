        k = np.random.choice(active_idxs)

        grad = np.zeros_like(a)

        if k < n:
            # positive shift
            # indices i where i + k < n
            grad[:n - k] += a[k:]

        step = eta * grad
        a_new = a - step

        # ensure nonnegativity
        a_new = np.maximum(a_new, 0.0)

        # renormalize sum to 1 (if sum zero, break)
        s = a_new.sum()
        if s < 1e-12:
            # revert small random jitter
            a_new = a
        else:
            a_new = a_new / s

        # evaluate new score
        new_score = evaluate_sequence(a_new.tolist())

            best_seq = a_new.copy()

def propose_candidate(seed=42,budget_s=1000,**kwargs):
    import time, numpy as np

    np.random.seed(seed)
    # Choose length n
    n = kwargs.get("length")
    if n is None:
        # try use global sequence length or fallback
        try:
            n = len(height_sequence_1)
            # maybe reduce for speed, say min(3000,n)
            n = min(3000, n)
        except Exception:
            n = 2000
    # Build init sequence
    init_type = kwargs.get("init","arcsine")
    if init_type == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif init_type == "uniform":
        a = np.full(n, 1.0/n)
    else:
        # arcsine (Catalan)
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for k in range(1, n):
            a[k] = a[k-1] * ((2*k-1)*(2*k))/(4.0*k*k)
        a /= a.sum()
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print("[init] n=", n, "score=", best_score)

    # Precompute power of two for FFT
    m = 1 << ((2*n - 1).bit_length())
    
    deadline = time.time() + budget_s - 0.1
    it = 0
    # parameters
    eta = 1.0
    decay = 0.995
    # keep history of recent scores to adapt step size
    while time.time() < deadline:
        # compute conv via FFT (real)
        fa = np.fft.rfft(a, n=m)
        conv = np.fft.irfft(fa * fa)[:2*n-1]
        max_val = conv.max()
        # early exit if we hit target
        if best_score < 1.503:
            break
        # pick active index (any of idx with max)
        idx_max = np.flatnonzero(conv == max_val)
        k = np.random.choice(idx_max)  # conv index
        # compute gradient for this k
        grad = np.zeros_like(a)
        # for each i, compute j = k - i
        # vectorized: positions where j in [0,n)
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]
        # apply exponentiated gradient step
        # step size adapt 
        step = eta
        # compute new a
        # exponentiated update
        a_new = a * np.exp(-step * grad)
        # renormalize
        sum_a_new = a_new.sum()
        if sum_a_new <= 0:
            # fallback to keep a unchanged
            break
        a_new /= sum_a_new
        # evaluate
        new_score = evaluate_sequence(a_new.tolist())
        if new_score < best_score:
            best_seq = a_new.copy()
            best_score = new_score
            a = a_new
            # optionally increase step size a bit to capitalize direction
            eta = min(1.0, eta / decay)  # increase??
            print(f"[improve] it={it} score={best_score:.6f} eta={eta:.5f}")
        else:
            # maybe accept with small prob
            # simple Simulated Annealing style

            if np.random.rand() < prob:
                a = a_new
        # decay step size gradually
        eta *= decay
        if eta < 1e-6:
            eta = 1e-6
        it += 1

    print("[final] it", it, "best", best_score)
    return best_seq.tolist()

propose_candidate
=================

A two‑stage optimizer for the step‑function evaluation:

1. **Shape search** – try a few symmetric Beta‑type families
   a_i ∝ (i+½)^{α‑1}·(n‑½‑i)^{α‑1} (α≈0.5 is the classical arcsine).
   The best member (according to a fast FFT‑based scoring) seeds the
   second stage.

2. **Projected exponentiated‑gradient descent** – repeatedly
   locate the shift where the self‑convolution attains its maximum,
   compute the exact sub‑gradient of that convolution entry,
   and update the sequence by a multiplicative (exponentiated‑gradient)
   step.  After each step the vector is projected back onto the
   probability simplex (non‑negative, unit sum).  The step size decays
   gradually; occasional random jitter helps escape plateaus.

The routine runs for the allotted time (default 1000 s) and returns
the best sequence found.  All numbers stay non‑negative; the evaluator
handles the final scaling automatically.
"""

# Helper: fast max‑convolution using FFT (linear convolution, no padding)
# ----------------------------------------------------------------------
def _max_conv_fft(x: np.ndarray) -> float:
    """Return max value of the linear convolution x*x using FFT."""
    n = x.shape[0]
    # length for FFT – next power of two >= 2n‑1
    m = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(fx * fx, n=m)[: 2 * n - 1]
    return float(conv.max())

# ----------------------------------------------------------------------
