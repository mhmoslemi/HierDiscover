    return seq


def _conv_max_and_index(seq: np.ndarray) -> tuple[float, int]:
    """Fast convolution using numpy (FFT for large n) – returns max value and argmax."""
    conv = np.convolve(seq, seq)          # linear convolution, length 2n-1
    max_val = float(conv.max())
    max_idx = int(conv.argmax())
    return max_val, max_idx


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Generate and iteratively improve a non‑negative step function.
    Returns a list of floats suitable for `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine the length of the candidate.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        n = 3000                      # default length if no prior sequence

    # ------------------------------------------------------------------
    # 2. Base sequence – inverse‑square‑root shape.
    # ------------------------------------------------------------------
    seq = np.array(generate_inverse_sqrt_sequence(n), dtype=float)

    # Initial score
    try:
        best_score = evaluate_sequence(seq.tolist())
    except Exception:
        # safety fallback – uniform sequence
        seq = np.ones(n, dtype=float)
        best_score = evaluate_sequence(seq.tolist())
    best_seq = seq.copy()

    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Iterative improvement – mass‑transfer moves + occasional global jitter.
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 0.1         # tiny margin for return
    no_improve_counter = 0
    # parameters (tuned empirically)
    transfer_frac = 0.05    # fraction of the donor entry transferred each move
    global_jitter_every = 2000   # perform a global jitter after this many failed moves
    jitter_batch = max(2, n // 200)   # how many entries to perturb globally
    jitter_sigma = 0.25               # log‑normal width for global jitter

    while time.time() < deadline:
        # ----- compute the current convolution maximum -----
        max_b, k_max = _conv_max_and_index(seq)

        # ----- contributions of each index to the current maximal shift -----
        # only indices i with 0 <= k_max-i < n contribute
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        contrib = np.zeros(n, dtype=float)
        # vectorised slice update
        idx = np.arange(i_start, i_end + 1, dtype=int)
        opp = k_max - idx
        # contribution = 2 * a_i * a_{k_max-i}   (double counted except on diagonal)
        contrib[idx] = 2.0 * seq[idx] * seq[opp]
        # correct the diagonal where i == k_max-i (i.e. idx == opp)
        diag = idx == opp
        contrib[idx[diag]] = seq[idx[diag]] * seq[idx[diag]]

        # ----- pick donor (largest contribution) and receiver (low contribution) -----
        donor = int(np.argmax(contrib))
        # try to find a receiver that does not affect the current maximal shift
        if i_start > 0:
            # there is a prefix of indices that do not contribute
            receiver = int(rng.integers(0, i_start))
        elif i_end < n - 1:
            # there is a suffix that does not contribute
            receiver = int(rng.integers(i_end + 1, n))
        else:
            # all indices contribute – pick a low‑contribution index
            # sample a few candidates and keep the one with smallest contribution
            cand = rng.integers(n, size=5)
            # avoid picking the donor again
            cand = cand[cand != donor]
            if cand.size == 0:
                continue
            receiver = int(cand[np.argmin(contrib[cand])])

        # ----- perform the mass transfer -----
        donor_amount = seq[donor]
        if donor_amount <= 0.0:
            # nothing to move – skip this iteration
            continue
        eps = transfer_frac * donor_amount
        eps = min(eps, donor_amount)   # keep donor non‑negative
        new_seq = seq.copy()
        new_seq[donor] -= eps
        new_seq[receiver] += eps

        # evaluate candidate
        try:
            cand_score = evaluate_sequence(new_seq.tolist())
        except Exception:
            # illegal candidate – ignore
            continue

        if cand_score < best_score:
            best_score = cand_score
            best_seq = new_seq.copy()
            seq = new_seq               # continue from the improved point
            no_improve_counter = 0
            print(f"[improve] score={best_score:.6f} (donor {donor}→receiver {receiver})")
            continue

        # ----- no improvement – possibly apply a global jitter -----
        no_improve_counter += 1
        if no_improve_counter >= global_jitter_every:
            # random log‑normal perturbation of a small subset of entries
            jitter = seq.copy()
            idxs = rng.choice(n, size=jitter_batch, replace=False)
            factors = np.exp(rng.normal(0.0, jitter_sigma, size=jitter_batch))
            jitter[idxs] *= factors
            jitter = np.maximum(jitter, 0.0)   # enforce non‑negativity
            try:
                jitter_score = evaluate_sequence(jitter.tolist())
            except Exception:
                jitter_score = np.inf
            if jitter_score < best_score:
                best_score = jitter_score
                best_seq = jitter.copy()
                seq = jitter
                print(f"[jitter] new best score={best_score:.6f}")
            no_improve_counter = 0

    # ------------------------------------------------------------------
    # 4. Return the best sequence found.
    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
