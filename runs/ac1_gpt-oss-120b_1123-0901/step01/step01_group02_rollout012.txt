"""
search for a step‑function with minimal value of
    E(a) = 2 * n * max_k (a * a)[k] / (sum a)^2

The program uses a smooth approximation of the maximum (log‑sum‑exp)
so that a gradient exists everywhere.  Gradients are obtained by
cross‑correlating the current weights with the soft‑max weights; the
convolutions are performed with FFTs for speed.  An Adam optimiser
with a decaying learning‑rate together with a slowly increasing
soft‑max temperature explores the space.  Small random jitters are
added periodically to escape shallow plateaus.  The best sequence seen
within the allotted wall‑clock time is returned.
"""

import time
import math
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Helper: fast linear convolution via FFT (real inputs)
def fast_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear convolution of two real 1‑D arrays using FFT."""
    n1 = x.shape[0]
    n2 = y.shape[0]
    nfft = 1 << ((n1 + n2 - 2).bit_length())    # next power of two
    fx = np.fft.rfft(x, n=nfft)
    fy = np.fft.rfft(y, n=nfft)
    fz = fx * fy
    z = np.fft.irfft(fz, n=nfft)
    return z[: n1 + n2 - 1]

    """

    Parameters

    kwargs : dict

    Returns

        The best sequence discovered within the time limit.
    """

    start_time = time.time()
    deadline = start_time + budget_s - 3.0               # safety margin

    # --------------------------------------------------------------
    # Problem size.  A few thousand entries give a good trade‑off

    n = 3000

    S = target_sum

    # --------------------------------------------------------------
