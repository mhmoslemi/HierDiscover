"""
Propose a candidate step‑function (non‑negative heights) that tries to minimise

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The search exploits the observation that a decreasing “power‑law” tail gives a
low maximum convolution, while a flat low‑plateau added after some index m can
increase the total L1 mass without raising the peak too much.  The algorithm
randomly samples tuples (p, γ, m, f) where

    a_i = (i + γ) ** (‑p)          for i < m
    a_i = max(a_i, f)             for i ≥ m   (a low constant “floor”)

and keeps the best sequence found.  After a coarse random phase a short
local‑refinement (Gaussian perturbations) is performed.  The total wall‑time
cannot exceed ``budget_s`` (default 1000 s).  The routine returns a plain
Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# The scorer ``evaluate_sequence`` is assumed to be available globally.


def _powerlaw(n: int, p: float, gamma: float) -> np.ndarray:
    """Return (i+γ)^(-p) for i=0..n‑1."""
    return (np.arange(n, dtype=float) + gamma) ** (-p)


def propose_candidate(seed: int = 123, budget_s: float = 1000) -> list[float]:
    """Search for a low‑score step function within the given time budget.

    The search uses a simple random‑and‑local‑refinement scheme over the
    parameter space (p, γ, m, floor).  The returned list contains the heights
    of the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # safety margin
    n = 3428                                  # length used by previous solutions

    # -----------------------------------------------------------------
    # 1) initialise with the best pure power‑law that is already known
    # -----------------------------------------------------------------
    best_p, best_gamma = 0.5002, 0.3210
    best_seq = _powerlaw(n, best_p, best_gamma)
    best_score = evaluate_sequence(best_seq.tolist())
    best_m, best_floor = n, 0.0               # “no floor” – pure power‑law
    print(f"[init] score={best_score:.6f} (pure power‑law)")

    # -----------------------------------------------------------------
    # 2) random exploration of (p,γ,m,floor) triples
    # -----------------------------------------------------------------
    # parameter bounds
    p_min, p_max = 0.45, 0.70
    gamma_min, gamma_max = 0.05, 5.0
    m_min, m_max = int(0.05 * n), int(0.90 * n)
    floor_min, floor_max = 0.0, 0.15        # a modest plateau height

    # number of random candidates (will stop earlier if time runs out)
    while time.time() < deadline:
        # decide which family to sample
        if np.random.rand() < 0.2:               # pure power‑law (no floor)
            p = np.random.uniform(p_min, p_max)
            gamma = np.random.uniform(gamma_min, gamma_max)
            seq = _powerlaw(n, p, gamma)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_p, best_gamma = p, gamma
                best_m, best_floor = n, 0.0
                print(f"[pure] new best p={p:.4f} γ={gamma:.4f} "
                      f"score={score:.6f}")
            continue

        # piecewise with a flat floor after index m
        p = np.random.uniform(p_min, p_max)
        gamma = np.random.uniform(gamma_min, gamma_max)
        m = np.random.randint(m_min, m_max)
        floor_val = np.random.uniform(floor_min, floor_max)

        seq = _powerlaw(n, p, gamma)
        # raise the tail to at least ``floor_val``
        seq[m:] = np.maximum(seq[m:], floor_val)

        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_p, best_gamma = p, gamma
            best_m, best_floor = m, floor_val
            print(f"[floor] new best p={p:.4f} γ={gamma:.4f} "
                  f"m={m} floor={floor_val:.5f} score={score:.6f}")

    # -----------------------------------------------------------------
    # 3) local refinement around the best found parameters
    # -----------------------------------------------------------------
    # standard deviations for Gaussian perturbations
    dp, dγ, dm, df = 0.008, 0.15, int(0.03 * n), 0.005
    for _ in range(8000):
        if time.time() > deadline:
            break

        # propose a new candidate close to the current best
        p_new = best_p + np.random.randn() * dp
        gamma_new = best_gamma + np.random.randn() * dγ
        m_new = int(round(best_m + np.random.randn() * dm))
        floor_new = best_floor + np.random.randn() * df

        # enforce bounds
        if not (p_min <= p_new <= p_max):
            continue
        if not (gamma_min <= gamma_new <= gamma_max):
            continue
        if not (m_min <= m_new <= m_max):
            continue
        if floor_new < 0.0:
            floor_new = 0.0
        if floor_new > floor_max:
            floor_new = floor_max

        seq = _powerlaw(n, p_new, gamma_new)
        seq[m_new:] = np.maximum(seq[m_new:], floor_new)

        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_p, best_gamma = p_new, gamma_new
            best_m, best_floor = m_new, floor_new
            print(f"[refine] improved p={p_new:.4f} γ={gamma_new:.4f} "
                  f"m={m_new} floor={floor_new:.5f} score={score:.6f}")
            # early exit if we already beat the target
            if best_score < 1.503:
                break

    # -----------------------------------------------------------------
    # 4) final result
    # -----------------------------------------------------------------
    print(f"Finished search – best score {best_score:.6f} "
          f"with p={best_p:.4f}, γ={best_gamma:.4f}, "
          f"m={best_m}, floor={best_floor:.5f}")
    return best_seq.tolist()
