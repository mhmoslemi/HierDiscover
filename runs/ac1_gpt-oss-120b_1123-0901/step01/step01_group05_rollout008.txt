"""
Propose a candidate step‑function sequence.

Algorithm
---------
1. Initialise a sequence `a` of length *n* (taken from the previous best
   construction if available, otherwise a modest default) with the classic
   inverse‑square‑root shape `a_i = binom(2*i,i)/4**i`.  The sequence is
   normalised to unit ℓ₁‑norm (the evaluation is scale‑invariant).

2. Repeatedly improve the shape by *gradient descent on the variance of the
   full convolution* `B = a * a`.  For a given `a` we have:

        L(a) = Σ_k (B_k - μ)²,   μ = mean(B)

   The gradient of `L` with respect to `a` is

        ∂L/∂a_i = 4· Σ_k B_k·a_{k-i} – 4·S³ / (2n‑1),

   where `S = Σ a_i`.  The first term is a convolution of `B` with the
   reversed sequence `a_rev`.  All convolutions are performed with FFTs in
   O(n log n) time.

3. In each iteration a step `η·grad` is taken (with a small, adaptive step
   size) and the resulting vector is projected onto the feasible set
   by clipping negatives to zero and re‑normalising the sum to one.
   If the new sequence yields a lower evaluation score it is kept;
   otherwise the step size is reduced.  Occasionally a tiny random
   multiplicative perturbation is tried to escape plateaus.

4. The loop runs until the time budget (default 1000 s) is exhausted or the
   target bound 1.503 is reached.  The best sequence found is returned.

The method focuses on flattening the convolution (making it almost constant),
which directly minimises the quantity `2·n·max(B)/sum(a)²` that the
`evaluate_sequence` function returns.
"""

import time
import numpy as np

def generate_inverse_sqrt_sequence(n: int) -> np.ndarray:
    """Return a NumPy array a_0,…,a_{n‑1} with a_i = binom(2*i,i)/4**i."""
    seq = np.empty(n, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, n):
        # recurrence a_i = a_{i‑1} * (2*i‑1) / (2*i)
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return seq


def compute_full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Return the full linear convolution a*a using FFT of length `fft_len`.
    The result has length 2*len(a)-1.
    """
    A = np.fft.rfft(a, n=fft_len)
    B_fft = A * A                      # element‑wise product → a*a in freq. domain
    conv = np.fft.irfft(B_fft, n=fft_len)[: 2 * len(a) - 1]
    return conv, B_fft                 # also return the frequency representation


def compute_score_from_conv(conv: np.ndarray, a_sum: float, n: int) -> float:
    """
    Compute the evaluation score given a convolution `conv`,
    the ℓ₁‑norm `a_sum` of the underlying sequence and length `n`.
    """
    if a_sum < 0.01:
        return float("inf")
    max_b = conv.max()
    return 2.0 * n * max_b / (a_sum * a_sum)

            a_pert[idxs] *= mult
            a_pert = np.clip(a_pert, 0.0, None)
            a_pert /= a_pert.sum()
            conv_pert, _ = compute_full_convolution(a_pert, fft_len_full)
            pert_score = compute_score_from_conv(conv_pert, a_pert.sum(), n)
            if pert_score < current_score:
                a = a_pert
                conv_full = conv_pert
                current_score = pert_score
                no_improve = 0
                step_factor = max(step_factor * 0.9, 0.1)
                if pert_score < best_score:
                    best_score = pert_score
                    best_seq = a_pert.copy()
                    print(f"[random improve] iter {iteration:6d}, score {best_score:.6f}")

        # ------------------------------------------------------------------
        # Early exit if target bound already reached
        # ------------------------------------------------------------------
        if best_score < 1.503:
            print("[target reached]")
            break

    # ----------------------------------------------------------------------
    # 4. Return the best sequence discovered.
    # ----------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f} after {iteration} iterations")
    return best_seq.tolist()
