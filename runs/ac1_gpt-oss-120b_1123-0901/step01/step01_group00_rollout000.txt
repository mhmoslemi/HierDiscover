def _convolve_full_fft(x, y):
    """Full convolution of two 1-D arrays using FFT."""
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    Z = X * Y
    conv = np.fft.irfft(Z, n=size)[:n]
    return conv

"""
Propose a sequence of non‑negative heights that minimizes
    2 * n * max(conv(a, a)) / (sum(a)**2)

We exploit two facts:

* The denominator is fixed by the simplex projection
  ( Σ a = √(2n) ), therefore the objective equals max(conv).
* The max is the limit of the smooth “log‑sum‑exp” function:
      f_α(a) = (1/α)·log Σ exp(α·conv_k(a)) .
  Its gradient involves only a single convolution:
      ∇f_α = 2·conv( w , a_rev )
  where w_k ∝ exp(α·conv_k) (soft‑max weights) and a_rev is a reversed.

The algorithm runs many short Adam‑optimisation runs with a continuation
on α (start small → smooth, end large → near the true max).  After each
step we Euclidean‑project back onto the simplex
{ x_i ≥ 0 , Σ x_i = √(2n) }.
Periodically the exact evaluation (via evaluate_sequence) is performed and
the best sequence seen so far is kept.

The method is completely self‑contained, uses only NumPy and the already
available evaluate_sequence, and respects the 1000 s time budget.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex Σ x = s , x ≥ 0
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x | x_i >= 0, sum x_i = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, v.shape[0] + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: full convolution using FFT (real input → real output)
# ----------------------------------------------------------------------
def _convolve_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full convolution of real vectors x and y using FFT."""
    n = len(x) + len(y) - 1
    # next power of two for efficient FFT
    size = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    Z = X * Y
    conv = np.fft.irfft(Z, n=size)[:n]
    return conv


# ----------------------------------------------------------------------
# Main routine required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    evaluation score.  The routine runs at most ``budget_s`` seconds and
    keeps the best sequence discovered.
    """
    np.random.seed(seed)
    # small safety margin for the outer deadline
    deadline = time.time() + budget_s - 0.5

    best_seq = None          # numpy array holding the best candidate
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Utility: evaluate a candidate and possibly update the global best.
    # ------------------------------------------------------------------
    def _maybe_update(a_vec, n):
        nonlocal best_seq, best_score
        try:
            score = evaluate_sequence(a_vec.tolist())
        except Exception:
            score = float("inf")
        if score < best_score:
            best_score = score
            best_seq = a_vec.copy()
            print(f"[{time.time():.2f}] NEW BEST = {best_score:.6f} (n={n})")
        return score

    # ------------------------------------------------------------------
    # Main search loop – a collection of short Adam runs with random restarts.
    # ------------------------------------------------------------------
    # Preference list of lengths (shorter runs are cheaper, larger n may give
    # better minima).  The list is deliberately diverse.
    candidate_lengths = [500, 800, 1200, 1500, 2000, 2500, 3000]

    while time.time() < deadline:
        # --------------------------------------------------------------
        # Pick a problem size and initialise a starting point.
        # --------------------------------------------------------------
        n = np.random.choice(candidate_lengths)

        # Try to seed from an existing construction if available.
        if 'height_sequence_1' in globals() and np.random.rand() < 0.4:
            src = np.array(height_sequence_1, dtype=float)
            if src.size >= n:
                a = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-4
                a = np.concatenate([src, pad])
        elif best_seq is not None and np.random.rand() < 0.3:
            # start from the best seen so far (trim / pad)
            src = best_seq
            if src.size >= n:
                a = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-4
                a = np.concatenate([src, pad])
        else:
            # pure random start
            a = np.random.rand(n)

        # Project onto the simplex Σ a = √(2n)
        target_sum = np.sqrt(2.0 * n)
        a = _project_onto_simplex(a, target_sum)

        # --------------------------------------------------------------
        # Adam optimiser initialisation
        # --------------------------------------------------------------
        lr = 0.08                     # base learning rate
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        # Continuation on the soft‑max temperature α
        alpha = 0.02                  # start very smooth
        alpha_max = 10.0
        # Number of Adam steps we allow in this run (based on remaining time)
        run_deadline = min(deadline, time.time() + 9.0)   # at most ~9 s per run
        it = 0

        # --------------------------------------------------------------
        # Optimisation loop for the current restart
        # --------------------------------------------------------------
        while time.time() < run_deadline:
            # Full autocorrelation
            conv = _convolve_full_fft(a, a)          # length 2n‑1
            max_conv = conv.max()

            # Gradually increase α (exponential schedule works well)
            if alpha < alpha_max:
                alpha = min(alpha_max, alpha * 1.045 + 0.001)

            # Soft‑max weights w_k ∝ exp(α·conv_k)
            # Subtract max for numerical stability
            exp_shift = np.exp(alpha * (conv - max_conv))
            w = exp_shift / exp_shift.sum()          # length 2n‑1, sums to 1

            # Gradient = 2·conv(w , a_rev)  (full length = 3n‑2)
            a_rev = a[::-1]
            grad_full = _convolve_full_fft(w, a_rev)  # length 3n‑2
            # Extract the n components that correspond to ∂/∂a_i
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]

            # Adam update
            it += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Keep feasibility
            a = _project_onto_simplex(a, target_sum)

            # ------------------------------------------------------------------
            # Periodic exact evaluation (every ~30 iterations) and occasional
            # random shake‑up to escape flat plateaus.
            # ------------------------------------------------------------------
            if it % 30 == 0:
                _maybe_update(a, n)

                # Very small chance of a perturbation
                if np.random.rand() < 0.015:
                    noise = np.random.randn(n) * (0.01 * target_sum / n)
                    a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)

            # Early exit if we already hit the target bound
            if best_score < 1.5:
                break

        # End of this restart – final exact check
        _maybe_update(a, n)

        # Global early stop if we beat the target
        if best_score < 1.5:
            break

    # ------------------------------------------------------------------
    # Fallback: uniform sequence (always feasible) if nothing was found.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    return best_seq.tolist()
