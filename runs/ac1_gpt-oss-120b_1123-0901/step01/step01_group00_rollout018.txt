def _lp_step(f, M, target_sum):
    # Input f length n, M is max conv value for f.
    n = f.shape[0]
    rows = 2 * n - 1
    # Build A_ub, shape (rows, n)
    # To avoid building dense matrix each time we could generate in sparse format.
    # We'll use scipy.sparse.csr_matrix for speed.
    
    import scipy.sparse as sp
    data = []
    row_ind = []
    col_ind = []
    for k in range(rows):
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        # col indices j = low..high
        cols = np.arange(low, high + 1, dtype=int)
        # f coefficients: f[k - cols]
        vals = f[k - cols]  # This works as f indexing with array
        row_ind.extend([k] * len(cols))
        col_ind.extend(cols.tolist())
        data.extend(vals.tolist())
    A_ub = sp.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))
    # bounds for variable b_i >=0
    bounds = [(0, None)] * n
    # objective: maximize sum(b) -> minimize -sum(b)
    c = np.full(n, -1.0)
    # right-hand side:
    b_ub = np.full(rows, M, dtype=float)
    # call linprog
    try:
        res = sp.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 5})
    except Exception as e:
        return None
    if not res.success:
        return None
    b = res.x
    # Scale to target sum if sum > 0
    sum_b = b.sum()
    if sum_b <= 0:
        return None
    scale = target_sum / sum_b
    b_scaled = b * scale
    # Clip to avoid >1000 due to scaling; but they will be small anyway.
    return b_scaled

def _lp_step(f, M, target_sum):
    n = f.shape[0]
    rows = 2 * n - 1
    # Build sparse matrix
    data = []
    row_inds = []
    col_inds = []
    for k in range(rows):
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        cols = np.arange(low, high + 1, dtype=int)
        vals = f[k - cols]   # f[k - col] yields directly
        # extend lists
        row_inds.extend([k] * len(cols))
        col_inds.extend(cols.tolist())
        data.extend(vals.tolist())
    A_ub = sp.csr_matrix((data, (row_inds, col_inds)), shape=(rows, n))
    b_ub = np.full(rows, M, dtype=np.float64)
    bounds = [(0, None)] * n
    c = np.full(n, -1.0, dtype=np.float64)  # maximize sum => minimize negative sum
    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        return None
    scale = target_sum / sum_b
    b_scaled = b * scale
    # Clip to stay within evaluate bounds
    b_scaled = np.clip(b_scaled, 0, 1000.0)
    return b_scaled

"""
Simple alternating LP optimizer for the autocorrelation‐maximisation problem.

Algorithm
---------

* The objective 2*n*max(conv(a,a))/sum(a)^2 is scale‑invariant.
  By projecting the vector onto the simplex
      {x_i ≥ 0 , Σ x_i = √(2 n)},
  the denominator becomes 2 n and the objective reduces to
  max_k (a * a)[k].

* For a current vector f we compute its maximal convolution value M.
  Keeping the sum fixed, we look for a direction g that maximises the
  linear functional Σ g_i while satisfying the linear constraints

        (f * g)[k] ≤ M   for all 0 ≤ k < 2 n‑1 ,   g_i ≥ 0 .

  This is a linear program (LP); its solution pushes mass to positions
  that do not increase any convolution entry beyond the current bound.

* The LP solution g is scaled back to the simplex (its ℓ₁‑norm becomes
  √(2 n)) and mixed with the current vector:

        f ← (1‑t)·f + t·g ,   with a small mixing factor t.

  The mixing keeps the sum unchanged, reduces the maximal convolution,
  and provides a new starting point for the next LP.

* The procedure is run repeatedly for many random restarts, different
  lengths n (200 … 1500) and random initial vectors (or a warm start from
  the previously discovered sequence `height_sequence_1`).  The best
  vector found across all runs is returned.

The implementation uses SciPy’s HiGHS linear‑programming solver with a
sparse constraint matrix, a fast Euclidean projection onto the simplex,
and occasional random perturbations to escape flat regions.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# --------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# --------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------
# Linear‑programming step: maximise Σ b_i  subject to  f∗b ≤ M
# --------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float, target_sum: float) -> np.ndarray | None:
    """
    Solve the LP
        maximise   Σ b_i
        subject to (f * b)[k] ≤ M   (k = 0 … 2n‑2)
                   b_i ≥ 0 .
    The returned vector is scaled to have ℓ₁‑norm = target_sum.
    Returns None if the LP fails.
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Build the sparse constraint matrix A_ub where
    #   A_ub[k, j] = f[k‑j]  if 0 ≤ k‑j < n, otherwise 0.
    data = []
    row_ind = []
    col_ind = []

    for k in range(rows):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        cols = np.arange(lo, hi + 1, dtype=int)
        vals = f[k - cols]               # f[k‑j]
        row_ind.extend([k] * len(cols))
        col_ind.extend(cols.tolist())
        data.extend(vals.tolist())

    A_ub = sp.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))
    b_ub = np.full(rows, M, dtype=np.float64)

    # maximise Σ b_i  → minimise −Σ b_i
    c = np.full(n, -1.0, dtype=np.float64)
    bounds = [(0, None)] * n

    try:
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                          method="highs", options={"presolve": True})
    except Exception:
        return None

    if not res.success or res.x is None:
        return None

    b = res.x
    s = b.sum()
    if s <= 0:
        return None
    # Scale to the required ℓ₁‑norm (the mixing step later keeps the sum fixed)
    scale = target_sum / s
    b_scaled = b * scale
    # Safety clipping – the evaluation function caps at 1000 anyway
    b_scaled = np.clip(b_scaled, 0.0, 1000.0)
    return b_scaled


# --------------------------------------------------------------------
# Run a single optimisation trajectory for a fixed length n
# --------------------------------------------------------------------
def _run_opt(n: int, init: np.ndarray, global_deadline: float) -> tuple[np.ndarray, float]:
    """
    Optimise a length‑n sequence starting from ``init``.
    Returns the best sequence found inside this run and its score
    (the maximal convolution entry, which equals the objective value
    because the vector lives on the simplex).
    """
    target_sum = np.sqrt(2.0 * n)
    f = _project_onto_simplex(init.astype(np.float64), target_sum)

    # Initial score (maximal convolution entry)
    best_seq = f.copy()
    best_score = np.max(np.convolve(f, f))

    # Mixing factor – a modest step helps stability
    t = 0.25
    max_iters = 60                     # hard cap per run
    it = 0

    while time.time() < global_deadline and it < max_iters:
        # current maximal autocorrelation value
        conv_f = np.convolve(f, f)
        M = np.max(conv_f)

        # LP to obtain a promising direction
        g = _lp_step(f, M, target_sum)
        if g is None:
            break

        # Convex combination and re‑project onto the simplex
        f_new = (1.0 - t) * f + t * g
        f_new = _project_onto_simplex(f_new, target_sum)

        # Evaluate the new candidate (no need to call evaluate_sequence –
        # the sum is already fixed, so the maximal convolution entry is the true score)
        new_score = np.max(np.convolve(f_new, f_new))

        if new_score < best_score:
            best_score = new_score
            best_seq = f_new.copy()
            print(f"[{time.time():.2f}]   New local best = {best_score:.6f} (n={n}, iter={it})")

        f = f_new
        it += 1

        # Occasionally jitter the current point to leave flat plateaus
        if np.random.rand() < 0.03:
            noise = np.random.randn(n) * (target_sum * 0.01 / n)
            f = _project_onto_simplex(np.maximum(f + noise, 0.0), target_sum)
            # after a shake‑up shrink the step a little
            t = max(0.1, t * 0.9)

    return best_seq, best_score


# --------------------------------------------------------------------

# --------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """

    The routine performs many random restarts, each consisting of a short
    alternating‑LP optimisation (the method described above).  The best
    sequence discovered before the time budget expires is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a small safety margin

    # ----------------------------------------------------------------
    # Main search loop – create independent restarts until time runs out
    # ----------------------------------------------------------------
    while time.time() < deadline:
        # Random length; allow a few very large runs if time permits
        if np.random.rand() < 0.2 and (deadline - time.time()) > 200:

            else:

        else:

            # Early exit if we pass the target threshold

    # ----------------------------------------------------------------

    # The evaluation function expects a plain Python list
