"""
propose_candidate
=================

Generate a non‑negative step function whose self‑convolution has a small
maximum.  Instead of the classical arcsine coefficients we use a simple
power‑law shape

    a_i ∝ (i+1)^{-p} ,

which for exponents p>½ yields a strictly smaller theoretical bound for the
evaluation function

    score = 2·n·max(conv(a,a)) / (sum(a)^2).

The constant limiting value of the score is

    f(p) = 2·(1-p)^2·B(1-p,1-p)      (B is the beta function),

and it decreases monotonically for p∈(½,1).  By scanning a few exponent
values we pick the one giving the lowest score for a moderate length
(≈ 3000).  The chosen exponent is then used to build a longer sequence
(up to the remaining time budget) and a short hill‑climbing refinement
is performed.

The whole routine respects the 1000 s budget, prints intermediate scores,
and finally returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: power‑law sequence a_i = (i+1)^{-p}
# ----------------------------------------------------------------------
def _power_law_sequence(n: int, p: float) -> np.ndarray:
    """Return the first n terms of (i+1)^{-p} as a numpy array."""
    idx = np.arange(1, n + 1, dtype=float)          # i+1, i=0…n-1
    seq = idx ** (-p)
    return seq / seq.sum()                          # normalise to sum = 1


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(budget_s: float = 1000.0,
                     seed: int = 42,
                     **kwargs):
    """
    Produce a candidate sequence of non‑negative heights.

    Parameters
    ----------
    budget_s : float, optional
        Total time budget (seconds).
    seed : int, optional
        Random seed for the stochastic refinement.
    **kwargs
        Ignored – kept for API compatibility.

    Returns
    -------
    list[float]
        Normalised sequence that (empirically) minimises the evaluation.
    """
    np.random.seed(seed)
    start = time.time()

    # --------------------------------------------------------------
    # 1. Quick sweep on a modest length to decide the exponent p.
    # --------------------------------------------------------------
    n_sweep = 3000                     # cheap enough for many trials
    p_grid = np.linspace(0.55, 0.90, 8)   # 0.55, 0.60, …, 0.90
    best_p = 0.6
    best_score = float("inf")

    for p in p_grid:
        if time.time() - start > budget_s * 0.25:
            break
        seq = _power_law_sequence(n_sweep, p)
        sc = evaluate_sequence(seq.tolist())        # pylint: disable=undefined-variable
        if sc < best_score:
            best_score, best_p = sc, p
        # log progress
        print(f"[sweep] p={p:.3f} → score={sc:.6f}")

    # --------------------------------------------------------------
    # 2. Build the final (larger) sequence using the selected exponent.
    #    Choose the length so that we still have time for a light refinement.
    # --------------------------------------------------------------
    time_used = time.time() - start
    remaining = budget_s - time_used
    # reserve ~30 % of the remaining budget for hill‑climbing
    climb_budget = remaining * 0.30
    # pick a length that fits the remaining time (convolution is roughly O(n log n))
    # a safe default is 15000; increase it if we have plenty of time.
    n_final = 15000
    if remaining > 600:                     # >10 min left → be generous
        n_final = 30000

    seq_arr = _power_law_sequence(n_final, best_p)

    # --------------------------------------------------------------
    # 3. Light stochastic refinement (hill‑climbing).
    #    Move a small fraction of mass from one position to another and keep

    # --------------------------------------------------------------
    try:
