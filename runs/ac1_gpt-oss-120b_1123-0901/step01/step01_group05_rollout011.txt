"""
Propose a candidate sequence using a simple LP‑based “active‑set’’ improvement.

The idea is the same as the method described in the statement:
1. Start from the classical inverse‑square‑root sequence
   a_i = C(2i,i)/4**i (exactly constant convolution = 1).
2. Scale it so that the total sum = sqrt(2*n); then the score equals the
   maximal convolution value M (initially ≈ 1.57).
3. Solve a linear program
        maximise   sum(b)
        s.t.       b >= 0
                   (a * b)[k] <= M   for every k
   The convolution (a*b) is linear in the variable b, therefore this is an
   LP.  Because a is very small near the end of the vector, the LP can push
   a lot of mass into those tail entries without violating the constraints,
   increasing the total sum while keeping the peak M unchanged.
4. Normalise the solution b to the same total sum and mix it with the old
   vector (convex combination).  The mixing strictly reduces the maximal
   convolution, i.e. the score.
5. Repeat the LP‑mixing steps while time permits, then finish with a cheap
   random‑local‑search refinement.

If cvxpy is unavailable the algorithm falls back to the original random
multiplicative perturbation approach.

The whole routine respects the 1000 s budget, uses only standard Python,
NumPy and (optionally) CVXPY, and returns a list of non‑negative floats.
"""

    seq = [1.0]                     # a_0 = 1

        seq.append(seq[-1] * (2 * i - 1) / (2 * i))
    return seq

        best_score = float('inf')
    best_seq = f.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Try to import cvxpy – the LP engine.
    # --------------------------------------------------------------
    try:
        import cvxpy as cp
        cvxpy_available = True
    except Exception as exc:
        cvxpy_available = False
        print("[info] cvxpy not available, falling back to cheap random search.")

    # --------------------------------------------------------------
    # 4. Main optimisation loop (LP + mixing).
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # keep a small safety margin
    mixing = 0.5      # convex‑mixing parameter (0<t<1)
    eps = 1e-8
    iteration = 0

    while time.time() < deadline:
        # current maximal convolution M (the score for a normalized vector)
        conv_f = np.convolve(f, f)
        M = float(conv_f.max())

        # ------------------------------------------------------------------
        # 4.1 LP: maximise total mass while keeping the convolution with the
        #       current f bounded by M.
        # ------------------------------------------------------------------
        if cvxpy_available:
            try:
                # variable b (non‑negative)
                b = cp.Variable(n, nonneg=True)

                # convolution expression (linear in b because f is a constant)
                conv_expr = cp.conv(cp.Constant(f), b)   # length 2n‑1

                # linear constraints: every entry ≤ M (+ tiny eps)
                constraints = [conv_expr <= M + eps]

                # objective: maximise sum(b)
                prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

                # solve – ECOS works well for dense LPs of this size
                prob.solve(solver='ECOS', warm_start=True, verbose=False,
                           max_iters=5000)

                if prob.status not in ("optimal", "optimal_inaccurate"):
                    raise RuntimeError(f"LP status {prob.status}")

                b_val = b.value
                if b_val is None:
                    raise RuntimeError("LP returned None for b")
                b_val = np.maximum(b_val, 0.0)   # safeguard against tiny negatives

                # --------------------------------------------------------------
                # 4.2 Normalise the LP solution to the target sum.
                # --------------------------------------------------------------
                sum_b = b_val.sum()
                if sum_b <= 0:
                    raise RuntimeError("LP produced zero mass")

                g = b_val * (target_sum / sum_b)   # same total sum as f

                # --------------------------------------------------------------
                # 4.3 Mix the new candidate with the current vector.
                # --------------------------------------------------------------
                f_new = (1.0 - mixing) * f + mixing * g
                # re‑normalise to eliminate numerical drift
                f_new = f_new * (target_sum / f_new.sum())
                # clip to the limits enforced by evaluate_sequence (optional safety)
                f_new = np.clip(f_new, 0.0, 1000.0)

                # --------------------------------------------------------------
                # 4.4 Evaluate the candidate.  If it improves, keep it.
                # --------------------------------------------------------------
                cand_score = evaluate_sequence(f_new.tolist())
                if cand_score < best_score - 1e-9:
                    best_score = cand_score
                    best_seq = f_new.copy()
                    f = f_new.copy()
                    print(f"[improve] iter {iteration}  score={best_score:.6f}  M={M:.6f}")
                else:
                    # Even if not better we still move to the mixed point – it may
                    # open new directions for the next LP.
                    f = f_new.copy()

                iteration += 1
                continue  # next iteration
            except Exception as e:
                # Any failure in the LP path – switch to simple local search.
                print(f"[lp error] iteration {iteration}: {e}")
                cvxpy_available = False   # disable further LP attempts

        # ------------------------------------------------------------------
        # 4.5 Simple random local search (fallback or after LP failure).
        # ------------------------------------------------------------------
        i = rng.integers(n)
        old_val = f[i]
        factor = np.exp(rng.normal(0.0, 0.02))
        new_val = max(0.0, old_val * factor)
        # keep total sum unchanged by a global renormalisation
        f_candidate = f.copy()
        f_candidate[i] = new_val
        f_candidate = f_candidate * (target_sum / f_candidate.sum())
        cand_score = evaluate_sequence(f_candidate.tolist())
        if cand_score < best_score - 1e-9:
            best_score = cand_score
            best_seq = f_candidate.copy()
            f = f_candidate.copy()
            print(f"[ls improve] iter {iteration}  score={best_score:.6f}")
        iteration += 1

    # --------------------------------------------------------------
    # 5. Final cheap random refinement (exploits any remaining time).
    # --------------------------------------------------------------
    final_deadline = start_time + budget_s - 0.1
    while time.time() < final_deadline:
        i = rng.integers(n)
        old_val = best_seq[i]
        factor = np.exp(rng.normal(0.0, 0.02))
        new_val = max(0.0, old_val * factor)
        cand = best_seq.copy()
        cand[i] = new_val
        cand = cand * (target_sum / cand.sum())
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score - 1e-9:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[final ls] improved to {best_score:.6f}")

    print(f"[final] best score={best_score:.6f}")
    # ensure we return Python list of floats
    return best_seq.tolist()
