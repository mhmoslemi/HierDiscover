"""
Gradient‑descent on the peak of the autocorrelation.

The evaluation metric is 2·n·max(a∗a)/(∑a)².
Because it is scale‑invariant we keep the sequence on the probability simplex
(∑a = 1, a_i ≥ 0).  Minimising the score then amounts to minimising the
maximum entry of the convolution.

Algorithm
---------
1. Initialise with the classical inverse‑square‑root shape
   a_i = C(2i,i)/4**i (the “binom” sequence).  If a previous best
   `height_sequence_1` is available we keep the better of the two.
2. Repeatedly:
   * Compute the full convolution c = a∗a and locate the index k where
     c is maximal.
   * The gradient of c_k w.r.t. a is (up to a factor 2) the reversed slice
     a_{k‑i}.  We take g_i = a_{k‑i} for all valid i.
   * Propose a new point a' = a – η·g, clip negatives to 0 and renormalise
     to sum‑1 (simple projection onto the simplex).
   * Accept the move if the new maximal convolution value decreases;
     otherwise reject and shrink the step size η.
   * Occasionally (≈1 % of iterations) apply a random multiplicative
     perturbation to a single entry to escape shallow plateaux.
3. After the main loop we perform a cheap random‑local‑search phase
   (multiplicative perturbations of single entries) for the remaining time.
4. Return the best sequence found (still on the simplex, i.e. sum≈1).

The whole search fits comfortably into the 1000 s budget (≈10 k convolution
evaluations for n≈3400).  The final sequence respects all security checks
performed by `evaluate_sequence`.
"""

import time
import numpy as np

def generate_inverse_sqrt_sequence(n: int) -> np.ndarray:
    """Return a NumPy array a_0…a_{n‑1} with a_i = binom(2i,i)/4**i.
    The recurrence a_{i+1}=a_i·(2i+1)/(2i+2) avoids large intermediate numbers."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project non‑negative vector v onto the probability simplex (sum = 1)."""
    # Classic algorithm of Duchi et al.
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero -> return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise exactly
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(v, 1.0 / n)
    else:
        w /= w_sum
    return w

    deadline_total = start_time + budget_s - 0.2   # tiny safety margin

    # ----------------------------------------------------------------------

    n = None

    # ----------------------------------------------------------------------

    # ----------------------------------------------------------------------
    seq0 = generate_inverse_sqrt_sequence(n)
    seq0 = seq0.astype(np.float64)
    seq0 /= seq0.sum()                     # place on the simplex

    best_seq = seq0.copy()
    # evaluate using the provided evaluator (adds security checks)
