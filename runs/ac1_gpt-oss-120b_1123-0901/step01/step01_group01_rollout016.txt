The algorithm consists of three phases:

# Helper: projection of a vector onto the simplex {x >= 0, sum(x)=1}
def _project_onto_simplex(v):
    """Project a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                   # all entries are non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]

        new_seq = _project_onto_simplex(new_seq)

        # (f) evaluate the true objective (cheap compared to the FFTs)
        score = evaluate_sequence(new_seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            last_improve = iter_no
            print(f"[Iter {iter_no:06d}] new best = {best_score:.6f}")

        # (g) adaptive schedule
        if iter_no - last_improve > 8000:          # stagnation → shrink step
            step *= 0.5
            last_improve = iter_no
            print(f"[Iter {iter_no:06d}] reducing step to {step:.6f}")

        # increase α slowly to tighten the approximation of the max
        if (iter_no % 2500) == 0 and alpha < 20000:
            alpha = min(alpha * 1.3, 20000)
            print(f"[Iter {iter_no:06d}] raising alpha to {alpha:.0f}")

        seq = new_seq
        iter_no += 1

    print(f"[Done] best objective = {best_score:.6f} after {iter_no} iterations.")
    return best_seq.tolist()
