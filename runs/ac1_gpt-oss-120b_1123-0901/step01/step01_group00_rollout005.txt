def _lp_improve(a, t, target_sum):
    n = a.shape[0]
    conv = np.convolve(a, a)
    M = conv.max()
    # Build constraint matrix
    N = 2 * n - 1
    A = np.zeros((N, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a
    # right side
    b_vec = np.full(N, M, dtype=np.float64)
    # Linear programme: maximise sum b_i => minimise -sum b_i
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, 1000.0) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
    if not res.success:
        return None, float('inf')
    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        return None, float('inf')
    scale = target_sum / sum_b
    g = b * scale
    # mix
    new_a = (1 - t) * a + t * g
    # Ensure feasibility (nonnegative, sum)
    new_a = np.maximum(new_a, 0.0)
    # small projection to enforce exact sum (to counter numerical error)
    new_a = _project_onto_simplex(new_a, target_sum)
    return new_a, M

def _grad_step(a, target_sum, step_scale):
    n = a.shape[0]
    conv = np.convolve(a, a)
    k_max = int(np.argmax(conv))
    # compute gradient of this entry
    low = max(0, k_max - (n - 1))
    high = min(n - 1, k_max)
    idx = np.arange(low, high + 1)
    partner = k_max - idx
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[partner]
    max_grad = grad.max()
    if max_grad <= 0:
        eta = 0.0
    else:
        eta = step_scale * conv[k_max] / max_grad
    a_new = a - eta * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = _project_onto_simplex(a_new, target_sum)
    return a_new

1.  **LP‑driven expansion** –  given a current vector `a` we compute the
    maximal autocorrelation value `M = max(conv(a,a))`.  We then solve the
    linear programme

        maximise   Σ b_i
        subject to a * b ≤ M               (convolution inequality)
                   b_i ≥ 0 , b_i ≤ 1000

    The optimal `b` can be scaled to the simplex Σ b_i = √(2 n) and mixed
    with `a`.  This step provably reduces the maximal convolution entry
    and therefore the evaluation value.

2.  **Projected gradient descent on the active index** – after a few LP
    updates we perform a few short gradient‑descent steps (the same
    idea used in the previous baseline) to fine‑tune the vector.

Several random restarts with different lengths ``n`` are performed within
the allotted time budget.  The best sequence discovered is returned.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex { x_i ≥ 0 , Σ x_i = s }.
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex of radius ``s``."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    n = v.shape[0]
    # algorithm of Wang & Carreira‑Perpiñán (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Build the Toeplitz‑like constraint matrix for (a * b)[k] ≤ M.
# ----------------------------------------------------------------------
def _build_constraint_matrix(a: np.ndarray) -> np.ndarray:
    """Return a matrix A (shape (2n‑1, n)) such that A @ b = a * b."""
    n = a.shape[0]
    N = 2 * n - 1
    A = np.zeros((N, n), dtype=np.float64)
    # column j contributes a to rows j … j+n‑1
    for j in range(n):
        A[j : j + n, j] = a
    return A


# ----------------------------------------------------------------------
# One LP‑driven improvement step.
# ----------------------------------------------------------------------
def _lp_improve(a: np.ndarray,
                target_sum: float,
                mixing: float = 0.2):
    """
    Perform a single LP‑based improvement and return the updated vector.
    Returns ``a`` unchanged if the LP fails.
    """
    from scipy.optimize import linprog

    n = a.shape[0]
    # current maximal convolution entry
    conv = np.convolve(a, a)
    M = conv.max()

    # build constraints a * b ≤ M
    A = _build_constraint_matrix(a)
    b_ub = np.full(A.shape[0], M, dtype=np.float64)

    # maximise Σ b_i  →  minimise -Σ b_i
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, 1000.0) for _ in range(n)]

    # solve LP
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return a  # keep the current point

    b = np.maximum(res.x, 0.0)          # safety, eliminate tiny negatives
    sum_b = b.sum()
    if sum_b <= 0:
        return a

    # scale to the simplex Σ a_i = target_sum
    scale = target_sum / sum_b
    g = b * scale

    # convex combination with a small mixing factor
    new_a = (1.0 - mixing) * a + mixing * g
    # keep feasibility (numerical drift)
    new_a = np.maximum(new_a, 0.0)
    new_a = _project_onto_simplex(new_a, target_sum)
    return new_a


# ----------------------------------------------------------------------
# A few projected‑gradient steps on the currently maximal convolution entry.
# ----------------------------------------------------------------------
def _gradient_step(a: np.ndarray,
                   target_sum: float,
                   base_step: float = 0.05) -> np.ndarray:
    """One gradient‑descent step on the index attaining the max convolution."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    k_max = int(np.argmax(conv))
    max_val = conv[k_max]

    # indices that actually contribute to conv[k_max]
    low = max(0, k_max - (n - 1))
    high = min(n - 1, k_max)
    idx = np.arange(low, high + 1, dtype=int)
    partner = k_max - idx
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[partner]

    # adaptive step size (avoid overshoot)
    max_grad = grad.max()
    eta = base_step * max_val / max_grad if max_grad > 0 else 0.0
    a_new = a - eta * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = _project_onto_simplex(a_new, target_sum)
    return a_new


# ----------------------------------------------------------------------
# Main entry point required by the challenge framework.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    evaluation score.  The routine runs for at most ``budget_s`` seconds
    and keeps the best sequence discovered.
    """
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate without calling the heavyweight
    # ``evaluate_sequence`` (the objective is scale‑invariant and we keep the
    # sum fixed to √(2 n), therefore the score equals max(conv)).
    # ------------------------------------------------------------------
    def _score_of(a_vec: np.ndarray) -> float:
        conv = np.convolve(a_vec, a_vec)
        return float(conv.max())

    # ------------------------------------------------------------------
    # Main restart loop.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Pick a problem size.  Larger ``n`` gives more freedom but makes
        # each LP solve more expensive – we stay in the 300‑1500 range.
        n = np.random.randint(300, 1501)

        target_sum = math.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Initialise a starting point.
        # ------------------------------------------------------------------
        if 'height_sequence_1' in globals() and np.random.rand() < 0.4:
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                a0 = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-3
                a0 = np.concatenate([src, pad])
        else:
            # Dirichlet‑like draw with concentration < 1 (spiky vectors)
            alpha = 0.4
            g = np.random.gamma(alpha, 1.0, size=n)
            a0 = g
        # enforce positivity and project onto the simplex
        a0 = np.maximum(a0, 1e-8)
        a0 = _project_onto_simplex(a0, target_sum)

        a = a0.copy()

        # ------------------------------------------------------------------
        # Alternate LP updates and a few gradient steps.
        # ------------------------------------------------------------------
        lp_iterations = np.random.randint(8, 15)      # cheap but enough
        for it in range(lp_iterations):
            if time.time() >= deadline:
                break

            # LP‑driven improvement
            a = _lp_improve(a, target_sum, mixing=0.25)

            # occasional shake‑up to escape flat regions
            if np.random.rand() < 0.03:
                noise = np.random.randn(n) * (0.02 * target_sum / n)
                a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)

            # evaluate and keep global best
            cur_score = _score_of(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] New best = {best_score:.6f} (n={n}, it={it})")

            # a few lightweight gradient steps for fine‑tuning
            for _ in range(3):
                a = _gradient_step(a, target_sum, base_step=0.04)
                cur_score = _score_of(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time():.2f}] New best (grad) = {best_score:.6f} (n={n})")

        # early stop if we already beat the target
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback: a safe uniform sequence if nothing was found.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    return best_seq.tolist()
