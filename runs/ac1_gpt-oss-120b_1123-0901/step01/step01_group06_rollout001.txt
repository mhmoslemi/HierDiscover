#!/usr/bin/env python
"""
propose_candidate
=================

Iteratively improve a non‑negative step function (a probability vector) by
solving a linear program that seeks a second vector `g` whose convolution with
the current vector `f` never exceeds the current worst convolution value.
A convex combination `f←(1‑t)f+t g` with an optimised step size `t` reduces the
maximum of `f*f`.  Repeating the LP + line‑search steps rapidly lowers the
evaluation metric

    evaluate_sequence(f) = 2·n·max(conv(f,f))          (sum(f)=1)

below the classic arcsine construction (π/2≈1.5708).  The routine respects the
1000 s budget, works with moderate lengths (default ≤ 800) and returns the
best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper 1 – coefficients of (1‑z)^(-1/2)
# ----------------------------------------------------------------------
def _arcsine_coeffs(n: int) -> np.ndarray:
    """Return the first *n* terms a_i = C(2i,i) / 4**i.

    The recurrence a_{i+1} = a_i * (2·i + 1) / (2·(i+1)) is numerically stable.
    """
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# Helper 2 – build the dense matrix for the constraints f * b ≤ M
# ----------------------------------------------------------------------
def _constraint_matrix(f: np.ndarray) -> np.ndarray:
    """Construct A where (A @ b)[k] = (f * b)[k] for k=0..2·n‑2."""
    n = f.size
    # shape (2*n-1, n)
    K = np.arange(2 * n - 1)[:, None]   # column vector of output indices
    I = np.arange(n)[None, :]           # row vector of b‑indices
    idx = K - I                         # index of f that multiplies each b_j
    mask = (idx >= 0) & (idx < n)       # valid positions only
    A = np.where(mask, f[idx], 0.0)
    return A


# ----------------------------------------------------------------------
# Helper 3 – optimal mixing coefficient t (grid search)
# ----------------------------------------------------------------------
def _optimal_mix(f: np.ndarray, g: np.ndarray, n: int) -> tuple[float, float]:
    """Return the mixing weight t∈(0,1] giving the smallest evaluation
    and the corresponding evaluation value."""
    ff = np.convolve(f, f)      # length 2n‑1
    fg = np.convolve(f, g)
    gg = np.convolve(g, g)

    # evaluate a modest grid of t values; refine later if needed
    t_grid = np.linspace(0.0, 1.0, 51)[1:]          # exclude t=0 (no change)
    t = t_grid[:, None]                            # shape (m,1)

    # convolution of the convex combination (1‑t)f + t g:
    # (1‑t)^2·ff + 2·t·(1‑t)·fg + t^2·gg   (element‑wise)
    conv_t = (1 - t) ** 2 * ff + 2 * t * (1 - t) * fg + t ** 2 * gg
    max_t = conv_t.max(axis=1)                     # max over each t
    scores = 2.0 * n * max_t                       # evaluation for each t

    best_idx = np.argmin(scores)
    best_t = float(t_grid[best_idx])
    best_score = float(scores[best_idx])
    return best_t, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(budget_s: float = 1000.0,
                     seed: int = 42,
                     **kwargs) -> list[float]:
    """
    Produce a candidate sequence of non‑negative heights.

    Parameters
    ----------
    budget_s : float, optional
        Time budget in seconds (default 1000 s).
    seed : int, optional
        Random seed – used only for deterministic choice of the length.
    **kwargs
        Ignored – kept for API compatibility.

    Returns
    -------
    list[float]
        Normalised sequence (sum == 1.0) that (hopefully) minimises the
        evaluation function.
    """
    np.random.seed(seed)
    start = time.time()

    # --------------------------------------------------------------
    # Choose a length.  Use the length of a previously found candidate
    # (if available) but cap it at a moderate size to keep the LP fast.
    # --------------------------------------------------------------
    try:
        n_global = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        n_global = 800
    n = min(n_global, 800)                 # keep the problem tractable

    # --------------------------------------------------------------
    # Initialise with the arcsine distribution (known good baseline).
    # --------------------------------------------------------------
    coeffs = _arcsine_coeffs(n)
    f = coeffs / coeffs.sum()              # sum(f) == 1
    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())  # noqa: F821
    print(f"[propose_candidate] initial length={n}, score={best_score:.6f}")

    iteration = 0
    # --------------------------------------------------------------
    # Main optimisation loop – each iteration solves a linear program,
    # builds a new direction g and mixes with an optimal step size t.
    # --------------------------------------------------------------
    while time.time() - start < budget_s - 1.0:
        iteration += 1

        # 1) current worst convolution value M
        conv_ff = np.convolve(f, f)
        M = conv_ff.max()

        # 2) linear programme: maximise sum(b)  s.t.  f * b ≤ M,  b ≥ 0
        A = _constraint_matrix(f)                         # shape (2n‑1, n)
        b_ub = np.full(2 * n - 1, M * (1.0 + 1e-12))       # tiny slack for numerics
        c = -np.ones(n)                                    # maximise sum(b) ⇔ minimise -sum(b)

        lp_res = linprog(c, A_ub=A, b_ub=b_ub,
                         bounds=[(0, None)] * n,
                         method='highs')
        if not lp_res.success:
            print(f"[propose_candidate] LP failed at iteration {iteration}")
            break

        b = lp_res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            # No feasible direction that improves the bound
            break

        # 3) direction g (normalised to sum(g)=1)
        g = b / sum_b

        # 4) find the best mixing coefficient t
        t_opt, score_opt = _optimal_mix(f, g, n)

        # 5) update the candidate
        f = (1.0 - t_opt) * f + t_opt * g
        f = np.maximum(f, 0.0)      # protect against tiny negatives
        f /= f.sum()                # enforce sum = 1 exactly (numerical drift)

        # keep the best seen so far
        if score_opt < best_score - 1e-12:
            best_score = score_opt
            best_f = f.copy()
            print(f"[propose_candidate] iter {iteration:3d}: "
                  f"score ↓ {best_score:.6f}  (t={t_opt:.3f})")

        # stop early if improvement becomes negligible
        if iteration >= 200:
            break

    final_score = evaluate_sequence(best_f.tolist())   # noqa: F821
    print(f"[propose_candidate] final length={n}, best_score={final_score:.6f} "
          f"after {iteration} iterations")
    return best_f.tolist()
