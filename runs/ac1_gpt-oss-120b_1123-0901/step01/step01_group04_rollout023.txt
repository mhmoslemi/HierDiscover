def proj_simplex(v):
    # Implementation using algorithm (Michelot 1986).
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def convolve_fft(x, y):
    m = len(x) + len(y) - 1
    nfft = 1 << ((m - 1).bit_length())
    X = np.fft.rfft(x, n=nfft)

    conv = np.fft.irfft(X*Y, n=nfft)[:m]
    return conv

"""Propose a candidate step‑function.

The algorithm starts from the discrete arcsine distribution (the
asymptotically optimal shape) and then performs a mixed optimisation:

* **Projected gradient descent** on a smooth approximation of the
  L∞‑norm of the autocorrelation.  The objective is
  (1/β)·log∑exp(β·conv) – the soft‑max of the convolution.  Gradients
  are obtained analytically and the iterate is projected onto the
  probability simplex (non‑negative entries summing to 1).

* **Targeted hill‑climbing** steps that move a small amount of mass from
  indices that contribute most to the currently maximal convolution entry
  toward indices that contribute the least.  Moves are kept only if they
  improve the true evaluation function.

Both phases are interleaved for a few‑seconds budget, keeping the best
sequence found.  The code works with any length (defaults to the length
of the global ``height_sequence_1`` when present) and respects the
security checks of ``evaluate_sequence``.
"""

import time
import numpy as np

# -----------------------------------------------------------
# Helper functions – defined at top level as required
# -----------------------------------------------------------

def convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution using an FFT of length a power of two.
    """
    m = len(x) + len(y) - 1
    nfft = 1 << ((m - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=nfft)
    fy = np.fft.rfft(y, n=nfft)
    conv = np.fft.irfft(fx * fy, n=nfft)[:m]
    return conv


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {w | w_i >= 0, sum w_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                # sort in descending order
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def generate_arcsine(n: int) -> np.ndarray:
    """
    Produce the un‑normalised arcsine coefficients
        a_k = (2k choose k) / 4**k ,   k = 0,…,n‑1
    and normalise them so that the sum equals 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    a /= a.sum()
    return a


def softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Return the soft‑max weights for a given convolution vector.
    The maximal entry receives weight 1, the others are scaled by
    exp(beta·(conv_i – max(conv))).
    """
    max_val = conv.max()
    # shift to avoid overflow, max weight becomes 1.0
    w_raw = np.exp(beta * (conv - max_val))
    w = w_raw / w_raw.sum()
    return w


def gradient_step(a: np.ndarray, beta: float, step: float) -> np.ndarray:
    """
    Perform one projected gradient descent step on the soft‑max
    approximation of the L∞‑norm.
    """
    # 1. smooth weights
    conv = convolve_fft(a, a)
    w = softmax_weights(conv, beta)

    # 2. gradient of the smooth objective:
    #    grad_i = 2 * Σ_k a_k * w_{i+k}
    #    which can be obtained by a convolution with the reversed weight vector.
    w_rev = w[::-1]
    full = convolve_fft(a, w_rev)                 # length = 3n‑2
    n = len(a)
    # slice and reverse to obtain the correct ordering
    grad = 2.0 * full[n - 1:2 * n - 1][::-1]

    # 3. explicit gradient step followed by projection onto the simplex
    a_new = a - step * grad
    a_proj = project_simplex(a_new)
    return a_proj


def targeted_hill_climb(a: np.ndarray, conv: np.ndarray, n: int) -> np.ndarray:
    """
    Try to move a tiny amount of mass from indices that contribute
    heavily to the current maximal convolution entry to those that
    contribute very little.  The move is kept only if the true score
    improves.
    """
    s_max = int(conv.argmax())
    idx = np.arange(n)
    partner = s_max - idx
    mask = (partner >= 0) & (partner < n)
    contrib = np.zeros_like(a)
    contrib[mask] = a[mask] * a[partner[mask]]

    # pick a few promising source / destination candidates
    top_k = 5
    bot_k = 5
    src_cand = np.argpartition(-contrib, top_k - 1)[:top_k]
    dst_cand = np.argpartition(contrib, bot_k - 1)[:bot_k]

    # try all pairs (src,dst) – stop at the first improvement
    for src in src_cand:
        for dst in dst_cand:
            if src == dst:
                continue
            # move a small fraction of the source mass
            delta = 0.01 * a[src]
            if delta <= 0.0:
                continue
            a_tmp = a.copy()
            a_tmp[src] -= delta
            a_tmp[dst] += delta
            conv_tmp = convolve_fft(a_tmp, a_tmp)
            new_score = 2.0 * n * conv_tmp.max()
            cur_score = 2.0 * n * conv.max()
            if new_score < cur_score:
                return a_tmp
    # no improvement – return the original vector
    return a

# -----------------------------------------------------------

    Produce a list of non‑negative heights that (hopefully) beats the
    previous best evaluation value.  The function respects the total
    time budget and returns the best sequence found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Determine the length of the step function
    # ------------------------------------------------------------------
    n = kwargs.get("length")
    if n is None:
        try:
            n = len(height_sequence_1)   # type: ignore[name-defined]
        except Exception:
            n = 6027                      # fallback (the previous run)

    # ------------------------------------------------------------------
    # 2. Initialise with the (discrete) arcsine distribution
    # ------------------------------------------------------------------
    a = generate_arcsine(n)

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Optimisation loop – mixture of gradient descent and hill climbing
    # ------------------------------------------------------------------
    it = 0
    beta = 5.0                # start of the soft‑max smoothing
    while time.time() < deadline:
        # current convolution and true score
        conv = convolve_fft(a, a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[new best] it={it:06d}, score={best_score:.6f}")

        # Alternate between a gradient step and a targeted move
        if it % 20 == 0:                     # every 20 iterations: gradient descent
            # gradually tighten the soft‑max approximation
            beta = min(2000.0, beta * 1.03)
            # step size decays with sqrt(iteration) – modest but safe
            step = 0.5 / np.sqrt(it + 1.0)
            a = gradient_step(a, beta, step)
        else:

            a = targeted_hill_climb(a, conv, n)

        if it % 500 == 0 and it > 0:

            a = project_simplex(a + jitter)

        it += 1

    print(f"[final] iterations={it}, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Return plain Python list as required by the evaluator
    # ------------------------------------------------------------------
    return best_seq.tolist()
