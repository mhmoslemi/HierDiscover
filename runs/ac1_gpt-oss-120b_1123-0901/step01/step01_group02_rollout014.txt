"""
Propose a candidate step‑function by continuous optimisation on the simplex.
The algorithm works in the “probability” domain p_i = a_i / Σa_i (Σa_i
is fixed to √(2n)).  The objective is 2 n · max(p∗p), i.e. minimise the
maximum entry of the self‑convolution of p.

We use a smooth approximation of the max:
    w_k = exp(α·b_k) / Σ_j exp(α·b_j)   (b = p∗p)
so the gradient becomes
    ∇ = 2·(w ⋆ p)                     (⋆ = cross‑correlation)

The gradient is obtained efficiently by FFTs.  An Adam‑style adaptive
step is applied, afterwards the vector is projected onto the simplex by
clipping negatives and renormalising.  The smoothing parameter α is
increased slowly (continuation) and occasional Gaussian jitter is added
to escape plateaus.  The best sequence seen is recorded and returned
when the time budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: next power of two (used for zero‑padding in FFT)
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# Helper: full convolution via FFT (real inputs)
def _conv_fft(a: np.ndarray, b: np.ndarray, size: int) -> np.ndarray:
    """Return the linear convolution a*b (real, full) using FFT of given size."""
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    C = A * B
    conv = np.fft.irfft(C, size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Helper: cross‑correlation w ⋆ p (linear, full) via FFT
def _corr_fft(p: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """Return the linear cross‑correlation (w ⋆ p) for lags 0..len(p)-1."""
    # FFT of p and w
    P = np.fft.rfft(p, size)
    W = np.fft.rfft(w, size)
    # correlation = IFFT(conj(P) * W)
    corr_full = np.fft.irfft(np.conj(P) * W, size)
    # offset where lag 0 starts
    offset = w.size - 1
    return corr_full[offset : offset + p.size]


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Wall‑clock time budget (seconds).  The function stops a few seconds
        before the limit to avoid termination.
    kwargs : dict
        Unused, kept for compatibility.

    Returns
    -------
    list[float]
        The best sequence of heights found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin so that the harness does not cut us off while printing
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Initialise a starting sequence (use the global SOTA if available)
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        # fallback: uniform vector of length 2100
        base = np.ones(2100, dtype=float)

    n = base.shape[0]
    target_sum = math.sqrt(2.0 * n)          # the normalisation used in the paper

    # Work with the probability vector p = a / target_sum, so Σ p = 1
    p = base.astype(float)
    p = p / p.sum()                          # ensure Σp = 1

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.15                                # base learning rate for Adam

    # Smoothing (softmax) parameter α
    alpha = 5.0
    alpha_mul = 1.001                         # multiplicative increase each iteration
    alpha_max = 200.0

    # FFT size large enough for all convolutions/correlations:
    # we need at least n + (2n-1) - 1 = 3n-2 points.
    size_fft = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Helper to compute the current objective from the convolution result
    def _objective_from_conv(conv_full: np.ndarray) -> float:
        """Return 2·n·max(conv) – the exact value that `evaluate_sequence` would give."""
        return float(2.0 * n * conv_full.max())

    # Compute initial objective
    conv_full = _conv_fft(p, p, size_fft)          # length 2n-1
    best_score = _objective_from_conv(conv_full)
    best_p = p.copy()

    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop
    iteration = 0
    beta1_pow = beta1
    beta2_pow = beta2
    eval_interval = max(1, n // 200)      # how often we check for improvement
    jitter_interval = 2000                # add random jitter occasionally
    jitter_scale = 0.005                  # relative jitter size (on the simplex)

    while time.time() < deadline:
        # ---------- convolution p * p ----------
        conv_full = _conv_fft(p, p, size_fft)        # length 2n‑1
        max_conv = conv_full.max()

        # ---------- soft‑max over convolution ----------
        # w_k = exp(α·(b_k - max)) / Σ exp(...)
        shifted = conv_full - max_conv               # stabilise exponentials
        exp_vals = np.exp(alpha * shifted, dtype=np.float64)
        w = exp_vals / exp_vals.sum()                # shape (2n‑1,)

        # ---------- gradient = 2·(w ⋆ p) ----------
        corr = _corr_fft(p, w, size_fft)             # length n, lag 0..n‑1
        grad = 2.0 * corr

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # projected gradient step on the simplex
        p_new = p - step
        p_new = np.maximum(p_new, 0.0)               # non‑negative orthant

        s = p_new.sum()
        if s > eps:
            p_new /= s                                # re‑normalise to Σp = 1
        else:
            # If we collapsed to zero (unlikely), fall back to uniform
            p_new = np.full_like(p, 1.0 / n)

        # occasional jitter to escape shallow plateaus
        if (iteration > 0) and (iteration % jitter_interval == 0):
            jitter = np.random.randn(n) * jitter_scale
            p_new = np.maximum(p_new + jitter, 0.0)
            p_new /= p_new.sum()

        # accept the new point
        p = p_new

        # ---------- bookkeeping ----------
        iteration += 1
        beta1_pow *= beta1
        beta2_pow *= beta2
        alpha = min(alpha_max, alpha * alpha_mul)

        # evaluate (cheap) every `eval_interval` iterations
        if iteration % eval_interval == 0:
            # the convolution computed earlier is still valid for this p
            # (it already corresponds to the current p because we recomputed it)
            cur_score = _objective_from_conv(conv_full)
            if cur_score < best_score:
                best_score = cur_score
                best_p = p.copy()
                print(f"[iter {iteration:06d}] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best absolute heights (scale back with target_sum)
    best_a = best_p * target_sum
    # Ensure we respect the hard limits of `evaluate_sequence`
    best_a = np.clip(best_a, 0.0, 1000.0)

    print(f"[done] iterations={iteration}, best score={best_score:.6f}")
    return best_a.tolist()
