def _softmax_obj_grad(a, lam):
    # a: ndarray shape (n,)
    n = a.shape[0]
    conv = np.convolve(a, a)
    # compute softmax probabilities
    # For stability, subtract max
    cmax = np.max(conv)
    exp_term = np.exp(lam * (conv - cmax))
    sum_exp = np.sum(exp_term)
    probs = exp_term / sum_exp
    # objective approximates (cmax + (1/lam) * np.log(sum_exp))
    L = cmax + np.log(sum_exp) / lam
    # gradient: 2 * correlation of probs with a
    corr = np.correlate(probs, a, mode='full')
    grad = 2.0 * corr[n-1:n-1 + n]   # slice length n
    return L, grad

"""
Search for a step‑function (non‑negative sequence) minimizing
    2 * n * max(conv(a, a)) / (sum(a)**2)

by smooth optimisation of the max using a soft‑max approximation.
The algorithm:

1. Work on the simplex {a_i ≥ 0, Σ a_i = √(2 n)} – the objective is
   scale invariant, so fixing the ℓ₁‑norm removes the trivial scaling.
2. Approximate max(conv) by L_λ = (1/λ)·log Σ exp(λ·conv).  For λ→∞
   L_λ → max(conv).  Its gradient is a simple correlation,
   enabling cheap first‑order updates.
3. Perform projected gradient descent with a back‑tracking line search.
   The step size is scaled with the average entry (√(2 n)/n) and
   gradually reduced as λ grows.
4. Several random restarts with different lengths (200 … 1800) are run
   within the time budget.  The best sequence found (lowest exact
   evaluation) is returned.
"""

import time
import numpy as np

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

Search for a step‑function (non‑negative sequence) minimizing

    2 * n * max(conv(a, a)) / (sum(a)**2)

by smooth optimisation of the max using a soft‑max approximation.
The algorithm:

1. Work on the simplex {a_i ≥ 0, Σ a_i = √(2 n)} – the objective is
   scale‑invariant, so fixing the ℓ₁‑norm removes the trivial scaling.
2. Approximate max(conv) by the smooth function
        L_λ(a) = (1/λ)·log Σ_k exp(λ·conv_k),
   whose gradient is a cheap correlation.
3. Perform projected gradient descent with a back‑tracking line search.
   The step size is scaled with the average entry (√(2n)/n) and
   reduced as λ grows.
4. Run many random restarts with different lengths (≈300 … 1800) inside
   the time budget.  The best sequence (lowest exact evaluation) is
   returned.

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
