def arcsine_sequence(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * ((2*k-1)*(2*k)) / (4.0 * k * k)
    a = a / a.sum()
    return a

def solve_lp(f, M):
    n = len(f)
    # define variable
    g = cp.Variable(n, nonneg=True)
    # constant f
    f_const = cp.Constant(f)
    # convolution f*g
    conv_expr = cp.conv(f_const, g)

"""
propose_candidate
=================

Optimise a non‑negative step function for the metric

    2·n·max(conv(a,a)) / (sum a)^2 .

The algorithm starts from the discrete arcsine distribution (the
theoretically optimal shape for the continuous problem) and repeatedly
does the following:

1. Compute the current maximum `M = max(f∗f)`.
2. Solve the linear programme

       maximise   Σ g_j
       subject to  (f∗g)_k ≤ M   for every k
                   g_j ≥ 0

   The optimum pushes as much mass as possible into indices where `f`
   is small, while keeping the mixed convolution below the current
   bound `M`.
3. Normalise the solution `g` to have the same ℓ₁‑mass as `f` and
   perform a tiny line‑search over convex combinations
   `α·f + (1‑α)·g` (`α∈[0,1]`).  
   The best combination that reduces the objective is kept.
4. If the line‑search fails to improve, a lightweight hill‑climbing
   move (transfer a small amount of mass from a large entry to a
   small one) is attempted.

The loop runs until the allocated time budget expires (≈ 1000 s) or
the target score `1.503` is reached.  The best sequence found is
returned as a plain Python list.

The implementation falls back to pure hill‑climbing when `cvxpy`
is unavailable.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional import of cvxpy – used for the linear programme.
# ----------------------------------------------------------------------
try:
    import cvxpy as cp
    _CVX_AVAILABLE = True
except Exception:                     # pragma: no cover
    _CVX_AVAILABLE = False


def _arcsine_distribution(n: int) -> np.ndarray:
    """Return the normalised arcsine distribution of length ``n``."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * ((2 * k - 1) * (2 * k)) / (4.0 * k * k)
    a /= a.sum()                         # ℓ₁‑normalise → sum == 1
    return a


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise Σ g_j   s.t.   (f∗g)_k ≤ M   (for all k)  and g ≥ 0.
    Returns a non‑negative ndarray ``g`` (not normalised).
    """
    n = f.shape[0]

    # Variable and constant that cvxpy will use
    g = cp.Variable(n, nonneg=True)                 # g_j ≥ 0
    f_const = cp.Constant(f)

    # Affine expression for the convolution f∗g
    # The result has length 2·n‑1.
    conv_expr = cp.conv(f_const, g)

    # Linear inequality constraints (element‑wise)
    constraints = [conv_expr <= M]

    # Linear objective – maximise total mass of g
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # ECOS works well for moderate‑size LPs.
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
               abstol=1e-8, reltol=1e-8, feastol=1e-8, verbose=False)

    if g.value is None:                     # pragma: no cover
        raise RuntimeError("LP solver failed to produce a solution.")
    g_val = np.asarray(g.value, dtype=np.float64)
    # Numerical noise can produce tiny negatives – clip them.
    np.maximum(g_val, 0.0, out=g_val)
    return g_val


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) improves
    upon the arcsine baseline.  The routine respects the global

    remains tractable.
    """
    start_time = time.time()
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1.  Choose a working length.
    # ------------------------------------------------------------------
    n = kwargs.get("length")
    if n is None:
        try:
            n = len(height_sequence_1)          # type: ignore[name-defined]
        except Exception:
            n = 1200                              # a size that LP can handle quickly
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if init.ndim == 1 and init.size == n:
            f = init.copy()
        else:
            raise ValueError
    except Exception:
        f = _arcsine_distribution(n)

    # Normalise to ℓ₁‑mass = 1 (the evaluation function is scale‑invariant).
    f /= f.sum()
    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    print(f"[init] n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Main optimisation loop.
    # ------------------------------------------------------------------
    iteration = 0
    # A modest grid for the line‑search over the mixing coefficient.
    alphas = np.linspace(0.0, 1.0, 21)

    while time.time() - start_time < budget_s - 0.5:

        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------
        g = None

                # Normalise g so it has the same ℓ₁‑mass as f.

            except Exception as e:               # pragma: no cover

        # If the LP could not be used (or failed), fall back to a random vector.
        if g is None:

        # Pre‑compute the three convolutions needed for the line‑search.
        conv_fg = np.convolve(f, g)

        # ------------------------------------------------------------------
        # 3.2  Explore convex combinations α·f + (1‑α)·g.
        # ------------------------------------------------------------------
        improved = False
        best_local_f = None

            combined = (alpha ** 2) * conv_f + ((1 - alpha) ** 2) * conv_g \

        else:

            # ------------------------------------------------------------------
            i, j = np.random.choice(n, size=2, replace=False)
            src, dst = (i, j) if f[i] > f[j] else (j, i)
            delta = 0.01 * min(f[src], f[dst])   # move 1 % of the smaller entry
            if delta > 0:

                    best_score = cur_score

        # ------------------------------------------------------------------

    # ----------------------------------------------------------------------

    print(f"[final] iterations={iteration}, best_score={final_score:.6f}")
    return best_f.tolist()
