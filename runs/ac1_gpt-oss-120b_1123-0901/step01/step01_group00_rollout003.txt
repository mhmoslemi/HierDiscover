def propose_candidate(seed=0,budget_s=1000.0):
    import time, numpy as np, math
    try:
        from scipy.optimize import linprog
    except ImportError:
        linprog = None
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1
    best_seq, best_score = None, float('inf')
    # helper evaluate and compute objective
    def compute_score(a):
        # Ensure non-negative, clip to 1000?
        # Actually evaluate_sequence will clip; but for quick we do own compute.
        n = len(a)
        conv = np.convolve(a, a)
        max_val = conv.max()
        sum_a = a.sum()
        if sum_a < 0.01: return float('inf')
        return 2 * n * max_val / (sum_a**2)
    # param families
    def try_shape(n, shape_func, params):
        for p in params:
            a = shape_func(n, p)
            a = np.maximum(a, 0.0)
            # scale to sum sqrt(2n)
            target = math.sqrt(2.0*n)
            s = target / a.sum() if a.sum()>0 else 0
            a = a * s
            if np.any(np.isnan(a)) or np.any(np.isinf(a)): continue
            score = evaluate_sequence(a.tolist())
            # or compute manually to reduce overhead
            nonlocal best_seq,best_score
            if score < best_score:
                best_score = score; best_seq = a.copy()
                print(f"New best {score:.6f} with shape {shape_func.__name__} param {p} n={n}")
    ```

"""
propose_candidate
-----------------

A multi‑phase black‑box optimiser for the step‑function evaluation

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The routine consists of :

1.  A quick sweep over several low‑dimensional parametric families
    (exponential, linear, quadratic, triangular).  Every candidate is
    normalised onto the simplex Σa_i = √(2 n) and evaluated with the
    provided ``evaluate_sequence``.  The best candidate so far is kept.

2.  A large‑scale sub‑gradient descent (with momentum) that minimises the
    infinity‑norm of the autocorrelation.  Convolutions are performed via
    FFT (≈ O(n log n)) so that thousands of iterations are affordable.  The
    iteration is stopped when the global deadline is reached or a target
    score (≈ 1.503) is achieved.

3.  If SciPy is available, a single linear‑programming refinement
    (the “LP‑step” from the SOTA description) is executed on the current
    best sequence: maximise Σg_j under the constraint (f * g) ≤ max(f * f).
    The resulting g is rescaled to the simplex and blended with f.
    This often pushes the bound a little further.

All helpers are defined at module level – no closures or lambdas – and the
function returns a plain list of non‑negative floats ready for the
evaluation harness.
"""

import math

# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection of a vector onto the simplex with sum `s`."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: autocorrelation via FFT (fast for n up to a few thousand)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the (full) convolution a * a using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Helper: safely evaluate a sequence using the external routine
# ----------------------------------------------------------------------
def _evaluate(seq: list) -> float:
    """Return the score from evaluate_sequence or +inf on error."""
    try:
        return evaluate_sequence(seq)
    except Exception:
        return float("inf")


# ----------------------------------------------------------------------
# Shape factories (non‑negative, not normalised)
# ----------------------------------------------------------------------
def _shape_exp(n: int, alpha: float) -> np.ndarray:
    return np.exp(-alpha * np.arange(n, dtype=float))


def _shape_exp_rev(n: int, alpha: float) -> np.ndarray:
    return np.exp(-alpha * np.arange(n - 1, -1, -1, dtype=float))


def _shape_linear_desc(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i)


def _shape_quad_desc(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i * i)


def _shape_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.minimum(i + 1.0, n - i)


# ----------------------------------------------------------------------
# Main routine required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list:
    """
    Search for a non‑negative height sequence that minimises the
    evaluation function.  The routine respects the given time budget and
    returns the best sequence found as a plain Python list.
    """
    # ------------------------------------------------------------------
    # Initialisation
    # ------------------------------------------------------------------
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Small utility used locally (no closures)
    # ------------------------------------------------------------------
    def _maybe_update(best, cand):
        """Update global best_seq / best_score if `cand` improves."""
        nonlocal best_seq, best_score
        score = _evaluate(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] New best = {best_score:.6f} (len={len(cand)})")
        return score

    # ------------------------------------------------------------------
    # Phase 1 – parametric families (very cheap)
    # ------------------------------------------------------------------
    lengths = list(range(200, 1601, 200)) + [1326, 1500]
    for n in lengths:
        target = math.sqrt(2.0 * n)

        # exponential decay
        for alpha in np.logspace(-4, -0.3, num=12):
            a = _shape_exp(n, alpha)
            a = a / a.sum() * target
            _maybe_update(best_score, a)

        # reversed exponential
        for alpha in np.logspace(-4, -0.3, num=12):
            a = _shape_exp_rev(n, alpha)
            a = a / a.sum() * target
            _maybe_update(best_score, a)

        # linear descending
        for beta in np.linspace(0.0, 2.0 / max(1, n - 1), num=10):
            a = _shape_linear_desc(n, beta)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            _maybe_update(best_score, a)

        # quadratic descending
        for beta in np.linspace(0.0, 1.0 / max(1, n ** 2), num=10):
            a = _shape_quad_desc(n, beta)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            _maybe_update(best_score, a)

        # triangular (symmetric) shape
        a = _shape_triangular(n)
        a = a / a.sum() * target
        _maybe_update(best_score, a)

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Helper: sub‑gradient descent with momentum on a single start vector
    # ------------------------------------------------------------------
    def _run_subgrad(init_vec: np.ndarray):
        """
        Perform many sub‑gradient steps on `init_vec` (already on the simplex).
        Updates the global best sequence whenever a better candidate is seen.
        """
        nonlocal best_seq, best_score
        a = init_vec.astype(float)
        n = a.shape[0]
        target = math.sqrt(2.0 * n)

        # momentum vector
        m = np.zeros_like(a)
        beta = 0.9          # momentum decay
        base_step = 0.2

        it = 0
        while time.time() < deadline:
            conv = _conv_self(a)
            kmax = int(np.argmax(conv))

            # gradient of the maximal convolution entry
            low = max(0, kmax - (n - 1))
            high = min(n - 1, kmax)
            idx = np.arange(low, high + 1, dtype=int)
            partner = kmax - idx
            grad = np.zeros_like(a)
            grad[idx] = 2.0 * a[partner]

            # direction = -grad (sub‑gradient of max‑conv)
            direction = -grad
            m = beta * m + (1.0 - beta) * direction

            # diminishing step size
            step = base_step / math.sqrt(it + 1)
            max_abs = np.max(np.abs(m))
            if max_abs == 0.0:
                max_abs = 1.0
            eta = step / max_abs

            a_new = a + eta * m
            a_new = _project_onto_simplex(a_new, target)

            # evaluate periodically (every few steps) to keep overhead low
            if it % 5 == 0:
                _maybe_update(best_score, a_new)

            a = a_new
            it += 1

            # early exit if we have already reached the target bound
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Phase 2 – heavy sub‑gradient refinement
    # ------------------------------------------------------------------
    # always try to improve the current best (if any)
    if best_seq is not None:
        _run_subgrad(np.array(best_seq, copy=True))

    # several random restarts
    while time.time() < deadline:
        # random length (kept moderate to stay fast)
        n = np.random.randint(300, 2001)

        # random exponential seed – good baseline
        alpha = np.random.rand() * 0.02 + 1e-5
        a0 = _shape_exp(n, alpha)
        a0 = a0 / a0.sum() * math.sqrt(2.0 * n)

        _run_subgrad(a0)

        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Phase 3 – optional LP refinement (requires SciPy)
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import linprog
    except Exception:
        linprog = None

    if linprog is not None and best_seq is not None and time.time() < deadline:
        # LP refinement may be costly; we run only a few iterations
        a = np.array(best_seq, copy=True)
        n = a.shape[0]
        target = math.sqrt(2.0 * n)

        while time.time() < deadline:
            conv = _conv_self(a)
            M = conv.max()

            rows = 2 * n - 1
            # Build the constraint matrix A for (a * g)_k ≤ M
            A = np.zeros((rows, n), dtype=float)
            for k in range(rows):
                low = max(0, k - (n - 1))
                high = min(n - 1, k)
                cols = np.arange(low, high + 1, dtype=int)
                A[k, cols] = a[k - cols]

            res = linprog(
                c=-np.ones(n),                # maximise Σg_j
                A_ub=A,
                b_ub=np.full(rows, M),
                bounds=(0, None),
                method="highs",
                options={"presolve": True},
            )
            if not res.success:
                break

            g = res.x
            s = g.sum()
            if s <= 0:
                break
            # rescale g to the simplex used by the rest of the algorithm
            g = g * (target / s)

            # convex combination step (t chosen modestly)
            t = 0.2
            a = (1 - t) * a + t * g

            # check improvement
            _maybe_update(best_score, a)

            # stop if improvement stalls
            if best_score < 1.503 or time.time() + 0.5 > deadline:
                break

    # ------------------------------------------------------------------
    # Fallback – uniform sequence if nothing else succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    return best_seq.tolist()
