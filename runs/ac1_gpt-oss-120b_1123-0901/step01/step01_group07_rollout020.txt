"""
Propose a step‑function (non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in three stages:

1. **Initialisation** –  use the best power‑law shape found previously
   (`p≈0.5, γ≈0.32`).  The sequence is scaled so that its ℓ₁‑norm equals
   √(2 n) (the value used by the reference solution).

2. **Gradient optimisation** –  minimise a smooth surrogate of the
   ``max`` operator.  For a current vector ``a`` we compute the full
   self‑convolution ``b = a * a``.  The smooth surrogate is the
   log‑sum‑exp function

       φ(a) = (1/α)·log( Σ exp(α·bₖ) ),

   whose gradient w.r.t. ``a`` can be written using a *single*
   convolution:

       ∇φ = 2 · conv( w , a[::-1] )            (restricted to n entries)

   where ``wₖ = exp(α·bₖ) / Σ exp(α·b)`` is a soft‑max over the
   convolution entries.  We increase the temperature parameter ``α`` from
   5 up to ≈35 during the run so that the optimisation gradually focuses
   on the true maximum.

   After each gradient step we project onto the feasible set
   (non‑negative values and a fixed ℓ₁‑norm).  A simple back‑off
   schedule on the step size keeps the method stable.

3. **Random restarts / perturbations** –  when progress stalls we
   inject tiny Gaussian noise or re‑initialise from a fresh random
   Dirichlet draw.  The best sequence seen over the whole 1000 s budget
   is returned.

All heavy work (convolutions, gradients) uses FFT‑based linear
convolution which is O(n log n) for n≈3400 and easily fits into the time
budget.

The function ``propose_candidate`` respects the required signature and
returns a plain Python list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real 1‑D arrays using FFT.
    The result length is len(x) + len(y) – 1.
    """
    n = len(x) + len(y) - 1
    N = _next_pow2(n)
    fx = np.fft.rfft(x, N)
    fy = np.fft.rfft(y, N)
    conv = np.fft.irfft(fx * fy, N)[:n]
    return conv


def _make_powerlaw_sequence(n: int, p: float, gamma: float) -> np.ndarray:
    """Return a_i = (i + gamma) ** (-p) scaled to ℓ₁ = √(2 n)."""
    seq = (np.arange(n, dtype=np.float64) + gamma) ** (-p)
    # scaling – the absolute scale does not affect the evaluation,
    # but we keep the sum at the traditional value to stay away from the
    # “sum too small” guard in ``evaluate_sequence``.
    target_sum = np.sqrt(2.0 * n)
    seq = seq * (target_sum / seq.sum())
    return seq


def _score(seq: np.ndarray) -> float:
    """Exact value of the evaluation function (no security checks)."""
    n = len(seq)
    b = np.convolve(seq, seq)          # linear convolution, length 2 n‑1
    max_b = b.max()
    s = seq.sum()
    if s < 1e-12:          # same guard as in evaluate_sequence
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _grad_softmax(seq: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate of max(conv(seq,seq)).
    Returned vector has length len(seq) and corresponds to ``∇φ``.
    """
    n = len(seq)
    b = np.convolve(seq, seq)                # length 2 n‑1
    # Stable soft‑max weights
    max_b = b.max()
    w = np.exp(alpha * (b - max_b))
    w /= w.sum()                             # now Σ w = 1
    # Convolution w * reversed(seq)
    conv_full = _conv_full_fft(w, seq[::-1])
    # Slice the part that corresponds to the gradient (indices n‑1 … n‑1+n‑1)
    grad = 2.0 * conv_full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list:
    """
    Search for a low‑score step‑function under the 1000 s budget.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.15   # tiny safety margin

    # --------------------------------------------------------------
    # 1) Parameters & initialisation
    # --------------------------------------------------------------
    n = 3428                                 # length used by the reference solution
    target_sum = np.sqrt(2.0 * n)            # ℓ₁‑norm we keep throughout the optimisation

    # start from the classic power‑law (p≈0.5, γ≈0.32) – this is already good
    cur = _make_powerlaw_sequence(n, p=0.5002, gamma=0.3210)

    best_seq = cur.copy()
    best_score = _score(best_seq)
    print(f"[init] score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Main optimisation loop
    # --------------------------------------------------------------
    iteration = 0
    # learning‑rate schedule (initial value, decays slowly)
    base_lr = 0.08
    # temperature schedule for the soft‑max (α grows from 5 to 35)
    alpha_start, alpha_end = 5.0, 35.0

    while time.time() < deadline:
        iteration += 1

        # ---- evaluate current sequence ----
        cur_score = _score(cur)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur.copy()
            print(f"[iter {iteration:06d}] new best score {best_score:.6f}")

        # ---- if we have run for too long, break out ----
        if time.time() >= deadline:
            break

        # ---- compute gradient (soft‑max surrogate) ----
        #   α is increased linearly with elapsed time so that we focus
        #   more on the true maximum later in the run.
        elapsed = time.time() + budget_s - deadline - (budget_s - deadline)
        frac = (time.time() - (deadline - budget_s + 0.15)) / budget_s
        alpha = alpha_start + (alpha_end - alpha_start) * min(max(frac, 0.0), 1.0)

        grad = _grad_softmax(cur, alpha)

        # ---- step size – normalise gradient magnitude ----
        #   Prevent overly large jumps; we use a decaying step size.
        grad_norm = np.linalg.norm(grad) + 1e-12
        lr = base_lr / (1.0 + iteration * 1e-4)          # simple decay
        step = lr * grad / grad_norm

        # ---- gradient descent update ----
        nxt = cur - step
        # project onto the non‑negative orthant
        nxt = np.maximum(0.0, nxt)

        # re‑scale to keep the ℓ₁‑norm constant (does not affect the score)
        s = nxt.sum()
        if s > 0:
            nxt = nxt * (target_sum / s)
        else:
            # extremely unlikely, but fall back to a random feasible point
            nxt = _make_powerlaw_sequence(n, p=0.5, gamma=0.3)

        cur = nxt

        # ---- occasional random perturbation to escape plateaus ----
        if iteration % 1123 == 0:
            # add tiny Gaussian noise, then re‑normalise
            noise = np.random.randn(n) * (target_sum * 0.001)
            cur = np.maximum(0.0, cur + noise)
            cur = cur * (target_sum / cur.sum())
            print(f"[iter {iteration:06d}] perturbation applied")

    # --------------------------------------------------------------
    # 3) Return the best sequence found
    # --------------------------------------------------------------
    print(
        f"Finished search – best score {best_score:.6f} after {iteration} iterations"
    )
    # ensure plain Python list as required by the scorer
    return best_seq.tolist()
