def adam_step(a, m, v, t, lr, beta1, beta2, epsilon, target_sum):

- compute grad.

- m = beta1*m + (1-beta1)*grad

- v = beta2*v + (1-beta2)*(grad**2)

- m_hat = m / (1 - beta1**t)

- v_hat = v / (1 - beta2**t)

- update = lr * m_hat / (np.sqrt(v_hat) + epsilon)

- a_new = a - update

- projection: clip >=0, rescale to target_sum.

Return a_new, new m, v.

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start+budget_s-2.0
    best_seq = None
    best_score = float('inf')
    n = 2100
    target_sum = math.sqrt(2.0 * n)

    # Starting vector: if global variable height_sequence_1 exists, use it else random
    try: base = np.array(height_sequence_1, dtype=float)
    except Exception: base = np.random.rand(n)
    # Ensure length matches n
    if base.size != n:
        # If length differs, either truncate or pad.
        base = np.resize(base, n)  # not ideal but ok.
    # Normalize
    base = base * (target_sum / base.sum())
    a = np.maximum(base, 0.0)
    # Ensure within bounds 0-1000
    np.clip(a,0,1000, out=a)

    # Initialize Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.05

    iteration = 0

    while time.time() < deadline:
        # Compute gradient rev (Subgradient)
        S = a.sum()
        if S == 0:
            break
        # Compute convolution
        b = np.convolve(a, a)
        M = b.max()
        # Determine max indices within tolerance
        # Compute tolerance as maybe 1e-12*max(1,M) or relative.
        tol = 1e-12 * max(1.0, M)
        max_idx = np.where(np.abs(b - M) <= tol)[0]
        # Build rev sum
        rev = np.zeros_like(a)
        nlen = a.shape[0]
        idx = np.arange(nlen)
        for k in max_idx:
            j = k - idx
            mask = (j >= 0) & (j < nlen)
            rev[mask] += a[j[mask]]
        # Average over number of max indices
        rev = rev / len(max_idx)
        # Compute gradient
        grad = (4.0 * n / (S * S)) * rev - (4.0 * n * M / (S ** 3))
        # Adam updates
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a_new = a - step
        # Projection
        a_new = np.maximum(a_new, 0.0)
        # Re-normalize sum
        sum_new = a_new.sum()
        if sum_new > 0:
            a_new = a_new * (target_sum / sum_new)
        else:
            a_new = a.copy()
        # Clip to 0-1000
        np.clip(a_new, 0.0, 1000.0, out=a_new)

        # Evaluate
        cur_score = evaluate_sequence(a_new.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            print(f"[{iteration}] new best {best_score:.6f}")
        # accept step if improves or occasionally accept worse for exploration:
        # We'll accept always, but ensure we don't diverge (maybe keep if cur_score is not inf)
        if np.isfinite(cur_score) and cur_score <= best_score * 1.01:  # accept if not terrible
            a = a_new
        else:
            # maybe keep old a, but add small random noise for diversification occasionally
            if np.random.rand() < 0.001:
                a = a_new
            # else keep a unchanged

        # Periodic permutation improvement
        if iteration % 5000 == 0:
            # try swapping
            a_perm = a.copy()
            # we will attempt up to, say, 2000 swaps or until time over
            swaps = 0
            while swaps < 2000 and time.time() < deadline:
                i, j = np.random.randint(0, n, size=2)
                if i == j:
                    continue
                # swap
                a_perm[i], a_perm[j] = a_perm[j], a_perm[i]
                # no need to renormalize as sum unchanged
                # evaluate
                score_perm = evaluate_sequence(a_perm.tolist())
                if score_perm < best_score:
                    best_score = score_perm
                    best_seq = a_perm.copy()
                    print(f"[swap] new best {best_score:.6f}")
                if score_perm <= cur_score:
                    # Accept swap
                    a = a_perm
                    cur_score = score_perm
                else:
                    # revert
                    a_perm[i], a_perm[j] = a_perm[j], a_perm[i]
                swaps += 1

        # Occasionally random perturbation to avoid stagnation
        if iteration % 2000 == 0:
            # add small Gaussian noise
            noise = np.random.randn(n) * target_sum * 1e-4
            a_noisy = a + noise
            a_noisy = np.maximum(a_noisy, 0.0)
            sum_noisy = a_noisy.sum()
            if sum_noisy > 0:
                a_noisy = a_noisy * (target_sum / sum_noisy)
            a_noisy = np.clip(a_noisy, 0.0, 1000.0)
            score_noisy = evaluate_sequence(a_noisy.tolist())
            if score_noisy < best_score:
                best_score = score_noisy
                best_seq = a_noisy.copy()
                print(f"[noise] new best {best_score:.6f}")
                a = a_noisy

        # Possibly restart after certain iterations if no improvement for long time
        # Let's implement simple early restart: if iteration % 50000 == 0, random reinitialize
        if iteration % 50000 == 0:
            # Random reinitialize
            init = np.random.rand(n)
            init = init * (target_sum / init.sum())
            a = init
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            print(f"[restart] iteration {iteration}")
            # evaluate and maybe update best_seq
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[restart] new best {best_score:.6f}")

    print(f"[done] iterations={iteration}, best_score={best_score:.6f}")
    return best_seq.tolist()

"""
Hybrid Adam + LP / permutation optimizer for the step‑function bound.

The evaluation functional
    E(a) = 2·n·max_k (a*a)[k] / (sum_i a_i)^2
is homogeneous, so we fix the L1‑norm to sqrt(2·n) (the same
normalisation used in the reference paper).  The algorithm works in three
phases:

1.  **LP‑initialisation** – solve a linear programme that maximises the
    total mass under the constraint (f * g)[·] ≤ max(f*f).  The resulting
    vector g is scaled to the required L1‑norm and used as a starting
    point; this already pushes the mass to “safe” positions (small
    entries of f) and gives a considerable drop from the uniform bound.

2.  **Projected Adam optimisation** – we minimise E by a projected Adam
    optimiser.  The gradient of max‑convolution is sub‑differentiable;
    we use the average of the gradients over all indices where the
    convolution attains its maximum (more stable than picking a single
    arg‑max).  After each step we project onto the non‑negative orthant,
    clip to the limits required by ``evaluate_sequence`` and re‑normalise
    to the fixed L1‑norm.

3.  **Permutation local search** – every few thousand Adam steps we
    perform a cheap hill‑climbing on the ordering of the entries:
    random swaps are accepted if they do not increase E.  This exploits
    the fact that the convolution depends heavily on the ordering of the
    heights.

The whole search runs for the allocated wall‑clock budget (≈ 1000 s);
the best sequence seen so far is returned.  All operations are
vectorised and the expensive convolution is reused for both the
gradient and the exact objective value, so the inner loop is cheap.

The code uses only NumPy, SciPy (linprog + toeplitz) and the supplied
``evaluate_sequence`` function.
"""

import time
import math
import numpy as np
import scipy.optimize
import scipy.linalg

# ----------------------------------------------------------------------
def _conv_and_max(a: np.ndarray):
    """Return convolution (a*a), its maximum value and all indices attaining it."""
    conv = np.convolve(a, a)
    M = conv.max()
    # tolerance for floating‑point equality
    tol = 1e-12 * max(1.0, M)
    max_idx = np.where(np.abs(conv - M) <= tol)[0]
    return conv, M, max_idx


def _gradient(a: np.ndarray, n: int, S: float, M: float, max_idx: np.ndarray):
    """
    Sub‑gradient of  E(a) = 2·n·M / S²  w.r.t. a.
    The gradient of M is the average of the contributions from all
    maximising shifts.
    """
    rev = np.zeros_like(a)
    idx = np.arange(n)

    for k in max_idx:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]

    # average over the active set
    rev /= max(1, len(max_idx))

    grad = (4.0 * n / (S * S)) * rev - (4.0 * n * M / (S ** 3))
    return grad


def _lp_optimize(f: np.ndarray, M: float):
    """
    Solve  max sum(g)  s.t.  f * g ≤ M  (component‑wise) , g ≥ 0 , g ≤ 1000.
    Returns the optimal g or None on failure.
    """
    n = f.shape[0]

    # build Toeplitz matrix that implements the convolution f * g
    col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    row = np.concatenate([f[:1], np.zeros(2 * n - 2, dtype=f.dtype)])
    A = scipy.linalg.toeplitz(col, row)          # shape (2n‑1, n)

    c = -np.ones(n)                               # maximise sum(g)  → minimise –sum(g)
    b = np.full(2 * n - 1, M)

    bounds = [(0.0, 1000.0)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                 method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None

                swaps += 1

        # --- light random jitter every few thousand iterations ----------
        if iteration % 2000 == 0:
            jitter = np.random.randn(n) * (target_sum * 5e-4 / n)
            a_jit = a + jitter
            a_jit = np.maximum(a_jit, 0.0)
            a_jit = a_jit * (target_sum / a_jit.sum())
            np.clip(a_jit, 0.0, 1000.0, out=a_jit)
            jit_score = _score(a_jit)
            if jit_score < best_score:
                best_score = jit_score
                best_seq = a_jit.copy()
                print(f"[jitter {iteration}] new best {best_score:.6f}")
                a = a_jit

        # --- occasional LP polishing (very cheap, at most a few times) ---
        if iteration % 30000 == 0:
            conv, M, _ = _conv_and_max(a)
            g_opt = _lp_optimize(a, M)
            if g_opt is not None:
                g_opt = np.clip(g_opt, 0.0, 1000.0)
                g_opt = g_opt * (target_sum / g_opt.sum())
                # mix with a small step size
                t_mix = 0.15
                a = (1.0 - t_mix) * a + t_mix * g_opt
                a = np.clip(a, 0.0, 1000.0)
                a = a * (target_sum / a.sum())
                # re‑evaluate
                cur_score = _score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[LP {iteration}] new best {best_score:.6f}")

    print(f"[done] iterations={iteration}, best_score={best_score:.6f}")
    # ensure the returned list respects the evaluate_sequence limits
    final = np.clip(best_seq, 0.0, 1000.0).tolist()
    return final
