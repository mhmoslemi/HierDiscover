"""
propose_candidate
=================

Search for a non‑negative sequence that minimizes the evaluation metric

    2·n·max(conv(a,a)) / (sum(a)**2)

The search consists of two stages.

1. **Parameter sweep** –  families of Beta‑shaped distributions
   (including the discrete arcsine law) are generated for the chosen
   length `n`.  The best among them is kept as a starting point.

2. **Targeted hill‑climbing** –  at each iteration the index `m`
   where the current convolution attains its maximum is identified.
   The contribution of each coefficient to this peak is
   `c[i] = a[i] * a[m-i]`.  Mass is moved from the index with the
   largest contribution to the one with the smallest contribution.
   The move size is a decreasing fraction of the source weight.
   The convolution is updated cheaply (O(n)) and the move is accepted
   only if the peak value does not increase.  The process runs until
   the allotted time is exhausted.

The algorithm respects all security checks performed in
`evaluate_sequence` – all entries are non‑negative, the sum is kept
above the required threshold and the final result is returned as a
plain Python list.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ------------------------------------------------------------

def beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta(α,β) distribution on n points.
    Points are taken at the mid‑points (k+0.5)/n.  The result is
    normalised to have sum 1.
    """
    # avoid exact 0 or 1 which would produce infinities for α<1 or β<1
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # Power law – safe for α,β > 0
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    # Numerical guard – clip extreme values that could cause overflow
    a = np.clip(a, 0.0, 1e8)
    total = a.sum()
    if total == 0.0:
        a[0] = 1.0  # fallback to a single spike
        total = 1.0
    return a / total   # sum == 1.0


def initial_arcsine(n: int) -> np.ndarray:
    """
    Return the discrete arcsine distribution of length `n`,
    normalised to sum 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * ((2.0 * k - 1.0) * (2.0 * k)) / (4.0 * k * k)
    a /= a.sum()
    return a


def conv_max(a: np.ndarray) -> float:
    """Return max of the autoconvolution of `a` (sum(a) == 1)."""
    return np.max(np.convolve(a, a))


def targeted_move(a: np.ndarray, conv: np.ndarray, eps_frac: float) -> tuple:
    """
    Perform a single targeted move on `a`:
      * identify the index `m` where `conv` is maximal,
      * compute contributions `c[i] = a[i] * a[m-i]`,
      * move a small amount of mass from the largest contributor
        to the smallest contributor.
    Returns the new array, the updated convolution and a boolean
    indicating whether the move was accepted.
    """
    n = a.shape[0]
    m = int(np.argmax(conv))                     # index of the peak
    # range of i that actually participates in the peak
    i_min = max(0, m - (n - 1))
    i_max = min(n - 1, m)
    idx = np.arange(i_min, i_max + 1, dtype=np.intp)
    j = m - idx                                   # partner indices
    contrib = a[idx] * a[j]

    src = idx[np.argmax(contrib)]                # biggest contributor
    dst = idx[np.argmin(contrib)]                # smallest contributor

    if a[src] <= 0.0:
        return a, conv, False

    eps = eps_frac * a[src]                      # move a fraction of the source weight
    if eps <= 0.0:
        return a, conv, False

    # propose new vector
    a_new = a.copy()
    a_new[src] -= eps
    a_new[dst] += eps

    # cheap O(n) update of the convolution:
    conv_new = conv.copy()
    # contribution of the source (removed)
    conv_new[src:src + n] -= 2.0 * eps * a
    # contribution of the destination (added)
    conv_new[dst:dst + n] += 2.0 * eps * a
    # quadratic correction term δ*δ
    two_src = 2 * src
    two_dst = 2 * dst
    mid = src + dst
    if two_src < conv_new.size:
        conv_new[two_src] += eps * eps
    if two_dst < conv_new.size:
        conv_new[two_dst] += eps * eps
    if mid < conv_new.size:
        conv_new[mid] -= 2.0 * eps * eps

    # accept only if the peak does not increase
    if conv_new.max() <= conv.max() + 1e-15:
        # recompute exact convolution for safety (still cheap)
        conv_exact = np.convolve(a_new, a_new)
        if conv_exact.max() <= conv.max() + 1e-15:
            return a_new, conv_exact, True
    return a, conv, False


# ------------------------------------------------------------
# Main entry point required by the platform
# ------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list:
    """
    Generate a candidate step‑function sequence.
    The function respects the global `height_sequence_1` if it exists,
    otherwise it falls back to a length of 6000.
    """
    np.random.seed(seed)
    start_time = time.time()

    # --------------------------------------------------------
    # 1.  Choose a length for the sequence.
    # --------------------------------------------------------
    n = kwargs.get("length")
    if n is None:
        try:
            n = len(height_sequence_1)   # type: ignore[name-defined]
        except Exception:
            n = 6000                     # a comfortable size for the search

    # --------------------------------------------------------
    # 2.  Scan a few parametrised families (Beta distributions)
    # --------------------------------------------------------
    best_seq = None
    best_score = np.inf
    families = []

    # a) the classic arcsine law (β = 0.5)
    families.append(initial_arcsine(n))

    # b) symmetric Beta(α,α) with α around 0.5
    for alpha in np.linspace(0.30, 0.70, 9):
        families.append(beta_distribution(n, alpha, alpha))

    # c) a few mildly skewed betas
    for alpha, beta in [(0.4, 0.6), (0.6, 0.4), (0.35, 0.65), (0.65, 0.35)]:
        families.append(beta_distribution(n, alpha, beta))

    # evaluate all candidates
    for cand in families:
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
    print(f"[init] length={n}, best initial score={best_score:.6f}")

    # --------------------------------------------------------
    # 3.  Targeted hill‑climbing starting from the best candidate.
    # --------------------------------------------------------
    a = best_seq.copy()
    conv = np.convolve(a, a)
    best_seq = a.copy()

    # parameters for the hill‑climbing
    eps_frac = 0.20          # start with 20 % of the source weight
    decay = 0.995            # exponential decay of step size
    it = 0
    max_iter = 500_000       # safety cap; the loop will also stop by time

    while time.time() - start_time < budget_s - 0.5 and it < max_iter:
        a, conv, improved = targeted_move(a, conv, eps_frac)
        if improved:
            cur_score = 2.0 * n * conv.max()   # equal to evaluate_sequence for sum==1
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[improve] it={it:06d}, score={best_score:.6f}")
        # slowly shrink the step size to allow finer adjustments later
        eps_frac *= decay
        it += 1

    print(f"[final] iterations={it}, best score={best_score:.6f}")
    # --------------------------------------------------------
    # 4.  Return the best sequence found (plain Python list).
    # --------------------------------------------------------
    return best_seq.tolist()
