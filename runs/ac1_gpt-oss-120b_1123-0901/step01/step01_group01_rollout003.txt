""" 
Implementation: simple geometric step function search.
Choose a sequence length n moderate (default 20000). 
For a range of decay factors c around ln(n) evaluate the objective
using the provided evaluate_sequence and keep the best. 
Finally return the best sequence.
"""

import time, math, numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # Choose a length that balances quality and runtime.
    # Large n reduces the theoretical bound ~ (ln n)^2 / n.
    # Using n = 20000 stays well within the time budget.
    n = 20000

    # Determine a base decay coefficient around ln(n)
    base = math.log(n)
    # We'll sweep a small integer neighbourhood
    candidates = [base - 2, base - 1, base, base + 1, base + 2]
    # also include a few small decays to be safe
    candidates = [c for c in candidates if c > 0.5]
    # ensure we stay within the budget `deadline`.
    best_seq = None
    best_score = float('inf')
    for c in candidates:
        if time.time() > deadline:
            break
        eps = c / n
        r = 1.0 - eps
        if r <= 0.0:
            continue
        # Build geometric progression a_i = r^i
        powers = np.exp(np.arange(n, dtype=np.float64) * math.log(r))
        # Normalize to sum 1 (any scaling leaves the objective invariant,
        # but fixing the sum makes the objective comparable)
        seq = powers / powers.sum()
        score = evaluate_sequence(seq.tolist())
        print(f"[Search] c={c:.3f}  r={r:.6f}  score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = seq.copy()

    # If we still have time, try a very fine search around the best c.
    if best_seq is not None:
        # refine via tiny perturbations of the decay factor
        # using a simple golden-section style loop
        lo = max(0.5, best_c - 1.0)
        hi = best_c + 1.0
        for _ in range(12):
            if time.time() > deadline:
                break
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            for c in (m1, m2):
                eps = c / n
                r = 1.0 - eps
                powers = np.exp(np.arange(n) * math.log(r))
                seq = powers / powers.sum()
                score = evaluate_sequence(seq.tolist())
                print(f"[Refine] c={c:.6f} score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    best_c = c
            # compare function values at m1 and m2
            # we need scores for m1, m2; we already computed above in loop maybe
            # for simplicity we just shrink interval towards the currently best
            if best_c == m1:
                hi = m2
            else:
                lo = m1

    # fallback in case none found (should not happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    print(f"Best objective found: {best_score:.6f} (n={n})")
    return best_seq.tolist()

"""
Simple geometric‑progression search.

The evaluation objective is scale‑invariant, so we may fix the
L1‑norm of the sequence to one.  For a geometric sequence
a_i = r^i the maximal entry of the self‑convolution is attained
either at shift 0 (a_0²) or at the centre (≈ n·r^{n‑1} / S²) where
S = Σ_{j=0}^{n‑1} r^j.  By choosing the decay factor

        c ≈ ln n   (i.e. r = 1 – c/n)

the two contributions become equal and the objective behaves like

        2·n·max_b ≈ 2·(ln n)² / n ,

which tends to zero as n grows.  The algorithm therefore:

1. Picks a moderate length `n` (20000 gives a score ≈ 0.01 in a few
   seconds).
2. Sweeps a handful of decay factors around `ln n`.
3. Evaluates each candidate using the provided `evaluate_sequence`.
4. Optionally refines the best factor with a tiny ternary search.
5. Returns the best sequence found.

All numbers are non‑negative; the runtime stays far below the 1000 s
budget.
"""

import time
import math
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """Return a step‑function (list of non‑negative heights) that minimises
    the given evaluation function.  The construction is a geometric
    progression with a decay factor tuned around `ln(n)`.
    """
    np.random.seed(seed)

    # reserve a small safety margin for the evaluator
    deadline = time.time() + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1. Choose a length: 20000 is a good trade‑off between quality and
    #    evaluation cost (convolution of 40000 elements is still fast).
    n = 20000

    # 2. Base decay factor – the theoretical optimum is around ln(n)
    base = math.log(n)

    # 3. Small grid around the base value
    cand_factors = [base - 2.0, base - 1.0, base, base + 1.0, base + 2.0]

    best_seq = None
    best_score = float("inf")
    best_factor = None

    # ------------------------------------------------------------------
    # 4. Scan the candidates
    for c in cand_factors:

            break

            continue

            continue

        log_r = math.log(r)

        seq = powers / powers.sum()          # fix L1‑norm to 1 (scale‑invariant)

        # evaluate the objective

            best_score = score
            best_seq = seq.copy()
            best_factor = c

    # 5. Optional tiny refinement around the best factor (ternary search)
    if best_seq is not None and time.time() < deadline:

        hi = best_factor + 1.0

        while time.time() < deadline and hi - lo > 1e-5:

            m2 = hi - (hi - lo) / 3.0
            scores = []
