def compute_objective(a):
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    max_conv = np.max(conv)
    n = a.shape[0]
    return float(2 * n * max_conv / (sum_a**2))

def gradient_for_k(a, k):
    n = a.shape[0]
    low = max(0, k - (n - 1))
    high = min(n - 1, k)
    idx = np.arange(low, high + 1)
    partner = k - idx
    grad = np.zeros_like(a)
    grad[idx] = 2 * a[partner]
    return grad

def combined_gradient(a, ks):
    grad = np.zeros_like(a)
    for k in ks:
        grad += gradient_for_k(a, k)
    if len(ks) > 0:
        grad /= len(ks)
    return grad

"""
Optimisation of a non‑negative step‐function for the objective

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm combines several ideas:

*   Every candidate is kept on the simplex
        {x_i ≥ 0 , Σ x_i = √(2·n)}  – the objective is scale‑invariant.
*   At each iteration the convolution is computed, a *small* set of
    indices attaining (or almost attaining) the maximal value is selected,
    and the (sub)‑gradient of those entries is summed.
*   The step size is chosen proportional to max(conv) / max|grad| (as in
    the reference method) and then the new point is projected back onto
    the simplex.
*   A light “Adam‑like’’ momentum (exponential moving averages of the
    gradient) is applied to stabilise the descent.
*   Random perturbations are injected occasionally to escape flat regions.
*   Many independent restarts with different lengths and initial shapes
    (uniform, cosine‑U‑shape, sinusoidal perturbations, previous best,
    random) are performed within the time budget.  The best sequence seen
    so far is refined at the end.

The implementation respects the required “no closures’’ rule – all helper
functions are defined at top level.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex of radius ``s``."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective (fast, no clipping)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2).  Returns np.inf for
    degenerate inputs."""
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    max_b = np.max(conv)
    n = a.shape[0]
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Gradient of a single convolution entry k
# ----------------------------------------------------------------------
def _grad_for_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of conv(a,a)[k] w.r.t. a.
    (∂/∂a_i) conv_k = 2 * a_{k-i} for indices that appear in the sum.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    partner = k - idx
    grad[idx] = 2.0 * a[partner]
    return grad


# ----------------------------------------------------------------------
# Combined gradient from a set of indices (sub‑gradient of the max)
# ----------------------------------------------------------------------
def _combined_gradient(a: np.ndarray, ks) -> np.ndarray:
    """Average of the gradients for the (few) indices in ``ks``."""
    if len(ks) == 0:
        return np.zeros_like(a)
    grad = np.zeros_like(a)
    for k in ks:
        grad += _grad_for_index(a, int(k))
    grad /= len(ks)
    return grad


# ----------------------------------------------------------------------
# Core optimisation routine for a fixed length ``n``
# ----------------------------------------------------------------------
def _run_optimisation(
    n: int,
    init: np.ndarray,
    target_sum: float,
    best_data: dict,
    deadline: float,
) -> None:
    """
    Perform a projected (sub)‑gradient descent on a single problem size.
    Updates ``best_data`` (keys ``'seq'`` and ``'score'``) whenever a better
    sequence is discovered.
    """
    # initialise feasible vector
    a = _project_onto_simplex(init.astype(float), target_sum)
    cur_score = _objective(a)

    if cur_score < best_data["score"]:
        best_data["score"] = cur_score
        best_data["seq"] = a.copy()
        print(f"[{time.time():.2f}] Init best = {cur_score:.6f} (n={n})")

    # hyper‑parameters
    base_step = 0.05          # scaling factor for the step length
    max_top = 5              # number of top convolution indices to use
    run_deadline = min(deadline, time.time() + 9.0)   # ~9 s per run

    # Adam‑style moving averages (optional – they just stabilise the direction)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    while time.time() < run_deadline:
        # current convolution and the set of (almost) maximal entries
        conv = np.convolve(a, a)
        max_val = conv.max()

        # pick a few indices with the largest values (sub‑gradient of the max)
        if max_top >= conv.size:
            top_ks = np.arange(conv.size)
        else:
            top_ks = np.argpartition(conv, -max_top)[-max_top:]

        # combined gradient
        grad = _combined_gradient(a, top_ks)

        # Adam style update (first‑ and second‑moment estimates)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size proportional to max(conv)/max|grad| (as in the reference)
        max_grad = np.max(np.abs(grad))
        if max_grad == 0.0:
            max_grad = 1.0
        eta = base_step * max_val / max_grad

        # Adam‑scaled descent direction
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        # move and re‑project onto the simplex
        a_new = _project_onto_simplex(a - step, target_sum)

        # evaluate the new point
        new_score = _objective(a_new)

        # acceptance rule – always accept an improvement; occasionally accept a
        # worsening move to help escape shallow local minima.
        if new_score < cur_score or np.random.rand() < 0.01:
            a = a_new
            cur_score = new_score
            if new_score < best_data["score"]:
                best_data["score"] = new_score
                best_data["seq"] = a.copy()
                print(f"[{time.time():.2f}] New best = {new_score:.6f} (n={n})")

        # random shake‑up (helps to explore flat regions)
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            cur_score = _objective(a)

        # very slowly decay the base step so that the algorithm fine‑tunes
        base_step *= 0.9995


# ----------------------------------------------------------------------
# Public entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 42,
    budget_s: float = 1000.0,
    **kwargs,
) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    evaluation score.  The routine runs for at most ``budget_s`` seconds,
    exploring many random restarts and refining the best candidate found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_data = {"seq": None, "score": float("inf")}

    # ------------------------------------------------------------------
    # Main optimisation loop – many independent restarts of varying size
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # choose a problem size (shorter runs for very large n)
        n = np.random.randint(200, 1500)

        # target sum = √(2·n) (any positive scaling works; this keeps the
        # previous normalisation used in the reference code)
        target_sum = np.sqrt(2.0 * n)

        # initialise from a handful of different patterns
        init_type = np.random.choice(
            ["uniform", "u_shape", "sin_perturb", "random", "prev"]
        )
        if init_type == "uniform":
            init = np.full(n, target_sum / n)
        elif init_type == "u_shape":
            # high at the edges, low in the centre (cosine‑squared pattern)
            x = np.arange(n)
            f = np.cos(np.pi * x / (n - 1)) ** 2
            init = f
        elif init_type == "sin_perturb":
            # uniform plus a very small sinusoidal wiggle
            base = np.full(n, target_sum / n)
            wiggle = np.sin(np.linspace(0, 2 * np.pi, n)) * (0.03 * target_sum / n)
            init = base + wiggle
        elif init_type == "random":
            init = np.random.rand(n)
        else:   # "prev" – reuse the current best (trim / pad as needed)
            if best_data["seq"] is not None:
                src = np.asarray(best_data["seq"], dtype=float)
                if src.size >= n:
                    init = src[:n].copy()
                else:
                    pad = np.random.rand(n - src.size) * 1e-3
                    init = np.concatenate([src, pad])
            else:
                init = np.random.rand(n)

        # run a short optimisation on this restart
        _run_optimisation(n, init, target_sum, best_data, deadline)

        # early exit if we have already reached the target bound
        if best_data["score"] < 1.51:
            break

    # ------------------------------------------------------------------
    # Final fine‑tuning on the best sequence found so far (if time permits)
    # ------------------------------------------------------------------
    if best_data["seq"] is not None:
        remaining = deadline - time.time()
        if remaining > 2.0:
            n_best = len(best_data["seq"])
            target_sum = np.sqrt(2.0 * n_best)
            # a longer run with a smaller base step for fine adjustments
            _run_optimisation(
                n_best,
                best_data["seq"].copy(),
                target_sum,
                best_data,
                time.time() + min(15.0, remaining),
            )

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded, return a safe uniform sequence
    # ------------------------------------------------------------------
    if best_data["seq"] is None:
        n = 500
        best_data["seq"] = np.full(n, np.sqrt(2.0 * n) / n)

    # The competition expects a plain Python list, not a NumPy array.
    return best_data["seq"].tolist()
