"""
Propose a step‑function (non‑negative vector) with a low value of

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The implementation mixes three ideas:

1. **Good analytical start** – a_i ∝ (i+1)^‑p with p≈0.5.  The
   continuous kernel 1/√x has an (almost) constant aperiodic
   autocorrelation, which is already close to the theoretical optimum.

2. **Projected (sub‑)gradient descent on the active constraints** – at
   each iteration the indices whose convolution value is within a small
   relative tolerance of the current maximum are collected; the gradient
   of the sum of those entries (∂/∂a_i Σ_k a_i a_{k‑i}) is
   `2 * a_{k‑i}`.  The step is taken opposite to that gradient and the
   vector is projected back onto the simplex
   {x_i ≥ 0, Σ x_i = √(2 n)} (the normalisation that makes the objective
   equal to `max(conv)`).

3. **Exploration** – occasional random jitter, a decaying step size and
   many independent restarts (different lengths, different exponents
   p, optional warm‑starts from the best solution seen so far).  The best
   candidate discovered before the time budget expires is returned.

All heavy work (convolution and gradients) is performed with FFTs,
so even thousands of iterations on vectors of a few thousand entries
fit comfortably into the 1000 s budget.

The function `propose_candidate` obeys the required signature and can be
called directly.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper 1 – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2 – convolution via FFT (real inputs → real output)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full (linear) convolution a * a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    # Numerical noise may produce tiny negative values – clip them.
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Helper 3 – indices whose convolution value is close to the current maximum
# ----------------------------------------------------------------------
def _active_indices(conv: np.ndarray, rel_tol: float = 5e-4) -> tuple[float, np.ndarray]:
    """Return (max_value, indices) where conv ≥ max – rel_tol·max."""
    max_val = conv.max()
    # absolute tolerance never goes below a tiny number to avoid empty set
    tol = max(1e-12, max_val * rel_tol)
    active = np.where(conv >= max_val - tol)[0]
    return max_val, active


# ----------------------------------------------------------------------
# Helper 4 – gradient of Σ_{k∈active} conv_k  with respect to a
# ----------------------------------------------------------------------
def _gradient(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Gradient of the sum of the active convolution entries."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]          # ∂/∂a_i Σ a_i a_{k-i} = 2 a_{k-i}
    return grad


# ----------------------------------------------------------------------
# Core optimisation routine for a fixed length `n`
# ----------------------------------------------------------------------
def _optimise_one_length(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    rel_tol: float = 5e-4,
    base_step: float = 0.12,
    momentum_beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the max‑convolution objective for a single length.
    Returns the best vector found and its (approximate) score.
    """
    target_sum = np.sqrt(2.0 * n)          # Σ a_i = √(2 n) → objective = max(conv)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # keep the best vector seen in this run
    conv = _conv_fft(a)
    max_val, _ = _active_indices(conv, rel_tol)
    best_local = a.copy()
    best_score = 2.0 * n * max_val / (a.sum() ** 2)

    # momentum term (Adam‑like but without second‑moment)
    velocity = np.zeros_like(a)

    step_factor = base_step
    iteration = 0
    while time.time() < run_deadline:
        # --------------------------------------------------------------
        # 1) compute convolution and active set
        # --------------------------------------------------------------
        conv = _conv_fft(a)
        max_val, active = _active_indices(conv, rel_tol)

        # --------------------------------------------------------------
        # 2) gradient of the sum of the active entries
        # --------------------------------------------------------------
        grad = _gradient(a, active)

        # --------------------------------------------------------------
        # 3) momentum + adaptive step size
        # --------------------------------------------------------------
        velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
        max_grad = np.max(velocity) if np.max(velocity) > 0 else 1.0
        eta = step_factor * max_val / max_grad

        a_new = a - eta * velocity
        a_new = _project_onto_simplex(a_new, target_sum)

        # --------------------------------------------------------------
        # 4) occasional random shake‑up (helps escaping flat plateaus)
        # --------------------------------------------------------------
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (target_sum * 0.001 / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            # after a shake‑up we become a bit more cautious
            step_factor *= 0.92

        # --------------------------------------------------------------
        # 5) evaluate – we use the analytical form (identical to evaluate_sequence)
        # --------------------------------------------------------------
        conv_new = _conv_fft(a_new)
        max_val_new = conv_new.max()
        score = 2.0 * n * max_val_new / (a_new.sum() ** 2)

        # --------------------------------------------------------------
        # 6) keep global best of this run
        # --------------------------------------------------------------
        if score < best_score:
            best_score = score
            best_local = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  local best = {score:.6f}")

        a = a_new
        # slowly decay the step factor – classic simulated‑annealing style
        step_factor *= 0.9997
        iteration += 1

    return best_local, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the evaluation
    function.  The routine runs for at most `budget_s` seconds and returns the
    best sequence discovered.

    Strategy:
        • Randomly choose a length n (≈ 800 … 3500).
        • Initialise with a power‑law shape a_i ∝ (i+1)^{-p} (p≈0.5),
          optionally warm‑started from the current global best.
        • Run a projected gradient descent that optimises all near‑maximal
          convolution entries simultaneously, with momentum and occasional
          stochastic perturbations.
        • Keep the overall best vector across many independent restarts.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # range of lengths to explore (large enough to beat the previous 1.524)
    min_n, max_n = 800, 3500

    # ------------------------------------------------------------------
    # Main restart loop – each restart gets a few seconds of work.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Choose a problem size.  As the deadline shrinks we stay modestly
        # small to keep each run cheap.
        remaining = deadline - time.time()
        if remaining < 30.0:
            n = np.random.randint(800, 1500)
        else:
            n = np.random.randint(min_n, max_n + 1)

        # ----- initialise -------------------------------------------------
        # Base shape: a_i = (i+1)^{-p} with a random exponent around 0.5.
        p = np.random.uniform(0.45, 0.55)
        idx = np.arange(1, n + 1, dtype=np.float64)
        init = idx ** (-p)

        # Small random jitter to avoid symmetry artefacts
        init += np.random.rand(n) * 1e-4

        # Occasionally warm‑start from the best vector we have seen so far.
        if best_seq is not None and np.random.rand() < 0.4:
            if best_seq.size >= n:
                init = best_seq[:n].copy()
            else:
                # pad the existing best with tiny noise to reach the new length
                pad = np.random.rand(n - best_seq.size) * 1e-4
                init = np.concatenate([best_seq, pad])

        # ----- allocate a time slice for this run -----------------------
        # 8 s per run is a good trade‑off: many restarts, each gets enough
        # iterations to converge a little.
        run_deadline = min(deadline, time.time() + 8.0)

        # ----- run the optimisation --------------------------------------
        seq_local, score_local = _optimise_one_length(
            n,
            init,
            run_deadline,
            rel_tol=5e-4,
            base_step=0.12,
            momentum_beta=0.9,
        )

        # ----- update global best ----------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (length={n})"
            )

        # Early‑exit if we have already passed the target threshold.
        if best_score < 1.45:   # comfortably below the 1.503 target
            break

    # ------------------------------------------------------------------
    # Fallback – in the extremely unlikely case that nothing was found.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    # Return a plain Python list as required by the evaluator.
    return best_seq.tolist()
```
