"""
Simple soft‑max gradient descent with Adam.

The score to minimise is
    E(a) = 2 * n * max_k (a * a)[k] / (sum_i a_i)^2 .
Because E is homogeneous we keep the sum fixed to sqrt(2n); scaling
does not change the objective but makes the optimisation numerically
stable.

The gradient of the exact max is not smooth – it depends on a single
convolution index.  To obtain a smoother landscape we replace the max
by a soft‑max (log‑sum‑exp) with a temperature β that is slowly raised.
For a given β the gradient of the soft‑max is a weighted sum of the
gradients of all convolution entries, which makes the optimiser less
prone to get stuck at a local plateau.

The optimiser uses the Adam update rule, projects onto the non‑negative
orthant, and rescales after each step to keep the prescribed total sum.
Occasional random jitter helps escape shallow minima.  The best sequence
found before the time budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution using a power‑of‑two FFT (real inputs)
def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Convolve two real 1‑D arrays using FFT.
    Returns the full discrete convolution of length len(x)+len(y)-1.
    """
    n = len(x) + len(y) - 1
    # next power of two for efficient FFT
    N = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    Z = X * Y
    conv = np.fft.irfft(Z, N)[:n]
    return conv


# ----------------------------------------------------------------------
# Compute the (soft) gradient of the objective for a given temperature β
def softmax_gradient(a: np.ndarray, beta: float, target_sum: float) -> tuple:
    """
    Compute Adam‑compatible gradient for the soft‑max approximation of the
    objective and return also the true convolution maximum.

    Parameters
    ----------
    a : np.ndarray
        Current sequence (shape (n,)).
    beta : float
        Soft‑max temperature (larger → closer to true max).
    target_sum : float
        Desired sum of a (sqrt(2*n)); used only for scaling the gradient.

    Returns
    -------
    grad : np.ndarray
        Gradient of the soft‑max objective (shape (n,)).
    max_conv : float
        The true maximum of the convolution a*a (used for the true score).
    """
    n = a.shape[0]
    # true convolution (needed for evaluation)
    conv = conv_fft(a, a)                # length 2n‑1
    max_conv = conv.max()                # true max for score reporting

    # ----- soft‑max over convolution entries -----
    # stability: shift by the maximum before exponentiation
    shift = conv.max()
    exp_term = np.exp(beta * (conv - shift))
    w = exp_term / exp_term.sum()        # soft‑max weights, sum to 1

    # soft‑max (log‑sum‑exp) value – used in the gradient denominator term
    M_approx = np.dot(w, conv)           # = (1/β) * log‑sum‑exp(β*conv)

    # ----- gradient of the soft‑max term -----
    # derivative of conv_k w.r.t a_p is 2 * a_{k-p} (zero out of range)
    a_rev = a[::-1]                      # reversed a
    # convolution of weights w with reversed a gives needed sums
    conv_rev = conv_fft(w, a_rev)        # length 3n‑2
    start = n - 1
    G = conv_rev[start:start + n]        # length n, each entry = Σ_i w_i a_{i-p}
    grad_M = 2.0 * G                     # dM_approx / da

    # ----- full gradient of the (soft‑max) objective -----
    S = a.sum()
    # derivative of 2*n*M / S^2  (with M replaced by M_approx)
    grad = (2.0 * n / (S * S)) * grad_M - (4.0 * n * M_approx / (S ** 3))
    return grad, max_conv
