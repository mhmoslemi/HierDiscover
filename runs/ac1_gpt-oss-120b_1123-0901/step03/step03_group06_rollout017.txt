"""
Propose a step‑function with a low self‑convolution peak.

Algorithm
---------
* Work with a probability vector `p` (sum = 1) and the canonical scaling
  `a = sqrt(2·n)·p`.  The objective is `2·n·max(p∗p)`.
* For each chosen length `n` we run a mirror‑descent (exponentiated‑gradient)
  optimisation on `log(p)`:
    - Compute the full self‑convolution `conv = p∗p` with an FFT.
    - Build a soft‑max weight `w_k = exp(β·(conv_k−M))` where `M = max(conv)`.
      The temperature `β` grows from a small value up to a large value
      (≈200) so that the gradient gradually focuses on the true maximisers.
    - The sub‑gradient of the smoothed max is
          rev = Σ_k w_k·p_{k‑i}   (a correlation, computed via FFT).
      The gradient of the original objective is `grad = 4·n·rev`.
    - Perform an EG update in log‑space, optionally adding jitter to
      escape plateaus, and re‑project onto the probability simplex.
* A multiscale schedule is used: start with a modest length, optimise,
  then up‑sample the best probability vector to a larger length and repeat.
  Several random restarts are performed at each scale.
* The best sequence encountered (converted back to the original scale)
  is returned.  All operations are O(n log n) and respect the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """Linear self‑convolution p∗p using a single FFT of size N."""
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    # keep the linear part (length 2·len(p)−1)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute `rev_i = Σ_k w_k·p_{k‑i}` for i=0..n‑1.
    This is a correlation and is obtained by an FFT.
    """
    n = p.shape[0]
    L = w.shape[0]                # = 2·n−1
    P = np.fft.rfft(p, n=N)
    W_rev = np.fft.rfft(w[::-1], n=N)   # reverse w before FFT
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    # keep the linear part of length n+L‑1
    conv_rev = conv_rev[: n + L - 1]
    # extract the correlation slice and reverse it back
    rev = conv_rev[L - n : L][::-1]
    return rev


def _project_onto_simplex(x: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Result sums to 1 and is non‑negative.
    """
    # implementation of the well‑known O(n log n) algorithm
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(x)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero; return uniform
        return np.full_like(x, 1.0 / len(x))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(x - theta, 0.0)
    return w


def _optimize_length(
    n: int,
    p_start: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 0.1,
    beta_end: float = 200.0,
    beta_ramp_iters: int = 100_000,
    jitter_interval: int = 5000,
    jitter_scale: float = 0.005,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimisation for a fixed length ``n``.
    Returns the best probability vector (sum = 1) and its objective value
    ``2·n·max(p∗p)``.
    """
    # prepare FFT size that works for both the self‑convolution (2n‑1)
    # and the correlation with a weight vector (n + (2n‑1) − 1 = 3n‑2)
    N = _next_pow2(3 * n - 2)

    # initialise log‑probabilities
    p = p_start.copy()
    logp = np.log(p + 1e-30)

    best_score = float("inf")
    best_p = p.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- self‑convolution -----
        conv = _full_convolution_fft(p, N)
        M = conv.max()
        cur_score = 2.0 * n * M

        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[len {n:5d} iter {it:7d}] new local best = {cur_score:.6f}")

        # ----- temperature schedule (soft‑max) -----
        if it <= beta_ramp_iters:
            beta = beta_start + (beta_end - beta_start) * (it / beta_ramp_iters)
        else:
            beta = beta_end

        # soft‑max weights (the denominator of the soft‑max is omitted;
        # the step‑size accounts for its magnitude)
        w = np.exp(beta * (conv - M))

        # ----- weighted correlation (sub‑gradient) -----
        rev = _weighted_rev_fft(p, w, N)

        # gradient of 2·n·max_β(p∗p)
        grad = 4.0 * n * rev

        # ----- step size (decaying) -----
        eta = (1.0 / (4.0 * n)) / (1.0 + it * 1e-4)

        # ----- EG update in log‑space -----
        logp -= eta * grad

        # occasional jitter to escape flat regions
        if it % jitter_interval == 0:
            logp += np.random.randn(n) * jitter_scale

        # ----- projection onto the simplex -----
        # avoid overflow by subtracting the maximum log‑value
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)
        p = exp_shift / exp_shift.sum()

    return best_p, best_score
