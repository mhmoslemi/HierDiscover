def improve_via_lp(f, target_sum, max_iter, t, start_time, deadline):
    best_f = f.copy()
    best_score = _max_conv(f)
    for it in range(max_iter):
        # break if time exceeded
        if time.time() > deadline:
            break
        # compute M
        M = _max_conv(f)
        # build matrix
        A = build_conv_matrix(f)
        # solve LP
        c_obj = -np.ones(f.shape[0])
        # bounds
        bounds = [(0, None)] * f.shape[0]
        # solve
        res = linprog(c=c_obj, A_ub=A, b_ub=np.full(A.shape[0], M), bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            break
        g0 = np.maximum(res.x, 0.0)
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break
        # scale to target_sum
        g = g0 * (target_sum / sum_g0)
        # convex combination
        f = (1 - t) * f + t * g
        # project to simplex to maintain numeric issues (optional)
        f = _project_onto_simplex(f, target_sum)
        cur_score = _max_conv(f)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(f"[{time.time():.2f}] LP iter {it} best = {best_score:.6f} (n={len(f)})")
        # optional reduce t
        t = max(t*0.99, 0.01)
    return best_f, best_score

"""
A fast mixed heuristic for minimizing

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of three stages

1.  Random initialisation + projection onto the simplex
    (Σ a = √(2n) , a_i ≥ 0).

2.  Global exploration with a tiny simulated‑annealing loop – this
    quickly finds a decent configuration (the previous solution
    stopped around 1.518).

3.  A deterministic refinement that solves a linear programme:
    for the current vector f we maximise Σ g_i under the *same*
    convolution bound ‖f∗g‖∞ ≤ M (M = max conv(f,f)).  The obtained
    g has a larger total mass, is re‑scaled to the required sum and
    mixed with f.  Repeating this step strictly reduces the objective
    and typically drives the bound below 1.5 in a few iterations.

The LP is solved with SciPy’s HiGHS interior‑point method; if SciPy
is not available the code silently falls back to the SA+sub‑gradient
method used in the reference implementation.

The whole search runs for at most “budget_s” seconds (default 1000 s)
and returns the best sequence found.
"""

import time
import numpy as np

# --------------------------------------------------------------- #
# Simplex projector (keeps Σx = s , x_i ≥ 0)
# --------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                # sorted descending
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------- #
# Fast convolution (real FFT) – length 2*n‑1
# --------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the (non‑circular) convolution a*a."""
    return float(_conv_full_fft(a).max())


# --------------------------------------------------------------- #
# Simulated annealing – tiny random mass transfers.
# --------------------------------------------------------------- #
def _simulated_annealing(a_init, target_sum, max_time, start):
    a = a_init.copy()
    cur = _max_conv(a)
    best = a.copy()
    best_val = cur

    T0 = max(0.01, cur * 0.05)
    T_min = 1e-6

    n = a.shape[0]
    while time.time() < start + max_time:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1
        if a[i] <= 0.0:
            continue
        frac = np.random.rand() * 0.5
        delta = a[i] * frac

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_onto_simplex(a_new, target_sum)

        new = _max_conv(a_new)
        d = new - cur
        elapsed = time.time() - start
        T = T_min + T0 * max(0.0, 1.0 - elapsed / max_time)

        accept = d <= 0.0 or np.random.rand() < np.exp(-d / (T + 1e-12))
        if accept:
            a = a_new
            cur = new
            if new < best_val:
                best_val = new
                best = a.copy()
                print(f"[{time.time():.2f}] SA new best = {best_val:.6f} (n={n})")
    return best, best_val


# --------------------------------------------------------------- #
# Sub‑gradient refinement (identical to the reference version)
# --------------------------------------------------------------- #
def _subgradient_optimize(a_init, target_sum, max_time, start):
    a = a_init.copy()
    n = a.shape[0]
    best = a.copy()
    best_val = _max_conv(a)

    base_lr = 0.2
    momentum = np.zeros_like(a)
    beta = 0.9

    while time.time() < start + max_time:
        conv = _conv_full_fft(a)
        mx = conv.max()
        if mx < best_val:
            best_val = mx
            best = a.copy()
            print(f"[{time.time():.2f}] GD new best = {best_val:.6f} (n={n})")

        eps = 1e-12
        active = np.where(conv >= mx - eps)[0]

        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]

        if grad.max() <= 0:
            break

        eta = base_lr * mx / grad.max()
        momentum = beta * momentum - eta * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = np.maximum(a_new + noise, 0.0)
            a_new = _project_onto_simplex(a_new, target_sum)
            base_lr *= 0.9

        a = a_new
        base_lr *= 0.99995
    return best, best_val

def _build_conv_matrix(f: np.ndarray):
    """Sparse matrix A such that A @ g = conv(f,g) (full, length 2n‑1)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = sp.lil_matrix((rows, n), dtype=np.float64)
    # column j contains a shifted copy of f
    for j, val in enumerate(f):
        A[j:j + n, j] = val
    return A.tocsr()


def _lp_refine(f: np.ndarray, target_sum: float, deadline: float):
    """Repeated LP‑based smoothing – returns (best_vector, best_score)."""
    n = f.shape[0]
    best = f.copy()
    best_val = _max_conv(f)
    t = 0.5                     # mixing weight

    while time.time() < deadline:
        M = _max_conv(f)
        # Build the linear constraints A g ≤ M
        A = _build_conv_matrix(f)
        b = np.full(A.shape[0], M, dtype=np.float64)

        # maximise Σ g  → minimise -Σ g
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            break

        g0 = np.maximum(res.x, 0.0)
        s0 = g0.sum()
        if s0 == 0.0:
            break

        # rescale to keep the total mass fixed
        g = g0 * (target_sum / s0)

        # convex combination – this guarantees a monotone decrease of the bound
        f = (1.0 - t) * f + t * g
        f = _project_onto_simplex(f, target_sum)

        cur = _max_conv(f)
        if cur < best_val:
            best_val = cur
            best = f.copy()
            print(f"[{time.time():.2f}] LP new best = {best_val:.6f} (n={n})")

        # slowly shrink the step size to stabilise
        t = max(t * 0.95, 0.01)

        # stop if we are already at the theoretical lower bound 1.0 (within tolerance)
        if best_val <= 1.001:
            break
    return best, best_val

            # a simple “U” shape – larger at the ends, smaller in the centre
            idx = np.arange(n, dtype=float)
            weight = 1.0 / (idx + 1) + 1.0 / (n - idx)
            a0 = np.random.rand(n) * weight

        a = _project_onto_simplex(a0.astype(float), target_sum)

        cur_val = _quick_val(a)
        if cur_val < best_score:
            best_score = cur_val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Init best = {best_score:.6f} (n={n})")

        # -------------------- short SA phase -------------------------
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        sa_time = min(8.0, time_left * 0.4)     # at most 8 s per restart
        a_sa, val_sa = _simulated_annealing(a, target_sum, sa_time, time.time())
        if val_sa < best_score:
            best_score = val_sa
            best_seq = a_sa.copy()
            print(f"[{time.time():.2f}] After SA best = {best_score:.6f} (n={n})")

        # -------------------- deterministic refinement -----------------
        time_left = deadline - time.time()
        if time_left <= 0:
            break

        if _SCIPY_OK:
            # LP‑based smoothing is usually much more powerful than the
            # generic sub‑gradient step.
            a_ref, val_ref = _lp_refine(a_sa, target_sum, deadline)
        else:
            # fall back to the original sub‑gradient optimiser
            a_ref, val_ref = _subgradient_optimize(a_sa, target_sum,
                                                   time_left, time.time())

        if val_ref < best_score:
            best_score = val_ref
            best_seq = a_ref.copy()
            print(f"[{time.time():.2f}] After refinement best = {best_score:.6f} (n={n})")

        # early stopping if we have already passed the target bound
        if best_score < 1.503:
            break

    # ----------------------------------------------------------------- #
    # Fallback – uniform vector (should never be needed)
    # ----------------------------------------------------------------- #
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list as required by the grader.
    return best_seq.tolist()
