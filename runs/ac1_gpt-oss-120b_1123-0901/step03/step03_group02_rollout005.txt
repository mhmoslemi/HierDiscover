"""
Optimisation of a non‑negative step‑function to minimise

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The objective is equal to the maximal value of the self‑convolution when the
sequence is projected onto the simplex Σ a_i = √(2 n).  The code below uses a
*progressive p‑norm* smooth approximation of the ∞‑norm together with an
Adam‑style update and periodic random “shakes”.  The p‑norm exponent grows
from 2 to a large value (default 256) during a run, thereby starting with a
very smooth landscape and ending with a sharp focus on the current maximum.
Various initialisations (power‑law, exponential taper, warm‑starts from the
current best, interpolation to different lengths) are tried in many short
restarts until the allocated budget (default 1000 s) is exhausted.

The algorithm generally achieves a lower maximal convolution than the original
soft‑max method, pushing the bound towards the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – Full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())      # next power‑of‑2
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative artefacts from rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Helper – Interpolate a sequence to a new length (linear interpolation)
# ----------------------------------------------------------------------
def _interpolate_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize `seq` to length `new_n` using linear interpolation."""
    if seq.size == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, seq.size)
    new_x = np.linspace(0.0, 1.0, new_n)
    return np.interp(new_x, old_x, seq)


# ----------------------------------------------------------------------
# Helper – Simple smoothing (3‑point moving average) to reduce high‑frequency noise
# ----------------------------------------------------------------------
def _smooth_sequence(a: np.ndarray) -> np.ndarray:
    """Apply a tiny 3‑point smoothing kernel to a (edge‑replicated)."""
    kernel = np.array([0.25, 0.5, 0.25], dtype=float)
    pad = 1
    a_pad = np.pad(a, (pad, pad), mode='edge')
    smooth = np.convolve(a_pad, kernel, mode='valid')
    return smooth


# ----------------------------------------------------------------------
# Optimisation of a single length using a progressive p‑norm smooth max
# ----------------------------------------------------------------------
def _optimise_one_length_pnorm(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    p_max: float = 256.0,
    lr: float = 0.08,
    beta1: float = 0.9,
    beta2: float = 0.999,
    adam_eps: float = 1e-8,
    eps_conv: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length `n` using a progressive p‑norm
    approximation of the max.  Returns the best sequence (NumPy array) and
    its evaluation score (float, i.e. max(conv)).
    """
    target_sum = np.sqrt(2.0 * n)          # Σ a_i = √(2 n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    best_seq = a.copy()
    conv = _full_convolution_fft(a, a)
    best_score = conv.max()

    # Adam‑style accumulators
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    start = time.time()
    iteration = 0

    while time.time() < run_deadline:
        # ----- 1) current self‑convolution and its maximum -----
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()

        # ----- 2) progressive p‑norm weight vector -----
        # fraction of time elapsed within this run
        t_frac = (time.time() - start) / max(run_deadline - start, 1e-12)
        p = 2.0 + (p_max - 2.0) * t_frac
        # weight proportional to (c_i)^{p‑1}
        w = (conv + eps_conv) ** (p - 1.0)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- 3) gradient of Σ w_k * c_k (cross‑correlation) -----
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)      # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)              # positions for gradient
        grad = 2.0 * conv2[idx]

        # ----- 4) Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** (iteration + 1))
        v_hat = v / (1.0 - beta2 ** (iteration + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # optional clipping of giant steps (protect against overshoot)
        max_step = 0.2 * target_sum / max(n, 1.0)   # roughly 20 % of the average entry
        cur_max = np.max(np.abs(step))
        if cur_max > max_step:
            step = step * (max_step / cur_max)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional random perturbation to escape plateaus
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (target_sum * 5e-4 / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)

        # small smoothing every few iterations (helps shape regularity)
        if iteration % 13 == 0:
            a_new = _smooth_sequence(a_new)
            a_new = _project_onto_simplex(a_new, target_sum)

        # ----- 5) evaluate candidate -----
        conv_new = _full_convolution_fft(a_new, a_new)
        score_new = conv_new.max()

        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  NEW local best = {best_score:.6f}")

        a = a_new
        iteration += 1

    # final verification using the provided evaluator (identical to max conv)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, float(final_score)


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the given
    evaluation function.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.

    Strategy
    --------
    * Randomly pick a length `n` (≈ 800 … 3500) and initialise a shape that
      is a power‑law, optionally tapered with an exponential factor.
    * Occasionally warm‑start from the current global best (truncated or
      interpolated to the new length).
    * Optimise each restart with the progressive‑p‑norm Adam method
      defined in ``_optimise_one_length_pnorm``.
    * Keep the overall best across all restarts; stop early if the bound
      drops below the target 1.503.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a previously known construction if present.
    # ------------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        score = evaluate_sequence(hs_arr.tolist())
        if score < best_score:
            best_score = float(score)
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start from global seed = {best_score:.6f}")
    except Exception:
        pass  # no previous construction available

    # ------------------------------------------------------------------
    # Main restart loop – each restart receives a modest time slice.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break

        # Choose problem size.  The schedule favours medium sizes later on.
        if remaining > 300:
            n = np.random.randint(800, 3500 + 1)
        elif remaining > 120:
            n = np.random.randint(600, 3000 + 1)
        else:
            n = np.random.randint(400, 2200 + 1)

        # ------------------- initial shape -----------------------------
        p_exp = np.random.uniform(0.3, 0.7)           # power‑law exponent
        idx = np.arange(1, n + 1, dtype=float)
        init = idx ** (-p_exp)

        # optional exponential taper
        if np.random.rand() < 0.35:
            tau_exp = np.random.uniform(0.4, 2.0) * n
            init *= np.exp(-idx / tau_exp)

        init += np.random.rand(n) * 1e-4   # tiny jitter

        # occasional warm‑start from the current global best
        if best_seq is not None and np.random.rand() < 0.45:
            if best_seq.size >= n:
                init = best_seq[:n].copy()
            else:
                init = _interpolate_sequence(best_seq, n)

        # allocate a time slice for this restart (≈ 6‑8 s or half of the budget)
        slice_time = min(8.0, remaining * 0.5)
        run_deadline = min(deadline, time.time() + slice_time)

        # run the optimiser for this length
        seq_local, score_local = _optimise_one_length_pnorm(
            n,
            init,
            run_deadline,
            p_max=256.0,
            lr=0.08,
            beta1=0.9,
            beta2=0.999,
            adam_eps=1e-8,
            eps_conv=1e-12,
        )

        # update the global best if improvement was found
        if score_local < best_score:
            best_score = float(score_local)
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (length={n})"
            )
            # early exit if we already beat the target bound
            if best_score < 1.503:
                print("[INFO] Target bound reached – exiting early.")
                break

    # ------------------------------------------------------------------
    # Fallback in case nothing was generated (should not happen).
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list as required by the evaluator.
    return best_seq.tolist()
