    beta2 = 0.999

def _softmax_gradient(a, w):
    # a length n, w length L=2n-1
    # compute correlation a ⋆ w
    n = a.shape[0]
    L = w.shape[0]  # should be 2n-1
    # reverse w
    w_rev = w[::-1]
    # full convolution length n+L-1 = 3n-2

    idx_start = L - 1
    # gather gradient

"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm
---------
* Work on several problem sizes (different lengths).  For each length we
  run a smooth‑gradient Adam optimiser.
* The objective (the maximal convolution value) is approximated by a
  soft‑max:   L_t(a) = (1/t)·log(∑_k exp(t·(a*a)_k)).  Its gradient is a
  cheap correlation which we compute with FFTs.
* Temperature *t* starts small (high smoothing) and is increased
  gradually; as t → ∞ the gradient becomes a sub‑gradient of the true
  max, so the optimiser automatically switches to a “hard’’ regime.
* After the Adam phase a very cheap stochastic local search (mass moves
  between two random positions) tries to improve the solution further.
* Multiple random restarts, occasional jitter and a simple schedule for
  the learning rate keep the search from stagnating.
* The best vector seen over all lengths is returned (after normalising
  the L1‑sum to √(2 n) – this does not change the objective because the
  evaluation function is scale‑invariant).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _full_conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Real FFT based full convolution of one‑dimensional real arrays ``a``
    and ``b``.  Returns the (n+m‑1)-length result.
    """
    n, m = a.shape[0], b.shape[0]
    N = _next_pow_two(n + m - 1)
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(A * B, n=N)
    return conv[: n + m - 1]


# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Returns (max value, full convolution) of ``a`` with itself.
    """
    conv = _full_conv_fft(a, a)               # length = 2*n-1
    return float(conv.max()), conv


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate  Σ_k w_k (a*a)_k  w.r.t. ``a``.
    ``w`` must have length 2*n‑1 and be non‑negative, sum(w)=1.
    The formula reduces to the correlation   g = 2·(a ⋆ w).
    """
    n = a.shape[0]
    L = w.shape[0]                     # = 2*n-1
    # correlation via convolution with the reversed weight vector
    w_rev = w[::-1]
    conv_full = _full_conv_fft(a, w_rev)    # length = n+L-1 = 3*n-2
    # indices that correspond to the correlation (see analysis)
    idx_start = L - 1                     # = 2*n-2
    grad = 2.0 * conv_full[idx_start - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
def _smooth_adam_optimize(init_a: np.ndarray,
                          n: int,
                          target_sum: float,
                          deadline: float,
                          max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the maximal autocorrelation using a
    temperature‑controlled softmax approximation.
    Returns the best vector and its true maximal convolution value.
    """
    a = init_a.astype(float).copy()
    # initialise – enforce the normalisation that the original code uses
    s = a.sum()
    if s > 0:
        a *= target_sum / s
    else:
        a.fill(target_sum / n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.06

    # temperature (smoothness) schedule
    t = 0.6                     # start fairly smooth
    t_max = 60.0                # never let it explode
    temp_factor = 1.015

    best_vec = a.copy()
    best_val = np.inf

    iteration = 0
    jitter_every = 2000

    while time.time() < deadline and iteration < max_iters:
        iteration += 1

        # ---- convolution and softmax weights --------------------------------
        M, conv = _conv_max_fft(a)                # conv length = 2*n-1
        # soft‑max (stable version)
        tmp = t * (conv - M)                       # max(conv) = M
        w = np.exp(tmp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ---- gradient -------------------------------------------------------
        grad = _softmax_gradient(a, w)

        # ---- Adam step -------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # simple decay for the learning rate
        lr = base_lr / (1.0 + 1e-5 * iteration)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- projection onto feasible set (non‑negative, correct L1) -------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a *= target_sum / s
        else:
            a.fill(target_sum / n)

        # ---- temperature increase -------------------------------------------
        t = min(t * temp_factor, t_max)

        # ---- occasional jitter ------------------------------------------------
        if iteration % jitter_every == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= target_sum / a.sum()

        # ---- keep best according to the true max ------------------------------
        if M < best_val:
            best_val = M
            best_vec = a.copy()
            print(f"[len {n:5d} iter {iteration:7d}] new best M = {best_val:.6f}")

    return best_vec, best_val


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  n: int,
                  target_sum: float,
                  steps: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap greedy local search that repeatedly moves a tiny amount of
    mass from one index to another if it reduces the maximal convolution.
    """
    best = a.copy()
    best_val, _ = _conv_max_fft(best)

    for step in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # move a small fraction of the source entry
        delta = best[i] * 0.02
        if delta <= 0.0:
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        # sum is unchanged, only need to enforce non‑negativity
        cand = np.maximum(cand, 0.0)

        M, _ = _conv_max_fft(cand)
        if M < best_val:
            best = cand
            best_val = M

    return best, best_val

            sub_deadline = time.time() + max_time

            # ----- smooth Adam optimisation ------------------------------------
            opt_seq, opt_val = _smooth_adam_optimize(
                init, n, target_sum, sub_deadline, max_iters=200_000
            )

            # ----- cheap local refinement ---------------------------------------
            if time.time() < deadline:
                ls_seq, ls_val = _local_search(
                    opt_seq, n, target_sum, steps=30_000, deadline=deadline
                )
                if ls_val < opt_val:
                    opt_seq, opt_val = ls_seq, ls_val

            # ----- update the global best ---------------------------------------
            if opt_val < best_val:
                best_val = opt_val
                best_seq = opt_seq.copy()
                print(f"[global] new best M = {best_val:.6f} (len={n})")

    # ------------------------------------------------------------------
    # Final sanity check – evaluate through the official function
    if best_seq is None:
        # should never happen, but fall back to a uniform vector
        best_seq = np.full(2100, 1.0, dtype=float)
        best_seq *= math.sqrt(2.0 * 2100) / best_seq.sum()

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[done] best reported M = {best_val:.6f}, official score = {final_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
