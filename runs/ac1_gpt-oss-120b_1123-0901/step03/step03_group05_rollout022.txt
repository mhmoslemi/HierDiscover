def project_to_simplex(a, target):
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0: a = a * (target / s)
    else: a = np.full_like(a, target / len(a))
    return a

def monotone_decreasing(a):
    n=len(a)
    # PAVA for decreasing
    # we invert sign to use increasing algorithm
    x = -a
    # apply isotonic regression to get increasing x
    # Use simple PAVA
    # each block has sum and count
    sum_block = []
    cnt_block = []
    for v in x:
        sum_block.append(v)
        cnt_block.append(1)
        # while last two block averages violate monotonicity
        while len(sum_block) >= 2 and sum_block[-2] / cnt_block[-2] > sum_block[-1] / cnt_block[-1]:
            # merge last two
            sb = sum_block[-2] + sum_block[-1]
            cb = cnt_block[-2] + cnt_block[-1]
            sum_block[-2] = sb
            cnt_block[-2] = cb
            sum_block.pop()
            cnt_block.pop()
    # Now expand blocks to values
    out = np.empty(n, dtype=float)
    idx = 0
    for s,c in zip(sum_block, cnt_block):
        avg = s / c
        out[idx:idx+c] = avg
        idx += c
    return -out  # revert sign -> decreasing
```

"""
Monotone‑projected sub‑gradient optimiser with an optional
simulated‑annealing (soft‑max) phase and a cheap local‑search.
The optimiser works for any length (up to ~8000) and keeps the L1
norm fixed to √(2 n).  By enforcing a monotone shape we can dramatically
lower the maximal value of the a*a convolution, which is exactly the
quantity returned by `evaluate_sequence`.  The routine explores a few
different lengths and several random restarts; the best sequence found
before the 1000 s budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """Full convolution a*a via FFT; return max value and the full array."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    C = A * A
    conv = np.fft.irfft(C, n=N)[: 2 * n - 1]
    return float(conv.max()), conv


def _conv_fft(a: np.ndarray, b: np.ndarray):
    """Convolution of two real vectors via FFT (full mode)."""
    n, m = a.shape[0], b.shape[0]
    N = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b, n=N)
    C = A * B
    return np.fft.irfft(C, n=N)[: n + m - 1]


def _rev_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    """Vector rev where rev[i] = a[k‑i] if the index is inside the support."""
    n = a.shape[0]
    rev = np.zeros_like(a)
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    rev[mask] = a[j[mask]]
    return rev


def _monotone_decreasing(a: np.ndarray) -> np.ndarray:
    """
    Project onto the cone of non‑negative, monotone decreasing vectors
    (isotonic regression with decreasing order).  Returns a new array.
    """
    # PAVA on the negative values to get decreasing order.
    x = -a.astype(float)
    sum_block = []
    cnt_block = []
    for v in x:
        sum_block.append(v)
        cnt_block.append(1)
        # merge while averages are decreasing (i.e. original a is increasing)
        while len(sum_block) >= 2 and (sum_block[-2] / cnt_block[-2]) > (sum_block[-1] / cnt_block[-1]):
            sb = sum_block[-2] + sum_block[-1]
            cb = cnt_block[-2] + cnt_block[-1]
            sum_block[-2] = sb
            cnt_block[-2] = cb
            sum_block.pop()
            cnt_block.pop()
    # expand the blocks
    out = np.empty_like(a)
    pos = 0
    for s, c in zip(sum_block, cnt_block):
        avg = s / c
        out[pos: pos + c] = avg
        pos += c
    return -out


def _subgradient_descent(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Projected sub‑gradient descent for `max(a*a)`.
    Returns the best vector found and its max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_M = float("inf")
    it = 0
    # base step size (controls the L2 magnitude of each move)
    base_step = 2.0  # will be multiplied by target_sum / ||grad||
    jitter_every = 3000
    random_active_prob = 0.4

    while time.time() < deadline and it < 2_000_000:
        it += 1
        M, conv = _conv_max_fft(a)

        # keep the best ever seen for this run
        if M < best_M:
            best_M = M
            best_a = a.copy()
            print(f"[subgrad n={n} it={it}] new best M = {best_M:.6f}")

        # active set (indices achieving (almost) the maximum)
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # ---------------- sub‑gradient computation ----------------
        if active.size == 0:
            # should never happen, but guard against it
            rev = np.zeros_like(a)
        elif active.size == 1 or np.random.rand() > random_active_prob:
            # pick a single active shift (fast)
            k = active[np.random.randint(active.size)]
            rev = _rev_for_shift(a, k)
        else:
            # average over all active shifts – cheap via FFT
            w = np.zeros(2 * n - 1, dtype=float)
            w[active] = 1.0 / active.size
            # convolution a * reversed(weights)
            rev_full = _conv_fft(a, w[::-1])
            rev = rev_full[2 * n - 2 - np.arange(n)]

        # gradient of max‑convolution (sub‑gradient) = 2 * rev
        grad = 2.0 * rev

        # ------------------- update ----------------------------
        # step magnitude = base_step * target_sum / ||grad||
        step = base_step * target_sum / (np.linalg.norm(grad) + 1e-12)
        # optional decay with iteration count
        step = step / math.sqrt(it)

        a = a - step * grad

        # ---------- projection onto feasible set ----------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # ---------- enforce monotone decreasing shape ----------
        a = _monotone_decreasing(a)

        # ---------- occasional jitter to avoid plateaus ----------
        if it % jitter_every == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            s = a.sum()
            if s > 0.0:
                a = a * (target_sum / s)

    # final evaluation of the best vector found
    final_M, _ = _conv_max_fft(best_a)
    return best_a, float(final_M)


def _local_search(a_start: np.ndarray, target_sum: float,
                  init_score: float, deadline: float,
                  max_proposals: int = 30_000):
    """
    Very cheap hill‑climb that repeatedly moves a small amount of mass
    from one index to another while keeping the L1 norm fixed.
    Accepts a move only if the maximal convolution value decreases.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_score = init_score
    proposals = 0
    while time.time() < deadline and proposals < max_proposals:
        proposals += 1
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        # choose a different receiver
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        # move up to 5 % of the donor's mass
        eps = a[i] * np.random.rand() * 0.05
        if eps <= 0.0:
            continue
        new_vec = a.copy()
        new_vec[i] -= eps
        new_vec[j] += eps
        # non‑negativity is guaranteed by construction
        M_new, _ = _conv_max_fft(new_vec)
        if M_new < best_score:
            a = new_vec
            best_score = M_new
            print(f"[local] improvement to {best_score:.6f} (proposal {proposals})")
    return a, float(best_score)
