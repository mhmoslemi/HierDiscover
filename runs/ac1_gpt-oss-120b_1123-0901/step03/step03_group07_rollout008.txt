    # returns projection of vector v onto probability simplex (nonnegative sum 1)
    u = np.sort(v)[::-1]

"""
Propose a candidate step‑function (non‑negative heights) that minimizes

    F(a) = 2·n·max_k (a∗a)[k] / (∑a_i)²

The algorithm works in three stages:

1. **Parameter sweep** – generate a diverse collection of well‑known
   window functions (Kaiser, Gaussian, Hann, Hamming, Blackman, etc.),
   a few random Dirichlet draws and the historic solution
   ``height_sequence_1`` (if present).  All candidates are normalised
   to sum‑to‑1 and evaluated with a fast FFT based routine; the best
   few are kept.

2. **Projected sub‑gradient descent** – starting from each of the
   selected seeds we minimise the peak of the auto‑convolution.
   For a given shift `k* = argmax (a∗a)` the sub‑gradient of the
   maximum is `∇max = 2·a` reversed and shifted.  After a gradient step
   we project back onto the probability simplex (Euclidean projection).
   The step‑size decays geometrically; occasional “shakes’’ (tiny random
   perturbations) help to escape shallow plateaus.

3. **Return the best sequence found** – the sequence with the lowest
   exact objective (computed with the same FFT routine) is converted
   to a plain Python list and returned.

The whole search respects a 1000 s wall‑clock budget, adapts the
iteration budget to the remaining time and prints progress messages.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities -------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(a: np.ndarray, size: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length ``size``.
    Returns a real array of length 2*len(a)-1.
    """
    n = a.shape[0]
    if size is None:
        size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    # numerical noise may produce tiny negatives – clip them
    np.maximum(conv, 0.0, out=conv)
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex:
    {x | x_i ≥ 0, sum x_i = 1}.
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    # find the largest rho satisfying the condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_fft(a: np.ndarray, size_fft: int = None) -> float:
    """
    Exact objective 2·n·max(a*a) / (sum(a)^2) using FFT.
    The vector ``a`` is assumed non‑negative; its sum need not be 1.
    """
    n = a.shape[0]
    conv = _full_convolution_fft(a, size=size_fft)
    max_b = conv.max()
    sum_a = a.sum()
    # Guard against division‑by‑zero (should never happen after projection)
    if sum_a < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a low‑value step‑function within the allotted time.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose a length that balances expressivity and speed.
    # Empirically n around 4000 works well inside the 1000 s budget.
    n = 4000
    size_fft = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 1) Build a pool of initial candidates (parameter sweep)
    candidates = []

    # a) historic solution, if the variable exists in the global scope
    try:
        hist = np.asarray(height_sequence_1, dtype=float)
        if hist.size == n:
            candidates.append(hist / hist.sum())
        else:
            # resize by simple linear interpolation if lengths differ
            idx = np.linspace(0, hist.size - 1, n)
            interpolated = np.interp(idx, np.arange(hist.size), hist)
            candidates.append(interpolated / interpolated.sum())
    except NameError:
        pass

    # b) Kaiser windows with a wide range of beta values
    for beta in np.linspace(0.0, 12.0, 20):
        w = np.kaiser(n, beta)
        candidates.append(w / w.sum())

    # c) Gaussian windows (sigma as a fraction of half‑length)
    half = (n - 1) / 2.0
    for sigma_frac in np.linspace(0.10, 0.45, 15):
        sigma = sigma_frac * half
        i = np.arange(n, dtype=float)
        w = np.exp(-0.5 * ((i - half) / sigma) ** 2)
        candidates.append(w / w.sum())

    # d) Sine‑power families (half‑sine raised to a power)
    for p in np.linspace(0.5, 4.0, 15):
        i = np.arange(1, n + 1, dtype=float)
        w = np.sin(np.pi * i / (n + 1)) ** p
        candidates.append(w / w.sum())

    # e) Classical raised‑cosine windows
    i = np.arange(n, dtype=float)
    # Hann
    w = 0.5 - 0.5 * np.cos(2 * np.pi * i / (n - 1))
    candidates.append(w / w.sum())
    # Hamming
    w = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    candidates.append(w / w.sum())
    # Blackman
    w = (0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4 * np.pi * i / (n - 1)))
    candidates.append(w / w.sum())

    # f) Simple linear ramps (ascending and descending)
    w = np.arange(1, n + 1, dtype=float)
    candidates.append(w / w.sum())
    w = np.arange(n, 0, -1, dtype=float)
    candidates.append(w / w.sum())

    # g) Random Dirichlet draws
    for _ in range(30):
        w = np.random.dirichlet(np.ones(n))
        candidates.append(w)

    # --------------------------------------------------------------
    # Evaluate all candidates quickly and keep the best few
    best_val = float("inf")
    best_seq = None
    top_candidates = []          # store (val, seq) tuples
    for idx, a in enumerate(candidates):
        if time.time() >= deadline:
            break
        val = _objective_fft(a, size_fft=size_fft)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        # keep the 8 best seeds for the second phase
        top_candidates.append((val, a))
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:8]   # keep only a handful

    print(f"[Init] Best pre‑search objective = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2) Projected sub‑gradient descent from each top seed
    # --------------------------------------------------------------
    # Common hyper‑parameters for the descent
    base_lr = 0.12                # initial learning‑rate
    decay = 0.9995                # per‑iteration decay
    shake_interval = 1.5          # seconds without improvement before shaking
    max_iters_per_seed = 20000    # hard cap per seed (will be cut by budget)

    global_best_val = best_val
    global_best_seq = best_seq.copy() if best_seq is not None else None

    for seed_id, (seed_val, a0) in enumerate(top_candidates, start=1):
        if time.time() >= deadline:
            break
        a = a0.copy()
        # ensure we start exactly on the simplex (sum = 1)
        a = _project_simplex(a)
        lr = base_lr
        last_improve_time = time.time()
        best_local_val = _objective_fft(a, size_fft=size_fft)

        for it in range(1, max_iters_per_seed + 1):
            # ---- stop if out of time ------------------------------------------------
            if time.time() >= deadline:
                break

            # ---- compute self‑convolution and locate the peak ------------------------
            conv = _full_convolution_fft(a, size=size_fft)
            max_b = conv.max()
            k_max = int(conv.argmax())                      # shift where the peak occurs

            # ---- sub‑gradient of the peak (2·a reversed & shifted) -------------------
            grad = np.zeros_like(a)
            j_start = max(0, k_max - (n - 1))
            j_end = min(n - 1, k_max)
            # indices where the gradient is non‑zero
            idxs = np.arange(j_start, j_end + 1, dtype=int)
            grad[idxs] = 2.0 * a[k_max - idxs]

            # ---- gradient step + projection onto simplex -------------------------------
            a_proposed = a - lr * grad
            a = _project_simplex(a_proposed)

            # ---- learning‑rate schedule ------------------------------------------------
            lr *= decay

            # ---- periodic evaluation ---------------------------------------------------
            if it % 25 == 0 or it == max_iters_per_seed:
                cur_val = _objective_fft(a, size_fft=size_fft)
                if cur_val < best_local_val:
                    best_local_val = cur_val
                    last_improve_time = time.time()
                if cur_val < global_best_val:
                    global_best_val = cur_val
                    global_best_seq = a.copy()
                    print(f"[Seed {seed_id:02d}] iter {it:05d}  new global best = {cur_val:.6f}")

            # ---- shake if we have been stagnant ---------------------------------------
            if time.time() - last_improve_time > shake_interval:
                # tiny random perturbation in log‑space → keep on simplex
                perturb = (np.random.rand(n) - 0.5) * 5e-4
                a = _project_simplex(a + perturb)
                last_improve_time = time.time()
                # reset learning‑rate a bit to escape possible flat region
                lr = base_lr * 0.5

        print(f"[Seed {seed_id:02d}] finished – best local objective = {best_local_val:.6f}")

    # --------------------------------------------------------------
    # 3) Return the best sequence found (converted to list)
    if global_best_seq is None:
        # fallback – uniform distribution
        global_best_seq = np.full(n, 1.0 / n)
        global_best_val = _objective_fft(global_best_seq, size_fft=size_fft)

    print(f"Final best objective = {global_best_val:.6f} (length {n})")
    # The evaluation routine will automatically clamp negatives and overly large values,
    # but we keep the sequence well‑behaved (non‑negative, sum≈1).
    return global_best_seq.tolist()
