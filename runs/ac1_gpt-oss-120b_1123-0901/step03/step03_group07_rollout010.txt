"""
propose_candidate – hybrid LP + sub‑gradient optimizer

The routine starts from the best known step function (the global
`height_sequence_1` if it exists, otherwise a random Dirichlet draw) and
iteratively improves it:

1. **Linear‑programming refinement**  
   For the current sequence `f` we compute `M = max_k (f*f)[k]`.  
   Holding `f` fixed, we solve the linear program

       maximise   Σ_j g_j
       subject to g ≥ 0
                  (f * g)[k] ≤ M   for a selected set of shifts k

   where `(f*g)[k]` is linear in `g`.  Because the LP only uses the `K`
   largest convolution entries of `f*f` (default `K=200`) the problem
   stays small.  The optimal `g` tends to concentrate its mass where `f`
   is small, thus offering a chance to lower the peak after mixing.

   The solution `g₀` is rescaled to the required normalisation
   `Σ g = √(2n)` and mixed with the current `f`:
       f ← (1‑t)·f + t·g     (t≈0.25).

   The process is repeated while the objective improves.

2. **Sub‑gradient polishing**  
   After the LP phase we run a projected sub‑gradient descent on the
   true objective `F(a)=2·n·max(a*a)/(∑a)²`.  At each step the gradient
   of the active convolution peak is computed analytically, a step is
   taken, negativity is clipped, and the vector is projected back onto
   the simplex with the prescribed sum.  Occasional tiny random
   perturbations (“shakes”) help to escape shallow plateaus.

Both stages respect the 1000 s wall‑clock budget, print progress and
return the best sequence (as a plain Python list) found so far.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT (fast for n≈3000)."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective used by the evaluator (scale‑invariant)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        theta = cssv[-1] / len(v)
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _solve_lp_max_sum(f: np.ndarray, M: float, topK: int = 200,
                      max_iters: int = 5000) -> np.ndarray | None:
    """
    Solve  max Σ g_j   s.t.   g ≥ 0, (f * g)[k] ≤ M for the `topK`
    largest entries of the self‑convolution of `f`.  Returns the optimal
    g as a dense NumPy array or None if the solver fails.
    """
    n = f.shape[0]
    conv_ff = _full_convolution(f)
    # indices of the K largest convolution values
    K = min(topK, len(conv_ff))
    top_idx = np.argpartition(-conv_ff, K)[:K]

    g = cp.Variable(n)
    constraints = [g >= 0]

    for k in top_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1)
        idx_g = k - idx_i                      # corresponding g‑indices
        # Linear expression Σ f_i * g_{k-i}
        expr = cp.sum(cp.multiply(f[idx_i], g[idx_g]))
        constraints.append(expr <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=max_iters, verbose=False)
    except Exception:          # safety net for unexpected solver crashes
        return None

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None
    return np.asarray(g.value).reshape(-1)

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a non‑negative step‑function (list of heights) with a low
    evaluation score.  The algorithm mixes a small LP refinement loop
    with a projected sub‑gradient polishing phase.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 3000 is a good trade‑off between expressiveness
    # and runtime for the available budget.
    n = 3000
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initialise from the globally provided best construction if possible.
    try:
        init = np.asarray(height_sequence_1, dtype=float)
        if init.size != n:
            raise ValueError
        f = init.copy()
    except Exception:
        # fall back to a random Dirichlet start
        f = np.random.dirichlet(np.ones(n))

    # Normalise to the prescribed ℓ₁‑norm
    f = f / f.sum() * target_sum

    best_seq = f.copy()
    best_val = _objective(f)
    print(f"[Init] objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  LP refinement phase
    outer = 0
    mix_ratio = 0.25          # t in the description
    topK = 200                # number of active convolution constraints

    while time.time() < deadline and outer < 15:
        outer += 1
        conv_ff = _full_convolution(f)
        M = conv_ff.max()

        # Solve the LP: maximise Σ g  s.t. (f*g)[k] ≤ M for topK shifts
        g0 = _solve_lp_max_sum(f, M, topK=topK, max_iters=3000)
        if g0 is None:
            print(f"[LP {outer}] solver failed – exiting LP phase.")
            break

        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            print(f"[LP {outer}] degenerate solution – exiting LP phase.")
            break

        # Rescale to keep the ℓ₁‑norm fixed, then mix with the current f
        g = g0 * (target_sum / sum_g0)
        f_candidate = (1.0 - mix_ratio) * f + mix_ratio * g
        f_candidate = _project_to_simplex(f_candidate, target_sum)

        val = _objective(f_candidate)
        print(f"[LP {outer:02d}] M={M:.3e}, Σg={sum_g0:.3f}, "
              f"value={val:.6f}")

        if val < best_val:
            best_val = val
            best_seq = f_candidate.copy()
            print(f"   ** new best after LP {outer}")

        # Accept the candidate for the next round
        f = f_candidate

        # Stop early if improvement stalls
        if outer > 1 and best_val / val > 1.001:
            print("[LP] improvement very small – terminating LP loop.")
            break

    # ------------------------------------------------------------------
    # 2️⃣  Sub‑gradient polishing phase
    it = 0
    last_improve = time.time()
    init_lr = 0.6
    decay = 0.9994

    while time.time() < deadline and it < 12000:
        it += 1
        # Compute full convolution and the active shift
        conv = _full_convolution(f)
        k_max = np.argmax(conv)
        shift = k_max - (n - 1)                # signed offset
        # Gradient of the active peak (see analysis)
        grad = np.zeros_like(f)
        if shift >= 0:
            s = shift
            grad[:n - s] += f[s:]
            grad[s:] += f[:n - s]
        else:
            s = -shift
            grad[s:] += f[:n - s]
            grad[:n - s] += f[s:]

        # Step‑size schedule
        lr = init_lr * (decay ** it)

        # Gradient step, enforce non‑negativity, project back to simplex
        f_new = f - lr * grad
        f_new = np.maximum(f_new, 0.0)
        f_new = _project_to_simplex(f_new, target_sum)

        # Occasionally add a tiny random shake to avoid stagnation
        if it % 800 == 0:
            eps = 1e-4
            perturb = (np.random.rand(n) - 0.5) * eps
            f_new = np.exp(np.log(f_new + 1e-30) + perturb)
            f_new = _project_to_simplex(f_new, target_sum)

        # Evaluate every few iterations
        if it % 20 == 0:
            val = _objective(f_new)
            if val < best_val:
                best_val = val
                best_seq = f_new.copy()
                last_improve = time.time()
                print(f"[SGD {it:05d}] new best = {val:.6f}")

        # Simple stopping condition if no improvement for a while
        if time.time() - last_improve > 2.5:
            # restart from the best seen so far with a small random kick
            f = best_seq + np.random.normal(scale=1e-4, size=n)
            f = np.maximum(f, 0.0)
            f = _project_to_simplex(f, target_sum)
            last_improve = time.time()
            print("[SGD] stagnation – restarting from best")
            continue

        f = f_new

    # ------------------------------------------------------------------
    print(f"Finished. Best objective = {best_val:.6f} (length {n})")
    return best_seq.tolist()
