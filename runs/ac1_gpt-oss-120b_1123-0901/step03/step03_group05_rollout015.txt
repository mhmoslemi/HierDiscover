import math, time, numpy as np

def compute_full_conv(a):
    n = a.shape[0]
    N = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A*A, n=N)[:2*n-1]
    return conv

def grad_smoothed_max(a, conv, gamma):
    maxc = conv.max()
    # compute softmax weights
    expvals = np.exp(gamma * (conv - maxc))
    sum_exp = expvals.sum()
    w = expvals / sum_exp
    # convolution of a and w
    # length of w = 2n-1
    n = a.shape[0]
    N = 1 << ((n + (2*n-1) - 1).bit_length())  # length for convolution a (n) + w (2n-1)
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(w, n=N)
    grad_full = np.fft.irfft(A*B, n=N)[:n]
    return grad_full

def project_simplex(a, target_sum):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a = a * (target_sum / s)
    else:
        a = np.full_like(a, target_sum / a.shape[0])
    return a

def optimize_length(n, time_deadline):
    target_sum = math.sqrt(2.0 * n)
    # initialize with arcsine shape
    idx = np.arange(n, dtype=float)
    # shift to avoid zero denominator
    eps = 1e-12
    a0 = 1.0 / np.sqrt((idx + 0.5) * (n - 0.5 - idx) + eps)
    a0 = project_simplex(a0, target_sum)

    # hyperparameters
    max_iters = 200_000  # may be limited by time
    init_lr = 0.5
    gamma_start = 0.5
    gamma_end = 30.0

    a = a0.copy()
    best_a = a.copy()
    best_M = float('inf')
    iteration = 0

    while time.time() < time_deadline and iteration < max_iters:
        conv = compute_full_conv(a)
        M = conv.max()
        if M < best_M:
            best_M = M
            best_a = a.copy()
            print(f"[len {n}] iter {iteration} new best M = {best_M:.6f}")
        # schedule
        gamma = gamma_start + (gamma_end - gamma_start) * iteration / max_iters
        # compute gradient
        grad = grad_smoothed_max(a, conv, gamma)
        # step size schedule
        lr = init_lr * (1.0 - iteration / max_iters) + 1e-4
        a = a - lr * grad
        # projection onto non-negative simplex
        a = project_simplex(a, target_sum)
        iteration += 1

    # Local simulated annealing refinement
    a = best_a.copy()
    best_M = best_M
    temperature = 0.05
    cooling = 0.9995
    proposals = 0
    while time.time() < time_deadline and proposals < 200_000:
        proposals += 1
        # choose donor i with mass > 0
        i = np.random.randint(n)
        if a[i] < 1e-12:
            continue
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        # transfer eps
        eps_transfer = a[i] * np.random.rand() * 0.02
        new_a = a.copy()
        new_a[i] -= eps_transfer
        new_a[j] += eps_transfer
        new_a = project_simplex(new_a, target_sum)
        conv_new = compute_full_conv(new_a)
        M_new = conv_new.max()
        if M_new < best_M:
            a = new_a
            best_M = M_new
            best_a = new_a.copy()
            print(f"[len {n}] SA improvement M={best_M:.6f} (prop {proposals})")
        else:
            # accept with prob
            if temperature > 0 and np.random.rand() < math.exp(-(M_new - best_M) / temperature):
                a = new_a
        temperature *= cooling
        if temperature < 1e-6:
            temperature = 1e-6

    return best_a, best_M

Propose a step‑function (non‑negative sequence) that minimises the
evaluate_sequence objective.

Algorithm
---------
1. Choose a relatively large length (default 8000).  Larger n gives a

   mass from a random donor index to a random receiver.  Moves that
   reduce the maximal convolution are always accepted; otherwise they are
   accepted with probability exp(‑Δ/T) with a temperature that cools
   geometrically.

5. Several restarts (arcsine‑initialised and random Dirichlet) are tried
   until the time budget (≈ 1000 s) is exhausted.  The best sequence found
   (smallest value of evaluate_sequence) is returned.

The whole procedure uses only NumPy and the supplied `evaluate_sequence`
function, runs completely in‑memory, and respects the 2‑CPU limit.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """
    Full (linear) autocorrelation a*a computed with FFT.
    Returns the complete convolution array of length 2*len(a)-1.
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return conv


def _conv_two_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of a (length n) with b (length m) using FFT.
    Returns array of length n+m-1.
    """
    n, m = a.shape[0], b.shape[0]
    N = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(A * B, n=N)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Projects onto the set {x ≥ 0, Σ x = target_sum}.
    Simple scaling after cutting negative entries is sufficient because the
    feasible set is a simplex intersected with the non‑negative orthant.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v = v * (target_sum / s)
    else:
        v = np.full_like(v, target_sum / v.shape[0])
    return v


# ----------------------------------------------------------------------
def _grad_smoothed_max(a: np.ndarray, conv: np.ndarray, gamma: float) -> np.ndarray:
    """
    Gradient of the smoothed max function
        Lγ(a) = (1/γ)·log Σ_k exp(γ·c_k) ,   c = a*a .
    The gradient is  (a * w) where w is the soft‑max of the convolution
    values (∑w_k = 1).  Only the first n entries of the full convolution
    are needed.
    """
    max_c = conv.max()
    # soft‑max weights (stable)
    exp_vals = np.exp(gamma * (conv - max_c))
    w = exp_vals / exp_vals.sum()                     # length 2n‑1

    # gradient = a * w   (linear convolution, take first n entries)
    grad_full = _conv_two_fft(a, w)                   # length 3n‑2
    grad = grad_full[: a.shape[0]]
    return grad


# ----------------------------------------------------------------------
def _smooth_adam_optimize(
    a_init: np.ndarray,
    n: int,
    target_sum: float,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    max_iter: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation with the smoothed‑max gradient described above.
    Returns the best sequence found (and its max‑convolution value).
    """
    a = a_init.copy()
    a = a * (target_sum / a.sum())       # enforce the ℓ1‑norm

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.3

    # smoothing schedule
    gamma_start, gamma_end = 0.5, 30.0

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # 1) current convolution and max‑value
        conv = _conv_full_fft(a)
        cur_max = conv.max()

        # 2) keep track of the best seen so far
        if cur_max < best_score:
            best_score = cur_max
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:7d}] new best max = {best_score:.6f}")

        # 3) smoothing parameter (slowly increase)
        gamma = gamma_start + (gamma_end - gamma_start) * it / max_iter

        # 4) gradient of the smoothed max
        grad = _grad_smoothed_max(a, conv, gamma)

        # 5) Adam update (descent direction = -grad)
        lr = lr0 / (1.0 + 1e-4 * it)            # simple decay
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # 6) projection onto the feasible simplex
        a = _project_to_simplex(a, target_sum)

        # 7) occasional tiny jitter to help escape plateaus
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = _project_to_simplex(a + jitter, target_sum)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _simulated_annealing_refine(
    a_start: np.ndarray,
    n: int,
    target_sum: float,
    cur_best: float,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap SA/hill‑climb that transfers a small fraction of the mass
    from a donor index to a receiver index while keeping the ℓ1‑norm fixed.
    Accepts improving moves always, otherwise with probability exp(-Δ/T).
    Temperature is geometrically cooled.
    """
    a = a_start.copy()
    best_val = cur_best
    best = a.copy()

    temperature = 0.05
    cooling = 0.9995
    moved = 0

    while time.time() < deadline and moved < max_moves:
        moved += 1

        # pick donor (must have some mass)
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue

        # pick receiver ≠ donor
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # small amount (up to 2 % of donor)
        delta = a[i] * np.random.rand() * 0.02
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_a = _project_to_simplex(new_a, target_sum)

        conv = _conv_full_fft(new_a)
        new_val = conv.max()

        if new_val < best_val:
            a = new_a
            best_val = new_val
            best = new_a.copy()
            print(f"[SA] improvement → {best_val:.6f} (move {moved})")
        else:
            # accept probabilistically
            if temperature > 0 and np.random.rand() < math.exp(-(new_val - best_val) / temperature):
                a = new_a

        temperature = max(temperature * cooling, 1e-6)

    return best, best_val


# ----------------------------------------------------------------------
def _arcsine_initial(n: int, target_sum: float) -> np.ndarray:
    """
    Continuous arcsine density on [0,1] discretised to length n,
    then scaled to the required ℓ1‑norm.
    """
    idx = np.arange(n, dtype=float)
    eps = 1e-12
    a = 1.0 / np.sqrt((idx + 0.5) * (n - 0.5 - idx) + eps)
    a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a step‑function that minimises evaluate_sequence.
    The routine respects the 1000 s time budget and returns the best
    sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a safety margin for the host (few seconds)
    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # Choose a single length (close to the allowed maximum) – larger n
    # gives a lower theoretical bound.
    n = 8000
    target = math.sqrt(2.0 * n)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) arcsine start + Adam optimisation + SA refinement
    a0 = _arcsine_initial(n, target)
    best_seq, best_score = _smooth_adam_optimize(
        a_init=a0,
        n=n,
        target_sum=target,
        best_seq=best_seq if best_seq is not None else a0,
        best_score=best_score,
        deadline=deadline,
        max_iter=120_000,
    )
    best_seq, best_score = _simulated_annealing_refine(
        a_start=best_seq,
        n=n,
        target_sum=target,
        cur_best=best_score,
        deadline=deadline,
        max_moves=200_000,
    )
    print(f"[phase 1] best max‐conv = {best_score:.6f} (len {n})")

    # ------------------------------------------------------------------
    # 2) One (or more) random Dirichlet restart if time permits
    while time.time() < deadline - 30.0:      # leave a few seconds for final I/O
        a_rand = np.random.rand(n)
        a_rand = _project_to_simplex(a_rand, target)

        cand_seq, cand_score = _smooth_adam_optimize(
            a_init=a_rand,
            n=n,
            target_sum=target,
            best_seq=best_seq,
            best_score=best_score,
            deadline=deadline,
            max_iter=80_000,
        )
        cand_seq, cand_score = _simulated_annealing_refine(
            a_start=cand_seq,
            n=n,
            target_sum=target,
            cur_best=cand_score,
            deadline=deadline,
            max_moves=150_000,
        )
        if cand_score < best_score:
            best_seq, best_score = cand_seq, cand_score
            print(f"[restart] new global best max = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final check: evaluate using the provided function (just for safety)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[done] final evaluate_sequence = {final_score:.6f}, length = {len(best_seq)}")
    return best_seq.tolist()
