def smooth_max_and_gradient(a, alpha):
    conv = np.convolve(a, a)
    maxc = conv.max()
    # For numerical stability, shift by maxc
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / exp_term.sum()
    # smooth max (approx)
    smooth_val = np.dot(w, conv)
    # compute gradient
    # w is length L = 2n-1
    # reversed w
    w_rev = w[::-1]
    # convolution a * w_rev using FFT for speed
    # Use np.fft.rfft with appropriate length.
    n = a.shape[0]
    L = len(w_rev)
    N = 1 << ((n + L - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(w_rev, n=N)
    conv2 = np.fft.irfft(A*B, n=N)[: n + L - 1]
    # gradient indices start at L-1 = 2n-2
    start = L - 1
    grad = 2.0 * conv2[start : start + n]
    return smooth_val, grad
```

def adam_softmax(a_init, n, target_sum, best_seq, best_score, deadline,
                 max_iter=200000, alpha_start=1.0, alpha_end=50.0,
                 lr0=0.05):
    a = a_init.copy()
    # normalize to target_sum
    sum_a = a.sum()
    if sum_a>0:
        a = a * (target_sum / sum_a)
    else:
        a = np.full_like(a, target_sum / n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    iteration = 0
    while time.time() < deadline and iteration < max_iter:
        # schedule alpha linearly increasing
        alpha = alpha_start + (alpha_end - alpha_start) * iteration / max_iter

        smooth_val, grad = smooth_max_and_gradient(a, alpha)

        # Adam update
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        lr = lr0 / (1 + 1e-4 * iteration)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # Evaluate actual max conv to check improvement
        M, _, _ = _conv_max_fft(a)  # we can use this as measuring metric
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n} iter {iteration}] new best {best_score:.6f}")

        # occasional jitter to escape plateau
        if iteration % 5000 == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
    return best_seq, best_score
```

"""
Propose a non‑negative step sequence that minimizes the evaluation score.
Algorithm
---------
1. Load the supplied construction `height_sequence_1` (fallback: uniform).
2. Explore several lengths (original  and modest enlargements up to ~8000).
3. For each length we run a few restarts:
   * Interpolation of the previous best sequence (to keep good structure).
   * Random Dirichlet‑like initialisations.
4. Each restart is optimised with a **soft‑max‑smoothed Adam** method:
   - The objective is   L(a)=∑ₖ wₖ·(a∗a)[k]  where  wₖ∝exp(α·(a∗a)[k]).
   - α is increased from a small to a large value (annealing), turning L
     from a smooth average to an approximation of the max‑convolution.
   - The gradient of L is computed analytically via FFTs:
        ∇L = 2·(a ⋆ w)   (⋆ = cross‑correlation).
   - Adam updates are projected onto the feasible set (a≥0, Σa = √(2·n)).
   - Periodic jitter is added to escape plateaus.
5. After the Adam phase a cheap **targeted local search** moves a small
   amount of mass from indices that contribute most to the current maximal
   convolution term to indices that contribute least.
6. The globally best sequence (lowest max‑convolution value) found before
   the time budget expires is returned as a plain list.

The whole search respects the 1000‑second budget; intermediate progress
(and improvements) are printed.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
def _fft_len(min_len: int) -> int:
    """Return the next power‑of‑two ≥ min_len (used for FFT padding)."""
    return 1 << (min_len - 1).bit_length()


# ------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear autocorrelation a∗a via FFT (length 2·n‑1)."""
    n = a.shape[0]
    N = _fft_len(2 * n - 1)
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return conv


# ------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """
    Full autocorrelation a∗a (FFT) and the maximal entry.
    Returns (max_value, index_of_max, full_array).
    """
    conv = _conv_full_fft(a)
    idx = int(conv.argmax())
    return float(conv.max()), idx, conv


# ------------------------------------------------------------
def _rev_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    """
    For a given shift index k, returns the vector rev where rev[i] = a[k-i]
    if the index is inside the support, otherwise 0.
    (Used to build the sub‑gradient of a single convolution entry.)
    """
    n = a.shape[0]
    rev = np.zeros_like(a)
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    rev[mask] = a[j[mask]]
    return rev


# ------------------------------------------------------------
def _smooth_max_and_gradient(a: np.ndarray, alpha: float):
    """
    Compute a smooth approximation of max(a∗a) and its gradient.
    The smooth max is the weighted average with weights wₖ ∝ exp(α·cₖ)
    where cₖ = (a∗a)[k].
    Returns (smooth_value, gradient_vector, current_max).
    """
    n = a.shape[0]

    # ----- convolution (full) ---------------------------------
    conv = _conv_full_fft(a)                # length 2n‑1
    max_c = conv.max()

    # ----- soft‑max weights ------------------------------------
    # shift by max for numerical stability
    exp_term = np.exp(alpha * (conv - max_c))
    w = exp_term / exp_term.sum()           # length 2n‑1

    # ----- gradient: ∇L = 2·(a ⋆ w) ----------------------------
    # cross‑correlation a⋆w can be obtained as a * reverse(w)
    w_rev = w[::-1]                         # also length 2n‑1
    N2 = _fft_len(n + w_rev.size - 1)       # enough for convolution length
    A = np.fft.rfft(a, n=N2)
    B = np.fft.rfft(w_rev, n=N2)
    corr_full = np.fft.irfft(A * B, n=N2)[: n + w_rev.size - 1]

    # the relevant slice (see derivation in the analysis)
    start = w_rev.size - 1                  # 2n‑2
    grad = 2.0 * corr_full[start : start + n]

    # ----- smooth max value ------------------------------------
    smooth_val = np.dot(w, conv)

    return smooth_val, grad, max_c


# ------------------------------------------------------------
def _adam_softmax_optimize(
    a_init: np.ndarray,
    n: int,
    target_sum: float,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    max_iters: int = 120_000,
    lr0: float = 0.05,
    alpha_start: float = 0.5,
    alpha_end: float = 30.0,
    jitter_interval: int = 5000,
):
    """
    Adam optimiser on the soft‑max‑smoothed max‑convolution objective.
    Returns possibly improved (best_seq, best_score).
    """
    a = a_init.copy()
    # normalise to required ℓ₁‑norm
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    else:
        a = np.full_like(a, target_sum / n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    iteration = 0
    while time.time() < deadline and iteration < max_iters:
        # linearly anneal α
        alpha = alpha_start + (alpha_end - alpha_start) * iteration / max_iters

        # smooth objective + gradient
        _, grad, cur_max = _smooth_max_and_gradient(a, alpha)

        # Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto feasible set
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # record improvement if any
        if cur_max < best_score:
            best_score = cur_max
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:7d}] new best = {best_score:.6f}")

        # occasional jitter to escape flat regions
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

    return best_seq, best_score


# ------------------------------------------------------------
def _local_search_targeted(
    a_start: np.ndarray,
    n: int,
    target_sum: float,
    best_score: float,
    deadline: float,
    max_moves: int = 40_000,
):
    """
    Very cheap hill‑climb that moves a small amount of mass from the
    most harmful index (w.r.t. the current maximal convolution term) to
    the least harmful one.  Accepts only improving moves; occasionally
    accepts a worsening move with tiny probability to escape local minima.
    """
    a = a_start.copy()
    for move in range(max_moves):
        if time.time() >= deadline:
            break

        # compute full convolution and the index of the current maximum
        conv = np.convolve(a, a)
        cur_max = conv.max()
        k_max = int(conv.argmax())

        # the reverse vector that tells, for each i, how much a[i] contributes to conv[k_max]
        rev = _rev_for_shift(a, k_max)          # shape (n,)

        # donor = index with largest contribution a[i]*rev[i]
        contribution = a * rev
        donor = int(np.argmax(contribution))

        # receiver = index with smallest a (or smallest contribution)
        receiver = int(np.argmin(a))

        if donor == receiver:
            # fallback: pick a random pair
            donor, receiver = np.random.choice(n, 2, replace=False)

        # amount to move – up to 4 % of the donor's mass
        eps = a[donor] * np.random.rand() * 0.04
        if eps <= 0.0:
            continue

        new_a = a.copy()
        new_a[donor] -= eps
        new_a[receiver] += eps

        # keep feasibility (non‑negativity already ensured)
        # (sum is unchanged)

        new_max = np.convolve(new_a, new_a).max()
        if new_max < cur_max - 1e-12:
            a = new_a
            cur_max = new_max
            if cur_max < best_score:
                best_score = cur_max
                best_seq = a.copy()
                print(f"[local] improved to {best_score:.6f} (move {move})")
        else:
            # small probability to accept a worsening move (simulated annealing style)
            if np.random.rand() < 1e-4:
                a = new_a

    return a, best_score
