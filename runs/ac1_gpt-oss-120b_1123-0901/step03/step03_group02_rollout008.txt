"""
Propose a non‑negative step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a)**2)

which is exactly the maximum value of the self‑convolution once the
sequence is projected onto the simplex Σ a_i = √(2 n).
The algorithm combines several ideas:

* Randomised restarts with lengths ranging from a few hundred to several
  thousand entries (larger lengths give a lower theoretical bound).
* Each restart runs a projected Adam optimiser on the smooth‑max
  approximation  τ·log Σ exp(conv/τ).  The temperature τ is exponentially
  cooled so that the optimisation finally targets the true max.
* The gradient of the smooth‑max is obtained analytically as a
  cross‑correlation; both the convolution and its gradient are computed
  with a real FFT (O(n log n)).
* Occasionally a small random shake‑up is applied to escape plateaus.
* The current global best sequence is used as a warm‑start (truncated or
  padded) for many restarts.
* After the time budget the best sequence found is returned as a plain
  Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (non‑negative, sum = s)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – Full linear convolution via real FFT
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power‑of‑2
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative artefacts caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Optimisation of a single length with Adam + smooth‑max (log‑sum‑exp)
# ----------------------------------------------------------------------
def _optimise_one_length_adam(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 0.001,
    lr: float = 0.1,
    tau_decay: float = 0.995,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    shake_prob: float = 0.008,
) -> tuple[np.ndarray, float]:
    """
    Run a projected Adam optimiser on the smooth‑max approximation of
    max(conv(a, a)).  Returns the best sequence found (as np.ndarray) and
    its exact score obtained from ``evaluate_sequence``.
    """
    target_sum = np.sqrt(2.0 * n)          # Σ a_i = √(2 n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # bookkeeping
    best_seq = a.copy()
    conv = _full_convolution_fft(a, a)
    best_score = conv.max()               # equal to the true objective

    # Adam moment vectors
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # initial temperature (relative to the current max)
    max_val = conv.max()
    tau = max(init_tau_factor * max_val, min_tau_factor * max_val)

    while time.time() < run_deadline:
        # --------------------------------------------------------------
        # 1) self‑convolution and maximum
        # --------------------------------------------------------------
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()

        # --------------------------------------------------------------
        # 2) smooth‑max weights (log‑sum‑exp) – temperature τ
        # --------------------------------------------------------------
        tau = max(tau * tau_decay, min_tau_factor * max_val)
        # stable softmax
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # 3) gradient of Σ w_k conv_k  =  2 * Σ_j a_j w_{i+j}
        # --------------------------------------------------------------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)          # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)                  # indices that give Σ_j a_j w_{i+j}
        grad = 2.0 * conv2[idx]

        # --------------------------------------------------------------
        # 4) Adam update (projected onto the simplex)
        # --------------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # --------------------------------------------------------------
        # 5) keep local best
        # --------------------------------------------------------------
        conv_new = _full_convolution_fft(a_new, a_new)
        score_new = conv_new.max()
        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {best_score:.6f}")

        a = a_new

        # occasional random shake‑up to get out of plateaus
        if np.random.rand() < shake_prob:
            noise = np.random.randn(n) * (target_sum * 0.0005 / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            # after a shake‑up make the optimiser a little more cautious
            lr *= 0.9

    # final evaluation with the supplied evaluator (ensures identical rounding)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the evaluation
    function.  The routine respects the time budget and returns the best
    sequence discovered.

    Strategy
    --------
    1. Warm‑start from a previously supplied global construction if present.
    2. Perform many independent restarts.  Early restarts explore short
       sequences (≈ 800‑2500) quickly; later restarts focus on longer
       sequences (up to ≈ 8000) where the theoretical lower bound is closer
       to 1.
    3. Each restart runs the Adam optimiser defined above with a decaying
       soft‑max temperature.
    4. The current global best is reused as a warm‑start (truncated/padded)
       for a substantial fraction of the restarts.
    5. The overall best sequence (as a plain Python list) is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a previously known construction if available.
    # ------------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        score = evaluate_sequence(hs_arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start from global seed = {score:.6f}")
    except Exception:
        pass  # no warm‑start supplied

    # ------------------------------------------------------------------
    # Main restart loop.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        elapsed = time.time() - start_time
        frac = elapsed / budget_s

        # --------------------------------------------------------------
        # Choose a problem size according to the current phase.
        # --------------------------------------------------------------
        if frac < 0.30:
            n = np.random.randint(800, 2500 + 1)
            slice_time = 5.0
        elif frac < 0.70:
            n = np.random.randint(1500, 5000 + 1)
            slice_time = 7.0
        else:
            n = np.random.randint(3000, 8000 + 1)
            slice_time = 10.0

        # --------------------------------------------------------------
        # Build an initial shape.
        # --------------------------------------------------------------
        p = np.random.uniform(0.3, 0.7)               # power‑law exponent
        idx = np.arange(1, n + 1, dtype=float)
        init = idx ** (-p)

        if np.random.rand() < 0.35:                  # optional exponential tail
            tau_exp = np.random.uniform(0.4, 2.0) * n
            init *= np.exp(-idx / tau_exp)

        init += np.random.rand(n) * 1e-4              # tiny jitter

        # Occasionally start from the current global best (truncated/padded)
        if best_seq is not None and np.random.rand() < 0.45:
            bs = best_seq
            if bs.size >= n:
                init = bs[:n].copy()
            else:
                pad = np.random.rand(n - bs.size) * 1e-4
                init = np.concatenate([bs, pad])

        # --------------------------------------------------------------
        # Run the optimiser for the chosen length.
        # --------------------------------------------------------------
        run_deadline = min(deadline, time.time() + min(slice_time, remaining * 0.6))

        seq_local, score_local = _optimise_one_length_adam(
            n,
            init,
            run_deadline,
            init_tau_factor=0.12,
            min_tau_factor=0.001,
            lr=0.12,
            tau_decay=0.995,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            shake_prob=0.008,
        )

        # --------------------------------------------------------------
        # Update the global best if an improvement was found.
        # --------------------------------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (length={n})"
            )
            # early exit if we already beat the target bound
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Fallback – should never happen, but keep a valid sequence just in case.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list as required by the external evaluator.
    return best_seq.tolist()
