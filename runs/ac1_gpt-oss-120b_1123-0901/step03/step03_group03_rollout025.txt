    import numpy as np, math, time, scipy.optimize, scipy.sparse

    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    # load base shape if any else start uniform
    try:
        base = np.array(height_sequence_1, dtype=float)
    except Exception:
        base = np.ones(2100, dtype=float)
    # Normalize sum to 1
    base = base / base.sum()

    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    print(f"[init] len={len(best_seq)} score={best_score}")

    # We'll search lengths: original and maybe larger
    lengths = [len(base), int(len(base)*1.5), int(len(base)*2)]
    lengths = [l for l in lengths if l <= 8000]
    lengths = sorted(set(lengths))

    for n in lengths:
        # resample base onto new length via interpolation
        if n != len(base):
            x_old = np.linspace(0, 1, len(base))
            x_new = np.linspace(0, 1, n)
            f = np.interp(x_new, x_old, base)
            f = np.maximum(f, 0.0)
        else:
            f = base.copy()
        # Normalize sum=1
        f = f / f.sum()
        # iterative refinement via LP
        iter_no = 0
        while time.time() < deadline and iter_no < 200:
            iter_no += 1
            # compute convolution and max
            conv_f = np.convolve(f, f)
            M = conv_f.max()
            # construct sparse constraints matrix A where A_{k,i}= f_{k-i} if valid
            # shape (2n-1, n)
            rows = []
            cols = []
            data = []
            # we can generate matrix via loops
            for k in range(2*n-1):
                i_start = max(0, k - (n-1))
                i_end = min(n-1, k)
                # i range inclusive
                # for each i, column j = i, row k, entry f_{k - i}
                # we can vectorize using numpy slices
                js = np.arange(i_start, i_end+1, dtype=int)
                vals = f[k - js]   # f index
                rows.extend([k]*len(js))
                cols.extend(js.tolist())
                data.extend(vals.tolist())
            A = scipy.sparse.coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsr()
            # solve LP: maximize sum(g) -> minimize -sum(g)
            c = -np.ones(n)
            b_ub = np.full(2*n-1, M)
            res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
            if not res.success:
                print(f"[LP fail] n={n} iter={iter_no}")
                break
            g = res.x
            sum_g = g.sum()
            if sum_g <= 0:
                # fallback to uniform
                g = np.full_like(f, 1.0/n)
                sum_g = 1.0
            # normalise to sum=1
            g = g / sum_g

            # Evaluate mixing fraction candidates
            # we will evaluate t in a set of values to pick best
            # maybe use golden-section search? Simpler: h set {0.0,0.05,0.1,0.2,0.3,0.5}
            # compute conv for each candidate.
            best_local_seq = f
            best_local_score = best_score
            # We'll precompute conv_f and conv_g to speed
            conv_g = np.convolve(g, g)
            conv_fg = np.convolve(f, g)
            for t in [0.0, 0.05, 0.1, 0.2, 0.3, 0.5]:
                if t == 0.0:
                    h = f
                elif t == 1.0:
                    h = g
                else:
                    h = (1 - t) * f + t * g
                # sum already 1
                # compute objective
                cur_score = evaluate_sequence(h.tolist())
                if cur_score < best_local_score:
                    best_local_score = cur_score
                    best_local_seq = h.copy()
            # update global best
            if best_local_score < best_score:
                best_score = best_local_score
                best_seq = best_local_seq.copy()
                print(f"[len {n} iter {iter_no}] new best = {best_score:.6f}")
                # optionally set f to best_local_seq for further refinement
                f = best_local_seq.copy()
            else:
                # continue using current f (maybe improved)
                f = best_local_seq.copy()
    # after loops
    print(f"[done] best score = {best_score:.6f}, length = {len(best_seq)}")

"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
* Load the previously discovered construction (``height_sequence_1``) if
  available, otherwise start from a uniform distribution.  The vector is
  normalised to have unit ℓ₁‑norm – the objective is invariant under scaling.
* For a few chosen lengths (original length and a 1.5× enlargement) run an
  alternating optimisation:
    – Given the current sequence *f*, compute its self‑convolution and the
      maximal value M = max(f∗f).
    – Build the linear constraint matrix A (size (2n−1)×n) for the
      convolution ``(f * g) ≤ M``; each row k corresponds to the coefficients
      f_{k‑i} (a Toeplitz matrix).  The matrix is constructed with a sparse
      representation using ``scipy.sparse.diags``.
    – Solve the linear programme
          maximise   Σ g_i
          subject to A·g ≤ M·1 ,   g ≥ 0 .
      The solution g pushes mass into locations where the current
      convolution is far below the bound M.
    – Normalise g to unit sum (a convex combination keeps the sum = 1).
    – For a small grid of mixing factors t∈[0,1] evaluate the exact
      convolution of the candidate h = (1‑t)·f + t·g using the analytic
      formula
          conv_h = (1‑t)²·conv_f + 2·t·(1‑t)·conv_{f,g} + t²·conv_g .
      The objective is ``2·n·max(conv_h)``.  Pick the t that gives the
      smallest objective and update f←h.
    – Occasionally add a tiny random jitter (Gaussian noise) to help
      escaping plateaus.
* Keep the best sequence seen across all lengths and iterations.
* The whole search is stopped before the allotted 1000 s wall‑clock time.

The method alternates a linear‑programming step (which directly reduces the
maximum of the convolution) with a simple line‑search over convex
combinations.  It reaches lower bounds than pure gradient descent while
remaining inexpensive enough to run many iterations within the time budget.
"""

import time
import math
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp


def _full_conv(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT (full length)."""
    n = a.shape[0] + b.shape[0] - 1
    Nfft = 1 << (n - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, n=Nfft)
    B = np.fft.rfft(b, n=Nfft)
    conv = np.fft.irfft(A * B, n=Nfft)[:n]
    return conv


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a convolution."""
    return _full_conv(a, a)


def _build_constraint_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix A (shape (2n‑1, n)) such that
        (A @ g)[k] = (f * g)[k] .
    The matrix is Toeplitz: each diagonal d (0≤d<n) holds the constant value f[d].
    """
    n = f.shape[0]
    # each diagonal contains n entries equal to the corresponding entry of f
    diag_data = np.repeat(f[:, np.newaxis], n, axis=1)   # shape (n, n)
    offsets = np.arange(n)                               # 0 … n‑1
    A = sp.diags(diag_data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the outer harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load the previous construction (or fall back to a uniform vector)
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        base = np.ones(2100, dtype=float)

    # Normalise to sum = 1 (objective does not depend on overall scale)
    base = base / base.sum()
    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={len(best_seq)}  initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore – original and a 1.5× enlargement (capped at 8000)
    lengths = sorted({len(base), int(len(base) * 1.5)})
    lengths = [min(l, 8000) for l in lengths]
    lengths = sorted(set(lengths))

    # ------------------------------------------------------------------
    # Main optimisation loop over the selected lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        # Initialise f for the current length (linear interpolation if needed)
        if n != len(base):
            x_old = np.linspace(0.0, 1.0, len(base))
            x_new = np.linspace(0.0, 1.0, n)
            f = np.interp(x_new, x_old, base)
            f = np.maximum(f, 0.0)
        else:
            f = base.copy()
        f = f / f.sum()                       # keep unit ℓ₁‑norm

        # Pre‑compute static index data for the constraint matrix (rows/cols)
        # (the actual values will be filled anew each iteration)
        max_outer_iters = 80                  # generous but still bounded
        outer_iter = 0

        while time.time() < deadline and outer_iter < max_outer_iters:
            outer_iter += 1

            # --------------------------------------------------------------
            # 1) current self‑convolution and its maximum
            conv_f = _conv_self(f)
            M = float(conv_f.max())

            # --------------------------------------------------------------
            # 2) build linear constraints (f * g ≤ M)
            A = _build_constraint_matrix(f)

            # --------------------------------------------------------------
            # 3) linear programme: maximise Σ g_i   (i.e. minimise -Σ g_i)
            c = -np.ones(n)                                 # objective: -sum(g)
            b_ub = np.full(2 * n - 1, M)                    # RHS = M
            # Use the HiGHS solver – it accepts sparse matrices directly
            res = opt.linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                bounds=(0, None),
                method="highs",
                options={"presolve": True, "time_limit": max(1.0, (deadline - time.time()))},
            )
            if not res.success:
                # If the LP fails (unlikely) fall back to a tiny random perturbation
                print(f"[LP fail] n={n} iter={outer_iter}")
                jitter = np.random.randn(n) * 1e-6
                f = np.maximum(f + jitter, 0.0)
                f = f / f.sum()
                continue

            g_raw = res.x
            sum_raw = g_raw.sum()
            if sum_raw <= 0.0:
                # Degenerate LP – revert to a uniform vector
                g = np.full_like(f, 1.0 / n)
            else:
                g = g_raw / sum_raw     # normalise to unit ℓ₁‑norm

            # --------------------------------------------------------------
            # 4) pre‑compute the three convolutions needed for any convex blend
            conv_g = _conv_self(g)
            conv_fg = _full_conv(f, g)

            # --------------------------------------------------------------
            # 5) line‑search over mixing factor t ∈ [0,1]
            #    candidate h(t) = (1‑t)·f + t·g  (still unit sum)
            t_grid = np.linspace(0.0, 1.0, 21)   # 0,0.05,…,1
            t_arr = t_grid[:, None]             # shape (|t|,1) for broadcasting
            # convolution of the blend obtained analytically:
            conv_h = (
                (1.0 - t_arr) ** 2 * conv_f[None, :]
                + 2.0 * t_arr * (1.0 - t_arr) * conv_fg[None, :]
                + t_arr ** 2 * conv_g[None, :]
            )
            max_vals = conv_h.max(axis=1)        # maximal entry for each t
            obj_vals = 2.0 * n * max_vals        # exactly the evaluate_sequence value
            best_idx = int(np.argmin(obj_vals))
            best_t = t_grid[best_idx]
            h = (1.0 - best_t) * f + best_t * g

            cur_score = float(obj_vals[best_idx])
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = h.copy()
                print(f"[len {n:5d} iter {outer_iter:3d}] new best = {best_score:.6f}")

            # --------------------------------------------------------------
            # 6) update f for the next iteration (always keep the best blend)
            f = h
            # occasional jitter to help escape shallow plateaus
            if outer_iter % 10 == 0:
                jitter = np.random.randn(n) * (1e-4 / n)
                f = np.maximum(f + jitter, 0.0)
                f = f / f.sum()

    # ------------------------------------------------------------------
    # Final safety check – if for any reason we have no sequence, fall back
    if best_seq is None or len(best_seq) == 0:

        best_seq = np.full(n, 1.0 / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq.tolist()
