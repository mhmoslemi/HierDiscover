"""
Propose a non‑negative step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

Algorithm
---------
* Use a projected Adam optimizer on the smooth‑max (log‑sum‑exp) surrogate
  of the max‑convolution.
* The iterate is always projected onto the simplex
  Σ a_i = √(2 n) (this makes the objective equal to the maximal
  convolution value).
* Gradient of the weighted sum Σ w_k (a*a)_k is computed via FFT as a
  cross‑correlation.
* Temperature τ of the soft‑max is decreased geometrically; Adam’s
  adaptive step‑sizes handle the scaling of the gradient.
* Many independent restarts with different lengths (up to ∼30 000) and
  diverse initial shapes (power‑law, sinusoidal, parabolic) are performed
  within the 1000 s budget.  The best sequence found is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (non‑negative, sum = s)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – Full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())      # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative values caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Optimisation of a single length using Adam + smooth‑max surrogate
# ----------------------------------------------------------------------
def _optimise_one_length_adam(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 0.001,
    learning_rate: float = 0.6,
    step_decay: float = 0.9995,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam‑based optimisation for a fixed length `n`.

    Returns the best sequence found (as a NumPy array) and its score
    obtained from ``evaluate_sequence``.
    """
    target_sum = math.sqrt(2.0 * n)            # Σ a_i = √(2 n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # keep the best sequence seen in this run
    best_seq = a.copy()
    conv = _full_convolution_fft(a, a)
    best_score = 2.0 * n * conv.max() / (target_sum ** 2)

    m = np.zeros_like(a)   # first moment (Adam)
    v = np.zeros_like(a)   # second moment (Adam)

    iteration = 0
    lr = learning_rate

    while time.time() < run_deadline:
        iteration += 1

        # ---- 1. self‑convolution and maximal entry ----
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()

        # ---- 2. smooth‑max weights (soft‑max) ----
        tau = max(init_tau_factor * max_val * (0.95 ** iteration),
                  min_tau_factor * max_val)
        # shift by max_val for stability
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)                      # all entries ≤ 1
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- 3. gradient of Σ w_k (a*a)_k ----
        w_rev = w[::-1]                         # reversed weighting vector
        conv2 = _full_convolution_fft(a, w_rev)  # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)        # indices giving Σ_j a_j w_{i+j}
        grad = 2.0 * conv2[idx]                 # shape (n,)

        # ---- 4. Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # ---- 5. keep local best ----
        conv_new = _full_convolution_fft(a_new, a_new)
        score_new = 2.0 * n * conv_new.max() / (target_sum ** 2)
        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {score_new:.6f}")

        a = a_new
        lr *= step_decay

        # ---- occasional random perturbation to escape plateaus ----
        if np.random.rand() < 0.015:
            noise = np.random.randn(n) * (target_sum * 0.001 / np.sqrt(n))
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            # reset Adam statistics after a shake‑up
            m.fill(0.0)
            v.fill(0.0)
            lr *= 0.9

    # final evaluation with the exact evaluator (ensures identical rounding)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score

    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a previously known construction if it exists.
    # ------------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        score = evaluate_sequence(hs_arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start from global seed = {score:.6f}")
    except Exception:
        pass  # no warm‑start available

    # ------------------------------------------------------------------
    # Main restart loop – each restart gets a time slice.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # Choose a length.  Early on we try larger problems; later we shrink
        # the range so that we can finish the last iterations quickly.
        if remaining > 400:
            n = np.random.randint(5000, 30000 + 1)
        elif remaining > 150:
            n = np.random.randint(3000, 15000 + 1)
        else:
            n = np.random.randint(1000, 8000 + 1)

        # ------------------- initialise a candidate shape -----------------
        shape_kind = np.random.choice(["power", "sin", "parab", "mix"])
        idx = np.arange(1, n + 1, dtype=float)

        if shape_kind == "power":
            p = np.random.uniform(0.2, 1.2)
            init = idx ** (-p)
            if np.random.rand() < 0.4:                     # optional exponential tail
                tau_exp = np.random.uniform(0.2, 1.5) * n
                init *= np.exp(-idx / tau_exp)
        elif shape_kind == "sin":
            t = np.arange(n, dtype=float) / (n - 1)
            exponent = np.random.uniform(0.5, 2.0)
            init = np.sin(np.pi * t) ** exponent
        elif shape_kind == "parab":
            # parabola that rises to the centre and falls symmetrically
            t = np.linspace(-1, 1, n)
            init = 1.0 - t * t
            init = np.clip(init, 0.0, None)
        else:  # mix – power law with sinusoidal modulation
            p = np.random.uniform(0.3, 0.9)
            init = idx ** (-p)
            init *= (1.0 + 0.2 * np.sin(2.0 * np.pi * idx / n))

        # tiny jitter to break symmetry
        init += np.random.rand(n) * 1e-4

        # occasional warm‑start from the current global best
        if best_seq is not None and np.random.rand() < 0.35:
            bs = best_seq
            if bs.size >= n:
                init = bs[:n].copy()
            else:
                pad = np.random.rand(n - bs.size) * 1e-4
                init = np.concatenate([bs, pad])

        # Allocate a time slice for this particular run (≈10 s but never
        # exceed the global deadline)
        slice_time = min(10.0, remaining * 0.5)
        run_deadline = time.time() + slice_time

        # Run the optimiser for the chosen length
        seq_local, score_local = _optimise_one_length_adam(
            n,
            init,
            run_deadline,
            init_tau_factor=0.12,
            min_tau_factor=0.001,
            learning_rate=0.6,
            step_decay=0.9995,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
        )

        # Update the global best if improvement was found
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (length={n})")
            # If we already beat the target comfortably we can stop early
            if best_score < 1.49:
                break

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that no candidate was generated.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # The evaluator expects a plain Python list.
    return best_seq.tolist()
