"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

The code improves on the previous soft‑max Adam implementation by
* running several random restarts (different initial shapes, including
  geometric, linear and random Dirichlet draws);
* using a two‑phase temperature schedule (smooth soft‑max early, hard
  max later) for a more accurate sub‑gradient;
* enlarging the learning rate a little and decaying it more aggressively;
* periodically sorting the vector in non‑increasing order – by the
  rearrangement inequality this cannot increase the maximal convolution;
* adding jitter more often to escape plateaus;
* allocating the 1000 s budget over many short runs instead of a single
  long run, which gives a better chance to escape local minima.

The optimiser works on a single length (the original length of the
known construction, plus a doubled length) and keeps the global best
vector seen so far.  All operations are performed with FFTs (O(N log N))
so even a few million iterations fit comfortably in the time limit.

The final sequence is returned as a plain Python list ready for the
harness.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level – no closures)

def _conv_full(a: np.ndarray, Nfft: int) -> np.ndarray:
    """Full convolution a * a (polynomial multiplication) via FFT."""
    A = np.fft.rfft(a, n=Nfft)
    # convolution by squaring in frequency domain
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * a.size - 1]
    return conv


def _softmax_weights(conv: np.ndarray, tau: float):
    """
    Stable soft‑max of the convolution values.
    Returns (weights w that sum to 1, smoothed value Lτ = τ·log∑exp(conv/τ)).
    """
    mx = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, mx
    # subtraction for numerical stability
    shift = conv - mx
    ex = np.exp(shift / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = mx + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev where rev[i] = Σ_k w[k] * a[k-i]  (zero‑filled outside range).
    Implemented by FFT: reverse w, convolve with a.
    """
    rev_full = np.fft.irfft(np.fft.rfft(w[::-1], n=Nfft) *
                            np.fft.rfft(a, n=Nfft), n=Nfft)
    # keep the part that corresponds to indices 0…len(a)-1 after reversal
    rev = rev_full[: len(w) + a.size - 1]
    start = 2 * a.size - 2          # index of a_rev[0] in the full convolution
    rev = rev[start - np.arange(a.size)]
    return rev


def _project_onto_simplex(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale a to have the given L1 sum (preserving non‑negativity)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    else:
        a = np.full_like(a, target_sum / a.size)
    return a


def _objective_from_conv(conv: np.ndarray, n: int, sum_a: float) -> float:
    """Exact objective value given convolution, length and sum."""
    return float(2.0 * n * conv.max() / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Core optimiser for a single initial vector

def _optimize_one_start(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    *,
    lr0: float = 0.10,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau_factor: float = 0.10,
    decay_rate: float = 25000.0,
    jitter_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the smoothed maximal self‑convolution.
    Returns the best vector found and its objective value.
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise and enforce the L1 constraint
    a = _project_onto_simplex(a0.copy(), target_sum)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # FFT size (next power of two >= 2*n-1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    best_seq = a.copy()
    # compute initial objective (exact)
    conv = _conv_full(a, Nfft)
    best_score = _objective_from_conv(conv, n, a.sum())

    # main loop
    it = 0
    while time.time() < deadline and it < max_iters:
        # 1) full convolution a*a
        conv = _conv_full(a, Nfft)
        M = float(conv.max())                     # plain max (used for reporting)

        # 2) temperature schedule
        # warm‑up: relatively large tau, then exponential decay
        tau = max(1e-12, tau_factor * M * math.exp(-it / decay_rate))

        # 3) soft‑max (or hard‑max when tau is tiny)
        w, Ltau = _softmax_weights(conv, tau)

        # 4) weighted reverse convolution = sub‑gradient of smoothed max
        rev = _weighted_rev(a, w, Nfft)

        # 5) gradient of the *scale‑invariant* objective (see original script)
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        # 6) Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 5e-5 * it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # 7) feasibility projection (non‑negative, correct L1 norm)
        a = _project_onto_simplex(a, target_sum)

        # 8) occasional sorting – non‑increasing order can only lower the max conv
        if (it % 5000) == 0:
            a = np.sort(a)[::-1]          # descending order
            a = _project_onto_simplex(a, target_sum)

        # 9) occasional jitter to break stagnation
        if (it % jitter_interval) == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = _project_onto_simplex(np.maximum(a + jitter, 0.0), target_sum)

        # 10) exact objective check (fast – we already have conv)
        cur_score = _objective_from_conv(conv, n, a.sum())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            # logging – prints are captured by the harness
            print(f"[len {n:5d} it {it:8d}] new best = {best_score:.6f}")

    # final check (in case the loop exited before updating best_seq)
    conv = _conv_full(best_seq, Nfft)
    best_score = _objective_from_conv(conv, n, best_seq.sum())
    return best_seq, best_score
