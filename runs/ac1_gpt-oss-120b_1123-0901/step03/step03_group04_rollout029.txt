"""
search for a non‑negative step‑function that minimises ``evaluate_sequence``.

The algorithm combines three ideas:

1. **Fast sub‑gradient descent** (Adam) on the exact sub‑gradient of the
   ``max(convolution)`` objective, exactly as in the reference solution.
   This gives a good starting point quickly.

2. **Random local moves** – cheap greedy mass transfers between two
   coordinates that further reduce the peak.

3. **Linear‑programming direction** – for the current vector `f`
   we solve the LP

        maximise    Σ g_j
        subject to  g_j ≥ 0
                    (f * g)[k] ≤ M   for all k

   where `M = max(f * f)`.  The solution `g` is a feasible direction that
   can carry more ℓ¹‑mass without increasing the current peak.
   Mixing the current vector with the rescaled `g` (convex combination)
   strictly reduces the peak.  The LP is built explicitly (size
   (2n‑1)×n) and solved with ``scipy.optimize.linprog`` (HiGHS).

All steps respect the constraints required by ``evaluate_sequence``:
non‑negative entries, no explicit normalisation (the optimisation works
with a fixed ℓ¹‑norm `√(2 n)`, which makes the returned value equal to the
max‑convolution).  The routine runs until the time budget (≈1000 s) is
exhausted and returns the best sequence seen.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # type: ignore

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """Return (max convolution value, full convolution array) using FFT."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]
    return float(conv.max()), conv


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, active_idxs: np.ndarray):
    """
    Sub‑gradient of the peak of a*a at the current point.
    For each active shift `k` we add the shifted copy of `a`
    and finally average over the active set.
    """
    n = a.shape[0]
    idx = np.arange(n)
    grad = np.zeros_like(a)

    for k in active_idxs:
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad[mask] += a[j[mask]]

    if active_idxs.size > 0:
        grad /= float(active_idxs.size)
    return grad


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   target_sum: float,
                   deadline: float,
                   max_iters: int = 5_000_000):
    """
    Adam optimiser on the exact sub‑gradient of the peak of a*a.
    Returns the best vector seen and its peak value.
    """
    a = init_a.copy()
    a *= target_sum / a.sum()               # normalise once

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.06
    iteration = 0
    jitter_interval = 3000

    best_vec = a.copy()
    best_val, _ = _conv_max_fft(a)

    while time.time() < deadline and iteration < max_iters:
        M, conv = _conv_max_fft(a)

        # active shifts (ties are handled as a whole set)
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # sub‑gradient of the max term
        rev = _subgradient(a, active)

        # gradient of the full objective  (the factor 2·n/(Σa)² is 1 after normalisation)
        coeff = 4.0 * n / (target_sum * target_sum)         # =2
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = base_lr / (1.0 + 1e-4 * iteration)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the feasible simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a *= (target_sum / s)
        else:
            a[:] = target_sum / n

        # occasional jitter
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.0009 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= (target_sum / a.sum())

        # keep the best point
        if M < best_val:
            best_val = M
            best_vec = a.copy()
            print(f"[adam] iter {iteration:7d}  new best M = {best_val:.6f}")

    return best_vec, best_val


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  n: int,
                  target_sum: float,
                  steps: int,
                  deadline: float):
    """
    Greedy stochastic local search: move a tiny fraction of mass from a
    random source index to a random destination if the peak improves.
    """
    best = a.copy()
    best_val, _ = _conv_max_fft(best)

    for step in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        delta = best[i] * 0.015   # 1.5 % of the source
        if delta <= 0:
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        M, _ = _conv_max_fft(cand)
        if M < best_val:
            best = cand
            best_val = M

    return best, best_val


# ----------------------------------------------------------------------
def _solve_lp_for_g(f: np.ndarray, M: float):
    """
    Solve
        maximise Σ g_j
        subject to   g_j ≥ 0
                     (f * g)[k] ≤ M   for all k
    The matrix of the linear constraints (size (2n‑1) × n) is built
    explicitly – the (k, j) entry equals f_{k‑j} when the index is
    valid, otherwise 0.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # build the constraint matrix
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    c = -np.ones(n, dtype=np.float64)               # maximise Σ g  → minimise -Σ g
    b = np.full(m, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x

        ls_seq, ls_val = _local_search(best_seq, n, target_sum,
                                       steps=200_000, deadline=deadline)
        if ls_val < best_val:
            best_val, best_seq = ls_val, ls_seq.copy()
            print(f"[local] improved best to M = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) LP‑based refinement loop
    f = best_seq.copy()
    while time.time() < deadline:
        M, _ = _conv_max_fft(f)
        if M < best_val:
            best_val, best_seq = M, f.copy()
            print(f"[global] new best after LP loop M = {best_val:.6f}")

        # stop if we are already close to target
        if best_val <= 1.503:
            print("[stop] reached target bound")
            break

        try:
            g0 = _solve_lp_for_g(f, M)
        except Exception as exc:
            print("[lp] failed:", exc)
            break

        # rescale to the required ℓ¹‑norm (both f and the scaled g will have it)
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            print("[lp] zero‑sum solution – aborting LP stage")
            break
        g = g0 * (target_sum / sum_g0)

        improved = False
        # try a few mixing coefficients (larger t gives stronger move)
        for t in (0.3, 0.5, 0.7):
            candidate = (1.0 - t) * f + t * g
            candidate = np.maximum(candidate, 0.0)
            candidate *= target_sum / candidate.sum()
            M_new, _ = _conv_max_fft(candidate)
            if M_new < M:
                f = candidate
                improved = True
                print(f"[lp] mix t={t:.2f} reduced M from {M:.6f} → {M_new:.6f}")
                break

        if not improved:
            # if no improvement with any mixing, exit the LP loop
            print("[lp] no improving mix found – terminating LP phase")
            break

        # occasional tiny jitter to avoid stagnation
        if np.random.rand() < 0.07:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            f = np.maximum(f + jitter, 0.0)
            f *= target_sum / f.sum()

    # ------------------------------------------------------------------
    # final evaluation with the official function
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[done] best reported max = {best_val:.6f}, official score = {final_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
