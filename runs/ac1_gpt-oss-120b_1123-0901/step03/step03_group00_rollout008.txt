"""
Propose a step‑function (non‑negative sequence) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

The core idea is to place the mass on a *Sidon* (B₂) set of indices.
For a Sidon set every sum i+j (including i=j) is unique, therefore the
self‑convolution of the resulting probability vector has at most one
contribution per entry.  If the set contains m indices and the weight on
each index is 1/m, then

        max(a*a) = 1/m² ,   sum(a)=1   ⇒   F = 2·n/m² .

Choosing the largest possible Sidon set inside a window of size n gives
m≈√(2n) and the theoretical lower bound F≈1.  The routine builds such a
set with a simple greedy random search (restarted a few times), evaluates
the exact objective via an FFT‑based convolution, and keeps the best
sequence found before the time budget expires.

The algorithm runs completely within the provided time limit, has no
file or network I/O, and returns a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """
    Return max(conv(a,a)) using an FFT of appropriate length.
    The convolution length is 2·len(a)-1.
    """
    n = a.shape[0]
    fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
def _build_sid_set(n: int,
                   m: int,
                   max_trials_per_elem: int = 200,
                   max_restarts: int = 5) -> np.ndarray | None:
    """
    Greedy construction of a Sidon (B₂) set of size m inside {0,…,n‑1}.

    Returns the sorted array of chosen indices or None if construction
    fails after `max_restarts` attempts.
    """
    for _ in range(max_restarts):
        pos = []                     # chosen indices
        sums = set()                 # all pairwise sums i+j (i<=j)

        for _ in range(m):
            # try a few random candidates for the next element
            found = False
            for _ in range(max_trials_per_elem):
                cand = np.random.randint(0, n)
                if cand in pos:
                    continue

                # check whether any new sum would collide
                collides = False
                # self‑sum
                if (cand + cand) in sums:
                    collides = True
                else:
                    for p in pos:
                        if (cand + p) in sums:
                            collides = True
                            break

                if collides:
                    continue

                # accept cand
                for p in pos:
                    sums.add(cand + p)
                sums.add(cand + cand)
                pos.append(cand)
                found = True
                break

            if not found:
                # could not place the current element – restart whole set
                break

        if len(pos) == m:
            return np.array(sorted(pos), dtype=np.int64)

    return None


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value step‑function sequence within the given time budget.

    The search builds Sidon‑set based candidates for several lengths,
    evaluates the exact objective and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_val = float("inf")

    # try larger problems first – they give more freedom
    lengths = [8192, 4096, 2048, 1024]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Trying length n={n} ===")

        # theoretical maximal Sidon size (≈√(2n))
        m_opt = int(np.sqrt(2 * n))

        # we also test a couple of slightly smaller sizes in case the exact
        # optimum cannot be achieved inside the interval.
        size_candidates = [m_opt, m_opt - 1, m_opt - 2]

        for m in size_candidates:
            if m < 2:
                continue
            if time.time() > deadline:
                break

            # a few independent attempts per (n,m) pair
            attempts = 5
            for att in range(attempts):
                if time.time() > deadline:
                    break

                pos = _build_sid_set(n, m,
                                     max_trials_per_elem=200,
                                     max_restarts=1)

                if pos is None:
                    # could not find a Sidon set of this size – try a smaller one
                    continue

                # construct the probability vector (equal weight on the Sidon set)
                a = np.zeros(n, dtype=np.float64)
                weight = 1.0 / m
                a[pos] = weight

                # exact evaluation (sum(a) == 1, so denominator = 1)
                max_c = _max_self_convolution(a)
                val = 2.0 * n * max_c   # same as evaluate_sequence(a)

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[n={n} m={m} att={att}] new best value: {val:.6f}")

                # early exit if we are already at the theoretical lower bound
                if best_val <= 1.01:
                    break
            if best_val <= 1.01:
                break
        if best_val <= 1.01:
            break

    # ------------------------------------------------------------------
    # Fallback: uniform distribution (should never be needed)
    if best_seq is None:
        n_fallback = 2000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = 2.0 * n_fallback * (1.0 / n_fallback)   # =2.0
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    # Convert to plain Python list as required by the external evaluator.
    return best_seq.tolist()
