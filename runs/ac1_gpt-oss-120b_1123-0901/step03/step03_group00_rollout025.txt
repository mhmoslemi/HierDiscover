def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np, time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # Candidate lengths (powers of two up to a safe limit)
    lengths = [256, 512, 1024, 2048, 4096, 8192]
    best_seq = None
    best_val = float('inf')
    for n in lengths:
        if time.time() > deadline:
            break
        a = np.arange(1, n+1, dtype=np.float64)
        a = a / a.sum()
        # compute evaluation using the provided function to guarantee correct handling
        val = evaluate_sequence(a.tolist())
        print(f"length={n}, val={val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a.copy()
    if best_seq is None:
        # fallback: simple uniform
        n = 1024
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    return best_seq.tolist()
