        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)
        p = exp_shift / exp_shift.sum()
    return best_p, best_score

def lp_mix(a, max_lp_iters=1, t=0.2, deadline=None):
    n = a.shape[0]
    S = math.sqrt(2.0 * n)
    # ensure sum close to S
    current_sum = a.sum()
    if abs(current_sum - S) > 1e-6:
        a = a * (S / current_sum)
    # compute M
    conv = np.convolve(a, a)
    M = conv.max()
    # Build A matrix
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = a  # because a is f here
    b = np.full(m, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximize sum g i.e. minimize -sum

    # Solve LP
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')

    else:
        g0 = res.x

    if sum_g0 <= 0:
        g = np.full(n, S / n)
    else:
        g = g0 * (S / sum_g0)  # rescale
    # Mix
    a_new = (1 - t) * a + t * g
    # Compute new M and return improvement flag
    conv_new = np.convolve(a_new, a_new)
    new_M = conv_new.max()
    improved = new_M < M  # maybe
    return a_new, improved

"""
Propose a low‑peak self‑convolution step function.

The algorithm works on the probability vector  p  (∑p = 1) and
optimises  max(p∗p)  by a combination of:

* Exponentiated‑gradient (mirror descent on the simplex) with a
  soft‑max approximation of the max.  The gradient is obtained by a
  weighted correlation computed by FFTs.  A schedule drives the
  soft‑max temperature β from a tiny value to a fairly large one,
  turning the smoothed objective into the true max.

* A linear‑programming “mixing’’ step (inspired by the paper cited
  in the prompt).  Given the current vector f, we solve
      maximise   Σ g_i
      subject to f∗g ≤ M   and   g ≥ 0,
  where M = max(f∗f).  The optimal g₀ is rescaled to the
  canonical ℓ₁‑norm √(2 n) and mixed with f:
      f ← (1‑t)·f + t·g .
  This monotone mixing often pushes the peak a little lower and helps
  the gradient method escape plateaus.

The search proceeds over a multiscale schedule of lengths (500 … 4000
elements).  For each length we initialise the probability vector by
interpolating the best solution found so far, run a medium‑length EG
phase, perform a few LP‑mixing steps, and finally refine again with
EG.  The best sequence (scaled back to the original amplitudes) is
returned.  All operations respect the 1000 s time budget.
"""

import time
import math
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """
    Full self‑convolution (p∗p) using a single FFT of size N.
    Returns real array of length 2·len(p)−1.
    """
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute   rev_i = Σ_k w_k·p_{k‑i}   for i = 0…n‑1.
    This is a correlation of p with the weight vector w and is obtained
    via FFT (one forward transform of p and one of the reversed weights).
    """
    n = p.shape[0]
    L = w.shape[0]                     # = 2·n‑1
    P = np.fft.rfft(p, n=N)
    W_rev = np.fft.rfft(w[::-1], n=N)
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    conv_rev = conv_rev[: n + L - 1]
    rev = conv_rev[L - n : L][::-1]
    return rev

def _eg_optimize(p_init: np.ndarray,
                 max_iters: int,
                 deadline: float,
                 beta_max: float = 30.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimisation of max(p∗p) for a fixed length.
    Returns the best probability vector found and its score
    (2·n·max(p∗p)).
    """
    n = p_init.shape[0]
    p = p_init.copy()
    # work in log‑space to keep the iterate on the simplex
    logp = np.log(p + 1e-30)

    # FFT size large enough for both the convolution (2n‑1) and the
    # correlation with the weight vector (≤3n‑2)
    N = _next_pow2(3 * n - 2)

    best_score = float("inf")
    best_p = p.copy()

    beta_start = 0.01
    beta_end = beta_max
    beta_len = max_iters // 2          # linear schedule

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- self‑convolution p∗p -----
        conv = _full_convolution_fft(p, N)
        M = conv.max()
        cur_score = 2.0 * n * M

        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[EG n={n:4d} it={it:6d}] new local best = {cur_score:.6f}")

        # ----- soft‑max weighting (smooth approximation of the max) -----
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_len)
        # w_k = exp(beta·(conv_k−M))  – the maximal entry gets weight 1
        w = np.exp(beta * (conv - M))

        # ----- correlation needed for the gradient -----
        rev = _weighted_rev_fft(p, w, N)

        # gradient of 2·n·max_β(p∗p)  (see derivation in the prompt)
        grad = 4.0 * n * rev

        # step size (decreasing)
        eta = (1.0 / (4.0 * n)) / (1.0 + it * 1e-4)

        # EG update in log‑space
        logp -= eta * grad

        # occasional jitter to escape plateaus
        if it % 4000 == 0:
            logp += np.random.randn(n) * 0.001

        # projection back onto the simplex (via normalisation)
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)   # avoid overflow
        p = exp_shift / exp_shift.sum()

    return best_p, best_score


# ----------------------------------------------------------------------
def _lp_mix(a: np.ndarray,
            t: float = 0.2,
            deadline: float = None) -> tuple[np.ndarray, bool]:
    """
    One linear‑programming mixing step.
    Input a has the canonical ℓ₁‑norm √(2·n).
    Returns a new vector (still with that norm) and a flag indicating
    whether the peak decreased.
    """
    n = a.shape[0]
    S = math.sqrt(2.0 * n)

    # ---- current peak M ----
    conv = np.convolve(a, a)
    M = conv.max()

    # ---- build the constraint matrix A (size (2n‑1)×n) ----
    # (f∗g)[k] = Σ_i f_i·g_{k‑i}, i.e. column i of A contains f shifted
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a                     # f_i appears in rows i … i+n‑1

    # ---- linear program: maximise Σ g_i  (i.e. minimise -Σ g_i) ----
    c = -np.ones(n, dtype=np.float64)               # -Σ g_i
    bounds = [(0.0, None)] * n
    b = np.full(m, M, dtype=np.float64)             # f∗g ≤ M

    # Solve with HiGHS (fast interior‑point solver)
    res = scipy.optimize.linprog(
        c, A_ub=A, b_ub=b, bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": max(0.0, (deadline - time.time()))}
        if deadline is not None else {}
    )

    if not res.success:
        # fallback: uniform distribution
        g0 = np.full(n, S / n, dtype=np.float64)
    else:
        g0 = res.x

    sum_g0 = g0.sum()
    if sum_g0 <= 0.0:
        g = np.full(n, S / n, dtype=np.float64)
    else:
        g = g0 * (S / sum_g0)          # rescale to the canonical ℓ₁‑norm

    # ---- mix and return ----
    a_new = (1.0 - t) * a + t * g
    # evaluate whether the peak actually went down
    improved = np.convolve(a_new, a_new).max() < M
    return a_new, improved

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine follows the multiscale scheme
    described above and returns the best sequence found before the time
    budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # tiny safety margin

    # --------------------------------------------------------------
    # Try to start from the historic construction if it exists.
    try:
        base = np.array(height_sequence_1, dtype=np.float64)  # type: ignore
    except Exception:
        base = np.ones(500, dtype=np.float64)

    n0 = base.shape[0]
    S0 = math.sqrt(2.0 * n0)
    base = base * (S0 / base.sum())       # normalise to the canonical norm
    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n0}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # Multiscale schedule – we keep the length modest so that the
    # LP steps stay affordable.
    length_schedule = [500, 800, 1200, 1700, 2300, 3000, 3800, 4600]
    length_schedule = [l for l in length_schedule if 300 <= l <= 5000]
    length_schedule = sorted(set(length_schedule))

    for n in length_schedule:
        if time.time() >= deadline:
            break

        # ----- initialise probability vector for this length -----
        if best_seq.shape[0] == n:
            p = best_seq / math.sqrt(2.0 * n)           # probability vector
        else:
            # interpolate the current best probability onto the new grid
            p_old = best_seq / math.sqrt(2.0 * best_seq.shape[0])
            x_old = np.linspace(0.0, 1.0, p_old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            p = np.interp(x_new, x_old, p_old)
            p = np.maximum(p, 0.0)
            p /= p.sum()

        print(f"\n[run] optimizing length = {n}")

        # ----------------------------------------------------------
        # Phase 1 – exponentiated gradient optimisation
        # Allocate a chunk of the remaining time for this phase.
        remaining = deadline - time.time()
        iters_phase1 = max(2000, int(min(30000, remaining * 10)))  # heuristic
        p_opt, local_score = _eg_optimize(
            p_init=p,
            max_iters=iters_phase1,
            deadline=deadline,
            beta_max=40.0,
        )
        a_opt = p_opt * math.sqrt(2.0 * n)
        cur_score = evaluate_sequence(a_opt.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_opt.copy()
            print(f"[len {n}] after EG phase 1 improved to {best_score:.6f}")

        # ----------------------------------------------------------
        # Phase 2 – a few LP mixing steps (each followed by a short EG)
        for mix_idx in range(3):
            if time.time() >= deadline:
                break
            a_opt, improved = _lp_mix(a_opt, t=0.18, deadline=deadline)
            cur_score = evaluate_sequence(a_opt.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_opt.copy()
                print(f"[len {n}] after LP {mix_idx+1} improved to {best_score:.6f}")

            # A short refinement with EG after the mixing
            p_refine = a_opt / math.sqrt(2.0 * n)
            remaining = deadline - time.time()
            iters_refine = max(1000, int(min(8000, remaining * 8)))
            p_refine, _ = _eg_optimize(
                p_init=p_refine,
                max_iters=iters_refine,
                deadline=deadline,
                beta_max=30.0,
            )
            a_opt = p_refine * math.sqrt(2.0 * n)
            cur_score = evaluate_sequence(a_opt.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_opt.copy()
                print(f"[len {n}] after EG refine {mix_idx+1} improved to {best_score:.6f}")

        # ----------------------------------------------------------
        # Phase 3 – a final long EG run on the current best for this length
        remaining = deadline - time.time()
        iters_final = max(2000, int(min(20000, remaining * 6)))
        p_final, _ = _eg_optimize(
            p_init=a_opt / math.sqrt(2.0 * n),
            max_iters=iters_final,
            deadline=deadline,
            beta_max=50.0,
        )
        a_final = p_final * math.sqrt(2.0 * n)
        cur_score = evaluate_sequence(a_final.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_final.copy()
            print(f"[len {n}] final EG improvement to {best_score:.6f}")

    # --------------------------------------------------------------
    print(f"\n[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
