def adam_optimize(a, steps, deadline):
    n=len(a)
    conv_fft_size = 1 << int(np.ceil(np.log2(2*n-1)))
    size_corr= 1 << int(np.ceil(np.log2(3*n)))
    u = np.log(a + 1e-30)
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1=0.9; beta2=0.999; eps=1e-8; base_lr=0.05; lam=0.5; lam_factor=1.02
    it=0
    while it<steps and time.time()<deadline:
        it+=1
        # compute a
        u_shift = u - np.max(u)
        a = np.exp(u_shift)
        a /= a.sum()
        # compute convolution
        conv = _full_convolution(a, size=conv_fft_size)
        w = _softmax_weights(conv, lam)
        # gradient
        grad_a = _softmax_gradient(a, w, size_corr=size_corr)
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)

        # Adam update
        m = beta1*m + (1-beta1)*grad_u
        v = beta2*v + (1-beta2)*(grad_u**2)
        m_hat = m/(1-beta1**it)
        v_hat = v/(1-beta2**it)
        lr = base_lr * (0.9995**it)
        u -= lr * m_hat/(np.sqrt(v_hat)+eps)
        lam *= lam_factor
    # final retrieve a
    u_shift = u - np.max(u)
    a = np.exp(u_shift)
    a /= a.sum()
    return a

"""
Hybrid optimizer for the step‑function bound.

The algorithm combines three ideas:

1. **Adam on a smooth‑max surrogate** – identical in spirit to the
   previous code but run for a limited number of iterations each
   outer cycle.

2. **Linear‑programming boost** – for the current sequence `a`,
   solve the LP  

        maximise   Σ b_i
        subject to  a * b ≤ M    (component‑wise)
                    b_i ≥ 0

   where `M = max (a * a)`.  The LP raises the ℓ¹‑mass while keeping the
   maximal convolution no larger than before.  Mixing the LP solution
   with the current sequence strictly improves the objective.

3. **Mix & repeat** – after a short Adam phase we perform the LP boost,
   mix the two sequences (convex combination) and start the next
   cycle.  The loop runs until the time budget expires (≈1000 s).

The routine works for any initial `height_sequence_1` (if present) or
starts from a random Dirichlet vector.  All sequences are kept on the
probability simplex (sum = 1), which does not affect the scale‑invariant
objective.
"""

import time
import numpy as np
from scipy import optimize as opt

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, exact on the linear convolution length)
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of appropriate length."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smooth‑max surrogate (soft‑max) and its gradient (used by Adam)
def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """w_k = exp(lam·conv_k) / Σ_j exp(lam·conv_j)."""
    m = conv.max()
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int = None) -> np.ndarray:
    """
    Gradient of φ(a)= log‑sum‑exp(lam·(a*a))  w.r.t. a.
    Result: ∇φ = 2·(w ★ a)   (cross‑correlation).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Exact objective (the same quantity that ``evaluate_sequence`` returns)
def _objective_val(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a))²."""
    n = a.shape[0]
    conv = _full_convolution(a)
    M = conv.max()
    s = a.sum()
    return float(2.0 * n * M / (s * s))


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
def _adam_optimize(a0: np.ndarray,
                   max_iters: int,
                   deadline: float) -> np.ndarray:
    """Run a few hundred Adam steps starting from ``a0``."""
    n = a0.shape[0]
    # FFT sizes – reused each iteration
    conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    corr_fft_size = 1 << int(np.ceil(np.log2(3 * n)))

    # Log‑softmax parametrisation
    u = np.log(a0 + 1e-30)

    # Adam state
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    # Hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.05
    lam = 1.0          # start of smoothing
    lam_factor = 1.02  # growth per iteration

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- probabilities from logits -----
        u_shift = u - np.max(u)
        a = np.exp(u_shift)
        a /= a.sum()

        # ----- smooth‑max weights -----
        conv = _full_convolution(a, size=conv_fft_size)
        w = _softmax_weights(conv, lam)

        # ----- gradient w.r.t. a -----
        grad_a = _softmax_gradient(a, w, size_corr=corr_fft_size)

        # ----- transform gradient to log‑space (soft‑max Jacobian) -----
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = base_lr * (0.9995 ** it)
        u -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- increase smoothing -----
        lam *= lam_factor

    # final projection onto simplex
    u_shift = u - np.max(u)
    a = np.exp(u_shift)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming boost (maximise ℓ¹‑mass under convolution bound)
def _lp_boost(a: np.ndarray, M: float, max_constraints: int = 200) -> np.ndarray:
    """
    Solve  maximise Σ b_i   subject to   a * b ≤ M  (component‑wise), b ≥ 0.
    Only the ``max_constraints`` most binding convolution indices are kept.
    Returns the optimal vector ``b`` (may be all‑zeros if LP fails).
    """
    n = a.shape[0]

    # full self‑convolution of a
    conv = _full_convolution(a)
    # pick the largest entries (indices where the bound is most tight)
    if max_constraints >= conv.size:
        idx = np.arange(conv.size)
    else:
        # argpartition gives unordered top‑k; we keep them as a set
        idx = np.argpartition(conv, -max_constraints)[-max_constraints:]

    # Build constraint matrix A_ub (rows = selected convolution indices)
    rows = np.zeros((len(idx), n), dtype=np.float64)
    for r, k in enumerate(idx):
        # b_j appears with coefficient a_{k-j} when 0≤j<n and 0≤k-j<n
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # indices j = lo … hi
        # corresponding a‑coefficients = a[k - j]
        # vectorised slice:
        rows[r, lo:hi + 1] = a[k - np.arange(lo, hi + 1)]

    # Linear program: maximise Σ b_i  → minimise -Σ b_i
    c = -np.ones(n, dtype=np.float64)   # objective
    ub = np.full(rows.shape[0], M, dtype=np.float64)   # right‑hand side

    # bounds for each variable b_i ≥ 0
    bounds = [(0.0, None)] * n

    # Solve with HiGHS (fast interior‑point/simplex hybrid)
    res = opt.linprog(c, A_ub=rows, b_ub=ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 5.0})
    if not res.success:
        # failed – fall back to a zero vector (will be ignored by the caller)
        return np.zeros_like(a)
    b = res.x
    # Normalise to keep the iterate on the simplex (not required for feasibility)
    if b.sum() > 0:
        b = b / b.sum()
    return b


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Hybrid Adam + LP search for a low‑value step function.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin for the caller

    # --------------------------------------------------------------
    # Choose a problem size.
    # 3000 works well on a modern CPU within the 1000 s budget.
    n = 3000

    # --------------------------------------------------------------
    # Initialise from a previously found construction if available.
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)) and len(init_seq) == n:
        a = np.array(init_seq, dtype=np.float64)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.random.dirichlet(np.ones(n))
        else:
            a = a / a_sum
    else:
        # fall back to a random Dirichlet start
        a = np.random.dirichlet(np.ones(n))

    # Keep track of the best solution found so far
    best_a = a.copy()
    best_val = _objective_val(best_a)

    print(f"[Init] objective = {best_val:.6f}")

    # --------------------------------------------------------------
    # Main outer loop – alternate short Adam phases with LP boosts.
    outer_iter = 0
    while time.time() < deadline:
        outer_iter += 1
        time_left = deadline - time.time()
        if time_left < 1.0:
            break

        # ------------------------------------------------------------------
        # 1️⃣  Adam optimisation (short burst)
        # ------------------------------------------------------------------
        # Allocate a modest number of steps – enough to make progress
        # but cheap enough to leave time for the LP phases.
        max_adam_steps = 1500
        a_adam = _adam_optimize(a, max_adam_steps, deadline)

        val_adam = _objective_val(a_adam)
        if val_adam < best_val:
            best_val = val_adam
            best_a = a_adam.copy()
            print(f"[Iter {outer_iter:03d}] after Adam  val={val_adam:.6f}")

        # ------------------------------------------------------------------
        # 2️⃣  Linear‑programming boost
        # ------------------------------------------------------------------
        # Compute current maximal self‑convolution
        conv = _full_convolution(a_adam)
        M = conv.max()

        # Solve LP to enlarge ℓ¹‑mass while respecting the bound M
        b_lp = _lp_boost(a_adam, M, max_constraints=250)

        # If the LP failed (zero vector) we skip mixing.
        if b_lp.sum() > 0:
            # Mix with a convex coefficient t (t=0.3 works well empirically)
            t = 0.3
            a_mix = (1.0 - t) * a_adam + t * b_lp
            # Project back to the simplex – keeps the scale invariant
            a_mix = np.maximum(a_mix, 0.0)
            if a_mix.sum() == 0.0:
                a_mix = np.random.dirichlet(np.ones(n))
            else:
                a_mix = a_mix / a_mix.sum()

            val_mix = _objective_val(a_mix)
            if val_mix < best_val:
                best_val = val_mix
                best_a = a_mix.copy()
                print(f"[Iter {outer_iter:03d}] after LP‑mix val={val_mix:.6f}")

            # Prepare for the next outer iteration
            a = a_mix
        else:
            # LP gave nothing useful – keep the Adam result as the next start.
            a = a_adam

        # ------------------------------------------------------------------
        # Optional: occasional random shake if we have stalled for a while.
        # ------------------------------------------------------------------
        if outer_iter % 7 == 0:
            # tiny perturbation in log‑space to escape flat regions
            eps_shake = 2e-4
            log_a = np.log(best_a + 1e-30)
            log_a += (np.random.rand(n) - 0.5) * eps_shake
            a = np.exp(log_a - np.max(log_a))
            a /= a.sum()
            # Re‑evaluate after the shake
            val_shake = _objective_val(a)
            if val_shake < best_val:
                best_val = val_shake
                best_a = a.copy()
                print(f"[Iter {outer_iter:03d}] shake improved val={val_shake:.6f}")

    # ----------------------------------------------------------------------
    # Final report
    print(f"Final best objective = {best_val:.6f} (length {len(best_a)})")
    # Return a plain Python list as required by the harness.
    return best_a.tolist()
