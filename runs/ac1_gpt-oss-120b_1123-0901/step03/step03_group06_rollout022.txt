def _project_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a high‑quality step‑function using Adam on the log‑probabilities.
The optimisation directly minimises the maximum entry of the self‑convolution
(= the evaluation score after the internal normalisation).  For a given
length n we treat the coefficients as a probability vector p (Σp=1) and
optimise the objective

    F(p) = 2·n·max_k (p * p)_k .

The sub‑gradient of max_k (p*p)_k is obtained analytically:
if K = {k | (p*p)_k = M} and M = max(p*p) then

    ∂F/∂p_i = 4·n· Σ_{k∈K} p_{k-i} ,

i.e. a correlation of the indicator of the active constraints with p.
We perform Adam updates on the log‑probabilities (mirror descent on the
simplex) and periodically re‑project onto the simplex.
A multiscale schedule (500 → 1000 → 1500 → 2100 → 3000 → 5000 → 8000)
is used: the best solution is up‑sampled to the next length and further
optimised.  The initial distribution for each length is a “U‑shaped”
Beta(½,½) (arcsine) density, which is known to produce a flatter
self‑convolution than a uniform start.  Several random restarts and a
tiny jitter are added to escape plateaus.  The whole search respects the
global time budget (default 1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """Linear self‑convolution p∗p (length 2·len(p)‑1) using FFT of size N."""
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _correlation_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute the correlation rev_i = Σ_k w_k·p_{k-i} for i=0..n‑1.
    w must have length 2·n‑1 (full convolution length).
    """
    n = p.shape[0]
    L = w.shape[0]                     # = 2·n‑1
    # FFT of p (size N)
    P = np.fft.rfft(p, n=N)
    # FFT of the reversed weight vector
    W_rev = np.fft.rfft(w[::-1], n=N)
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    conv_rev = conv_rev[: n + L - 1]   # keep linear part
    # Extract rev = conv_rev[L‑n : L] and reverse
    rev = conv_rev[L - n : L][::-1]
    return rev


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returns a vector w with w_i ≥ 0 and Σ w_i = 1.
    """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / len(v))
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _arcsine_init(n: int) -> np.ndarray:
    """
    Initialise a probability vector with a Beta(½,½) (U‑shaped) density.
    The density ∝ 1/√(x(1‑x)), discretised and normalised.
    """
    i = np.arange(n, dtype=np.float64) + 0.5        # avoid exact 0/1
    x = i / n
    p = 1.0 / np.sqrt(x * (1.0 - x))
    p = np.nan_to_num(p, nan=0.0, posinf=0.0)
    p = np.maximum(p, 1e-12)                       # avoid log(0)
    p /= p.sum()
    return p

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Starting point (if a previous construction is available)
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
        n0 = base.shape[0]
        S0 = math.sqrt(2.0 * n0)
        base = base * (S0 / base.sum())
    except Exception:
        # fallback: arcsine shape of moderate length
        n0 = 800
        base = _arcsine_init(n0) * math.sqrt(2.0 * n0)

    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n0}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Multiscale schedule (progressively larger lengths)
    length_schedule = [500, 1000, 1500, 2100, 3000, 5000, 8000]
    length_schedule = [l for l in length_schedule if l >= 300]
    length_schedule = sorted(set(length_schedule))

    # optimisation hyper‑parameters (common to all lengths)
    max_iters_per_len = 150_000          # safety cap per length
    jitter_interval = 4000
    jitter_scale = 0.0008
    beta_start = 0.1
    beta_end = 30.0                      # a larger final β than before
    beta_ramp_iters = 80_000             # linear growth of β
    lr_initial = 0.25                    # initial Adam learning‑rate
    lr_decay = 1e-5                      # simple decay factor
    beta1 = 0.9                          # Adam first‑moment decay
    beta2 = 0.999                        # Adam second‑moment decay
    eps_adam = 1e-8

    for n in length_schedule:
        if time.time() >= deadline:
            break

        # ----- initialise probability vector for the new length -----
        if best_seq.shape[0] == n:
            p0 = best_seq / math.sqrt(2.0 * n)
        else:
            # up‑sample the current best (probability) shape
            p_old = best_seq / math.sqrt(2.0 * best_seq.shape[0])
            old_x = np.linspace(0.0, 1.0, p_old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            p0 = np.interp(new_x, old_x, p_old)
            p0 = np.maximum(p0, 1e-12)
            p0 /= p0.sum()
        print(f"[run] start length = {n}")

        # FFT size big enough for both self‑convolution (2n‑1) and
        # correlation with a weight vector (3n‑2)
        N = _next_pow2(3 * n - 2)

        # ------------------------------------------------------------------
        # Run a few random restarts; the first one uses the up‑sampled p0,
        # the others start from a fresh arcsine or Dirichlet draw.
        n_restarts = 3
        best_local_p = None
        best_local_score = float("inf")

        for restart in range(n_restarts):
            if time.time() >= deadline:
                break

            if restart == 0:
                p = p0.copy()
            else:
                # a fresh random start (Dirichlet α=0.5 gives U‑shaped draws)
                raw = np.random.gamma(0.5, 1.0, size=n)
                p = raw / raw.sum()

            # Adam state (operates on log‑probabilities)
            logp = np.log(p + 1e-30)
            m = np.zeros_like(logp)
            v = np.zeros_like(logp)

            cur_best_score = float("inf")
            cur_best_p = p.copy()

            it = 0
            while time.time() < deadline and it < max_iters_per_len:
                it += 1

                # ----- self‑convolution -----
                conv = _full_convolution_fft(p, N)
                M = conv.max()
                cur_score = 2.0 * n * M

                # track best local solution
                if cur_score < cur_best_score:
                    cur_best_score = cur_score
                    cur_best_p = p.copy()
                    print(f"[len {n:5d} restart {restart} iter {it:7d}] new local best = {cur_score:.6f}")

                # ----- gradient of the (soft) max -----
                # smooth‑max weight schedule
                beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_ramp_iters)
                w = np.exp(beta * (conv - M))          # w_k ∈ (0,1], w_max = 1

                # correlation (rev) = Σ_k w_k·p_{k-i}
                rev = _correlation_fft(p, w, N)

                # gradient of 2·n·max (soft) w.r.t p
                grad = 4.0 * n * rev

                # ------ Adam update on log‑probabilities ------
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * (grad * grad)

                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)

                lr = lr_initial / (1.0 + lr_decay * it)

                logp -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                # occasional jitter to escape plateaus
                if it % jitter_interval == 0:
                    logp += np.random.randn(n) * jitter_scale

                # ----- project back onto the simplex (via exponentiation) -----
                max_log = np.max(logp)
                exp_shift = np.exp(logp - max_log)       # avoid overflow
                p = exp_shift / exp_shift.sum()

                # early termination if we have not improved for a while
                if it % 5000 == 0 and cur_best_score < best_local_score * 0.999:
                    # found a new significant improvement – keep going
                    pass
                if it % 20000 == 0 and (time.time() > deadline - 30):
                    # close to the deadline – stop early
                    break

            # keep the best of the restarts
            if cur_best_score < best_local_score:
                best_local_score = cur_best_score
                best_local_p = cur_best_p.copy()

        # ------------------------------------------------------------------
        # Convert the best probability vector for this length back to the
        # raw heights (scale factor S = √(2·n)).
        S = math.sqrt(2.0 * n)
        a_opt = best_local_p * S
        cur_score = evaluate_sequence(a_opt.tolist())

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_opt.copy()
            print(f"[len {n}] improved global best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback safety net (should never be needed)
    if best_seq is None:
        n = 800
        S = math.sqrt(2.0 * n)
        best_seq = np.full(n, S / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
