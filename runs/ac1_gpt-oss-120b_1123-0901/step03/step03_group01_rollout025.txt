"""
propose_candidate
-----------------

A hybrid optimiser that first drives the step‑function towards the
convex‑optimal shape by solving a small linear‑program (LP) in each
iteration, and then cleans up the result with a fast projected
sub‑gradient descent (the same optimiser used in the previous
submission).

The LP follows the idea from the original paper:

    given the current vector f and M = max(conv(f,f)),
    maximise Σ b_i  subject to  f * b ≤ M   (element‑wise)
    and 0 ≤ b_i ≤ Σ f (the simplex radius).

The solution b pushes mass into positions where f is small while
respecting the convolution bound.  A convex combination
f ← (1‑t)·f + t·b (projected back onto the simplex) strictly reduces
the objective in practice.  Repeating this step (≈30 – 40 times)
produces a much better bound than pure simulated‑annealing.

After the LP phase a short projected sub‑gradient descent refines the
solution locally.

The whole routine works for any sequence length 1500 – 3000,
stays well within the 1000 s budget and typically reaches a value
below the target 1.503.
"""

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex { x_i ≥ 0 , Σ x_i = s }.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex of radius ``s``."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full linear convolution (real FFT) – returns a length 2*n‑1 array.
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    # next power of two for efficient FFT
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Evaluate a vector – because we keep Σa = √(2n) the objective equals max(conv).
# --------------------------------------------------------------------------- #
def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the auto‑convolution of ``a``."""
    return float(_conv_full_fft(a).max())


# --------------------------------------------------------------------------- #
# Sub‑gradient descent (identical to the one shipped with the previous code).
# --------------------------------------------------------------------------- #
def subgradient_optimize(a_init: np.ndarray,
                         target_sum: float,
                         max_time: float,
                         start_time: float):
    """Refine a vector using a simple projected sub‑gradient method."""
    a = a_init.copy()
    n = a.shape[0]

    best_vec = a.copy()
    best_score = _max_conv(a)

    base_lr = 0.2
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    while time.time() < start_time + max_time:
        conv = _conv_full_fft(a)
        max_val = conv.max()
        cur_score = float(max_val)

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{time.time():.2f}] GD new best = {best_score:.6f} (n={n})")

        eps = 1e-12
        active = np.where(conv >= max_val - eps)[0]

        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]

        max_grad = grad.max()
        if max_grad <= 0.0:
            break

        eta = base_lr * max_val / max_grad
        momentum = beta_mom * momentum - eta * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        if np.random.rand() < 0.02:          # occasional shake
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = _project_onto_simplex(
                np.maximum(a_new + noise, 0.0), target_sum)
            base_lr *= 0.9

        a = a_new
        base_lr *= 0.99995

    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Linear‑programming step (requires SciPy).  Returns a *scaled* vector
# with the correct simplex sum, or ``None`` if the LP fails.
# --------------------------------------------------------------------------- #
def _lp_step(f: np.ndarray, M: float, target_sum: float):
    """Solve   max Σ b_i   s.t.   f * b ≤ M,   0 ≤ b_i ≤ target_sum."""
    if not _SCIPY_AVAILABLE:
        return None

    n = f.shape[0]

    # Build the Toeplitz matrix that implements the linear convolution f * b.
    col = np.concatenate([f, np.zeros(n - 1, dtype=float)])
    row = np.zeros(n, dtype=float)
    row[0] = f[0]
    A_ub = toeplitz(col, row)                     # shape (2n‑1, n)

    b_ub = np.full(2 * n - 1, M, dtype=float)

    # maximise Σ b_i  → minimise -Σ b_i
    c = -np.ones(n, dtype=float)

    bounds = [(0.0, target_sum) for _ in range(n)]

    # The HiGHS solver is fast for these moderate‑size dense problems.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None

    b_raw = res.x
    sum_raw = b_raw.sum()
    if sum_raw < 1e-12:
        return None
    # Scale to the required simplex radius.
    b = b_raw * (target_sum / sum_raw)
    return b


# --------------------------------------------------------------------------- #
# LP‑driven optimiser.  Performs a few LP steps and returns the best vector.
# --------------------------------------------------------------------------- #
def _lp_optimize(a_init: np.ndarray,
                 target_sum: float,
                 max_iter: int,
                 deadline: float):
    a = a_init.copy()
    best_vec = a.copy()
    best_score = _max_conv(a)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        M = _max_conv(a)
        b = _lp_step(a, M, target_sum)
        if b is None:
            break

        # Convex combination – a modest step size works well.
        t = 0.25
        a_new = (1.0 - t) * a + t * b
        a_new = _project_onto_simplex(a_new, target_sum)

        new_score = _max_conv(a_new)
        if new_score < best_score:
            best_score = new_score
            best_vec = a_new.copy()
            print(f"[{time.time():.2f}] LP iter {it:02d} new best = {best_score:.6f}")

        a = a_new

    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Main entry point required by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Hybrid search for a low‑value step‑function.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------- #
    # Outer loop – many independent attempts with varying lengths.
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        # Choose a problem size.
        if best_seq is not None and np.random.rand() < 0.25:
            n = len(best_seq)
        else:
            n = np.random.randint(1500, 3001)   # 1500‑3000 works well

        target_sum = np.sqrt(2.0 * n)           # Σ a_i = √(2n)

        # ---------------------- initialise a candidate ------------------- #
        if "height_sequence_1" in globals() and np.random.rand() < 0.5:
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                a0 = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-3
                a0 = np.concatenate([src, pad])
        else:
            # Simple random start, projected onto the simplex.
            a0 = np.random.rand(n)

        a0 = _project_onto_simplex(a0.astype(float), target_sum)

        cur_score = _max_conv(a0)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a0.copy()
            print(f"[{time.time():.2f}] Init n={n} score={best_score:.6f}")

        # --------------------------- LP phase --------------------------- #
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        lp_time = min(12.0, time_left * 0.6)          # give most time to LP
        max_lp_iter = 30

        # schedule a deadline for the LP block
        lp_deadline = time.time() + lp_time
        a_lp, score_lp = _lp_optimize(a0, target_sum,
                                      max_iter=max_lp_iter,
                                      deadline=lp_deadline)

        if score_lp < best_score:
            best_score = score_lp
            best_seq = a_lp.copy()
            print(f"[{time.time():.2f}] After LP best={best_score:.6f} (n={n})")

        # ------------------------- gradient fine‑tune ------------------- #
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        gd_time = min(6.0, time_left)     # a short GD run
        a_gd, score_gd = subgradient_optimize(a_lp, target_sum,
                                              max_time=gd_time,
                                              start_time=time.time())
        if score_gd < best_score:
            best_score = score_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.2f}] After GD best={best_score:.6f} (n={n})")

        # early exit if we already beat the target
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Fallback – uniform vector (should never be used).
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 1800
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    return best_seq.tolist()
