    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """Linear self‑convolution (p∗p) of length 2·len(p)−1 using FFT size N."""
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k·p_{k−i} for i=0..n−1.
    This is the correlation of p with w (implemented via FFT).
    """
    n = p.shape[0]
    L = w.shape[0]                # = 2·n−1
    P = np.fft.rfft(p, n=N)
    # Reverse w for correlation
    W_rev = np.fft.rfft(w[::-1], n=N)
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    conv_rev = conv_rev[: n + L - 1]          # linear part
    # Extract the slice corresponding to i=0..n‑1 and reverse it
    rev = conv_rev[L - n : L][::-1]
    return rev


def _simplex_normalise(v: np.ndarray) -> np.ndarray:
    """
    Simple (fast) projection onto the probability simplex:
    set negative entries to zero and renormalise to sum 1.
    If the vector becomes all zero, fall back to the uniform distribution.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


def _optimize_adam_length(
    n: int,
    p_start: np.ndarray,
    deadline: float,
    max_iters: int,
    beta_start: float = 0.5,
    beta_end: float = 400.0,
    beta_len: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation for a fixed length ``n``.
    Returns the best probability vector (sum = 1) found and its objective
    ``2·n·max(p∗p)`` (the value returned by ``evaluate_sequence`` after scaling).
    """
    # FFT size sufficient for both the convolution (2n‑1) and the correlation
    N = _next_pow2(3 * n - 2)

    p = p_start.copy()
    # Adam state
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    # learning‑rate schedule (large at the start, decays)
    alpha0 = 0.2

    best_score = float("inf")
    best_p = p.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ---- compute auto‑convolution and smooth max ----
        conv = _full_convolution_fft(p, N)
        M = conv.max()                     # true max (used for reporting)
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            # occasional progress report
            print(
                f"[len {n:5d} it {it:6d}] new best = {best_score:.6f}",
                flush=True,
            )

        # ----- soft‑max weighting (beta grows linearly) -----
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_len)
        # shift so that the largest entry becomes 0 → exponent ≤ 0
        shift = conv - M
        w_raw = np.exp(beta * shift)          # w_raw ≤ 1, max = 1
        w = w_raw / w_raw.sum()               # Σ w = 1

        # ----- correlation rev_i = Σ_k w_k·p_{k−i} -----
        rev = _weighted_rev_fft(p, w, N)

        # ----- gradient of the smooth objective -----
        # ∇ = 2·n·(rev + p)   (see derivation in the docstring)
        grad = 2.0 * n * (rev + p)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate that decays gently with iteration count
        alpha = alpha0 / (1.0 + it * 2e-4)

        p = p - alpha * m_hat / (np.sqrt(v_hat) + eps)

        # ----- jitter to escape plateaus (every few thousand iters) -----
        if it % 3000 == 0:
            p += np.random.randn(n) * 2e-4

        # ----- project back onto the simplex -----
        p = _simplex_normalise(p)

    return best_p, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The method combines multiscale Adam optimisation with soft‑max smoothing.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Keep a safety margin for the test harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Initialise from the previously discovered construction if it exists
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        # fallback: modest uniform start of length 500
        base = np.ones(500, dtype=float)

    n0 = base.shape[0]
    S0 = math.sqrt(2.0 * n0)
    base = base * (S0 / base.sum())          # enforce Σa = √(2n)

    # Convert to probability vector p (sum = 1) for the optimiser
    best_p = base / S0
    best_score = evaluate_sequence(base.tolist())
    best_n = n0
    print(f"[init] length={n0}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Multiscale schedule – lengths we will explore
    length_schedule = [300, 600, 900, 1200, 1500, 1800,
                       2100, 2600, 3000, 3500, 4000, 5000, 6000, 8000]
    length_schedule = [l for l in length_schedule if 300 <= l <= 8000]
    length_schedule = sorted(set(length_schedule))

    # Global best carrier
    global_best_p = best_p
    global_best_score = best_score
    global_best_n = best_n

    # ------------------------------------------------------------------
    for n in length_schedule:
        if time.time() >= deadline:
            break

        # ---- initialise probability vectors for this length ----
        # 1) Upsampled version of the current global best
        p_up = np.interp(
            np.linspace(0.0, 1.0, n),
            np.linspace(0.0, 1.0, global_best_p.shape[0]),
            global_best_p,
        )
        p_up = np.maximum(p_up, 0.0)
        p_up = _simplex_normalise(p_up)

        # 2) Random Dirichlet start
        p_rand = np.random.dirichlet(np.ones(n))

        # Determine a reasonable iteration budget for this size
        base_iters = 20000
        max_iters = max(500, int(base_iters * (500.0 / n)))

        print(f"\n[run] length={n}, max_iters≈{max_iters}", flush=True)

        # ----- first run: up‑sampled start -----
        p_opt, score = _optimize_adam_length(
            n=n,
            p_start=p_up,
            deadline=deadline,
            max_iters=max_iters,
        )
        a_opt = p_opt * math.sqrt(2.0 * n)
        cur_score = evaluate_sequence(a_opt.tolist())
        if cur_score < global_best_score:
            global_best_score = cur_score
            global_best_p = p_opt
            global_best_n = n
            print(
                f"[len {n:5d}] improved global best (up‑sample) = {global_best_score:.6f}",
                flush=True,
            )

        # ----- second run: random start -----
        p_opt, score = _optimize_adam_length(
            n=n,
            p_start=p_rand,
            deadline=deadline,
            max_iters=max_iters,
        )
        a_opt = p_opt * math.sqrt(2.0 * n)
        cur_score = evaluate_sequence(a_opt.tolist())
        if cur_score < global_best_score:
            global_best_score = cur_score
            global_best_p = p_opt
            global_best_n = n
            print(
                f"[len {n:5d}] improved global best (random) = {global_best_score:.6f}",
                flush=True,
            )

    # ------------------------------------------------------------------
    # Final fine‑tuning on the best length found so far (more iterations, sharper β)
    if time.time() < deadline:
        print(
            f"\n[final] fine‑tuning on length={global_best_n}",
            flush=True,
        )
        p_fine, fine_score = _optimize_adam_length(
            n=global_best_n,
            p_start=global_best_p,
            deadline=deadline,
            max_iters=50000,
            beta_start=0.2,
            beta_end=800.0,
            beta_len=20000,
        )
        a_fine = p_fine * math.sqrt(2.0 * global_best_n)
        fine_score = evaluate_sequence(a_fine.tolist())
        if fine_score < global_best_score:
            global_best_score = fine_score
            global_best_p = p_fine
            print(
                f"[final] new global best after fine‑tune = {global_best_score:.6f}",
                flush=True,
            )

    # ------------------------------------------------------------------
    # Assemble final sequence
    final_a = global_best_p * math.sqrt(2.0 * global_best_n)
    print(
        f"[done] best score = {global_best_score:.6f}, length = {global_best_n}",
        flush=True,
    )
    return final_a.tolist()
