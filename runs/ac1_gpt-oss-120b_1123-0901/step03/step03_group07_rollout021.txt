def proj_simplex(v):
    # using method from Condat 2016
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low–value step‑function.

The method is a simple sub‑gradient descent on the exact
∞‑norm of the self‑convolution.  For a current vector a (sum(a)=1,
a_i≥0) we locate the index k where (a*a)[k] is maximal and
use the analytical gradient of that convolution entry:
∂(a*a)[k]/∂a_i = 2·a_{k‑i} (zero when the index is out of range).
The update is performed on the log‑softmax parametrisation
(u_i = log a_i) with an Adam optimiser; after each step
the probabilities are obtained by a soft‑max, guaranteeing the
simplex constraints automatically.

* a few different random initialisations are tried,
* a “shake” (tiny random perturbation of the logits) is applied
  when progress stalls,
* the continuation parameter λ is kept large from the start, so
  the gradient is essentially the true sub‑gradient of the max‑convolution,
* the inner loop runs for many more iterations than the reference
  implementation (≈30 000), and a limited number of restarts are
  performed, allowing the optimiser to converge deeper within the
  1000 s budget.

The best sequence encountered over all restarts is returned as a plain
Python list, ready to be fed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _max_gradient(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Gradient of the convolution entry with index ``max_idx``
    w.r.t. the original vector a (exact sub‑gradient of the max).
    """
    n = a.shape[0]
    grad = np.zeros_like(a)

    # the only non‑zero part lies where i + j = max_idx
    start_i = max(0, max_idx - (n - 1))
    end_i = min(n - 1, max_idx)          # inclusive
    i = np.arange(start_i, end_i + 1, dtype=np.int64)
    j = max_idx - i
    grad[i] = 2.0 * a[j]
    return grad


def _objective(a: np.ndarray) -> float:
    """Exact evaluation used by ``evaluate_sequence`` (scale‑invariant)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    # sum_a is guaranteed > 0 by construction
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list:
    """
    Search for a sequence of non‑negative heights with a low evaluation score.
    Returns the best list found before the time budget expires.
    """
    np.random.seed(seed)

    deadline = time.time() + budget_s - 0.3   # safety margin

    # ------------------------------------------------------------------
    # Length of the step‑function.
    # 2000 gives a good trade‑off between expressive power and speed.
    n = 2000

    # Pre‑computed FFT length (next power of two ≥ 2n‑1)
    conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.03          # moderate learning rate

    best_val = float("inf")
    best_seq = None

    restart = 0
    while time.time() < deadline:
        restart += 1

        # -------------------- random initialisation --------------------
        mode = restart % 5
        if mode == 0:
            a0 = np.random.dirichlet(np.ones(n))
        elif mode == 1:
            # geometric decay
            r = np.random.uniform(0.2, 0.95)
            a0 = r ** np.arange(n, dtype=np.float64)
            a0 /= a0.sum()
        elif mode == 2:
            # half‑sine shape
            i = np.arange(1, n + 1, dtype=np.float64)
            a0 = np.sin(np.pi * i / (n + 1))
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        elif mode == 3 and 'height_sequence_1' in globals():
            # reuse the previous best construction if available (crop/extend)
            old = np.array(height_sequence_1, dtype=np.float64)
            if old.size < n:
                # pad with zeros
                a0 = np.concatenate([old, np.zeros(n - old.size)])
            else:
                # down‑sample by simple averaging
                step = old.size // n
                a0 = old[: step * n].reshape(n, step).mean(axis=1)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            # a few random spikes – encourages sparsity
            a0 = np.zeros(n)
            spikes = np.random.choice(n, size=n // 50, replace=False)
            vals = np.random.rand(spikes.size)
            a0[spikes] = vals
            a0 += 1e-6
            a0 /= a0.sum()

        # Log‑space parametrisation (soft‑max)
        u = np.log(a0 + 1e-30)

        # Adam state
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        # Continue flag for inner loop
        it = 0
        last_improve = time.time()

        # ------------------------------------------------------------------
        # Run a relatively long inner optimisation (≈30 k iterations)
        while time.time() < deadline:
            it += 1

            # ---- probabilities from logits (numerically stable) ----
            u_shift = u - np.max(u)
            a = np.exp(u_shift)
            a /= a.sum()                     # guarantees a ∈ simplex

            # ---- compute convolution and locate the maximal entry ----
            conv = _full_convolution(a, size=conv_fft_size)
            max_idx = int(conv.argmax())

            # ---- exact gradient of the active convolution entry ----
            grad_a = _max_gradient(a, max_idx)

            # ---- convert to log‑space (soft‑max Jacobian) ----
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            # ---- Adam update ----
            m = beta1 * m + (1.0 - beta1) * grad_u
            v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

            it_float = float(it)
            m_hat = m / (1.0 - beta1 ** it_float)
            v_hat = v / (1.0 - beta2 ** it_float)

            # learning‑rate decay (slow)
            lr = base_lr * (0.9997 ** it)

            u -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # ---- exact evaluation every 25 iterations ----
            if it % 25 == 0:
                val = _objective(a)
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[Restart {restart:03d}] iter {it:06d}  new best = {val:.6f}")

            # ---- shake if we are stuck for >1.2 s ----
            if time.time() - last_improve > 1.2:
                # add a tiny random perturbation in log‑space
                u += (np.random.rand(n) - 0.5) * 6e-4
                m.fill(0.0)
                v.fill(0.0)
                last_improve = time.time()

            # ---- stop after a reasonable number of steps ----
            if it >= 30000:
                break

        print(f"[Restart {restart:03d}] finished it={it}, best so far = {best_val:.6f}")

    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (objective = 2)
        best_seq = np.full(n, 1.0 / n)

    print(f"Final best objective = {best_val:.6f} (length {n})")
    return best_seq.tolist()
