def propose_candidate(...):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2

    best_val = np.inf
    best_seq = None

    # decide candidate lengths: maybe start n=3000, then if time left, increase to n=6000
    n_list = [3000, 5000, 8000]  # but need to adapt to time: start with smaller
    for n in n_list:
        # check if time left enough for at least one restart.
        if time.time() + 10 > deadline: break
        # precompute FFT sizes
        conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_size = 1 << int(np.ceil(np.log2(3 * n)))

        restart = 0
        while time.time() < deadline:
            restart += 1

            # initialization
            mode = restart % 3
            if mode == 0:
                a = np.random.dirichlet(np.ones(n))
            elif mode == 1:
                r = np.random.uniform(0.2,0.95)
                a = np.power(r, np.arange(n, dtype=np.float64))
                a /= a.sum()
            else:
                i = np.arange(1, n+1, dtype=np.float64)
                a = np.sin(np.pi*i/(n+1))
                a = np.maximum(a, 0)
                a /= a.sum()

            # transform to logit space sorted decreasing
            u = np.log(a + 1e-30)
            # Sort u decreasing to enforce monotonic shape
            u = np.sort(u)[::-1]

            # Adam states
            m = np.zeros_like(u)
            v = np.zeros_like(u)

            lam = 1.0
            lam_factor = 1.015
            base_lr = 0.04
            beta1 = 0.9
            beta2 = 0.999
            eps = 1e-8

            it = 0
            last_improve = time.time()
            while time.time() < deadline:
                it += 1

                u_shift = u - np.max(u)

assistantfinal**Solution Overview**

The objective is scale‑invariant, therefore we keep the vector on the simplex
(`∑aᵢ = 1`).  
For a vector `a` the evaluation value is  

```
F(a) = 2·n·max_k (a * a)[k] / (∑aᵢ)² = 2·n·max_k (a * a)[k]      (∑aᵢ = 1)
```

The non‑smooth `max` is replaced by a smooth‑max (soft‑max) with a
temperature `λ`.  The surrogate

```
φ_λ(a) = Σ_k w_k·(a*a)[k] ,      w_k = exp(λ·c_k) / Σ_j exp(λ·c_j)
```

is differentiable; its gradient is `∇φ_λ = 2·(w ★ a)` where `★` denotes
cross‑correlation.  Using Adam on the logits (`u` with `a = softmax(u)`)
gives a fast first‑order method.

**Key improvements over the previous code**

* **Monotone shape** – the optimal step function is believed to be
  decreasing.  After each Adam step the logits are sorted in decreasing
  order, which forces the probabilities to stay monotone.  This reduces the
  peak of the self‑convolution considerably.
* **Length sweep** – larger vectors have more flexibility.  
  The algorithm tries several lengths (`n = 3000, 5000, 8000`) and keeps the
  best result.
* **More aggressive continuation** – `λ` grows slowly (`λ ← λ·1.015`) and
  the learning‑rate is decayed gently; this allows the optimisation to
  follow the true `max` more closely.
* **Stagnation ”shake”** – if no improvement happens for 1.5 s a tiny random
  perturbation is added to the logits and the Adam moments are reset.
* **Final refinement** – after the multi‑length search a longer Adam run
  (up to the remaining time budget) is performed on the best candidate.

All heavy work (FFT‑based convolutions) is `O(n log n)` and the code stays
well inside the 2‑CPU limit.

```python
"""
Propose a step‑function (non‑negative sequence) that minimizes
evaluate_sequence().  The algorithm uses an Adam optimiser on a smooth‑max
surrogate of max(a*a) and enforces a monotone decreasing shape by sorting
the logits at each iteration.  Several vector lengths are tried and the
best result is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max weights w_k = exp(lam·conv_k) / Σ_j exp(lam·conv_j)."""
    m = conv.max()
    # safe exponentiation – shift by the maximum
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:                # should never happen, but guard against it
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int = None) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate w.r.t. the probabilities ``a``.
    The formula is  ∇φ = 2·(w ★ a) .
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    # slice that corresponds to indices 0 … n‑1 of the original vector
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


def _objective(a: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence (scale‑invariant)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) achieves an
    evaluation score lower than the previous best (≈ 1.52).

    The routine:
        * tries several vector lengths (3000, 5000, 8000);
        * for each length performs many random restarts of an Adam optimiser
          on a smooth‑max surrogate;
        * after every Adam update the logits are sorted decreasingly
          → the probabilities stay monotone;
        * λ (smoothness) is increased gradually (continuation);
        * occasional tiny random “shakes’’ avoid stagnation;
        * the best sequence seen so far is kept and finally refined
          with a longer run if time permits.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin for the caller

    # lengths to explore – larger n gives more expressive power but costs more time
    lengths = [3000, 5000, 8000]

    best_val = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Main multi‑length search
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # pre‑compute FFT sizes that are reused many times
        conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_size = 1 << int(np.ceil(np.log2(3 * n)))

        restart = 0
        while time.time() < deadline:
            restart += 1

            # --------------------------------------------------------------
            # 1) random initialisation (a mixture of families)
            # --------------------------------------------------------------
            mode = restart % 3
            if mode == 0:
                a0 = np.random.dirichlet(np.ones(n))
            elif mode == 1:
                r = np.random.uniform(0.2, 0.95)
                a0 = np.power(r, np.arange(n, dtype=np.float64))
                a0 /= a0.sum()
            else:
                i = np.arange(1, n + 1, dtype=np.float64)
                a0 = np.sin(np.pi * i / (n + 1))
                a0 = np.maximum(a0, 0.0)
                a0 /= a0.sum()

            # logits (log‑softmax parameters) – enforce monotonicity immediately
            u = np.log(a0 + 1e-30)
            u = np.sort(u)[::-1]                     # decreasing order

            # Adam state
            m = np.zeros_like(u)
            v = np.zeros_like(u)

            # continuation parameters
            lam = 1.0
            lam_factor = 1.015

            # Adam hyper‑parameters
            beta1 = 0.9
            beta2 = 0.999
            eps = 1e-8
            base_lr = 0.04

            it = 0
            last_improve = time.time()

            # --------------------------------------------------------------
            # 2) inner optimisation loop
            # --------------------------------------------------------------
            while time.time() < deadline:
                it += 1

                # ----- probabilities from logits (soft‑max) -----
                u_shift = u - np.max(u)               # numeric stability
                a = np.exp(u_shift)
                a /= a.sum()                           # stays on the simplex

                # ----- convolution & smooth‑max weights -----
                conv = _full_convolution(a, size=conv_fft_size)
                w = _softmax_weights(conv, lam)

                # ----- gradient w.r.t. a -----
                grad_a = _softmax_gradient(a, w, size_corr=corr_fft_size)

                # ----- convert to gradient w.r.t. logits -----
                dot = np.dot(a, grad_a)
                grad_u = a * (grad_a - dot)

                # ----- Adam update -----
                m = beta1 * m + (1.0 - beta1) * grad_u
                v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)

                lr = base_lr * (0.9995 ** it)

                u = u - lr * m_hat / (np.sqrt(v_hat) + eps)

                # ----- enforce monotone decreasing logits -----
                u = np.sort(u)[::-1]

                # ----- increase smoothness -----
                lam *= lam_factor

                # ----- occasional exact evaluation -----
                if it % 20 == 0:
                    val = _objective(a)
                    if val < best_val:
                        best_val = val
                        best_seq = a.copy()
                        last_improve = time.time()
                        print(f"[n={n} Restart {restart:03d}] it {it:05d}  new best = {val:.6f}")

                # ----- shake if we have been stagnant -----
                if time.time() - last_improve > 1.5:
                    u += (np.random.rand(n) - 0.5) * 1e-3
                    m.fill(0.0)
                    v.fill(0.0)
                    last_improve = time.time()

                # ----- stop inner loop after a reasonable number of steps -----
                if it >= 12000:
                    break

            print(f"[n={n} Restart {restart:03d}] finished it={it}, best overall = {best_val:.6f}")

            # small safety check – avoid too many restarts when time is short
            if time.time() + 5.0 > deadline:
                break

    # ------------------------------------------------------------------
    # 3) final long‑run refinement on the best candidate (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = len(best_seq)
        conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_size = 1 << int(np.ceil(np.log2(3 * n)))

        u = np.log(best_seq + 1e-30)
        u = np.sort(u)[::-1]

        m = np.zeros_like(u)
        v = np.zeros_like(u)

        lam = 2.0                     # start a little smoother
        lam_factor = 1.010

        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        base_lr = 0.03
        it = 0
        last_improve = time.time()

        while time.time() < deadline:
            it += 1
            u_shift = u - np.max(u)
            a = np.exp(u_shift)
            a /= a.sum()

            conv = _full_convolution(a, size=conv_fft_size)
            w = _softmax_weights(conv, lam)
            grad_a = _softmax_gradient(a, w, size_corr=corr_fft_size)

            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            m = beta1 * m + (1.0 - beta1) * grad_u
            v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = base_lr * (0.9995 ** it)
            u = u - lr * m_hat / (np.sqrt(v_hat) + eps)

            u = np.sort(u)[::-1]               # keep monotone shape

            lam *= lam_factor

            if it % 20 == 0:
                val = _objective(a)
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[Refine] it {it:05d}  new best = {val:.6f}")

            if time.time() - last_improve > 2.0:
                u += (np.random.rand(n) - 0.5) * 5e-4
                m.fill(0.0)
                v.fill(0.0)
                last_improve = time.time()

            if it >= 25000:                    # generous upper bound
                break

    # ------------------------------------------------------------------
    # 4) fallback (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, 1.0 / n)

    print(f"Final best objective = {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
```
