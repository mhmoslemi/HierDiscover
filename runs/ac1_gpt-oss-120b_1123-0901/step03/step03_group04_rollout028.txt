"""
Propose a non‑negative step function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm Overview
------------------
1. **Exploratory families** – For a handful of problem sizes we scan
   simple parametrised families (exponential decay and power‑law decay)
   that admit a closed form convolution.  Each candidate is normalised
   to the required ℓ₁‑norm ( √(2 n) ) and its peak convolution is evaluated.
   The best candidate so far is kept.

2. **Global refinement** – Starting from the current best vector we run a
   projected Adam optimiser that works on the exact sub‑gradient of the
   peak convolution.  The optimiser respects the non‑negativity and
   normalisation constraints by projection after every step and
   injects a tiny Gaussian jitter occasionally to escape plateaux.
   Several random restarts are performed while time permits.

3. **Targeted local search** – Using the result of Adam we perform a
   greedy local optimisation.  In each iteration the shift attaining the
   maximum convolution is identified; the pair of indices contributing
   the largest product to that shift is located and a small amount of
   mass is moved from that pair to two random positions.  The move is
   kept if it lowers the peak.  This step is cheap yet effective at
   shaving off the last decimals.

All steps respect the 1000‑second wall‑clock budget.  The routine prints
progress messages and finally returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """
    Compute the maximal value of the full convolution a*a and the
    shift index where it is attained.  Uses an FFT of length equal to
    the next power of two for speed.
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    max_val = float(conv.max())
    argmax = int(conv.argmax())
    return max_val, argmax, conv


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, active_idxs: np.ndarray):
    """
    Sub‑gradient of max_{k} (a*a)[k] at the current point.
    The gradient is the average of the linear gradients corresponding
    to the active shifts (those attaining the maximum).
    """
    n = a.shape[0]
    idx = np.arange(n)
    grad = np.zeros_like(a)

    for k in active_idxs:
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad[mask] += a[j[mask]]

    if active_idxs.size > 0:
        grad /= float(active_idxs.size)          # average -> a valid sub‑gradient
    return grad


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   target_sum: float,
                   deadline: float,
                   max_iters: int = 10_000_000):
    """
    Adam optimiser on the exact sub‑gradient of the peak convolution.
    Returns the best vector encountered and its peak value.
    """
    a = init_a.copy()
    # initial normalisation
    s = a.sum()
    if s > 0:
        a *= target_sum / s
    else:
        a[:] = target_sum / n

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.06
    iteration = 0
    jitter_interval = 5000

    # keep the best point seen so far
    best_vec = a.copy()
    best_val, _, _ = _conv_max_fft(a)

    while time.time() < deadline and iteration < max_iters:
        # ----------------------------------------------------------
        # 1. convolution and active set
        M, _, conv = _conv_max_fft(a)

        # treat all shifts whose value is within tolerance of the maximum as active
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # ----------------------------------------------------------
        # 2. sub‑gradient of the max term (no scaling – we minimise M directly)
        grad = _subgradient(a, active)

        # ----------------------------------------------------------
        # 3. Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # simple decay schedule for the learning rate
        lr = base_lr / (1.0 + 1e-4 * iteration)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----------------------------------------------------------
        # 4. projection onto the feasible set
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a *= target_sum / s
        else:
            a[:] = target_sum / n

        # ----------------------------------------------------------
        # 5. occasional jitter to escape flat plateaux
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= target_sum / a.sum()

        # ----------------------------------------------------------
        # 6. keep the best solution seen so far
        if M < best_val:
            best_val = M
            best_vec = a.copy()
            print(f"[adam] n={n:5d}, iter={iteration:7d}, new best M={best_val:.6f}")

    return best_vec, best_val


# ----------------------------------------------------------------------
def _targeted_local_search(a: np.ndarray,
                           n: int,
                           target_sum: float,
                           steps: int,
                           deadline: float):
    """
    Greedy local search that repeatedly reduces the contribution of the
    shift attaining the current maximum.  In each step we:
        * locate the shift k* with maximal convolution,
        * locate the pair (i, j) = (i, k*−i) giving the largest product,
        * move a small fraction of mass from i and j to two random
          positions (keeping the total sum constant).
    The move is kept only if it improves the maximal convolution.
    """
    best = a.copy()
    best_val, _, conv = _conv_max_fft(best)

    for step in range(steps):
        if time.time() >= deadline:
            break

        # locate the worst shift
        k_star = int(conv.argmax())

        # indices contributing to that shift
        i_start = max(0, k_star - (n - 1))
        i_end = min(n - 1, k_star)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k_star - i_vals

        # product of each pair
        prod = best[i_vals] * best[j_vals]
        max_pair_idx = int(prod.argmax())
        i = i_vals[max_pair_idx]
        j = j_vals[max_pair_idx]

        # amount to move – a modest fraction of the smaller partner
        delta = min(best[i], best[j]) * 0.02
        if delta <= 0:
            continue

        # choose two distinct random destinations (different from i and j)
        t1, t2 = np.random.randint(0, n, size=2)
        while t1 == i or t1 == j:
            t1 = np.random.randint(0, n)
        while t2 == i or t2 == j or t2 == t1:
            t2 = np.random.randint(0, n)

        cand = best.copy()
        cand[i] -= delta
        cand[j] -= delta
        cand[t1] += delta / 2.0
        cand[t2] += delta / 2.0

        # keep feasibility
        if cand[i] < 0 or cand[j] < 0:
            continue
        cand = np.maximum(cand, 0.0)
        cand *= target_sum / cand.sum()

        cand_val, _, _ = _conv_max_fft(cand)
        if cand_val < best_val:
            best = cand
            best_val = cand_val
            conv = _conv_max_fft(best)[2]   # remember the full convolution for next loop
            # occasional progress report
            if step % 2000 == 0:
                print(f"[local] step={step:6d}, improved M={best_val:.6f}")

        # small random jitter every few thousand steps
        if step % 5000 == 0:
            jitter = np.random.randn(n) * 0.0005 * target_sum / n
            cand = np.maximum(best + jitter, 0.0)
            cand *= target_sum / cand.sum()
            cand_val, _, _ = _conv_max_fft(cand)
            if cand_val < best_val:
                best = cand
                best_val = cand_val
                conv = _conv_max_fft(best)[2]

    return best, best_val

    for n in length_candidates:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # -- exponential family -------------------------------------------------
        for lam in np.logspace(-5, -1, 25):
            if time.time() >= deadline:
                break
            a = np.exp(-lam * np.arange(n, dtype=float))
            a *= target_sum / a.sum()
            _maybe_update(a)

        # -- power‑law family ---------------------------------------------------
        for p in np.linspace(0.1, 2.5, 20):
            if time.time() >= deadline:
                break
            idx = np.arange(1, n + 1, dtype=float)
            a = idx ** (-p)
            a *= target_sum / a.sum()
            _maybe_update(a)

        # -- a few random Dirichlet starts ------------------------------------
        for _ in range(4):
            if time.time() >= deadline:
                break
            rand = np.random.exponential(scale=1.0, size=n)
            a = rand * (target_sum / rand.sum())
            _maybe_update(a)

    # --------------------------------------------------------------
    # 2. bring in the previously discovered construction if available
    try:
        prev = np.array(height_sequence_1, dtype=float)  # type: ignore
        if prev.size > 0:
            n = prev.shape[0]
            target = math.sqrt(2.0 * n)
            prev = prev * (target / prev.sum())
            _maybe_update(prev)
    except Exception:
        pass

    # If nothing was generated (should not happen) fall back to uniform
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, 1.0, dtype=float)
        best_seq *= target_sum / best_seq.sum()
        best_val = _conv_max_fft(best_seq)[0]
        print("[fallback] using uniform start")

    # --------------------------------------------------------------
    # 3. refined Adam optimisation (primary refinement)
    time_left = deadline - time.time()
    adam_time = min(time_left * 0.5, 300.0)
    adam_deadline = time.time() + adam_time

    print(f"[adam] starting refinement for length {len(best_seq)} for up to {adam_time:.1f}s")
    opt_seq, opt_val = _adam_optimize(best_seq, len(best_seq),
                                      math.sqrt(2.0 * len(best_seq)),
                                      adam_deadline,
                                      max_iters=15_000_000)
    _maybe_update(opt_seq)

    # occasional extra random‑restart Adam runs while time permits
    while time.time() < adam_deadline - 5.0:
        n = len(best_seq)
        target = math.sqrt(2.0 * n)
        rand = np.random.exponential(scale=1.0, size=n)
        init = rand * (target / rand.sum())
        seq, val = _adam_optimize(init, n, target, adam_deadline, max_iters=5_000_000)
        _maybe_update(seq)

    # --------------------------------------------------------------
    # 4. targeted local search (final polishing)
    if time.time() < deadline:
        steps = 20_000
        print(f"[local] starting targeted local search (up to {steps} steps)")
        ls_seq, ls_val = _targeted_local_search(best_seq, len(best_seq),
                                                math.sqrt(2.0 * len(best_seq)),
                                                steps=steps,
                                                deadline=deadline)
        _maybe_update(ls_seq)

    # --------------------------------------------------------------
    # final official evaluation
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[done] best reported M = {best_val:.6f}, official score = {final_score:.6f}, length = {len(best_seq)}")
    return best_seq.tolist()
