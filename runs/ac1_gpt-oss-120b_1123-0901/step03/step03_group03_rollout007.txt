"""
Propose a non‑negative step‑function minimizing ``evaluate_sequence``.

Algorithm
---------
*   Retrieve the best known construction (``height_sequence_1``) if
    available, otherwise start from a uniform vector.
*   Work with the *probability* representation  
    `p_i = a_i / Σ a` (so Σ p = 1, p ≥ 0).  After optimisation the final
    amplitudes are `a_i = p_i * √(2 n)`.
*   Optimise `p` on the simplex by **exponentiated‑gradient Adam**.
    -   The exact objective is `max (a∗a)`.  For the probability vector
        it is `max (p∗p)` because the normalisation factor `√(2 n)` is
        constant.
    -   A smooth approximation of the max is the log‑sum‑exp  
        `Lτ(p)=τ·log Σ exp((p∗p)/τ)`.  Its gradient w.r.t. `p` is
        `2·rev`, where `rev_i = Σ_k w_k p_{k‑i}` and `w` are the
        soft‑max weights of the convolution values.
    -   Using the Jacobian of the softmax (the “mirror map” of the
        simplex) we obtain the gradient w.r.t. the unconstrained
        logarithmic parameters `u` (`p = softmax(u)`):  

        `grad_u = p * (grad_p – (grad_p·p))`.

    -   Adam updates are performed on `u`.  The parameters are projected
        back to the simplex automatically by the soft‑max.
    -   Temperature `τ` starts as a fraction of the current maximum
        convolution value and decays exponentially (`τ = τ₀·e⁻ᵗ/τ_decay`).
    -   Occasionally a tiny Gaussian jitter is added to `u` to escape
        plateaus.
*   The optimiser is run for several lengths (original length and a few
    enlargements, up to 8000 entries).  For each length the current best
    sequence (`p`) is interpolated, normalised and refined by the same
    optimiser.
*   Throughout the search the exact objective `max(a∗a)` (which equals the
    value returned by ``evaluate_sequence`` after normalisation) is
    monitored; the globally best sequence is kept.
*   The routine respects a 1000 s wall‑clock limit and returns the best
    sequence found as a plain Python list.
"""

# Helper functions (no closures)

def _softmax_weights(conv: np.ndarray, tau: float):
    """
    Stable soft‑max of the convolution values.
    Returns the weight vector w (summing to 1) and the smoothed maximum
    Lτ = τ·log ∑exp(conv/τ).
    """
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """
    Compute rev_i = Σ_k w_k a_{k‑i} (indices outside 0…len(a)-1 ignored)
    via FFT.  ``a`` and ``w`` are real vectors, ``Nfft`` >= 2*len(a)-1.
    """
    n = a.shape[0]
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + n - 1]
    idx_start = 2 * n - 2
    rev = conv_rev[idx_start - np.arange(n)]
    return rev

    lr0 = 0.12                # base learning rate for the log‑parameters
    tau_factor = 0.20         # initial τ = τ_factor·M
    decay_rate = 70_000.0     # slower decay → smoother landscape longer
    jitter_interval = 10_000
    jitter_scale = 0.02       # relative jitter on u (log‑probabilities)

    # FFT size (next power‑of‑two ≥ 2·n‑1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    u = init_u.copy()
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    iteration = 0
    while time.time() < deadline and iteration < max_iters:
        # ---------- probability vector ----------
        # use numerically stable softmax
        u_shift = u - np.max(u)
        p = np.exp(u_shift)
        p_sum = p.sum()
        if p_sum == 0.0:
            p = np.full_like(u, 1.0 / n)
        else:
            p = p / p_sum

        # ---------- convolution ----------
        P_fft = np.fft.rfft(p, n=Nfft)
        conv = np.fft.irfft(P_fft * P_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ---------- smooth max ----------
        tau = max(1e-15, tau_factor * M * math.exp(-iteration / decay_rate))
        w, _ = _softmax_weights(conv, tau)

        # ---------- gradient wrt p ----------
        # dL/da = 2 * rev   (rev_i = Σ_k w_k p_{k‑i})
        rev = 2.0 * _weighted_rev(p, w, Nfft)   # shape (n,)
        grad_p = rev

        # ---------- gradient wrt log‑probabilities ----------
        dot = np.dot(grad_p, p)          # scalar = grad_p·p
        grad_u = p * (grad_p - dot)      # mirror‑gradient on the simplex

        # ---------- Adam update ----------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u * grad_u)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        lr = lr0 / (1.0 + 5e-5 * iteration)
        u = u - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---------- occasional jitter ----------
        if iteration % jitter_interval == 0:
            u += np.random.randn(n) * jitter_scale

        # ---------- track best ----------
        if M < best_M:
            best_M = M
            best_p = p.copy()
            print(f"[len {n:5d} iter {iteration:8d}] new best M = {best_M:.6f}")

    return best_p, best_M


def _interpolate_to_length(vec: np.ndarray, new_len: int) -> np.ndarray:
    """
    Linear interpolation of a vector onto a new length (preserving shape).
    """
    old_len = vec.shape[0]
    if new_len == old_len:
        return vec.copy()
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    interpolated = np.interp(new_x, old_x, vec)
    # ensure strict positivity for the log‑parameterisation
    interpolated = np.maximum(interpolated, 1e-12)
    interpolated /= interpolated.sum()
    return interpolated

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.

    Parameters
    ----------
    seed: int
        Random seed for reproducibility.
    budget_s: float
        Wall‑clock budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best sequence found (plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the evaluation harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load a previously discovered construction, fallback to uniform
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
        if base.ndim != 1:
            raise ValueError("height_sequence_1 must be 1‑d")
    except Exception:
        # a modest default length – 2100 was used in earlier runs
        base = np.ones(2100, dtype=float)

    n0 = base.shape[0]
    # Normalise to the required L1‑norm √(2 n0)
    target_sum0 = math.sqrt(2.0 * n0)
    base = base * (target_sum0 / base.sum())
    # Probability version (sums to 1)
    base_p = np.maximum(base, 1e-12)
    base_p = base_p / base_p.sum()

    # initialise global best
    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n0}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore – original and several enlargements (capped at 8000)
    length_factors = [1.0, 1.25, 1.5, 2.0, 2.5, 3.0]
    lengths = sorted({int(round(n0 * f)) for f in length_factors})
    lengths = [l for l in lengths if l >= 300]
    lengths = [min(l, 8000) for l in lengths]
    lengths = sorted(set(lengths))

    # ------------------------------------------------------------------
    # Main optimisation loop over the selected lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        # ---- initialise log‑probabilities for length n ----
        if n == n0:
            init_p = base_p.copy()
        else:
            init_p = _interpolate_to_length(base_p, n)
        # log‑parameterisation (stable: subtract max later)
        init_u = np.log(np.maximum(init_p, 1e-30))

        # ---- allocate iteration budget (more for smaller n) ----
        # Rough scaling: 5 million iters for n≈2100, inversely with n.
        max_iters = int(5_000_000 * 2100 / n)
        max_iters = max(200_000, max_iters)   # at least a few hundred‑k
        # but respect remaining time (very rough estimate: 5 µs per iter)
        remaining = deadline - time.time()
        est_iters = int(remaining * 2e5)  # assume ~5 µs/iter → 2e5 iters per sec
        max_iters = min(max_iters, max(1_000, est_iters))

        print(f"[run] length={n}, max_iters≈{max_iters}")

        # ---- optimise this length ----
        best_p, best_M = _optimize_length_eg(
            n=n,
            init_u=init_u,
            best_p=base_p if n == n0 else init_p,
            best_M=best_score,
            deadline=deadline,
            max_iters=max_iters,
        )

        # convert best probability vector back to amplitudes
        target_sum = math.sqrt(2.0 * n)
        candidate_seq = best_p * target_sum

        # evaluate exact objective (should equal best_M, but we call for safety)
        cand_score = evaluate_sequence(candidate_seq.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = candidate_seq.copy()
            print(f"[global] new overall best = {best_score:.6f} (len={n})")

    # ------------------------------------------------------------------
    # Fallback in case nothing was found (should never happen)
    if best_seq is None or best_seq.size == 0:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
