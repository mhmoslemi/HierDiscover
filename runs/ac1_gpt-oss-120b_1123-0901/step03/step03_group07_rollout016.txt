def _refine_lp(f0, deadline):
    import numpy as np, time
    from scipy.optimize import linprog
    from scipy.linalg import toeplitz

    f = f0.astype(np.float64).copy()
    # ensure non-negative
    f = np.maximum(f, 0.0)
    # normalize to sum 1 (any positive sum works)
    total = f.sum()
    if total == 0:
        f = np.full_like(f, 1.0 / f.shape[0])
        total = 1.0
    f = f / total
    best_val = evaluate_sequence(f.tolist())
    best_f = f.copy()
    n = len(f)
    # predefine variables for LP
    eps = 1e-12
    while time.time() < deadline:
        # compute current max conv
        conv_f = np.convolve(f, f)
        M = conv_f.max()
        # build convolution matrix A (2n-1, n)
        col = np.concatenate([f, np.zeros(n-1, dtype=f.dtype)])
        row = np.concatenate([[f[0]], np.zeros(n-1, dtype=f.dtype)])
        A = toeplitz(col, row)  # dense matrix
        # Solve LP: maximize sum(g) subject to A g <= M, g >= 0
        c = -np.ones(n, dtype=float)  # maximize sum => minimize -sum
        b = np.full(A.shape[0], M, dtype=float)
        bounds = [(0, None)] * n
        # Using high-performance method 'highs'
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
        if not res.success:
            # fallback: break
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= eps:
            # cannot improve
            break
        # scale to keep sum same as f
        g = g0 * (f.sum() / sum_g0)
        # try direct replacement if improves
        val_g = evaluate_sequence(g.tolist())
        if val_g < best_val:
            f = g
            best_val = val_g
            best_f = f.copy()
            print(f"Direct replace improvement val={val_g:.6f}")
            continue
        # otherwise try convex combination with step size t
        t = 0.5
        improved = False
        while t > 1e-4:
            f_new = (1 - t) * f + t * g
            val_new = evaluate_sequence(f_new.tolist())
            if val_new < best_val:
                f = f_new
                best_val = val_new
                best_f = f.copy()
                improved = True
                print(f"Blend t={t:.4f} improved val={val_new:.6f}")
                break
            t *= 0.5
        if not improved:
            # no improvement, break loop
            break
    return best_f, best_val

import time, numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

def propose_candidate(seed=0, budget_s=1000):
    """
    Search for a non‑negative step function that minimises
        2·n·max_k (a*a)[k] / (∑ a_i)²
    The method first performs a few random (or down‑sampled) start‑points,
    then iteratively improves each start point by solving a linear program
    that maximises the total mass under the current maximal autocorrelation
    bound.  The solution is scaled to the original total mass and blended
    with the current sequence.  After the LP‑phase a short Adam‑refinement
    (smooth‑max surrogate) is applied.  The best sequence found before the
    time budget expires is returned as a plain Python list.
    """
    import warnings
    warnings.filterwarnings("ignore", category=RuntimeWarning)
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ----- helpers -------------------------------------------------
    def _evaluate(a):
        # wrapper that silently handles the built‑in checks
        return evaluate_sequence(a.tolist())

    def _max_conv(a):
        return np.convolve(a, a).max()

    def _build_conv_matrix(f):
        n = f.shape[0]
        col = np.concatenate([f, np.zeros(n-1, dtype=f.dtype)])
        row = np.concatenate([[f[0]], np.zeros(n-1, dtype=f.dtype)])
        return toeplitz(col, row)    # (2n‑1) × n

    # ----- initial candidates --------------------------------------
    candidates = []

    # try to reuse the sequence that the platform already generated
    try:
        global height_sequence_1
        base = np.asarray(height_sequence_1, dtype=float)
        # down‑sample to a moderate size (≈2000) – linear interpolation in index space
        target_n = 2000
        if base.size > target_n:
            idx = np.linspace(0, base.size-1, target_n, dtype=int)
            base = base[idx]
        elif base.size < target_n:
            target_n = base.size
        base = np.maximum(base, 0.0)
        if base.sum() == 0:
            base = np.full_like(base, 1.0/base.size)
        base = base / base.sum()
        candidates.append(base)
    except Exception:
        pass

    # a handful of random Dirichlet starts (moderate size)
    target_n = 2000
    for _ in range(5):
        a = np.random.dirichlet(np.ones(target_n))
        candidates.append(a)

    # ----- main optimisation loop ---------------------------------
    best_val = float('inf')
    best_seq = None

    for start_id, a0 in enumerate(candidates):
        if time.time() >= deadline:
            break
        print(f"[Start {start_id+1}/{len(candidates)}] length={a0.size}")

        # ------------------------------------------------------------------
        # Phase 1 – LP based monotone improvement
        # ------------------------------------------------------------------
        f = a0.copy()
        n = f.size
        # keep the total mass fixed (normalise to 1)
        f = f / f.sum()
        it_lp = 0
        last_val = _evaluate(f)
        best_val = min(best_val, last_val)
        best_seq = f.copy() if last_val == best_val else best_seq
        no_improve_lp = 0

        while time.time() < deadline:
            it_lp += 1
            M = _max_conv(f)

            # build Toeplitz matrix for the constraints A·g ≤ M
            A = _build_conv_matrix(f)
            c = -np.ones(n)               # maximise Σ g_i
            b = np.full(A.shape[0], M)

            # solve LP – we give the remaining time as a simple warm‑start limit
            remaining = max(0.0, deadline - time.time())
            lp_res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                             method='highs', options={'time_limit': remaining})

            if not lp_res.success:
                # if the solver gives up we exit the LP phase for this start point
                break

            g0 = lp_res.x
            sum_g0 = g0.sum()
            if sum_g0 <= 1e-12:
                break

            # scale g0 to the same total mass as f
            g = g0 * (f.sum() / sum_g0)

            # try the scaled vector directly
            val_g = _evaluate(g)
            if val_g < last_val:
                f = g
                last_val = val_g
                no_improve_lp = 0
                print(f"[LP {it_lp:03d}] direct replace → {val_g:.6f}")
                continue

            # otherwise blend with a modest step size; backtrack if it does not improve
            t = 0.5
            improved = False
            while t > 1e-4:
                f_candidate = (1 - t) * f + t * g
                val_cand = _evaluate(f_candidate)
                if val_cand < last_val:
                    f = f_candidate
                    last_val = val_cand
                    improved = True
                    print(f"[LP {it_lp:03d}] blend t={t:.4f} → {val_cand:.6f}")
                    break
                t *= 0.5

            if not improved:
                no_improve_lp += 1
                if no_improve_lp >= 5:
                    # shake – add a tiny random perturbation and renormalise
                    noise = (np.random.rand(n) - 0.5) * 1e-3
                    f = np.maximum(f + noise, 0.0)
                    f /= f.sum()
                    no_improve_lp = 0
                    print(f"[LP {it_lp:03d}] shake")
                # if we cannot improve after a blend, exit LP loop – further LP steps are unlikely to help
                break

        # ------------------------------------------------------------------
        # Phase 2 – short Adam optimisation on a smooth‑max surrogate
        # ------------------------------------------------------------------
        # we keep the best result from the LP phase as a starting point
        u = np.log(f + 1e-30)                     # log‑softmax parameters
        m = np.zeros_like(u)
        v = np.zeros_like(u)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        base_lr = 0.05
        lam = 1.0
        lam_factor = 1.02
        it_adam = 0
        last_improvement = time.time()
        best_local_val = last_val

        # pre‑compute sizes for FFT based convolutions (faster than np.convolve)
        conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        size_corr = 1 << int(np.ceil(np.log2(3 * n)))

        while time.time() < deadline and it_adam < 3000:
            it_adam += 1
            # probabilities from logits
            u_shift = u - np.max(u)
            a = np.exp(u_shift)
            a /= a.sum()

            # full convolution of a with itself (FFT for speed)
            fa = np.fft.rfft(a, conv_fft_size)
            conv = np.fft.irfft(fa * fa, conv_fft_size)[: 2 * n - 1]

            # smooth‑max weights
            m_conv = conv.max()
            w = np.exp(lam * (conv - m_conv))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of smooth‑max w.r.t. a : 2·(w ★ a)
            rev_a = a[::-1]
            fw = np.fft.rfft(w, size_corr)
            frev = np.fft.rfft(rev_a, size_corr)
            corr = np.fft.irfft(fw * frev, size_corr)
            grad_a = 2.0 * corr[n - 1 : n - 1 + n]

            # convert gradient to log‑space (soft‑max Jacobian)
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_u
            v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

            m_hat = m / (1.0 - beta1 ** it_adam)
            v_hat = v / (1.0 - beta2 ** it_adam)

            lr = base_lr * (0.9995 ** it_adam)
            u -= lr * m_hat / (np.sqrt(v_hat) + eps)

            lam *= lam_factor

            # exact evaluation every 30 iterations
            if it_adam % 30 == 0:
                cur_val = _evaluate(a)
                if cur_val < best_local_val:
                    best_local_val = cur_val
                    best_f = a.copy()
                    last_improvement = time.time()
                    print(f"[Adam] iter {it_adam:04d} new best {cur_val:.6f}")

                if time.time() - last_improvement > 2.0:
                    # small random shake in log‑space
                    u += (np.random.rand(n) - 0.5) * 5e-4
                    m.fill(0.0)
                    v.fill(0.0)
                    last_improvement = time.time()

        # pick the best result obtained in this start‑point
        final_candidate = best_f if best_local_val < last_val else f
        final_val = best_local_val if best_local_val < last_val else last_val

        if final_val < best_val:
            best_val = final_val
            best_seq = final_candidate.copy()
            print(f"[GLOBAL] new overall best {best_val:.6f}")

    # ----------------------------------------------------------------------
    # final safety – ensure we return a Python list with non‑negative numbers
    # ----------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution of modest length
        n = 1000
        best_seq = np.full(n, 1.0 / n)

    # Clip to the domain accepted by evaluate_sequence (0 … 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Finished – best value {best_val:.6f} length {best_seq.size}")
    return best_seq.tolist()

"""
Search for a non‑negative step function that minimises

    F(a) = 2·n·max_k (a*a)[k] / (∑ a_i)²

The algorithm consists of two stages.

Stage 1 – LP refinement.
    For a fixed sequence f we compute M = max(conv(f,f)).
    The linear program
        maximise   Σ_j g_j
        subject to  conv(f,g) ≤ M   (one inequality per convolution index)
                    g_j ≥ 0
    returns a vector g₀ that carries more total mass while still respecting the
    current peak autocorrelation.  Scaling g₀ back to the original total mass
    and mixing it with f (or, if it improves, replacing f outright) strictly
    reduces the objective.  Repeating this step yields a monotonically improving
    sequence.

Stage 2 – short Adam optimisation.
    After the LP phase a few Adam steps on the smooth‑max surrogate
    (as in the reference code) are applied to polish the result.

Several initialisations (the historic `height_sequence_1` if present and a few
random Dirichlet draws) are tried; the best sequence found before the
1000 s budget expires is returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list[float]:
    """
    Return a step‑function (list of non‑negative heights) with a low
    evaluation score.  The routine respects the 1000 s time budget,
    combines an LP‑based monotone improvement with a short Adam
    polishing phase and finally returns the best sequence discovered.
    """

    # Helper utilities
    # ------------------------------------------------------------------
    import warnings

    warnings.filterwarnings("ignore", category=RuntimeWarning)

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

        """Wrap the supplied evaluate_sequence (expects a Python list)."""
        return evaluate_sequence(seq.tolist())

    def _max_conv(seq: np.ndarray) -> float:
        """Maximum of the linear self‑convolution."""
        return np.convolve(seq, seq).max()

    def _conv_matrix(f: np.ndarray) -> np.ndarray:
        """Return the (2n‑1)×n matrix A such that A·g = conv(f,g)."""
        n = f.shape[0]
        col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
        row = np.concatenate([[f[0]], np.zeros(n - 1, dtype=f.dtype)])
        return toeplitz(col, row)  # dense matrix, shape (2n‑1, n)

    # Choose a moderate length (≈2000) – large enough to be expressive,
    # small enough to keep the LP tractable.

    initial_vectors = []

        base = np.maximum(base, 0.0)
        if base.sum() == 0.0:
            base = np.full_like(base, 1.0 / base.size)
        base = base / base.sum()
        initial_vectors.append(base)
        print(f"[Init] using down‑sampled height_sequence_1 (len={base.size})")
    except Exception:
        pass

    # add a few random Dirichlet starts
    for _ in range(5):
        a = np.random.dirichlet(np.ones(TARGET_N))
        initial_vectors.append(a)

    best_val = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Main loop over the different start points
    # ------------------------------------------------------------------
    for start_id, a0 in enumerate(initial_vectors, 1):
        if time.time() >= deadline:
            break
        n = a0.shape[0]
        print(f"[Start {start_id}/{len(initial_vectors)}] n={n}")

        # Phase 1 – LP based monotone improvement
        # --------------------------------------------------------------
        f = a0.copy()
        f = f / f.sum()                     # normalise to total mass 1
        it_lp = 0
        last_val = _evaluate(f)
        if last_val < best_val:
