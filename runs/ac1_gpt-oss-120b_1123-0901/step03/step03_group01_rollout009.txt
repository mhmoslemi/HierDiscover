"""
Optimise a non‑negative step‑function a = (a0,…,a_{n‑1}) under the simplex
constraint Σ a_i = √(2n) so that the objective

    2·n·max(conv(a,a)) / (Σ a_i)² = max(conv(a,a))

is as small as possible.  

The core optimiser minimises a smooth approximation of the ℓ∞‑norm
(log‑sum‑exp of the autocorrelation) with an Adam optimiser and a
projection onto the simplex.  The smoothing parameter α is increased
during the run so the algorithm first performs a global search and
then focuses on the true maximum.  After the smooth phase a short
projected sub‑gradient refinement (the previously‑used method) is
applied.  Multiple restarts with different lengths and initialisations
are performed within the allotted budget.

The returned list is a plain Python list of non‑negative heights.

# Simplex projection (Euclidean projection onto {x_i ≥ 0, Σ x_i = s})
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (Σx = s, x ≥ 0)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                       # sort descending
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Convolution utilities (real FFT, full convolution)
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution of a with itself (length 2·len(a)-1) via FFT."""
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


def _conv_full_fft_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via FFT."""
    n = len(x)
    m = len(y)
    n_fft = 1 << ((n + m - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X * Y, n=n_fft)[: n + m - 1]
    return conv


def _max_conv(a: np.ndarray) -> float:
    """Return max(conv(a,a))."""
    return float(_conv_full_fft(a).max())


# --------------------------------------------------------------------------- #
# Smooth‑max optimiser (log‑sum‑exp + Adam) --------------------------------- #
# --------------------------------------------------------------------------- #
def smooth_max_optimize(a_init: np.ndarray,
                        target_sum: float,
                        deadline: float,
                        start_time: float):
    """
    Minimise max(conv(a,a)) using a smooth ℓ∞ surrogate.
    Runs until time.time() >= deadline.
    Returns (best_vector, best_score).
    """
    a = a_init.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    lr = 0.5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)
    t = 1

    best_a = a.copy()
    best_score = _max_conv(a)

    while time.time() < deadline:
        # current autocorrelation
        conv = _conv_full_fft(a)
        max_val = conv.max()

        # continuation on α: start with a smooth surrogate and increase.
        elapsed = (time.time() - start_time) / max(1e-12, deadline - start_time)
        alpha = 0.5 + 19.5 * elapsed          # range [0.5, 20]

        # stable soft‑max weights
        scaled = conv - max_val               # ≤ 0
        w = np.exp(alpha * scaled)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            # all under‑flow → concentrate on the maximal entry
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w = w / w_sum

        # gradient of the smooth max: 2·(w * a_rev) convolution
        a_rev = a[::-1]
        grad_full = _conv_full_fft_general(w, a_rev)          # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]             # relevant slice

        # Adam update
        m1 = beta1 * m1 + (1.0 - beta1) * grad
        m2 = beta2 * m2 + (1.0 - beta2) * (grad * grad)
        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)
        step = lr * m1_hat / (np.sqrt(m2_hat) + eps)
        a = a - step

        # keep feasibility (non‑negative + simplex)
        a = _project_onto_simplex(a, target_sum)

        # occasional random shake to avoid stagnation
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (0.001 * target_sum)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)

        # evaluate current objective
        cur_score = _max_conv(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] smooth_opt new best = {best_score:.6f} (n={n})")
        t += 1

    return best_a, best_score


# --------------------------------------------------------------------------- #
# Projected sub‑gradient optimiser (same as the previous version) ----------- #
# --------------------------------------------------------------------------- #
def subgradient_optimize(a_init: np.ndarray,
                         target_sum: float,
                         max_time: float,
                         start_time: float):
    """Refine a vector using a projected sub‑gradient method."""
    a = a_init.copy()
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _max_conv(a)

    base_lr = 0.2
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    while time.time() < start_time + max_time:
        conv = _conv_full_fft(a)
        max_val = conv.max()
        cur_score = float(max_val)

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{time.time():.2f}] GD new best = {best_score:.6f} (n={n})")

        # active indices (where conv is (almost) maximal)
        eps = 1e-12
        active = np.where(conv >= max_val - eps)[0]

        grad = np.zeros_like(a)
        for k in active:
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]

        max_grad = grad.max()
        if max_grad <= 0.0:
            break

        eta = base_lr * max_val / max_grad

        momentum = beta_mom * momentum - eta * grad
        a_new = a + momentum

        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            base_lr *= 0.9

        a = a_new
        base_lr *= 0.99995

    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Main entry point ---------------------------------------------------------- #
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5      # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------- #
    # Helper to evaluate a vector (fast, exact max of the autocorrelation)
    # ------------------------------------------------------------------- #
    def _evaluate(a):
        return _max_conv(a)

    # ------------------------------------------------------------------- #
    # Outer loop – many restarts with varying lengths.
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        # Problem size: use a moderate range; sometimes reuse current best size.
        if best_seq is not None and np.random.rand() < 0.2:
            n = len(best_seq)
        else:
            n = np.random.randint(900, 2501)   # 900‑2500 works well in practice

        target_sum = np.sqrt(2.0 * n)           # Σ a_i = √(2n)

        # ---------------------------- initialise --------------------------- #
        if "height_sequence_1" in globals() and np.random.rand() < 0.5:
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-3
                init = np.concatenate([src, pad])
        elif best_seq is not None and np.random.rand() < 0.5:
            src = np.asarray(best_seq, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-3
                init = np.concatenate([src, pad])
        else:
            # Random Dirichlet‑like start
            init = np.random.rand(n)

        a0 = _project_onto_simplex(init.astype(float), target_sum)

        cur_score = _evaluate(a0)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a0.copy()
            print(f"[{time.time():.2f}] Init best = {best_score:.6f} (n={n})")

        # --------------------- allocate time for this restart ---------- #
        time_left = deadline - time.time()
        if time_left <= 0:
            break

        # Give roughly half the remaining time to the smooth optimiser,
        # the rest to the sub‑gradient refinement.
        phase_time = min(8.0, time_left * 0.5)
        opt_deadline = time.time() + phase_time

        # ----- smooth‑max optimisation (Adam + log‑sum‑exp) ----- #
        a_smooth, score_smooth = smooth_max_optimize(
            a0, target_sum, opt_deadline, time.time()
        )
        if score_smooth < best_score:
            best_score = score_smooth
            best_seq = a_smooth.copy()
            print(f"[{time.time():.2f}] After smooth_opt best = {best_score:.6f} (n={n})")

        # ----- sub‑gradient refinement (local optimiser) ----- #
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        phase_time = min(6.0, time_left)

        a_gd, score_gd = subgradient_optimize(
            a_smooth, target_sum, phase_time, time.time()
        )
        if score_gd < best_score:
            best_score = score_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.2f}] After GD best = {best_score:.6f} (n={n})")

        # Early exit if we have already beaten the target bound.
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Fallback – uniform vector (should never be used).
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 1500
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    return best_seq.tolist()
