"""
Sub‑gradient descent on the exact max‑convolution objective.

The objective to minimise is

    F(a) = 2·n·max_k (a * a)[k]    (the sum of a is kept equal to 1)

where “*’’ denotes linear self‑convolution.  For a fixed index k* the
gradient of max_k (a*a)[k] w.r.t. a is

    ∇_a  (a*a)[k*] = 2·a_rev_shift ,

i.e. 2·a_{k*‑i} for the overlapping indices and 0 elsewhere.
The algorithm repeatedly

* computes the convolution by FFT,
* finds the index k* attaining the current maximum,
* builds the (sub‑)gradient for that index,
* takes a small step opposite to the gradient,
* projects the result back onto the probability simplex (non‑negative,
  sum‑to‑one).

A few random restarts are performed (the first one starts from the
previous best sequence `height_sequence_1` if it exists).  After a
restart the step‑size decays geometrically; occasional tiny random
“shakes’’ prevent stagnation.  The best sequence found before the
budget expires is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length ``size``.
    Returns the array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * a.shape[0] - 1]


def _max_conv_gradient(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] at the current maximising index ``k``.
    For the overlapping region the gradient entry i is 2·a[k‑i].
    Outside the region it is zero.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    low = max(0, k - (n - 1))
    high = min(n - 1, k)
    i = np.arange(low, high + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * a[j]
    return g


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0 , Σ x_i = 1 }.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    v = np.maximum(v, 0.0)                     # negativity is never helpful
    if v.sum() == 0.0:
        # uniform fallback (should not happen in practice)
        return np.full_like(v, 1.0 / v.shape[0])

    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety: fall back to uniform
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    Search for a non‑negative step‑function (list of heights) with a low
    evaluation score.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence found.
    """

    deadline = time.time() + budget_s - 0.3   # tiny safety margin

    restart = 0

        restart += 1

        # ---- initialise a new candidate ---------------------------------

        else:
