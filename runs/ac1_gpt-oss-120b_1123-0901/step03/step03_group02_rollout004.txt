"""
Propose a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The code is a hybrid of several ideas that proved useful in
previous experiments:

*   A projected Adam optimiser on the simplex.  The gradient is the
    cross‑correlation of the current vector with the soft‑max weights
    of its own self‑convolution.  The temperature τ of the soft‑max
    decays exponentially, moving the optimisation from a smooth
    surrogate of the max towards the true max.

*   Random “shake‑up’’ steps (Gaussian noise) and a cheap local
    hill‑climbing phase after each Adam run.  These non‑gradient moves
    help escape plateaus and fine‑tune the solution.

*   Many independent restarts with diverse lengths (400 … 3500)
    and diverse initial shapes (power‑law, exponential, uniform).
    Occasionally a restart is warm‑started from the current global
    best (truncated or padded).

*   A very small amount of “up‑sampling’’ logic: a good solution for a
    shorter length can be linearly interpolated to a longer length and
    refined further.  This lets the optimiser explore many shapes
    quickly before committing to a larger problem size.

The optimiser works completely in the simplex (Σaᵢ = √(2 n)), which
makes the evaluation equal to the peak of the self‑convolution.
All helper functions are top‑level; the entry point is
``propose_candidate``.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy import sparse

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (sum = s, non‑negative)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – Full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power‑of‑2
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # Numerical clean‑up
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Adam optimiser for a fixed length using a smooth‑max surrogate.
# ----------------------------------------------------------------------
def _optimise_one_length_adam(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 0.001,
    learning_rate: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    shake_prob: float = 0.015,
) -> tuple[np.ndarray, float]:
    """
    Adam‑based optimisation of a fixed length `n`.

    Returns the best sequence (as a NumPy array) and its exact score
    obtained from ``evaluate_sequence``.
    """
    target_sum = np.sqrt(2.0 * n)          # Σ a_i = √(2 n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_iter = 0

    best_seq = a.copy()
    conv = _full_convolution_fft(a, a)
    best_score = 2.0 * n * conv.max() / (target_sum ** 2)

    while time.time() < run_deadline:
        # --------------------------------------------------------------
        # 1) self‑convolution and its maximum
        # --------------------------------------------------------------
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()

        # --------------------------------------------------------------
        # 2) smooth‑max weights.
        #    τ starts relatively large (smooth) and shrinks to min_tau.
        # --------------------------------------------------------------
        tau = max(init_tau_factor * max_val * (0.96 ** t_iter),
                  min_tau_factor * max_val)

        # soft‑max (shifted by max for numerical stability)
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            # fallback – uniform
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # 3) gradient of Σ w_k conv_k  (cross‑correlation a * w_rev)
        # --------------------------------------------------------------
        w_rev = w[::-1]
        corr = _full_convolution_fft(a, w_rev)      # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)            # picks the right slice
        grad = 2.0 * corr[idx]                      # ∂/∂a_i

        # --------------------------------------------------------------
        # 4) Adam update (in the ambient space, before projection)
        # --------------------------------------------------------------
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        step = -learning_rate * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a + step

        # --------------------------------------------------------------
        # 5) Simplex projection (keeps Σ a_i = target_sum)
        # --------------------------------------------------------------
        a_new = _project_onto_simplex(a_new, target_sum)

        # --------------------------------------------------------------
        # 6) Occasional random shake‑up to escape plateaus
        # --------------------------------------------------------------
        if np.random.rand() < shake_prob:
            noise = np.random.randn(n) * (target_sum * 5e-4 / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)

        # --------------------------------------------------------------
        # 7) Evaluate (using the cheap max‑conv proxy) and keep global best
        # --------------------------------------------------------------
        conv_new = _full_convolution_fft(a_new, a_new)
        score_new = 2.0 * n * conv_new.max() / (target_sum ** 2)

        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  Adam new best = {best_score:.6f}")

        a = a_new

    # Final evaluation with the official evaluator (identical rounding)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Very cheap local hill‑climbing (mass redistribution)
# ----------------------------------------------------------------------
def _local_search(a: np.ndarray, target_sum: float, max_iter: int,
                  time_deadline: float) -> np.ndarray:
    """
    Starting from `a`, repeatedly move a tiny amount of mass from one
    index to another if it reduces the peak of the self‑convolution.
    Returns the improved vector (may be unchanged).
    """
    n = a.shape[0]
    best = a.copy()
    conv_best = _full_convolution_fft(best, best)
    max_best = conv_best.max()
    step_scale = 0.02 * target_sum / n   # typical transferred mass

    for _ in range(max_iter):
        if time.time() >= time_deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # propose moving `delta` from i to j
        delta = step_scale * np.random.rand()
        if best[i] < delta:
            continue        # cannot take more than we have

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        # no need to re‑project – total sum unchanged, non‑negativity kept
        conv = _full_convolution_fft(cand, cand)
        cur_max = conv.max()
        if cur_max < max_best:
            best = cand
            max_best = cur_max
            # print occasional improvements
            print(f"[{time.time():.2f}] local improvement: new max = {max_best:.6f}")

    return best

**Explanation of the improvements**

1. **Adam optimiser** – Replaces the simple momentum SGD with Adam, which
   automatically rescales the step per coordinate.  This yields more
   stable and faster convergence for the highly non‑convex landscape.

2. **Exponential temperature schedule** – The soft‑max temperature τ
   starts relatively high (smooth surrogate) and decays exponentially
   to a very small value.  As τ → 0 the gradient becomes the exact
   sub‑gradient of the max, allowing the optimiser to finish with a
   tight bound.

3. **Shake‑up noise** – Periodic Gaussian perturbations (controlled by
   `shake_prob`) keep the iterates from getting stuck in flat regions.

4. **Local hill‑climbing** – After each Adam run a cheap stochastic
   hill‑climbing phase moves tiny amounts of mass between entries.
   Because the objective is cheap to evaluate (FFT‑based convolution)
   this extra refinement often improves the bound by a few thousandths.

5. **Multi‑restart with diverse initialisations** – Random power‑law
   exponents, optional exponential tails, and occasional warm‑starts from
   the current global best give a rich set of starting points.

6. **Upsampling trick** – Good shapes discovered for a shorter length
   are linearly interpolated to a longer length and refined again.
   This lets the optimizer explore many shapes quickly before dedicating
   time to a larger problem.

7. **Time‑aware scheduling** – Each restart receives a bounded slice
