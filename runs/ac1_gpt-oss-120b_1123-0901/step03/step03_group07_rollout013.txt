"""
Hybrid Adam‑based optimizer for the self‑convolution bound.

The algorithm repeatedly launches short Adam runs on a smooth‑max surrogate
of   F(a)=2·n·maxₖ (a*a)[k]/(∑a)² .
* a is parametrised by logits u (soft‑max), guaranteeing a≥0 and Σa=1.
* A temperature‑like λ controls the smooth‑max; it is increased slowly
  (continuation) so the optimisation first sees a smooth landscape and
  later focuses on the true maximum.
* Several families of random initialisations (Dirichlet, exponential,
  half‑sine, sineⁿ, quadratic, bimodal) are used to diversify the search.
* If the optimisation stagnates for a few seconds a small random shake
  of the logits is applied and the Adam moments are reset.
* After the main restarts a final fine‑tuning stage starts from the best
  sequence found so far with a higher λ and a smaller learning‑rate.
* The whole routine respects a user‑supplied time budget (default 1000 s)
  and returns the best sequence (plain list of floats) discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max weights wₖ = exp(lam·convₖ) / Σⱼ exp(lam·convⱼ)."""
    m = conv.max()
    # exponent is non‑positive → no overflow, only possible under‑flow
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:                     # safety – should not happen
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int = None) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate with respect to the original
    probabilities ``a``.
    The closed form is  ∇ = 2·(w ★ a) .
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    # slice that corresponds to indices 0 … n‑1 of the original vector
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad
