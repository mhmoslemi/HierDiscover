"""
Optimise a non‑negative step‑function for the “evaluate_sequence” task.

The true objective after normalising the L1‑norm to sqrt(2·n) is simply the
maximum entry of the full convolution a*a (the auto‑correlation).
The optimisation below works with a *smooth* approximation of the max,
the log‑sum‑exp (soft‑max) with a temperature τ.  For large τ the loss
is smooth and gradients are informative everywhere; τ is gradually
decreased so that the optimiser finally concentrates on the true peak.
The gradient of the soft‑max loss can be written analytically and is
computed efficiently with FFT‑based convolutions.  Adam is used for the
updates, together with projection onto the feasible set (non‑negative,
fixed L1‑norm).  Random restarts, occasional jitter and a short greedy
local‑search are added.  The routine explores several problem sizes
and returns the best sequence found before the 1000 s budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Return the full convolution a*a (length 2·len(a)-1) using FFT.
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, tau: float):
    """
    Compute a smooth approximation of max(a*a) (log‑sum‑exp) and its
    gradient w.r.t. the vector *a*.

    Returns
    -------
    smooth_val : float
        tau * log( sum_k exp( (c_k) / tau ) )   where c = a*a.
    grad : np.ndarray
        ∂smooth_val/∂a   (shape same as a)
    """
    n = a.shape[0]

    # ----- convolution a*a -------------------------------------------------
    conv = _full_convolution(a)                     # shape (2n‑1,)

    # ----- soft‑max weights -------------------------------------------------
    max_c = conv.max()
    shift = conv - max_c                           # for numerical stability
    exp_shift = np.exp(shift / tau)
    sum_exp = exp_shift.sum()
    w = exp_shift / sum_exp                        # w_k = soft‑max weight

    # ----- gradient:  g_i = 2 * Σ_k w_k * a_{k‑i}  (zero outside) ----------
    # This is a cross‑correlation of w with a.  It can be obtained as
    #   g = 2 * (w_rev  *  a)[offset‑i]  where w_rev = w[::-1].
    w_rev = w[::-1]                               # length 2n‑1
    N = 1 << ((len(w_rev) + n - 1).bit_length())  # FFT size for the correlation
    W = np.fft.rfft(w_rev, n=N)
    A = np.fft.rfft(a, n=N)
    corr = np.fft.irfft(W * A, n=N)[: len(w_rev) + n - 1]

    offset = len(w_rev) - 1                        # = 2n‑2
    idx = offset - np.arange(n)                   # slice that gives g_i
    grad = 2.0 * corr[idx]

    # ----- smooth objective value -------------------------------------------
    smooth_val = max_c + tau * np.log(sum_exp)

    return smooth_val, grad, conv.max()


# ----------------------------------------------------------------------
def _adam_smooth_optimize(init_a: np.ndarray,
                          n: int,
                          target_sum: float,
                          deadline: float,
                          max_iters: int = 200_000):
    """
    Adam optimiser on the smooth max‑convolution loss.
    Returns the best sequence seen (according to the *true* max) and its max.
    """
    a = init_a.copy()
    # normalise to the correct L1 norm
    s = a.sum()
    if s > 0:
        a *= (target_sum / s)
    else:
        a[:] = target_sum / n

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.08

    # temperature schedule
    tau = 0.5
    tau_min = 1e-4
    tau_decay = 0.995

    best_vec = a.copy()
    best_val = np.inf

    iteration = 0
    while time.time() < deadline and iteration < max_iters:
        iteration += 1

        # smooth loss, its gradient and the *true* max of the convolution
        smooth_val, grad, true_max = _smooth_max_and_grad(a, tau)

        # keep best according to the true objective
        if true_max < best_val:
            best_val = true_max
            best_vec = a.copy()
            print(f"[len {n:5d} iter {iteration:8d}] "
                  f"new best max = {best_val:.6f}  τ={tau:.5f}")

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = base_lr / (1.0 + 1e-4 * iteration)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection: non‑negative + L1 normalisation
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a *= (target_sum / s)
        else:
            a[:] = target_sum / n

        # occasional tiny jitter to escape flat plateaus
        if iteration % 5000 == 0:
            jitter = np.random.randn(n) * 1e-4 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= target_sum / a.sum()

        # cool the temperature
        if tau > tau_min:
            tau *= tau_decay

    return best_vec, best_val


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  target_sum: float,
                  steps: int,
                  deadline: float):
    """
    Very cheap greedy local search: move a small fraction of mass from a
    random source index to a random destination index if the move reduces
    the true max‑convolution.
    """
    n = a.shape[0]
    best = a.copy()
    best_val, _ = _full_convolution(best).max(), None

    for step in range(steps):
        if time.time() >= deadline:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j or best[i] <= 0:
            continue
        delta = best[i] * 0.02                      # 2 % of the source mass
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # sum stays constant, only non‑negativity to check
        if cand[i] < 0:
            continue
        cur_max = _full_convolution(cand).max()
        if cur_max < best_val:
            best = cand
            best_val = cur_max
    return best, best_val
