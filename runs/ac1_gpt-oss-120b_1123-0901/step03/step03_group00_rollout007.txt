"""
Propose a step‑function sequence that (approximately) minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

using a two‑phase optimisation:
  1. Adam on a smooth‑max surrogate (log‑sum‑exp) to quickly lower the bound.
  2. Projected sub‑gradient descent on the exact max‑convolution objective
     (gradient = 2·(w ★ a) where w is the indicator of the maximal
     convolution entries).

Multiple problem sizes are tried (up to ~16 k entries) and several
re‑starts (including warm‑starts from the global variable
`height_sequence_1`).  The routine respects a 1000 s time budget and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using an FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)], i = 0 … n‑1.
    Linear cross‑correlation needed for the gradient.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    n = a.shape[0]
    start = n - 1
    return corr[start:start + n]  # length n


# ----------------------------------------------------------------------
# Simplex projection (Euclidean projection onto {x≥0, Σx=1})
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the probability simplex."""
    # based on algorithm of Wang & Carreira‑Perpiñán (ICML 2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective (sum of a is assumed to be 1, but we keep the denominator)
def _objective(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a (a may not sum to 1)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Adam optimisation on the smooth‑max surrogate
def _adam_optimize(a0: np.ndarray,
                   n_iters: int,
                   lr: float,
                   lam_start: float,
                   lam_factor: float,
                   deadline: float,
                   verbose_prefix: str = ""):
    """
    Perform Adam updates on the soft‑max logits of a.
    Returns (best_a, best_val, last_a, last_val).
    """
    n = a0.shape[0]
    # logits initialisation
    x = np.log(a0 + 1e-15)

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # FFT sizes (powers of two)
    fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))  # len(w)+len(a)-1 ≤ 3n

    best_a = None
    best_val = np.inf
    last_improve = time.time()
    lam = lam_start

    for t in range(1, n_iters + 1):
        if time.time() > deadline:
            break

        # ----- soft‑max (guarantees simplex) -----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ----- self‑convolution -----
        conv = _full_convolution(a, fft_len=fft_len_conv)
        max_c = conv.max()

        # ----- smooth‑max weights -----
        # w ∝ exp(lam·(conv - max_c))
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.shape[0]
        else:
            w /= w_sum

        # ----- gradient of the surrogate w.r.t. a -----
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=fft_len_corr)

        # ----- back‑prop through soft‑max -----
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, 1e6)

        # ----- exact objective (cheap because sum(a)≈1) -----
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"{verbose_prefix}[iter={t:05d}] new best {val:.6f}")

        # ----- stagnation handling -----
        if time.time() - last_improve > 5.0:
            # small random perturbation of logits
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"{verbose_prefix}[iter={t}] shake (stagnation)")

    # final a (may be the last iterate)
    final_a = np.exp(x - np.max(x))
    final_a = final_a / final_a.sum()
    final_val = _objective(final_a)

    return best_a, best_val, final_a, final_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution objective
def _subgradient_optimize(a0: np.ndarray,
                          max_iters: int,
                          step0: float,
                          deadline: float,
                          verbose_prefix: str = ""):
    """
    Sub‑gradient descent on 2·n·max(a*a) / (∑a)².
    a stays on the probability simplex via Euclidean projection.
    Returns the best (a, value) pair encountered.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    # FFT size for cross‑correlation (w length = 2n‑1)
    fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a)
        max_c = conv.max()

        # indices attaining (almost) the maximum
        eps = 1e-12
        I = np.where(conv >= max_c - eps)[0]
        if I.size == 0:
            w = np.zeros_like(conv)
        else:
            w = np.zeros_like(conv)
            w[I] = 1.0 / I.size

        # gradient of max‑conv w.r.t. a
        grad = 2.0 * _cross_correlation(w, a, fft_len=fft_len_corr)

        # diminishing step size
        eta = step0 / np.sqrt(t)

        a = a - eta * grad
        a = _project_onto_simplex(a)

        val = _objective(a)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"{verbose_prefix}[sgd iter={t:05d}] new best {val:.6f}")

        # occasional random shake to escape flat plateaus
        if t % 2000 == 0:
            a += (np.random.rand(n) - 0.5) * 1e-3
            a = _project_onto_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main driver
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best sequence (list of floats) found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper to create an initial probability vector for a given length.
    def _initial_vector(n, restart_id):
        # Warm‑start from the globally best of matching length
        if restart_id == 0 and best_seq is not None and len(best_seq) == n:
            a = np.array(best_seq, dtype=np.float64)
            a = np.maximum(a, 0.0)
            if a.sum() < 0.01:
                a = np.random.dirichlet(np.ones(n))
            else:
                a = a / a.sum()
            print(f"[n={n}] warm start from previous best")
            return a

        # Warm‑start from the provided legacy construction (if present)
        if 'height_sequence_1' in globals():
            cand = np.array(height_sequence_1, dtype=np.float64)
            if cand.shape[0] >= n:
                cand = cand[:n]
            else:
                cand = np.pad(cand, (0, n - cand.shape[0]))
            cand = np.maximum(cand, 0.0)
            if cand.sum() < 0.01:
                cand = np.random.dirichlet(np.ones(n))
            else:
                cand = cand / cand.sum()
            print(f"[n={n}] start from height_sequence_1")
            return cand

        # Random Dirichlet (spreads mass uniformly)
        a = np.random.dirichlet(np.ones(n))
        return a

    # ------------------------------------------------------------------
    # Problem sizes – larger n gives more flexibility but costs more time.
    lengths = [1024, 2048, 4096, 8192, 16384]

    # For each length we perform a few independent restarts.
    restarts_per_length = 4

    for n in lengths:
        if time.time() > deadline:
            break
        print("\n=== Optimising length n =", n, "===")

        for r in range(restarts_per_length):
            if time.time() > deadline:
                break

            # --------------------------------------------------------------
            # 1) initialise a probability vector (sum to 1)
            a0 = _initial_vector(n, r)

            # --------------------------------------------------------------
            # 2) Adam on smooth‑max surrogate
            print(f"[n={n} restart={r}] start Adam phase")
            adam_best_a, adam_best_val, last_a, last_val = _adam_optimize(
                a0=a0,
                n_iters=15000,
                lr=0.05,
                lam_start=0.5,
                lam_factor=1.02,
                deadline=deadline,
                verbose_prefix=f"[n={n} r={r}] "
            )
            # keep the best from Adam
            if adam_best_val < best_val:
                best_val = adam_best_val
                best_seq = adam_best_a.copy()
                print(f"[n={n} r={r}] global best improved to {best_val:.6f}")

            # --------------------------------------------------------------
            # 3) Sub‑gradient refinement (starting from Adam's best)
            print(f"[n={n} r={r}] start sub‑gradient phase")
            sg_best_a, sg_best_val = _subgradient_optimize(
                a0=adam_best_a,
                max_iters=8000,
                step0=0.1,
                deadline=deadline,
                verbose_prefix=f"[n={n} r={r}] "
            )
            if sg_best_val < best_val:
                best_val = sg_best_val
                best_seq = sg_best_a.copy()
                print(f"[n={n} r={r}] global best improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen but protects against empty result.
    if best_seq is None:
        n_fallback = 2000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = _objective(best_seq)
        print("Fallback sequence used, objective =", best_val)

    # Convert to plain Python list for the external evaluator.
    return best_seq.tolist()
