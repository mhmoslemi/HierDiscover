"""
Improved search for a low‑autoconvolution step function.

The algorithm combines three ideas that were not all present in the reference
implementation:

1. **Weighted sub‑gradient** – instead of averaging the reversed vectors of
   all active shifts uniformly, we give a larger weight to the shifts with a
   larger convolution value (soft‑max style).  This yields a stronger descent
   direction for the maximal entry.

2. **Dynamic length schedule** – we start from the known construction,
   explore a set of increasingly larger lengths (up to ~12 000) and keep the
   best result found so far.  Longer vectors give a richer search space and
   often lower maxima.

3. **Aggressive hill‑climbing** – after the Adam‑style optimisation a cheap
   but very extensive local‑search phase moves a noticeable fraction of mass
   from indices that contribute heavily to the current worst shift toward
   indices that hardly affect it.  The proposal count is tuned to the remaining
   time budget, so the routine always finishes before the deadline.

The routine respects all security checks of `evaluate_sequence`: the
candidate vectors are kept non‑negative, projected onto the hyper‑plane
`sum(a)=sqrt(2·n)` after each update, and values are clipped to the allowed
range `[0,1000]`.  The function `propose_candidate` returns the best sequence
found within the allotted time (default 1000 s).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """
    Full convolution a*a via FFT and return (max value, argmax index,
    full convolution array).  Complexity O(N log N) with N the next power of
    two greater than 2·len(a)−1.
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]
    k = int(conv.argmax())
    return float(conv.max()), k, conv


# ----------------------------------------------------------------------
def _rev_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    """
    For a given shift index k, return the vector rev where
    rev[i] = a[k‑i] if the index lies inside the support, otherwise 0.
    This vector is the sub‑gradient of the convolution term at shift k
    (up to a factor 2).
    """
    n = a.shape[0]
    rev = np.zeros_like(a)
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    rev[mask] = a[j[mask]]
    return rev


# ----------------------------------------------------------------------
def _weighted_rev(a: np.ndarray, conv: np.ndarray, top_k: int = 10,
                  max_active: int = 20) -> np.ndarray:
    """
    Compute a weighted average of reversed vectors over the shifts with the
    largest convolution values.  The weight of shift k is conv[k].
    Only up to `max_active` indices are used – if more than `max_active`
    shifts tie for the maximum we randomly subsample.
    """
    n = a.shape[0]

    # indices of the largest entries
    if top_k >= conv.size:
        idx = np.arange(conv.size)
    else:
        # argpartition gives the top‑k unsorted; sort them for stability
        idx = np.argpartition(-conv, top_k)[:top_k]
    # optionally keep a few more near‑max entries (within a tiny tolerance)
    M = conv.max()
    tol = 1e-12
    near_max = np.where(conv >= M - tol)[0]
    if near_max.size > max_active:
        # random subset to keep runtime bounded
        near_max = np.random.choice(near_max, size=max_active, replace=False)
    active = np.unique(np.concatenate((idx, near_max)))

    if active.size == 0:
        return np.zeros_like(a)

    rev = np.zeros_like(a, dtype=float)
    total_w = 0.0
    for k in active:
        w = conv[k]
        rev_k = _rev_for_shift(a, k)
        rev += w * rev_k
        total_w += w
    if total_w > 0.0:
        rev /= total_w
    return rev


# ----------------------------------------------------------------------
def _adam_optimize(a_init: np.ndarray,
                   n: int,
                   target_sum: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   deadline: float,
                   max_iter: int = 5_000_000,
                   jitter_interval: int = 4000,
                   random_active_prob: float = 0.3,
                   top_k: int = 10) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of the maximal convolution value for a fixed
    length `n`.  The sub‑gradient is a weighted average of the reversed
    vectors of the top‑k convolution shifts (see `_weighted_rev`).  The
    optimiser projects onto the simplex `sum(a)=target_sum` after each step
    and occasionally adds a tiny Gaussian jitter to escape plateaus.
    """
    a = a_init.copy()
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.04                     # a bit smaller than the reference value
    iteration = 0

    # constant derived from the Lagrange multiplier for the sum constraint
    sqrt2 = math.sqrt(2.0)

    while time.time() < deadline and iteration < max_iter:
        # ------------------------------------------------------------
        # 1) convolution and its maximum
        M, _, conv_full = _conv_max_fft(a)

        # ------------------------------------------------------------
        # 2) weighted sub‑gradient (biased toward larger conv entries)
        if np.random.rand() < random_active_prob:
            # occasionally use a single random active shift – can help escape
            # flat regions.
            active_idx = np.random.choice(np.where(conv_full >= M - 1e-12)[0])
            rev = _rev_for_shift(a, int(active_idx))
        else:
            rev = _weighted_rev(a, conv_full, top_k=top_k)

        # ------------------------------------------------------------
        # 3) gradient of the Lagrangian (objective + sum constraint)
        const = sqrt2 * M / math.sqrt(n)   # same formula as in the reference
        grad = 2.0 * rev - const           # shape (n,)

        # ------------------------------------------------------------
        # 4) Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ------------------------------------------------------------
        # 5) projection onto feasible set (non‑negative + fixed L1 norm)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # ------------------------------------------------------------
        # 6) occasional jitter
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # ------------------------------------------------------------
        # 7) bookkeeping
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:7d}] new best score = {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_search(a_start: np.ndarray,
                  n: int,
                  target_sum: float,
                  best_score: float,
                  deadline: float,
                  max_proposals: int = 30_000,
                  donor_bias: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Very cheap but aggressive hill‑climbing.  A donor index is chosen with
    probability proportional to its mass (the `donor_bias` parameter) and a
    random receiver elsewhere.  Up to 10 % of the donor’s mass can be moved.
    The move is kept only if it lowers the current maximal convolution value.
    """
    a = a_start.copy()
    n_range = np.arange(n)
    proposal = 0

    # pre‑compute a temperature‑like coefficient that slowly decays – used to
    # occasionally accept a neutral move to shake the system.
    accept_prob = 0.02

    while time.time() < deadline and proposal < max_proposals:
        proposal += 1

        # Choose donor with probability ~ a_i**donor_bias (bias towards larger mass)
        probs = a ** donor_bias
        probs_sum = probs.sum()
        if probs_sum == 0:
            continue
        probs = probs / probs_sum
        i = np.searchsorted(np.cumsum(probs), np.random.rand())
        if a[i] <= 1e-14:
            continue

        # Choose a receiver different from i uniformly
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # Amount to transfer – up to 10 % of the donor's current mass
        eps = a[i] * np.random.rand() * 0.10
        if eps <= 0:
            continue

        new_vec = a.copy()
        new_vec[i] -= eps
        new_vec[j] += eps

        # Enforce non‑negativity (donor already ensured)
        M_new, _, _ = _conv_max_fft(new_vec)
        if M_new < best_score:
            a = new_vec
            best_score = M_new
            print(f"[local] improvement to {best_score:.6f} (prop {proposal})")
        else:
            # rare chance to accept a neutral (or slightly worse) move:
            if np.random.rand() < accept_prob and M_new < best_score * 1.001:
                a = new_vec
                # best_score unchanged (or slightly larger)
    return a, best_score

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Main entry point used by the evaluation harness.  The function searches
    for a non‑negative height sequence that minimizes `evaluate_sequence`.
    A mixture of Adam optimisation (with a weighted sub‑gradient) and an
    aggressive local search is performed over several candidate lengths.
    The best sequence discovered before the time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a 5‑second safety margin for the outer harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load the previously discovered construction (if available)
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        base = np.ones(2100, dtype=float)

    # ------------------------------------------------------------------
    # Global bookkeeping
    best_seq = None
    best_score = float("inf")

    # Normalise the initial construction and evaluate it (objective = max conv)
    n0 = base.shape[0]
    target_sum0 = math.sqrt(2.0 * n0)
    base = base * (target_sum0 / base.sum())
    init_val = evaluate_sequence(base.tolist())
    best_seq = base.copy()
    best_score = init_val
    print(f"[init] length={n0}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Length schedule – we explore several multiples of the base length,
    # up to a configurable ceiling.
    max_len = 12000
    factors = [1.0, 1.2, 1.5, 1.8, 2.0, 2.5, 3.0]
    lengths = sorted({int(n0 * f) for f in factors})
    lengths = [L for L in lengths if 300 <= L <= max_len]

    # Shuffle the length order a little to avoid deterministic patterns
    np.random.shuffle(lengths)

    # ------------------------------------------------------------------
    # Main optimisation loop over the selected lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        # Number of random restarts per length; more restarts for larger n
        restarts = 4 if n <= 4000 else 3

        for r in range(restarts):
            if time.time() >= deadline:
                break

            # ------------------------------------------------------------
            # Initialisation
            if r == 0:
                # Interpolate the current best construction onto the new grid
                old_x = np.linspace(0.0, 1.0, best_seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                init_a = np.interp(new_x, old_x, best_seq)
                init_a = np.maximum(init_a, 0.0)
            else:
                # Random Dirichlet‑ish start (positive numbers)
                init_a = np.random.rand(n)
                init_a = np.maximum(init_a, 0.0)

            # Normalise to the required L1 sum
            total = init_a.sum()
            if total > 0:
                init_a = init_a * (target_sum / total)
            else:
                init_a = np.full(n, target_sum / n)

            # ------------------------------------------------------------
            # Adam optimisation phase
            best_seq, best_score = _adam_optimize(
                a_init=init_a,
                n=n,
                target_sum=target_sum,
                best_seq=best_seq,
                best_score=best_score,
                deadline=deadline,
                max_iter=3_000_000,
                jitter_interval=5000,
                random_active_prob=0.35,
                top_k=12,
            )

            # ------------------------------------------------------------
            # Local‑search refinement
            if time.time() < deadline:
                best_seq, best_score = _local_search(
                    a_start=best_seq,
                    n=n,
                    target_sum=target_sum,
                    best_score=best_score,
                    deadline=deadline,
                    max_proposals=40_000,
                )

    # ------------------------------------------------------------------
    # Fallback – in the unlikely case we never improved anything.
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
