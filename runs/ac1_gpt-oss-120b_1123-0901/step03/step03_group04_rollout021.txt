"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm overview
------------------
* A set of problem sizes (lengths) around the known construction is
  explored.
* For each length several random restarts are performed.
* Each restart is refined by a *smoothed* Adam optimiser:
    – The max‑convolution is approximated by a log‑sum‑exp (temperature
      τ) – a smooth upper bound of the true maximum.
    – The gradient of this smooth bound is obtained analytically via
      FFT‑based convolutions (no explicit loops over shifts).
    – Adam updates are applied, followed by a projection onto the
      feasible set (non‑negative entries and the L1‑norm fixed to
      √(2 n)).
    – τ is gradually annealed, turning the smooth bound into the true
      max and helping the optimiser escape plateaus.
* After the Adam phase a cheap stochastic local‑search (mass transfer
  between two random positions) is run for a few‑thousand steps.
* The best sequence seen across all lengths and restarts is returned.
* All timings respect the 1000 s budget; the routine stops early if the
  deadline is reached.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full convolution of two real vectors using a real FFT.
    Returns a real array of length len(a)+len(b)-1.
    """
    n = a.shape[0]
    m = b.shape[0]
    L = 1 << ((n + m - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=L)
    B = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(A * B, n=L)
    return conv[: n + m - 1]


# --------------------------------------------------------------
def _conv_and_grad(a: np.ndarray, tau: float):
    """
    Compute the full self‑convolution a*a, a smooth approximation of its
    maximum (log‑sum‑exp with temperature τ) and the gradient of the
    smooth maximum w.r.t. a.

    Returns
    -------
    conv_self : ndarray, length 2n‑1
        a * a.
    smooth_max : float
        τ·log Σ_k exp(conv_k/τ)  (numerically stable).
    grad_smoothed : ndarray, shape (n,)
        ∂smooth_max / ∂a .
    """
    n = a.shape[0]
    # size large enough for both convolutions (a*a) and (p_rev*a)
    L = 1 << ((3 * n - 2).bit_length())
    # FFT of a (used twice)
    A = np.fft.rfft(a, n=L)

    # ----- self‑convolution -------------------------------------------------
    conv_self = np.fft.irfft(A * A, n=L)[: 2 * n - 1]

    # ----- smooth maximum (log‑sum‑exp) -------------------------------------
    shift = conv_self.max()                     # for numerical stability
    # all entries ≤ shift, so (conv‑shift)/τ ≤ 0 ⇒ exp ≤ 1, no overflow
    exp_term = np.exp((conv_self - shift) / tau)
    sum_w = exp_term.sum()
    p = exp_term / sum_w                        # soft‑max weights (sum to 1)
    smooth_max = shift + tau * np.log(sum_w)

    # ----- gradient of the smooth maximum ----------------------------------
    # gradient = 2 * Q a  where Q_{i,j}=p_{i+j}
    p_rev = p[::-1]                             # reversed weights
    B = np.fft.rfft(p_rev, n=L)                 # FFT of reversed p
    conv_rev = np.fft.irfft(A * B, n=L)[: 3 * n - 2]

    m_len = 2 * n - 1                           # length of a*a
    # slice = conv_rev[m_len-n : m_len] (size n) – reversed order needed
    slice_ = conv_rev[m_len - n : m_len]
    g = slice_[::-1]                            # shape (n,)
    grad_smoothed = 2.0 * g                     # ∂smooth_max/∂a

    return conv_self, smooth_max, grad_smoothed


# --------------------------------------------------------------
def _smooth_adam_optimize(init_a: np.ndarray,
                          n: int,
                          target_sum: float,
                          deadline: float,
                          max_iters: int = 10_000_000):
    """
    Adam optimisation using the smooth max‑convolution described above.
    Returns the best vector seen (according to the true max) and its
    objective value (the true max, i.e. evaluate_sequence after normalising).
    """
    a = init_a.copy()
    s = a.sum()
    if s > 0.0:
        a *= (target_sum / s)
    else:
        a[:] = target_sum / n

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.04

    # initial temperature (smoothness)
    conv0 = _conv_fft(a, a)
    delta0 = conv0.max() - conv0.min()
    tau = max(delta0 * 0.5, 0.01)      # start fairly smooth
    min_tau = 1e-5

    iteration = 0
    best_vec = a.copy()
    best_val = float(conv0.max())     # true max (objective after normalising)
    print(f"[opt] start length={n}, init max={best_val:.6f}")

    while time.time() < deadline and iteration < max_iters:
        # ------------------------------------------------------------
        # 1. convolution, smooth max and its gradient
        conv_self, smooth_max, grad_smooth = _conv_and_grad(a, tau)

        # true max (needed for bookkeeping & final objective)
        M = conv_self.max()
        # ------------------------------------------------------------
        # 2. gradient of the full objective
        sum_a = a.sum()
        coeff = (2.0 * n) / (sum_a * sum_a)        # 2n / sum^2
        const = (4.0 * n * smooth_max) / (sum_a ** 3)
        grad = coeff * grad_smooth - const       # shape (n,)

        # ------------------------------------------------------------
        # 3. Adam step
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = base_lr / (1.0 + 1e-4 * iteration)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ------------------------------------------------------------
        # 4. projection onto feasible set
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a *= (target_sum / s)
        else:
            a[:] = target_sum / n

        # ------------------------------------------------------------
        # 5. keep best according to true max
        if M < best_val:
            best_val = M
            best_vec = a.copy()
            print(f"[len {n:5d} iter {iteration:7d}] new best max={best_val:.6f}")

        # ------------------------------------------------------------
        # 6. temperature annealing
        tau = max(tau * 0.99, min_tau)

        # occasional jitter to get out of flat regions
        if iteration % 4000 == 0:
            jitter = np.random.randn(n) * 0.0008 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

    return best_vec, best_val


# --------------------------------------------------------------
def _local_search(a: np.ndarray,
                  n: int,
                  target_sum: float,
                  steps: int,
                  deadline: float):
    """
    Very cheap stochastic local search: repeatedly move a tiny amount of
    mass from one random index to another if the move reduces the true
    max‑convolution.
    """
    best = a.copy()
    best_val, _ = _conv_fft(best, best).max(), None
    for step in range(steps):
        if time.time() >= deadline:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * 0.02
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # projection keeps sum unchanged and non‑negativity is already satisfied
        M = _conv_fft(cand, cand).max()
        if M < best_val:
            best = cand
            best_val = M
    return best, best_val
