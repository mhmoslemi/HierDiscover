def propose_candidate(seed=42, budget_s=1000, **kwargs):
    # Setup
    import numpy as np, math, time

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # Determine lengths
    lengths = [500, 1000, 2100, 4000, 6000, 8000]
    best_seq = None
    best_score = float('inf')
    # try each length up to some time
    for n in lengths:
        if time.time() >= deadline:
            break
        S = math.sqrt(2.0 * n)
        # init sequence: arcsine shape
        xs = (np.arange(n) + 0.5) / n
        eps = 1e-12
        w = 1.0 / np.sqrt(np.maximum(xs * (1 - xs), eps))
        a = w * (S / w.sum())
        # evaluate
        score = evaluate_sequence(a.tolist())
        # keep best
        if score < best_score:
            best_seq = a.copy()
            best_score = score
            print(f"[init] n={n} score={score:.6f}")

        # Local improvement using directed gradient descent
        # In each iteration compute rev and transfer mass from high rev to low rev
        # limit number of iterations based on remaining time (e.g., up to 200k)
        it = 0
        # parameters
        max_iter = 1000000
        # step fraction schedule
        # We will attempt to run while time left > small threshold
        while time.time() < deadline and it < max_iter:
            # compute convolution max and active indices
            # use FFT for speed
            conv = np.fft.irfft(np.fft.rfft(a, n=1 << ((2 * n - 1).bit_length()))**2)[:2*n-1]
            M = conv.max()
            # stop early if not improvement possible??? Not needed.
            # compute active set where conv is near M
            tol = 1e-12
            active = np.where(conv >= M - tol)[0]
            # compute rev average
            rev = np.zeros_like(a)
            idx = np.arange(n)
            for k in active:
                j = k - idx
                # mask for valid indices
                mask = (j >= 0) & (j < n)
                rev[mask] += a[j[mask]]
            if active.size > 0:
                rev /= float(active.size)
            # find donor and receiver
            donor = np.argmax(rev)    # max gradient
            receiver = np.argmin(rev) # min gradient
            if donor == receiver or rev[donor] <= rev[receiver] + 1e-15:
                # cannot improve further
                break
            # transfer amount proportional to donor's mass and a learning factor
            # learning factor schedule: decays with it
            lr = 0.1 / (1.0 + it * 1e-5)  # small
            delta = lr * a[donor]
            if delta < 1e-12:
                break
            # apply transfer
            a[donor] -= delta
            a[receiver] += delta
            # project onto simplex (non-negative sum fixed)
            # due to transfer, sum stays same, positivity maintained
            # but ensure no negative small rounding
            if a[donor] < 0:
                # adjust rounding errors
                a[receiver] += a[donor]  # a[donor] is negative small
                a[donor] = 0.0

            it += 1
            # occasionally recompute score and update best
            if it % 500 == 0:
                cur_score = evaluate_sequence(a.tolist())
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[len {n}] it {it} new best {best_score:.6f}")
        # after loop
    # fallback uniform if none
    if best_seq is None:
        n = 2100
        S = math.sqrt(2.0 * n)
        best_seq = np.full(n, S / n)
        best_score = evaluate_sequence(best_seq.tolist())
    print(f"[done] best_score={best_score:.6f}, len={best_seq.shape[0]}")
    return best_seq.tolist()

import time, math, numpy as np

def _conv_max_fft(a):
    n = a.shape[0]
    N = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A*A, n=N)[:2*n-1]
    return float(conv.max()), conv

def _project_simplex(v, s):
    # from Duchi et al. 2008
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(n)+1) > 0)[0]
    if rho.size==0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def _gen_beta_shape(n, alpha, eps=1e-12):
    # beta(alpha, alpha) pdf up to constant
    xs = (np.arange(n)+0.5)/n
    w = np.power(np.maximum(xs*(1-xs), eps), alpha-1.0)
    return w

def _initial_arcsine(n):
    # alpha=0.5
    return _gen_beta_shape(n, 0.5)

def _adam_opt(a_init, target_sum, max_iter, deadline):
    a = a_init.copy()
    a = a * (target_sum / a.sum())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05
    sqrt2 = math.sqrt(2.0)
    n = a.shape[0]
    best_seq = a.copy()
    best_score = float('inf')
    it = 0
    while time.time() < deadline and it < max_iter:
        # compute convolution max and active set
        M, conv = _conv_max_fft(a)
        # identify active indices where conv close to max
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]
        # compute subgradient (average reversed)
        rev = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            rev[mask] += a[j[mask]]
        if active.size:
            rev /= float(active.size)
        # gradient of objective (the subgradient plus term for the sum constraint)
        # we work directly on max convolution, the sum constraint is enforced by projection
        grad = rev  # direction to *increase* M; we will descend
        # perform a step of projected gradient descent with Adam like update
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        lr = lr0 / (1 + 1e-4 * it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection onto simplex (nonnegative sum = target_sum)
        a = _project_simplex(a, target_sum)
        # evaluate current objective value
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
        # occasional jitter
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-3 * target_sum / n
            a = np.maximum(a + jitter, 0)
            a = _project_simplex(a, target_sum)
    return best_seq, best_score

def _local_search(a_start, target_sum, max_proposals, deadline):
    a = a_start.copy()
    n = a.shape[0]
    best_score = float('inf')
    # compute initial score
    M, _ = _conv_max_fft(a)
    best_score = M
    best_a = a.copy()
    proposals = 0
    while time.time() < deadline and proposals < max_proposals:
        proposals += 1
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        # prefer moving from high gradient indices
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        # amount up to 5% of donor's mass
        delta = a[i] * np.random.rand() * 0.05
        if delta <= 0:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_a = _project_simplex(new_a, target_sum)  # ensure feasibility
        M_new, _ = _conv_max_fft(new_a)
        if M_new < best_score:
            a = new_a
            best_score = M_new
            best_a = new_a.copy()
            # small print optionally
            # print(f"[local] improvement {best_score:.6f}")
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The search proceeds as follows:
      1. Several lengths are examined (≈ 2000–6000).
      2. For each length a family of Beta‑shaped initial vectors
         p_i∝(x_i(1−x_i))^{α−1} is generated (α∈{0.5,0.55,0.6,…,1.0}).
      3. Each candidate is normalised to the required L1‑norm
         ( √(2 n) ) and a short projected‑Adam optimisation is run.
      4. The Adam result is refined by a cheap hill‑climb that repeatedly
         transfers a small amount of mass from a high‑gradient index to a
         low‑gradient one, keeping the sum fixed.
      5. The best sequence seen before the time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = float('inf')

    # lengths to try – we keep them moderate so that many iterations fit
    base_len = 2100
    length_factors = [0.9, 1.0, 1.2, 1.5, 2.0]
    lengths = sorted({int(base_len * f) for f in length_factors})
    lengths = [l for l in lengths if 300 <= l <= 8000]

    # shape parameters (α for Beta(α,α)). 0.5 is arcsine, >0.5 tends to uniform.
    alphas = [0.5, 0.55, 0.6, 0.65, 0.7, 0.8, 0.9, 1.0]

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        # pre‑compute a few starting shapes
        for alpha in alphas:
            if time.time() >= deadline:
                break
            # initialise
            w = _gen_beta_shape(n, alpha, eps=1e-12)
            a0 = w * (target_sum / w.sum())
            # quick evaluate
            cur_score = evaluate_sequence(a0.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a0.copy()
                print(f"[init] n={n} α={alpha:.2f} score={cur_score:.6f}")

            # run a few Adam steps (limited iteration count)
            iter_deadline = deadline - 0.1  # reserve a bit of time for local search
            a_opt, score_opt = _adam_opt(a0, target_sum, max_iter=200_000, deadline=iter_deadline)
            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                print(f"[adam] n={n} α={alpha:.2f} score={score_opt:.6f}")

            # local search refinement
            if time.time() < deadline:
                local_deadline = deadline - 0.05
                a_refined, score_refined = _local_search(
                    a_opt, target_sum, max_proposals=50_000, deadline=local_deadline
                )
                if score_refined < best_score:
                    best_score = score_refined
                    best_seq = a_refined.copy()
                    print(f"[local] n={n} α={alpha:.2f} score={score_refined:.6f}")

            # early exit if we've already passed the target threshold
            if best_score <= 1.503:
                print("[target reached] exiting early")
                break

    # fallback guard
    if best_seq is None:
        n = base_len
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()

import time
import math
import numpy as np

def _conv_max_fft(a: np.ndarray):
    """Return max value of full autocorrelation a*a and the full convolution array."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    max_val = float(conv.max())
    return max_val, conv

def _project_simplex(v: np.ndarray, s: float):
    """Euclidean projection of v onto the simplex {x>=0, sum x = s}."""
    # based on Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _beta_shape(n: int, alpha: float, eps: float = 1e-12):
    """Return an (unnormalised) Beta(alpha,alpha) density sampled on n points."""
    xs = (np.arange(n) + 0.5) / n  # mid‑point rule
    w = np.power(np.maximum(xs * (1.0 - xs), eps), alpha - 1.0)
    return w

def _directed_move(a: np.ndarray, target_sum: float, M: float, conv: np.ndarray):
    """
    Perform a small deterministic transfer of mass from a high‑contributing
    index (with respect to the currently maximal shift) to a low‑weight index.
    Returns a new vector and a flag indicating whether the move improved the
    maximal convolution.
    """
    n = a.shape[0]
    # shift with maximal convolution
    k = int(conv.argmax())
    # indices that actually participate in this shift
    i_low = max(0, k - (n - 1))
    i_high = min(k, n - 1)
    idx = np.arange(i_low, i_high + 1)
    partner = k - idx
    contrib = a[idx] * a[partner]
    donor = idx[np.argmax(contrib)]
    # choose a receiver that currently carries little mass
    receiver = np.argmin(a)
    if donor == receiver:
        return a, False
    delta = a[donor] * 0.05  # move up to 5 % of the donor's mass
    if delta <= 0.0:
        return a, False
    new_a = a.copy()
    new_a[donor] -= delta
    new_a[receiver] += delta
    new_a = _project_simplex(new_a, target_sum)
    M_new, _ = _conv_max_fft(new_a)
    if M_new < M:
        return new_a, True
    else:
        return a, False

def _random_move(a: np.ndarray, target_sum: float):
    """Randomly transfer a small amount of mass; accepts only improvements."""
    n = a.shape[0]
    i = np.random.randint(n)
    if a[i] <= 1e-12:
        return a, False
    j = np.random.randint(n - 1)
    if j >= i:
        j += 1
    delta = a[i] * np.random.rand() * 0.05
    if delta <= 0.0:
        return a, False
    new_a = a.copy()
    new_a[i] -= delta
    new_a[j] += delta
    new_a = _project_simplex(new_a, target_sum)
    M_new, _ = _conv_max_fft(new_a)
    M_cur, _ = _conv_max_fft(a)
    if M_new < M_cur:
        return new_a, True
    else:
        return a, False

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Strategy:
      * Work with a single length (the known 2100‑item construction) – this keeps
        the FFT cost modest and gives plenty of time for fine‑grained search.
      * Initialise from a Beta(α,α) family (α from 0.4 to 1.0).  α=0.5 corresponds
        to the continuous arcsine distribution that is known to flatten the
        self‑convolution.
      * Keep the best of those simple candidates.
      * Refine the best candidate with a mixed hill‑climb that alternates
        *directed* mass transfers (targeting the currently maximal convolution
        shift) with *random* transfers, always accepting only improvements.
      * The refinement runs until the wall‑clock budget is exhausted.
    The function returns the best sequence found before the deadline.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # spare margin for safety

    # ------------------------------------------------------------------
    n = 2100                                   # size of the search space
    target_sum = math.sqrt(2.0 * n)

    # Grid of shape parameters – 0.5 is the arcsine law; we also explore
    # slightly flatter and slightly more peaked shapes.
    alphas = [0.4, 0.45, 0.5, 0.55, 0.6, 0.65, 0.7, 0.8, 0.9, 1.0]

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1.  Scan the simple Beta‑family
    for alpha in alphas:
        if time.time() >= deadline:
            break
        w = _beta_shape(n, alpha, eps=1e-12)
        a = w * (target_sum / w.sum())
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[init] α={alpha:.2f} score={score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Mixed hill‑climb refinement
    if best_seq is None:
        # fallback – uniform vector
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform start")

    a = best_seq.copy()
    it = 0
    last_jitter = time.time()
    jitter_interval = 2000          # introduce a tiny perturbation every few thousand steps

    while time.time() < deadline:
        it += 1
        M, conv = _conv_max_fft(a)

        # occasionally try a directed move; otherwise try a random move
        if np.random.rand() < 0.6:
            a_new, improved = _directed_move(a, target_sum, M, conv)
        else:
            a_new, improved = _random_move(a, target_sum)

        if improved:
            a = a_new
            best_score = M
            print(f"[improve] iter {it:6d} new best {best_score:.6f}")

        # jitter to escape possible plateaus
        if time.time() - last_jitter > jitter_interval * 1e-4:
            jitter = np.random.randn(n) * 1e-4 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a, target_sum)
            last_jitter = time.time()

        # early exit if we have already beaten the target
        if best_score <= 1.503:
            print("[target reached] exiting early")
            break

    # ------------------------------------------------------------------
    print(f"[done] best score = {best_score:.6f}, length = {n}")
    return a.tolist()

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """
    Full autocorrelation a*a via FFT and returns the maximal value and the
    whole convolution array.
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return float(conv.max()), conv


# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, s: float):
    """
    Euclidean projection of v onto the simplex {x ≥ 0, sum x = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_shape(n: int, alpha: float, eps: float = 1e-12):
    """
    Un‑normalised discretisation of the symmetric Beta(α,α) density:
        w_i ∝ (x_i (1‑x_i))^{α‑1},
    where x_i = (i+0.5)/n ∈ (0,1).
    """
    xs = (np.arange(n) + 0.5) / n
    w = np.power(np.maximum(xs * (1.0 - xs), eps), alpha - 1.0)
    return w


# ----------------------------------------------------------------------
def _directed_move(a: np.ndarray, target_sum: float,
                   cur_max: float, conv: np.ndarray):
    """
    Transfer a small amount of mass from a position that *most* contributes
    to the currently maximal shift to the globally smallest entry.
    Returns the new vector and a flag indicating an improvement.
    """
    n = a.shape[0]
    k = int(conv.argmax())               # shift with maximal autocorrelation
    i_low = max(0, k - (n - 1))
    i_high = min(k, n - 1)
    idx = np.arange(i_low, i_high + 1)
    partner = k - idx
    contrib = a[idx] * a[partner]        # contribution of each pair to conv_k
    donor = idx[np.argmax(contrib)]      # index that hurts the most
    receiver = np.argmin(a)              # a point that holds little mass

    if donor == receiver:
        return a, False

    delta = a[donor] * 0.05               # move up to 5 % of donor's mass
    if delta <= 0.0:
        return a, False

    new_a = a.copy()
    new_a[donor] -= delta
    new_a[receiver] += delta
    new_a = _project_simplex(new_a, target_sum)

    new_max, _ = _conv_max_fft(new_a)
    if new_max < cur_max:
        return new_a, True
    return a, False


# ----------------------------------------------------------------------
def _random_move(a: np.ndarray, target_sum: float):
    """
    Very cheap random transfer of a small amount of mass.
    Accepted only if the maximal autocorrelation value decreases.
    """
    n = a.shape[0]
    i = np.random.randint(n)
    if a[i] <= 1e-12:
        return a, False
    j = np.random.randint(n - 1)
    if j >= i:
        j += 1
    delta = a[i] * np.random.rand() * 0.05
    if delta <= 0.0:
        return a, False

    new_a = a.copy()
    new_a[i] -= delta
    new_a[j] += delta
    new_a = _project_simplex(new_a, target_sum)

    new_max, _ = _conv_max_fft(new_a)
    cur_max, _ = _conv_max_fft(a)
    if new_max < cur_max:
        return new_a, True
    return a, False

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    The algorithm works on a single, moderately sized length (2100) and
    proceeds in three stages:

    1.  Initialise from a family of symmetric Beta(α,α) densities
        (α spans from 0.4 to 1.0, α = 0.5 is the continuous arcsine law).
        The best of these simple candidates is kept.

    2.  Refine the selected candidate with a mixed hill‑climb:
        – with probability 0.6 a *directed* move is attempted – it transfers
          mass from the index that contributes most to the current maximal
          convolution shift to the global minimum.
        – otherwise a pure *random* move is tried.
        Only moves that strictly decrease the maximal convolution value are
        accepted.  Periodically a tiny Gaussian jitter is added to help the
        search escape flat plateaus.

    3.  The loop runs until the allotted wall‑clock budget expires (a safety
        margin of 0.5 s is reserved).  The best sequence found is returned as a
        plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Fixed problem size – large enough to be expressive but still cheap
    n = 2100
    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------

    best_seq = None
    best_score = float('inf')
    for alpha in alphas:
        if time.time() >= deadline:
            break

        score = evaluate_sequence(a.tolist())

        print("[fallback] using uniform start")

    # --------------------------------------------------------------
    # 2️⃣  Mixed hill‑climb refinement
    a = best_seq.copy()
    it = 0
    last_jitter = time.time()
    while time.time() < deadline:
        it += 1

        # directed move with probability 0.6
        if np.random.rand() < 0.6:

        else:
            a_new, improved = _random_move(a, target_sum)

        if improved:

            cur_max, _ = _conv_max_fft(a)   # recompute after a successful move
            best_score = cur_max
            print(f"[improve] iter {it:6d} new best {best_score:.6f}")

        # tiny jitter every few thousand iterations to shake out of plateaus
        if time.time() - last_jitter > 0.3:
            jitter = np.random.randn(n) * 1e-4 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = _project_simplex(a, target_sum)
            last_jitter = time.time()

        # early stop if we have already passed the target
        if best_score <= 1.503:
            print("[target reached] exiting early")
            break

    # --------------------------------------------------------------
    print(f"[done] best score = {best_score:.6f}, length = {n}")
    return a.tolist()
