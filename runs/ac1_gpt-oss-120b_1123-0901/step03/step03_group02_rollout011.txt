"""
Projected Adam optimizer with a soft‑max proxy for the L∞ auto‑correlation.
The search explores many random lengths (≈1 k–5 k) and many random
initialisations (power‑law, exponential tails, Dirichlet samples).  Each
run performs a few seconds of Adam‑style gradient descent on the
smooth‑max of the self‑convolution, projects the iterate onto the
non‑negative simplex Σ a_i = √(2 n) and keeps the best sequence found.
Occasional random “shakes” help escape plateaus.  The routine respects
the given time budget (default 1000 s) and returns the best sequence as
a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Σ x_i = s,  x_i ≥ 0
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1, dtype=v.dtype)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power‑of‑2
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate possible tiny negatives introduced by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max weighted sum Σ w_k (a*a)_k
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute  ∇_a Σ_k w_k (a*a)_k  = 2 * a ⊛ w,
    where ⊛ denotes cross‑correlation.  Implemented via FFT.
    """
    n = a.shape[0]
    # correlation a with reversed weights
    w_rev = w[::-1]
    conv2 = _full_convolution_fft(a, w_rev)        # length = n + len(w) - 1 = 3n‑2
    idx = (2 * n - 2) - np.arange(n)               # positions giving Σ_j a_j w_{i+j}
    grad = 2.0 * conv2[idx]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser for a single length (projected onto the simplex)
# ----------------------------------------------------------------------
def _adam_optimize_one_length(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 0.001,
    step_factor: float = 0.6,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    shake_prob: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of the soft‑max of the self‑convolution.
    Returns the best sequence (as an ndarray) and its objective value
    (the maximal convolution entry, which equals the value returned by
    ``evaluate_sequence`` when Σ a_i = √(2 n)).
    """
    target_sum = np.sqrt(2.0 * n)                     # Σ a_i = √(2 n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # keep track of the best point seen so far
    conv = _full_convolution_fft(a, a)
    best_score = conv.max()
    best_seq = a.copy()

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0

    while time.time() < run_deadline:
        # ---- 1) current convolution & smooth‑max weights -----------------
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()

        tau = max(init_tau_factor * max_val * (0.95 ** it),
                  min_tau_factor * max_val)

        # soft‑max centred at the maximum, normalised to a probability vector
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ---- 2) gradient of the weighted sum ---------------------------
        grad = _gradient_from_weights(a, w)

        # ---- 3) Adam update -------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Scale the step so that it is roughly a fraction of max_val
        max_grad = np.max(np.abs(grad)) + 1e-12
        dyn_lr = step_factor * max_val / max_grad

        step = -dyn_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a + step

        # ---- 4) enforce constraints ------------------------------------
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # ---- 5) evaluate ------------------------------------------------
        conv_new = _full_convolution_fft(a_new, a_new)
        score_new = conv_new.max()
        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {best_score:.6f}")

        a = a_new

        # occasional random shake‑up to leave flat plateaus
        if np.random.rand() < shake_prob:
            noise = np.random.randn(n) * (target_sum * 0.001 / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            m.fill(0.0)
            v.fill(0.0)

    # final evaluation via the provided evaluator (ensures identical rounding)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score

        # ----- initialise a candidate shape ----------------------------
        target_sum = np.sqrt(2.0 * n)

        # power‑law base (exponent in [0.3, 0.7])
        p = np.random.uniform(0.3, 0.7)
        idx = np.arange(1, n + 1, dtype=float)
        init = idx ** (-p)

        # occasionally apply an exponential taper
        if np.random.rand() < 0.35:
            tau_exp = np.random.uniform(0.5, 2.5) * n
            init *= np.exp(-idx / tau_exp)

        init += np.random.rand(n) * 1e-4            # break perfect symmetry

        # warm‑start from the current global best with some probability
        if best_seq is not None and np.random.rand() < 0.45:
            bs = best_seq
            if bs.size >= n:
                init = bs[:n].copy()
            else:
                pad = np.random.rand(n - bs.size) * 1e-4
                init = np.concatenate([bs, pad])

        # scale to the required sum (the optimiser will project again)
        init = init * (target_sum / np.maximum(init.sum(), 1e-12))

        # allocate a time slice for this optimisation run
        slice_time = min(8.0, remaining * 0.5)
        run_deadline = min(deadline, time.time() + slice_time)

        # ---------------------------------------------------------------
        # optimise this length
        # ---------------------------------------------------------------
        seq_local, score_local = _adam_optimize_one_length(
            n,
            init,
            run_deadline,
            init_tau_factor=0.12,
            min_tau_factor=0.001,
            step_factor=0.6,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            shake_prob=0.02,
        )

        # update global best if improved
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (length={n})"
            )
            # early exit if we have already passed a very good bound
            if best_score < 1.49:
                break

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that no candidate was generated.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1200
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # The evaluator expects a plain Python list.
    return best_seq.tolist()
