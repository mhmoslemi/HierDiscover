def differential_evolution_optimize(n, target_sum, max_time, start_time, init_seq=None):
    popsize = 15
    F = 0.8
    Cr = 0.9
    # init population
    population = []
    fitness = []
    # include init_seq optionally if provided and size n
    for i in range(popsize):
        if init_seq is not None and i==0:
            a = init_seq.copy()
        else:
            # random Dirichlet by exponential
            a_raw = np.random.exponential(1.0, size=n)
            a = a_raw / a_raw.sum() * target_sum
        population.append(a)
        fitness.append(_max_conv(a))
    # find best
    best_idx = int(np.argmin(fitness))
    best_vec = population[best_idx].copy()
    best_f = fitness[best_idx]
    # DE loop
    while time.time() < start_time + max_time:
        # For each individual
        for i in range(popsize):
            # Choose r1, r2, r3 distinct and not i
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = np.random.choice(idxs, 3, replace=False)
            a = population[r1]
            b = population[r2]
            c = population[r3]
            mutant = a + F * (b - c)
            # ensure positivity before projection
            # Crossover
            cross = np.random.rand(n) < Cr
            # ensure at least one True to avoid empty
            if not np.any(cross):
                cross[np.random.randint(0,n)] = True
            trial = np.where(cross, mutant, population[i])
            # projection onto simplex
            trial = _project_onto_simplex(trial, target_sum)
            # evaluate
            f_trial = _max_conv(trial)
            # selection
            if f_trial <= fitness[i]:
                population[i] = trial
                fitness[i] = f_trial
                # update best
                if f_trial < best_f:
                    best_f = f_trial
                    best_vec = trial.copy()
        # optional adapt F, Cr: may not needed
    return best_vec, best_f

    n = len(a)
    best = a.copy()
    best_val = _max_conv(a)

    # parameters
    alpha = 1.0
    # We'll increase alpha gradually to focus more on true max, e.g., after each iteration multiply by 1.01 up to maybe 50.
    # step size schedule: start with gamma = 0.5 * target_sum / n (some typical size), then adapt.
    step = 0.5 * target_sum / n  # initial step

    while time.time() < start_time + max_time:
        conv = _conv_full_fft(a)
        max_val = conv.max()
        # compute smooth max gradient
        # adjust alpha based on progress
        if alpha < 20.0:
            alpha *= 1.01

        # compute softmax weights
        max_shift = conv.max()
        exp_vals = np.exp(alpha * (conv - max_shift))
        w = exp_vals / exp_vals.sum()

        # compute gradient 
        # correlation via convolution
        conv_rev = np.convolve(a, w[::-1])
        grad = 2.0 * conv_rev[n-1:2*n-1][::-1]

        # Normalization of gradient step
        # Use step size proportional to max_val / np.linalg.norm(grad)
        grad_norm = np.linalg.norm(grad) + 1e-12
        eta = step * max_val / grad_norm

        a_new = a - eta * grad
        # enforce nonnegativity
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # evaluate actual max for tracking
        new_score = _max_conv(a_new)

        if new_score < best_val:
            best_val = new_score
            best = a_new.copy()
            # print progress
            print(f"[{time.time():.2f}] SmoothGD new best = {best_val:.6f} (n={n})")

        a = a_new

        # optional: if no improvement after many steps, break
    return best, best_val

"""
propose_candidate
~~~~~~~~~~~~~~~~~

A hybrid optimizer for the self‑convolution minimisation problem.

The search consists of many restarts.  For each restart we:
  * pick a length n (roughly 1500‑3500),
  * generate a diverse seed (random Dirichlet, U‑shape, exponential,
    or a previously found good construction),
  * run a *smooth‑max* gradient descent that minimises a softmax
    approximation of the maximum convolution entry.  The smoothing
    parameter α is increased gradually (continuation method), turning the
    problem from smooth to the true ℓ∞‑objective.
  * polish the result with the original projected sub‑gradient optimiser
    on the exact active (max‑tying) convolution entries.

Both stages keep the ℓ1‑norm fixed (Σa = √(2n)) by projecting onto the
simplex after every update.  The code is fully vectorised, uses FFTs for
convolutions, and respects the 1000 s time budget.  Progress is printed
so the user can see improvements.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex { x_i ≥ 0 , Σ x_i = s }.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex of radius ``s``."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full convolution (real FFT) – returns length 2*n‑1 array.
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Evaluate a vector – because Σa = √(2n) the objective reduces to max(conv).
# --------------------------------------------------------------------------- #
def _max_conv(a: np.ndarray) -> float:
    return float(_conv_full_fft(a).max())


# --------------------------------------------------------------------------- #
# Smooth‑max gradient descent (continuation on α).
# --------------------------------------------------------------------------- #
def _smooth_max_descent(a_init: np.ndarray,
                        target_sum: float,
                        max_time: float,
                        start_time: float):
    """
    Minimise a smoothed version of the max‑convolution using gradient
    descent with projection onto the simplex.
    """
    a = a_init.copy()
    n = a.shape[0]

    best_vec = a.copy()
    best_val = _max_conv(a)

    # hyper‑parameters
    alpha = 0.5                # start smoothing
    alpha_max = 50.0
    step_scale = 0.5 * target_sum / n   # base step size
    it = 0

    while time.time() < start_time + max_time:
        it += 1

        # --- convolution and smooth weights ---------------------------------
        conv = _conv_full_fft(a)                     # length 2n‑1
        max_val = conv.max()

        # increase smoothing gradually
        if alpha < alpha_max:
            alpha *= 1.015

        # soft‑max (stable)
        shift = max_val
        exp_vals = np.exp(alpha * (conv - shift))
        w = exp_vals / exp_vals.sum()                # weights, length 2n‑1

        # --- gradient of the smoothed objective ---------------------------
        # grad = 2 * correlation(a, w)
        w_rev = w[::-1]
        n_fft = 1 << ((n + (2 * n - 1) - 1).bit_length())   # ≥ 3n‑2
        A_fft = np.fft.rfft(a, n=n_fft)
        W_fft = np.fft.rfft(w_rev, n=n_fft)
        corr_full = np.fft.irfft(A_fft * W_fft, n=n_fft)[: (3 * n - 2)]
        # slice the part corresponding to shifts 0…n‑1 and reverse
        grad = 2.0 * corr_full[n - 1: 2 * n - 1][::-1]

        # --- step size ----------------------------------------------------
        grad_norm = np.linalg.norm(grad) + 1e-12
        eta = step_scale * max_val / grad_norm

        # --- update --------------------------------------------------------
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional random jitter to escape shallow minima
        if it % 100 == 0 and np.random.rand() < 0.4:
            noise = np.random.randn(n) * (0.02 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0),
                                          target_sum)

        # track best real objective (true max‑convolution)
        cur_val = _max_conv(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best_vec = a_new.copy()
            print(f"[{time.time():.2f}] Smooth‑GD new best = {best_val:.6f}"
                  f" (n={n})")

        a = a_new

        # early break if we are extremely close to the target
        if best_val < 1.503:
            break

    return best_vec, best_val


# --------------------------------------------------------------------------- #
# Projected sub‑gradient descent on the active convolution entries.
# --------------------------------------------------------------------------- #
def _subgradient_optimize(a_init: np.ndarray,
                          target_sum: float,
                          max_time: float,
                          start_time: float):
    """
    Refine a vector using a sub‑gradient method that focuses on the
    active (max‑tying) convolution entries.
    """
    a = a_init.copy()
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _max_conv(a)

    base_lr = 0.2          # learning rate
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    while time.time() < start_time + max_time:
        conv = _conv_full_fft(a)
        max_val = conv.max()
        cur_score = float(max_val)

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{time.time():.2f}] Sub‑GD new best = {best_score:.6f}"
                  f" (n={n})")

        # active indices (those that attain the max, up to tiny tolerance)
        eps = 1e-12
        active = np.where(conv >= max_val - eps)[0]

        # gradient of sum of active entries
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]

        max_grad = grad.max()
        if max_grad <= 0.0:
            break

        eta = base_lr * max_val / max_grad

        momentum = beta_mom * momentum - eta * grad
        a_new = a + momentum

        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional random shake
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0),
                                          target_sum)
            base_lr *= 0.9

        a = a_new
        base_lr *= 0.99995

    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Main entry point expected by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    best_seq = None            # numpy array of the best candidate
    best_score = float("inf")  # objective value (max convolution)

    # ------------------------------------------------------------------- #
    # Helper to try a single restart.
    # ------------------------------------------------------------------- #
    def _run_restart():
        nonlocal best_seq, best_score

        # -------- decide length -----------------------------------------
        if best_seq is not None and np.random.rand() < 0.25:
            n = len(best_seq)
        else:
            n = np.random.randint(1500, 3501)

        target_sum = np.sqrt(2.0 * n)   # Σ a_i = √(2n)

        # -------- generate an initial seed -------------------------------
        init_type = np.random.choice(['dirichlet', 'u_shape', 'exp', 'seed'])
        if init_type == 'seed' and 'height_sequence_1' in globals():
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-3
                init = np.concatenate([src, pad])
        elif init_type == 'u_shape':
            # U‑shape with a random exponent
            p = 0.5 + np.random.rand() * 2.5   # 0.5 … 3.0
            idx = np.arange(n, dtype=float)
            weight = (1.0 / (idx + 1) + 1.0 / (n - idx)) ** p
            init = np.random.rand(n) * weight
        elif init_type == 'exp':
            beta = np.random.rand() * 5.0      # 0 … 5
            idx = np.arange(n, dtype=float)
            dist = np.minimum(idx, n - 1 - idx)
            weight = np.exp(-beta * dist)
            init = np.random.rand(n) * weight
        else:  # dirichlet / generic exponential
            init = np.random.exponential(scale=1.0, size=n)

        a0 = _project_onto_simplex(init.astype(float), target_sum)

        # -------- allocation of time for this restart -------------------
        time_left = deadline - time.time()
        if time_left <= 0:
            return
        # roughly half for smooth‑max GD, half for sub‑gradient polishing
        phase1 = min(9.0, time_left * 0.55)
        a_opt, sc_opt = _smooth_max_descent(a0, target_sum,
                                            phase1, time.time())
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Global best = {best_score:.6f} (n={n})")

        # second phase
        time_left = deadline - time.time()
        if time_left <= 0:
            return
        phase2 = min(6.0, time_left)
        a_opt2, sc_opt2 = _subgradient_optimize(a_opt, target_sum,
                                                phase2, time.time())
        if sc_opt2 < best_score:
            best_score = sc_opt2
            best_seq = a_opt2.copy()
            print(f"[{time.time():.2f}] Global best after sub‑GD = {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------- #
    # Main restart loop.
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        _run_restart()
        # early exit if we reached the target bound
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Fallback – uniform vector if nothing was found (should never happen).
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return as plain Python list
    return best_seq.tolist()
