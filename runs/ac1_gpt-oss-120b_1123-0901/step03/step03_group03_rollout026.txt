"""
Linear‑programming based improvement of a non‑negative step function.

The evaluation function equals the maximal entry of the full convolution
`a * a`.  For a fixed length `n` we iteratively improve a candidate vector `a`
as follows

1. Normalise `a` so that Σa = √(2 n) (the objective is scale‑invariant).
2. Let M = max(a∗a).  All vectors `b ≥ 0` satisfying
       (a * b)_k ≤ M   (k = 0 … 2n‑2)
   form a polytope.  Maximising Σb under these linear constraints pushes
   mass into directions that keep the convolution bounded.
3. Solve the linear programme with `scipy.optimize.linprog`
   (highs solver).  The constraint matrix is a Toeplitz matrix that
   implements the convolution `a * b`.
4. Rescale the optimiser output `b` to the required ℓ₁‑norm
   ( Σb → √(2 n) ) obtaining `g`.
5. Update `a ← (1‑t)·a + t·g` with a small mixing factor `t`.
   This convex combination preserves Σa and never increases the maximal
   convolution value (by the classic exchange argument).
6. Repeat until no improvement or the time budget expires.

We run this scheme on several lengths (original length and a few
enlargements) and from several random starts (uniform, Dirichlet,
interpolated previous best).  The best vector encountered is returned.
The implementation respects a 1000 s wall‑clock limit and prints
progress information.
"""

import time
import math
import numpy as np
import scipy.linalg
import scipy.optimize


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a∗a (numpy's O(n²) is fine for n≈3000)."""
    return np.convolve(a, a)


def _build_toeplitz(f: np.ndarray) -> np.ndarray:
    """
    Build the Toeplitz matrix T (shape (2n‑1, n)) such that
    T @ b = f * b (full convolution) for any b.
    """
    n = f.shape[0]
    first_col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    first_row = np.concatenate([f[:1], np.zeros(n - 1, dtype=f.dtype)])
    return scipy.linalg.toeplitz(first_col, first_row)


def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σb   s.t.   (f * b) ≤ M,   b ≥ 0
    via a linear programme (converted to a minimisation).
    Returns the optimal `b` (may be dense) or None on failure.
    """
    n = f.shape[0]
    # Build convolution matrix once per LP call.
    A = _build_toeplitz(f)                # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=f.dtype)

    # maximise Σb  → minimise -Σb
    c = -np.ones(n, dtype=f.dtype)
    bounds = [(0.0, None) for _ in range(n)]

    res = scipy.optimize.linprog(
        c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs", options={"presolve": True}
    )
    # explicit free of the potentially large matrix for the next call
    del A
    if res.success and res.x is not None:
        # Clip tiny negatives that can appear due to numerical noise.
        return np.maximum(res.x, 0.0)
    return None


def _lp_improve(
    a: np.ndarray,
    target_sum: float,
    deadline: float,
    max_outer_iters: int = 30,
    t: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Perform the alternating LP improvement described above.
    Returns the best vector encountered and its maximal convolution value.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_M = _full_convolution(a).max()

    it = 0
    while time.time() < deadline and it < max_outer_iters:
        M = _full_convolution(a).max()
        # Solve the LP; feasibility is guaranteed (b = a works).
        b = _solve_lp(a, M)
        if b is None:
            break

        sum_b = b.sum()
        if sum_b <= 0.0:
            break

        # Scale to the required ℓ₁ norm.
        g = b * (target_sum / sum_b)

        # Convex combination; both a and g already sum to target_sum.
        a_new = (1.0 - t) * a + t * g
        # Numerical safety.
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum > 0.0:
            a_new = a_new * (target_sum / a_new_sum)

        M_new = _full_convolution(a_new).max()
        if M_new < best_M - 1e-9:
            best_M = M_new
            best_a = a_new.copy()
            print(f"[LP iter {it:3d}]   new best M = {best_M:.6f}")

        a = a_new
        it += 1

        # Occasionally inject a tiny jitter to escape shallow plateaus.
        if it % 7 == 0:
            jitter = np.random.randn(n) * (0.005 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

    return best_a, best_M

            )

            if M_opt < best_M - 1e-9:
                best_M = M_opt
                best_seq = a_opt.copy()
                print(f"[len {n:5d} init {idx}] new global best M = {best_M:.6f}")

    # ------------------------------------------------------------------
    # Final safety: evaluate through the provided function for reporting.
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[done] best sequence length = {best_seq.shape[0]}, final score = {final_score:.6f}")

    return best_seq.tolist()
