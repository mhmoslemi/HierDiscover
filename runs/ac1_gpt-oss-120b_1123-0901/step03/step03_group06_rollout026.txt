def optimize(p_start, n, deadline):

    # set N = next_pow2(2*n-1)
    N = _next_pow2(2*n - 1)

    # Use initial logp as log(p + eps)
    logp = np.log(p_start + 1e-30)

    best_score_local = float('inf')
    best_p_local = None

    it = 0
    while time.time() < deadline:
        it += 1

        # compute p = exp(logp) normalized
        # stable: subtract max
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)
        p = exp_shift / np.sum(exp_shift)

        # compute conv via FFT
        P = np.fft.rfft(p, n=N)
        conv = np.fft.irfft(P * P, n=N)[:2*n-1]
        M = conv.max()
        cur_score = 2.0 * n * M

        if cur_score < best_score_local:
            best_score_local = cur_score
            best_p_local = p.copy()
            # print for debug
            print(f"[len {n:4d} iter {it:7d}] best local score {cur_score:.6f}")

        # gradient using max index (exact)
        k_max = np.argmax(conv)
        # compute rev using vectorized slice
        indices = k_max - np.arange(n)
        mask = (indices >= 0) & (indices < n)
        rev = np.zeros_like(p)
        rev[mask] = p[indices[mask]]
        grad = 4.0 * n * rev

        # step size
        # Using eta = 1/(4*n) / np.sqrt(it) may be too small; we set eta = 1.0/(4.0*n) / (1.0 + it * 1e-5) (slightly slower decay)

    # after loop final best_p_local maybe None if never computed; fallback to p_start
    if best_p_local is None:
        best_p_local = p_start

    return best_p_local, best_score_local

"""
Optimiser for the “self‑convolution peak” problem.

The score of a non‑negative step‑function `a` of length `n` is  

    score = 2·n·max_k (p∗p)_k ,   with  p = a / √(2 n) , Σ p = 1.

We minimise the L∞ norm of the auto‑convolution of `p`.  The algorithm
works on the probability vector `p` :

* Exponentiated‑gradient (mirror descent on the simplex) on `log(p)`.
* At each iteration we compute the full auto‑convolution with a single
  FFT, locate the index where it attains its maximum and use the exact
  sub‑gradient
        ∂max(p∗p)/∂p_i = 2·p_{k_max-i}
  (zero outside the support).  The gradient of the objective is
        ∇ = 4·n·rev .
* A simple schedule for the step‑size η and occasional Gaussian jitter are
  used to escape plateaux.
* A multiscale schedule gradually enlarges the vector length; the current
  best probability vector is interpolated to the new length before optimisation.
* The whole run respects the 1000 s budget, split automatically between the
  different lengths.  The best sequence encountered is returned.

The optimiser is deterministic apart from the random jitter (seeded for
reproducibility) and usually beats the previous 1.519159 bound within the
time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(p: np.ndarray, N: int) -> np.ndarray:
    """
    Full (linear) self‑convolution of a real vector `p` using one FFT of size `N`.
    Returns the real array of length 2·len(p)‑1.
    """
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _interpolate_prob(old_p: np.ndarray, new_n: int) -> np.ndarray:
    """
    Interpolate a probability vector `old_p` (sum = 1) to length `new_n`
    using linear interpolation on the unit interval.
    """
    old_n = old_p.shape[0]
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_p = np.interp(new_x, old_x, old_p)
    new_p = np.maximum(new_p, 0.0)
    s = new_p.sum()
    if s == 0:
        # fallback to uniform
        return np.full(new_n, 1.0 / new_n)
    return new_p / s


def _eg_optimize(p_start: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimisation for a fixed length `n`.

    Parameters
    ----------
    p_start : np.ndarray
        Initial probability vector (sum = 1, non‑negative).
    n : int
        Length of the vector.
    deadline : float
        Unix time after which the routine must return.

    Returns
    -------
    best_p : np.ndarray
        Probability vector with the lowest observed objective.
    best_score : float
        Corresponding value of 2·n·max(p∗p).
    """
    # FFT size – at least 2·n‑1, rounded up to a power of two for speed.
    N = _next_pow2(2 * n - 1)

    # work in log‑space to stay on the simplex automatically
    logp = np.log(p_start + 1e-30)

    best_score_local = float("inf")
    best_p_local = None

    it = 0
    # main loop – runs until the per‑length deadline is reached
    while time.time() < deadline:
        it += 1

        # ----- project back onto the simplex -----
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)          # avoid overflow
        p = exp_shift / exp_shift.sum()              # p is now a probability vector

        # ----- self‑convolution -----
        conv = _full_convolution(p, N)
        M = conv.max()
        cur_score = 2.0 * n * M

        if cur_score < best_score_local:
            best_score_local = cur_score
            best_p_local = p.copy()
            print(f"[len {n:5d} iter {it:7d}] new local best = {cur_score:.6f}")

        # ----- exact sub‑gradient of the max -----
        k_max = int(conv.argmax())
        # rev_i = p_{k_max-i} when the index is inside the array, otherwise 0
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        rev = np.zeros_like(p)
        rev[mask] = p[idx[mask]]
        grad = 4.0 * n * rev  # ∇ (2·n·max) = 4·n·rev

        # ----- step‑size schedule -----
        # start with η≈1/(4n) and decay like 1/√t
        eta = (1.0 / (4.0 * n)) / (np.sqrt(it))

        # ----- EG update (log‑space) -----
        logp -= eta * grad

        # occasional jitter to help escape plateaus
        if it % 4000 == 0:
            logp += np.random.randn(n) * 0.001

        # safety cap – prevents pathological endless loops
        if it >= 800_000:
            break

    # Fallback in the extremely unlikely case that nothing was recorded
    if best_p_local is None:
        best_p_local = p_start.copy()
        best_score_local = 2.0 * n * _full_convolution(p_start, N).max()

    return best_p_local, best_score_local

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the 1000 s budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness (a few seconds)
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # initialise from the previously discovered construction if it exists
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        # sensible default – uniform vector of moderate size
        base = np.full(1200, 1.0, dtype=float)

    n0 = base.shape[0]
    S0 = math.sqrt(2.0 * n0)
    base = base * (S0 / base.sum())                 # enforce Σa = √(2 n)
    p0 = base / S0                                 # probability vector (Σp = 1)

    # Global best tracking
    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n0}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # multiscale schedule – we focus on the range where the previous result lived
    # (≈2100) but also explore a bit below and above.
    length_schedule = [800, 1200, 1600, 2000, 2400, 2800, 3200, 3600, 4000, 5000]
    # keep only lengths that are reasonable given the remaining time
    length_schedule = [l for l in length_schedule if 300 <= l <= 8000]
    length_schedule = sorted(set(length_schedule))

    remaining_lengths = len(length_schedule)

    for idx, n in enumerate(length_schedule):
        if time.time() >= deadline:
            break

        # allocate a fair slice of the remaining budget to the current length
        remaining = deadline - time.time()
        slices_left = remaining_lengths - idx
        per_len_deadline = time.time() + remaining / slices_left * 0.95  # a tiny safety margin

        # ----- initialise probability vector for this length -----
        if best_seq.shape[0] == n:
            p_init = best_seq / math.sqrt(2.0 * n)   # because Σa = √(2 n)
        else:
            # interpolate the current global best (probability) to the new length
            p_old = best_seq / math.sqrt(2.0 * best_seq.shape[0])
            p_init = _interpolate_prob(p_old, n)

        print(f"[run] start optimisation for length = {n}")

        # ----- run EG optimisation for this length -----
        p_opt, local_score = _eg_optimize(
            p_start=p_init,
            n=n,
            deadline=per_len_deadline,
        )

        # Convert back to the original scale (a = √(2 n)·p)
        S = math.sqrt(2.0 * n)
        a_opt = p_opt * S
        cur_score = evaluate_sequence(a_opt.tolist())

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_opt.copy()
            print(f"[len {n}] improved global best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # final polishing on the best length (if time permits)
    if time.time() < deadline:
        n_best = best_seq.shape[0]
        remaining = deadline - time.time()
        # give the last polish up to half of the remaining budget
        polish_deadline = time.time() + remaining * 0.5

        print(f"[polish] final refinement on length = {n_best}")
        p_start = best_seq / math.sqrt(2.0 * n_best)
        p_opt, _ = _eg_optimize(p_start, n_best, polish_deadline)
        a_opt = p_opt * math.sqrt(2.0 * n_best)
        cur_score = evaluate_sequence(a_opt.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_opt.copy()
            print(f"[polish] improved global best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # safety fallback – should never trigger
    if best_seq is None or best_seq.size == 0:
        n = 800
        S = math.sqrt(2.0 * n)
        best_seq = np.full(n, S / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
