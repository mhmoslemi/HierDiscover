        w = np.exp((conv - max_val) / tau)

"""
Propose a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm mixes three ideas:

1.  Wide‑range random initialisations (power‑law, sinusoidal, Dirichlet …).
2.  A soft‑max (log‑sum‑exp) surrogate of the max‑convolution together with
    an Adam‑style optimiser.  The vector is projected onto the simplex
    Σa_i = √(2 n) after every update, guaranteeing that the objective equals
    the maximal self‑convolution value.
3.  A brief simulated‑annealing (SA) phase that moves a small amount of mass
    between two coordinates; this can escape the basin found by gradient
    descent.

The search runs for at most ``budget_s`` seconds (default 1000 s), keeping
track of the best sequence discovered across many independent restarts and
different lengths.  The final candidate is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (non‑negative, sum = s)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – Full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power‑of‑2
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative artefacts
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Random initialisation strategies (return a raw positive vector)
# ----------------------------------------------------------------------
def _random_initialisation(n: int) -> np.ndarray:
    """Return a random positive vector of length n (not yet normalised)."""
    r = np.random.rand()
    if r < 0.35:
        # Dirichlet / Gamma – essentially a random point on the simplex
        x = np.random.gamma(shape=1.0, scale=1.0, size=n)
    elif r < 0.65:
        # Power‑law tail a_i ∝ (i+1)^{-p}
        p = np.random.uniform(0.2, 1.5)
        idx = np.arange(1, n + 1, dtype=float)
        x = idx ** (-p)
        if np.random.rand() < 0.4:          # optional exponential taper
            tau = np.random.uniform(0.4, 2.0) * n
            x *= np.exp(-idx / tau)
    else:
        # Half‑sine (or cosine) window, optionally raised to a power
        q = np.random.uniform(0.5, 2.0)
        idx = np.arange(1, n + 1, dtype=float)
        x = np.sin(np.pi * idx / (n + 1)) ** q
    # tiny jitter to break perfect symmetry
    x += np.random.rand(n) * 1e-6
    return x


# ----------------------------------------------------------------------
# Soft‑max surrogate optimiser using Adam (gradient descent on a smooth max)
# ----------------------------------------------------------------------
def _optimise_one_length_adam(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 0.001,
    alpha: float = 0.6,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of a fixed length ``n``.
    Returns the best sequence found (numpy array) and the exact evaluation
    (via ``evaluate_sequence``) of that sequence.
    """
    target_sum = np.sqrt(2.0 * n)          # Σ a_i = √(2 n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    best_seq = a.copy()
    best_score = 2.0 * n * _full_convolution_fft(a, a).max() / (target_sum ** 2)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    iteration = 1
    while time.time() < run_deadline:
        # ----- self‑convolution and smooth‑max weights -----
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        tau = max(init_tau_factor * max_val * (0.95 ** iteration),
                  min_tau_factor * max_val)

        w = np.exp((conv - max_val) / tau)    # all entries ≤ 1
        w_sum = w.sum()
        if w_sum > 0.0:
            w /= w_sum
        else:                                 # degenerate case – uniform
            w.fill(1.0 / w.size)

        # ----- gradient of Σ w_k conv_k -----
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)   # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)           # positions for ∑ a_j w_{i+j}
        grad = 2.0 * conv2[idx]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # ----- keep the best local candidate -----
        conv_new = _full_convolution_fft(a_new, a_new)
        score_new = 2.0 * n * conv_new.max() / (target_sum ** 2)

        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] n={n} new local best = {best_score:.6f}")

        a = a_new
        iteration += 1

        # occasional random jitter to escape plateaus
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (target_sum * 5e-4 / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            m.fill(0.0)
            v.fill(0.0)

    # final exact evaluation (identical to the external evaluator)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Very cheap simulated‑annealing refinement (moves a small amount of mass)
# ----------------------------------------------------------------------
def _simulated_annealing(
    a_start: np.ndarray,
    run_deadline: float,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    """
    Simple SA that repeatedly moves a tiny mass from one coordinate to another.
    Returns the best sequence discovered and its exact evaluation.
    """
    a = a_start.copy()
    cur_score = _full_convolution_fft(a, a).max()
    best_seq = a.copy()
    best_score = cur_score

    # initial temperature loosely based on current objective
    T0 = max(cur_score * 0.3, 1e-3)
    it = 0

    n = a.shape[0]

    while time.time() < run_deadline:
        # exponential cooling
        T = T0 * np.exp(-0.0005 * it)

        # pick two distinct indices
        i, j = np.random.choice(n, 2, replace=False)
        if a[i] <= 0.0:
            it += 1
            continue

        # move a small fraction of mass from i to j
        delta = a[i] * np.random.rand() * 0.08       # ≤ 8 % of a_i
        a[i] -= delta
        a[j] += delta

        # keep the sum unchanged – no projection required

        # evaluate
        new_score = _full_convolution_fft(a, a).max()
        dE = new_score - cur_score

        # Metropolis acceptance
        accept = False
        if dE <= 0.0:
            accept = True
        else:
            if T > 0 and np.random.rand() < np.exp(-dE / T):
                accept = True

        if accept:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] SA new best = {best_score:.6f}")
        else:
            # revert the move
            a[i] += delta
            a[j] -= delta

        it += 1

    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the given evaluation
    function.  The routine respects the ``budget_s`` second limit and returns
    the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin to avoid overrunning the timeout
    deadline = start_time + budget_s - 0.2

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optional warm‑start from a previously discovered construction
    # ------------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        score = evaluate_sequence(hs_arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start from global seed = {score:.6f}")
    except Exception:
        pass  # no warm‑start available

    # ------------------------------------------------------------------
    # Main search loop – many independent restarts
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 6.0:                     # not enough time for a fresh run
            break

        # ----- pick a problem size -------------------------------------------------
        if remaining > 300:
            n = np.random.randint(1500, 5000 + 1)
        elif remaining > 120:
            n = np.random.randint(1000, 3800 + 1)
        else:
            n = np.random.randint(600, 2500 + 1)

        # ----- initialise a random candidate ---------------------------------------
        init = _random_initialisation(n)

        # ----- sometimes seed from the current global best -------------------------
        if best_seq is not None and np.random.rand() < 0.35:
            if best_seq.size >= n:
                init = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.size) * 1e-6
                init = np.concatenate([best_seq, pad])

        # ----- allocate a slice of the remaining budget for Adam optimisation -------
        slice_time = min(remaining * 0.4, 12.0)           # at most ~12 s per run
        run_deadline = min(deadline, time.time() + slice_time)

        seq_local, score_local = _optimise_one_length_adam(
            n,
            init,
            run_deadline,
            init_tau_factor=0.12,
            min_tau_factor=0.001,
            alpha=0.6,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
        )

        # ----- record improvement ---------------------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] GLOBAL new best after Adam = {best_score:.6f} (n={n})")
            if best_score < 1.49:       # comfortable margin below the target bound
                break

        # ----- optional SA refinement on the same length ----------------------------
        remaining_after_adam = deadline - time.time()
        if remaining_after_adam > 5.0:
            sa_deadline = min(deadline, time.time() + min(6.0, remaining_after_adam * 0.5))
            seq_sa, score_sa = _simulated_annealing(seq_local, sa_deadline,
                                                    target_sum=np.sqrt(2.0 * n))
            if score_sa < best_score:
                best_score = score_sa
                best_seq = seq_sa.copy()
                print(f"[{time.time():.2f}] GLOBAL new best after SA = {best_score:.6f} (n={n})")
                if best_score < 1.49:
                    break

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that no candidate was generated.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = np.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.2f}] Fallback uniform candidate, score = {best_score:.6f}")

    # The evaluator expects a plain Python list.
    return best_seq.tolist()
