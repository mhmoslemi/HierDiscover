def solve_lp(f, M):
    n = len(f)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # For j = k - i
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])

    if res.success:
        g = res.x
        return g
    else:
        return None

from scipy.optimize import linprog

def solve_lp_for_f(f, M):
    n = len(f)
    # Build A_ub (2n-1) x n
    rows = np.empty(n*n, dtype=np.int32)
    cols = np.empty(n*n, dtype=np.int32)
    data = np.empty(n*n, dtype=np.float64)
    idx = 0
    for i in range(n):
        rows[idx: idx+n] = i + np.arange(n)
        cols[idx: idx+n] = np.arange(n)

def _solve_lp_for_f(f, M):
    n = len(f)
    # Build A as described
    # Make row and col arrays using np.arange
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)  # shape (n*n,)
    cols = np.tile(np.arange(n), n)  # each block of n entries for each i
    data = np.repeat(f, n)  # each f[i] repeated n times
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # constraints
    b = np.full(2*n-1, M, dtype=np.float64)
    # objective - maximize sum(g) -> minimize -sum(g)
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None
    return res.x

"""
Non‑negative step‑function optimiser.
------------------------------------
The algorithm is a hybrid of three ideas:

1. **Soft‑max (smooth‑max) gradient descent** – the same fast sub‑gradient
   method used in the reference solution, but with a slightly slower
   temperature decay and a longer per‑restart budget.  This quickly finds
   a good local shape for a given length `n`.

2. **Geometric‑shape warm‑starts** – before the gradient descent we
   generate a few simple exponential (geometric) sequences and pick the
   one with the smallest raw convolution maximum.  These provide
   diverse starting points that are often closer to a good optimum than
   a plain power‑law.

3. **Linear‑programming polishing** – after the gradient phase we solve
   a small LP:

        maximise   Σ g_j
        subject to g_j ≥ 0,
                   (f * g)_k ≤ M   for all k,

   where `M = max(f * f)`.  The LP is linear because `f` is fixed.
   The optimum `g` is mixed back into `f` (`f←(1‑t)f+t g`) and the
   process is repeated a few times.  This step implements the
   “alternating LP” method described in the statement and often pushes
   the objective a little further.

All helpers are top‑level so they survive the sandbox restart.
The main entry point is `propose_candidate`, which respects the 1000 s
budget, prints progress, and finally returns the best sequence found
as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy import sparse as sp

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex of radius s (non‑negative, sum = s)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – Full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power‑of‑2
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # remove tiny negative artefacts caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Soft‑max sub‑gradient descent for a fixed length
# ----------------------------------------------------------------------
def _optimise_one_length_softmax(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 0.001,
    step_factor: float = 0.6,
    step_decay: float = 0.9993,
    momentum_beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent (smooth‑max) for a fixed length `n`.
    Returns the best sequence (NumPy array) and its evaluator score.
    """
    target_sum = np.sqrt(2.0 * n)          # Σ a_i = √(2 n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # keep the best seen so far
    conv = _full_convolution_fft(a, a)
    best_score = 2.0 * n * conv.max() / (target_sum ** 2)   # equals conv.max()
    best_seq = a.copy()

    velocity = np.zeros_like(a)
    iteration = 0

    while time.time() < run_deadline:
        # ---- self‑convolution -------------------------------------------------
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()

        # ---- smooth‑max weights -----------------------------------------------
        tau = max(init_tau_factor * max_val * (0.95 ** iteration),
                  min_tau_factor * max_val)

        scaled = (conv - max_val) / tau
        w = np.exp(scaled)                 # all entries ≤ 1
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of Σ w_k conv_k (cross‑correlation) ----------------------
        w_rev = w[::-1]
        corr = _full_convolution_fft(a, w_rev)   # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)          # picks the needed slice
        grad = 2.0 * corr[idx]

        # ---- momentum & adaptive step -----------------------------------------
        if momentum_beta > 0.0:
            velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
            step_dir = velocity
        else:
            step_dir = grad

        max_grad = np.max(np.abs(step_dir)) + 1e-12
        eta = step_factor * max_val / max_grad
        a_new = a - eta * step_dir
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # ---- evaluate ---------------------------------------------------------
        conv_new = _full_convolution_fft(a_new, a_new)
        score_new = 2.0 * n * conv_new.max() / (target_sum ** 2)

        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {score_new:.6f}")

        a = a_new
        step_factor *= step_decay
        iteration += 1

        # occasional shake‑up to escape plateaus
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (target_sum * 5e-4 / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            step_factor *= 0.9

    # final evaluation using the provided evaluator (ensures identical rounding)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “LP polishing” step)
# ----------------------------------------------------------------------
def _solve_lp_for_f(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve:
        maximise   Σ g_j
        subject to g_j ≥ 0,
                   (f * g)_k ≤ M   for all k.
    Returns the optimal g (not yet rescaled) or None on failure.
    """
    n = f.shape[0]
    # Build sparse matrix A where A[k, j] = f[k‑j] if 0≤k‑j<n else 0
    i = np.arange(n).repeat(n)                # each i repeated n times
    j = np.tile(np.arange(n), n)              # each block of n columns
    rows = i + j                               # k = i + j
    cols = j
    data = np.repeat(f, n)                    # f[i] repeated n times
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    b = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n
    # maximise Σ g → minimise −Σ g
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x


def _lp_refine(seq: np.ndarray, target_sum: float,
               lp_iters: int = 3, mix_start: float = 0.5) -> np.ndarray:
    """
    Apply a few LP polishing steps to `seq`.  Returns a refined sequence.
    """
    f = seq.copy()
    for it in range(lp_iters):
        conv = _full_convolution_fft(f, f)
        M = conv.max()
        g = _solve_lp_for_f(f, M)
        if g is None:
            break
        # scale g to the correct total mass
        sum_g = g.sum()
        if sum_g == 0:
            break
        g = g * (target_sum / sum_g)

        # mixing factor slowly annealed
        t = mix_start * (0.8 ** it)
        f = (1.0 - t) * f + t * g
        f = _project_onto_simplex(f, target_sum)
    return f


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the
    evaluation function within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a previously known construction if present.
    # ------------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        score = evaluate_sequence(hs_arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start from global seed = {score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Main optimisation loop – many independent restarts.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # pick a problem size (explore wide early, focus later)
        if remaining > 400:
            n = np.random.randint(1200, 3500 + 1)
        elif remaining > 150:
            n = np.random.randint(800, 3000 + 1)
        else:
            n = np.random.randint(500, 2000 + 1)

        target_sum = np.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Choose an initial shape – we test a few simple families and keep
        # the one with the smallest raw convolution maximum.
        # ------------------------------------------------------------------
        candidates = []

        # 1) power‑law
        p = np.random.uniform(0.2, 0.9)
        idx = np.arange(1, n + 1, dtype=float)
        cand = idx ** (-p)
        candidates.append(cand)

        # 2) geometric (exponential) shape
        r = np.random.uniform(0.95, 0.995)                # decay factor
        geom_sum = (1.0 - r ** n) / (1.0 - r)
        C = target_sum / geom_sum
        cand = C * (r ** np.arange(n))
        candidates.append(cand)

        # 3) mixed power‑law + exponential tail
        p = np.random.uniform(0.3, 0.7)
        cand = idx ** (-p)
        if np.random.rand() < 0.5:
            tau = np.random.uniform(0.5, 2.0) * n
            cand = cand * np.exp(-idx / tau)
        candidates.append(cand)

        # occasionally use the current global best as a seed (truncated/padded)
        if best_seq is not None and np.random.rand() < 0.4:
            bs = best_seq
            if bs.size >= n:
                cand = bs[:n].copy()
            else:
                pad = np.random.rand(n - bs.size) * 1e-4
                cand = np.concatenate([bs, pad])
            candidates.append(cand)

        # evaluate each cheap candidate (only need max(conv) for comparison)
        best_local = None
        best_local_val = np.inf
        for cand in candidates:
            cand = np.maximum(cand, 0.0)
            cand = _project_onto_simplex(cand, target_sum)
            conv = _full_convolution_fft(cand, cand)
            val = conv.max()
            if val < best_local_val:
                best_local_val = val
                best_local = cand

        # ------------------------------------------------------------------
        # Gradient descent (soft‑max) – allocate a slice of the remaining time.
        # ------------------------------------------------------------------
        slice_time = min(12.0, remaining * 0.55)
        run_deadline = time.time() + slice_time
        seq_local, score_local = _optimise_one_length_softmax(
            n,
            best_local,
            run_deadline,
            init_tau_factor=0.12,
            min_tau_factor=0.001,
            step_factor=0.6,
            step_decay=0.9993,
            momentum_beta=0.9,
        )

        # ------------------------------------------------------------------
        # LP polishing – a few cheap refinements.
        # ------------------------------------------------------------------
        seq_refined = _lp_refine(seq_local, target_sum, lp_iters=3, mix_start=0.5)
        final_score = evaluate_sequence(seq_refined.tolist())

        if final_score < best_score:
            best_score = final_score
            best_seq = seq_refined.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (n={n})"
            )
            if best_score < 1.503:
                # reached the target, we can stop early
                break

    # ------------------------------------------------------------------
    # Fallback – should never happen but keeps the function safe.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = np.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    # Return a plain Python list as required by the evaluator.
    return best_seq.tolist()
