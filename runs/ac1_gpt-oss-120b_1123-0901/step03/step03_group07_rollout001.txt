"""
Hybrid gradient/annealing search for a low‑value step‑function.

The algorithm consists of several independent restarts.  For each restart
the vector `a` (length n) is represented in log‑space and optimised with
Adam on a smooth‑max surrogate of the objective

        F(a) = 2·n·max_k (a*a)[k] / (∑ a_i)² .

The surrogate uses a soft‑max with a temperature λ that is increased
slowly (continuation).  After the Adam phase a cheap stochastic hill‑climb
(move of a tiny amount of mass between two positions) is performed to
further polish the best candidate seen so far.

Different initialisations (Dirichlet, exponential decay, half‑sine) and
different lengths (2000‑5000) are used in order to explore a rich
search space.  The best sequence found before the time budget expires is
returned as a plain Python list, ready to be fed to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, reusable for any length)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Soft‑max weights  w_k = exp(lam·conv_k) / Σ_j exp(lam·conv_j).

    The maximum of ``conv`` is subtracted for numerical stability; because
    lam>0 the exponent is never positive, avoiding overflow.
    """
    m = conv.max()
    # exponent is non‑positive → no overflow, only possible underflow
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:                     # safety fallback (should not happen)
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int = None) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate w.r.t. the original probabilities ``a``.
    The formula is  ∇φ = 2·(w ★ a) .
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    # slice that corresponds to indices 0 … n‑1 of the original vector
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


def _objective(a: np.ndarray) -> float:
    """Exact evaluation (scale‑invariant)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function (list of heights) that minimises
    the provided evaluation function.  The routine respects the given time
    budget (seconds) and returns the best sequence found.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.

    Returns
    -------
    list[float]
        The best sequence discovered (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.25   # small safety margin

    # ------------------------------------------------------------------
    # Hyper‑parameters (can be tweaked on the fly)
    # ------------------------------------------------------------------
    # candidate lengths – a mix of moderate and larger sizes
    length_choices = (2000, 2500, 3000, 3500, 4000, 4500, 5000)

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.07          # initial learning rate for logits
    lr_decay = 0.9996       # multiplicative decay per iteration

    # λ‑continuation
    lam_start = 0.5
    lam_factor = 1.004      # growth factor per iteration (slow)
    lam_cap = 1e4           # avoid overflow in exp

    best_val = float("inf")
    best_seq = None

    restart_counter = 0
    # ------------------------------------------------------------------
    # Outer loop – many short restarts, each may finish early if time runs out
    # ------------------------------------------------------------------
    while time.time() < deadline:
        restart_counter += 1

        # --------------------------------------------------------------
        # Choose a problem size and initialise a
        # --------------------------------------------------------------
        n = int(np.random.choice(length_choices))
        # pick one of three families for diversification
        mode = restart_counter % 3
        if mode == 0:
            # Dirichlet (roughly uniform but random)
            a = np.random.dirichlet(np.ones(n))
        elif mode == 1:
            # geometric decay with a random ratio
            r = np.random.uniform(0.25, 0.90)
            powers = np.power(r, np.arange(n, dtype=np.float64))
            a = powers / powers.sum()
        else:
            # half‑sine shape (strictly positive)
            i = np.arange(1, n + 1, dtype=np.float64)
            a = np.sin(np.pi * i / (n + 1))
            a = np.maximum(a, 0.0)
            a = a / a.sum()

        # Log‑space parametrisation (soft‑max)
        u = np.log(a + 1e-30)          # avoid log(0)

        # Adam state
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        # Pre‑compute FFT sizes (helps avoid repeated log2 calculations)
        conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        size_corr = 1 << int(np.ceil(np.log2(3 * n)))   # safe for w★a

        # Continuation variable
        lam = lam_start

        # Tracking best candidate inside this restart
        local_best_val = float("inf")
        local_best_a = None

        it = 0
        last_improve_time = time.time()

        # --------------------------------------------------------------
        # Inner optimisation loop (Adam on smooth‑max)
        # --------------------------------------------------------------
        # We cap the number of iterations to keep the global budget reasonable.
        max_iter_this_restart = 15000
        while time.time() < deadline and it < max_iter_this_restart:
            it += 1

            # ----- probabilities from logits (numerically stable) -----
            u_shift = u - np.max(u)
            a = np.exp(u_shift)
            a /= a.sum()

            # ----- convolution and soft‑max weights -----
            conv = _full_convolution(a, size=conv_fft_size)
            w = _softmax_weights(conv, lam)

            # ----- gradient w.r.t. a (smooth‑max surrogate) -----
            grad_a = _softmax_gradient(a, w, size_corr=size_corr)

            # ----- convert to gradient in log‑space (soft‑max Jacobian) -----
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            # ----- Adam update -----
            m = beta1 * m + (1.0 - beta1) * grad_u
            v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = base_lr * (lr_decay ** it)
            u -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # ----- λ continuation (slow growth, capped) -----
            lam = min(lam * lam_factor, lam_cap)

            # ----- periodic exact evaluation -----
            if it % 40 == 0:
                val = _objective(a)
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[Restart {restart_counter:03d}] iter {it:05d} "
                          f"new global best = {val:.6f}")
                    last_improve_time = time.time()
                if val < local_best_val:
                    local_best_val = val
                    local_best_a = a.copy()

            # ----- simple stagnation shake -----
            if time.time() - last_improve_time > 1.2:
                # tiny random perturbation in log‑space to escape flat regions
                u += (np.random.rand(n) - 0.5) * 4e-4
                m.fill(0.0)
                v.fill(0.0)
                last_improve_time = time.time()

        # --------------------------------------------------------------
        # Local stochastic refinement (hill‑climb / simulated‑annealing)
        # --------------------------------------------------------------
        if local_best_a is not None:
            a_refine = local_best_a.copy()
            # Number of refinement steps is tuned to remaining budget
            remaining = deadline - time.time()
            steps = int(min(8000, remaining * 30))   # roughly 30 eval/s
            for _ in range(steps):
                # pick two distinct indices
                i, j = np.random.choice(n, size=2, replace=False)
                if a_refine[i] == 0.0:
                    continue
                # move a modest fraction of mass from i to j
                eps_move = np.random.rand() * 0.08   # up to 8% of a[i]
                delta = eps_move * a_refine[i]
                a_new = a_refine.copy()
                a_new[i] -= delta
                a_new[j] += delta
                # No need to renormalise – total mass stays constant
                val = _objective(a_new)
                if val < best_val:
                    best_val = val
                    best_seq = a_new.copy()
                    print(f"[Restart {restart_counter:03d}] hill‑climb improvement = {val:.6f}")
                if val < local_best_val:
                    local_best_val = val
                    a_refine = a_new
                else:
                    # occasional acceptance of worse move (simulated annealing)
                    # temperature schedule decays with iteration count
                    temp = 0.02 * (1.0 - _ / steps)   # linear cooling
                    if np.random.rand() < math.exp(-(val - local_best_val) / max(temp, 1e-12)):
                        a_refine = a_new
            # after refinement, keep the refined version as a candidate
            if a_refine is not None and _objective(a_refine) < best_val:
                best_val = _objective(a_refine)
                best_seq = a_refine.copy()

        # --------------------------------------------------------------
        # End of one restart – log status
        # --------------------------------------------------------------
        print(f"[Restart {restart_counter:03d}] length={n} it={it} "
              f"best so far = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the (very unlikely) case no candidate was found.
    # ------------------------------------------------------------------
    if best_seq is None:
        # uniform vector of length 2000 as a safe default
        n = 2000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"Search finished after {restart_counter} restarts. "
          f"Best objective = {best_val:.6f} (length {len(best_seq)})")
    # Convert to plain Python list as required by the evaluation harness.
    return best_seq.tolist()
