"""
Propose a step‑function sequence that (approximately) minimises

        F(a) = 2·n·max(conv(a,a)) / (sum(a)²)

under a_i ≥ 0.  The search uses several complementary ideas:

* **Soft‑max parametrisation** – the vector of logits `x` is transformed by
  a soft‑max to a probability vector `a`.  This guarantees non‑negativity
  and Σa_i = 1, making the objective scale‑invariant.
* **Smooth‑max surrogate** – instead of the hard max we optimise a
  log‑sum‑exp of the convolution values with a temperature `λ` that is
  increased geometrically (continuation).  The gradient of this surrogate
  is obtained analytically via a cross‑correlation, allowing efficient
  FFT‑based computation.
* **Adam optimiser** – unbiased first‑ and second‑moment estimates give a
  robust, adaptive update rule for the logits.
* **Multi‑scale restarts & warm‑starts** – the algorithm explores several
  lengths (`n` up to 8192).  For each length we perform a number of
  restarts, warm‑starting from the current global best, from the legacy
  `height_sequence_1` (if present) or from a random point.
* **Stagnation handling** – if no improvement is seen for a few seconds
  the logits are jittered slightly to escape flat regions.
* **Local hill‑climbing** – after the gradient phase a cheap pair‑wise
  perturbation of the probability vector is performed for a few thousand
  steps; any improvement is kept.
* **Time‑budget awareness** – the routine stops automatically before the
  allotted 1000 s, returning the best sequence found so far.

All heavy work (convolution, cross‑correlation) is performed with FFTs,
so each iteration costs O(n log n) and fits comfortably inside the
budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)−1) using FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the surrogate gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    n = a.shape[0]
    start = n - 1
    return corr[start:start + n]  # length n


def upsample_to_length(seq: np.ndarray, target_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length ``target_len``."""
    if seq.size == target_len:
        return seq.copy()
    x_old = np.arange(seq.size)
    x_new = np.linspace(0, seq.size - 1, target_len)
    return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a low evaluation
    score.  The routine respects the given time budget, prints progress
    and finally returns the best sequence (as a plain Python list).
    """
    np.random.seed(seed)

    # safety margin for interpreter shutdown
    deadline = time.time() + budget_s - 0.5

    best_seq = None          # numpy array, Σa = 1
    best_val = float("inf")  # current best evaluation (exact)

    # ------------------------------------------------------------------
    # helper to initialise logits from a probability vector
    def logits_from_probs(p: np.ndarray) -> np.ndarray:
        return np.log(p + 1e-15)

    # ------------------------------------------------------------------
    # Problem sizes – start with the largest to give it the most time.
    lengths = [8192, 4096, 2048, 1024]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")

        # FFT sizes (powers of two) for the two convolutions we need.
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))  # ≥ (2n‑1)+n‑1

        # Rough allocation of restarts: larger n ⇒ fewer restarts.
        remaining = deadline - time.time()
        # want at least a few seconds per restart; clamp to a sensible range
        restarts = max(1, int(remaining // 30))
        restarts = min(restarts, 12)

        for r in range(restarts):
            if time.time() > deadline:
                break

            # -------------------------------------------------------------
            # Initialise logits (x).  Try warm‑starts before falling back to random.
            if best_seq is not None and len(best_seq) == n:
                # Warm‑start from the best sequence seen so far (same length)
                prob = np.maximum(best_seq, 0.0)
                prob_sum = prob.sum()
                if prob_sum > 0:
                    prob = prob / prob_sum
                else:
                    prob = np.random.dirichlet(np.ones(n))
                x = logits_from_probs(prob)
                print(f"[n={n} restart={r}] warm‑start from global best")
            elif "height_sequence_1" in globals():
                # Warm‑start from the legacy construction, interpolating if needed
                cand = np.asarray(height_sequence_1, dtype=np.float64)
                cand = upsample_to_length(cand, n)
                cand = np.maximum(cand, 0.0)
                s = cand.sum()
                if s < 0.01:
                    prob = np.random.dirichlet(np.ones(n))
                else:
                    prob = cand / s
                x = logits_from_probs(prob)
                print(f"[n={n} restart={r}] warm‑start from height_sequence_1")
            else:
                # Pure random initialise
                x = np.random.randn(n) * 0.1
                print(f"[n={n} restart={r}] random initialise")

            # -------------------------------------------------------------
            # Adam state
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8

            # Learning‑rate schedule (larger n → slightly larger start)
            lr = 0.12 if n >= 4096 else 0.07

            # Smooth‑max temperature
            lam = 0.5
            lam_factor = 1.02   # geometric increase

            t_iter = 0
            max_iters = 50000            # safety cap per restart
            last_improve = time.time()   # for stagnation detection

            # -------------------------------------------------------------
            while time.time() < deadline and t_iter < max_iters:

                a = softmax(x)                     # a_i ≥ 0, Σa_i = 1
