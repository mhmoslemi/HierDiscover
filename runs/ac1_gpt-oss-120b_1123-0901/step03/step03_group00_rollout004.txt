"""
Propose a step‑function (non‑negative sequence) that minimizes

        F(a) = 2·n·max(a*a) / (sum(a)²)

using a hybrid strategy:

* **Phase 1 – Continuous optimisation**  
  Optimise the vector in the probability simplex with Adam on a smooth‑max
  surrogate of the convolution peak.  Multiple sequence lengths, several
  initialisations (random, Dirichlet, exponential, warm‑starts from the
  previous best or from ``height_sequence_1``) are tried.  The optimiser
  works with FFT‑based convolutions for speed.

* **Phase 2 – Local refinement**  
  Starting from the best candidate found in phase 1 we perform a cheap
  stochastic hill‑climbing: a small amount of mass is transferred from one
  index to another while the total sum stays constant.  A move is kept
  only if the exact evaluation (via ``evaluate_sequence``) improves.

The routine respects the supplied time budget (default 1000 s) and finally
returns the best sequence as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _fft_len(target: int) -> int:
    """Smallest power‑of‑two not smaller than *target*."""
    return 1 << int(np.ceil(np.log2(target)))


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    ``fft_len`` must be a power of two ≥ 2·len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)],  i = 0…n‑1.
    Implemented via FFT: correlation = ifft(fft(w)·fft(rev(a))).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    n = a.shape[0]
    start = n - 1
    return corr[start:start + n]


def _objective_from_conv(conv: np.ndarray, n: int, sum_a: float) -> float:
    """Exact objective used by ``evaluate_sequence`` (ignoring clipping)."""
    # ``conv`` is a*a, max over all lag positions
    max_c = conv.max()
    return 2.0 * n * max_c / (sum_a * sum_a)


def _logits_from_probs(p: np.ndarray) -> np.ndarray:
    """Safe conversion of a probability vector to log‑its."""
    return np.log(p + 1e-15)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a low‑value step function within the given time budget.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper to create different initial logit vectors
    def _init_logits(n: int, mode: str, warm_seq=None):
        if mode == "warm" and warm_seq is not None:
            a = np.asarray(warm_seq, dtype=np.float64)
            if a.shape[0] != n:
                # truncate or pad with zeros
                if a.shape[0] > n:
                    a = a[:n]
                else:
                    a = np.pad(a, (0, n - a.shape[0]))
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 0.01:
                a = np.random.dirichlet(np.ones(n))
            else:
                a /= s
            return _logits_from_probs(a)

        if mode == "dirichlet":
            a = np.random.dirichlet(np.ones(n))
            return _logits_from_probs(a)

        if mode == "exp":
            a = np.random.exponential(scale=1.0, size=n)
            a /= a.sum()
            return _logits_from_probs(a)

        if mode == "gauss":
            a = np.abs(np.random.randn(n))
            a /= a.sum()
            return _logits_from_probs(a)

        # fallback: tiny Gaussian noise around uniform
        a = np.full(n, 1.0 / n) + 0.01 * np.random.randn(n)
        a = np.clip(a, 0.0, None)
        a /= a.sum()
        return _logits_from_probs(a)

    # ------------------------------------------------------------------
    # Problem sizes we will explore
    lengths = [1024, 2048, 4096, 8192]

    # Main optimisation loop
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        fft_len_conv = _fft_len(2 * n - 1)
        fft_len_corr = _fft_len(3 * n)          # generous size for cross‑corr

        # Number of restarts for this length (adjustable)
        restarts = 6

        for r in range(restarts):
            if time.time() > deadline:
                break

            # -------------------------------------------------------------
            # Choose an initialisation strategy
            if best_seq is not None and len(best_seq) == n:
                init_mode = "warm"
                warm = best_seq
                print(f"[restart {r}] warm start from global best")
            elif 'height_sequence_1' in globals():
                # Use the supplied historic construction
                init_mode = "warm"
                warm = globals()['height_sequence_1']
                print(f"[restart {r}] warm start from height_sequence_1")
            else:
                # Randomly pick a diverse scheme
                init_mode = np.random.choice(["dirichlet", "exp", "gauss"])
                warm = None
                print(f"[restart {r}] random init mode = {init_mode}")

            x = _init_logits(n, init_mode, warm_seq=warm)

            # Adam state
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.03                     # modest step size for stability
            lam = 0.7                     # initial smooth‑max temperature
            lam_factor = 1.02

            # Iteration limits
            max_iters = 12000
            t = 0
            last_improve = time.time()

            while time.time() < deadline and t < max_iters:
                # ---------- simplex projection (softmax) ----------
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)            # stability trick
                a = exp_x / exp_x.sum()               # Σa = 1, a_i≥0

                # ---------- self‑convolution ----------
                conv = _full_convolution(a, fft_len_conv)
                max_c = conv.max()

                # ---------- smooth‑max weight ----------
                # w_k ∝ exp(lam·(conv_k – max_c))
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # ---------- gradient w.r.t. a ----------
                grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

                # ---------- back‑prop through softmax ----------
                avg = np.dot(a, grad_a)
                grad_x = a * (grad_a - avg)

                # ---------- Adam update ----------
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # ---------- temperature schedule ----------
                lam = min(lam * lam_factor, 1e5)

                # ---------- exact objective (fast) ----------
                # sum(a) == 1 by construction, so denominator = 1
                val = 2.0 * n * max_c

                # ---------- record best ----------
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} r={r} it={t:05d}] new best = {val:.6f}")

                # ---------- stagnation handling ----------
                if time.time() - last_improve > 4.0:
                    # random shake of logits to escape plateau
                    x += (np.random.rand(n) - 0.5) * 5e-3
                    last_improve = time.time()
                    print(f"[n={n} r={r} it={t}] shake (stagnation)")

    # ------------------------------------------------------------------
    # If nothing was found (unlikely) fall back to a uniform distribution.
    if best_seq is None:
        n_fallback = 2000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = _objective_from_conv(
            _full_convolution(best_seq, _fft_len(2 * n_fallback - 1)),
            n_fallback,
            best_seq.sum(),
        )
        print("Fallback to uniform sequence, value:", best_val)

    # ------------------------------------------------------------------
    # Phase 2 – inexpensive local stochastic refinement
    # ---------------------------------------------------------------
    def _local_refine(seq: np.ndarray, max_steps: int):
        """Move a tiny amount of mass between random positions if it improves."""
        n_local = seq.shape[0]
        cur_val = evaluate_sequence(seq.tolist())
        start = time.time()
        for step in range(max_steps):
            if time.time() > deadline:
                break
            # pick donor and receiver
            i, j = np.random.randint(0, n_local, size=2)
            if i == j or seq[i] == 0.0:
                continue
            # transfer a fraction of donor's mass
            frac = np.random.uniform(0.001, 0.02)
            delta = seq[i] * frac
            new_seq = seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
            # evaluate exact objective (includes clipping etc.)
            val = evaluate_sequence(new_seq.tolist())
            if val < cur_val:
                seq = new_seq
                cur_val = val
                # keep the best globally as well
                nonlocal best_seq, best_val
                if val < best_val:
                    best_val = val
                    best_seq = seq.copy()
                # occasional progress report
                if step % 500 == 0:
                    print(
                        f"[local refine] step {step:05d} improve to {val:.6f}"
                    )
        return seq

    # Apply the local refinement – number of steps scales with remaining time
    remaining = max(0.0, deadline - time.time())
    if remaining > 2.0:                        # at least a couple of seconds left
        # decide how many steps we can afford (roughly 2 ms per step on avg)
        max_local_steps = int(min(20000, remaining * 300))
        print(f"\n--- Local refinement: up to {max_local_steps} steps ---")
        best_seq = _local_refine(best_seq.copy(), max_local_steps)

    # ------------------------------------------------------------------
    # Convert final numpy array to plain Python list
    final_list = best_seq.tolist()
    print("\n=== Finished ===")
    print(f"Best objective value found: {best_val:.6f}")
    print(f"Length of final sequence: {len(final_list)}")
    return final_list
