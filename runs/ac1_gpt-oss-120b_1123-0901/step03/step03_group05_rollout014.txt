def build_convolution_ub_matrix(f):
    n = f.shape[0]
    # number of rows = 2*n - 1
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)
    pos = 0
    for i, fi in enumerate(f):
        start = i
        end = i + n
        # rows slice
        rows[pos:pos+n] = np.arange(start, end, dtype=np.int32)
        cols[pos:pos+n] = i
        data[pos:pos+n] = fi
        pos += n
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
    return A

def build_conv_ub_matrix(f):
    n = len(f)
    # Build row indices: for each i, rows i to i+n-1
    row_inds = np.arange(n).reshape(-1, 1) + np.arange(n)  # shape (n, n)
    row_inds = row_inds.ravel()
    col_inds = np.repeat(np.arange(n), n)
    data = np.repeat(f, n)
    A = scipy.sparse.csr_matrix((data, (row_inds, col_inds)), shape=(2*n-1, n))
    return A

"""
Propose a step‑function (non‑negative vector) with a low maximal
autoconvolution coefficient.

Algorithm
---------
1.  Load the previous construction (global ``height_sequence_1``) and
    normalise it so that Σa = sqrt(2·n).  This gives a reasonable starting
    point.

2.  For several lengths (≈2000–3000) initialise a few candidate vectors:
    – linear interpolation of the previous construction,
    – random Dirichlet draws.

3.  **LP‑mixing phase** – for each candidate run a few outer iterations:
        •  Compute the current bound M = max (a * a).
        •  Build the sparse convolution matrix A such that (A@g)_k = (a*g)_k.
        •  Solve the linear program
               maximise   Σ g_i
               subject to A g ≤ M·1 ,   g ≥ 0 .
           (SciPy’s high‑performance “highs” solver is used.)
        •  Scale the solution to the required ℓ₁–norm and mix it with the
           current vector: a ← (1‑t)·a + t·g_scaled (t = 0.5).
        •  Update the best vector if max‑convolution decreases.

   The LP step moves mass from indices that contribute heavily to the
   current maximal coefficient to positions that are “safer”.

4.  **Fine‑tuning** – the result of the LP phase is refined with a
    projected Adam optimiser (same sub‑gradient that was used in the
    previous SOTA code) for a limited number of iterations.

5.  **Local search** – a cheap coordinate‑swap hill‑climb (mass transfer
    between two indices) is performed for a few ten‑thousands of proposals.

6.  The best sequence found before the time budget expires is returned as
    a plain Python list.  All steps respect the 1000 s budget and the
    positivity/normalisation constraints required by ``evaluate_sequence``.
"""

import time
import math
import numpy as np
import scipy.sparse
import scipy.optimize

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """Full autoconvolution (polynomial square) via FFT.
    Returns (max value, full convolution array)."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return float(conv.max()), conv


# ----------------------------------------------------------------------
def _build_conv_ub_matrix(f: np.ndarray):
    """Build sparse matrix A (shape (2n‑1, n)) such that (A @ g)_k = (f * g)_k."""
    n = f.shape[0]
    # rows[i, j] = i + j  (i = column index, j = row offset inside column)
    rows = (np.arange(n).reshape(-1, 1) + np.arange(n)).ravel()
    cols = np.repeat(np.arange(n), n)
    data = np.repeat(f, n)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


# ----------------------------------------------------------------------
def _lp_optimize(a_init: np.ndarray,
                 target_sum: float,
                 deadline: float,
                 max_outer_iters: int = 6,
                 mix_factor: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Iterative linear‑programming mixing.
    Starting from a_init, repeatedly:
        – maximise Σg  under the current max‑convolution bound,
        – scale g to the required ℓ₁‑norm,
        – mix it with the current vector.
    Returns the best vector and its max‑convolution value.
    """
    a = a_init.copy()
    a = a * (target_sum / a.sum())          # enforce ℓ₁‑norm
    best_a = a.copy()
    best_M, _ = _conv_max_fft(a)

    it = 0
    while time.time() < deadline and it < max_outer_iters:
        it += 1
        cur_M, _ = _conv_max_fft(a)

        # ---- linear programme ------------------------------------------------
        A = _build_conv_ub_matrix(a)               # (2n‑1) × n
        b = np.full(2 * a.shape[0] - 1, cur_M)

        c = -np.ones(a.shape[0])                    # maximise sum(g)
        bounds = [(0, None)] * a.shape[0]

        res = scipy.optimize.linprog(c,
                                     A_ub=A,
                                     b_ub=b,
                                     bounds=bounds,
                                     method='highs',
                                     options={'presolve': True})
        if not res.success:
            # If the LP fails we abort the LP phase for this vector.
            print(f"[LP] linprog failed (iter {it}): {res.message}")
            break

        g = res.x
        if g.sum() <= 1e-12:
            break

        # ---- scale and mix ---------------------------------------------------
        g_scaled = g * (target_sum / g.sum())
        a = (1.0 - mix_factor) * a + mix_factor * g_scaled
        # re‑normalise (numerical drift)
        a = a * (target_sum / a.sum())

        new_M, _ = _conv_max_fft(a)
        if new_M < best_M:
            best_M = new_M
            best_a = a.copy()
            print(f"[LP iter {it}] improvement: M = {best_M:.6f}")

    return best_a, best_M

def _rev_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    """Reversed slice for a given shift index k (used by the Adam optimiser)."""
    n = a.shape[0]
    rev = np.zeros_like(a)
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    rev[mask] = a[j[mask]]
    return rev


def _subgradient_average(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Average reversed vector over all active shift indices."""
    n = a.shape[0]
    rev = np.zeros_like(a)
    idx = np.arange(n)
    for k in active:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    if active.size > 0:
        rev /= float(active.size)
    return rev


def _adam_optimize(a_init: np.ndarray,
                   n: int,

    Projected Adam optimiser of max‑convolution (same objective as before).
    Shorter iteration budget – it is used only as a fine‑tuner after the
    LP phase.
    """
    a = a_init.copy()
    a = a * (target_sum / a.sum())

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.07
    coeff = 2.0
    sqrt2 = math.sqrt(2.0)

    iteration = 0
    while time.time() < deadline and iteration < max_iter:
        M, _, conv_full = _conv_max_fft(a)
        tol = 1e-12
        active = np.where(conv_full >= M - tol)[0]

        if active.size and np.random.rand() < random_active_prob:
            k = np.random.choice(active)
            rev = _rev_for_shift(a, k)
        else:
            rev = _subgradient_average(a, active)

        const = sqrt2 * M / math.sqrt(n)
        grad = coeff * rev - const

        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # occasional jitter
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[Adam] new best M = {best_score:.6f} (iter {iteration})")
    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_search(a_start: np.ndarray,
                  n: int,
                  target_sum: float,
                  best_score: float,
                  deadline: float,
                  max_proposals: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb: move a small amount of mass from one index to
    another only if the maximal convolution coefficient decreases.
    """
    a = a_start.copy()
    proposal = 0
    while time.time() < deadline and proposal < max_proposals:
        proposal += 1
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        eps = a[i] * np.random.rand() * 0.05
        if eps <= 0:
            continue
        new_vec = a.copy()
        new_vec[i] -= eps
        new_vec[j] += eps
        M_new, _ = _conv_max_fft(new_vec)
        if M_new < best_score:
            a = new_vec
            best_score = M_new
            print(f"[local] improvement to {best_score:.6f} (proposal {proposal})")
    return a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step sequence that minimises
    ``evaluate_sequence``.  Returns the best list found before the time
    budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # ------------------------------------------------------------------
    # Load previous construction if available
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        base = np.ones(2100, dtype=float)

    n0 = base.shape[0]
    target_sum0 = math.sqrt(2.0 * n0)
    base = base * (target_sum0 / base.sum())
    best_seq = base.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] length={n0}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore (moderate sizes – keep LP memory feasible)
    length_factors = [1.0, 1.1, 1.25, 1.4, 1.6, 2.0]
    lengths = sorted({int(n0 * f) for f in length_factors})
    max_len = 3000               # keep LP tractable
    lengths = [l for l in lengths if 300 <= l <= max_len]

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # three restarts per length: interpolation + random + random‑spike
        for restart in range(3):
            if time.time() >= deadline:
                break

            # ----- initialisation ------------------------------------------------
            if restart == 0:
                # Interpolation of the previously best vector onto new grid
                old_x = np.linspace(0.0, 1.0, n0)
                new_x = np.linspace(0.0, 1.0, n)
                a0 = np.interp(new_x, old_x, best_seq)
                a0 = np.maximum(a0, 0.0)
            elif restart == 1:
                # Dirichlet‑like random start
                a0 = np.random.rand(n)
            else:
                # Exponential‑decay start (more mass at the beginning)
                decay = np.random.uniform(0.96, 0.998)
                a0 = np.exp(-decay * np.arange(n))
                a0 = a0 * np.random.rand()   # random scale
                a0 = np.maximum(a0, 0.0)

            a0 = a0 * (target_sum / a0.sum())

            # ----- LP mixing phase ------------------------------------------------
            a_lp, M_lp = _lp_optimize(a0,
                                      target_sum,
                                      deadline,
                                      max_outer_iters=6,
                                      mix_factor=0.5)

            cur_score = M_lp
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_lp.copy()
                print(f"[lp] length={n}, restart={restart}, new best M={best_score:.6f}")

            # ----- small Adam fine‑tuning -----------------------------------------
            if time.time() < deadline:
                a_adam, M_adam = _adam_optimize(a_lp,
                                                n,
                                                target_sum,
                                                best_seq,
                                                best_score,
                                                deadline,
                                                max_iter=150_000,
                                                jitter_interval=2500,
                                                random_active_prob=0.45)
                if M_adam < best_score:
                    best_score = M_adam
                    best_seq = a_adam.copy()
                    print(f"[adam] length={n}, restart={restart}, M={best_score:.6f}")

            # ----- cheap local search ---------------------------------------------
            if time.time() < deadline:
                a_loc, M_loc = _local_search(best_seq,
                                             n,
                                             target_sum,
                                             best_score,
                                             deadline,
                                             max_proposals=20_000)
                if M_loc < best_score:
                    best_score = M_loc
                    best_seq = a_loc.copy()
                    print(f"[local] length={n}, restart={restart}, M={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
```

This implementation combines a **linear‑programming mixing** step (which makes large, informed moves in the simplex) with a short **Adam** fine‑tuning and a final cheap **local‑search** hill‑climb.  
It explores several lengths (up to ~3000 entries) and several random restarts, always respecting the 1000 s time budget.  The returned list satisfies the non‑negativity and ℓ₁‑normalisation required by `evaluate_sequence`.
