def _adam_optimize(n, init_a, deadline, max_iters=500000, best_seq=None, best_score=math.inf):
    target_sum = math.sqrt(2.0*n)
    a = init_a.copy()
    # Project onto sum constraint
    a = a * (target_sum / a.sum())
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    lr0=0.08
    tau_factor = 0.3
    decay_rate = 15000.0
    # Determine FFT size once
    Nfft = 1 << ((2*n-1).bit_length())
    iter=0
    while time.time() < deadline and iter < max_iters:
        # compute conv via FFT
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[:2*n-1]
        M = conv.max()
        # compute smoothed max via softmax
        tau = max(1e-12, tau_factor * M * math.exp(-iter/decay_rate))
        w, Ltau = _softmax_weights(conv, tau)
        # compute rev
        rev = _weighted_rev(a, w, Nfft)
        # gradient
        coeff = 4.0 * n / (target_sum**2)
        const = 4.0 * n * Ltau / (target_sum**3)
        grad = coeff * rev - const
        # Adam update
        iter += 1
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad*grad)
        m_hat = m / (1-beta1**iter)
        v_hat = v / (1-beta2**iter)
        lr = lr0 / (1 + 1e-4*iter)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # projection onto feasible set
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)
        # jitter every few iterations
        if iter % 4000 == 0:
            jitter = np.random.randn(n) * (target_sum/n) * 0.001
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
        # evaluate exact objective (equal to M)
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            # log
            print(f"[len {n:5d} iter {iter:8d}] new best = {best_score:.6f}")
    return best_seq, best_score

def _hillclimb(a, cur_score, deadline, max_iters=200000):
    n = len(a)
    target_sum = a.sum()  # already equal sqrt(2n)
    # conversion for evaluation pure using evaluate_sequence
    temp = 0.1
    step_frac = 0.02 # fraction of target_sum per move
    # precompute random indices pool
    it = 0
    while time.time() < deadline and it < max_iters:
        # dynamic step size reduction
        step = step_frac * target_sum * (0.99**(it/5000))
        # pick i with a[i] > step
        # maybe weighted by a to larger mass
        idx_i = np.random.choice(n, p=a/target_sum)  # bigger mass more likely
        if a[idx_i] < step:
            # reduce step
            step = a[idx_i] * 0.9
        # pick j different
        idx_j = np.random.randint(0, n-1)
        if idx_j >= idx_i:
            idx_j += 1
        # propose transfer
        new_a = a.copy()
        new_a[idx_i] -= step
        new_a[idx_j] += step
        # ensure non-neg
        # evaluate
        new_score = evaluate_sequence(new_a.tolist())
        # acceptance
        if new_score < cur_score:
            a = new_a
            cur_score = new_score
        else:
            # probability to accept
            if np.random.rand() < math.exp(-(new_score - cur_score)/temp):
                a = new_a
                cur_score = new_score
        # anneal temperature
        it += 1
        if it % 1000 == 0:
            # gradually lower temp
            temp *= 0.99
        # update best global maybe
        # we could return best found globally; track best.
        # if new_score < best_global: update best_global. In propose we maintain best.
    return a, cur_score

"""
Improved search for a low‑peak step function.

Algorithm
---------
* Load the previously discovered shape (global ``height_sequence_1``) if
  available, otherwise start from a uniform vector.
* Try several lengths (the original length and a few enlargements up to
  8000 entries).  For each length run a **soft‑max Adam** optimiser:
    – The true max of the self‑convolution is approximated by a
      temperature‑smoothed soft‑max ``Lτ``.  The gradient of ``Lτ`` is
      obtained analytically with a single FFT (weighted reverse
      convolution).
    – Adam updates are projected onto the feasible set
      ``a ≥ 0, Σa = √(2 n)``.
    – A relatively large initial temperature (τ ≈ 30 % of the current
      maximum) and a slow exponential decay give a smooth landscape
      early on and a fine search later.
    – Small Gaussian jitter is injected every few thousand iterations
      to escape shallow plateaus.
* The best vector over all lengths is kept.
* **Fine‑tuning:**  With the remaining time a second Adam stage is run
  with τ → 0 (i.e. a one‑hot weight at the current maximal convolution
  index).  This directly minimises the exact objective ``max(a∗a)`` and
  typically yields a further noticeable drop.
* The routine monitors the wall‑clock and stops well before the
  1000 s limit, returning the best sequence found.

The code uses only ``numpy`` and the already‑imported ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, tau: float):
    """
    Stable soft‑max of the convolution values.
    Returns the weight vector w (summing to 1) and the smoothed maximum
    Lτ = τ·log ∑exp(conv/τ).  When tau==0 a hard max is returned.
    """
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    # subtract the maximum for numerical stability
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """
    Compute rev_i = Σ_k w_k · a_{k‑i}  (indices outside 0…n‑1 ignored)
    via FFT.  ``a`` length = n, ``w`` length = 2 n‑1.
    """
    n = a.shape[0]
    # FFT of a (zero‑padded to length Nfft)
    A_fft = np.fft.rfft(a, n=Nfft)
    # FFT of the reversed weight vector
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    # Convolution of w_rev with a
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # keep the valid part (length = len(w)+n‑1)
    conv_rev = conv_rev[: len(w) + n - 1]
    # the required slice
    idx_start = 2 * n - 2
    rev = conv_rev[idx_start - np.arange(n)]
    return rev


# ----------------------------------------------------------------------
# Adam optimiser for a given length
# ----------------------------------------------------------------------


def _adam_optimize(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    *,
    lr0: float = 0.08,
    tau_factor: float = 0.30,
    decay_rate: float = 50000.0,
    jitter_interval: int = 3000,
    jitter_scale: float = 0.001,
):
    """
    Adam optimisation of the smoothed maximum of the self‑convolution.
    ``init_a`` is assumed to already satisfy the L1‑norm constraint.
    Returns the (possibly) improved best sequence and its exact score.
    """
    target_sum = math.sqrt(2.0 * n)

    a = init_a.copy()
    a = a * (target_sum / a.sum())          # enforce exact L1 norm

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    Nfft = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1

    iteration = 0
    while time.time() < deadline and iteration < max_iters:
        # --------------------------------------------------------------
        # 1) exact convolution a∗a (full) via FFT
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # --------------------------------------------------------------
        # 2) soft‑max smoothing of the max
        tau = max(1e-14, tau_factor * M * math.exp(-iteration / decay_rate))
        w, Ltau = _softmax_weights(conv, tau)

        # --------------------------------------------------------------
        # 3) weighted reverse convolution (sub‑gradient of the smooth max)
        rev = _weighted_rev(a, w, Nfft)

        # --------------------------------------------------------------
        # 4) Adam gradient step (including the normalisation term)
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # --------------------------------------------------------------
        # 5) projection onto the feasible set (a ≥ 0, Σa = √(2 n))
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # --------------------------------------------------------------
        # 6) occasional jitter to escape plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (target_sum / n) * jitter_scale
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # --------------------------------------------------------------
        # 7) exact objective value (after normalisation it equals M)
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:8d}] new best = {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Fine‑tuning stage – directly minimises the exact max using a one‑hot
# weight (τ → 0).  This is just another Adam run with a much smaller
# learning rate and no temperature smoothing.
# ----------------------------------------------------------------------


def _fine_tune_max(
    a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    lr0: float = 0.02,
    jitter_interval: int = 5000,
    jitter_scale: float = 0.0005,
):
    """
    Refine ``a`` by descending the genuine objective
    ``max(a∗a)``.  The gradient is taken at the current maximal
    convolution index (i.e. a one‑hot weight).  Returns the refined
    sequence and its exact objective value.
    """
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)

    a = a.copy()
    a = a * (target_sum / a.sum())

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    Nfft = 1 << ((2 * n - 1).bit_length())

    iteration = 0
    while time.time() < deadline and iteration < max_iters:
        # convolution and location of the maximum
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        k_max = int(conv.argmax())
        M = float(conv[k_max])

        # one‑hot weight at the maximal index
        w = np.zeros_like(conv)
        w[k_max] = 1.0

        rev = _weighted_rev(a, w, Nfft)

        coeff = 4.0 * n / (target_sum ** 2)
        const = 4.0 * n * M / (target_sum ** 3)   # here Lτ == M
        grad = coeff * rev - const

        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # projection
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # jitter (helps to break symmetry)
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (target_sum / n) * jitter_scale
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # occasional logging
        if iteration % 5000 == 0:
            print(f"[fine] len {n:5d} iter {iteration:7d} cur max {M:.6f}")

    # final exact score
    final_score = float(np.max(np.fft.irfft(
        np.fft.rfft(a, n=Nfft) * np.fft.rfft(a, n=Nfft), n=Nfft
    )[: 2 * n - 1]))
    return a, final_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted time.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # slight safety margin

    # ------------------------------------------------------------------
    # 1) Load a sensible starting shape
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        # fallback: uniform vector of length 2100
        base = np.ones(2100, dtype=float)

    n0 = base.shape[0]
    target_sum0 = math.sqrt(2.0 * n0)
    base = base * (target_sum0 / base.sum())
    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n0:,d}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Define lengths to explore (original and a few enlargements)
    length_factors = [1.0, 1.2, 1.5, 1.8, 2.0]
    lengths = sorted({int(n0 * f) for f in length_factors})
    # keep them reasonable
    lengths = [l for l in lengths if 300 <= l <= 8000]
    lengths = sorted(set(lengths))

    # ------------------------------------------------------------------
    # 3) Optimise each length with a (moderately long) Adam run
    for n in lengths:
        if time.time() >= deadline:
            break

        # Initialise a vector of the required length
        if n == n0:
            init_a = base.copy()
        else:
            # Linear interpolation of the base shape onto the new grid
            old_x = np.linspace(0.0, 1.0, n0)
            new_x = np.linspace(0.0, 1.0, n)
            init_a = np.interp(new_x, old_x, base)
            init_a = np.maximum(init_a, 0.0)

        # Enforce the correct L1‑norm for the new length
        target_sum = math.sqrt(2.0 * n)
        init_a = init_a * (target_sum / init_a.sum())

        # Rough iteration budget: more for smaller n, less for larger n
        # Allocate a fraction of the remaining time to this length.
        time_left = deadline - time.time()
        # heuristic: aim for about 0.15 s per 1000 iterations at n≈2000
        approx_iter = int(min(800_000, max(100_000, 5_000_000 * n0 / n * time_left / budget_s)))
        max_iters = max(200_000, approx_iter)

        print(f"[run] length={n:,d}, max_iters≈{max_iters:,d}")

        # Run the optimiser (single restart – the Adam loop already contains jitter)
        best_seq, best_score = _adam_optimize(
            n=n,
            init_a=init_a,
            deadline=deadline,
            max_iters=max_iters,
            best_seq=best_seq,
            best_score=best_score,
            lr0=0.08,
            tau_factor=0.30,
            decay_rate=50000.0,
            jitter_interval=3000,
            jitter_scale=0.001,
        )

    # ------------------------------------------------------------------
    # 4) Final fine‑tuning on the best candidate found so far
    if time.time() < deadline:
        print("[fine] starting exact‑max fine‑tuning")
        fine_seq, fine_score = _fine_tune_max(
            best_seq,
            deadline,
            max_iters=200_000,
            lr0=0.02,
            jitter_interval=5000,
            jitter_scale=0.0005,
        )
        if fine_score < best_score:
            best_seq, best_score = fine_seq, fine_score

    # ------------------------------------------------------------------
    # 5) Return result
    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
