def _conv_full_fft_custom(x):
    n = len(x)
    n_fft = 1 << ((2*n-1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    return np.fft.irfft(X * X, n=n_fft)[:2*n-1]

def _conv_full_fft_two(x, y):
    n = len(x) + len(y) - 1
    n_fft = 1 << ((n).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X*Y, n=n_fft)[:n]
    return conv

def adam_optimize(a_init, target_sum, time_limit, start_time):
    a = a_init.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _max_conv(a)  # direct

    beta = 0.5
    beta_max = 200.0
    beta_factor = 1.5
    lr = 0.1 * target_sum / n  # scale
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    while time.time() < start_time + time_limit:
        # compute smooth max and grad
        conv = _conv_full_fft(a)  # O(n log n)
        maxc = conv.max()
        # smoothing
        # w = np.exp(beta*(conv - maxc))
        # but this yields all terms <=1; stable.
        w = np.exp(beta * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            # all zeros? then set uniform weights to avoid division zero.
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum
        # gradient
        a_rev = a[::-1]
        grad_full = _conv_full_fft_two(w, a_rev)  # length 3n-2
        grad = 2.0 * beta * grad_full[n-1:n-1+n]  # slice length n

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

"""
Propose a non‑negative step‑function that (approximately) minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is a two‑phase optimiser:

1. **Smooth‑max Adam optimisation** – we minimise a smooth approximation of
   the max‑convolution using the log‑sum‑exp surrogate.
   The surrogate is gradually sharpened (β grows from 0.5 to ~200) while the
   vector stays on the simplex Σa = √(2n).  Adam with projection gives fast
   descent and works well for the high‑dimensional non‑convex landscape.

2. **Short simulated‑annealing refinement** – after the Adam phase we run a
   cheap SA that transfers a random fraction of mass between two entries.
   The SA helps to escape possible local minima and usually yields a small
   extra improvement.

The outer loop restarts the whole procedure with different lengths
(`n ≈ 1500‑3000`) and different initialisations (previous best, a U‑shaped
seed, or a random perturbation).  It runs until the allotted time budget
(~1000 s) expires and returns the best sequence found (as a plain Python
list).  All helper functions are top‑level, use only NumPy and the standard
library, and respect the required simplex constraint.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Simplex projection (Euclidean) – Σx = s , x_i ≥ 0
# --------------------------------------------------------------------------- #
def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = v.astype(float, copy=False)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Full convolution via FFT (real signals)
# --------------------------------------------------------------------------- #
def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a (length 2*len(a)-1) using FFT."""
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n = len(x) + len(y) - 1
    n_fft = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X * Y, n=n_fft)[:n]
    return conv


# --------------------------------------------------------------------------- #
# Max of the self‑convolution (the value we really want to minimise)
# --------------------------------------------------------------------------- #
def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the a*a convolution."""
    return float(conv_self_fft(a).max())


# --------------------------------------------------------------------------- #
# Smooth‑max (log‑sum‑exp) and its gradient w.r.t. a.
# --------------------------------------------------------------------------- #
def smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Compute a smooth approximation of max(conv) and its gradient.

    The smooth max is log‑sum‑exp(beta * conv) / beta, but we only need the
    gradient, which can be expressed through the softmax weights.
    """
    n = a.shape[0]
    conv = conv_self_fft(a)          # length 2n-1
    maxc = conv.max()
    # softmax weights (numerically stable)
    w = np.exp(beta * (conv - maxc))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum
    # gradient = 2 * beta * (w * a_rev) convolution, sliced to length n
    a_rev = a[::-1]
    grad_full = conv_fft(w, a_rev)   # length 3n-2
    grad = 2.0 * beta * grad_full[n - 1 : n - 1 + n]
    # The smooth max value itself (not required for optimisation, only for
    # optional logging) is maxc + np.log(w_sum) / beta, but we return maxc.
    return maxc, grad


# --------------------------------------------------------------------------- #
# Adam optimiser on the smooth‑max surrogate.
# --------------------------------------------------------------------------- #
def adam_optimize(a_init: np.ndarray,
                  target_sum: float,
                  max_seconds: float,
                  start_time: float):
    """Adam optimisation of the smooth max of a*a with simplex projection."""
    a = a_init.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = max_convolution(a)

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # schedule for the smoothness parameter β
    beta = 0.5
    beta_max = 200.0
    beta_factor = 1.5

    # base learning rate (roughly 30 % of a typical entry)
    lr = 0.3 * target_sum / n

    while time.time() < start_time + max_seconds:
        # ----- smooth max & gradient -----
        cur_max, grad = smooth_max_and_grad(a, beta)

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # ----- feasibility -----
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        # ----- keep best -----
        cur_val = max_convolution(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam new best = {best_val:.6f} (n={n}, β={beta:.1f})")

        # ----- increase smoothness -----
        if beta < beta_max:
            beta = min(beta * beta_factor, beta_max)

        # occasional tiny random jitter to avoid stagnation
        if np.random.rand() < 0.01:
            jitter = np.random.randn(n) * (target_sum * 0.001 / n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = project_onto_simplex(a, target_sum)

    return best_a, best_val


# --------------------------------------------------------------------------- #
# Simple simulated‑annealing refinement (mass transfer between two slots)
# --------------------------------------------------------------------------- #
def simulated_annealing(a_init: np.ndarray,
                        target_sum: float,
                        max_seconds: float,
                        start_time: float):
    """Perform a cheap SA to possibly improve a given candidate."""
    a = a_init.copy()
    n = a.shape[0]

    cur_score = max_convolution(a)
    best_vec = a.copy()
    best_score = cur_score

    # temperature schedule (starts at ~5 % of the current value)
    T0 = max(0.01, cur_score * 0.05)
    T_min = 1e-6

    while time.time() < start_time + max_seconds:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1
        if a[i] <= 0.0:
            continue

        # move a random fraction (up to 50 %) of a[i] from i to j
        frac = np.random.rand() * 0.5
        delta = a[i] * frac

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = project_onto_simplex(a_new, target_sum)

        new_score = max_convolution(a_new)
        delta_score = new_score - cur_score

        elapsed = time.time() - start_time
        frac_time = max(0.0, 1.0 - elapsed / max_seconds)
        T = T_min + T0 * frac_time

        if delta_score <= 0.0 or np.random.rand() < np.exp(-delta_score / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_vec = a.copy()
                print(f"[{time.time():.2f}] SA new best = {best_score:.6f} (n={n})")

    return best_vec, best_score
