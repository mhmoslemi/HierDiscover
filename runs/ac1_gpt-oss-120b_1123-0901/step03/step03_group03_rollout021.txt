def project_simplex(v, s):
    # v is np array, s = target sum.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def optimize_adam(a_init, length, deadline, max_iters=10_000_000):
    n = length
    target = math.sqrt(2.0*n)
    a = a_init.copy()
    a = a * (target / a.sum())
    m = np.zeros_like(a); v = np.zeros_like(a); beta1...
    iteration = 0
    Nfft = 1 << ((2*n-1).bit_length())
    best_local_a = a.copy()
    best_local_score = max_conv_fft(a)
    while time.time() < deadline and iteration < max_iters:
        # compute conv
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[:2*n-1]
        M = conv.max()
        # compute weights using softmax
        tau = max(1e-8, 0.2 * M * math.exp(-iteration / 15000.0))
        w,_ = _softmax_weights(conv, tau)
        # rev gradient (subgradient)
        rev = _weighted_rev(a, w, Nfft)
        grad = 2.0 * rev
        iteration+=1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m/(1-beta1**iteration)
        v_hat = v/(1-beta2**iteration)
        lr = 0.05 / (1 + 1e-4 * iteration)
        a = a - lr * m_hat/(np.sqrt(v_hat)+1e-8)
        # project onto simplex
        a = project_onto_simplex(a, target)
        # occasional jitter
        if iteration % 5000 == 0:
            jitter = np.random.randn(n) * (0.001 * target / n)
            a = a + jitter
            a = project_onto_simplex(a, target)
        # check improvement
        cur_score = M
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_a = a.copy()
            print(f"[len {n}] iter {iteration} new local best {cur_score:.6f}")
    return best_local_a, best_local_score

def project_onto_simplex(v, s):
    # v: any real vector; s>0. Returns projection onto {x >= 0, sum x = s}
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]

"""
Propose a non‑negative step‑function (vector) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
* Initialise from the existing construction ``height_sequence_1`` (if it
  exists) and from a small portfolio of simple shapes (uniform,
  sine, exponential, Dirichlet samples).  All candidates are projected
  onto the feasible simplex {a ≥ 0, Σa = √(2 n)}.
* For each chosen length a few starting points are refined by a
  projected Adam optimiser that minimises a *smooth* version of the
  maximal self‑convolution.  The smooth max is realised with a soft‑max
  (log‑sum‑exp) whose temperature is annealed.
* The Adam step uses the exact gradient of the soft‑max:
      ∇ Lτ = 2·rev,
  where ``rev_i = Σ_k w_k a_{k‑i}`` and ``w`` are the soft‑max weights.
  After the gradient step the iterate is projected onto the simplex
  using the Euclidean projection (no ad‑hoc rescaling).
* A tiny Gaussian jitter is injected every few thousand iterations to
  escape plateaus.
* The optimisation runs until the global time budget (≈ 1000 s) is exhausted,
  always keeping the best sequence seen (measured by the exact
  ``max(a∗a)`` after normalisation, which equals the value returned by
  ``evaluate_sequence``).
* The algorithm explores several lengths (original, 1.25×, 1.5× and
  2×) and multiple random restarts; the most promising sequence is
  returned as a plain Python list.
"""

import time
import math
import numpy as np


def _softmax_weights(conv: np.ndarray, tau: float):
    """
    Stable soft‑max of the convolution values.
    Returns a weight vector w (summing to 1) and the smoothed maximum
    Lτ = τ·log ∑exp((conv)/τ).
    """
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """
    Compute rev_i = Σ_k w_k a_{k‑i}
    (indices outside 0…len(a)-1 are ignored) using FFTs.
    """
    n = a.shape[0]
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + n - 1]          # keep the valid part
    idx_start = 2 * n - 2
    rev = conv_rev[idx_start - np.arange(n)]
    return rev


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}.
    Returned array has the same shape as v.
    """
    if s < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero; return the zero vector with correct sum (impossible)
        return np.zeros_like(v)
    rho = ind[cond][-1]               # the last index where condition holds
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _max_conv_fft(a: np.ndarray) -> float:
    """
    Compute max(a ∗ a) using a single FFT pair.
    The result equals the objective value once Σa = √(2 n).
    """
    n = a.shape[0]
    Nfft = 1 << ((2 * n - 1).bit_length())
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
    return float(conv.max())


def _adam_optimize(
    a_init: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 10_000_000,
    tau_factor: float = 0.20,
    decay_rate: float = 20000.0,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser for the smooth maximal self‑convolution.
    Returns the best sequence found and its exact max‑convolution value.
    """
    target_sum = math.sqrt(2.0 * n)

    # start from the supplied a and enforce exact simplex constraint
    a = a_init.copy()
    a = _project_simplex(a, target_sum)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.05
    iteration = 0
    jitter_interval = 5000

    # FFT size (next power‑of‑two ≥ 2·n‑1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    best_local_a = a.copy()
    best_local_score = _max_conv_fft(a)

    while time.time() < deadline and iteration < max_iters:
        # 1) exact convolution via FFT
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # 2) soft‑max smoothing of the max
        tau = max(1e-12, tau_factor * M * math.exp(-iteration / decay_rate))
        w, Ltau = _softmax_weights(conv, tau)

        # 3) weighted reverse convolution (sub‑gradient of the smooth max)
        rev = _weighted_rev(a, w, Nfft)

        # 4) Adam gradient step (formula from the original derivation)
        coeff = 2.0                           # 4·n / (target_sum²) = 2
        const = 2.0 * Ltau / target_sum       # 4·n·Lτ / (target_sum³) = 2·Lτ/target_sum
        grad = coeff * rev - const

        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # 5) projection onto the feasible simplex
        a = _project_simplex(a, target_sum)

        # 6) occasional jitter to shake the iterate
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = a + jitter
            a = _project_simplex(a, target_sum)

        # 7) update best candidate (exact objective)
        cur_score = M
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_a = a.copy()
            print(f"[len {n:5d} iter {iteration:8d}] new local best = {best_local_score:.6f}")

    return best_local_a, best_local_score

        # ------------------------------------------------------------------
        # Determine a per‑length iteration budget (more for smaller n)
        max_iters = int(5_000_000 * 2100 / n)
        max_iters = max(5_000, max_iters)  # guarantee some work

        # Run the Adam optimiser from the chosen start
        seq, score = _adam_optimize(
            a_init=local_best_a,
            n=n,
            deadline=deadline,
            max_iters=max_iters,
        )

        # Update global best if improved
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[global] new best length {n} score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety check – evaluate with the provided function
    if best_seq is None:
        n = 2100
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")
    else:
        # Convert to plain list for the harness and compute final score
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[done] final score = {final_score:.6f}, length = {best_seq.shape[0]}")

    return best_seq.tolist()
```

This implementation builds on the earlier soft‑max Adam approach but adds several
enhancements:

* **Euclidean projection onto the simplex** – a numerically sound way of
  enforcing `a ≥ 0` and `Σa = √(2 n)` after each update.
* **Richer initialisation** – uniform, sine, exponential and Dirichlet‑sampled
  vectors, plus interpolation of the historic best solution.
* **Dynamic iteration budgeting** – more iterations for shorter sequences,
  guaranteeing a substantial amount of optimisation time even under the
  1000 s wall‑clock limit.
* **Logging** – progress messages show per‑length improvements and the current
  global best.

The optimizer runs until the overall deadline, constantly tracking the best
candidate.  The returned list conforms to the expected format for the scoring
harness.
