"""
Improved search for low‑peak self‑convolution step functions.

Algorithm
---------
* Represent the sequence as a probability vector p (Σp = 1) and
  minimise 2·n·max(p∗p).  The scaling factor √(2·n) is applied at the
  very end – the objective is invariant to overall scaling.
* Optimise p by Exponentiated‑Gradient (mirror descent on the simplex)
  using the *exact* sub‑gradient of the max‑convolution rather than a
  soft‑max approximation.  For a given p we compute the full
  auto‑convolution (FFT, O(n log n)) and identify the active shift(s)
  where the maximum is attained (a tolerance of 1e‑12 relative to the
  maximum is used).  The sub‑gradient is
        grad = 4·n· Σ_{k active}  p_{k‑i}
  which is efficiently obtained by a single FFT‑based correlation.
* An Adam‑style adaptive step‑size is applied to the log‑probabilities,
  followed by a projection onto the probability simplex.
* The optimisation is performed on a multiscale schedule:
  500 → 1000 → 2000 → 4000 → 8000.  For each new length the current
  best probability vector is interpolated to the larger grid.
* Several random restarts (Dirichlet draws) are performed at each
  length; the best overall sequence is returned.
* Small Gaussian jitter is added periodically to avoid stagnation.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """Linear self‑convolution p∗p using a single FFT of size N."""
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k·p_{k‑i}   (i = 0..n‑1)
    using a single FFT (correlation of p with reversed w).
    """
    n = p.shape[0]
    L = w.shape[0]                     # = 2·n‑1
    # FFT of p (size N)
    P = np.fft.rfft(p, n=N)
    # FFT of reversed weights
    W_rev = np.fft.rfft(w[::-1], n=N)
    # Convolution of p with reversed w
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    conv_rev = conv_rev[: n + L - 1]    # keep the linear part
    # Extract rev = conv_rev[L‑n : L] and reverse the order
    rev = conv_rev[L - n : L][::-1]
    return rev


def _optimize_eg_exact(
    n: int,
    p_start: np.ndarray,
    best_score_global: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimisation (Adam on log‑probabilities)
    for a fixed length n using the exact sub‑gradient of the max
    convolution.
    Returns the best probability vector (sum = 1) and its objective
    value 2·n·max(p∗p).
    """
    # hyper‑parameters
    max_iters = 250_000                 # safety cap
    jitter_interval = 3000
    jitter_scale = 0.002
    tol_rel = 1e-12                     # tolerance for active set

    # initialise log‑probabilities (avoid log(0))
    eps = 1e-30
    logp = np.log(p_start + eps)

    # FFT size works for both the self‑convolution (2n‑1) and the
    # correlation with the weight vector (3n‑2).
    N = _next_pow2(3 * n - 2)

    # Adam state
    m = np.zeros_like(p_start)
    v = np.zeros_like(p_start)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    base_lr = 1.0                       # effective step on log‑space

    best_local_score = float("inf")
    best_local_p = None

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ---- project onto simplex ----
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)      # numerical stability
        p = exp_shift / exp_shift.sum()

        # ---- self‑convolution ----
        conv = _full_convolution_fft(p, N)
        M = conv.max()

        cur_score = 2.0 * n * M
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_p = p.copy()
            print(f"[len {n:5d} iter {it:7d}] new local best = {cur_score:.6f}")

        # ---- exact sub‑gradient of max (active set) ----
        # mask of active indices
        mask = conv >= M - max(tol_rel * M, 1e-15)
        w = mask.astype(float)                  # weights = 1 on active shifts
        rev = _weighted_rev_fft(p, w, N)        # Σ_k w_k·p_{k‑i}

        # gradient of 2·n·max(p∗p) w.r.t. p (in log‑space we need the raw gradient)
        grad = 4.0 * n * rev                     # ∇_p (2·n·M)

        # ---- Adam update on log‑probabilities ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        logp -= step

        # occasional jitter to escape plateaus
        if it % jitter_interval == 0:
            logp += np.random.randn(n) * jitter_scale

    if best_local_p is None:        # should never happen
        best_local_p = p_start.copy()
        best_local_score = 2.0 * n * _full_convolution_fft(best_local_p, N).max()

    return best_local_p, best_local_score

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Initialise from the previously discovered construction if present
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        # fall back to a simple uniform vector of modest length
        base = np.full(500, 1.0, dtype=float)

    # Normalise to the canonical sum S = sqrt(2·n) (so that Σa = S)
    n0 = base.shape[0]
    S0 = math.sqrt(2.0 * n0)
    base = base * (S0 / base.sum())

    # Global best tracking
    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n0}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Multiscale schedule – lengths that comfortably fit in the time budget
    length_schedule = [500, 1000, 2000, 4000, 8000]
    length_schedule = [l for l in length_schedule if l >= 300]
    length_schedule = sorted(set(length_schedule))

    for n in length_schedule:
        if time.time() >= deadline:
            break

        print(f"[run] start optimisation for length = {n}")

        # ---- initialise probability vector for this length ----
        # Interpolate the current best probability vector to the new size.
        p_old = best_seq / math.sqrt(2.0 * best_seq.shape[0])   # recover p
        old_x = np.linspace(0.0, 1.0, p_old.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        p_init = np.interp(new_x, old_x, p_old)
        p_init = np.maximum(p_init, 0.0)
        if p_init.sum() == 0.0:
            p_init = np.full(n, 1.0 / n)
        else:
            p_init /= p_init.sum()

        # ----- Run a few random restarts (including the interpolated start) -----
        attempts = 3
        for attempt in range(attempts):
            if time.time() >= deadline:
                break

            if attempt == 0:
                p_start = p_init.copy()
            else:
                # Dirichlet‑distributed random start (concentrated but diverse)
                concentration = 0.5   # low concentration → sparse starts
                p_start = np.random.gamma(concentration, 1.0, size=n)
                p_start /= p_start.sum()

            p_opt, local_score = _optimize_eg_exact(
                n=n,
                p_start=p_start,
                best_score_global=best_score,
                deadline=deadline,
            )

            # Convert back to the original scale (a = S·p)
            S = math.sqrt(2.0 * n)
            a_opt = p_opt * S
            cur_score = evaluate_sequence(a_opt.tolist())

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_opt.copy()
                print(f"[len {n}] improved global best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – if nothing has been found (unlikely), fall back.
    if best_seq is None or best_seq.size == 0:
        n = 500
        S = math.sqrt(2.0 * n)
        best_seq = np.full(n, S / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
