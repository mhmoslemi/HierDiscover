"""
Propose a non‑negative step‑function (height sequence) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The evaluator forces the sum to be √(2 n), therefore the objective equals the
maximum entry of the self‑convolution.  The algorithm combines several
techniques:

1. **Random restarts with diverse lengths** (n ≈ 1500‑5000).  Each restart
   receives a time slice and starts from a symmetric power‑law/​sinusoid
   shape, possibly warm‑started from the current global best.

2. **Projected gradient descent on a smoothed max**.
   * The smooth‑max uses a temperature τ that decays slowly.
   * Gradient of the weighted convolution is obtained by a single FFT‑based
     cross‑correlation.
   * Nesterov‑style momentum is employed.
   * After each gradient step the vector is projected onto the simplex
     Σ a_i = √(2 n) and forced to be mirror‑symmetric (a_i = a_{n‑1‑i}).

3. **Occasional random shake‑ups** to escape plateaus.

4. **Local refinement** after the main budget is exhausted:
   small pairwise mass transfers are accepted only if they improve the exact
   objective (computed with the evaluator).  This hill‑climbing stage can
   squeeze out the last bits of improvement.

The whole search respects the 1000 s budget and prints progress.  The best
sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – Full linear convolution via real FFT (fast for any length)
# ----------------------------------------------------------------------
def full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())  # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative rounding errors
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Gradient of the (smoothed) max of the self‑convolution
# ----------------------------------------------------------------------
def weighted_gradient(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Given a (length n) and a weight vector w (length 2n‑1) that sums to 1,
    compute grad_i = 2 * Σ_j a_j * w_{i+j}.
    """
    n = a.shape[0]
    w_rev = weights[::-1]
    conv2 = full_convolution_fft(a, w_rev)            # length = 3n‑2
    idx = (2 * n - 2) - np.arange(n)                  # slice giving Σ_j a_j w_{i+j}
    grad = 2.0 * conv2[idx]
    return grad


# ----------------------------------------------------------------------
# Optimise a single length by projected (Nesterov) gradient descent
# ----------------------------------------------------------------------
def optimise_one_length(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 0.001,
    step_factor: float = 0.8,
    step_decay: float = 0.9997,
    momentum_beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Perform a projected gradient descent on a smoothed max for a fixed length.
    Returns the best sequence encountered (as numpy array) and its
    exact evaluation (via evaluate_sequence).
    """
    target_sum = np.sqrt(2.0 * n)          # forced L1 norm
    a = project_onto_simplex(init.astype(np.float64), target_sum)

    # enforce exact symmetry to start
    a = (a + a[::-1]) * 0.5
    a = project_onto_simplex(a, target_sum)

    best_seq = a.copy()
    conv = full_convolution_fft(a, a)
    best_score = 2.0 * n * conv.max() / (target_sum ** 2)   # equals conv.max()
    velocity = np.zeros_like(a)

    iteration = 0
    while time.time() < run_deadline:
        # ----- convolution and max -----
        conv = full_convolution_fft(a, a)
        max_val = conv.max()

        # ----- smooth‑max weights -----
        tau = max(
            init_tau_factor * max_val * (0.97 ** iteration),
            min_tau_factor * max_val,
        )
        # soft‑max centred at max_val for numerical stability
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient (cross‑correlation) -----
        grad = weighted_gradient(a, w)

        # ----- momentum (Nesterov style) -----
        if momentum_beta > 0.0:
            velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
            direction = velocity
        else:
            direction = grad

        # ----- step size scaling -----
        max_abs = np.max(np.abs(direction)) + 1e-12
        eta = step_factor * max_val / max_abs
        a_new = a - eta * direction

        # ----- feasibility projection -----
        a_new = np.maximum(a_new, 0.0)
        # enforce symmetry
        a_new = (a_new + a_new[::-1]) * 0.5
        a_new = project_onto_simplex(a_new, target_sum)

        # ----- evaluate candidate -----
        conv_new = full_convolution_fft(a_new, a_new)
        score_new = 2.0 * n * conv_new.max() / (target_sum ** 2)

        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {best_score:.6f}")

        a = a_new
        step_factor *= step_decay
        iteration += 1

        # occasional random shake‑up to jump out of shallow minima
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (target_sum * 5e-4 / n)
            a = project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            a = (a + a[::-1]) * 0.5
            step_factor *= 0.9

    # final exact evaluation (guards against rounding differences)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Local refinement by tiny pairwise mass transfers
# ----------------------------------------------------------------------
def refine_sequence(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Hill‑climb the exact objective by moving small amounts of mass between two
    positions.  The sum stays constant, non‑negativity is enforced.
    """
    n = seq.shape[0]
    a = seq.copy()
    best_score = evaluate_sequence(a.tolist())
    # start with a relative step size; shrink gradually
    epsilon = np.sqrt(2.0 * n) * 0.02 / n   # ~2 % of average entry
    it = 0
    while time.time() < deadline and it < 200000:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        if a[i] == 0.0:
            continue
        # propose moving a random fraction of a[i] (capped by epsilon)
        delta = min(a[i], epsilon) * np.random.rand()
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta
        # enforce symmetry after each move (keeps search space reduced)
        a_candidate = (a_candidate + a_candidate[::-1]) * 0.5
        # No need to re‑project: sum unchanged, non‑negativity holds by construction
        score_candidate = evaluate_sequence(a_candidate.tolist())
        if score_candidate < best_score:
            best_score = score_candidate
            a = a_candidate
            epsilon *= 0.995   # tighten step as we improve
            print(f"[{time.time():.2f}] refinement improved to {best_score:.6f}")
        else:
            epsilon *= 1.001   # very slowly increase to keep exploration
        it += 1
    return a, best_score

# Main entry point required by the evaluator

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the evaluation
    function.  The routine runs for at most ``budget_s`` seconds and returns
    the best sequence found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2   # small safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Warm‑start from a previously discovered construction, if any.
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        warm_score = evaluate_sequence(hs_arr.tolist())
        if warm_score < best_score:
            best_score = warm_score
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] warm‑start from global seed = {warm_score:.6f}")
    except Exception:
        pass  # nothing to warm‑start from

    # --------------------------------------------------------------
    # Main restart loop – each restart gets a time slice.
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        # Choose a problem size; favor larger n early on.
        if remaining > 400:
            n = np.random.randint(1800, 5000 + 1)
        elif remaining > 150:
            n = np.random.randint(1300, 4000 + 1)
        else:
            n = np.random.randint(900, 3000 + 1)

        # ----- initial shape -----
        # Symmetric power‑law plus optional sinusoidal taper.
        p = np.random.uniform(0.35, 0.75)          # exponent
        idx = np.arange(1, n + 1, dtype=float)
        init = idx ** (-p)

        # Optional sinusoidal weight to encourage symmetry and zero ends.
        if np.random.rand() < 0.4:
            init *= np.sin(np.pi * idx / (n + 1)) ** np.random.uniform(0.5, 2.0)

        # Tiny jitter to break ties.
        init += np.random.rand(n) * 1e-5

        # Occasionally warm‑start from the current global best (truncated/padded).
        if best_seq is not None and np.random.rand() < 0.45:
            bs = best_seq
            if bs.size >= n:
                init = bs[:n].copy()
            else:
                pad = np.random.rand(n - bs.size) * 1e-5
                init = np.concatenate([bs, pad])

        # Allocate a time slice for this restart.

        # Run the optimiser for this length.

            n,

            )
            if best_score < 1.503:      # target achieved – stop early
                break

    # ------------------------------------------------------------------
    # Final local refinement on the global best (uses remaining budget).
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:

        refined_seq, refined_score = refine_sequence(best_seq, refine_deadline)
        if refined_score < best_score:

            best_seq = refined_seq

            )

    # ------------------------------------------------------------------
    if best_seq is None:

        target = np.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    # Return a plain list for the evaluator.
    return best_seq.tolist()
