"""
smooth‑max Adam optimiser for the autoconvolution L∞ problem.

We minimise    max_k (a * a)[k]   under a_i ≥ 0 and  Σ a_i = √(2n).
The non‑smooth “max” is approximated by a soft‑max
    f_α(a) = (1/α)·log Σ exp(α·(a*a)[k]),
with a schedule that slowly raises α (more “max‑like”) while an
Adam optimiser moves the vector.  After the Adam phase a cheap
coordinate‑swap hill‑climb refines the solution.  Several lengths and
random restarts are tried; the best sequence found before the time
budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _full_autoconv_fft(a: np.ndarray, N: int) -> np.ndarray:
    """
    Full (linear) autocorrelation a * a using an FFT of size N.
    Returns the first 2·len(a)-1 entries.
    """
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, p: np.ndarray, N: int) -> np.ndarray:
    """
    Gradient of  f_α(a) = (1/α) log Σ exp(α·conv)  w.r.t. a,
    where p_i = soft‑max_i(α·conv_i).
    The formula is  grad = 2·corr(a, p) (linear cross‑correlation).
    Both a and p are real; the correlation is computed via FFT.
    """
    A = np.fft.rfft(a, n=N)
    P = np.fft.rfft(p, n=N)
    # linear correlation (no wrap‑around because N ≥ len(a)+len(p)-1)
    corr = np.fft.irfft(np.conj(A) * P, n=N)[: a.shape[0]]
    return 2.0 * corr


# ----------------------------------------------------------------------
def _adam_smooth_max(
    a_init: np.ndarray,
    n: int,
    target_sum: float,
    deadline: float,
    max_iter: int = 200_000,
    init_lr: float = 0.05,
    lr_decay: float = 1e-4,
    init_alpha: float = 0.5,
    alpha_factor: float = 1.02,
    alpha_max: float = 30.0,
    jitter_interval: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max objective.
    Returns the best vector seen and its maximal autocorrelation value.
    """
    # normalise start
    a = a_init.copy()
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # FFT size (must accommodate a (n) + conv (2n‑1) → 3n‑2)
    N = _next_pow2(3 * n - 2)

    best_a = a.copy()
    # initial max‑convolution value
    conv = _full_autoconv_fft(a, N)
    best_val = float(conv.max())

    alpha = init_alpha
    iteration = 0

    while time.time() < deadline and iteration < max_iter:
        # ----- 1. current convolution & soft‑max weights -----------------
        conv = _full_autoconv_fft(a, N)               # length 2n‑1
        cur_max = float(conv.max())
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[len {n:5d} iter {iteration:7d}] new best max = {best_val:.6f}")

        # soft‑max probabilities (stable)
        scaled = alpha * conv
        scaled -= scaled.max()          # for numerical stability
        exp_scaled = np.exp(scaled)
        p = exp_scaled / exp_scaled.sum()

        # ----- 2. gradient (smooth max) --------------------------------
        grad = _gradient_softmax(a, p, N)

        # ----- 3. Adam update -----------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = init_lr / (1.0 + lr_decay * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- 4. projection (non‑negative, fixed L1) -----------------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:   # practically never happens
            a = np.full_like(a, target_sum / n)

        # ----- 5. schedule α (make the approximation sharper) -----
        if iteration % 1000 == 0:
            alpha = min(alpha_max, alpha * alpha_factor)

        # ----- 6. occasional jitter to escape plateaus ---------------
        if jitter_interval and iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

    return best_a, best_val

def _local_search(
    a_start: np.ndarray,
    n: int,
    target_sum: float,
    cur_best: float,
    deadline: float,
    max_proposals: int = 20_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb: repeatedly move a tiny amount of mass from one
    index to another; keep the move only if the maximal autocorrelation
    decreases.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_val = cur_best
    N = _next_pow2(3 * n - 2)

    proposals = 0
    while time.time() < deadline and proposals < max_proposals:
        proposals += 1
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move up to 5 % of donor mass
        eps = a[i] * np.random.rand() * 0.05
        if eps <= 0.0:
            continue

        new_vec = a.copy()
        new_vec[i] -= eps
        new_vec[j] += eps

        # maintain feasibility (sum already preserved)
        conv = _full_autoconv_fft(new_vec, N)
        new_max = float(conv.max())
        if new_max < best_val - 1e-12:
            best_val = new_max
            best_a = new_vec.copy()
            a = new_vec
            print(f"[local] improvement to {best_val:.6f} (prop {proposals})")
    return best_a, best_val

        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        base = None

    best_seq = None
    best_score = float("inf")

    if base is not None:
        n0 = base.shape[0]
        target0 = math.sqrt(2.0 * n0)
        base = base * (target0 / base.sum())
        best_seq = base.copy()
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[init] length={n0}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore
    n0 = base.shape[0] if base is not None else 2100
    length_factors = [0.9, 1.0, 1.2, 1.5, 2.0]
    lengths = sorted({int(n0 * f) for f in length_factors})
    extra_lengths = [3000, 4000, 5000, 6000, 8000]
    lengths = sorted(set(lengths + extra_lengths))
    lengths = [l for l in lengths if 500 <= l <= 10_000]

    # ------------------------------------------------------------------
    # Main optimisation loop
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # number of restarts per length
        restarts = 3
        for r in range(restarts):
            if time.time() >= deadline:
                break

            # ---- initialise ------------------------------------------------
            if r == 0 and base is not None:
                # linear interpolation from the existing base vector
                old_x = np.linspace(0.0, 1.0, n0)
                new_x = np.linspace(0.0, 1.0, n)
                a0 = np.interp(new_x, old_x, base)
                a0 = np.maximum(a0, 0.0)
            else:
                # random positive start (Dirichlet‑like)
                a0 = np.random.rand(n)
                a0 = np.maximum(a0, 0.0)

            a0 = a0 * (target_sum / a0.sum())

            # ---- Adam optimisation (smooth max) ---------------------------
            a_opt, max_val = _adam_smooth_max(
                a_init=a0,
                n=n,
                target_sum=target_sum,
                deadline=deadline,
                max_iter=200_000,
                init_lr=0.06,
                lr_decay=1e-4,
                init_alpha=0.5,
                alpha_factor=1.02,
                alpha_max=35.0,
                jitter_interval=3000,
            )

            # ---- cheap local refinement -----------------------------------
            a_opt, max_val = _local_search(
                a_start=a_opt,
                n=n,
                target_sum=target_sum,
                cur_best=max_val,
                deadline=deadline,
                max_proposals=30_000,
            )

            # ---- update global best ----------------------------------------
            if max_val < best_score:
                best_score = max_val
                best_seq = a_opt.copy()
                print(f"[global] new overall best: length={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    if best_seq is None:
        n = n0
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # return a plain‑list; the outer harness will call evaluate_sequence()
    return best_seq.tolist()
