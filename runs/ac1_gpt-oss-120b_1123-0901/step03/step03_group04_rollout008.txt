def full_conv_fft(a):
    # returns full convolution a * a using next power of 2.
    N = 1 << ((2 * len(a) - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)
    return conv[:2*len(a)-1]

def weighted_corr_fft(w, a):
    # compute gradient = 2 * correlation w * a (valid range).
    n = len(a)
    a_rev = a[::-1]
    N = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=N)
    Ar = np.fft.rfft(a_rev, n=N)
    corr = np.fft.irfft(W * Ar, n=N)
    # extract indices n-1 to n-1 + n -1 = 2n-2
    start = n - 1
    grad = 2 * corr[start:start + n]
    return grad

"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm
---------
* The objective (after fixing the ℓ1‑norm to √(2 n)) is simply the maximum
  entry of the full auto‑convolution  a*a .
* Directly optimising the non‑smooth max is difficult, therefore we minimise
  a smooth approximation:
        fλ(a) = log‑sum‑exp( λ·(a*a) ) / λ
  which converges to the true max as λ → ∞.
* For a given λ we obtain a cheap gradient:
        ∇a fλ = 2·correlation( w , a )
  where w is the soft‑max weight vector
        w_k = exp( λ·conv_k ) / Σ_j exp( λ·conv_j )
  and correlation denotes the “valid’’ correlation of the weight vector
  with the current sequence.
  All convolutions / correlations are computed by FFT for speed.
* An Adam optimiser is run on this smooth objective, slowly increasing λ
  (a homotopy) so that the optimiser first sees a well‑behaved landscape and
  finally refines the true max‑value.
* After the Adam phase a cheap greedy local‑search (transfer a tiny amount
  of mass between two random positions) is performed to polish the result.
* Several random restarts are used; the best sequence seen (according to the
  exact max‑convolution) is kept.  The whole search respects the 1000 s wall‑clock
  budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """
    Full convolution a*a using FFT (real‑to‑real).  Returns the array of
    length 2·n‑1 (no truncation).  The FFT length is the next power of two.
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)
    return conv[: 2 * n - 1]


def _conv_max_fft(a: np.ndarray):
    """Return max value of the full convolution and the full convolution array."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)
    conv = conv[: 2 * n - 1]
    return float(conv.max()), conv


def _weighted_corr_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute 2·correlation(w, a) (valid part, length n) using FFT.
    Mathematically:   g_i = 2·Σ_k w_{i+k}·a_k   for i=0,…,n‑1.
    """
    n = a.shape[0]
    a_rev = a[::-1]                      # reversed a
    # convolution length: len(w) + len(a_rev) - 1 = (2n‑1) + n - 1 = 3n‑2
    N = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(w, n=N)
    Ar = np.fft.rfft(a_rev, n=N)
    corr = np.fft.irfft(W * Ar, n=N)
    start = n - 1                         # index of first “valid’’ element
    grad = 2.0 * corr[start:start + n]    # length n
    return grad


# ----------------------------------------------------------------------
def _adam_optimize_smooth(init_a: np.ndarray,
                          n: int,
                          target_sum: float,
                          deadline: float,
                          max_iters: int = 10_000_000):
    """
    Adam optimiser on the smooth approximation of max‑convolution.
    Returns the best vector (according to the exact max‑convolution) and its
    exact max value.
    """
    a = init_a.copy()
    # enforce exact ℓ1‑norm
    s = a.sum()
    if s > 0:
        a *= (target_sum / s)
    else:
        a[:] = target_sum / n

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.02
    jitter_interval = 4000
    jitter_scale = 0.001

    # schedule for the smoothness parameter λ
    lambda_sched = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0]
    stage_len = 3000                     # iterations per λ‑stage

    iteration = 0
    best_a = a.copy()
    best_val, _ = _conv_max_fft(best_a)

    while time.time() < deadline and iteration < max_iters:
        # choose current λ
        lam = lambda_sched[min(iteration // stage_len,
                              len(lambda_sched) - 1)]

        # full convolution (needed for both objective & weights)
        conv = _conv_full_fft(a)
        M = float(conv.max())            # exact max for bookkeeping

        # -------- smooth max and its gradient --------
        # weights = soft‑max( λ·conv )   (stable computation)
        shift = conv - M                 # ≤ 0
        exp_shift = np.exp(lam * shift)
        w = exp_shift / exp_shift.sum()  # shape (2n‑1,)

        # gradient of the smooth objective w.r.t a
        grad = _weighted_corr_fft(w, a)

        # -------- Adam update --------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = base_lr / (1.0 + 1e-4 * iteration)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # -------- projection onto feasible set (non‑negative, fixed ℓ1) --------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a *= (target_sum / s)
        else:
            a[:] = target_sum / n

        # -------- occasional jitter to escape plateaus --------
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * jitter_scale * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= (target_sum / a.sum())

        # -------- keep the best exact solution --------
        if M < best_val:
            best_val = M
            best_a = a.copy()
            print(f"[len {n:5d} iter {iteration:7d}] new best exact M = {best_val:.6f}")

    return best_a, best_val

def _local_search(a: np.ndarray,
                  n: int,
                  target_sum: float,
                  steps: int,
                  deadline: float):
    """
    Very cheap stochastic local search: repeatedly move a tiny fraction of
    mass from one index to another if the move reduces the exact max‑convolution.
    """
    best = a.copy()
    best_val, _ = _conv_max_fft(best)

    for step in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * 0.02
        if delta <= 0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # projection onto the simplex is not needed – sum stays constant
        cand = np.maximum(cand, 0.0)

        M, _ = _conv_max_fft(cand)
        if M < best_val:
            best = cand
            best_val = M

    return best, best_val
