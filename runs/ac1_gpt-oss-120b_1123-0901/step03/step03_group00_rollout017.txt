    # Build matrix shape (m, n)

def cross_corr(w, a):
    n = a.shape[0]
    # compute correlation w ★ a: length n
    # w length Lc = 2*n - 1
    # pad both sequences to length L = power of two >= (len(w) + n -1)
    L = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    # convolution of w with reversed a
    rev_a = a[::-1]
    fw = np.fft.rfft(w, L)
    fr = np.fft.rfft(rev_a, L)
    corr = np.fft.irfft(fw * fr, L)
    # extract indices start=n-1 ... start+n-1
    start = n - 1
    return corr[start:start + n]

"""
search for a low‑value step‑function (non‑negative sequence) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)²)

where * denotes linear convolution.  The algorithm consists of two phases:

1.  A large‑scale “smooth‑max Adam” optimisation on the simplex (same idea as
    the reference code, but with a few tuned hyper‑parameters, more restarts
    and a longer stagnation timeout).

2.  A short “LP refinement” stage.  For a fixed sequence f the LP

        maximise    1ᵀg
        subject to   f * g ≤ M        (component‑wise)
                     g ≥ 0

   where * is linear convolution and M = max(f*f), yields a direction g that
   can be blended into f while keeping the convolution bound.  Repeating a
   few blending steps lowers the peak of the self‑convolution further.

Both phases run under the global time budget (default 1000 s).  The best
sequence discovered over several lengths (256 … 2048) is returned as a plain
list.  The code uses only NumPy, SciPy and the standard library.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Helper functions (all defined at top level – required by the platform)
# ----------------------------------------------------------------------


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns a probability vector."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑two length."""
    n = a.shape[0]
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute the cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) defined as in the reference code.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    L = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, L)
    fa = np.fft.rfft(rev_a, L)
    corr_full = np.fft.irfft(fw * fa, L)
    start = n - 1
    return corr_full[start:start + n]


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the linear convolution matrix A such that (A @ g)[k] = (f * g)[k].
    The shape is (2·n-1, n) where n = len(f).
    """
    n = f.shape[0]
    m = 2 * n - 1
    rows = np.arange(m)[:, None]          # (m,1)
    cols = np.arange(n)[None, :]          # (1,n)
    idx = rows - cols                     # (m,n) contains the index into f
    valid = (idx >= 0) & (idx < n)
    A = np.zeros((m, n), dtype=np.float64)
    A[valid] = f[idx[valid]]
    return A


def _objective_val(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) for a vector a with sum(a)=1."""
    n = a.shape[0]
    conv = _conv_fft(a)
    return 2.0 * n * conv.max()


def _lp_refine(a: np.ndarray,
               max_iter: int = 10,
               alpha: float = 0.3,
               deadline: float = None) -> tuple[np.ndarray, float]:
    """
    Perform a few Frank‑Wolfe / LP refinement steps on a given probability vector.
    Returns the (potentially) improved vector and its objective value.
    """
    n = a.shape[0]
    best_val = _objective_val(a)
    best_a = a.copy()

    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break

        # current convolution bound
        conv = _conv_fft(a)
        M = conv.max()

        # Build the LP matrix A (size (2n‑1)×n)
        A = _build_conv_matrix(a)

        # maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # Solve the LP
        res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "disp": False},
        )
        if not res.success:
            # if the LP fails we abort the refinement for this vector
            break

        g = res.x
        s = g.sum()
        if s <= 0:
            break
        g = g / s                     # normalise to a probability vector

        # convex blend
        a_new = (1.0 - alpha) * a + alpha * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new == 0.0:
            break
        a_new = a_new / s_new

        val_new = _objective_val(a_new)
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            # keep the improvement for the next LP step
            a = a_new
            # make the blend a little smaller over time
            alpha = max(alpha * 0.9, 0.05)
        else:
            # no improvement – stop early
            break

    return best_a, best_val


def _initial_candidates(n: int) -> list[np.ndarray]:
    """
    Produce a handful of deterministic starting probability vectors for a given length.
    They are all normalised to sum to 1.
    """
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # symmetric triangular (increase then decrease)
    tri = np.arange(1, n // 2 + 1, dtype=np.float64)
    if n % 2 == 1:
        tri = np.concatenate([tri, tri[-2::-1]])
    else:
        tri = np.concatenate([tri, tri[::-1]])
    cand.append(tri / tri.sum())

    # sinusoidal (small at the edges, larger in the middle)
    x = np.arange(n, dtype=np.float64)
    sin = np.sin(np.pi * (x + 0.5) / n)
    cand.append(sin / sin.sum())

    # power‑law family (p = 0.5, 1, 2) – symmetric
    for p in (0.5, 1.0, 2.0):
        base = (np.minimum(x, n - 1 - x) + 1) ** p
        cand.append(base / base.sum())

    return cand


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative sequence with a low evaluation score.
    The routine respects the given time budget, prints progress and finally
    returns the best sequence (as a plain Python list) that it found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper: evaluate a candidate (objective) – uses our own fast implementation.
    def _update_best(a: np.ndarray, val: float, tag: str = "") -> None:
        nonlocal best_seq, best_val
        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            print(f"{tag} NEW BEST: {val:.6f}")

    # ------------------------------------------------------------------
    # Problem sizes – we stay modest because the LP refinement scales quadratically.
    lengths = [256, 512, 1024, 2048]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # Determine a sensible number of restarts (more restarts for smaller n)
        if n <= 512:
            restarts = 6
        elif n <= 1024:
            restarts = 5
        else:
            restarts = 4

        # pre‑compute FFT length for the convolution (used in the smooth‑max loop)
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))   # satisfies w length + a length - 1

        for r in range(restarts):
            if time.time() > deadline:
                break

            # ------------------------------------------------------------------
            # Initialise the probability vector a (sum = 1)
            # try a deterministic seed first, otherwise a random Dirichlet
            init_candidates = _initial_candidates(n)
            if r < len(init_candidates):
                a = init_candidates[r].copy()
                tag = f"[n={n} restart={r}] deterministic start"
            elif best_seq is not None and len(best_seq) == n:
                a = np.array(best_seq, dtype=np.float64)
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 0.01:
                    a = np.random.dirichlet(np.ones(n))
                else:
                    a = a / s
                tag = f"[n={n} restart={r}] warm‑start from global best"
            else:
                a = np.random.dirichlet(np.ones(n))
                tag = f"[n={n} restart={r}] random start"

            # convert to logits for the Adam optimiser
            logits = np.log(a + 1e-15)

            # Adam state
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.07

            # smooth‑max temperature
            lam = 0.5
            lam_factor = 1.025

            it = 0
            max_iters = 25000
            last_improve = time.time()
            # ------------------------------------------------------------------
            while time.time() < deadline and it < max_iters:
                # ----- logits → probability (softmax) -----
                a = _softmax(logits)

                # ----- convolution and smooth‑max weights -----
                conv = _conv_fft(a)                     # length 2·n‑1
                max_c = conv.max()
                # smooth‑max weights
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w = w / w_sum

                # ----- gradient of the surrogate w.r.t. a -----
                grad_a = 2.0 * _cross_corr(w, a)        # shape (n,)

                # ----- back‑prop through the softmax (gradient wrt logits) -----
                avg = np.dot(a, grad_a)
                grad_logits = a * (grad_a - avg)

                # ----- Adam update -----
                it += 1
                m = beta1 * m + (1.0 - beta1) * grad_logits
                v = beta2 * v + (1.0 - beta2) * (grad_logits * grad_logits)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)
                logits = logits - lr * m_hat / (np.sqrt(v_hat) + eps)

                # ----- temperature schedule -----
                lam = min(lam * lam_factor, 1e6)

                # ----- exact objective (sum(a)=1) -----
                val = 2.0 * n * max_c
                _update_best(a, val, tag + f" iter={it:05d}")

                # ----- stagnation handling -----
                if time.time() - last_improve > 6.0:
                    # small random perturbation of the logits to escape flat regions
                    logits += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()
                    # (no break – continue optimisation)

            # ------------------------------------------------------------------
            # LP refinement – run only if we still have time left
            if time.time() < deadline:
                refined_a, refined_val = _lp_refine(
                    a,
                    max_iter=12,
                    alpha=0.35,
                    deadline=deadline,
                )
                _update_best(refined_a, refined_val, f"[n={n} LP‑refine]")

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded (highly unlikely) return a uniform vector.
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Convert to plain Python list as required by the external tester.
    return best_seq.tolist()
