"""
search.py – Minimise the peak of a self‑convolution step‑function.

Algorithm
---------
* **Parameterisation** – work with a probability vector `p` on the simplex
  (p_i ≥ 0, Σp_i = 1) and set the actual coefficients a_i = S·p_i where
  S = √(2·n).  The target value equals  max_k (a*a)_k = S²·max_k(p*p)_k,
  i.e. minimising the *soft‑max* of the auto‑convolution of `p`.

* **Smooth objective** – for a temperature β the smooth approximation of
  `max` is  

        φ_β(p) = max(conv) + log( Σ exp(β·(conv−max(conv))) ) / β

  where `conv = p * p` (full convolution).  Its gradient w.r.t. `p`
  is `2·rev`, where `rev_i = Σ_k w_k·p_{k−i}` and `w_k` are the soft‑max
  weights defined from `conv`.

* **Optimization** – we optimise the logits `x` of the softmax
  (`p = softmax(x)`).  Using the chain rule the gradient w.r.t. `x` is

        ∇_x φ_β = p ⊙ ( g − (p·g) ),   g = 2·rev .

  The problem is smooth and unconstrained in `x`.  We call
  ``scipy.optimize.minimize`` with the L‑BFGS‑B method.  Several
  increasing β‑phases (0.5 → 5 → 20 → 80) are performed, each phase warm‑starting
  from the solution of the preceding phase.  For every length we perform a few
  random restarts (including the current best interpolated solution).

* **Multi‑scale schedule** – start with a modest length (≈500) and
  progressively up‑sample the current best probability vector to larger
  lengths (up to 8000).  Interpolation on the unit interval preserves the
  shape while respecting the simplex constraint.

* **Local refinement** – after the L‑BFGS stage we run a short
  exponentiated‑gradient (EG) refinement with a large β (≈200) to push the
  exact maximum down further.

* **Time management** – the routine stops well before the supplied
  budget (default 1000 s) and returns the best coefficient list found.
"""

import time
import math
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """
    Full self‑convolution of a vector `p` using a single FFT of size `N`.
    Returns the real array of length 2·len(p)−1.
    """
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k·p_{k−i} for i = 0..n−1 (correlation).
    Implemented via FFT for O(n log n) cost.
    """
    n = p.shape[0]
    L = w.shape[0]                # 2·n−1
    P = np.fft.rfft(p, n=N)
    W_rev = np.fft.rfft(w[::-1], n=N)
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    conv_rev = conv_rev[: n + L - 1]          # keep linear part
    rev = conv_rev[L - n : L][::-1]
    return rev


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smooth objective (log‑sum‑exp of the convolution) --------------------
def _make_obj_and_grad(beta: float, n: int, N: int):
    """
    Return a function f(x) that, given logits `x`, computes
        value = φ_β(softmax(x))
    and its gradient w.r.t. `x`.
    """
    def f(x: np.ndarray):
        # probabilities
        p = _softmax(x)

        # convolution p * p
        conv = _full_convolution_fft(p, N)
        max_val = conv.max()

        # smooth‑max weights
        w = np.exp(beta * (conv - max_val))

        # correlation rev_i = Σ_k w_k·p_{k−i}
        rev = _weighted_rev_fft(p, w, N)

        # gradient of the smooth max w.r.t. p
        g = 2.0 * rev                       # shape (n,)

        # objective value (smooth max)
        obj = max_val + np.log(w.sum()) / beta

        # chain rule to get gradient w.r.t. logits x
        pg = np.dot(p, g)                   # scalar <p,g>
        grad = p * (g - pg)                 # shape (n,)

        return obj, grad
    return f


# ----------------------------------------------------------------------
# Exponentiated‑gradient fine‑tuner ------------------------------------
def _refine_eg(p_start: np.ndarray,
               n: int,
               deadline: float,
               max_iters: int = 30_000,
               beta_final: float = 200.0) -> np.ndarray:
    """
    Short EG refinement with a very large β (hard max).  Operates directly
    on the probability vector `p`.  Returns the refined `p`.
    """
    p = p_start.copy()
    logp = np.log(p + 1e-30)

    # FFT size usable for both convolution and correlation
    N = _next_pow2(3 * n - 2)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        conv = _full_convolution_fft(p, N)
        M = conv.max()

        # hard max → weight vector is an indicator of the peak entries
        # use a very steep soft‑max as a proxy (β = beta_final)
        w = np.exp(beta_final * (conv - M))

        rev = _weighted_rev_fft(p, w, N)
        grad = 2.0 * rev                     # ∇_p φ_∞  (hard max)

        # step‑size schedule
        eta = 1.0 / (2.0 * n) / (1.0 + it * 1e-4)

        logp -= eta * grad

        # occasional jitter to escape plateaus
        if it % 3000 == 0:
            logp += np.random.randn(n) * 0.001

        # project back to simplex
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)
        p = exp_shift / exp_shift.sum()

    return p

    """

    max_restarts = 3
    max_iter_per_phase = 80                # L‑BFGS iterations per β

    for restart in range(max_restarts):
        if time.time() >= deadline:

        if restart == 0:

        else:
            # random perturbation of the log‑probabilities
            logp0 = np.random.randn(n) * 0.1

        logp = logp0.copy()
        # sequential β‑phases, warm‑starting each one

            if time.time() >= deadline:

            # minimise φ_β using L‑BFGS‑B

                method="L-BFGS-B",

        S = math.sqrt(2.0 * n)

                best_local_p = p_refined.copy()

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
