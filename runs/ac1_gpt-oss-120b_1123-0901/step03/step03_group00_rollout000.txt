"""
propose_candidate
=================

Search for a non‑negative step‑function (a probability vector) that
minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

where ``a*a`` is the ordinary linear convolution.
The objective is scale‑invariant, so we enforce ``sum(a)=1`` and
simply minimise ``2·n·max(conv)``.

The optimiser consists of three stages:

1. **Adam on a smooth‑max surrogate**  
   The hard max is replaced by a log‑sum‑exp with a temperature λ.
   λ is increased geometrically (continuation) so the optimisation
   gradually focuses on the true peak.  The probabilities are
   parametrised by logits (soft‑max) to keep the simplex constraints
   exact.  Several lengths (2048 … 16384) and random restarts are
   tried; the best vector is kept.

2. **Stochastic local refinement**  
   Starting from the best vector, a simple hill‑climbing / simulated
   annealing step moves a tiny amount of mass between two random
   coordinates while keeping the total sum fixed.  Accepted moves
   lower the exact objective; a temperature schedule allows occasional
   uphill steps to escape shallow local minima.

3. **Final check** – the best sequence found is returned as a plain
   Python list.  All operations respect the 1000 s time budget.

The implementation uses only NumPy (no external solvers) and runs
comfortably within the allotted time on a single core.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities ----------------------------------------------------
def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using an FFT of length `fft_len`."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)] for i=0…n‑1.
    The result has length n, the same as ``a``.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
def _logits_from_probs(p: np.ndarray) -> np.ndarray:
    """Convert a probability vector (sum = 1) to logits (log‑softmax)."""
    return np.log(p + 1e-15)


def _softmax_logits(x: np.ndarray) -> np.ndarray:
    """Stable softmax → simplex point."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()

    Search for a step‑function (non‑negative sequence) with a low value
    of ``evaluate_sequence``.  The routine respects the supplied time
