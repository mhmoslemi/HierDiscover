def _cross_corr(a, w):
    """
    Compute cross-correlation of a (len n) with w (len 2n-1):
    out[k] = sum_i a[i] * w[i + k] for k=0..n-1.

    # Select appropriate indices
    idx = (L - 1) - np.arange(n)
    out = c[idx]
    return out

Hybrid projected‑gradient optimizer for the self‑convolution bound.

The objective to minimise is

    2·n·max(conv(a,a)) / (sum(a)**2)

Because the function is homogeneous we fix the ℓ1‑norm
    Σ a_i = √(2 n) .
Then the objective equals simply max(conv(a,a)).

The algorithm works in two phases:

* **Exploration** – many random problem sizes (n≈1500…5000) are tried.

  projected gradient descent on a smooth log‑sum‑exp surrogate of the
  max‑convolution.  This quickly discovers good candidates and keeps the

  remaining time with a long projected‑gradient run.  The surrogate’s

  with Euclidean projection onto the simplex guarantees feasibility
  (non‑negativity and Σa_i = √(2 n)).

All linear algebra uses FFT‑based convolutions, so each iteration costs
O(n log n).  The code respects the 1000 s budget, prints progress, and
returns a plain Python list of non‑negative heights.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex {x_i ≥ 0 , Σ x_i = s}.
# --------------------------------------------------------------------------- #
def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex of radius ``s``."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full convolution via FFT (real signals).
# --------------------------------------------------------------------------- #
def conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a * a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


def max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a (used as the objective since Σa = √(2n))."""
    return float(conv_full_fft(a).max())


# --------------------------------------------------------------------------- #
# Cross‑correlation needed for the gradient of the smooth surrogate.
# --------------------------------------------------------------------------- #
def cross_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute, for k = 0 … n‑1,
        out[k] = Σ_i a[i] * w[i + k]      (zero‑padding outside bounds)

    ``a`` has length n, ``w`` has length 2·n‑1.
    """
    n = a.shape[0]
    L = w.shape[0]                # = 2·n‑1
    w_rev = w[::-1]               # reverse w
    size = 1 << ((n + L - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(w_rev, n=size)
    c = np.fft.irfft(A * B, n=size)[: n + L - 1]
    idx = (L - 1) - np.arange(n)   # map to required k‑values
    return c[idx]


# --------------------------------------------------------------------------- #
# Projected gradient descent on the log‑sum‑exp surrogate of max(conv).
# --------------------------------------------------------------------------- #
def pgd_optimize(a_init: np.ndarray,
                 target_sum: float,
                 max_iter: int,
                 max_time: float,
                 beta_start: float = 5.0,
                 lr_start: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Refine ``a_init`` (already on the simplex) by minimising
    f(a) = (1/β)·log Σ_i exp(β·conv_i) using projected gradient descent.
    Returns the best vector seen and its max‑convolution value.
    """
    start = time.time()
    a = a_init.copy()
    best = a.copy()
    best_score = max_conv(a)

    beta = beta_start
    lr = lr_start

    for it in range(max_iter):
        # time check
        if time.time() - start >= max_time:
            break

        conv = conv_full_fft(a)
        m = conv.max()

        # smooth weights (soft‑max) – numerically stable
        exp_term = np.exp(beta * (conv - m))
        w = exp_term / exp_term.sum()          # length 2·n‑1, sums to 1

        # gradient of the surrogate
        grad = 2.0 * cross_corr(a, w)          # shape n

        # gradient step + projection onto simplex
        a = a - lr * grad
        a = np.maximum(a, 0.0)
        a = project_onto_simplex(a, target_sum)

        # evaluate true objective (max‑convolution)
        cur_score = max_conv(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time():.2f}] PGD iter {it:5d} new best = {best_score:.6f} (n={len(a)})")

        # schedule updates
        lr *= 0.9995                         # gentle decay
        if (it + 1) % 500 == 0:              # make the surrogate sharper
            beta = min(beta * 1.1, 100.0)

    return best, best_score


# --------------------------------------------------------------------------- #
# Simple U‑shaped warm‑start (larger values near the ends).
# --------------------------------------------------------------------------- #
def init_U_shape(n: int) -> np.ndarray:
    """Return a randomised U‑shaped vector of length n."""
    idx = np.arange(n, dtype=float)
    weight = 1.0 / (idx + 1) + 1.0 / (n - idx)   # larger at the two ends
    a = np.random.rand(n) * weight
    return a


# --------------------------------------------------------------------------- #
# Main entry point expected by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Hybrid search for a low‑value step‑function.
    Uses a quick exploratory stage followed by a long PGD refinement.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5      # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------- #
    # If a previous good construction is available, seed from it.
    # ------------------------------------------------------------------- #
    if 'height_sequence_1' in globals():
        seq0 = np.asarray(height_sequence_1, dtype=float)
        n0 = seq0.shape[0]
        tsum0 = np.sqrt(2.0 * n0)
        seq0 = project_onto_simplex(seq0, tsum0)
        sc0 = max_conv(seq0)
        best_seq = seq0.copy()
        best_score = sc0
        print(f"[{time.time():.2f}] Seed from height_sequence_1 => {best_score:.6f} (n={n0})")

    # ------------------------------------------------------------------- #
    # Phase 1 – rapid exploration with many short PGD runs.
    # ------------------------------------------------------------------- #
    while time.time() < deadline - 100.0:          # leave ≈100 s for deep work
        # Pick a problem size
        if best_seq is not None and np.random.rand() < 0.15:
            n = len(best_seq)                     # occasionally reuse current best size
        else:
            n = np.random.randint(1500, 5000)     # range that balances quality & speed
        target_sum = np.sqrt(2.0 * n)

        # Initialise a candidate
        a0 = init_U_shape(n)
        a0 += np.random.rand(n) * 1e-3                # tiny random perturbation
        a0 = project_onto_simplex(a0, target_sum)

        cur = max_conv(a0)
        if cur < best_score:
            best_score = cur
            best_seq = a0.copy()
            print(f"[{time.time():.2f}] Explor init   best = {best_score:.6f} (n={n})")

        # Allocate a modest budget for this restart (≈30 s or 40 % of remaining)
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        phase_time = min(30.0, time_left * 0.4)

        a_opt, sc_opt = pgd_optimize(a0,
                                     target_sum,
                                     max_iter=2000,
                                     max_time=phase_time,
                                     beta_start=5.0,
                                     lr_start=0.5)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.2f}] Explor PGD   best = {best_score:.6f} (n={n})")

        if best_score < 1.503:          # reached target bound – stop early
            break

    # ------------------------------------------------------------------- #
    # Phase 2 – long refinement of the best candidate found so far.
    # ------------------------------------------------------------------- #
    if best_seq is not None:
        n = len(best_seq)
        target_sum = np.sqrt(2.0 * n)
        time_left = deadline - time.time()
        if time_left > 0:
            print(f"[{time.time():.2f}] Deep refinement start (n={n}, time_left={time_left:.1f}s)")
            a_refined, sc_refined = pgd_optimize(best_seq,
                                                 target_sum,
                                                 max_iter=20000,
                                                 max_time=time_left,
                                                 beta_start=5.0,
                                                 lr_start=0.5)
            if sc_refined < best_score:
                best_score = sc_refined
                best_seq = a_refined.copy()
                print(f"[{time.time():.2f}] Deep refinement best = {best_score:.6f}")

    # ------------------------------------------------------------------- #
    # Fallback – uniform vector (should never be needed).
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list for the grader.
    return best_seq.tolist()
