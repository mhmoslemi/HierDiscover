*   We then solve the linear program

# Helper: Euclidean projection onto the simplex { x_i ≥ 0 , Σ x_i = s }.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project `v` onto the simplex of radius `s` (non‑negative and sum = s)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full convolution via FFT (real, returns length 2*n‑1 array).
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Evaluate the objective – with the normalisation Σa = √(2n) the objective is
# exactly max(conv(a, a)).
# --------------------------------------------------------------------------- #
def _max_conv(a: np.ndarray) -> float:
    return float(_conv_full_fft(a).max())


# --------------------------------------------------------------------------- #
# Build a sparse matrix representing the linear map g ↦ f * g.
#   (f * g)_k = Σ_i f_i * g_{k-i}   →   row k, column j has coefficient f_{k-j}
# --------------------------------------------------------------------------- #
def _conv_constraint_matrix(f: np.ndarray) -> sp.csr_matrix:
    n = f.shape[0]
    i = np.repeat(np.arange(n, dtype=np.int64), n)          # shape n*n
    j = np.tile(np.arange(n, dtype=np.int64), n)           # shape n*n
    rows = i + j                                            # 0 … 2n-2
    cols = j
    data = f[i]
    # Build in COO format then convert to CSR (fast for linprog)
    A = sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n), dtype=np.float64)
    return A.tocsr()

# --------------------------------------------------------------------------- #
# Alternating LP + mixing step.
# --------------------------------------------------------------------------- #
def _lp_mixing_step(f: np.ndarray,
                    target_sum: float,
                    mixing_factor: float,
                    max_time: float,
                    start_time: float):
    """
    Perform a single LP optimisation for the current vector `f`,
    scale the solution to the required sum and mix it back into `f`
    using `mixing_factor` (0 < mixing_factor ≤ 1).
    Returns the new vector and the achieved max convolution.
    """
    n = f.shape[0]
    M = _max_conv(f)

    # Build the convolution constraint matrix for the current f
    A = _conv_constraint_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise sum(g)  <=> minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # Solve the LP with HiGHS (fast, sparse aware)
    res = opt.linprog(c,
                      A_ub=A,
                      b_ub=b,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True})

    if not res.success:
        # In the unlikely event of failure we simply return the unchanged vector.
        print(f"[{time.time():.2f}] LP failed (status={res.status}); skipping step.")
        return f, M

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return f, M

    # Scale to the required total mass
    g = g * (target_sum / sum_g)

    # Mix and re‑project (tiny numerical drift may appear)
    f_new = (1.0 - mixing_factor) * f + mixing_factor * g
    f_new = _project_onto_simplex(f_new, target_sum)

    new_score = _max_conv(f_new)
    return f_new, new_score
