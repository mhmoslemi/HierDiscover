def project_simplex(v, s):
    # Based on standard algorithm: sort v descending, compute cum sum - s, find rho, compute theta.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v, s):
    # v is vector to project onto simplex sum s
    # Works even if v sum > s or < s.
    # Implementation from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u > cssv / np.arange(1, len(v) + 1))[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
* Load the known construction ``height_sequence_1`` (if present) and
  normalise it to Σa = √(2 n).  This vector is used as the base shape.
* Several lengths are explored (the original length and a few extensions).
  For each length a parametric optimisation is performed in two phases:
    – **Phase 1 (smooth)** – a soft‑max approximation of the maximal
      autocorrelation is minimised with Adam.  The temperature τ decays
      exponentially so that the landscape first looks smooth and later
      becomes exact.
    – **Phase 2 (exact)** – the true max‑convolution is tackled using its
      sub‑gradient (the average of the gradients at all maximizing
      indices).  Adam is kept, but the weight vector is a hard one‑hot
      (or evenly split) on the active indices.
* After every Adam step the vector is projected onto the feasible set
  (a_i ≥ 0 and Σa = √(2 n)) using an Euclidean projection onto the
  simplex.
* Occasionally a tiny Gaussian jitter is added to help escape plateaus.
* The best sequence (lowest exact objective value) seen over all
  lengths, restarts and phases is returned.
* The routine respects a 1000 s wall‑clock budget.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# -------------------------------------------------------------
# Helper functions
# -------------------------------------------------------------

def _softmax_weights(conv: np.ndarray, tau: float) -> Tuple[np.ndarray, float]:
    """Return stable soft‑max weights (sum to 1) and the smoothed max."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k * a_{k‑i} (indices outside 0…n‑1 ignored)
    using one pair of FFTs.
    """
    n = a.shape[0]
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + n - 1]          # keep the valid part
    idx_start = 2 * n - 2
    rev = conv_rev[idx_start - np.arange(n)]
    return rev


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond].max()
        theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
# Core optimiser for a single length
# -------------------------------------------------------------

def _optimize_length(
    n: int,
    init_a: np.ndarray,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    max_iters: int,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser consisting of a smooth phase (soft‑max) followed by an
    exact phase (sub‑gradient of the true max).  The best sequence seen
    (according to the *exact* objective) is returned.
    """
    target_sum = math.sqrt(2.0 * n)

    # normalise the starting vector
    a = init_a.copy()
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.05                     # initial learning rate
    jitter_interval = 5000

    # annealing schedule for the smooth phase
    tau_factor = 0.20
    decay_rate = 20000.0

    # FFT size (next power‑of‑two ≥ 2·n‑1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    # split the iterations: 30 % smooth, the rest exact
    phase_switch = int(0.30 * max_iters)

    iteration = 0
    while iteration < max_iters and time.time() < deadline:
        # ---------------------------------------------------------
        # 1) compute the full self‑convolution via FFT
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # 2) keep the best exact solution seen so far
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:9d}] new best = {best_score:.6f}")

        # ---------------------------------------------------------
        # 3) build the weight vector w
        if iteration < phase_switch:
            # smooth phase – soft‑max of the convolution values
            tau = max(1e-12, tau_factor * M * math.exp(-iteration / decay_rate))
            w, _ = _softmax_weights(conv, tau)
        else:
            # exact phase – average over all tight indices
            max_idx = np.where(conv >= M - 1e-12)[0]
            w = np.zeros_like(conv)
            w[max_idx] = 1.0 / max_idx.size

        # ---------------------------------------------------------
        # 4) weighted reverse convolution (sub‑gradient of the weighted sum)
        rev = _weighted_rev(a, w, Nfft)

        # gradient of the *exact* objective (or of the smooth surrogate)
        grad = 2.0 * rev

        # ---------------------------------------------------------
        # 5) Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # learning‑rate schedule (slow decay)
        lr = lr0 / (1.0 + iteration * 1e-5)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---------------------------------------------------------
        # 6) projection onto the feasible simplex (a ≥ 0, Σa = target_sum)
        a = _project_onto_simplex(a, target_sum)

        # ---------------------------------------------------------
        # 7) occasional jitter to escape shallow plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = a + jitter
            a = _project_onto_simplex(a, target_sum)

    return best_seq, best_score
