def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_val = float('inf')
    # helper functions: conv etc defined top-level

    # define lengths, maybe 4096,8192,16384 maybe adapt based on time left.
    lengths = [1024, 2048, 4096, 8192, 16384]
    # but may prune based on time

        if time.time() > deadline:
            break
        print(f'Optimising length {n}')
        # compute FFT lengths for conv & cross
        conv_len = 2*n - 1
        fft_len_conv = 1 << int(np.ceil(np.log2(conv_len)))
        fft_len_corr = 1 << int(np.ceil(np.log2(conv_len + n - 1)))  # about 3n
        
        # prepare initial candidates list
        candidates = []
        # Warm starts from global best if length matches
        if best_seq is not None and len(best_seq) == n:

            else:

            candidates.append(cand)

    if best_seq is None:
        n_default = 1024
        best_seq = np.full(n_default, 1.0/n_default)
    # convert to list
    return best_seq.tolist()

"""
Propose a step‑function sequence that (approximately) minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

under a_i ≥ 0.

Algorithm
---------
* Work on several lengths (1024 … 16384).  For each length many
  *initial candidates* are created:
    - warm start from the current best,
    - a copy of the global ``height_sequence_1`` (if present),
    - a random Dirichlet draw,
    - a family of edge‑concentrated power‑law shapes,
    - a family of two‑Gaussian edge shapes.
* Every candidate is refined with two optimisation phases
    1. Adam on a *smooth‑max* surrogate of the max‑convolution.
       The surrogate temperature λ is increased geometrically, the
       logits are updated in the simplex via the soft‑max trick.
    2. A few steps of a sub‑gradient method on the *exact* max.
       The sub‑gradient is obtained from the active set of maximal
       convolution entries; an exponentiated‑gradient update keeps the
       iterate in the probability simplex.
* After each exact evaluation the global best sequence is updated.
* The whole search respects a 1000 s time budget; stagnation in an
  optimisation run triggers an early stop or a tiny random shake.
* The returned object is a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper to evaluate the exact objective (sum(a) is assumed ≈ 1)
    def _exact_objective(a: np.ndarray, n: int) -> float:
        conv = _full_convolution(a, fft_len_conv_cache[n])
        return 2.0 * n * conv.max() / (a.sum() ** 2)

    # ------------------------------------------------------------------
    # Pre‑compute FFT lengths for all lengths we will try.
    lengths = [1024, 2048, 4096, 8192, 16384]
    fft_len_conv_cache = {}
    fft_len_corr_cache = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_len_conv_cache[n] = 1 << int(np.ceil(np.log2(conv_len)))
        # for cross‑correlation we need length conv_len + n - 1 ≤ 3·n
        fft_len_corr_cache[n] = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        fft_len_conv = fft_len_conv_cache[n]
        fft_len_corr = fft_len_corr_cache[n]

        # --------------------- build a list of initial candidates ----------
        candidates = []

        # warm start from the globally best (if length matches)
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.array(best_seq, dtype=np.float64))

        # copy of the historic solution if it exists
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            cand = np.asarray(hseq, dtype=np.float64)
            if cand.shape[0] > n:
                cand = cand[:n]
            else:
                cand = np.pad(cand, (0, n - cand.shape[0]))
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s < 0.01:
                cand = np.random.dirichlet(np.ones(n))
            else:
                cand = cand / s
            candidates.append(cand)

        # pure random start
        candidates.append(np.random.dirichlet(np.ones(n)))

        # power‑law edge‑concentrated shapes  ((i+1)*(n-i))**α  , α<0
        idx = np.arange(n, dtype=np.float64) + 1.0
        prod = idx * (n - idx + 1.0)
        for alpha in np.linspace(-2.0, -0.1, 9):
            shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
            if shape.sum() > 0:
                shape /= shape.sum()
                candidates.append(shape)

        # two‑Gaussian edge shapes  exp(-γ·i²)+exp(-γ·(n‑1‑i)²)
        for gamma in [0.2, 0.5, 1.0, 2.0, 5.0]:
            i = np.arange(n, dtype=np.float64)
            shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            shape /= shape.sum()
            candidates.append(shape)

        # ------------------------------------------------------------------
        # Optimise every candidate
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # make a proper probability vector
            a = a0 / a0.sum()
            # initialise logits for the Adam phase
            logits = np.log(a + 1e-15)

            # ---- Adam on smooth‑max surrogate ----------------------------
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.05
            lam = 0.5
            lam_factor = 1.02

            # stop when no improvement for a few seconds
            last_improve = time.time()
            max_iters = int(15000 * 1024 / n) + 2000   # a few thousand per length
            for t in range(1, max_iters + 1):
                # soft‑max → simplex
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                # convolution and smooth‑max weights
                conv = _full_convolution(a, fft_len_conv)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # gradient of the surrogate
                grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

                # back‑prop through soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # temperature schedule
                lam = min(lam * lam_factor, 1e6)

                # exact objective (sum(a) ≈ 1)
                val = 2.0 * n * max_c / (a.sum() ** 2)

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Adam iter={t:05d}] new best {val:.6f}")

                # stagnation handling – break after a few seconds without gain
                if time.time() - last_improve > 4.0:
                    # small random “shake” to possibly escape flat regions
                    logits += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()
                    # continue for a few more iterations (do not exit yet)
                if time.time() > deadline:
                    break

            # ---- Sub‑gradient phase on the true max -----------------------
            # Start from the last a (already on the simplex)
            a = a / a.sum()
            eta = 0.6                     # step size for exponentiated update
            sub_last_improve = time.time()
            for sub_it in range(2000):
                conv = _full_convolution(a, fft_len_conv)
                max_c = conv.max()
                # which indices are active?
                active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
                if not active.any():
                    w = np.zeros_like(conv)
                else:
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()

                # sub‑gradient of max(conv) w.r.t a
                grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

                # exponentiated‑gradient step (keeps a ≥ 0)
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:                     # avoid collapse
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                # evaluate
                conv = _full_convolution(a, fft_len_conv)
                max_c = conv.max()
                val = 2.0 * n * max_c / (a.sum() ** 2)

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad iter={sub_it}] new best {val:.6f}")

                # early stop if stuck
                if time.time() - sub_last_improve > 4.0:
                    break
                if time.time() > deadline:
                    break

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (very unlikely)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return plain Python list as required by the external evaluator
    return best_seq.tolist()
