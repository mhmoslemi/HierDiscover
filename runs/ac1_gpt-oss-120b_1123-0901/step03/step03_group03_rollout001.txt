def optimize_length(n, init_a, deadline, best_score_global):

    target_sum = sqrt(2*n)
    a = init_a.copy()
    a = a * (target_sum / a.sum())
    # Adam variables
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8

    # Compute Nfft
    Nfft = 1 << ((2*n -1).bit_length())

    # parameters
    lr_initial = 0.03
    tau_factor_initial = 0.5  # start with relatively large tau
    tau_decay_rate = 40000.0  # slower decay

    # track best local
    best_local_a = a.copy()
    best_local_score = float('inf')
    no_improve_counter = 0
    max_no_improve = 200_000

    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # compute convolution and M
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[:2*n-1]
        M = float(conv.max())
        # compute smooth max weight
        tau = max(1e-12, tau_factor_initial * M * math.exp(-iteration / tau_decay_rate))
        w, Ltau = _softmax_weights(conv, tau)

        # gradient: derivative of Ltau wrt a
        rev = _weighted_rev(a, w, Nfft)

        # gradient of objective: we want to minimize M (or Ltau)
        # Using similar derivation: grad = 4 n rev / (target_sum^2) - const
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)

        lr = lr_initial / (1.0 + 1e-4 * iteration)

        # suggestion: try step size scaling
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0:
            a_new = a_new * (target_sum / s)
        else:
            a_new = np.full_like(a_new, target_sum / n)

        a = a_new

        # occasional jitter

"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.

Algorithm
---------
* The objective after normalising Σ a = √(2 n) is simply
  M = max_k (a∗a)[k].  We therefore minimise the maximal self‑convolution.
* To obtain a smooth surrogate we replace the max by a soft‑max
  Lτ = τ·log Σ exp(conv/τ) with an annealed temperature τ.
* The gradient of Lτ w.r.t. a is computed analytically (single FFT) and
  fed to an Adam optimiser.  After each step we project onto the feasible
  set (a_i ≥ 0, Σ a_i = √(2 n)).
* Several lengths are explored (smaller and larger than the provided baseline)
  and for each length we run the optimiser for a budgeted number of iterations.
  Periodic jitter and an early‑stop on stagnation help to escape plateaus.
* The best vector seen over all runs is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, tau: float):
    """
    Stable soft‑max of the convolution values.
    Returns the weight vector w (summing to 1) and the smoothed maximum
    Lτ = τ·log ∑exp((conv)/τ).
    """
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """
    Compute   rev_i = Σ_k w_k a_{k‑i}
    (indices outside 0…len(a)−1 are ignored) using FFTs.
    """
    n = a.shape[0]
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + n - 1]
    idx_start = 2 * n - 2
    rev = conv_rev[idx_start - np.arange(n)]
    return rev


def _project_simplex(x: np.ndarray, target_sum: float):
    """
    Euclidean projection of a non‑negative vector onto the simplex
    {y ≥ 0, Σ y = target_sum}.  A simple scaling after clipping is sufficient
    because all entries remain non‑negative.
    """
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s > 0.0:
        x = x * (target_sum / s)
    else:
        # degenerate case – fill uniformly
        x = np.full_like(x, target_sum / x.shape[0])
    return x


def _optimize_length_with_budget(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iters: int,
    best_score_global: float,
):
    """
    Adam optimiser for a given length `n`.
    Returns (best_local_sequence, best_local_score, possibly improved global score).
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise and enforce normalisation
    a = init_a.copy()
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.04

    # soft‑max temperature schedule
    tau_factor = 0.5          # initial τ = τ_factor·M
    decay_rate = 50000.0      # slower exponential decay

    # FFT size (next power‑of‑two ≥ 2·n‑1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    best_local_score = float("inf")
    best_local_a = a.copy()
    no_improve_counter = 0
    max_no_improve = 120_000  # stop if no improvement for many iter

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # 1) exact self‑convolution via FFT
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # --------------------------------------------------------------
        # 2) smooth approximation of the max
        tau = max(1e-12, tau_factor * M * math.exp(-it / decay_rate))
        w, Ltau = _softmax_weights(conv, tau)

        # --------------------------------------------------------------
        # 3) gradient of the smooth max (sub‑gradient of Lτ)
        rev = _weighted_rev(a, w, Nfft)

        # --------------------------------------------------------------
        # 4) gradient of the exact objective (via the smooth surrogate)
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        # --------------------------------------------------------------
        # 5) Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # 6) projection onto the feasible set
        a = _project_simplex(a_new, target_sum)

        # --------------------------------------------------------------
        # 7) occasional jitter to perturbe the trajectory
        if it % 4000 == 0:

            a = _project_simplex(a + jitter, target_sum)

        # 8) keep best local solution and possibly improve global best
        if M < best_local_score - 1e-12:
            best_local_score = M
            best_local_a = a.copy()
            no_improve_counter = 0
            if M < best_score_global:
                best_score_global = M
                print(

                )
        else:
            no_improve_counter += 1

        if no_improve_counter > max_no_improve:
            # stagnation – quit this length early
            break

    return best_local_a, best_local_score, best_score_global
