"""
search for a step‑function (non‑negative vector) that minimises
evaluate_sequence().  The algorithm works on several lengths.
For each length we start from the known construction (interpolated)
or a random Dirichlet draw, then run a customised Adam sub‑gradient
optimiser on the maximal convolution value (the objective after fixing
the L¹‑norm to sqrt(2·n)).  After the gradient phase a cheap
targeted local search moves a small amount of mass from indices that
contribute most to the current peak towards indices that contribute
least, accepting only improving moves (with a tiny simulated‑annealing
fallback).  The best vector seen before the overall deadline is
returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """
    Compute the full (linear) autocorrelation a*a via FFT and return
    the maximal value and its index.  The convolution length is 2*n-1.
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]
    k = int(conv.argmax())
    return float(conv.max()), k, conv


# ----------------------------------------------------------------------
def _rev_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    """
    For a given shift index k, returns the vector rev where
    rev[i] = a[k-i] if that index lies in [0,n), otherwise 0.
    """
    n = a.shape[0]
    rev = np.zeros_like(a)
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    rev[mask] = a[j[mask]]
    return rev


# ----------------------------------------------------------------------
def _adam_optimize(
    a_init: np.ndarray,
    n: int,
    target_sum: float,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    max_iter: int = 1_500_000,
    jitter_interval: int = 2000,
    random_active_prob: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style sub‑gradient descent for the maximal convolution value.
    The gradient is taken as the reversed vector for the (random) active
    shift, made zero‑mean to keep the L¹‑norm approximately constant.
    After each step we project onto the feasible set (non‑negative,
    correct L¹‑norm).  The function updates the global best sequence /
    score whenever an improvement is found.
    """
    a = a_init.copy()
    a = a * (target_sum / a.sum())          # normalise once

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.10
    iteration = 0

    while time.time() < deadline and iteration < max_iter:
        # ---- compute full convolution and identify active indices ----
        M, k_max, conv_full = _conv_max_fft(a)
        tol = 1e-12
        active = np.where(conv_full >= M - tol)[0]

        # ---- sub‑gradient: either a random active shift or the peak ----
        if active.size and np.random.rand() < random_active_prob:
            k = int(np.random.choice(active))
        else:
            k = k_max
        rev = _rev_for_shift(a, k)

        # gradient of the (scale‑invariant) objective
        grad = rev - rev.mean()                     # zero‑sum direction

        # ---- Adam update -------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- projection (non‑negative + re‑normalisation) ----------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a *= (target_sum / s)
        else:                         # practically never happens
            a.fill(target_sum / n)

        # ---- occasional jitter to escape plateaus ------------------
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 1e-3 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= target_sum / a.sum()

        # ---- keep global best --------------------------------------
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:7d}] new best M = {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _targeted_local_search(
    a_start: np.ndarray,
    n: int,
    target_sum: float,
    best_score: float,
    deadline: float,
    max_proposals: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climb that repeatedly moves a tiny amount of mass from the
    index contributing most to the current maximal convolution towards an
    index contributing least.  Only improving moves are kept; a very weak
    simulated‑annealing acceptance is added to avoid premature stagnation.
    """
    a = a_start.copy()
    cur_M, cur_k, _ = _conv_max_fft(a)
    start_time = time.time()
    T0 = cur_M * 0.05         # initial temperature (5 % of the peak)
    for prop in range(max_proposals):
        if time.time() >= deadline:
            break

        # contribution of each index to the current peak
        rev = _rev_for_shift(a, cur_k)
        contribution = a * rev

        # donor = index with largest (positive) contribution
        i = int(np.argmax(contribution))
        if a[i] <= 1e-14:
            continue

        # receiver = index with smallest rev value (least impact on the peak)
        # avoid the donor itself
        rev_excluding_i = rev.copy()
        rev_excluding_i[i] = np.inf
        j = int(np.argmin(rev_excluding_i))

        # amount to transfer (up to 5 % of donor’s mass)
        eps = a[i] * np.random.rand() * 0.05
        if eps <= 0.0:
            continue

        new_a = a.copy()
        new_a[i] -= eps
        new_a[j] += eps

        # projection not needed – mass conserved and non‑negativity kept
        M_new, k_new, _ = _conv_max_fft(new_a)

        # ------- acceptance test ------------------------------------
        accept = False
        if M_new < cur_M:
            accept = True
        else:
            # very weak simulated‑annealing fallback
            elapsed = time.time() - start_time
            fraction = elapsed / (deadline - start_time + 1e-12)
            T = max(T0 * (1.0 - fraction), 1e-6)
            if np.random.rand() < math.exp(-(M_new - cur_M) / T):
                accept = True

        if accept:
            a = new_a
            cur_M = M_new
            cur_k = k_new
            if cur_M < best_score:
                best_score = cur_M
                best_seq = a.copy()
                print(f"[targeted] new global best M = {best_score:.6f} (proposal {prop})")

    return a, best_score
