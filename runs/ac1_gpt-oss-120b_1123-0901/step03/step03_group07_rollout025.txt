    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.1

    best_val = float('inf')
    best_seq = None

    restart = 0

        restart += 1

        # choose initialization
        mode = restart % 5

"""
Propose a step‑function (non‑negative heights) with a low evaluation score.

Algorithm
---------
We minimise the exact objective

    F(a) = 2·n·max_k (a*a)[k] / (∑ a_i)²

using a mirror‑descent (soft‑max) parametrisation of the vector and Adam
updates on the true (hard‑max) gradient.
* a is represented as a = softmax(u) → all entries are ≥0 and sum to 1.
* At each iteration the full linear self‑convolution a*a is obtained with an
  FFT, the index of its maximum (k*) and the corresponding value M are
  extracted.
* The gradient of M w.r.t. a is cheap: ∂M/∂a_i = 2·a_{k*‑i} whenever the
  index lies inside the vector, otherwise 0.
* The gradient of F follows from the chain rule and is then mapped to the
  log‑space parameters u (the Jacobian of the soft‑max).
* Adam (β₁=0.9, β₂=0.999) with a gently decaying learning‑rate provides
  adaptive step‑sizes.  When progress stalls a tiny random “shake’’ in
  log‑space is added to escape shallow plateaus.
* Several random restarts (Dirichlet, geometric, half‑sine, symmetric,
  and a perturbation of the current best) are performed until the time
  budget (≈1000 s) is exhausted.  The best sequence found is returned as a
  plain Python list, ready for `evaluate_sequence`.

The method differs from the previous code by using the exact sub‑gradient
instead of a smooth‑max surrogate, by adding a dedicated projection onto
the simplex via the soft‑max parametrisation, and by exploring a few more
initial families (including symmetric shapes).  This usually yields a
lower bound than the original 1.519 … after the same amount of time.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << int(np.ceil(np.log2(x)))


def _full_convolution_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``size``."""
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * a.shape[0] - 1]


def _grad_max_convolution(a: np.ndarray, k_max: int) -> np.ndarray:
    """Gradient of max_{k} (a*a)[k] w.r.t. a when the max occurs at k_max."""
    n = a.shape[0]
    g = np.zeros_like(a)
    start = max(0, k_max - (n - 1))
    end = min(n - 1, k_max)
    idx = np.arange(start, end + 1, dtype=np.int64)
    # for each i in [start,end] we have ∂/∂a_i a_i a_{k-i} + a_{k-i} a_i = 2·a_{k-i}
    g[idx] = 2.0 * a[k_max - idx]
    return g


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """Return a list of non‑negative heights that (hopefully) beats the
    current best bound.  The routine runs for at most *budget_s* seconds.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2  # safety margin

    # ------------------------------------------------------------------
    # Length of the vector.
    # Use the length of a previously discovered solution if it exists,
    # otherwise fall back to 3000 (the length that was used before).
    global height_sequence_1
    try:
        n = len(height_sequence_1)  # type: ignore[name‑defined]
    except Exception:
        n = 3000

    conv_fft_size = _next_pow2(2 * n - 1)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.05

    best_val = float("inf")
    best_seq = None

    restart = 0
    while time.time() < deadline:
        restart += 1

        # --------------------------- initialise --------------------------
        mode = restart % 5
        if mode == 0:
            a = np.random.dirichlet(np.ones(n))
        elif mode == 1:
            # geometric decay
            r = np.random.uniform(0.2, 0.95)
            a = (r ** np.arange(n, dtype=np.float64))
            a /= a.sum()
        elif mode == 2:
            # half‑sine shape
            i = np.arange(1, n + 1, dtype=np.float64)
            a = np.sin(np.pi * i / (n + 1))
            a = np.maximum(a, 0.0)
            a /= a.sum()
        elif mode == 3:
            # symmetric about the centre
            half = (n + 1) // 2
            half_vec = np.random.dirichlet(np.ones(half))
            if n % 2 == 0:
                a = np.concatenate([half_vec, half_vec[::-1]])
            else:
                a = np.concatenate([half_vec, half_vec[-2::-1]])
        else:
            # perturb current best if we have one of the same length
            if best_seq is not None and len(best_seq) == n:
                a = np.array(best_seq) + 0.001 * np.random.rand(n)
                a = np.maximum(a, 0.0)
                a /= a.sum()
            else:
                a = np.random.rand(n)
                a /= a.sum()

        # log‑space (soft‑max) parametrisation
        u = np.log(a + 1e-30)

        # Adam state
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        it = 0
        last_improve = time.time()

        # ---------------------------------------------------------------
        while time.time() < deadline:
            it += 1

            # ---- soft‑max to obtain the current probability vector ----
            u_shift = u - np.max(u)               # numerical stability
            a = np.exp(u_shift)
            S = a.sum()
            a /= S

            # ---- convolution and max index/value -------------------
            conv = _full_convolution_fft(a, conv_fft_size)
            k_max = int(np.argmax(conv))
            M = conv[k_max]

            # ---- gradient of the max convolution ------------------
            gM = _grad_max_convolution(a, k_max)

            # ---- gradient of the objective F(a) --------------------
            # F = 2·n·M / S²
            grad_a = (2.0 * n / (S * S)) * gM - (4.0 * n * M / (S * S * S)) * np.ones_like(a)

            # ---- map gradient to log‑space (soft‑max Jacobian) ----
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            # ---- Adam update ---------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad_u
            v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = base_lr * (0.9995 ** it)          # gentle decay
            u -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # ---- occasional exact evaluation -----------------------
            if it % 30 == 0:
                F_val = 2.0 * n * M / (S * S)     # exact objective (no clamping)
                if F_val < best_val:
                    best_val = F_val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(
                        f"[Restart {restart:03d}] iter {it:06d}  new best = {F_val:.6f}"
                    )

            # ---- shake if we have been idle too long ---------------
            if time.time() - last_improve > 1.2:
                u += (np.random.rand(n) - 0.5) * 1e-3
                m.fill(0.0)
                v.fill(0.0)
                last_improve = time.time()

            # ---- limit the inner loop to keep the total time sane ---
            if it >= 12000:
                break

        # status line after each restart
        print(
            f"[Restart {restart:03d}] finished it={it}, best so far = {best_val:.6f}"
        )

    # ------------------------------------------------------------------
    # If nothing was found (should not happen) fall back to uniform.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # final check with the provided evaluator (clamping etc.)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final objective according to evaluate_sequence = {final_score:.6f}")
    except Exception:
        pass

    # return as a plain Python list
    return best_seq.tolist()
