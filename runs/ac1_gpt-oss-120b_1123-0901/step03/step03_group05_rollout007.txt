def _simulated_annealing(a, n, target_sum, deadline, cur_score, max_iters=1_000_000):

    iteration = 0
    max_iters = int(1e6)  # adjust

    # Introduce dynamic max_move_frac: start large, reduce
    while time.time() < deadline and iteration < max_iters:

        rev = _subgradient_average(a, active)  # subgradient

        prob = rev / rev.sum() if rev.sum() > 0 else None
        if prob is not None:

            i = np.random.randint(n)
        # ensure a[i] > 0

        max_frac = 0.3 if iteration < max_iters * 0.3 else 0.1 if iteration < max_iters * 0.7 else 0.03

        new_a[j] += eps

def _adam_optimize(a_init, target_sum, max_iter, deadline):
    a = a_init.copy()
    a = a * (target_sum / a.sum())
    best_seq = a.copy()
    best_score = _conv_max_fft(a)[0]  # M

    lr = 0.2  # initial learning rate
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    iter = 0
    while time.time() < deadline and iter < max_iter:
        iter += 1
        # compute conv
        M, _, conv = _conv_max_fft(a)
        # compute active set
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]
        # compute subgrad (average reversed)
        rev = _subgradient_average(a, active)
        # subtract mean to keep sum unchanged (centering)
        rev_centered = rev - rev.mean()
        # Adam update
        grad = rev_centered
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**iter)
        v_hat = v / (1 - beta2**iter)
        # step size schedule: base_lr / sqrt(iter) perhaps
        step = lr / (1 + 1e-4 * iter)
        a = a - step * m_hat / (np.sqrt(v_hat) + 1e-8)
        # proj non-neg
        a = np.maximum(a, 0)
        # renorm sum
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / len(a))
        # jitter occasional
        if iter % 2000 == 0:
            jitter = np.random.randn(len(a)) * 0.001 * target_sum / len(a)
            a = np.maximum(a + jitter, 0)
            a = a * (target_sum / a.sum())
        # evaluate
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[adam len {len(a)} iter {iter}] new best {best_score:.6f}")
    return best_seq, best_score

"""
Hybrid projected sub‑gradient descent + simulated‑annealing search for a
non‑negative step‑function that minimises the evaluate_sequence objective.
The algorithm works in three phases for each chosen length:

1. **Initialisation** – interpolate the known construction or draw a random
   positive vector and renormalise to the required ℓ1‑norm √(2 n).

2. **Projected sub‑gradient descent** – the objective is  M = max_k (a*a)[k].
   The sub‑gradient of M is the average of the reversed vector over all
   active shifts (those attaining the maximum).  After a gradient step the
   iterate is projected onto the feasible set (non‑negative, fixed sum).

3. **Fine‑grained simulated annealing** – random mass transfers between
   indices are accepted if they improve M or with a temperature‑dependent
   probability.  Transfers are biased by the sub‑gradient: mass is taken
   from indices with large gradient (they contribute most to the peak)
   and moved to indices with small gradient.

The search is performed for several lengths (the original length and a few
scaled versions).  The best sequence found before the time budget expires
is returned.  All heavy work (convolution, sub‑gradient) is done with
NumPy/FFT, keeping the per‑iteration cost low enough to explore millions
of proposals within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """
    Full autocorrelation a*a via FFT.
    Returns (max_value, argmax_index, full_convolution_array).
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]
    k = int(conv.argmax())
    return float(conv.max()), k, conv


# ----------------------------------------------------------------------
def _subgradient_average(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Average of the reversed vector a w.r.t. the active shift indices.
    For each active shift k we add a shifted copy of a and finally average.
    """
    n = a.shape[0]
    rev = np.zeros_like(a)
    idx = np.arange(n)
    for k in active:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    if active.size > 0:
        rev /= float(active.size)
    return rev


# ----------------------------------------------------------------------
def _projected_subgradient(a_init: np.ndarray,
                           target_sum: float,
                           deadline: float,
                           max_iter: int = 300_000,
                           base_lr: float = 0.2,
                           jitter_interval: int = 2_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with Adam‑style moment estimates.
    Returns the best vector seen and its max‑convolution value.
    """
    a = a_init.copy()
    a *= target_sum / a.sum()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_seq = a.copy()
    best_score, _, _ = _conv_max_fft(a)

    iteration = 0
    while time.time() < deadline and iteration < max_iter:
        iteration += 1

        # 1) current autocorrelation and active set
        M, _, conv = _conv_max_fft(a)
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # 2) sub‑gradient (average reversed active vectors)
        grad = _subgradient_average(a, active)

        # 3) centre the gradient so the sum constraint is respected after the step
        grad -= grad.mean()

        # 4) Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = base_lr / (1.0 + 1e-4 * iteration)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # 5) projection onto the feasible set
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a *= target_sum / s
        else:
            a = np.full_like(a, target_sum / n)

        # 6) occasional jitter to escape plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= target_sum / a.sum()

        # 7) bookkeeping
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[sgd len {n:5d} iter {iteration:7d}] new best M = {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _anneal_optimize(a_start: np.ndarray,
                     target_sum: float,
                     deadline: float,
                     max_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing on the mass‑transfer neighbourhood.
    Moves are biased by the sub‑gradient (large gradient ⇒ donor,
    small gradient ⇒ receiver).  Returns the best vector seen.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score, _, _ = _conv_max_fft(a)

    # temperature schedule
    T0, Tmin = 0.20, 0.001
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # recompute sub‑gradient only occasionally (every 50 iters)
        if it % 50 == 0:
            M, _, conv = _conv_max_fft(a)
            tol = 1e-12
            active = np.where(conv >= M - tol)[0]
            grad = _subgradient_average(a, active)
            # probabilities for donor / receiver
            donor_prob = grad / (grad.sum() + 1e-12)
            recv_prob = (grad.max() - grad + 1e-12)
            recv_prob /= recv_prob.sum()
        # pick donor / receiver
        i = np.random.choice(n, p=donor_prob)
        if a[i] <= 1e-15:
            continue
        j = np.random.choice(n, p=recv_prob)
        if j == i:
            continue

        # move a fraction of donor's mass (fraction shrinks as we go)
        if it < max_iters * 0.3:
            frac = 0.25
        elif it < max_iters * 0.7:
            frac = 0.10
        else:
            frac = 0.03
        eps = a[i] * np.random.rand() * frac
        if eps <= 0.0:
            continue

        new_a = a.copy()
        new_a[i] -= eps
        new_a[j] += eps
        # keep feasibility
        new_a = np.maximum(new_a, 0.0)
        # sum is preserved (up to floating error); renormalise just in case
        s = new_a.sum()
        if s > 0.0:
            new_a *= target_sum / s
        else:
            new_a = np.full_like(new_a, target_sum / n)

        M_new, _, _ = _conv_max_fft(new_a)

        delta = M_new - best_score
        T = T0 * (1 - it / max_iters) + Tmin * (it / max_iters)
        accept = (M_new < best_score) or (np.random.rand() < math.exp(-delta / T))
        if accept:
            a = new_a
            if M_new < best_score:
                best_score = M_new
                best_seq = new_a.copy()
                print(f"[anneal iter {it:7d}] new best M = {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _cheap_local_search(a_start: np.ndarray,
                        target_sum: float,
                        deadline: float,
                        max_proposals: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb that moves a small amount of mass from one index
    to another only if the maximum autocorrelation decreases.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_score, _, _ = _conv_max_fft(a)
    proposal = 0
    while time.time() < deadline and proposal < max_proposals:
        proposal += 1
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        eps = a[i] * np.random.rand() * 0.05          # up to 5 % of donor
        new_a = a.copy()
        new_a[i] -= eps
        new_a[j] += eps
        new_a = np.maximum(new_a, 0.0)
        new_a *= target_sum / new_a.sum()
        M_new, _, _ = _conv_max_fft(new_a)
        if M_new < best_score:
            a = new_a
            best_score = M_new
            print(f"[local] improvement to {best_score:.6f} (prop {proposal})")
    return a, best_score
