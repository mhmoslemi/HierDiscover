def _adam_optimize_length(n, p_start, deadline):
    max_iters = 300_000
    p = p_start.copy()
    logp = np.log(p + 1e-30)
    m = np.zeros_like(logp)
    v = np.zeros_like(logp)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    N = _next_pow2(3 * n - 2)
    best_local_score = float('inf')
    best_local_p = p.copy()
    beta_start = 0.1
    beta_end = 50.0
    beta_len = 100_000
    # iteration
    for it in range(1, max_iters+1):
        if time.time() >= deadline: break
        conv = _full_convolution_fft(p, N)
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_p = p.copy()
            # optionally print
        # compute subgradient:
        # Use hard max: difference tolerance maybe 1e-12
        tol = 1e-12
        active = conv >= M - tol
        # if many active, use weights in proportion to conv (or uniform)
        if active.sum() > 1:
            # use softmax weights for stability
            beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_len)
            w = np.exp(beta * (conv - M))
        else:
            w = active.astype(float)  # only 1s for max indices
        # correlation (rev)
        rev = _weighted_rev_fft(p, w, N)
        # gradient of 2n * M, in log space:
        grad = 4.0 * n * rev
        # Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        eta = 0.1 / (np.sqrt(v_hat) + eps)   # step size
        logp -= eta * m_hat
        # jitter occasionally:
        if it % 5000 == 0:
            logp += np.random.randn(n) * 0.001
        # project to simplex: exp(logp) normalized:
        max_logp = np.max(logp)
        exp_shift = np.exp(logp - max_logp)
        p = exp_shift / exp_shift.sum()
    return best_local_p, best_local_score

def lp_refine(p, M, deadline):
    n = len(p)
    # Build A_ub (2n-1) x n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # j ranges from i_start..i_end (since j = i)
        # Actually for given k, g_j coefficient = p_{k-j}
        # So we set columns j = i_start..i_end with p_{k-j}
        j_vals = np.arange(i_start, i_end+1)
        A[k, j_vals] = p[k - j_vals]
    # now solve LP: maximize sum(g), i.e., c = -np.ones(n) for minimization of -sum(g)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    # b_ub = M * np.ones(2n-1)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success:
        return p  # no improvement
    g = res.x
    sum_g = g.sum()
    if sum_g <= 1e-12:
        return p
    g_norm = g / sum_g
    # mixing weight t
    t = 0.5
    new_p = (1 - t) * p + t * g_norm
    # project onto simplex (ensure sum=1)
    new_p = np.maximum(new_p, 0)
    new_p /= new_p.sum()
    return new_p
```

"""
Multi‑scale Adam‑based mirror descent for the convolution‑max problem.

The evaluation function is scale‑invariant: for a sequence a with
Σa = √(2·n) the score equals max(a∗a).  Therefore we work with a
probability vector p (Σp=1) and minimise 2·n·max(p∗p).  The algorithm
is:

* Interpolate the current best probability vector to a new length.
* Optimise p on the simplex using Adam on the logarithm of p.
  The gradient of the smoothed maximum is obtained by a soft‑max
  weighting of the full autocorrelation and a single FFT‑based
  correlation (O(n log n) per iteration).
* The temperature (β) of the soft‑max grows from 0.1 to a large value
  (≈50) so the update eventually follows the true sub‑gradient of the
  max.
* Occasional jitter and a bias‑corrected Adam step help to escape
  plateaus.
* After the gradient phase a cheap linear‑programming refinement
  (SciPy linprog) is performed on the current p: maximise Σg under
  (p∗g)≤M (M = current max) and mix the resulting g back into p.
* The process is repeated on a schedule of increasing lengths.
  The best sequence (scaled back to Σa=√(2·n)) found before the time
  budget expires is returned.
"""

import time
import math
import numpy as np
try:
    # SciPy is optional – used only for a final LP refinement.
    from scipy.optimize import linprog
    _has_scipy = True
except Exception:  # pragma: no cover
    _has_scipy = False

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """
    Full linear convolution p∗p using a single FFT of size N.
    Returns the real array of length 2·len(p)−1.
    """
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Correlation Σ_k w_k·p_{k-i} for i=0..n−1
    (i.e. rev = p  ★  w^rev ).
    Obtained by a single pair of FFTs.
    """
    n = p.shape[0]
    L = w.shape[0]                # = 2·n−1
    P = np.fft.rfft(p, n=N)
    W_rev = np.fft.rfft(w[::-1], n=N)
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    conv_rev = conv_rev[: n + L - 1]
    rev = conv_rev[L - n : L][::-1]
    return rev


# ----------------------------------------------------------------------
def _adam_optimize_length(p_start: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Adam‑based mirror descent on the simplex for a fixed length ``n``.
    Returns the best probability vector found and its (unscaled) score
    2·n·max(p∗p).
    """
    # hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    jitter_interval = 5000
    jitter_scale = 0.001

    # soft‑max temperature schedule
    beta_start = 0.1
    beta_end = 50.0
    beta_len = 100_000          # iterations over which β grows linearly

    p = p_start.copy()
    logp = np.log(p + 1e-30)     # stay in log‑space (mirror descent)

    m = np.zeros_like(logp)     # first‑moment estimate
    v = np.zeros_like(logp)     # second‑moment estimate

    N = _next_pow2(3 * n - 2)    # safe size for both convolutions

    best_score = float("inf")
    best_p = p.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- convolution and current objective -----
        conv = _full_convolution_fft(p, N)
        M = conv.max()
        cur_score = 2.0 * n * M

        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[len {n:5d} iter {it:7d}] new local best = {cur_score:.6f}")

        # ----- soft‑max weighting for a smooth sub‑gradient -----
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_len)
        w = np.exp(beta * (conv - M))          # w_k ∈ (0,1], max=1

        # ----- gradient (correlation with the weight vector) -----
        rev = _weighted_rev_fft(p, w, N)       # shape (n,)

        # Gradient of 2·n·max_β(p∗p) w.r.t. p
        grad = 4.0 * n * rev

        # ----- Adam update in log‑space -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Step‑size – a modest base value works well with the log‑parameterisation
        step = 0.05 / (np.sqrt(v_hat) + eps)
        logp -= step * m_hat

        # occasional jitter to break ties / escape plateaus
        if it % jitter_interval == 0:
            logp += np.random.randn(n) * jitter_scale

        # ----- project back onto the simplex (exponentiate & normalise) -----
        max_log = np.max(logp)                     # avoid overflow
        exp_shift = np.exp(logp - max_log)
        p = exp_shift / exp_shift.sum()

    return best_p, best_score


# ----------------------------------------------------------------------
def _lp_refine(p: np.ndarray,
               max_val: float,
               mixing: float = 0.5) -> np.ndarray:
    """
    One linear‑programming refinement step.
    Solve      max Σ g_j
    s.t.      (p ∗ g)_k ≤ max_val   for all k
              g_j ≥ 0
    The solution g is normalised to a probability vector and mixed
    back into p with weight ``mixing``.
    Returns the mixed probability vector.
    """
    if not _has_scipy:
        return p    # no LP capability – simply return the input

    n = p.shape[0]
    rows = 2 * n - 1

    # Build Toeplitz constraint matrix A (rows × n)
    # A_{k,j} = p_{k-j} if 0 ≤ k-j < n else 0
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # columns j = i_start … i_end  correspond to p_{k-j}
        cols = np.arange(i_start, i_end + 1)
        A[k, cols] = p[k - cols]

    c = -np.ones(n)                     # linprog minimises, so maximise Σg ⇔ minimise −Σg
    bounds = [(0.0, None) for _ in range(n)]
    b_ub = np.full(rows, max_val)

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method="highs", options={"presolve": True})
    except Exception:  # pragma: no cover
        return p

    if not res.success:
        return p

    g = np.maximum(res.x, 0.0)
    sum_g = g.sum()
    if sum_g < 1e-12:
        return p

    g_norm = g / sum_g                     # probability vector
    new_p = (1.0 - mixing) * p + mixing * g_norm
    new_p = np.maximum(new_p, 0.0)
    new_p /= new_p.sum()
    return new_p


# ----------------------------------------------------------------------
def _interpolate_prob(old_p: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = old_p.shape[0]
    if old_n == new_n:
        return old_p.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_p = np.interp(x_new, x_old, old_p)
    new_p = np.maximum(new_p, 0.0)
    new_p /= new_p.sum()
    return new_p
