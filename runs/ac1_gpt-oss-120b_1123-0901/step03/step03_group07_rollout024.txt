def simplex_projection(v):
    """Projects vector v onto the probability simplex."""
    # ensure non-negative? v can be any real.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1000):

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # Choose n
    default_n=5000
    n = default_n
    if "height_sequence_1" in globals():
        seq0 = list(globals()["height_sequence_1"])
        n = len(seq0)
        # we will start with that as initial candidate in first restart.

    # Precompute FFT size
    conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))

    # Adam hyperparameters
    beta1 = 0.9; beta2 = 0.999; eps = 1e-8; base_lr = 0.05

    best_val= float('inf')
    best_seq = None

    restart = 0
    while time.time() < deadline:
        restart += 1

        # Initialize u based on seed or random or previous best
        if restart == 1 and "height_sequence_1" in globals():
            # use given sequence
            a0 = np.array(seq0, dtype=np.float64)
            a0 = a0 / a0.sum()
            u = np.log(a0 + 1e-30)
        else:
            # random Dirichlet
            a0 = np.random.dirichlet(np.ones(n))
            u = np.log(a0 + 1e-30)

        # Adam state
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        # iteration
        it = 0
        last_improve = time.time()
        while time.time() < deadline:
            it += 1

            # compute probabilities
            u_shift = u - np.max(u)  # for numeric stability
            a = np.exp(u_shift)
            a /= a.sum()

            # compute convolution via FFT
            fa = np.fft.rfft(a, conv_fft_size)
            conv = np.fft.irfft(fa * fa, conv_fft_size)[:2 * n - 1]

            # max conv and index
            idx = int(np.argmax(conv))
            max_b = conv[idx]

            # compute subgradient of max conv wrt a
            grad_a = np.zeros_like(a)
            # term a[idx - i]
            i_start = max(0, idx - (n - 1))
            i_end = min(idx, n - 1)
            idx_range = np.arange(i_start, i_end + 1)
            grad_a[idx_range] += a[idx - idx_range]

            # term a[i - idx]
            j_start = max(idx, 0)  # idx >=0
            j_end = min(n - 1, idx + n - 1)
            j_range = np.arange(j_start, j_end + 1)
            grad_a[j_range] += a[j_range - idx]

            # diagonal double for even idx
            if idx % 2 == 0:
                i_mid = idx // 2
                if 0 <= i_mid < n:
                    grad_a[i_mid] += a[i_mid]  # add second contribution

            # scale gradient for objective (2*n factor)
            grad_a = grad_a * (2.0 * n)   # because F = 2*n * max_conv

            # convert to gradient in log-space (softmax)
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_u
            v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

            it_float = float(it)
            m_hat = m / (1.0 - beta1 ** it_float)
            v_hat = v / (1.0 - beta2 ** it_float)

            lr = base_lr / np.sqrt(it_float)  # decaying learning rate

            u -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # evaluate every 50 iterations
            if it % 50 == 0:
                val = 2.0 * n * max_b  # because sum a=1
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[Restart {restart:03d}] iter {it:06d}  new best = {val:.6f}")

            if it >= 20000:

    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # final list convert
    print(f"Final best objective = {best_val:.6f} (length {n})")
    return best_seq.tolist()

"""
Propose a non‑negative step function (list of heights) that minimizes

    F(a) = 2·n·max_k (a * a)[k] / (∑ a_i)²

The algorithm uses a mirror‑descent style optimiser (Adam) on the exact
sub‑gradient of the max‑convolution.  The probabilities are kept in a
soft‑max (log‑space) parametrisation, guaranteeing non‑negativity and a
unit sum.  A simple “shake” is applied when progress stalls.
Multiple random restarts are performed until the time budget expires.
If a previously found sequence is available as the global variable
`height_sequence_1` it is used as the first starting point.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, fft_size: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length ``fft_size``.
    Returns the full 2·n‑1 result.
    """
    fa = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(fa * fa, fft_size)[: 2 * a.shape[0] - 1]
    return conv


def _subgradient_max_conv(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Sub‑gradient of the maximum convolution entry with respect to ``a``.
    ``idx`` is the index where the maximum occurs.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    if idx < n:
        # term a[idx‑i] for i = 0 … idx
        g[: idx + 1] += a[idx :: -1]          # a[idx], a[idx‑1], … , a[0]
        # term a[i‑idx] for i = idx … n‑1
        g[idx:] += a[: n - idx]               # a[0], a[1], … , a[n‑idx‑1]
    else:
        # only the a[idx‑i] term survives
        start = idx - (n - 1)                 # first i where idx‑i is in range
        # slice a[n‑1 : start‑1 : -1] yields a[n‑1], a[n‑2], … , a[start]
        g[start:] += a[n - 1 : start - 1 : -1]

    # diagonal term (i = idx/2) appears with factor 2 when idx is even
    if idx % 2 == 0:
        i_mid = idx // 2
        if 0 <= i_mid < n:
            g[i_mid] += a[i_mid]

    return g


def _objective_from_conv(max_conv: float, n: int) -> float:
    """Exact objective when the sum of the coefficients is normalised to 1."""
    return float(2.0 * n * max_conv)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.
    Returns a plain Python list suitable for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3               # tiny safety margin

    # --------------------------------------------------------------
    # length of the sequence – 5000 gives a good trade‑off between
    # expressive power and speed (FFT size will be 16384).
    n = 5000
    fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))      # power of two >= 2n‑1

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.05                                         # for the logits

    best_val = float("inf")
    best_seq = None
    restart = 0

    while time.time() < deadline:
        restart += 1

        # ------------------- initialise logits (log‑softmax) -------------
        if restart == 1 and "height_sequence_1" in globals():
            # reuse the provided sequence if it matches the chosen length
            seq0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if seq0.size != n:
                # resize by simple interpolation if lengths differ
                x_old = np.linspace(0, 1, seq0.size)
                x_new = np.linspace(0, 1, n)
                seq0 = np.interp(x_new, x_old, seq0)
            seq0 = np.clip(seq0, 0.0, None)
            if seq0.sum() == 0:
                seq0 = np.full_like(seq0, 1.0 / n)
            else:
                seq0 = seq0 / seq0.sum()
            u = np.log(seq0 + 1e-30)
        else:
            # random Dirichlet start (uniform on the simplex)
            a0 = np.random.dirichlet(np.ones(n))
            u = np.log(a0 + 1e-30)

        # Adam state
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        it = 0
        last_improve = time.time()

        # ---------------------------- optimisation loop -----------------
        while time.time() < deadline:
            it += 1

            # ---- probabilities (soft‑max) ----
            u_shift = u - np.max(u)                # improve numerical stability
            a = np.exp(u_shift)
            a /= a.sum()                           # now ∑a_i = 1

            # ---- convolution & max index ----
            conv = _full_convolution_fft(a, fft_size)
            idx = int(np.argmax(conv))
            max_b = conv[idx]                      # this is exactly max(a*a)

            # ---- exact sub‑gradient of the max term ----
            g_raw = _subgradient_max_conv(a, idx)  # shape (n,)

            # ---- gradient w.r.t. the logits (soft‑max Jacobian) ----
            # the true objective is 2·n·max_b, so gradient = 2·n·g_raw
            dot = np.dot(a, g_raw)                 # ⟨a , g_raw⟩
            grad_u = 2.0 * n * a * (g_raw - dot)   # chain rule for soft‑max

            # ---- Adam update ----
            m = beta1 * m + (1.0 - beta1) * grad_u
            v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

            it_f = float(it)
            m_hat = m / (1.0 - beta1 ** it_f)
            v_hat = v / (1.0 - beta2 ** it_f)

            # learning‑rate decay (square‑root schedule)
            lr = base_lr / np.sqrt(it_f)

            u -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # ---- occasional exact evaluation (every 30 iterations) ----
            if it % 30 == 0:
                val = _objective_from_conv(max_b, n)
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[Restart {restart:03d}] it {it:06d}  new best = {val:.6f}")

            # ---- shake when stuck for >1.5 s ----
            if time.time() - last_improve > 1.5:
                u += (np.random.rand(n) - 0.5) * 5e-4   # tiny random log‑perturbation
                m.fill(0.0)
                v.fill(0.0)
                last_improve = time.time()

            # ---- modest iteration cap per restart (adjustable) ----
            if it >= 20000:
                break

        # status line after each restart
        print(f"[Restart {restart:03d}] finished it={it}, best so far = {best_val:.6f}")

        # if the remaining time is tiny, break early
        if time.time() + 5.0 > deadline:
            break

    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector
        best_seq = np.full(n, 1.0 / n)

    print(f"Final best objective = {best_val:.6f} (length {n})")
    # convert to plain list for the external evaluator
    return best_seq.tolist()
