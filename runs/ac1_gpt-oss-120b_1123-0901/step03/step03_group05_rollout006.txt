import time, math, numpy as np

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0
    best_seq = None
    best_score = float('inf')
    
    # define lengths
    lengths = [int(L) for L in [500, 1000, 2000, 4000, 7000, 10000] if L <= 12000]
    
    # Loop over lengths
    for n in lengths:
        target_sum = math.sqrt(2.0 * n)
        # allocate portion of time for this length
        len_deadline = time.time() + (deadline - time.time()) * 0.5  # half time on search, half later
        # random candidates
        while time.time() < len_deadline:
            # support size randomly around sqrt(2n)
            scale = np.random.uniform(0.5, 2.0)
            m = int(scale * math.sqrt(2.0 * n))
            m = max(2, min(m, n))
            # choose support
            support = np.random.choice(n, size=m, replace=False)
            a = np.zeros(n)
            a[support] = target_sum / m
            # evaluate
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[rand] n={n} m={m} new best {best_score:.6f}")
        
        # after random search, local improvement
        if best_seq is not None and time.time() < deadline:
            # hill climbing: try swap from support to outside
            a = best_seq.copy()
            n_range = np.arange(n)
            a_sum = a.sum()
            for _ in range(20000):
                if time.time() >= deadline:
                    break
                # pick donor from support
                supp_idx = np.where(a > 1e-12)[0]
                if supp_idx.size == 0:
                    break
                i = np.random.choice(supp_idx)
                # choose j not in support
                zero_idx = np.where(a <= 1e-12)[0]
                if zero_idx.size == 0:
                    break
                j = np.random.choice(zero_idx)
                # transfer epsilon fraction of donor
                eps = a[i] * np.random.rand() * 0.1
                if eps <= 0:
                    continue
                a_new = a.copy()
                a_new[i] -= eps
                a_new[j] += eps
                # evaluate
                new_score = evaluate_sequence(a_new.tolist())
                if new_score < best_score:
                    best_seq = a_new.copy()
                    best_score = new_score
                    a = a_new
                    print(f"[swap] n={n} improved to {best_score:.6f}")
    
    # final fallback: uniform if none found
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best score {best_score:.6f} length {len(best_seq)}")
    return best_seq.tolist()

"""
Random‑sparse + local‑swap optimizer for the max‑autoconvolution objective.

The evaluation metric is   M = max_k (a*a)[k]   after the vector a has been
scaled to the required ℓ₁‑norm √(2·n).  A perfectly flat convolution would give
M≈1 (the theoretical lower bound).  Dense vectors (e.g. the uniform sequence)
produce M≈2, while a carefully chosen sparse support can bring the peak well
below 1.5.

Algorithm
---------

1. **Initialisation**
   * Load the previously discovered construction (`height_sequence_1`) if it
     exists and keep it as the first incumbent.
   * Choose a handful of problem sizes `n` (from 500 up to ≈10000).  Larger
     `n` offers more degrees of freedom for flattening the convolution.

2. **Random sparse sampling**
   * For a given length `n` we repeatedly draw a random support set `S` of
     size `m`.  The size is sampled around √(2n) (the “natural” scale) but
     allowed to vary between 0.4·√(2n) and 2.0·√(2n).
   * On the support we either put an *equal* weight (`√(2n)/m`) or a *random*
     positive weight (drawn from `Exp(1)`) that is subsequently renormalised
     to the required ℓ₁‑norm.
   * The candidate is evaluated with `evaluate_sequence`.  Whenever it beats
     the incumbent we store it.

3. **Cheap local refinement**
   * Starting from the best vector found so far we run a simple hill‑climb:
       – pick a donor index `i` (currently >0) and a receiver index `j`
         (currently zero);
       – move a small fraction of the donor’s mass to the receiver;
       – keep the move only if the maximal convolution value decreases.
   * The refinement is repeated until the time budget expires or a large
     number of proposals have been tried.

4. **Fallback**
   * If for any reason no candidate survived, fall back to a uniform vector.

The whole search stops after the allotted budget (`budget_s` seconds) and
returns the best list of heights found.  The method uses only NumPy, no
external solvers, and runs comfortably within the 2‑CPU limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _eval(a: np.ndarray) -> float:
    """Convenient wrapper – converts to list before calling the provided evaluator."""
    return evaluate_sequence(a.tolist())


# ----------------------------------------------------------------------
def _random_sparse_candidate(n: int, target_sum: float) -> np.ndarray:
    """
    Build a random sparse vector of length ``n`` with ℓ₁‑norm = ``target_sum``.
    The support size ``m`` is sampled near √(2·n) and the values on the support
    are either equal or drawn from an exponential law (chosen uniformly at
    random).
    """
    # typical scale for the support
    base_m = math.sqrt(2.0 * n)
    scale = np.random.uniform(0.4, 2.0)          # allow a wide range of sparsities
    m = int(max(2, min(n, scale * base_m)))

    # random support without replacement
    support = np.random.choice(n, size=m, replace=False)

    # choose between equal weights and random positive weights
    if np.random.rand() < 0.5:
        # equal weights – the simplest possible shape
        values = np.full(m, target_sum / m, dtype=float)
    else:
        # random positive weights (exponential distribution gives many tiny values)
        raw = np.random.exponential(scale=1.0, size=m)
        values = raw * (target_sum / raw.sum())

    a = np.zeros(n, dtype=float)
    a[support] = values
    return a


# ----------------------------------------------------------------------
def _hill_climb(
    a: np.ndarray,
    n: int,
    target_sum: float,
    best_score: float,
    deadline: float,
    max_proposals: int = 40000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap coordinate‑swap hill‑climb.
    Repeatedly moves a small amount of mass from a positive entry to a zero entry.
    Accepts a move only if the maximal convolution value decreases.
    """
    a_cur = a.copy()
    # Pre‑allocate the zero‑index mask for speed
    while time.time() < deadline and max_proposals > 0:
        max_proposals -= 1

        # donors: indices with a positive mass
        donor_idx = np.where(a_cur > 1e-12)[0]
        if donor_idx.size == 0:
            break
        i = np.random.choice(donor_idx)

        # receivers: currently zero entries (or effectively zero)
        zero_idx = np.where(a_cur <= 1e-12)[0]
        if zero_idx.size == 0:
            break
        j = np.random.choice(zero_idx)

        # transfer a modest fraction of donor's mass
        eps = a_cur[i] * np.random.rand() * 0.07   # up to 7 % of the donor
        if eps <= 0.0:
            continue

        a_new = a_cur.copy()
        a_new[i] -= eps
        a_new[j] += eps

        # keep exact ℓ₁‑norm (numerical drift may appear)
        a_new *= target_sum / a_new.sum()

        new_score = _eval(a_new)
        if new_score < best_score - 1e-9:          # strict improvement
            a_cur = a_new
            best_score = new_score
            print(f"[hill] improved to {best_score:.6f} (len={n})")
    return a_cur, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation metric.
    The routine mixes random sparse sampling, a light hill‑climbing phase and
    a fallback to the previously known construction.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0       # small safety margin

    # ------------------------------------------------------------------
    # Load the previously discovered construction if it exists
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        # Normalise to the required ℓ₁‑norm for its length
        n0 = base.shape[0]
        target0 = math.sqrt(2.0 * n0)
        base = np.maximum(base, 0.0)
        if base.sum() > 0.0:
            base = base * (target0 / base.sum())
        else:
            base = np.full(n0, target0 / n0)
        best_seq = base.copy()
        best_score = _eval(best_seq)
        print(f"[init] length={n0}, score={best_score:.6f}")
    except Exception:
        # No previous construction – start from a uniform vector of modest size
        n0 = 1500
        target0 = math.sqrt(2.0 * n0)
        best_seq = np.full(n0, target0 / n0)
        best_score = _eval(best_seq)
        print("[init] no previous sequence – using uniform fallback")

    # ------------------------------------------------------------------
    # Choose a set of lengths to explore.
    # We keep the total work roughly proportional to the remaining budget.
    length_candidates = [500, 1000, 2000, 4000, 8000, 12000]
    max_n = 15000                               # hard upper bound to keep FFT cheap
    lengths = [l for l in length_candidates if l <= max_n]

    # Main optimisation loop over the selected lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # Allocate a fraction of the remaining time to this length.
        # Roughly 30 % of the remaining budget is spent on random sampling,
        # the rest is kept for the final hill‑climb.
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        sampling_deadline = time.time() + 0.30 * remaining

        # --------------------------------------------------------------
        # Random sparse sampling phase
        while time.time() < sampling_deadline:
            cand = _random_sparse_candidate(n, target_sum)
            score = _eval(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"[rand] n={n}, score={best_score:.6f}")

        # --------------------------------------------------------------
        # Light local refinement (swap‑hill‑climb)
        if time.time() < deadline:
            best_seq, best_score = _hill_climb(
                a=best_seq,
                n=n,
                target_sum=target_sum,
                best_score=best_score,
                deadline=deadline,
                max_proposals=20000,
            )

    # ------------------------------------------------------------------
    # Final safety net: if for some reason we have no candidate, fall back
    # to a uniform vector of a modest length.
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = _eval(best_seq)
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
