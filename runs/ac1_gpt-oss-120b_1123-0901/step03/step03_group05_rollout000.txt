def conv_max(a):
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[:2 * n - 1]
    M = conv.max()
    return float(M), conv

"""
Hybrid LP‑gradient optimiser for the step‑function maximal autocorrelation
problem.

The algorithm works for a fixed length `n` and a target L1‑norm
`S = sqrt(2 n)`.  Starting from an initial non‑negative vector `a`
the following loop is performed until the time budget expires:

1. Compute the full self‑convolution `c = a * a` by FFT and obtain the
   maximal value `M = max(c)`.  The set of *active* shifts where the
   convolution is (almost) equal to `M` is identified.

2. For the current vector `a` the convolution with an *unknown* vector
   `b` is linear:
        (a * b)[k] = Σ_i a_i * b_{k‑i}.
   Keeping the maximal value at most `M` for all active shifts yields a
   collection of linear inequalities `A b ≤ M`.  Adding the constraints
   `b ≥ 0` we solve the linear program

        maximise   1ᵀ b          (equivalently minimise –1ᵀ b)
        subject to A b ≤ M ,   b ≥ 0.

   The solution `b` is a feasible direction that tends to increase the
   total mass while respecting the active convolution bound.

3. Scale `b` to the required L1‑norm (`g = b · S / Σ b`).  A few trial
   step sizes `t` are examined and the convex combination
        a' = (1‑t) a + t g
   is accepted if it reduces the maximal convolution value.
   If no improvement occurs we inject a tiny random jitter to escape
   plateaus.

4. After the main loop a cheap coordinate‑swap hill‑climb (`local_search`)
   moves a small amount of mass between two positions whenever this
   strictly lowers the maximal convolution.

The outer driver (`propose_candidate`) explores several lengths (the
original length and a few enlargements), uses the previously discovered
construction (`height_sequence_1`) for interpolation, adds random
Dirichlet restarts, and keeps the best sequence seen across all runs.
All operations respect the non‑negativity and L1‑norm constraints.
"""

import time
import math
import random
import numpy as np
from scipy.optimize import linprog


def _conv_max(a: np.ndarray):
    """
    Full autocorrelation a*a via FFT and return (max value, full convolution).
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return float(conv.max()), conv


def _build_A_ub(f: np.ndarray, active_shifts):
    """
    Build the matrix A for the linear constraints A @ b <= M,
    where each row corresponds to an active shift k and
    row[k][j] = f[k‑j] (zero when index out of range).
    """
    n = f.shape[0]
    m = len(active_shifts)
    if m == 0:
        return np.empty((0, n), dtype=np.float64)
    A = np.empty((m, n), dtype=np.float64)
    idx = np.arange(n)
    for row, k in enumerate(active_shifts):
        j = k - idx
        mask = (j >= 0) & (j < n)
        A[row, mask] = f[j[mask]]
        # entries where mask is False stay zero
    return A


def _solve_lp_direction(f: np.ndarray, active_shifts, M):
    """
    Solve the LP:
        maximise Σ b_i  (implemented as minimise -Σ b_i)
        s.t.    A b <= M  for all active shifts,
                b >= 0.
    Returns the optimal vector b (may be all zeros if LP fails).
    """
    A = _build_A_ub(f, active_shifts)
    if A.shape[0] == 0:
        # no active constraints – any non‑negative vector works;
        # return a uniform vector as a harmless direction.
        return np.full_like(f, 1.0)
    b_ub = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones_like(f)                    # minimise -Σ b_i  → maximise Σ b_i
    bounds = [(0, None)] * f.shape[0]

    # Use the high‑performance HiGHS simplex/MIP solver
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    # fallback – return zero direction (will trigger jitter)
    return np.zeros_like(f)


def _hybrid_optimize(init_a: np.ndarray, n: int,
                     target_sum: float, deadline: float):
    """
    Hybrid optimiser that repeatedly:
        * identifies active shifts,
        * solves the LP for a feasible direction,
        * tries a few step sizes,
        * injects jitter when stuck.
    Returns the best vector found and its maximal convolution value.
    """
    a = init_a.copy()
    # enforce the L1‑norm exactly
    s = a.sum()
    a = a * (target_sum / s) if s > 0 else np.full(n, target_sum / n)

    best_a = a.copy()
    best_M, _ = _conv_max(a)

    # step schedule (large → small) and jitter frequency
    t_schedule = [0.30, 0.20, 0.10, 0.05, 0.02, 0.01]
    jitter_every = 250
    iteration = 0
    max_iter = 2000

    while time.time() < deadline and iteration < max_iter:
        iteration += 1

        M, conv = _conv_max(a)
        tol = max(1e-12, M * 1e-12)
        active = np.where(conv >= M - tol)[0]

        # 1) LP direction
        b = _solve_lp_direction(a, active, M)

        # 2) scale to the required L1‑norm
        sum_b = b.sum()
        if sum_b > 0:
            g = b * (target_sum / sum_b)
        else:
            g = a.copy()                     # fallback to current point

        improved = False
        # 3) try several mixing coefficients
        for t in t_schedule:
            cand = (1.0 - t) * a + t * g
            cand = np.maximum(cand, 0.0)
            cand = cand * (target_sum / cand.sum())
            M_cand, _ = _conv_max(cand)
            if M_cand < best_M - 1e-12:
                a = cand
                best_a = cand.copy()
                best_M = M_cand
                improved = True
                print(f"[len {n:5d} iter {iteration:5d}] step {t:.3f} → M {best_M:.6f}")
                break

        # 4) if no improvement, jitter occasionally
        if not improved and (iteration % jitter_every == 0):
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

    return best_a, best_M


def _local_search(a_start: np.ndarray, n: int,
                  target_sum: float, cur_M: float,
                  deadline: float, max_proposals: int = 30000):
    """
    Simple coordinate‑swap hill‑climb.  Moves a small amount of mass from one
    index to another if the maximal convolution value strictly decreases.
    """
    a = a_start.copy()
    best_M = cur_M
    proposal = 0
    while time.time() < deadline and proposal < max_proposals:
        proposal += 1
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        # pick a different index j
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        # transfer up to 5 % of the donor's mass
        eps = a[i] * np.random.rand() * 0.05
        if eps <= 0:
            continue
        new_a = a.copy()
        new_a[i] -= eps
        new_a[j] += eps
        M_new, _ = _conv_max(new_a)
        if M_new < best_M - 1e-12:
            a = new_a
            best_M = M_new
            print(f"[local] improvement to {best_M:.6f} (prop {proposal})")
    return a, best_M

    """

    random.seed(seed)

    best_seq = base.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] length={n0}, initial score={best_score:.6f}")

            if time.time() >= deadline:

            # ------- hybrid optimisation -------
            seq_opt, M_opt = _hybrid_optimize(init_a, n, target_sum, deadline)

            # Evaluate via the provided function (should equal M_opt)
            score_opt = evaluate_sequence(seq_opt.tolist())
            if score_opt < best_score:
                best_score = score_opt
                best_seq = seq_opt.copy()
                print(f"[update] len {n} restart {r} new best score {best_score:.6f}")

            # ------- cheap local search -------
            if time.time() < deadline:
                seq_loc, M_loc = _local_search(seq_opt, n, target_sum,
                                               M_opt, deadline,
                                               max_proposals=20000)
                score_loc = evaluate_sequence(seq_loc.tolist())
                if score_loc < best_score:
                    best_score = score_loc
                    best_seq = seq_loc.copy()
                    print(f"[local‑update] len {n} restart {r} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – never should happen, but keep a safe uniform vector
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
