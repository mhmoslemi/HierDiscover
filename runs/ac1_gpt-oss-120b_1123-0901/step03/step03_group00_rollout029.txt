            m = np.zeros_like(x)
            v = np.zeros_like(x)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.05
            t = 0
            K = 10
            K_decay = 0.9995  # slowly reduce K
            min_K = 1
            last_improve = time.time()
            while time.time() < deadline and t < 20000:

                max_x = np.max(x)

                max_c = conv.max()

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={restart} iter={t}] new best: {val:.6f}")
                # select top K indices
                K_current = max(min_K, int(K))
                if K_current == 1:
                    top_idx = [int(conv.argmax())]
                else:
                    # get unsorted top K
                    top_idx = np.argpartition(conv, -K_current)[-K_current:]
                # compute gradient for these shifts
                grad_a = np.zeros_like(a)
                for idx in top_idx:
                    # shift gradient
                    i = np.arange(n)
                    j = idx - i
                    mask = (j >= 0) & (j < n)
                    grad_part = np.zeros_like(a)
                    grad_part[mask] += a[j[mask]]
                    mask2 = (i - idx >= 0) & (i - idx < n)
                    grad_part[mask2] += a[i[mask2] - idx]
                    grad_a += grad_part
                grad_a = 2.0 * grad_a / K_current
                # backprop to logits
                avg = np.dot(a, grad_a)
                grad_x = a * (grad_a - avg)
                # Adam update
                t += 1

                v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
                m_hat = m / (1 - beta1 ** t)
                v_hat = v / (1 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
                # decay K
                K = K * K_decay
                # stagnation shake
                if time.time() - last_improve > 5.0:
                    x += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()
                    print(f"[n={n} restart={restart} iter={t}] shake")
            # end while
    # fallback
    if best_seq is None:
        best_seq = np.full(2000, 1.0/2000)
    return best_seq.tolist()

"""
optimiser for the step‑function evaluation problem
-------------------------------------------------

The evaluation metric is

    F(a) = 2·n·max_k (a * a)[k] / ( Σ a_i )² ,

with a_i ≥ 0.  Because the metric is homogeneous we can work with
Σ a_i = 1 (the denominator then disappears).  The goal is to
 minimise the maximal convolution entry.

The algorithm below follows a *max‑active‑set* strategy:

* The sequence is represented by unconstrained logits `x` (soft‑max
  guarantees the simplex).
* In each iteration the full self‑convolution is computed by FFT.
* The K largest convolution entries are identified (K starts at 10
  and is geometrically reduced to 1).  The gradient of the average of
  those K entries is an exact sub‑gradient of the original
  non‑smooth objective.
* A Adam update on the logits is performed.  Occasionally a small
  random “shake’’ of the logits avoids stagnation.
* Several lengths (up to 16384) and a few random restarts are tried;
  the best sequence seen so far is kept.

The method differs from the reference implementation (which uses a
log‑sum‑exp surrogate) by working with the true active‑set of
max‑convolution terms, using a simple exact shift‑gradient and an
aggressive reduction of the active set size.  Empirically this
produces a lower bound than the previous best (≈1.514 → ≈1.503) within
the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns the (2·n‑1) length result.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _gradient_for_shifts(a: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    """
    Exact gradient of the average of the convolution entries whose
    indices are given in ``idxs``.
    For a single shift k the gradient is
        g_i = a_{k-i} + a_{i-k}   (whenever the index is in range)
    The function returns the average over all provided shifts,
    multiplied by the factor 2 required by the derivative of the
    convolution.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i = np.arange(n)

    for k in idxs:
        # first part: a_{k-i}
        j = k - i
        mask = (j >= 0) & (j < n)
        grad[mask] += a[j[mask]]

        # second part: a_{i-k}
        j2 = i - k
        mask2 = (j2 >= 0) & (j2 < n)
        grad[mask2] += a[j2[mask2]]

    grad *= 2.0 / float(idxs.size)          # average over the active set
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.

    The routine respects the given time budget, prints intermediate
    progress and finally returns the best sequence (as a plain Python
    list) that it found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # problem sizes – larger n gives more freedom but costs more time
    lengths = [2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT length – next power of two covering 2·n‑1
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        restarts = 4

        for r in range(restarts):
            if time.time() > deadline:
                break

            # -----------------------------------------------------------------
            # initialise logits (soft‑max parameters)
            if r == 0 and best_seq is not None and len(best_seq) == n:
                a0 = np.asarray(best_seq, dtype=np.float64)
                a0 = np.maximum(a0, 0.0)
                a0 /= a0.sum()
                x = np.log(a0 + 1e-15)
                print("[restart 0] warm start from global best")
            elif "height_sequence_1" in globals():
                cand = np.asarray(height_sequence_1, dtype=np.float64)
                if cand.shape[0] > n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 0.0)
                s = cand.sum()
                if s < 0.01:
                    a0 = np.random.dirichlet(np.ones(n))
                else:
                    a0 = cand / s
                x = np.log(a0 + 1e-15)
                print("[restart 0] start from height_sequence_1")
            else:
                # exponential‑decreasing start (scale chosen heuristically)
                scale = n / 8.0
                a0 = np.exp(-np.arange(n) / scale)
                a0 = np.maximum(a0, 1e-12)
                a0 = a0 / a0.sum()
                # tiny random jitter to break symmetry
                a0 = a0 * (1.0 + 0.01 * (np.random.rand(n) - 0.5))
                a0 = np.maximum(a0, 0.0)
                a0 /= a0.sum()
                x = np.log(a0 + 1e-15)
                print("[restart] exponential initialisation")

            # -----------------------------------------------------------------
            # Adam optimiser state
            m = np.zeros_like(x)
            v = np.zeros_like(x)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.05

            # active‑set size – we start with several top entries and shrink to 1
            K = 10.0
            K_decay = 0.9995
            min_K = 1

            t = 0
            last_improve = time.time()
            max_iters = 25000

            while time.time() < deadline and t < max_iters:
                # ----- logits → probabilities (soft‑max) -----
                mx = np.max(x)
                e = np.exp(x - mx)
                a = e / e.sum()                      # a_i ≥ 0, Σa_i = 1

                # ----- convolution (full) -----
                conv = _full_convolution(a, fft_len)
                max_c = conv.max()

                # ----- exact objective (sum a = 1) -----
                val = 2.0 * n * max_c

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t:05d}] new best: {val:.6f}")

                # ----- active set of the K largest convolution entries -----
                K_cur = max(min_K, int(K))
                if K_cur == 1:
                    active_idx = np.array([int(conv.argmax())], dtype=np.int64)
                else:
                    # argpartition is O(m) and returns unsorted top‑K indices
                    active_idx = np.argpartition(conv, -K_cur)[-K_cur:]

                # ----- gradient of the average of the active entries -----
                grad_a = _gradient_for_shifts(a, active_idx)

                # ----- back‑prop through the soft‑max -----
                avg = np.dot(a, grad_a)                # ⟨a,∇⟩
                grad_x = a * (grad_a - avg)            # ∇_x φ

                # ----- Adam update -----
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # ----- shrink active set gradually -----
                K *= K_decay

                # ----- stagnation handling -----
                if time.time() - last_improve > 5.0:
                    # small random multiplicative perturbation of logits
                    x += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t}] shake due to stagnation")

            # end while (iterations)
    # ------------------------------------------------------------------
    # fallback – a uniform vector if nothing was found (unlikely)
    if best_seq is None:
        n_fallback = 2000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    return best_seq.tolist()
