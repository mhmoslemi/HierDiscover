def _optimize_eg(p_start, n, deadline, max_iters=100000):
    p = p_start.copy()
    # ensure p sum = 1
    p = p / p.sum()
    logp = np.log(p + 1e-30)
    N = _next_pow2(3*n - 2)
    best_local_score = float('inf')
    best_local_p = p.copy()
    it = 0
    beta_start = 0.1
    beta_end = 8.0
    beta_len = 50000
    while time.time() < deadline and it < max_iters:
        it += 1
        # compute conv (full)
        P = np.fft.rfft(p, n=N)
        conv = np.fft.irfft(P * P, n=N)[:2*n-1]
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_p = p.copy()
        # gradient: rev = correlation with weights
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_len)
        # compute softmax weights
        w = np.exp(beta * (conv - M))
        # correlation rev = sum_k w_k * p_{k-i}
        # Equivalent: convolution of p with reversed w
        # Use FFT: p and reversed w.
        # Compute reverr
        # we can compute rev via convolution:
        # rev_i = sum_k w_k * p_{k-i}
        # This is correlation; we can compute via FFT:
        # fft(p) * fft(w_reverse). Same as earlier code.
        # We'll compute rev via FFT.
        # Precompute N for conv_rev size.
        # w_rev = w[::-1]
        W_rev = np.fft.rfft(w[::-1], n=N)
        rev = np.fft.irfft(np.fft.rfft(p, n=N) * W_rev, n=N)
        rev = rev[:n + (2*n - 1) - 1] # length n + L - 1 = 3n-2? Actually earlier used more steps, but we only need rev length n: rev = rev[L-n:L][::-1] where L=2n-1.
        # We'll compute as earlier:
        L = 2*n - 1
        rev_full = rev[: n + L - 1]
        rev = rev_full[L - n : L][::-1]
        grad = 4.0 * n * rev
        eta = (1.0 / (4.0 * n)) / (1.0 + it * 1e-4)
        logp -= eta * grad
        # optionally jitter
        if it % 5000 == 0:
            logp += np.random.randn(n) * 0.0005
        # project back onto simplex
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)
        p = exp_shift / exp_shift.sum()
    # Return best local p
    return best_local_p, best_local_score

"""
Propose a low‑peak step‑function by a hybrid of
exponentiated‑gradient (mirror descent on the simplex) and a fast
O(n) simulated‑annealing refinement.

The algorithm works on the probability vector p (∑p=1).  The score that
evaluate_sequence returns is 2·n·max(p∗p) and therefore independent of
the overall scale of the step heights.  We keep p non‑negative, use a
FFT for the EG phase (fast gradient) and then improve the result with a
mass‑transfer SA that updates the full self‑convolution incrementally
(in O(n) per move) by exploiting the linearity of the convolution.
Multiple length scales and several initial shapes (uniform, beta‑shaped
U‑distributions) are tried; the best construction found before the
time budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """Full self‑convolution p∗p using one FFT of size N."""
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k·p_{k-i}   (i = 0…n‑1)
    via a correlation (FFT‑based).
    """
    n = p.shape[0]
    L = w.shape[0]                 # = 2·n‑1
    # FFT of p
    P = np.fft.rfft(p, n=N)
    # FFT of reversed weights
    W_rev = np.fft.rfft(w[::-1], n=N)
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    conv_rev = conv_rev[: n + L - 1]          # keep linear part
    # Extract rev = conv_rev[L‑n : L] and reverse
    rev = conv_rev[L - n : L][::-1]
    return rev


# ----------------------------------------------------------------------
def _optimize_eg(p_start: np.ndarray,
                 n: int,
                 deadline: float,
                 max_iters: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient minimisation of 2·n·max(p∗p) for a fixed length.
    Returns the best probability vector seen and its (approx.) score.
    """
    p = p_start.copy()
    p = np.maximum(p, 1e-30)               # avoid log(0)
    p /= p.sum()
    logp = np.log(p)

    N = _next_pow2(3 * n - 2)               # size for both convolutions
    best_score = float("inf")
    best_p = p.copy()

    beta_start = 0.1
    beta_end = 8.0
    beta_len = max_iters // 2
    jitter_interval = 2000
    jitter_scale = 5e-4

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        # --- self‑convolution ---
        conv = _full_convolution_fft(p, N)
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[EG  n={n:5d}  it={it:6d}]  new local best = {cur_score:.6f}")

        # --- soft‑max weights (smooth max) ---
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_len)
        w = np.exp(beta * (conv - M))       # w_k ≤ 1, w_max = 1

        # --- gradient (correlation) ---
        rev = _weighted_rev_fft(p, w, N)    # length n
        grad = 4.0 * n * rev                # ∇_p (2·n·M_β)

        # --- step‑size (decaying) ---
        eta = (1.0 / (4.0 * n)) / (1.0 + it * 1e-4)

        # --- EG update in log‑space ---
        logp -= eta * grad

        # occasional jitter to avoid stagnation
        if it % jitter_interval == 0:
            logp += np.random.randn(n) * jitter_scale

        # --- projection onto the simplex (KL‑projection) ---
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)  # avoid overflow
        p = exp_shift / exp_shift.sum()

    return best_p, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(p0: np.ndarray,
                         conv0: np.ndarray,
                         n: int,
                         score0: float,
                         deadline: float,
                         max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Fast O(n) SA that moves a small amount of mass from one index to another.
    The convolution array is updated incrementally.
    Returns the best probability vector found and its score.
    """
    p = p0.copy()
    conv = conv0.copy()
    cur_score = score0
    best_score = cur_score
    best_p = p.copy()

    # temperature schedule (exponential decay)
    T0 = 0.02
    tau = max_iters / 5.0

    step_factor = 0.25          # initial fraction of source mass to move
    min_step = 1e-5

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        # pick a source index with non‑zero mass
        i = np.random.randint(n)
        if p[i] <= 1e-12:
            continue
        # pick a different target index
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # amount to transfer
        delta = step_factor * p[i]
        if delta < min_step:
            delta = min_step
        # keep source non‑negative
        if delta > p[i]:
            delta = p[i]

        # keep a copy of p for possible rollback
        p_old = p.copy()

        # --- update convolution (linear in n) ---
        # conv[i:i+n] -= 2*delta * p_old
        conv[i:i + n] -= 2.0 * delta * p_old
        # conv[j:j+n] += 2*delta * p_old
        conv[j:j + n] += 2.0 * delta * p_old
        # quadratic corrections
        conv[2 * i] += delta * delta
        conv[2 * j] += delta * delta
        conv[i + j] -= 2.0 * delta * delta

        # --- update probabilities ---
        p[i] -= delta
        p[j] += delta

        # compute new score
        new_M = conv.max()
        new_score = 2.0 * n * new_M

        # acceptance test
        T = T0 * math.exp(-it / tau)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if np.random.rand() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_p = p.copy()
        else:
            # rollback both p and conv
            # revert convolution with the *original* p_old
            conv[i:i + n] += 2.0 * delta * p_old
            conv[j:j + n] -= 2.0 * delta * p_old
            conv[2 * i] -= delta * delta
            conv[2 * j] -= delta * delta
            conv[i + j] += 2.0 * delta * delta
            p = p_old

        # slowly shrink step size
        if it % 5000 == 0:
            step_factor = max(min_step, step_factor * 0.97)

    return best_p, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Hybrid EG + SA search for the step‑function that minimises
    ``evaluate_sequence``.  Returns the best height sequence found before
    the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Load a previously discovered construction if it exists
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        if base.ndim != 1 or base.size == 0:
            raise Exception()
    except Exception:
        base = None

    # initialise best probability vector (sum = 1)
    if base is not None:
        best_p = np.maximum(base, 0.0)
        best_p /= best_p.sum()
        best_n = best_p.shape[0]
    else:
        best_n = 500
        best_p = np.full(best_n, 1.0 / best_n)

    def score_from_prob(p: np.ndarray, n: int) -> float:
        """Exact objective value 2·n·max(p∗p)."""
        N = _next_pow2(2 * n - 1)
        conv = _full_convolution_fft(p, N)
        return 2.0 * n * conv.max()

    # store best overall
    best_score = score_from_prob(best_p, best_n)
    print(f"[init] length={best_n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Length schedule (increasing degrees of freedom)
    length_schedule = [500, 1000, 2000, 4000, 8000]
    length_schedule = [l for l in length_schedule if l >= 300 and l <= 8000]

    for n in length_schedule:
        if time.time() >= deadline:
            break

        # ---- initialise p for this length (interpolation) ----
        if best_n != n:
            old_x = np.linspace(0.0, 1.0, best_n)
            new_x = np.linspace(0.0, 1.0, n)
            p_interp = np.interp(new_x, old_x, best_p)
            p_interp = np.maximum(p_interp, 0.0)
            p_interp /= p_interp.sum()
        else:
            p_interp = best_p.copy()

        # collection of diverse starting points
        inits = [p_interp]

        # beta‑shaped U‑distributions (edges heavy)
        xs = np.linspace(0.0, 1.0, n)
        for a in (0.25, 0.35, 0.45):            # a < 1 gives U‑shape
            beta_shape = np.power(xs + 1e-12, a - 1) * np.power(1.0 - xs + 1e-12, a - 1)
            beta_shape /= beta_shape.sum()
            inits.append(beta_shape)

        # uniform
        inits.append(np.full(n, 1.0 / n))

        # ------------------------------------------------------------------
        for init_id, p_start in enumerate(inits):
            if time.time() >= deadline:
                break

            # ---- EG phase (quick but strong descent) ----
            p_eg, _ = _optimize_eg(p_start, n, deadline)

            # evaluate after EG
            cur_score = score_from_prob(p_eg, n)
            if cur_score < best_score:
                best_score = cur_score
                best_p = p_eg.copy()
                best_n = n
                print(f"[EG  n={n:5d} init={init_id}]  global best = {best_score:.6f}")

            # ---- SA refinement ----
            # compute full convolution once (FFT based, O(n log n))
            N_conv = _next_pow2(2 * n - 1)
            conv = _full_convolution_fft(p_eg, N_conv)
            # SA loop (fast O(n) updates)
            p_sa, sa_score = _simulated_annealing(
                p0=p_eg,
                conv0=conv,
                n=n,
                score0=cur_score,
                deadline=deadline,
                max_iters=120_000,
            )

            # final evaluation of SA result (using exact conv)
            final_score = score_from_prob(p_sa, n)
            if final_score < best_score:
                best_score = final_score
                best_p = p_sa.copy()
                best_n = n
                print(f"[SA  n={n:5d} init={init_id}]  global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Produce the final step‑function (scale to the canonical √(2n) norm)
    S = math.sqrt(2.0 * best_n)
    final_seq = (best_p * S).tolist()
    print(f"[done] best score = {best_score:.6f}, length = {best_n}")
    return final_seq
