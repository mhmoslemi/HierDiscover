"""
Propose a step‑function sequence that minimizes the evaluate_sequence objective.

Algorithm
---------

* **Preparation** – obtain a starting construction (`height_sequence_1`) if
  available, otherwise fall back to a uniform vector.

* **Length sweep** – explore several lengths (≈ 2100, 3200, 4800, 6400, 8000).
  Larger lengths give a lower theoretical average of the convolution and thus
  a better chance to reduce the peak.

* **Sub‑gradient descent** – for a fixed length we run a simple projected
  sub‑gradient method:

    - compute the full autocorrelation `c = a * a` by FFT,
    - locate a shift `k` attaining the current maximum (`max(c)`),
    - the sub‑gradient of `max(c)` w.r.t. `a` is `2 * rev_k`,
      where `rev_k[i] = a[k-i]` for valid indices and `0` otherwise.
    - take a step `a ← a – η·rev_k` (η decays as 1/√t),
      project onto the feasible set (non‑negative, fixed L¹‑norm).

    Occasionally a tiny random jitter is added to escape plateaus.

* **Greedy local search** – after the descent phase we perform a cheap hill‑climb
  that repeatedly moves a small amount of mass from a “donor’’ index to a
  “receiver’’ index if the move reduces the maximal convolution value.
  The search stops when the time budget is exhausted.

* **Best‑ever bookkeeping** – the best sequence (lowest `max(a*a)`) encountered
  over all lengths and restarts is kept and finally returned.

The whole routine respects the 1000 s budget, prints intermediate progress,
and requires only NumPy and the already‑imported `evaluate_sequence` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """
    Full non‑circular convolution a*a via FFT.
    Returns (max_value, argmax_index, full_convolution_array).
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())        # next power of two
    A = np.fft.rfft(a, n=N)
    B = A * A                                   # element‑wise product
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]    # keep only linear part
    idx = int(conv.argmax())
    return float(conv.max()), idx, conv


def _rev_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient for a given shift k:
    rev[i] = a[k-i] if (k-i) in [0, n), else 0.
    """
    n = a.shape[0]
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    rev = np.zeros_like(a)
    rev[mask] = a[j[mask]]
    return rev


# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray,
                         S: float,
                         deadline: float,
                         max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the objective   max_k (a*a)[k]
    while keeping a ≥ 0 and sum(a) = S.
    Returns the best vector seen and its corresponding max‑convolution.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_M = float("inf")

    # initial step‑size – about the typical magnitude of a_i
    eta0 = 0.5 * S / n

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        M, k_max, conv = _conv_max_fft(a)

        # record improvement
        if M < best_M:
            best_M = M
            best_a = a.copy()
            print(f"[len {n:5d} it {it:7d}] new best M = {best_M:.6f}")

        # choose a random active shift (all shifts attaining the max are safe)
        active = np.where(conv >= M - 1e-12)[0]
        k = np.random.choice(active)
        rev = _rev_for_shift(a, k)

        # decreasing step size
        eta = eta0 / math.sqrt(it)

        # gradient step (sub‑gradient of max is +2*rev, we move opposite)
        a = a - eta * rev

        # projection onto the feasible simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (S / s)
        else:
            a = np.full_like(a, S / n)

        # occasional jitter to disturb possible stagnation
        if it % 4000 == 0:
            jitter = np.random.randn(n) * 0.001 * S / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (S / a.sum())

    return best_a, best_M


# ----------------------------------------------------------------------
def _local_search(a_start: np.ndarray,
                  S: float,
                  cur_best: float,
                  deadline: float,
                  max_proposals: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Very cheap greedy hill‑climb: repeatedly shift a small amount of mass
    from a donor index to a receiver index, accepting only moves that lower
    the maximal convolution value.
    """
    a = a_start.copy()
    n = a.shape[0]
    best_M = cur_best

    for prop in range(max_proposals):
        if time.time() >= deadline:
            break

        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        eps = a[i] * np.random.rand() * 0.05   # up to 5 % of donor mass
        if eps <= 0:
            continue

        new_a = a.copy()
        new_a[i] -= eps
        new_a[j] += eps

        # sum is unchanged → no need to renormalise
        M_new, _, _ = _conv_max_fft(new_a)

        if M_new < best_M - 1e-12:
            a = new_a
            best_M = M_new
            print(f"[local] improvement to {best_M:.6f} (prop {prop})")

    return a, best_M

    """
    Main entry point used by the evaluation harness.
    Searches for a sequence of non‑negative heights that minimises
    evaluate_sequence (i.e. the peak of a*a after normalising the L¹‑norm).
    Returns the best sequence discovered before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0    # small safety margin

    # ------------------------------------------------------------------
    # Load the previously found construction, if it exists.
    try:

    except Exception:
        base = None

    # Global best bookkeeping
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Lengths to explore (original length + several enlargements)
    base_len = base.shape[0] if base is not None else 2100
    length_factors = [1.0, 1.2, 1.4, 1.6, 2.0]
    lengths = sorted({int(base_len * f) for f in length_factors})
    max_len = 8000
    lengths = [l for l in lengths if 300 <= l <= max_len]

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        S = math.sqrt(2.0 * n)          # required L¹‑norm

        for restart in range(3):
            if time.time() >= deadline:
                break

            # ----- initialise -------------------------------------------------
            if restart == 0 and base is not None:
                # linear interpolation of the existing base onto the new grid
                old_x = np.linspace(0.0, 1.0, base_len)
                new_x = np.linspace(0.0, 1.0, n)

                a0 = np.maximum(a0, 0.0)

                a0 = np.random.rand(n)

            # enforce the required sum

            # ----- sub‑gradient descent --------------------------------------
            # allocate an iteration budget (roughly proportional to remaining time)
            time_left = deadline - time.time()

            # we give each length a few hundred thousand steps at most.
            max_iters = int(min(300_000, max(10_000, time_left * 300)))
