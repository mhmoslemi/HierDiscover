"""
propose_candidate – search for a low‑peak self‑convolution step function.

Algorithm
---------
* Work with the probability vector p ( Σp = 1 ) and the canonical scaling
  a = sqrt(2·n)·p, because the objective simplifies to 2·n·max(p∗p).

* For a fixed length n we run a *projected Adam* optimiser on the log‑probabilities.
  – The gradient of the (smoothed) maximum of the convolution is obtained
    via a weighted correlation, computed efficiently with a single FFT.
  – In the first few thousand iterations a soft‑max weight
    w_k = exp(β·(conv_k−M)) is used (β growing from 0.1 to 30)
    to provide a smooth sub‑gradient.
  – Afterwards the hard active set (indices where conv_k is within a tiny
    tolerance of the current maximum) is used, which is exactly the
    sub‑gradient of the true max.
  – Adam’s moment estimates give an adaptive step‑size; we finally project
    back onto the simplex by exponentiation and normalisation.

* Multi‑scale search:
  – A schedule of increasing lengths (≈ 300 … 8000) is processed.
  – For each length we try several restarts:
        • the up‑sampled best solution found so far,
        • a few random Dirichlet initialisations.
  – After each optimisation run we keep the best vector (according to
    the exact formula 2·n·max(p∗p)).

* The whole search respects the given time budget (default 1000 s).  All
  heavy work is done with NumPy FFTs (O(n log n) per iteration), so
  thousands of iterations comfortably fit in the limit.

The function returns the best sequence of heights (list[float]) discovered
before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """
    Compute the full linear self‑convolution p∗p using a single FFT of size N.
    Returns a real array of length 2·len(p)−1.
    """
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k·p_{k−i}   (i = 0..n−1)
    This is a correlation of p with w and is obtained via FFT.
    """
    n = p.shape[0]          # length of p
    L = w.shape[0]          # = 2·n−1
    # FFT of p (size N)
    P = np.fft.rfft(p, n=N)
    # FFT of reversed weights
    W_rev = np.fft.rfft(w[::-1], n=N)
    # convolution of p with reversed w
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    conv_rev = conv_rev[: n + L - 1]          # keep the linear part
    # Extract rev = conv_rev[L‑n : L] and reverse it
    rev = conv_rev[L - n : L][::-1]
    return rev


# ----------------------------------------------------------------------
def _optimize_adam_length(
    n: int,
    p_start: np.ndarray,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Adam‑based optimisation for a fixed length ``n``.
    Returns the best probability vector (sum = 1) and the associated score
    (2·n·max(p∗p)).
    """
    # ---------- hyper‑parameters ----------
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5 / n                     # base learning‑rate (scaled by n)
    jitter_interval = 3000
    jitter_scale = 0.001

    soft_iters = 6000                # number of iterations with soft‑max
    beta_start = 0.1
    beta_end = 30.0

    # ---------- initial state ----------
    logp = np.log(p_start + 1e-30)   # stay in log‑space
    m = np.zeros_like(p_start)       # first moment
    v = np.zeros_like(p_start)       # second moment

    # FFT size sufficient for both self‑convolution (2n‑1) and correlation (3n‑2)
    N = _next_pow2(3 * n - 2)

    best_local_score = float("inf")
    best_local_p = None

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- probability vector (simplex projection) -----
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)       # avoid overflow
        p = exp_shift / exp_shift.sum()

        # ----- self‑convolution and current objective -----
        conv = _full_convolution_fft(p, N)
        M = conv.max()
        cur_score = 2.0 * n * M

        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_p = p.copy()
            print(f"[len {n:5d} iter {it:7d}] new local best = {cur_score:.6f}")

        # ----- gradient of (smoothed) max -----
        if it <= soft_iters:
            beta = beta_start + (beta_end - beta_start) * (it / soft_iters)
            w = np.exp(beta * (conv - M))          # un‑normalised soft‑max weights
        else:
            # hard active set (sub‑gradient of the true max)
            eps_active = 1e-12
            active = (conv >= M - eps_active).astype(float)
            cnt = active.sum()
            if cnt > 0:
                w = active / cnt                    # averaging over the active set
            else:
                w = active

        # correlation needed for the gradient
        rev = _weighted_rev_fft(p, w, N)
        grad = 4.0 * n * rev                        # ∇_p (2·n·max) ≈ 4n·rev

        # ----- Adam update in log‑space -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        update = lr * m_hat / (np.sqrt(v_hat) + eps)

        logp -= update

        # occasional jitter to escape plateaus
        if it % jitter_interval == 0:
            logp += np.random.randn(n) * jitter_scale

    # fallback in case no improvement was ever recorded
    if best_local_p is None:
        best_local_p = p
        best_local_score = cur_score

    return best_local_p, best_local_score
