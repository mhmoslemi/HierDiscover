"""
Propose a low‑peak self‑convolution step function.

Algorithm
---------
* The search works on probability vectors `p` (sum = 1).  The actual
  step heights are `a = sqrt(2·n)·p`.  The objective to minimise is
  `max(a * a) = 2·n·max(p * p)`.
* For each length we run two phases:
  1. **Exponentiated‑gradient (EG)** on the simplex with a soft‑max
     approximation of the maximum.  The gradient is obtained by a
     weighted correlation computed by FFTs.  EG quickly finds a good
     shape.
  2. **LP refinement** – an iteration of the classic “LP‑update”
     from the state‑of‑the‑art paper:
        maximise Σ g_j  subject to  g_j≥0  and  f∗g ≤ M,
     where `f` is the current sequence and `M = max(f∗f)`.
     The LP is solved with `scipy.optimize.linprog`.  The resulting
     `g` is scaled to the correct ℓ₁‑norm and mixed with `f`.
     Repeating this step strictly decreases the peak.
* Several lengths (up to ~3000, where the LP is still tractable) and
  a few random restarts are tried.  The best sequence seen (according
  to `evaluate_sequence`) is returned.
* All heavy work is done with FFTs (O(n log n)) and the LP
  (dense `n×n` matrix) is limited to modest sizes, keeping the total
  running time well below the 1000 s budget.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """Full self‑convolution p∗p using a single FFT of size N."""
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k·p_{k−i}  (correlation of p with w).
    Returned array has length len(p).
    """
    n = p.shape[0]
    L = w.shape[0]                     # = 2·n‑1
    P = np.fft.rfft(p, n=N)
    W_rev = np.fft.rfft(w[::-1], n=N)  # reverse w
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    conv_rev = conv_rev[: n + L - 1]   # linear part
    rev = conv_rev[L - n : L][::-1]
    return rev


# ----------------------------------------------------------------------
def _optimize_eg(p_start: np.ndarray,
                 n: int,
                 deadline: float,
                 max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimisation for a fixed length `n`.
    Returns the best probability vector `p` (sum = 1) found
    and the corresponding `max(p∗p)`.
    """
    p = p_start.copy()
    # keep probabilities in log‑space to avoid underflow
    logp = np.log(p + 1e-30)

    # FFT size that works both for the self‑convolution (2n‑1)
    # and for the correlation with the weight vector (≈3n‑2)
    N = _next_pow2(3 * n - 2)

    best_M = float("inf")
    best_p = p.copy()

    # schedule parameters – tuned for a good trade‑off
    jitter_interval = 3000
    jitter_scale = 0.001
    beta_start = 0.1
    beta_end = 100.0            # sharper approximation of the max
    beta_len = 120_000          # linear increase of beta

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- self‑convolution -----
        conv = _full_convolution_fft(p, N)
        M = conv.max()

        if M < best_M:
            best_M = M
            best_p = p.copy()
            # occasional progress report
            print(f"[EG n={n:5d} it={it:7d}] new local best max(p∗p) = {M:.8f}")

        # ----- soft‑max weights (smooth approximation of the max) -----
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_len)
        w = np.exp(beta * (conv - M))      # w_max = 1, others ≤ 1

        # ----- weighted rev (correlation) -----
        rev = _weighted_rev_fft(p, w, N)

        # ----- gradient of the smoothed max -----
        grad = 4.0 * n * rev               # ∇_p (2·n·M_β) in log‑space

        # ----- step‑size (decaying) -----
        eta = (1.0 / (4.0 * n)) / (1.0 + it * 1e-4)

        # ----- EG update -----
        logp -= eta * grad

        # occasional jitter to escape flat regions
        if it % jitter_interval == 0:
            logp += np.random.randn(n) * jitter_scale

        # ----- project back onto the simplex -----
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)   # avoid overflow
        p = exp_shift / exp_shift.sum()

    return best_p, best_M


# ----------------------------------------------------------------------
def _lp_refine(a_curr: np.ndarray,
               n: int,
               deadline: float,
               max_steps: int = 8,
               t_initial: float = 0.5) -> tuple[np.ndarray, float]:
    """
    LP‑based refinement (the classic LP‑update).
    Starts from `a_curr` (already with sum = √(2n)).
    Returns a refined sequence and its `evaluate_sequence` score.
    """
    S = math.sqrt(2.0 * n)          # required ℓ₁‑norm
    f = a_curr.copy()
    # FFT size for fast convolution checks
    N = _next_pow2(3 * n - 2)

    def _max_conv(vec: np.ndarray) -> float:
        P = np.fft.rfft(vec, n=N)
        conv = np.fft.irfft(P * P, n=N)[: 2 * n - 1]
        return conv.max()

    M = _max_conv(f)
    best_f = f.copy()
    best_M = M
    t = t_initial

    for step in range(max_steps):
        if time.time() >= deadline:
            break

        # ----- build convolution matrix A (size (2n‑1) × n) -----
        # each diagonal i (offset i) contains the constant f[i]
        diag_arrays = [np.full(n, f[i]) for i in range(n)]
        offsets = np.arange(n)
        A = sp.diags(diag_arrays, offsets, shape=(2 * n - 1, n), format="csr")

        # ----- linear programme: maximise Σ g_j subject to f∗g ≤ M -----
        c = -np.ones(n)                                 # maximise sum(g)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        res = opt.linprog(c,
                          A_ub=A,
                          b_ub=b_ub,
                          bounds=bounds,
                          method="highs",
                          options={"presolve": True})

        if not res.success:
            print(f"[LP step {step+1}] linprog failed:", res.message)
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break

        # Scale g to have the correct ℓ₁‑norm (S)
        g_scaled = g * (S / sum_g)

        # ---- convex combination ----
        f_new = (1.0 - t) * f + t * g_scaled
        f_new = np.maximum(f_new, 0.0)               # guard against tiny negatives
        # renormalise exactly (numerical drift)
        f_new = f_new * (S / f_new.sum())

        M_new = _max_conv(f_new)

        if M_new < best_M - 1e-12:
            best_M = M_new
            best_f = f_new.copy()
            f = f_new
            M = M_new
            print(f"[LP n={n}] step {step+1} improved max = {M_new:.8f}")
        else:
            # no improvement – shrink the mixing factor
            t *= 0.5
            if t < 1e-4:
                break

    final_score = evaluate_sequence(best_f.tolist())
    return best_f, final_score

    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # initialise from the provided construction if available
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        base = np.ones(500, dtype=float)                 # fallback uniform

    # turn it into a probability vector (sum = 1)
    p_global = base / base.sum()
    best_p = p_global.copy()
    best_n = len(p_global)
    best_a = best_p * math.sqrt(2.0 * best_n)
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[init] length={best_n} initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # length schedule – limited to ~3000 to keep the LP tractable
    length_schedule = [500, 800, 1200, 1800, 2500, 3000]
    if best_n not in length_schedule:
        length_schedule.append(best_n)
    length_schedule = sorted(set(length_schedule))

    for n in length_schedule:
        if time.time() >= deadline:
            break

        # number of random restarts for this length
        restarts = 2 if n != best_n else 1

        for restart_id in range(restarts):
            if time.time() >= deadline:
                break

            # ----- initialise probability vector for this length -----
            if restart_id == 0:
                # interpolate the current best probability vector
                x_old = np.linspace(0.0, 1.0, best_p.shape[0])
                x_new = np.linspace(0.0, 1.0, n)
                p0 = np.interp(x_new, x_old, best_p)
                p0 = np.maximum(p0, 0.0)
                p0 /= p0.sum()
            else:
                # random Dirichlet‑like start
                p0 = np.random.rand(n)
                p0 = np.maximum(p0, 0.0)
                p0 /= p0.sum()

            # ----- Phase 1: EG optimisation -----
            p_opt, max_conv = _optimize_eg(p_start=p0,
                                           n=n,
                                           deadline=deadline,
                                           max_iters=180_000)

            a_opt = p_opt * math.sqrt(2.0 * n)
            cur_score = evaluate_sequence(a_opt.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_p = p_opt.copy()
                best_n = n
                best_a = a_opt.copy()
                print(f"[update] EG gave new global best: n={n} score={cur_score:.6f}")

            # ----- Phase 2: LP refinement (only for moderate sizes) -----
            if n <= 2500 and time.time() < deadline:
                a_ref, ref_score = _lp_refine(a_curr=a_opt,
                                              n=n,
                                              deadline=deadline,
                                              max_steps=8,
                                              t_initial=0.5)
                if ref_score < best_score:
                    best_score = ref_score
                    best_p = a_ref / math.sqrt(2.0 * n)
                    best_n = n
                    best_a = a_ref.copy()
                    print(f"[update] LP refinement improved global best: n={n} score={ref_score:.6f}")

    # --------------------------------------------------------------
    # fallback – should never be needed
    if best_a is None:
        n = 500
        S = math.sqrt(2.0 * n)
        best_a = np.full(n, S / n)
        best_score = evaluate_sequence(best_a.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_n}")
    return best_a.tolist()
