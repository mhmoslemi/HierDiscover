def adam_optimize(p_start, n, deadline):

   logp = np.log(p_start + 1e-30)

   m = np.zeros_like(logp)

   v = np.zeros_like(logp)

   t = 0

   best_score, best_p = compute_score(p_start), p_start

   while time.time() < deadline and t < max_iters:

        t += 1

        # compute p

        max_log = np.max(logp)

        exp_log = np.exp(logp - max_log)  # stable

        p = exp_log / exp_log.sum()

        # compute convolution and max

        conv = _full_convolution_fft(p, N)

        M = conv.max()

        cur_score = 2.0 * n * M

        if cur_score < best_score:

            best_score = cur_score

            best_p = p.copy()

            print(...)

        # compute smoothing weight w

        beta = beta_start + (beta_end - beta_start) * min(1.0, t / beta_len)

        # w = exp(beta*(conv - M))

        w = np.exp(beta * (conv - M))

        # compute rev

        rev = _weighted_rev_fft(p, w, N)

        # gradient

        grad = 4.0 * n * rev

        # Adam updates

        m = beta1 * m + (1 - beta1) * grad

        v = beta2 * v + (1 - beta2) * (grad**2)

        m_hat = m / (1 - beta1**t)

        v_hat = v / (1 - beta2**t)

        lr_t = lr * (np.sqrt(1 - beta2**t) / (1 - beta1**t))

        step = lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)

        # update logp (descend)

        logp -= step

        # jitter

        if t % jitter_interval == 0:

            logp += rng.normal(scale=jitter_scale, size=n)

   return best_p, best_score

        if it % jitter_interval == 0:
            logp += np.random.randn(n) * jitter_scale
        # early exit if no improvement for many iterations? optional
        # keep loop
    return best_p, best_score

"""
Multi‑scale Adam‑based optimisation of the maximal self‑convolution.
--------------------------------------------------------------------
* Treat the step heights a_i as a scaled probability vector:
      a = √(2·n) · p ,   Σ p_i = 1 ,   p_i ≥ 0.
  The evaluation becomes  evaluate = 2·n·max_k (p∗p)_k.

* Optimise p on the simplex by gradient descent in the log‑domain
  (mirror descent / exponentiated‑gradient).  The gradient is the
  correlation of p with a soft‑max weighting of the convolution
  (smooth approximation of the max).  An Adam‑style momentum update
  gives faster convergence than plain EG.

* A temperature β is increased from a small value to a large one
  (β≈30) so that early iterations see a smooth landscape and later
  iterations focus on the true maximum.

* A multiscale schedule: start with short vectors, optimise, then
  up‑sample the current best to a larger length and continue.  For
  lengths ≤3000 a few random restarts are tried; for larger lengths
  only the deterministic up‑sample is used (to respect the time
  budget).

* The routine respects a 1000‑second wall‑clock budget, performs only
  cheap FFT‑based convolutions, and finally returns the best sequence
  (as a plain Python list) found before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """
    Full self‑convolution of a probability vector p (real FFT).
    Returns the real array of length 2·len(p)−1.
    """
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k · p_{k‑i} for i = 0 … n‑1.
    This is a correlation of p with the weight vector w and is obtained
    via a single convolution in the frequency domain.
    """
    n = p.shape[0]
    L = w.shape[0]                # L = 2·n−1
    # FFT of p (already zero‑padded to size N)
    P = np.fft.rfft(p, n=N)
    # FFT of reversed weight vector
    W_rev = np.fft.rfft(w[::-1], n=N)
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    conv_rev = conv_rev[: n + L - 1]          # keep linear part
    # Extract the desired slice and reverse it
    rev = conv_rev[L - n : L][::-1]
    return rev


# ----------------------------------------------------------------------
def _adam_optimize(p_start: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for a fixed length ``n``.
    Works on log‑probabilities (mirror descent on the simplex) and
    minimises the smoothed maximum of the auto‑convolution.
    Returns the best probability vector (sum = 1) and its unsmoothed score
    ``2·n·max(p∗p)``.
    """
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05                      # base learning rate

    # schedule for the smoothing temperature β
    beta_start = 0.1
    beta_end = 30.0
    beta_len = 100_000              # iterations over which β is ramped

    jitter_interval = 4000
    jitter_scale = 1e-3

    # initialise log‑probabilities
    logp = np.log(p_start + 1e-30)
    m = np.zeros_like(logp)
    v = np.zeros_like(logp)

    # FFT size sufficient for both the self‑convolution (2n‑1) and the
    # correlation (3n‑2)
    N = _next_pow2(3 * n - 2)

    best_score = float("inf")
    best_p = None
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- recover p from log‑domain (stable softmax) -----
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)
        p = exp_shift / exp_shift.sum()

        # ----- compute full self‑convolution and current objective -----
        conv = _full_convolution_fft(p, N)
        M = conv.max()
        cur_score = 2.0 * n * M                 # this is exactly evaluate_sequence(...)

        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[n={n:5d} it={it:6d}] new local best = {cur_score:.6f}")

        # ----- soft‑max weights (smooth approximation of the max) -----
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_len)
        w = np.exp(beta * (conv - M))          # un‑normalised weights, max(w)=1

        # ----- correlation needed for the gradient -----
        rev = _weighted_rev_fft(p, w, N)

        # ----- gradient of the smoothed objective (as in the EG paper) -----
        grad = 4.0 * n * rev                    # ∇_logp (2·n·M_β)

        # ----- Adam update in log‑space -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr_t = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        step = lr_t * m_hat / (np.sqrt(v_hat) + eps)

        logp -= step

        # ----- jitter to shake out of plateaus -----
        if it % jitter_interval == 0:
            logp += np.random.randn(n) * jitter_scale

    # ``best_p`` is guaranteed to be set because the loop runs at least once
    return best_p, best_score

    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n0}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Multiscale length schedule (extendable up to ~12 k)
    length_schedule = [500, 1000, 1500, 2100, 3000, 5000, 8000, 12000]
    length_schedule = [l for l in length_schedule if 300 <= l <= 12000]
    length_schedule = sorted(set(length_schedule))

    # Fixed budget per optimisation run (seconds).  It is deliberately
    # modest so that several lengths and restarts fit into the overall
    # 1000‑second wall‑clock.
    max_run_time = 80.0

    # ------------------------------------------------------------------
    for n in length_schedule:
        if time.time() >= deadline:
            break

        # ----- initialise probability vector for this length -----
        if best_seq.shape[0] == n:
            p_init = best_seq / math.sqrt(2.0 * n)
        else:
            # linear interpolation of the current best probability vector
            p_old = best_seq / math.sqrt(2.0 * best_seq.shape[0])
            old_x = np.linspace(0.0, 1.0, p_old.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            p_init = np.interp(new_x, old_x, p_old)
            p_init = np.maximum(p_init, 0.0)
            s = p_init.sum()
            if s == 0.0:
                p_init[:] = 1.0 / n
            else:
                p_init /= s

        # ----- decide how many restarts to attempt (more for small n) -----
        if n <= 3000:
            restarts = [p_init]

            # a uniform Dirichlet (α=1) random start
            rand1 = np.random.rand(n)
            rand1 = np.maximum(rand1, 1e-12)
            rand1 /= rand1.sum()
            restarts.append(rand1)

            # a heavy‑tailed Dirichlet (α=0.5) start
            gamma = np.random.gamma(shape=0.5, scale=1.0, size=n)
            heav = gamma / gamma.sum()
            restarts.append(heav)
        else:
            # for the largest lengths we keep only the deterministic up‑sample
            restarts = [p_init]

        for run_id, p_start in enumerate(restarts, start=1):
            if time.time() >= deadline:
                break

            # Allocate a dedicated deadline for this run
            run_deadline = min(deadline, time.time() + max_run_time)

            print(f"[run] length={n:5d} restart={run_id}/{len(restarts)}")
            p_opt, _ = _adam_optimize(p_start, n, run_deadline)

            # Scale back to the original height vector
            a_opt = p_opt * math.sqrt(2.0 * n)
            cur_score = evaluate_sequence(a_opt.tolist())

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_opt.copy()
                print(f"[global] new best score={best_score:.6f} (len={n})")

    # ------------------------------------------------------------------
    # Final refinement on the overall best length, if time permits
    if time.time() < deadline - 5.0:
        n_best = best_seq.shape[0]
        p_start = best_seq / math.sqrt(2.0 * n_best)
        print(f"[refine] final polishing on length={n_best}")
        p_opt, _ = _adam_optimize(p_start, n_best, deadline)
        best_seq = p_opt * math.sqrt(2.0 * n_best)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[refine] refined score={best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # Convert to plain Python list for the harness
    return best_seq.tolist()
