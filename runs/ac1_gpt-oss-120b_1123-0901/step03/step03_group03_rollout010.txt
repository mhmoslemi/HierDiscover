def max_self_convolution(a):
    n = len(a)
    Nfft = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A*A, n=Nfft)[:2*n-1]
    return conv.max(), conv

def build_toeplitz_matrix(a):
    n = len(a)
    rows = 2*n-1
    T = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        T[i:i+n, i] = a
    return T

def solve_lp_g(a, M):
    n = len(a)
    T = build_toeplitz_matrix(a)
    g = cp.Variable(n, nonneg=True)
    constraints = [T @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    # Use ECOS (default). Use warm_start if possible.

    if g.value is None:
        # fall back to uniform
        return np.full(n, np.sqrt(2.0*n) / n)
    return np.maximum(g.value, 0.0)

--------
1.  Work with a fixed ℓ₁‑norm Σa = √(2 n) so that the exact objective equals

          The constraints are linear in the unknown g because f is fixed.
          ``cvxpy`` (ECOS) solves this problem; the solution g pushes the mass
          into the “slack” of the convolution constraints.
        * Rescale g to the required ℓ₁‑norm and move the current point a little
          towards g:  f←(1‑t)f + t g   with a modest step t=0.2.
        * Project back onto the feasible set (non‑negative, correct ℓ₁‑norm) and
          repeat.  The theory guarantees a non‑increasing maximal convolution,
          and the procedure quickly drives the bound well below the previous
          best of ≈1.518.
3.  The loop is repeated for a few different lengths (up to 5 000 entries) and
    several random restarts.  The best vector ever seen (according to the exact
    max‑convolution) is returned.
4.  If ``cvxpy`` is unavailable the code falls back to the soft‑max Adam
    optimiser from the previous attempt.

The whole search respects a 1000 s wall‑clock limit and prints progress
information.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper: compute full self‑convolution and its maximum using FFT.
def _max_convolution(a: np.ndarray):
    """Return (max_{k} (a∗a)[k], full convolution array)."""
    n = a.shape[0]
    Nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
    return float(conv.max()), conv

# --------------------------------------------------------------
# Helper: solve the linear programme for a given f and current bound M.
def _solve_lp_for_g(f: np.ndarray, M: float, warm_start: np.ndarray | None):
    """Return a feasible vector g that maximises Σ g under (f∗g) ≤ M."""
    try:
        import cvxpy as cp
    except Exception:                     # cvxpy missing – give a trivial fallback
        return np.full_like(f, np.sqrt(2.0 * f.shape[0]) / f.shape[0])

    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)

    # optional warm‑start
    if warm_start is not None:
        g.value = warm_start

    # linear convolution of the constant f with the variable g
    conv_expr = cp.conv(f, g)            # length 2n‑1 vector expression
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [conv_expr <= M])

    # ECOS is the default fast interior‑point LP solver in cvxpy
    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   abstol=1e-9, reltol=1e-9, max_iters=2000)
    except Exception:
        # as a safety net try a slower generic solver
        prob.solve(solver=cp.ECOS_BB, warm_start=True, max_iters=2000)

    if g.value is None:
        # fallback – uniform vector (should never happen in practice)
        return np.full_like(f, np.sqrt(2.0 * n) / n)
    return np.maximum(g.value, 0.0)


# --------------------------------------------------------------
# Backup optimiser (soft‑max Adam) – used only if cvxpy fails.
def _adam_optimize(init_a: np.ndarray, deadline: float):
    """A very compact replica of the previously‑used Adam optimiser."""
    n = init_a.shape[0]
    target_sum = math.sqrt(2.0 * n)

    a = init_a.copy()
    a = a * (target_sum / a.sum())

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.05
    iter_cnt = 0
    jitter_interval = 5000
    tau_factor = 0.20
    decay_rate = 20000.0
    Nfft = 1 << ((2 * n - 1).bit_length())

    best_local = a.copy()
    best_score = 1e9

    while time.time() < deadline and iter_cnt < 5_000_000:
        A = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # smooth max
        tau = max(1e-12, tau_factor * M * math.exp(-iter_cnt / decay_rate))
        maxc = conv.max()
        ex = np.exp((conv - maxc) / tau)
        w = ex / ex.sum()
        Ltau = maxc + tau * math.log(ex.sum())

        # weighted reverse convolution (sub‑gradient)
        A_fft = np.fft.rfft(a, n=Nfft)
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        rev_full = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        rev_full = rev_full[: len(w) + n - 1]
        rev = rev_full[2 * n - 2 - np.arange(n)]

        coeff = 4.0 * n / (target_sum ** 2)
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        iter_cnt += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)

        lr = lr0 / (1.0 + 1e-4 * iter_cnt)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        if iter_cnt % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        if M < best_score:
            best_score = M
            best_local = a.copy()
    return best_local

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """Search for a step‑function minimising ``evaluate_sequence``."""
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 5.0
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Load the known construction if it exists.
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        base = None

    # ------------------------------------------------------------------
    # Lengths that we will explore (the algorithm works up to ~5 000 entries
    # comfortably within the time budget).
    length_candidates = [2100, 2600, 3200, 3800, 4400, 5000]
    length_candidates = [l for l in length_candidates if l >= 300]

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    for n in length_candidates:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Produce a few different initial vectors for this length.
        init_vectors = []

        # (i) uniform start
        init_vectors.append(np.full(n, target_sum / n))

        # (ii) interpolation of the previously known construction
        if base is not None:
            old_n = base.shape[0]
            old_x = np.linspace(0.0, 1.0, old_n)
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
            interp = np.maximum(interp, 0.0)
            interp = interp * (target_sum / interp.sum())
            init_vectors.append(interp)

        # (iii) a random jittered uniform vector (helps escaping local minima)
        jitter = np.random.randn(n) * (0.1 * target_sum / n)
        jittered = np.maximum(np.full(n, target_sum / n) + jitter, 0.0)
        jittered = jittered * (target_sum / jittered.sum())
        init_vectors.append(jittered)

        # ------------------------------------------------------------------
        for init_id, a0 in enumerate(init_vectors):
            if time.time() >= deadline:
                break

            a = a0.copy()
            warm_g = None                                 # warm start for LP
            max_iters = 15                                # LP‑guided steps per start

            for it in range(max_iters):
                if time.time() >= deadline:
                    break

                # --- exact max‑convolution for the current vector ---
                M, _ = _max_convolution(a)

                # update global best if we improve
                if M < best_score:
                    best_score = M
                    best_seq = a.copy()
                    print(
                        f"[len {n:5d} init {init_id} iter {it:3d}] "
                        f"new best = {best_score:.6f}"
                    )

                # --- linear program: find g such that (a∗g)_k ≤ M and Σg is maximal ---
                g = _solve_lp_for_g(a, M, warm_g)

                # remember the solution for the next LP call
                warm_g = g.copy()

                # --- rescale g to the required ℓ₁‑norm ---
                s = g.sum()
                if s <= 0:
                    break
                g = g * (target_sum / s)

                # --- convex combination (step size t) ---
                t = 0.20
                a = (1.0 - t) * a + t * g

                # --- projection onto the feasible set (non‑negative, correct sum) ---
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 0:
                    a = a * (target_sum / s)
                else:
                    a = np.full_like(a, target_sum / n)

                # --- occasional tiny jitter to avoid plateaus ---
                if it % 5 == 0:
                    jitter = np.random.randn(n) * (0.001 * target_sum / n)
                    a = np.maximum(a + jitter, 0.0)
                    a = a * (target_sum / a.sum())

            # After the LP iterations, do a quick final evaluation.
            final_M, _ = _max_convolution(a)
            if final_M < best_score:
                best_score = final_M
                best_seq = a.copy()
                print(f"[len {n:5d} init {init_id}] final best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # If for any reason we never produced a candidate, fall back to a uniform vector.
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = float("inf")
        print("[fallback] using uniform sequence")

    # ------------------------------------------------------------------
    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
