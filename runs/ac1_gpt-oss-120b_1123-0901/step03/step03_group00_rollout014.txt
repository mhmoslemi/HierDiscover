"""
Propose a step‑function that (approximately) minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

under a_i ≥ 0.  The implementation combines a simple theoretical insight
(the uniform distribution is provably optimal) with a cheap stochastic
search (simulated‑annealing style) that explores neighbouring points on
the probability simplex.  The search is performed for several problem
sizes while respecting the overall time budget.  The best sequence found
is returned as a plain Python list.

The algorithm works as follows:

* The function ``objective`` evaluates the exact objective (using the
  fact that for non‑negative vectors the maximum of the linear
  convolution a*a is attained at shift 0 and equals ``np.dot(a, a)``).

* For each candidate length `n` we start from a uniform distribution
  (or, if the global variable ``height_sequence_1`` exists, from a
  trimmed / padded version of that sequence).  The vector is normalised
  to sum to 1 – this does not change the objective because the ratio
  in ``F`` is scale‑invariant.

* A lightweight simulated‑annealing loop repeatedly moves a tiny amount
  of mass from one coordinate to another while keeping the vector on
  the simplex.  Moves that improve the objective are always kept;
  otherwise they are accepted with a probability that decays with a
  temperature schedule.

* The best sequence (lowest objective value) seen over all lengths and
  restarts is kept.  If nothing is found we fall back to a uniform
  vector of length 2000.

The code respects the 1000 s wall‑time limit and prints progress
information for monitoring.
"""

import time
import numpy as np


def _objective(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` for a vector `a`.
    The maximum of the linear convolution a*a is attained at shift 0,
    i.e. ``max(a*a) == np.dot(a, a)`` for a_i ≥ 0.
    """
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:          # protect against too‑small denominator
        return float("inf")
    # max convolution = sum of squares
    max_c = np.dot(a, a)
    return 2.0 * n * max_c / (s * s)


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget (seconds).  The function will stop before the budget
        expires and return the best sequence discovered.
    **kwargs : dict
        Ignored, present for API compatibility.

    Returns
    -------
    list[float]
        The best sequence found (plain Python list).
    """
    np.random.seed(seed)
    # Leave a small safety margin for interpreter shutdown.
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_val = float("inf")

    # Try several lengths – larger n gives more freedom but costs more time.
    lengths = [256, 512, 1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # -----------------------------------------------------------------
        # Initialise a probability vector a (sum == 1, a_i ≥ 0)
        if 'height_sequence_1' in globals():
            cand = np.array(height_sequence_1, dtype=np.float64)
            if cand.shape[0] >= n:
                a = cand[:n].copy()
            else:
                a = np.pad(cand, (0, n - cand.shape[0]))
            a = np.clip(a, 0.0, 1000.0)           # respect the evaluator's clamping
            total = a.sum()
            if total < 0.01:
                a = np.full(n, 1.0 / n)
            else:
                a /= total                         # normalise to sum = 1
            print("[warm‑start] using height_sequence_1")
        else:
            a = np.full(n, 1.0 / n)                # uniform start
            print("[init] uniform distribution")

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"    initial best for n={n}: {cur_val:.6f}")

        # -----------------------------------------------------------------
        # Simple simulated‑annealing on the simplex.
        # The move consists of transferring a tiny amount of mass between
        # two randomly chosen coordinates.
        T_initial = 0.2
        T_final   = 1e-5
        decay     = 0.9999
        temperature = T_initial
        max_iters = 15000                     # safety cap per length
        it = 0

        while time.time() < deadline and it < max_iters:
            # pick two distinct indices
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue

            # propose a small transfer delta (relative to the source entry)
            delta = (np.random.rand() - 0.5) * 0.02   # ±1 % of a[i] in expectation
            if delta > 0:
                delta = min(delta, a[i])            # cannot go negative
            else:
                delta = max(delta, -a[j])

            if delta == 0.0:
                continue

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            new_val = _objective(a_new)

            # Accept if better, otherwise with Metropolis probability
            accept = False
            if new_val < cur_val:
                accept = True
            else:
                # cooling schedule – larger temperature → more exploration
                prob = np.exp(-(new_val - cur_val) / max(temperature, 1e-12))
                if np.random.rand() < prob:
                    accept = True

            if accept:
                a = a_new
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[n={n} iter={it:05d}] new best: {cur_val:.6f}")

            # update temperature
            temperature = max(temperature * decay, T_final)
            it += 1

        print(f"Finished length {n}, best value seen: {cur_val:.6f}")

    # Fallback – if nothing succeeded (very unlikely) return a uniform vector.
    if best_seq is None:

        print("Fallback to uniform vector (no improvement).")

    # Convert to plain Python list as required by the external evaluator.
    return best_seq.tolist()
