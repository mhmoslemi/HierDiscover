"""
Propose a low‑peak self‑convolution step function.

The algorithm proceeds in three phases:
1. Multi‑scale Exponentiated‑Gradient (EG) optimisation.
   The coefficient vector a is represented as a probability vector p
   (sum p = 1) with a = √(2·n)·p.  A smoothed maximum of the full
   auto‑convolution is minimised by EG on the simplex.  The smoothing
   parameter β is increased from 0.05 to 30 during the run, and occasional
   jitter is added to escape plateaus.

2. Local refinement.
   Using the (sub‑)gradient of the true maximum (obtained by a
   correlation with the active peak indices) we repeatedly move a small
   amount of mass from the coordinate with the largest gradient to the
   one with the smallest gradient.  This simple “push‑mass” step is
   repeated under a time budget and provides a fine‑grained improvement
   over the EG result.

3. Length schedule.
   Starting from the provided construction (or a uniform seed) we
   optimise a series of lengths [500, 1000, 2000, 4000, 8000] by
   interpolating the current best probability vector to the new size.
   Each length receives a time slice; the best sequence over all lengths
   is returned.

All operations are performed with FFT‑based convolutions (O(n log n)),
so the routine easily fits into the 1000 s budget while exploring
sequences up to 8000 entries.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """
    Full auto‑convolution of a probability vector p using FFT.
    Returns the real array of length 2·len(p)‑1.
    """
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k·p_{k‑i} for i=0..n‑1.
    Implements correlation of p with the (reversed) weight vector w.
    """
    n = p.shape[0]
    L = w.shape[0]                 # = 2·n‑1
    P = np.fft.rfft(p, n=N)
    W_rev = np.fft.rfft(w[::-1], n=N)
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    conv_rev = conv_rev[: n + L - 1]          # linear part
    rev = conv_rev[L - n : L][::-1]           # extract and reverse
    return rev


# ----------------------------------------------------------------------
def _eg_optimize(p_start: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimisation for a fixed length ``n``.
    Returns the best probability vector (sum = 1) and its objective value
    2·n·max(p∗p).
    """
    # hyper‑parameters (chosen to be aggressive yet stable)
    beta_start = 0.05
    beta_end = 30.0
    beta_len = 80_000            # iterations over which β is increased
    jitter_interval = 2000
    jitter_scale = 0.005
    max_iters = 500_000          # safety cap – loop will stop earlier by deadline

    p = p_start.copy()
    logp = np.log(p + 1e-30)      # work in log‑space to keep p on the simplex
    N = _next_pow2(3 * n - 2)     # FFT size covering both convolutions

    best_score = float("inf")
    best_p = p.copy()
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- self‑convolution and current objective -----
        conv = _full_convolution_fft(p, N)
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[EG n={n:5d} iter={it:7d}] new best = {cur_score:.6f}")

        # ----- smooth‑max weights -----
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_len)
        w = np.exp(beta * (conv - M))   # w_k ≤ 1, w_max = 1

        # ----- weighted rev (correlation) -----
        rev = _weighted_rev_fft(p, w, N)

        # ----- gradient of the smoothed objective -----
        grad = 4.0 * n * rev

        # ----- EG step (decreasing step‑size) -----
        eta = (1.0 / (4.0 * n)) / (1.0 + it * 1e-4)
        logp -= eta * grad

        # occasional jitter to escape flat regions
        if it % jitter_interval == 0:
            logp += np.random.randn(n) * jitter_scale

        # ----- project back onto the simplex -----
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)     # prevent overflow
        p = exp_shift / exp_shift.sum()

    return best_p, best_score


# ----------------------------------------------------------------------
def _local_refine(p_start: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple gradient‑based local search.
    At each step mass is moved from the index with the largest gradient
    (i.e. most responsible for the current peak) to the index with the
    smallest gradient.  The step size is a small fraction of the donor
    mass and is slowly annealed.
    """
    p = p_start.copy()
    N = _next_pow2(3 * n - 2)

    conv = _full_convolution_fft(p, N)
    best_obj = 2.0 * n * conv.max()
    best_p = p.copy()
    step_frac = 0.02
    it = 0
    tolerance = 1e-12

    while time.time() < deadline:
        it += 1

        conv = _full_convolution_fft(p, N)
        M = conv.max()

        # active (near‑max) indices – use a tiny tolerance to capture ties
        active = (conv >= M - tolerance).astype(float)

        # sub‑gradient of the true max
        rev = _weighted_rev_fft(p, active, N)
        grad = rev                     # scaling constant irrelevant for ordering

        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))
        if i_max == i_min:
            # random perturbation if gradient is flat
            p += np.random.randn(n) * 1e-4
            p = np.clip(p, 0, None)
            if p.sum() == 0:
                p = np.full(n, 1.0 / n)
            else:
                p /= p.sum()
            continue

        delta = min(p[i_max], step_frac * p[i_max])
        if delta <= 0:
            continue

        p_candidate = p.copy()
        p_candidate[i_max] -= delta
        p_candidate[i_min] += delta

        # evaluate candidate
        conv_cand = _full_convolution_fft(p_candidate, N)
        obj_cand = 2.0 * n * conv_cand.max()

        if obj_cand < best_obj - 1e-12:
            best_obj = obj_cand
            best_p = p_candidate.copy()
            p = p_candidate
            # optionally keep a tighter step fraction after improvement
            step_frac = min(0.02, step_frac * 1.01)
            # print improvement occasionally
            if it % 500 == 0:
                print(f"[refine n={n:5d} iter={it:7d}] improved = {best_obj:.6f}")
        else:
            # no improvement – reduce step size
            step_frac *= 0.99
            if step_frac < 1e-5:
                # small random restart around the current best
                p = best_p + np.random.randn(n) * 1e-4
                p = np.clip(p, 0, None)
                p /= p.sum()
                step_frac = 0.02

    return best_p, best_obj
