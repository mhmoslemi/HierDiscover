"""
Propose a step‑function (non‑negative heights) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm combines three ingredients:

1. **Simplex projection** – keep the vector on the simplex
   Σ a_i = √(2 n), a_i ≥ 0.

2. **Linear‑programming update** – given a current vector f,
   solve the LP
       maximise   Σ b_i
       subject to b_i ≥ 0,
                  (f * b)_k ≤ M   for all k,
   where M = max_k (f * f)_k.
   The LP gives a “direction’’ g that can increase the total mass
   without raising the maximal convolution.  After solving we rescale
   g to the required ℓ₁‑norm and replace f ← (1‑t) f + t g.

3. **Optional sub‑gradient polishing** – after the LP‑driven phase
   we run a few steps of the sub‑gradient optimiser from the
   original reference to clean up remaining slack.

The whole process is repeated with different lengths (≈1000–2500)
and random restarts until the allotted budget (≈1000 s) is exhausted.
The best sequence encountered is returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex { x_i ≥ 0 , Σ x_i = s }.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex of radius ``s``."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full convolution via real FFT – returns length 2*n‑1 array.
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Evaluate a vector (with Σa = √(2n) the objective equals max(conv)).
# --------------------------------------------------------------------------- #
def _max_conv(a: np.ndarray) -> float:
    return float(_conv_full_fft(a).max())


# --------------------------------------------------------------------------- #
# Sub‑gradient optimiser (same spirit as the reference implementation).
# Used only for a short polishing phase.
# --------------------------------------------------------------------------- #
def _subgradient_optimize(a_init: np.ndarray,
                          target_sum: float,
                          max_time: float,
                          start_time: float):
    a = a_init.copy()
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _max_conv(a)

    base_lr = 0.2
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    while time.time() < start_time + max_time:
        conv = _conv_full_fft(a)
        cur_score = float(conv.max())

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{time.time():.2f}] GD polishing best = {best_score:.6f} (n={n})")

        # active (max‑tying) indices – tiny epsilon for exact ties
        eps = 1e-12
        active = np.where(conv >= cur_score - eps)[0]

        # gradient of the sum of the active entries
        grad = np.zeros_like(a)
        for k in active:
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]

        max_grad = grad.max()
        if max_grad <= 0.0:
            break

        eta = base_lr * cur_score / max_grad
        momentum = beta_mom * momentum - eta * grad
        a_new = a + momentum

        # enforce feasibility
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional jitter
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            base_lr *= 0.9

        a = a_new
        base_lr *= 0.99995

    return best_vec, best_score

            if sum_b <= 0.0:
                break

            # Rescale to the required ℓ₁‑norm and blend.
            g = b * (target_sum / sum_b)
            f = (1.0 - t) * f + t * g
            f = _project_onto_simplex(f, target_sum)

            # Decay the blending factor.
            t *= 0.9

            # Small random jitter every few iterations to avoid stagnation.
            if np.random.rand() < 0.04:
                noise = np.random.randn(n) * (0.002 * target_sum / n)
                f = _project_onto_simplex(np.maximum(f + noise, 0.0), target_sum)

        # --------------------------------------------------------------- #
        # Polishing phase – a brief sub‑gradient descent.
        # --------------------------------------------------------------- #
        time_left = deadline - time.time()
        if time_left > 3.0 and best_seq is not None:
            # run polishing on the current best vector (or on f if it is better)
            polish_vec = best_seq if best_seq is not None else f
            polish_vec, polish_score = _subgradient_optimize(
                polish_vec, target_sum, max_time=time_left, start_time=time.time())
            if polish_score < best_score:
                best_score = polish_score
                best_seq = polish_vec.copy()
                print(f"[{time.time():.2f}] Polished best = {best_score:.6f} (n={n})")

        # Early exit if we already beat a reasonable target.
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Fallback – uniform vector (should never be needed).
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 1500
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return as a plain Python list.
    return best_seq.tolist()
