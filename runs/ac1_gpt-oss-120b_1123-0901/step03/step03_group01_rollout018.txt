def _conv_fft(a, b):
    n = len(a) + len(b) - 1
    n_fft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    conv = np.fft.irfft(A * B, n=n_fft)[:n]
    return conv

def optimize_sequence(n, time_budget, start_time, init_seq=None):
    target_sum = np.sqrt(2.0 * n)
    if init_seq is None:
        # build initial
        idx = np.arange(n)
        p = np.random.uniform(0.5, 2.5)   # exponent
        weight = 1.0/(np.power(idx+1, p)) + 1.0/(np.power(n-idx, p))
        init = np.random.rand(n) * weight
    else:
        # if init_seq length not n, adjust
        if len(init_seq) >= n:
            init = init_seq[:n].copy()
        else:
            pad = np.random.rand(n - len(init_seq)) * 1e-3
            init = np.concatenate([init_seq, pad])
    a = _project_onto_simplex(init.astype(float), target_sum)

    best_a = a.copy()
    best_score = _max_conv(a)

    # Adam parameters
    lr = 0.5 / np.sqrt(n)  # base LR
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    alpha = 1.0
    alpha_max = 30.0
    alpha_inc = 1.05
    max_iter = 1000000  # large

    while time.time() < start_time + time_budget and t < max_iter:
        t += 1

        # Compute conv of a with itself
        conv = _conv_full_fft(a)  # shape (2n-1)
        cur_score = float(conv.max())

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # log
            print(f"[{time.time():.2f}] n={n} iter={t} new best = {best_score:.6f}")

        # softmax weights
        # For stability use cur_score as max
        exps = np.exp(alpha * (conv - cur_score))
        w = exps / exps.sum()

        # Compute gradient
        w_rev = w[::-1]
        conv_w_a = _conv_fft(w, a)            # length = 3n-2
        conv_a_wrev = _conv_fft(a, w_rev)
        Lw = len(w)
        idx = Lw - 1 - np.arange(n)
        grad = conv_w_a[:n] + conv_a_wrev[idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad

"""
Improved optimizer for the step‑function challenge.

The algorithm searches over several problem sizes.  For each size it
initialises a “U‑shaped’’ vector (large values near the ends, small in
the middle) and then refines it with a smoothed version of the
max‑convolution objective:

    f(a) = max_k (a * a)[k]

Because the evaluation routine normalises the sum of the coefficients
to √(2 n), minimising f(a) is equivalent to minimising the supplied
objective.  The smoothing uses a soft‑max (temperature α) which makes
the objective differentiable; we obtain its gradient analytically and
apply an Adam optimiser with a projection onto the simplex
{a_i ≥ 0, Σ a_i = √(2 n)}.  The temperature α is increased slowly so
that the optimisation first follows a smooth landscape and then
focuses on the true maximum.  Several restarts with different lengths
are performed until the time budget expires.  The best sequence found
is returned as a plain Python list.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex { x_i ≥ 0 , Σ x_i = s }.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex of radius ``s``."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                               # descending sort
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full convolution (real FFT) – returns a length 2*n‑1 array.
# --------------------------------------------------------------------------- #
def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of a and b using real FFT."""
    n = a.shape[0] + b.shape[0] - 1
    n_fft = 1 << ((n - 1).bit_length())                 # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    conv = np.fft.irfft(A * B, n=n_fft)[:n]
    return conv


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution of a (i.e. a * a)."""
    return _conv_fft(a, a)


# --------------------------------------------------------------------------- #
# Evaluate the objective (max of the convolution) – convenient shortcut.
# --------------------------------------------------------------------------- #
def _max_conv(a: np.ndarray) -> float:
    """Return max_k (a*a)[k].  The outer wrapper will multiply by 2 n /
    (sum a)^2, but we keep Σa = √(2n) so the value is exactly the objective.
    """
    return float(_conv_full_fft(a).max())


# --------------------------------------------------------------------------- #
# Gradient of the smoothed max‑convolution (soft‑max temperature α).
# --------------------------------------------------------------------------- #
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute ∇  ( (1/α)·log Σ_k exp(α·conv_k) )
    with respect to a.  The formula reduces to a weighted sum of the
    gradients of the individual convolution entries.
    """
    # soft‑max weights (stable)
    max_c = conv.max()
    w = np.exp(alpha * (conv - max_c))
    w /= w.sum()

    # w_rev is w reversed; needed for the second term of the gradient
    w_rev = w[::-1]

    # convolution w * a  (length 3n‑2)
    conv_w_a = _conv_fft(w, a)

    # convolution a * w_rev  (same length)
    conv_a_wrev = _conv_fft(a, w_rev)

    Lw = w.shape[0]                     # = 2n‑1
    n = a.shape[0]

    # indices for the second term: Lw‑1‑i
    idx = Lw - 1 - np.arange(n)
    grad = conv_w_a[:n] + conv_a_wrev[idx]
    return grad


# --------------------------------------------------------------------------- #
# Adam‑style optimisation of a single problem size.
# --------------------------------------------------------------------------- #
def _optimize_once(n: int,
                   time_budget: float,
                   start_time: float,
                   init_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Optimise a vector of length n for at most ``time_budget`` seconds and
    return the best vector together with its max‑convolution value.
    """
    target_sum = np.sqrt(2.0 * n)       # Σa = √(2 n)  → objective = max(conv)

    # ----- initial vector ----------------------------------------------------
    if init_seq is not None:
        if init_seq.size >= n:
            a = init_seq[:n].copy()
        else:
            pad = np.random.rand(n - init_seq.size) * 1e-3
            a = np.concatenate([init_seq, pad])
    else:
        # U‑shaped seed:   a_i ∝ 1/(i+1)^p + 1/(n‑i)^p   (p sampled)
        p = np.random.uniform(0.7, 2.2)
        idx = np.arange(n, dtype=float)
        weight = 1.0 / np.power(idx + 1.0, p) + 1.0 / np.power(n - idx, p)
        a = np.random.rand(n) * weight

    a = _project_onto_simplex(a.astype(float), target_sum)

    # ----- Adam parameters ---------------------------------------------------
    lr = 0.5 / np.sqrt(n)               # base learning rate
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # temperature for the soft‑max – start small, increase slowly
    alpha = 1.0
    alpha_max = 30.0
    alpha_inc = 1.05

    best_a = a.copy()
    best_score = _max_conv(a)

    # ----- main loop ---------------------------------------------------------
    while time.time() < start_time + time_budget:
        t += 1

        # current convolution and objective
        conv = _conv_full_fft(a)
        cur_score = float(conv.max())

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] n={n} iter={t} new best = {best_score:.6f}")

        # gradient of the smooth approximation
        grad = _softmax_gradient(a, conv, alpha)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # projection onto the feasible simplex
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional random shake to escape flat regions
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0),
                                          target_sum)
            m.fill(0.0)
            v.fill(0.0)

        a = a_new

        # schedule updates
        if t % 200 == 0:
            alpha = min(alpha * alpha_inc, alpha_max)
        lr *= 0.9999                     # gentle decay

        # early exit if we have already crossed the target bound
        if best_score < 1.503:
            break

    return best_a, best_score
