"""
search_improved.py
------------------

A fully self‑contained optimiser for the “step‑function” challenge.

The scoring function is::

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv` is the linear convolution (full, length 2·n‑1).  The
objective is scale‑invariant, therefore we keep the L1‑norm fixed to
`target = sqrt(2·n)` throughout the search.

The optimiser combines three complementary ideas:

1. **Evolutionary population** – a small population (≈12) of candidate
   vectors is evolved by mutation and simple crossover.  Fitness is the
   raw peak of the convolution (`max(conv)`).  The population is kept
   feasible by projecting on the non‑negative simplex after every
   operation.

2. **Annealed mutation strength** – the Gaussian mutation standard‑deviation
   decays slowly with wall‑clock time, allowing large exploratory steps
   early and fine‑grained exploitation later.

3. **Greedy local refinement** – after the evolutionary phase a cheap
   hill‑climber repeatedly transfers a tiny amount of mass from a random
   “source’’ index to a random “target’’ index and keeps the move if it
   improves the peak.

The algorithm evaluates many thousands of candidates inside the 1000 s
budget, tries a few problem sizes (original length and several
multiples) and always returns the best sequence seen.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray) -> float:
    """
    Return the maximal value of the full convolution a*a.
    Uses an FFT with zero‑padding to the next power‑of‑two for speed.
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())        # next power of two ≥ 2·n‑1
    A = np.fft.rfft(a, n=N)
    B = A * A                                   # element‑wise product → FFT of conv
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]    # keep the linear part only
    return float(conv.max())


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project a non‑negative vector onto the affine set {x ≥ 0, Σx = target_sum}.
    The operation is simply a scaling after clipping negatives.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0:
        v *= (target_sum / s)
    else:                                   # degenerate case – fall back to uniform
        v[:] = target_sum / v.size
    return v


# ----------------------------------------------------------------------
def _mutate(parent: np.ndarray,
            sigma: float,
            target_sum: float) -> np.ndarray:
    """
    Produce a child by adding isotropic Gaussian noise (standard‑dev sigma)
    and projecting back onto the feasible simplex.
    """
    child = parent + np.random.randn(parent.size) * sigma
    child = _project_onto_simplex(child, target_sum)
    return child


# ----------------------------------------------------------------------
def _random_simplex_vector(n: int, target_sum: float) -> np.ndarray:
    """
    Draw a random point on the positive simplex using an exponential
    (Dirichlet‑like) distribution and scale to `target_sum`.
    """
    rnd = np.random.exponential(scale=1.0, size=n)
    rnd = rnd * (target_sum / rnd.sum())
    return rnd


# ----------------------------------------------------------------------
def _local_refine(a: np.ndarray,
                  target_sum: float,
                  max_iters: int,
                  deadline: float) -> np.ndarray:
    """
    Very cheap stochastic hill‑climber: repeatedly move a tiny fraction
    of mass from one index to another if the move lowers the peak
    of the convolution.
    """
    best = a.copy()
    best_val = _conv_max_fft(best)

    n = a.shape[0]
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # move a small fraction (2 %) of the source entry
        delta = 0.02 * best[i]
        if delta <= 0.0:
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_onto_simplex(cand, target_sum)

        val = _conv_max_fft(cand)
        if val < best_val:
            best = cand
            best_val = val
    return best


# ----------------------------------------------------------------------
def _evolutionary_optimize(n: int,
                           target_sum: float,
                           deadline: float,
                           base_vec: np.ndarray = None) -> tuple[np.ndarray, float]:
    """
    Evolutionary strategy with a tiny population.
    Returns the best vector found and its raw peak value (max(conv)).
    """
    pop_size = 12
    # initialise population
    population = [_random_simplex_vector(n, target_sum) for _ in range(pop_size)]
    if base_vec is not None and base_vec.shape[0] == n:
        population[0] = base_vec.copy()

    # keep a uniform individual as a safe reference
    population[1] = np.full(n, target_sum / n, dtype=float)

    # mutation strength – start relatively large and cool down
    sigma = 0.12 * target_sum / n          # about 12 % of the average entry
    sigma_final = 0.005 * target_sum / n   # final precision ~0.5 % of average
    total_time = deadline - time.time()
    decay_per_sec = (sigma_final / sigma) ** (1.0 / total_time)

    best_vec = None
    best_val = float('inf')

    generation = 0
    while time.time() < deadline:
        generation += 1

        # 1️⃣ evaluate current population
        fitness = [_conv_max_fft(ind) for ind in population]

        # 2️⃣ keep the elite (best half)
        order = np.argsort(fitness)
        population = [population[i] for i in order]
        fitness = [fitness[i] for i in order]

        # update global best
        if fitness[0] < best_val:
            best_val = fitness[0]
            best_vec = population[0].copy()
            print(f"[gen {generation:5d}] new global best = {best_val:.6f}")

        elite = population[: pop_size // 2]

        # 3️⃣ produce the next generation
        offspring = []
        # mutation of each elite parent
        for parent in elite:
            child = _mutate(parent, sigma, target_sum)
            offspring.append(child)

        # crossover (simple averaging) to fill the rest of the pool
        while len(offspring) < pop_size:
            p1, p2 = np.random.choice(elite, size=2, replace=False)
            child = 0.5 * (p1 + p2)
            child = _project_onto_simplex(child, target_sum)
            offspring.append(child)

        population = offspring

        # 4️⃣ cool the mutation strength
        sigma = max(sigma_final, sigma * decay_per_sec)

        # occasional jitter to shake out of plateaus
        if generation % 200 == 0:
            for idx in range(pop_size):
                jitter = np.random.randn(n) * 0.001 * target_sum / n
                population[idx] = _project_onto_simplex(population[idx] + jitter,
                                                       target_sum)

    return best_vec, best_val
