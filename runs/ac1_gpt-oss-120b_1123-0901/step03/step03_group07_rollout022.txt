"""
Propose a step‐function (non‑negative vector) that minimizes

        F(a) = 2·n·max(a∗a) / (∑a)²

using a multi‑restart Adam optimiser on a smooth‑max surrogate.
The optimiser works on a soft‑max (log‑space) parametrisation so the
probability simplex is satisfied automatically.  Several tricks are
added on top of the reference implementation:

* 4 different lengths (500, 1000, 2000, 3000) are explored – short
  runs are cheap and give good “seeds’’ for the longer runs.
* The first restart with each length can reuse the global variable
  ``height_sequence_1`` (if it exists) as a warm‑start.
* A slower λ‑continuation (λ grows by 1 %) gives a smoother approach
  to the true max‑operator.
* After a period of stagnation a stronger random “shake’’ is applied,
  and the Adam moments are reset.
* The inner optimiser performs up to 12 000 iterations, evaluating
  the exact objective every 15 steps; the best vector found overall
  is returned as a plain Python list.

The function respects the 1000 s time budget: it stops a few
tenths of a second before the deadline and prints progress messages.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length ``size``.
    """
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Soft‑max weights  w_k = exp(lam·conv_k) / Σ_j exp(lam·conv_j).
    """
    # shift for numerical stability
    m = conv.max()
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:                     # safety – should not happen
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray,
                      size_corr: int = None) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate with respect to the probability
    vector ``a``.  Formula: ∇φ = 2·(w ★ a) .
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    # slice that corresponds to indices 0 … n‑1 of the original vector
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


def _objective(a: np.ndarray) -> float:
    """
    Exact evaluation used by ``evaluate_sequence`` – the quantity we try
    to minimise.  ``a`` is assumed to be a non‑negative uint‑vector;
    it is not normalised inside the routine.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    # sum_a is guaranteed > 0 by construction
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000) -> List[float]:
    """
    Return a step‑function (list of non‑negative heights) with a low
    evaluation score.  The method combines several random restarts,
    a soft‑max Adam optimiser and a simple continuation on the smoothing
    parameter λ.  It also tries a few different sequence lengths.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.25  # safety margin

    # lengths that give a good trade‑off between expressiveness and speed
    candidate_lengths = [500, 1000, 2000, 3000]

    best_val = float("inf")
    best_seq = None

    restart_counter = 0
    while time.time() < deadline:
        restart_counter += 1

        # ------------------------------------------------------------------
        # Pick a length (cycle through the list to distribute work)
        n = candidate_lengths[restart_counter % len(candidate_lengths)]

        # FFT sizes – recomputed for each length
        conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        size_corr = 1 << int(np.ceil(np.log2(3 * n)))

        # ------------------------------------------------------------------
        # Initialise the probability vector ``a`` -------------------------------------------------
        if restart_counter == 1 and 'height_sequence_1' in globals():
            # warm start from the global best if it has the right size
            h = np.array(height_sequence_1, dtype=np.float64)
            if h.size == n:
                a = h.copy()
                a = np.clip(a, 0.0, 1000.0)
                a_sum = a.sum()
                if a_sum < 0.01:                # avoid the “too small’’ guard
                    a = np.full(n, 1.0 / n)
                else:
                    a /= a_sum
                u = np.log(a + 1e-30)
                warm = True
            else:
                warm = False
        else:
            warm = False

        if not warm:
            mode = restart_counter % 4
            if mode == 0:
                a = np.random.dirichlet(np.ones(n))
            elif mode == 1:
                r = np.random.uniform(0.2, 0.95)
                powers = np.power(r, np.arange(n, dtype=np.float64))
                a = powers / powers.sum()
            elif mode == 2:
                i = np.arange(1, n + 1, dtype=np.float64)
                a = np.sin(np.pi * i / (n + 1))
                a = np.maximum(a, 0.0)
                a = a / a.sum()
            else:
                # a tiny perturbation of the uniform vector
                a = np.full(n, 1.0 / n) + (np.random.rand(n) - 0.5) * 1e-4
                a = np.clip(a, 0.0, None)
                a = a / a.sum()

            u = np.log(a + 1e-30)          # logits – keep in log‑space

        # ------------------------------------------------------------------
        # Adam bookkeeping
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        base_lr = 0.06                     # a slightly larger starting LR
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        # λ‑continuation
        lam = 0.5
        lam_factor = 1.01                  # slower increase → smoother max

        it = 0
        last_improve = time.time()

        # ------------------------------------------------------------------
        # Main optimisation loop for this restart
        while time.time() < deadline:
            it += 1

            # ----- soft‑max to obtain a probability vector -----
            u_shift = u - np.max(u)                # improve numeric stability
            a = np.exp(u_shift)
            a_sum = a.sum()
            if a_sum == 0.0:                       # should never happen
                a = np.full(n, 1.0 / n)
                a_sum = 1.0
            a /= a_sum

            # ----- convolution and smooth‑max weights -----
            conv = _full_convolution(a, size=conv_fft_size)
            w = _softmax_weights(conv, lam)

            # ----- gradient w.r.t. a (smooth‑max surrogate) -----
            grad_a = _softmax_gradient(a, w, size_corr=size_corr)

            # ----- convert to gradient in log‑space (account for soft‑max Jacobian) -----
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            # ----- Adam update -----
            m = beta1 * m + (1.0 - beta1) * grad_u
            v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # gentle learning‑rate decay
            lr = base_lr * (0.9994 ** it)
            u -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # ----- increase smoothness -----
            lam *= lam_factor

            # ----- exact evaluation every few iterations -----
            if it % 15 == 0:
                val = _objective(a)
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[Restart {restart_counter:03d} | n={n}] "
                          f"iter {it:05d}  new best = {val:.6f}")

            # ----- shake if stagnating -----
            if time.time() - last_improve > 1.2:
                # a slightly larger random perturbation in log‑space
                u += (np.random.rand(n) - 0.5) * 1e-3
                m.fill(0.0)
                v.fill(0.0)
                last_improve = time.time()

            # ----- stop after a reasonable number of iterations -----
            if it >= 12000:
                break

        # status after each restart
        print(f"[Restart {restart_counter:03d}] length {n} finished it={it}, "
              f"best so far = {best_val:.6f}")

        # early exit if we already beat the target (saves time)
        if best_val < 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (guaranteed finite score)
    if best_seq is None:
        best_seq = np.full(3000, 1.0 / 3000)

    print(f"Final best objective = {best_val:.6f} (length {best_seq.shape[0]})")
    # Return a plain Python list – evaluate_sequence expects that.
    return best_seq.tolist()
