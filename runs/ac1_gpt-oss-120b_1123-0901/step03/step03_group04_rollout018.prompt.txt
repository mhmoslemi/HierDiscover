def _conv_max_fft(a: np.ndarray):
    """

    convolution array (real).  Uses an FFT of length equal to the next
    power of two for speed.

    n = a.shape[0]
    # Size for zero‑padding: next power of two >= 2*n-1
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)

    conv = np.fft.irfft(B, n=N)

    return max_val, conv

def _subgradient(a: np.ndarray, active_idxs: np.ndarray):
    """
    Compute a sub‑gradient of the max‑convolution at the current point.
    The gradient component for index i equals the average of a_{k-i} over
    all active shift positions k (with out‑of‑range indices ignored).
