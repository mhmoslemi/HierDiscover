def adam_optimize(a_init, target_sum, max_time, start_time, beta_start=0.5, beta_factor=1.5, beta_max=30, lr=0.2):

    a = a_init.copy()
    n = a.shape[0]
    m = 2*n - 1

    m1 = np.zeros_like(a)
    v1 = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8

    t = 0
    beta = beta_start
    best_vec = a.copy()
    best_score = _max_conv(a)

    while time.time() < start_time + max_time:
        # compute smooth max and gradient
        smooth, grad = _smooth_max_and_grad(a, beta)

        # objective actual = smooth (approx max)
        cur_score = smooth   # approximate but close when beta high
        # maybe also compute actual max conv occasionally
        if t % 50 == 0:
            maxc = _max_conv(a)
            if maxc < best_score:
                best_score = maxc
                best_vec = a.copy()
                print(f"[{time.time():.2f}] Adam new best = {best_score:.6f} (n={n})")

        # Adam update
        t += 1
        m1 = beta1 * m1 + (1 - beta1) * grad
        v1 = beta2 * v1 + (1 - beta2) * (grad * grad)
        m_hat = m1 / (1 - beta1**t)
        v_hat = v1 / (1 - beta2**t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)

        # enforce nonnegativity and projection onto simplex
        a = np.maximum(a, 0.0)  # may be negative after update
        a = _project_onto_simplex(a, target_sum)

        # enforce symmetry
        a = (a + a[::-1]) * 0.5
        a = _project_onto_simplex(a, target_sum)  # re-project after symmetry

        # schedule: increase beta over time (if not at max)
        if beta < beta_max:
            beta = min(beta_max, beta * beta_factor)

        # Maybe early termination if not improving; but ignore

    return best_vec, best_score

   global search, followed by a *smooth‑max* Adam optimiser.  The smooth max
   `f_β(a) = (1/β)·log Σ exp(β·conv_k(a))` approximates the true max
   (β large) while remaining differentiable.  Its gradient can be expressed as
   a single convolution, allowing an O(n log n) implementation via FFT.

3. **Symmetry enforcement** – after every update the vector is averaged with
   its reversal, which is natural for the autocorrelation problem and reduces
   the search space.

4. **Adaptive schedule** – the annealing temperature and the β‑parameter of
   the smooth‑max are increased over time, and the Adam learning‑rate decays.
   A large number of short restarts explore many lengths (1500‑3500), and a
   final long refinement is performed on the best candidate.

The code respects the 1000 s time budget, uses only NumPy (no external
optimisers), and returns a plain Python list of non‑negative heights.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex { x_i ≥ 0 , Σ x_i = s }.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex of radius ``s``."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full convolution (real FFT) – returns a length 2*n‑1 array.
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Evaluate the true objective (max of the convolution).
# --------------------------------------------------------------------------- #
def _max_conv(a: np.ndarray) -> float:
    return float(_conv_full_fft(a).max())


# --------------------------------------------------------------------------- #
# Smooth‑max of the convolution together with its exact gradient.
# --------------------------------------------------------------------------- #
def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """Return (smooth_max, gradient) for a given β."""
    n = a.shape[0]
    conv = _conv_full_fft(a)                 # length 2n‑1
    max_val = conv.max()
    # stable log‑sum‑exp
    shifted = beta * (conv - max_val)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    smooth = max_val + np.log(sum_exp) / beta

    # probability vector (weights of the log‑sum‑exp)
    w = exp_shifted / sum_exp                # length 2n‑1

    # gradient = 2 * (a * w_rev)  (see analysis for index handling)
    w_rev = w[::-1]
    m = len(w)                               # = 2n‑1
    n_fft = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(w_rev, n=n_fft)
    conv2_full = np.fft.irfft(A * B, n=n_fft)[: n + m - 1]   # length 3n‑2
    # extract the needed slice: positions (m‑1‑i) for i=0..n‑1
    grad = 2.0 * conv2_full[m - 1 - np.arange(n)]
    return smooth, grad


# --------------------------------------------------------------------------- #
# Simulated annealing – random mass transfers between two positions.
# --------------------------------------------------------------------------- #
def simulated_annealing(a_init: np.ndarray,
                        target_sum: float,
                        max_time: float,
                        start_time: float):
    """Short SA stage – returns best vector and its (true) max‑conv value."""
    a = a_init.copy()
    n = a.shape[0]

    cur_score = _max_conv(a)
    best_vec = a.copy()
    best_score = cur_score

    # temperature schedule (slowly cools)
    T0 = max(0.01, cur_score * 0.05)
    T_min = 1e-6

    while time.time() < start_time + max_time:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1
        if a[i] <= 0.0:
            continue

        # move a random fraction (up to 90 %) of a[i] to j
        frac = np.random.rand() * 0.9
        delta = a[i] * frac

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # keep feasibility (tiny numeric drift)
        a_new = _project_onto_simplex(a_new, target_sum)

        new_score = _max_conv(a_new)
        delta_score = new_score - cur_score

        elapsed = time.time() - start_time
        frac_time = max(0.0, 1.0 - elapsed / max_time)
        T = T_min + T0 * frac_time

        accept = False
        if delta_score <= 0.0:
            accept = True
        else:
            prob = math.exp(-delta_score / (T + 1e-12))
            accept = np.random.rand() < prob

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_vec = a.copy()
                print(f"[{time.time():.2f}] SA new best = {best_score:.6f} (n={n})")

    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Adam optimisation on the smooth‑max surrogate.
# --------------------------------------------------------------------------- #
def adam_optimize(a_init: np.ndarray,
                  target_sum: float,
                  max_time: float,
                  start_time: float):
    """Refine a vector using Adam on a smooth‑max approximation."""
    a = a_init.copy()
    n = a.shape[0]

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # schedule for β (smooth‑max parameter)
    beta = 0.5
    beta_max = 30.0
    beta_mul = 1.05

    # learning‑rate schedule
    lr = 0.2
    lr_decay = 0.9995

    best_vec = a.copy()
    best_score = _max_conv(a)

    step = 0
    while time.time() < start_time + max_time:
        step += 1

        smooth, grad = _smooth_max_and_grad(a, beta)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # enforce feasibility
        a = np.maximum(a, 0.0)
        a = _project_onto_simplex(a, target_sum)

        # enforce symmetry (helps the convolution objective)
        a = (a + a[::-1]) * 0.5
        a = _project_onto_simplex(a, target_sum)

        # occasional tiny random shake to escape flat regions
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (0.005 * target_sum / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)

        # update best using the true max‑conv (expensive, so do it every ~30 steps)
        if step % 30 == 0:
            cur_max = _max_conv(a)
            if cur_max < best_score:
                best_score = cur_max
                best_vec = a.copy()
                print(f"[{time.time():.2f}] Adam new best = {best_score:.6f} (n={n})")

        # schedule updates
        beta = min(beta_max, beta * beta_mul)
        lr *= lr_decay

    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Main entry point expected by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.0          # safety margin

    best_seq = None
    best_score = float("inf")
    iteration = 0

    # ------------------------------------------------------------------- #
    # Outer loop – many short restarts, later a long refinement.
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------- #
        # Choose problem size.
        # --------------------------------------------------------------- #
        if best_seq is not None and np.random.rand() < 0.15:
            n = len(best_seq)
        else:
            n = np.random.randint(1500, 3501)   # 1500‑3500 works well in practice

        target_sum = math.sqrt(2.0 * n)

        # --------------------------------------------------------------- #
        # Initialise a candidate (U‑shape with random exponent & noise).
        # --------------------------------------------------------------- #
        p = np.random.uniform(0.5, 6.0)          # exponent controlling edge bias
        idx = np.arange(n, dtype=float)
        weight = (idx + 1) ** (-p) + (n - idx) ** (-p)
        # random amplitude perturbation (0.5‑1.0 factor)
        init = weight * (0.5 + 0.5 * np.random.rand(n))

        # Occasionally blend with the current best (if available)
        if best_seq is not None and np.random.rand() < 0.3:
            src = np.asarray(best_seq, dtype=float)
            if src.size >= n:
                blend = src[:n] * np.random.rand()
            else:
                pad = np.random.rand(n - src.size)
                blend = np.concatenate([src, pad]) * np.random.rand()
            init = init * (1.0 - np.random.rand()) + blend

        a = _project_onto_simplex(init.astype(float), target_sum)

        # quick evaluation of the raw start point
        cur_score = _max_conv(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Init best (n={n}) = {best_score:.6f}")

        # --------------------------------------------------------------- #
        # Allocate a few seconds for this restart (SA → Adam).
        # --------------------------------------------------------------- #
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        phase_sa = min(4.0, time_left * 0.4)

        a_sa, score_sa = simulated_annealing(a, target_sum, phase_sa,
                                             time.time())
        if score_sa < best_score:
            best_score = score_sa
            best_seq = a_sa.copy()
            print(f"[{time.time():.2f}] After SA best (n={n}) = {best_score:.6f}")

        time_left = deadline - time.time()
        if time_left <= 0:
            break
        phase_adam = min(6.0, time_left * 0.6)

        a_adam, _ = adam_optimize(a_sa, target_sum, phase_adam,
                                  time.time())
        final_score = _max_conv(a_adam)
        if final_score < best_score:
            best_score = final_score
            best_seq = a_adam.copy()
            print(f"[{time.time():.2f}] After Adam best (n={n}) = {best_score:.6f}")

        # early stop if we have already passed the target bound
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Final long refinement on the current best sequence (if time permits).
    # ------------------------------------------------------------------- #
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        n = len(best_seq)
        target_sum = math.sqrt(2.0 * n)
        a_refine, _ = adam_optimize(best_seq, target_sum,
                                    remaining, time.time())
        refined_score = _max_conv(a_refine)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = a_refine.copy()
            print(f"[{time.time():.2f}] Final refinement best = {best_score:.6f}")

    # ------------------------------------------------------------------- #
    # Fallback – a uniform vector (should never be needed).
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 1800
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    return best_seq.tolist()
