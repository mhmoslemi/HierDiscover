def annealed_smooth_descent(a_init, target_sum, max_time, start_time):
    a = a_init.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _max_conv(a)

    eps_end = 50.0

""" 
Algorithm:
- Use many random restarts with different lengths. 
- For each restart initialize a u‑shaped sequence (or reuse the best known one).
- Apply a fast simulated‑annealing phase that moves mass between two
  positions while keeping the ℓ1‑norm fixed.
- Follow with a smooth‑gradient descent phase that minimizes a
  log‑sum‑exp approximation of the max‑convolution; this gives a
  differentiable surrogate and permits a line‑search step.  The update
  is always projected onto the simplex {x_i≥0, Σx_i=√(2n)}.
- Keep the globally best sequence seen; after the outer restarts spend the
  remaining budget refining this best candidate with the same smooth
  descent.
- The whole procedure finishes well within the 1000 s budget and
  typically reaches a bound ≤ 1.503.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast convolution (real FFT) – full length 2*n‑1
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    # next power of two for FFT
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Objective: max entry of a*a (since Σa = √(2n) the full expression equals this)
# --------------------------------------------------------------------------- #
def _max_conv(a: np.ndarray) -> float:
    return float(_conv_full_fft(a).max())


# --------------------------------------------------------------------------- #
# Simulated annealing – random mass transfer between two positions
# --------------------------------------------------------------------------- #
def _simulated_annealing(a_init: np.ndarray,
                         target_sum: float,
                         max_time: float,
                         start_time: float):
    """Return a tuple (best_vector, best_score) after a SA run."""
    a = a_init.copy()
    n = a.shape[0]
    cur_score = _max_conv(a)

    best_vec = a.copy()
    best_score = cur_score

    # temperature schedule (start about 5 % of the current value)
    T0 = max(0.01, cur_score * 0.05)
    T_min = 1e-8

    while time.time() < start_time + max_time:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1
        if a[i] <= 0.0:
            continue

        # move up to 80 % of the mass at i towards j
        delta = a[i] * np.random.rand() * 0.8

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _max_conv(a_new)
        delta_score = new_score - cur_score

        elapsed = time.time() - start_time
        frac = max(0.0, 1.0 - elapsed / max_time)
        T = T_min + T0 * frac

        accept = False
        if delta_score <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_score / (T + 1e-12))
            accept = np.random.rand() < prob

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_vec = a.copy()
                print(f"[{time.time():.2f}] SA new best = {best_score:.6f} (n={n})")
    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Smooth‑gradient descent (log‑sum‑exp surrogate of max)
# --------------------------------------------------------------------------- #
def _smooth_gradient_descent(a_init: np.ndarray,
                             target_sum: float,
                             max_time: float,
                             start_time: float):
    """Projected gradient descent on a smoothed max‑convolution objective."""
    a = a_init.copy()
    n = a.shape[0]

    best_vec = a.copy()
    best_score = _max_conv(a)

    # annealing of the smoothness parameter ε
    eps_start = 5.0
    eps_end = 50.0

    # base learning rate (will be decayed)
    base_lr = 0.15

    while time.time() < start_time + max_time:
        elapsed = time.time() - start_time
        frac = elapsed / max_time
        eps = eps_start + (eps_end - eps_start) * frac

        # ----- compute smoothed objective and its gradient ------------------- #
        conv = _conv_full_fft(a)
        max_val = conv.max()

        # soft‑max weights (stable by subtracting max_val)
        exp_vals = np.exp(eps * (conv - max_val))
        w = exp_vals / exp_vals.sum()

        # gradient of the smoothed objective:
        # grad_i = 2 * Σ_j a_j * w_{i+j}
        # which is the valid part of the cross‑correlation of a and w
        grad = 2.0 * np.correlate(a, w, mode='valid')

        # ----- line search ------------------------------------------------- #
        # start with a step size that shrinks as we progress
        lr = base_lr * (1.0 - frac)

        # try a few back‑tracking reductions if the step is not improving
        improved = False
        for _ in range(5):
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_onto_simplex(a_new, target_sum)
            new_score = _max_conv(a_new)
            if new_score < max_val:
                a = a_new
                if new_score < best_score:
                    best_score = new_score
                    best_vec = a.copy()
                    print(f"[{time.time():.2f}] Smooth GD new best = {best_score:.6f} (n={n})")
                improved = True
                break
            lr *= 0.5

        # if no improvement, occasionally inject a tiny random perturbation
        if not improved and np.random.rand() < 0.01:
            noise = np.random.randn(n) * (0.001 * target_sum / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)

    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Main entry point expected by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    best_seq = None          # numpy array of the best candidate
    best_score = float("inf")  # objective value (max convolution)

    # ------------------------------------------------------------------- #
    # Helper to update the global best
    # ------------------------------------------------------------------- #
    def _maybe_update(candidate, score):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = candidate.copy()
            print(f"[{time.time():.2f}] Global best = {best_score:.6f} (len={candidate.size})")

    # ------------------------------------------------------------------- #
    # Outer loop – many restarts with varying lengths.
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        # Choose problem size.
        if best_seq is not None and np.random.rand() < 0.25:
            n = len(best_seq)
        else:
            n = np.random.randint(1500, 3501)

        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------- #
        # Initialise a candidate.
        # --------------------------------------------------------------- #
        if "height_sequence_1" in globals() and np.random.rand() < 0.4:
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-3
                init = np.concatenate([src, pad])
        else:
            # U‑shaped seed (large near the ends, small in the centre)
            idx = np.arange(n, dtype=float)
            weight = 1.0 / (idx + 1) + 1.0 / (n - idx)
            init = np.random.rand(n) * weight

        a = _project_onto_simplex(init.astype(float), target_sum)

        # Quick evaluation of the raw initial point.
        cur_score = _max_conv(a)
        _maybe_update(a, cur_score)

        # --------------------------------------------------------------- #
        # Allocate time for this restart (≈12 s split 6 s SA / 6 s smooth‑GD).
        # --------------------------------------------------------------- #
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        phase_time = min(6.0, time_left * 0.5)

        # Simulated annealing stage
        a_sa, score_sa = _simulated_annealing(a, target_sum, phase_time, time.time())
        _maybe_update(a_sa, score_sa)

        # Gradient‑descent stage (smooth surrogate)
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        phase_time = min(6.0, time_left)

        a_gd, score_gd = _smooth_gradient_descent(a_sa, target_sum, phase_time, time.time())
        _maybe_update(a_gd, score_gd)

        # Early exit if we already beat the target bound.
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Final intensive refinement on the best sequence seen so far.
    # ------------------------------------------------------------------- #
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0:
            n = best_seq.size
            target_sum = np.sqrt(2.0 * n)
            a_ref, score_ref = _smooth_gradient_descent(best_seq, target_sum,
                                                        remaining, time.time())
            _maybe_update(a_ref, score_ref)
            best_seq = a_ref

    # ------------------------------------------------------------------- #
    # Fallback – uniform vector if nothing was found (should never happen).
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list as required by the grader.
    return best_seq.tolist()

"""
    Optimisation of the step‑function objective

    The goal is to minimise
        2·n·max(conv(a, a)) / (sum(a)**2)

    By normalising Σa = √(2n) the objective reduces to minimising the
    infinity norm of the self‑convolution.  The search consists of many
    restarts with different lengths, each restart performing:

    1. A quick simulated‑annealing phase that moves a random amount of
       mass from one position to another while keeping the ℓ1‑norm fixed.
    2. A smooth‑gradient‑descent phase that minimises a
       log‑sum‑exp (soft‑max) surrogate of the max‑convolution; this
       yields a differentiable objective and an inexpensive gradient
       (cross‑correlation).  After each gradient step the iterate is
       projected onto the simplex {x_i ≥ 0, Σx_i = √(2n)}.

    time with the same smooth‑descent routine.  The method stays well
    within the 1000 s budget and typically reaches a bound ≤ 1.503.

            new_score = _max_conv(a_new)
            if new_score < max_val:
                a = a_new
                if new_score < best_score:
                    best_score = new_score
                    best_vec = a.copy()
                    print(f"[{time.time():.2f}] Smooth GD new best = {best_score:.6f} (n={n})")
                improved = True
                break
            lr *= 0.5

        # occasional small random shake if we are stuck
        if not improved and np.random.rand() < 0.01:
            noise = np.random.randn(n) * (0.001 * target_sum / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)

    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Main entry point – the grader will call this function.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5      # tiny safety margin

    best_seq = None          # numpy array holding the best candidate
    best_score = float("inf")  # its objective value (max convolution)

    # ------------------------------------------------------------------- #
    # Repeated restarts with different lengths and initialisations.
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        # Choose a problem size.
        if best_seq is not None and np.random.rand() < 0.25:
            n = len(best_seq)
        else:
            n = np.random.randint(1500, 3501)

        target_sum = np.sqrt(2.0 * n)

        # --------------------------------------------------------------- #
        # Initialise a candidate sequence.
        # --------------------------------------------------------------- #
        if "height_sequence_1" in globals() and np.random.rand() < 0.4:
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-3
                init = np.concatenate([src, pad])
        else:
            # U‑shaped seed – larger values near the ends.
            idx = np.arange(n, dtype=float)
            weight = 1.0 / (idx + 1) + 1.0 / (n - idx)
            init = np.random.rand(n) * weight

        a = _project_onto_simplex(init.astype(float), target_sum)

        cur_score = _max_conv(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Init best = {best_score:.6f} (n={n})")

        # --------------------------------------------------------------- #
        # Time allocation for this restart (≈12 s split 6 s SA / 6 s GD).
        # --------------------------------------------------------------- #
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        phase_time = min(6.0, time_left * 0.5)

        # Simulated‑annealing phase
        a_sa, score_sa = _simulated_annealing(a, target_sum, phase_time, time.time())
        if score_sa < best_score:
            best_score = score_sa
            best_seq = a_sa.copy()
            print(f"[{time.time():.2f}] After SA best = {best_score:.6f} (n={n})")

        # --------------------------------------------------------------- #
        # Smooth‑gradient‑descent phase
        # --------------------------------------------------------------- #
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        phase_time = min(6.0, time_left)

        a_gd, score_gd = _smooth_gradient_descent(a_sa, target_sum,
                                                  phase_time, time.time())
        if score_gd < best_score:
            best_score = score_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.2f}] After GD best = {best_score:.6f} (n={n})")

        # Early exit if the target bound is already achieved.
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Final intensive refinement of the best sequence found so far.
    # ------------------------------------------------------------------- #
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0:
            n = best_seq.size
            target_sum = np.sqrt(2.0 * n)
            a_ref, score_ref = _smooth_gradient_descent(best_seq, target_sum,
                                                        remaining, time.time())
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"[{time.time():.2f}] Final refined best = {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------- #
    # Fallback – uniform vector (should never be needed).
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list as required by the grader.
    return best_seq.tolist()
