"""
Hybrid optimizer for the step‑function bound.

The algorithm works in two phases.

Phase 1 – Linear‑programming refinement
    * Given a current non‑negative vector f (∑f=1) compute M = max(f∗f).
    * Solve the LP   maximise   1ᵀb   s.t.   b≥0  and   f∗b ≤ M
      (the convolution f∗b is linear in the unknown b).
    * Mix the solution b back into f with a relaxation factor t∈(0,1).
      This guarantees that the supremum of the self‑convolution never
      increases.  Repeating the step quickly drives the objective down.

Phase 2 – Stochastic hill‑climbing
    * Starting from the best vector obtained in phase 1, repeatedly move a
      small amount of mass from one coordinate to another.
    * The new vector is accepted only if the exact objective
      F(a)=2·n·max(a∗a)/(∑a)² improves.
    * A simple annealing schedule (decrease of the step size when no
      improvement is seen) allows us to escape shallow plateaus.

Both phases respect the time budget (default 1000 s) and keep all
coefficients non‑negative and normalised to sum = 1, which leaves the
objective scale‑invariant.  The final result is returned as a plain
Python list that can be fed directly to ``evaluate_sequence``.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must stay at top level – no closures)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Exact objective  F(a)=2·n·max(a*a)/(∑a)² .  Assumes a≥0."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    # sum_a is guaranteed >0 because we normalise to 1
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _conv_matrix(f: np.ndarray):
    """
    Sparse matrix C such that C @ b == f ∗ b (linear convolution).
    Shape is (2·n‑1, n) where n = len(f).
    """
    from scipy.sparse import diags
    n = f.shape[0]
    # each diagonal offset d (0 … n‑1) contains the constant f[d]
    # build a (n, n) array where each row d is f[d] repeated n times
    data = np.tile(f, (n, 1))
    offsets = np.arange(n)                     # 0 … n‑1
    C = diags(data, offsets, shape=(2 * n - 1, n), format='csc')
    return C
