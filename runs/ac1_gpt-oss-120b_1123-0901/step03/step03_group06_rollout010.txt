"""
Improved search for a low‑peak self‑convolution step function.

Algorithm
---------
* The vector a is represented as a probability vector p with Σp = 1 and a = S·p,
  where S = sqrt(2·n) guarantees that the evaluation function equals max(a*a).
* For a fixed length n we minimise  max(p∗p)  by Adam optimisation on the
  log‑probabilities (mirror descent on the simplex).  The gradient of the
  smoothed maximum is obtained with a soft‑max weighting of the convolution
  entries; the required correlation is computed with a single FFT pair.
* A multi‑scale schedule is used: start with a modest length, optimise,
  then up‑sample the current best probability vector to a larger length.
* For each length a few random restarts (interpolated best + Dirichlet draws)
  are performed; the best local solution becomes the seed for the next scale.
* Time is split evenly over the remaining lengths so the whole search respects
  the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """Full self‑convolution p∗p (real output of length 2·len(p)‑1)."""
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k·p_{k‑i}  for i=0..n‑1.
    This is a correlation of p with the weight vector w.
    """
    n = p.shape[0]                 # length of p
    L = w.shape[0]                 # = 2·n‑1
    P = np.fft.rfft(p, n=N)
    # convolution of p with reversed w gives the required correlation
    W_rev = np.fft.rfft(w[::-1], n=N)
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    conv_rev = conv_rev[: n + L - 1]            # keep linear part
    rev = conv_rev[L - n : L][::-1]            # extract the n entries
    return rev


# ----------------------------------------------------------------------
def _interpolate_prob(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Interpolate an old probability vector (derived from a step sequence)
    onto a new length.
    """
    old_n = old_seq.shape[0]
    # convert to a probability vector (Σp = 1)
    p_old = old_seq / math.sqrt(2.0 * old_n)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    p_new = np.interp(new_x, old_x, p_old)
    p_new = np.maximum(p_new, 0.0)
    p_new /= p_new.sum()
    return p_new


# ----------------------------------------------------------------------
def _optimize_adam(p_start: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the simplex (log‑probabilities) for a fixed length.
    Returns the best probability vector found and its corresponding max‑conv
    value (which equals the evaluate_sequence score because Σa = √(2n)).
    """
    # ----- initialise -----
    p = p_start.copy()
    p = p / p.sum()
    logp = np.log(p + 1e-30)                     # work in log‑space

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    pow_beta1, pow_beta2 = 1.0, 1.0
    alpha = 0.5                                   # base step size

    # soft‑max schedule for the smoothed peak
    beta_start, beta_end = 0.1, 80.0
    beta_len = 50000                               # linear increase over this many iters

    # FFT size that works for both convolution (2n‑1) and correlation (3n‑2)
    N = _next_pow2(3 * n - 2)

    best_score = float("inf")
    best_p = p.copy()
    it = 0
    jitter_interval = 4000
    jitter_scale = 0.001

    while time.time() < deadline and it < 200_000:
        it += 1

        # ----- convolution and current objective -----
        conv = _full_convolution_fft(p, N)
        M = conv.max()                     # max(p∗p)

        if M < best_score:
            best_score = M
            best_p = p.copy()
            if it % 1000 == 0:
                print(f"[len {n:5d} iter {it:7d}] new local best = {best_score:.6f}")

        # ----- soft‑max weights (normalized) -----
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_len)
        w = np.exp(beta * (conv - M))
        w_sum = w.sum()
        if w_sum == 0.0:            # numerically impossible, but guard anyway
            w_norm = w
        else:
            w_norm = w / w_sum

        # ----- gradient of the smoothed maximum (via correlation) -----
        rev = _weighted_rev_fft(p, w_norm, N)     # Σ_k w_k p_{k-i}
        grad = 4.0 * n * rev                       # ∇_p (2·n·smooth_max)

        # ----- Adam update in log‑space -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        pow_beta1 *= beta1
        pow_beta2 *= beta2
        m_hat = m / (1.0 - pow_beta1)
        v_hat = v / (1.0 - pow_beta2)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        logp -= step

        # occasional jitter to escape plateaus
        if it % jitter_interval == 0:
            logp += np.random.randn(n) * jitter_scale

        # ----- project back onto the probability simplex -----
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)   # prevent overflow
        p = exp_shift / exp_shift.sum()

    return best_p, best_score

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the evaluation harness
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Load a previously discovered construction if it exists; otherwise start
    # from a simple uniform vector of length 500.
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        base = np.ones(500, dtype=float)

    n0 = base.shape[0]
    S0 = math.sqrt(2.0 * n0)
    base = base * (S0 / base.sum())          # normalise Σa = √(2n)

    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n0}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Multi‑scale schedule (lengths grow geometrically, capped at 10000)
    length_schedule = [500, 800, 1300, 1900, 2600, 3500, 4600, 6000, 8000, 10000]
    length_schedule = [l for l in length_schedule if l <= 12000]
    length_schedule = sorted(set(length_schedule))

    for idx, n in enumerate(length_schedule):
        if time.time() >= deadline:
            break

        # Allocate a fair share of the remaining time to this length
        lengths_left = len(length_schedule) - idx
        remaining_time = deadline - time.time()
        if lengths_left <= 0:
            break
        segment_deadline = time.time() + 0.9 * remaining_time / lengths_left

        # ----- initialise start points -----
        starts = []

        # 1) up‑sample the current global best (probability view)
        p_interp = _interpolate_prob(best_seq, n)
        starts.append(p_interp)

        # 2) a few random Dirichlet initialisations (sparse or dense)
        for _ in range(2):
            alpha = 0.5                     # smaller → sparser starts
            rnd = np.random.gamma(alpha, 1.0, size=n)
            rnd = np.maximum(rnd, 0.0)
            rnd /= rnd.sum()
            starts.append(rnd)

        # ----------------------------------------------------------
        # Run Adam optimisation for each start vector
        for sidx, p_start in enumerate(starts):
            if time.time() >= segment_deadline:
                break
            print(f"[len {n}] start {sidx + 1}/{len(starts)}")
            p_opt, local_score = _optimize_adam(p_start, n, segment_deadline)

            # Convert back to step heights (a = S·p, S = √(2·n))
            S = math.sqrt(2.0 * n)
            a_opt = p_opt * S
            cur_score = evaluate_sequence(a_opt.tolist())

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_opt.copy()
                print(f"[len {n}] new global best = {best_score:.6f}")

        # ----------------------------------------------------------
        # Optional quick fine‑tuning of the current global best at this length
        if time.time() < segment_deadline:
            p_cur = best_seq / math.sqrt(2.0 * n)
            p_fine, _ = _optimize_adam(p_cur, n, segment_deadline)
            a_fine = p_fine * math.sqrt(2.0 * n)
            fine_score = evaluate_sequence(a_fine.tolist())
            if fine_score < best_score:
                best_score = fine_score
                best_seq = a_fine.copy()
                print(f"[len {n}] fine‑tuned best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback (should never be taken)
    if best_seq is None or best_seq.size == 0:
        n = 500
        S = math.sqrt(2.0 * n)
        best_seq = np.full(n, S / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # Convert to plain list for the caller
    return best_seq.tolist()
