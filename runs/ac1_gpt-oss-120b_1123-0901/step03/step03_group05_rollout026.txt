def simulated_annealing(a, target_sum, best_score, deadline, max_moves=200000):
    cur = a.copy()
    cur_score = conv_max_fft(cur)[0]
    best_local_seq = cur.copy()
    best_local_score = cur_score
    n = len(cur)
    temp = cur_score * 0.1  # initial temperature 10% of current score
    decay = 0.99995  # multiply each move

    for move in range(max_moves):
        if time.time() >= deadline:
            break
        # propose move
        i = np.random.randint(n)
        if cur[i] <= 1e-15:
            continue
        j = np.random.randint(n)
        if j == i:
            continue
        # move epsilon
        eps = cur[i] * np.random.rand() * 0.1  # up to 10%
        if eps <= 0:
            continue
        new_vec = cur.copy()
        new_vec[i] -= eps
        new_vec[j] += eps
        # compute score
        new_score = conv_max_fft(new_vec)[0]
        delta = new_score - cur_score
        if delta < 0 or np.random.rand() < math.exp(-delta / temp):
            # accept
            cur = new_vec
            cur_score = new_score
            if new_score < best_local_score:
                best_local_score = new_score
                best_local_seq = new_vec.copy()
                # maybe update global best inside caller when returned
        # cooling
        temp *= decay
        # optionally reheat if no improvement
        if move % 10000 == 0:
            # maybe increase temperature slightly based on progress
            temp = max(temp, cur_score * 0.01)
    return best_local_seq, best_local_score

def _conv_max_fft(a):
    n = a.shape[0]
    N = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A*A, n=N)[:2*n-1]
    return float(conv.max())

def _build_piecewise(p, n):
    k = len(p)
    seg = n // k
    rem = n % k
    parts = []
    for i in range(k):
        length = seg + (1 if i < rem else 0)
        parts.append(np.full(length, p[i], dtype=float))
    return np.concatenate(parts)

def _objective_piecewise(p, n, target_sum):
    # p must be non‑negative; DE keeps it in bounds
    a = _build_piecewise(p, n)
    s = a.sum()
    if s <= 0:
        return 1e9
    a = a * (target_sum / s)
    # Avoid overflow/NaNs – if any element > 1000 the evaluate_sequence clamps,
    # but that would break our gradient; penalise such solutions.
    if np.any(a > 1000):
        return 1e9
    return _conv_max_fft(a)

            break
        i = np.random.randint(n)
        if a[i] <= 1e-14:

                print(f"[SA] improvement to {best_loc:.6f}")

        if move % 5000 == 0 and temp < 1e-6:

"""
Search for a non‑negative step function that minimises the maximal value of its
self‑convolution (the value returned by evaluate_sequence).  The algorithm
works on several problem sizes and uses three complementary phases:

1.  **Low‑dimensional global search** – the vector is approximated by a
    piecewise‑constant function with a small number of segments.  Differential
    evolution (SciPy) explores the segment heights without any gradient
    information.

2.  **Adam optimisation** – the best candidates from the global phase are
    refined with a projected Adam optimiser that follows the sub‑gradient of
    the max‑convolution (identical to the approach used in the reference code
    but with a few tuned hyper‑parameters).

3.  **Simulated‑annealing local search** – a cheap hill‑climb that moves a
    small amount of mass between two positions.  Worse moves are occasionally
    accepted with a temperature schedule, allowing the method to escape shallow
    plateaus.

All phases respect the constraints (non‑negative entries, fixed L1‑norm).
Multiple lengths (up to 8000) are tried; the best sequence found before the
time budget expires is returned as a plain Python list.
"""

import time
import math
import numpy as np
from scipy.optimize import differential_evolution


# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray) -> float:
    """Full self‑convolution via FFT and return its maximal value."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return float(conv.max())


def _conv_max_fft_full(a: np.ndarray):
    """
    Return (max value, index of the max, the whole convolution array).
    Used by the gradient based optimiser.
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    max_val = float(conv.max())
    idx = int(conv.argmax())
    return max_val, idx, conv


def _rev_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    """Reverse‑shifted copy needed for the sub‑gradient of max‑convolution."""
    n = a.shape[0]
    rev = np.zeros_like(a)
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    rev[mask] = a[j[mask]]
    return rev


def _subgradient_average(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Average of the reversed vectors for all active shift indices."""
    n = a.shape[0]
    rev = np.zeros_like(a)
    idx = np.arange(n)
    for k in active:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    if active.size:
        rev /= float(active.size)
    return rev


# ----------------------------------------------------------------------
def _build_piecewise(p: np.ndarray, n: int) -> np.ndarray:
    """
    Expand a low‑dimensional parameter vector `p` (segment heights) into a full
    length‑`n` step function by repeating each height over an almost equal
    number of indices.
    """
    k = len(p)
    seg = n // k
    rem = n % k
    parts = []
    for i in range(k):
        length = seg + (1 if i < rem else 0)
        parts.append(np.full(length, p[i], dtype=float))
    return np.concatenate(parts)


def _objective_piecewise(p: np.ndarray, n: int, target_sum: float) -> float:
    """
    Objective for differential evolution:
    build the full vector, normalise to the required L1‑norm and return the
    maximal convolution value (the evaluate_sequence score when the sum is
    fixed).
    """
    a = _build_piecewise(p, n)
    s = a.sum()
    if s <= 0.0:
        return 1e9
    a = a * (target_sum / s)
    if np.any(a > 1000.0):               # safety – values >1000 are clipped by the defender
        return 1e9
    return _conv_max_fft(a)


class PiecewiseObjective:
    """Callable wrapper required by scipy.optimize (no closure / lambda)."""

    def __init__(self, n: int, target_sum: float):
        self.n = n
        self.target_sum = target_sum

    def __call__(self, p: np.ndarray) -> float:
        return _objective_piecewise(p, self.n, self.target_sum)


# ----------------------------------------------------------------------
def _adam_optimize(
    a_init: np.ndarray,
    n: int,
    target_sum: float,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    max_iter: int = 300_000,
    jitter_interval: int = 3000,
    random_active_prob: float = 0.4,
):
    """
    Adam optimiser for the max‑convolution objective (fixed length).  Returns
    the best sequence seen so far together with its score.
    """
    a = a_init.copy()
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.03
    iteration = 0

    sqrt2 = math.sqrt(2.0)
    coeff = 2.0

    while time.time() < deadline and iteration < max_iter:
        # --- 1) convolution and active set -------------------------------------------------
        M, _, conv_full = _conv_max_fft_full(a)
        tol = 1e-12
        active = np.where(conv_full >= M - tol)[0]

        # --- 2) sub‑gradient ----------------------------------------------------------------
        if active.size and np.random.rand() < random_active_prob:
            k_sel = np.random.choice(active)
            rev = _rev_for_shift(a, int(k_sel))
        else:
            rev = _subgradient_average(a, active)

        # --- 3) gradient of the scalarised objective ----------------------------------------
        const = sqrt2 * M / math.sqrt(n)
        grad = coeff * rev - const

        # --- 4) Adam update -----------------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # --- 5) projection (non‑negative + normalisation) ------------------------------------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # --- 6) occasional jitter ------------------------------------------------------------
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # --- 7) bookkeeping ------------------------------------------------------------------
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[Adam n={n} iter={iteration}] new best = {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(
    a_start: np.ndarray,
    n: int,
    target_sum: float,
    best_score: float,
    deadline: float,
    max_moves: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap SA / hill‑climb that moves a small amount of mass from one index to
    another.  It accepts worsening moves with probability exp(-Δ/T) where the
    temperature T slowly cools.
    """
    a = a_start.copy()
    cur_score = _conv_max_fft(a)
    best_seq = a.copy()
    best_loc = cur_score

    # initial temperature – a fraction of the current score
    temp = max(cur_score * 0.1, 1e-3)
    decay = 0.9995

    for mv in range(max_moves):
        if time.time() >= deadline:
            break

        i = np.random.randint(n)
        if a[i] <= 1e-14:
            continue
        # pick a different index j
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        eps = a[i] * np.random.rand() * 0.05
        if eps <= 0.0:
            continue

        new_a = a.copy()
        new_a[i] -= eps
        new_a[j] += eps

        new_score = _conv_max_fft(new_a)
        delta = new_score - cur_score
        if delta < 0.0 or np.random.rand() < math.exp(-delta / max(temp, 1e-9)):
            a = new_a
            cur_score = new_score
            if new_score < best_loc:
                best_loc = new_score
                best_seq = new_a.copy()
                print(f"[SA] improvement to {best_loc:.6f}")

        temp *= decay
        if mv % 5000 == 0 and temp < 1e-6:
            temp = max(cur_score * 0.1, 1e-6)

    return best_seq, best_loc


# ----------------------------------------------------------------------
def _optimize_one_length(
    n: int,
    init_candidates: list[np.ndarray],
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Optimise a specific length `n`.  Performs a global DE search on a
    piece‑wise constant parametrisation, then refines every supplied initial
    candidate with Adam + SA.  Returns the best sequence found for this length.
    """
    target_sum = math.sqrt(2.0 * n)

    # --- 1) piece‑wise DE (very low dimensional) -----------------------------------------
    dim = max(4, min(12, n // 200))            # between 4 and 12 segments
    bounds = [(0.0, target_sum) for _ in range(dim)]

    # allocate a modest amount of iterations (30) – enough for a quick improvement
    de_iters = 30
    try:
        de_result = differential_evolution(
            PiecewiseObjective(n, target_sum),
            bounds=bounds,
            maxiter=de_iters,
            polish=False,
            seed=0,
            disp=False,
        )
        p_opt = de_result.x
        a_opt = _build_piecewise(p_opt, n)
        a_opt = a_opt * (target_sum / a_opt.sum())
        de_score = _conv_max_fft(a_opt)
        best_seq = a_opt.copy()
        best_score = de_score
        print(f"[DE n={n}] score={de_score:.6f}")
    except Exception as e:
        print(f"[DE] failed for n={n}: {e}")
        best_seq = None
        best_score = float("inf")

    # --- 2) Adam + SA for each initial candidate -----------------------------------------
    for idx, a0 in enumerate(init_candidates):
        if time.time() >= deadline:
            break

        # Adam phase (shorter than the reference implementation)
        a_adam, score_adam = _adam_optimize(
            a_init=a0,
            n=n,
            target_sum=target_sum,
            best_seq=best_seq if best_seq is not None else a0,
            best_score=best_score,
            deadline=deadline,
            max_iter=200_000,
            jitter_interval=2500,
            random_active_prob=0.45,
        )
        if score_adam < best_score:
            best_score = score_adam
            best_seq = a_adam.copy()
            print(f"[Adam n={n} init={idx}] new best={best_score:.6f}")

        # SA refinement
        if time.time() < deadline:
            a_sa, score_sa = _simulated_annealing(
                a_start=best_seq,
                n=n,
                target_sum=target_sum,
                best_score=best_score,
                deadline=deadline,
                max_moves=25_000,
            )
            if score_sa < best_score:
                best_score = score_sa
                best_seq = a_sa.copy()
                print(f"[SA n={n}] improved to {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Main entry point used by the competition harness.  Searches for a step
    function of non‑negative heights that minimises `evaluate_sequence`.  The
    function respects the 1000 s time budget and returns the best vector found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # safety margin

    # ------------------------------------------------------------------
    # Load a previously discovered construction (if present)
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        base = np.full(2100, 1.0, dtype=float)

    n0 = base.shape[0]
    target_sum0 = math.sqrt(2.0 * n0)
    base = base * (target_sum0 / base.sum())
    init_score = evaluate_sequence(base.tolist())
    best_seq = base.copy()
    best_score = init_score
    print(f"[init] length={n0}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Choose a set of lengths to explore – larger vectors often allow a lower
    # maximal convolution value.  We keep them ≤ 8000 for speed.
    length_factors = [1.0, 1.5, 2.0, 3.0]
    lengths = sorted({int(n0 * f) for f in length_factors}, reverse=True)
    lengths = [l for l in lengths if 300 <= l <= 8000]

    # ------------------------------------------------------------------
    # Main optimisation loop over the selected lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        print(f"\n[search] trying length={n}")

        target_sum = math.sqrt(2.0 * n)

        # ----- build a handful of diverse starting vectors -----------------
        init_candidates = []

        # 1) linear interpolation of the historic base
        old_x = np.linspace(0.0, 1.0, n0)
        new_x = np.linspace(0.0, 1.0, n)
        interp = np.interp(new_x, old_x, base)
        init_candidates.append(interp)

        # 2) linearly decreasing profile
        idx = np.arange(n)
        lin = (n - idx).astype(float)
        lin = lin / lin.sum() * target_sum
        init_candidates.append(lin)

        # 3) quadratic decreasing profile
        quad = (n - idx) ** 2.0
        quad = quad / quad.sum() * target_sum
        init_candidates.append(quad)

        # 4) gentle exponential decay
        alpha = 0.0015 * n          # a modest decay that works for any size
        expo = np.exp(-alpha * idx / n)
        expo = expo / expo.sum() * target_sum
        init_candidates.append(expo)

        # 5) random Dirichlet‑like start (sorted decreasing)
        rnd = np.random.rand(n)
        rnd = np.sort(rnd)[::-1]
        rnd = rnd / rnd.sum() * target_sum
        init_candidates.append(rnd)

        # ----- optimise this length ----------------------------------------
        seq, scr = _optimize_one_length(n, init_candidates, deadline)

        if seq is not None and scr < best_score:
            best_score = scr
            best_seq = seq.copy()
            print(f"[global] new best (len={n}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform vector")

    print(f"\n[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
