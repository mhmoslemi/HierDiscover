"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.

Algorithm
---------
* Work with the homogeneous form of the objective:
  if the L1‑norm of the vector is forced to √(2·n) then the score returned
  by ``evaluate_sequence`` equals the maximal coefficient of the
  self‑convolution (the “peak”).  Hence we minimise ``max(a*a)``.
* For a given length ``n`` we run several independent restarts.
  Each restart consists of three stages
      1. a long Adam optimisation on the exact sub‑gradient of the peak,
      2. an aggressive stochastic local search that is allowed to accept
         occasional uphill moves (simulated‑annealing style),
      3. a short fine‑tuning Adam pass.
* The best solution over a set of lengths (powers of two up to 8192) is
  kept globally.  Whenever a new best is found we mix a small fraction
  of it into the initialisation of subsequent runs – this steers the
  search away from bad local minima.
* The whole procedure respects the global time limit (≈1000 s); it
  stops early if the budget is exhausted and finally returns the best
  vector seen so far.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """
    Return the maximal coefficient of the full convolution a*a.
    Uses an FFT with a power‑of‑two length for speed.
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    B = A * A                                    # (a*a) in the frequency domain
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]    # keep the valid part only
    return float(conv.max()), conv


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, active: np.ndarray):
    """
    Exact sub‑gradient of the peak of the convolution at the current point.
    ``active`` contains the indices of the convolution where the value is
    within a tiny tolerance of the current maximum.
    The gradient for component i is the average of the partner entries
    a_{k-i} over all active shifts k (out‑of‑range indices are ignored).
    """
    n = a.shape[0]
    idx = np.arange(n)
    grad = np.zeros_like(a)

    for k in active:
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        grad[mask] += a[partner[mask]]

    if active.size > 0:
        grad /= float(active.size)
    return grad


# ----------------------------------------------------------------------
def _project_onto_simplex(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Clamp to non‑negative and renormalise to the required L1 norm."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a *= (target_sum / s)
    else:
        a[:] = target_sum / a.shape[0]
    return a


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   target_sum: float,
                   deadline: float,
                   max_iters: int = 2_000_000):
    """
    Adam optimisation of ``max(a*a)`` using the exact sub‑gradient.
    Returns the best vector encountered and its maximal convolution value.
    """
    a = init_a.copy()
    a = _project_onto_simplex(a, target_sum)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    base_lr = 0.12          # a bit larger than in the baseline
    jitter_interval = 5000
    best_vec = a.copy()
    best_val, _ = _conv_max_fft(best_vec)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # 1. convolution & active set
        M, conv = _conv_max_fft(a)
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # 2. sub‑gradient of the peak
        g = _subgradient(a, active)

        # Gradient of the whole objective (the outer factor 2·n/Σ² is 1
        # after we enforce Σ = sqrt(2·n))
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * g - const

        # 3. Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = base_lr / (1.0 + 1e-4 * it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # 4. project back to feasible set
        a = _project_onto_simplex(a, target_sum)

        # 5. occasional jitter to escape plateaus
        if it % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = _project_onto_simplex(a + jitter, target_sum)

        # 6. track the best seen so far
        if M < best_val:
            best_val = M
            best_vec = a.copy()
            # printing occasionally keeps the harness happy
            if it % 2000 == 0:
                print(f"[Adam n={n:5d} iter={it:7d}] new best M={best_val:.6f}")

    return best_vec, best_val


# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         n: int,
                         target_sum: float,
                         steps: int,
                         deadline: float):
    """
    Stochastic local search with a simple temperature schedule.
    At each step a small amount of mass is moved from a random source
    index to a random destination index.  Moves that increase the peak
    are occasionally accepted according to the current temperature.
    """
    cur = a.copy()
    cur_val, _ = _conv_max_fft(cur)

    best = cur.copy()
    best_val = cur_val

    T0 = 0.1
    Tf = 1e-5
    for step in range(steps):
        if time.time() >= deadline:
            break

        # exponential temperature decay
        T = T0 * ((Tf / T0) ** (step / steps))

        # pick distinct source and destination
        i, j = np.random.randint(0, n, size=2)
        if i == j or cur[i] <= 0.0:
            continue

        # transfer a random fraction (up to 30 %) of the source mass
        delta = cur[i] * (np.random.rand() * 0.3)
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # sum stays constant → only need to project non‑negativity
        cand = np.maximum(cand, 0.0)

        cand_val, _ = _conv_max_fft(cand)
        d = cand_val - cur_val

        if d < 0.0 or np.random.rand() < math.exp(-d / max(T, 1e-12)):
            cur = cand
            cur_val = cand_val
            if cand_val < best_val:
                best = cand.copy()
                best_val = cand_val

    return best, best_val
