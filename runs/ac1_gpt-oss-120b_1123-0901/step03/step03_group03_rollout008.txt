def _softmax_weights(conv: np.ndarray, tau: float):
    """
    Stable soft‑max of the convolution values.
    Returns a weight vector w (summing to 1) and the smoothed maximum Lτ.
    """
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    # subtract max for numerical stability
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(p: np.ndarray, w: np.ndarray, Nfft: int):
    """
    Compute the weighted reverse convolution:
        rev_i = Σ_k w_k * p_{k-i}
    (indices outside 0…len(p)-1 are ignored) using FFTs.
    """
    # FFT of the probability vector (zero‑padded)

    # FFT of the reversed weight vector
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    # Convolution in the frequency domain
    conv_rev = np.fft.irfft(W_rev_fft * P_fft, n=Nfft)

    conv_rev = conv_rev[: len(w) + p.shape[0] - 1]
    # extract the slice that corresponds to rev_i
    idx_start = 2 * p.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(p.shape[0])]
    return rev


def _project_onto_simplex(v: np.ndarray):
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1}.  Uses the O(n) algorithm of Wang & Carreira‑Perpiñán.
    """
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core optimisation routine for a single length and a single initialise
# ----------------------------------------------------------------------


def _optimize_one_length(
    n: int,
    init_p: np.ndarray,
    best_p: np.ndarray,
    best_score: float,
    deadline: float,
    max_iters: int,
):
    """
    Adam optimisation of the smooth surrogate of ``2·n·max(conv(p,p))``.
    Works on the probability vector `p` (sum = 1).  Returns the best
    probability vector found and the corresponding (exact) objective.
    """
    # FFT size – next power of two ≥ 2·n‑1 (re‑used for the whole run)
    Nfft = 1 << ((2 * n - 1).bit_length())

    # initialise Adam state
    p = init_p.copy()
    p = _project_onto_simplex(p)                # safety
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.08                                   # a bit larger than before
    iteration = 0
    jitter_interval = 4000

    # temperature schedule parameters
    tau_factor = 0.25
    decay_rate = 15000.0

    while time.time() < deadline and iteration < max_iters:
        # --------------------------------------------------------------
        # exact self‑convolution of the probability vector
        P_fft = np.fft.rfft(p, n=Nfft)
        conv = np.fft.irfft(P_fft * P_fft, n=Nfft)[: 2 * n - 1]

        M = float(conv.max())                     # max(conv(p,p))

        # exact objective (no scaling because Σp = 1)
        cur_obj = 2.0 * n * M

        # global best update (using the exact value, no evaluate_sequence call)
        if cur_obj < best_score:
            best_score = cur_obj
            best_p = p.copy()
            print(
                f"[len {n:5d}  it {iteration:9d}] new best = {best_score:.6f}"
            )

        # --------------------------------------------------------------
        # smooth approximation of the max
        tau = max(1e-12, tau_factor * M * math.exp(-iteration / decay_rate))
        w, Ltau = _softmax_weights(conv, tau)

        # --------------------------------------------------------------
        # gradient of the smoothed objective:
        #   d/dp (2·n·Lτ) = 2·n·(2·rev) = 4·n·rev
        rev = _weighted_rev(p, w, Nfft)
        grad = 4.0 * n * rev

        # --------------------------------------------------------------
        # Adam parameter update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # projection onto the simplex (ensures non‑negativity and Σp = 1)
        p = _project_onto_simplex(p)

        # --------------------------------------------------------------
        # occasional jitter to help escape plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 / n)
            p = _project_onto_simplex(p + jitter)

    return best_p, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget (≈1000 s).

    Returns a plain Python list of heights.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the harness
    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # Base construction (if available)
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        base = np.ones(2100, dtype=float)

    n0 = int(base.shape[0])
    # normalise base to a probability vector (sum = 1)
    base = base / base.sum()

    # ------------------------------------------------------------------
    # Lengths to explore (original and a few enlarged versions)
    length_factors = [1.0, 1.25, 1.5, 2.0]
    lengths = sorted({max(200, int(n0 * f)) for f in length_factors})
    lengths = [min(l, 8000) for l in lengths]          # upper bound on size

    # ------------------------------------------------------------------
    # Global best (probability vector + objective)
    best_p = base.copy()
    best_score = 2.0 * n0 * np.max(np.convolve(best_p, best_p))  # exact objective

    # ------------------------------------------------------------------
    # Helper to produce diverse initial probability vectors
    def _initial_vectors(n):
        # uniform
        uni = np.full(n, 1.0 / n)

        # discretised arcsine on [0,1]
        x = (np.arange(n) + 0.5) / n
        arcs = 1.0 / np.sqrt(x * (1.0 - x))
        arcs = arcs / arcs.sum()

        # exponential decay
        exp = np.exp(-5.0 * np.arange(n) / n)
        exp = exp / exp.sum()

        # random Dirichlet
        rand = np.random.dirichlet(np.ones(n))

        # interpolated base (if its length differs)
        if n != base.shape[0]:
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
            interp = np.maximum(interp, 0.0)
            interp = interp / interp.sum()
        else:
            interp = base.copy()

        return [uni, arcs, exp, rand, interp]

    # ------------------------------------------------------------------
    # Main optimisation loop over all lengths and initialisers
    for n in lengths:
        if time.time() >= deadline:
            break

        print(f"[run] length = {n}")

        # allocate a per‑run iteration budget (larger vectors get fewer iters)
        max_iters = int(2_000_000 * 2100 / n)
        max_iters = max(1_000, max_iters)

        inits = _initial_vectors(n)

        for idx, p0 in enumerate(inits):
            if time.time() >= deadline:
                break
            best_p, best_score = _optimize_one_length(
                n=n,
                init_p=p0,
                best_p=best_p,
                best_score=best_score,
                deadline=deadline,
                max_iters=max_iters,
            )
            # we have already printed improvements inside the optimiser

    # ------------------------------------------------------------------
    # Convert the best probability vector to the final height sequence
    final_n = best_p.shape[0]
    final_scale = math.sqrt(2.0 * final_n)          # Σa = √(2 n)
    final_seq = (best_p * final_scale).tolist()

    # final sanity check (optional – harmless if it exceeds the time budget)
    try:
        final_val = evaluate_sequence(final_seq)
        print(f"[done] best objective = {final_val:.6f}, length = {final_n}")
    except Exception:
        # if the harness kills us for being too slow we just announce what we have
        print(f"[done] length = {final_n}")

    return final_seq
