def lp_step(a, M):
    n = len(a)
    # Build sparse matrix data for constraints
    rows, cols, data = [], [], []
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # For each i in that range, j = k - i
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(a[i])
    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)  # all constraints <= M
    c = -np.ones(n)
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        return None
    b = res.x
    return b

def solve_lp(a, M):

   n = len(a)

   offsets = np.arange(n)

   data = [np.full(n, a[i]) for i in range(n)] # each diagonal data

   A_ub = sp.diags(data, offsets, shape=(2*n-1, n), format='csr')

   b_ub = np.full(2*n-1, M, dtype=float)

   c = -np.ones(n)

   res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')

   if not res.success: return None

   return res.x

"""
search for a non‑negative step‑function that minimises ``evaluate_sequence``.
Algorithm (≈ 1000 s budget)

1. Load the previous construction (``height_sequence_1``) or start from a uniform
   vector and normalise it to Σa = √(2 n).

2. Random‑sample a few hundred Dirichlet vectors with different concentration
   parameters – keep the best one.

3. Refine the best vector with a smooth‑max Adam optimiser (the implementation
   from the previous state‑of‑the‑art, but with a larger learning‑rate,
   a slightly faster temperature decay and more frequent jitter).

4. Perform a tiny linear‑programming step: for the current a we solve
   ``max Σ b_j``  subject to  b_j ≥ 0 and (a * b)_k ≤ M  (M = max(a*a)).
   The optimiser returns a new vector g; we mix a←(1‑t)·a + t·g
   (t decays geometrically).  A handful of such LP updates often lower the
   max‑convolution further.

5. Upsample the current shape to a larger length (× 1.5, bounded by 5000),
   re‑normalise and run the Adam optimiser again (shorter iteration budget).

6. Final greedy hill‑climbing: repeatedly move a tiny amount of mass from a
   random index i to j, update the convolution incrementally and keep the
   move only if it reduces the exact max‑convolution.

7. Return the best sequence found (as a plain list).
"""

import time
import math
import numpy as np

# optional – used in the LP refinement stage
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
    _HAS_SCIPY = True
except Exception:          # pragma: no cover
    _HAS_SCIPY = False


def _softmax_weights(conv: np.ndarray, tau: float):
    """stable soft‑max of a vector; returns (weights, smooth_max)."""
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, float(conv.max())
    maxc = conv.max()
    ex = np.exp((conv - maxc) / tau)
    s = ex.sum()
    w = ex / s
    L = maxc + tau * np.log(s)
    return w, float(L)


def _weighted_rev(a: np.ndarray, w: np.ndarray, nfft: int):
    """
    Compute rev_i = Σ_k w_k a_{k-i} (with out‑of‑range entries ignored)
    using a single FFT.
    """
    A = np.fft.rfft(a, n=nfft)
    Wrev = np.fft.rfft(w[::-1], n=nfft)
    rev = np.fft.irfft(A * Wrev, n=nfft)
    rev = rev[: len(a) + len(w) - 1]      # keep the valid part
    idx = 2 * len(a) - 2
    rev = rev[idx - np.arange(len(a))]
    return rev


def _max_convolution(a: np.ndarray) -> float:
    """exact max of a∗a (full linear convolution)."""
    n = a.shape[0]
    Nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
    return float(conv.max())


def _adam_optimize(
    a: np.ndarray,
    target_sum: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    *,
    lr0: float = 0.07,
    tau_factor: float = 0.25,
    decay_rate: float = 15000.0,
    jitter_interval: int = 2000,
    max_iters: int = 10_000_000,
    prefix: str = "",
):
    """Adam optimiser for the smooth max of a∗a."""
    n = a.shape[0]
    a = a * (target_sum / a.sum())

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    it = 0
    Nfft = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline and it < max_iters:
        # 1) exact convolution and its maximum
        A = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # 2) smooth max (soft‑max) with annealed temperature
        tau = max(1e-12, tau_factor * M * math.exp(-it / decay_rate))
        w, Ltau = _softmax_weights(conv, tau)

        # 3) sub‑gradient of the smooth max
        rev = _weighted_rev(a, w, Nfft)

        # 4) Adam step (gradient of the original objective after normalisation)
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # 5) projection onto a≥0, Σa = target_sum
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # 6) occasional jitter to escape shallow plateaus
        if it % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # 7) exact objective (after normalisation it equals M)
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"{prefix}[adam iter {it:7d}] new best = {best_score:.6f}")

    return best_seq, best_score


def _solve_lp_for_b(a: np.ndarray, M: float):
    """solve   max Σ b_j   s.t.  b≥0  and  a∗b ≤ M."""
    if not _HAS_SCIPY:                     # pragma: no cover
        return None
    n = a.shape[0]

    # build Toeplitz (convolution) matrix A_ub (2n‑1)×n as a sparse CSR matrix
    offsets = np.arange(n, dtype=int)
    data = [np.full(n, a[i], dtype=float) for i in range(n)]
    A_ub = _sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")

    b_ub = np.full(2 * n - 1, M, dtype=float)
    c = -np.ones(n, dtype=float)          # maximise Σb -> minimise –Σb

    # modern SciPy uses the HiGHS solver, which is fast for these sizes
    res = _opt.linprog(c, A_ub=A_ub, b_ub=b_ub,
                       bounds=(0, None), method="highs")
    if not res.success:
        return None
    return res.x


def _lp_refine(
    a: np.ndarray,
    target_sum: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_steps: int = 12,
):
    """perform a few LP‑based updates (complete the iteration described in the statement)."""
    n = a.shape[0]
    for step in range(max_steps):
        if time.time() >= deadline:
            break
        M = _max_convolution(a)
        b = _solve_lp_for_b(a, M)
        if b is None:
            break
        B = b.sum()
        if B <= 0:
            break
        # rescales b to the required ℓ₁‑norm
        g = b * (target_sum / B)

        # mixing factor that shrinks geometrically
        t = 0.5 * (0.8 ** step)
        a = (1.0 - t) * a + t * g
        a = np.maximum(a, 0.0)
        a = a * (target_sum / a.sum())

        cur = _max_convolution(a)
        if cur < best_score:
            best_score = cur
            best_seq = a.copy()
            print(f"[LP step {step:2d}] new best = {best_score:.6f}")
    return best_seq, best_score


def _hill_climb(
    a: np.ndarray,
    target_sum: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_iters: int = 20000,
):
    """tiny greedy coordinate‑wise search – move a little mass from i to j."""
    n = a.shape[0]
    # pre‑compute convolution once – we will keep it up‑to‑date incrementally
    Nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # transfer a tiny fraction of the mass at i to j
        max_delta = min(a[i], 0.01 * target_sum / n)
        delta = max_delta * np.random.rand()
        if delta <= 0.0:
            continue

        # build the two shift vectors (a placed at positions i and j)
        shift_i = np.zeros_like(conv)
        shift_i[i : i + n] = a
        shift_j = np.zeros_like(conv)
        shift_j[j : j + n] = a

        # new convolution after the move (exact formula dconv = 2·δ·(shift_j‑shift_i))
        conv_new = conv + 2.0 * delta * (shift_j - shift_i)
        new_score = float(conv_new.max())

        if new_score < best_score:
            # accept the move
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            best_score = new_score
            best_seq = a.copy()
            print(f"[hill {it:5d}] new best = {best_score:.6f}")

    return best_seq, best_score
