def _full_convolution(x, y):
    n = x.shape[0] + y.shape[0] - 1

    conv = np.fft.irfft(X*Y, n=n_fft)[:n]

"""
propose_candidate
-----------------

Search algorithm for the minimum of

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

under the constraints a_i ≥ 0 and sum(a) = √(2 n).

The search consists of many short restarts.  For each restart we

1. pick a length n (1500 – 4500) and build a *U‑shaped* seed
   a_i ∝ (i+1)^{-p} + (n-i)^{-p}  (p is random).
2. project the seed onto the simplex Σ a_i = √(2 n).
3. run a fast **Adam‑style projected gradient descent** on a
   smooth approximation of the max‑convolution:
   we minimise the soft‑max
        F(a) = (1/α)·log Σ_k exp(α·(a*a)[k])
   with α slowly increased from 1 to ≈30.
   The gradient of F can be written as a convolution and is
   computed by FFTs, so each iteration costs O(n log n).
4. after each iteration we re‑project onto the simplex.
   Random “shakes’’ are occasionally added to escape shallow minima.
5. the best vector seen over all restarts is returned as a plain
   Python list.

The whole procedure respects the 1000 s time budget, prints progress
messages and falls back to a uniform vector if nothing is found.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------------------- #
# Simplex projection (Euclidean) – Σ x_i = s , x_i ≥ 0
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Return the Euclidean projection of v onto {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Full convolution via FFT (real inputs)
# --------------------------------------------------------------------------- #
def _full_convolution(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of real vectors x and y."""
    n = x.shape[0] + y.shape[0] - 1
    n_fft = 1 << ((n - 1).bit_length())  # next power of two
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X * Y, n=n_fft)[:n]
    return conv


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Convolution of a with itself (full length = 2·len(a)‑1)."""
    return _full_convolution(a, a)


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a (full convolution)."""
    return float(_conv_full_fft(a).max())


# --------------------------------------------------------------------------- #
# Adam‑style projected gradient descent on a smoothed max‑convolution.
# --------------------------------------------------------------------------- #
def _optimize_softmax(a_init: np.ndarray,
                      target_sum: float,
                      max_time: float,
                      start_time: float):
    """Minimise the soft‑max of the autocorrelation of a."""
    a = a_init.copy()
    n = a.shape[0]

    best_vec = a.copy()
    best_score = _max_conv(a)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    it = 0
    # α will be increased from 1 to 30 during the run
    alpha_start = 1.0
    alpha_end = 30.0

    while time.time() < start_time + max_time:
        it += 1
        # ----- convolution of the current iterate -----
        conv = _conv_full_fft(a)
        max_val = conv.max()

        # ----- schedule for the soft‑max temperature -----
        elapsed = time.time() - start_time
        prog = elapsed / max_time                     # 0 … 1
        alpha = alpha_start + prog * (alpha_end - alpha_start)

        # ----- soft‑max weights (sum to 1) -----
        # stable exponentiation
        exps = np.exp(alpha * (conv - max_val))
        w = exps / exps.sum()

        # ----- gradient of the smoothed objective -----
        a_rev = a[::-1]                               # reverse for correlation
        grad_full = _full_convolution(w, a_rev)       # length = 3·n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]       # slice to size n

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule – start at 0.5 and decay to a tiny value
        lr = 0.5 * (1.0 - prog) + 1e-4

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # ----- feasibility projection -----
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        a = a_new

        cur_score = _max_conv(a)
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{time.time():.2f}] GD iter {it} new best {best_score:.6f} (n={n})")

        # occasional random shake to escape plateaus
        if np.random.rand() < 0.008:
            noise = np.random.randn(n) * (0.001 * target_sum / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)

    return best_vec, best_score
