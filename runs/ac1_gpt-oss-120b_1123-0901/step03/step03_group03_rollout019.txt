```
def solve_qcqp_small_n(n):
    import cvxpy as cp
    a = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    constraints = [cp.sum(a) == cp.sqrt(2.0 * n)]
    # Build Q_i matrices
    for i in range(n):
        Q = np.zeros((n, n), dtype=float)
        # j from max(0,i-(n-1)) to min(n-1,i)
        jmin = max(0, i - (n - 1))
        jmax = min(n-1, i)
        # for each j, k = i - j
        for j in range(jmin, jmax+1):
            k = i - j
            if 0 <= k < n:
                Q[j, k] = 1.0
        constraints.append(cp.quad_form(a, Q) <= t)
    # t >= 0
    constraints.append(t >= 0)
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver='ECOS', warm_start=True, max_iters=50000)
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return None, None
    return a.value, t.value
```

def generate_window(n, typ, param=None):
    i = np.arange(n, dtype=float)
    if typ == "hann":
        w = 0.5 * (1 - np.cos(2*np.pi*i/(n-1)))
    elif typ == "hamming":
        w = 0.54 - 0.46*np.cos(2*np.pi*i/(n-1))
    elif typ == "blackman":
        w = 0.42 - 0.5*np.cos(2*np.pi*i/(n-1)) + 0.08*np.cos(4*np.pi*i/(n-1))
    elif typ == "bartlett":
        w = 1 - np.abs((i - (n-1)/2) / ((n-1)/2))
    elif typ == "tukey":
        alpha = param if param is not None else 0.5
        # tukey function:
        w = np.ones_like(i)
        if alpha > 0:
            # fractional region
            period = (n-1)/2
            left = i < alpha * period
            right = i > (n-1) - alpha * period
            # compute left taper
            w[left] = 0.5*(1 + np.cos(np.pi*(2*i[left]/(alpha*(n-1)) - 1)))
            # compute right taper
            w[right] = 0.5*(1 + np.cos(np.pi*(2*i[right]/(alpha*(n-1)) - 2/alpha + 1)))
    elif typ == "gaussian":
        sigma = param if param is not None else 0.4  # relative to half-length
        center = (n-1)/2
        w = np.exp(-0.5*((i - center) / (sigma * center))**2)
    elif typ == "power":
        p = param if param is not None else 2
        center = (n-1)/2
        d = np.abs(i - center)
        half_width = (n-1)/2  # full support
        w = np.maximum(0, 1 - (d/half_width)**p)
    else:
        raise ValueError
    return w
```

def tukey_window(N, alpha):
    N = int(N)
    if alpha <= 0:
        return np.ones(N)
    if alpha >= 1:
        return np.hanning(N)  # or 0.5*(1-cos)
    n = np.arange(N)
    w = np.ones(N)
    per = (alpha * (N - 1)) / 2.0
    left = n < per

"""
Non‑negative step‑function optimizer.

The algorithm consists of two stages:
1. **Window‑search** – cheap sweep over several classic window shapes
   (Hann, Hamming, Blackman, Bartlett, Tukey, Gaussian, Power‑law).
   Each window is normalised to Σa = √(2 n) and evaluated.  The best
   window provides a strong starting point.
2. **Smooth‑max Adam optimisation** – a refined gradient descent that
   minimises the smooth approximation
        Lτ = τ·log∑exp((a∗a)/τ)
   of the maximal self‑convolution.  The gradient is obtained analytically
   by a single FFT and the update is projected onto the feasible set
   (a≥0, Σa = √(2 n)).  The temperature τ is annealed and occasional
   Gaussian jitter helps to escape plateaus.  The routine runs until the
   overall time budget (≈1000 s) is exhausted, always keeping the best
   sequence seen.

The returned sequence is a plain list of non‑negative floats ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: generate a family of windows
def generate_window(n: int, typ: str, param=None) -> np.ndarray:
    """Return a non‑negative window of length ``n``.
    ``typ`` can be 'hann', 'hamming', 'blackman', 'bartlett',
    'tukey', 'gaussian' or 'power'.  ``param`` supplies the extra
    shape parameter where needed (alpha for Tukey, sigma for Gaussian,
    p for Power‑law)."""
    i = np.arange(n, dtype=float)

    if typ == "hann":
        w = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    elif typ == "hamming":
        w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    elif typ == "blackman":
        w = (
            0.42
            - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
            + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
        )
    elif typ == "bartlett":
        # triangular (Bartlett) window
        w = 1.0 - np.abs((i - (n - 1) / 2.0) / ((n - 1) / 2.0))
    elif typ == "tukey":
        # Tukey (tapered cosine) window
        alpha = 0.5 if param is None else float(param)
        if alpha <= 0.0:
            w = np.ones_like(i)
        elif alpha >= 1.0:
            w = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))  # Hann
        else:
            w = np.ones_like(i)
            per = alpha * (n - 1) / 2.0
            left = i < per
            right = i > (n - 1 - per)
            # left taper
            w[left] = 0.5 * (
                1.0
                + np.cos(
                    np.pi * (2.0 * i[left] / (alpha * (n - 1)) - 1.0)
                )
            )
            # right taper
            w[right] = 0.5 * (
                1.0
                + np.cos(
                    np.pi
                    * (2.0 * i[right] / (alpha * (n - 1)) - 2.0 / alpha + 1.0)
                )
            )
    elif typ == "gaussian":
        # sigma is relative to half‑length, default 0.4
        sigma = 0.4 if param is None else float(param)
        centre = (n - 1) / 2.0
        denom = sigma * centre
        if denom == 0:
            w = np.zeros_like(i)
        else:
            w = np.exp(-0.5 * ((i - centre) / denom) ** 2)
    elif typ == "power":
        # w = max(0, 1 - (|i-centre|/half_width)^p)
        p = 2.0 if param is None else float(param)
        centre = (n - 1) / 2.0
        half = centre
        d = np.abs(i - centre)
        with np.errstate(divide="ignore", invalid="ignore"):
            w = np.maximum(0.0, 1.0 - (d / half) ** p)
    else:
        raise ValueError(f"unknown window type {typ}")

    # Ensure non‑negativity (numerical noise)
    w = np.clip(w, 0.0, None)
    return w


# ----------------------------------------------------------------------
# Soft‑max utilities used by the Adam optimiser
def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of ``conv``.  Returns (weights, smoothed max)."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """Compute rev_i = Σ_k w_k a_{k‑i} (zero‑pad outside range) via FFT."""
    # FFT of a (zero-padded)
    A_fft = np.fft.rfft(a, n=Nfft)
    # FFT of reversed weight
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # Keep only the part of length len(w)+len(a)-1
    conv_rev = conv_rev[: len(w) + len(a) - 1]
    # Slice the required n elements (see original implementation)
    idx_start = 2 * len(a) - 2
    rev = conv_rev[idx_start - np.arange(len(a))]
    return rev


# ----------------------------------------------------------------------
# Adam optimiser for a fixed length
def _optimize_length(
    n: int,
    init_a: np.ndarray,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    max_iters: int = 10_000_000,
):
    """
    Smooth‑max Adam optimiser for length ``n``.
    ``init_a`` is assumed already normalised (Σa = √(2 n)).
    The function updates the global best (``best_seq``, ``best_score``)
    whenever a strictly better max‑convolution is found.
    """
    target_sum = math.sqrt(2.0 * n)

    a = init_a.copy()
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.05
    iteration = 0
    jitter_interval = 3000  # more frequent than before

    tau_factor = 0.30   # a slightly larger initial temperature
    decay_rate = 25000.0

    # FFT size (next power‑of‑two ≥ 2·n‑1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline and iteration < max_iters:
        # ---- exact self‑convolution via FFT ----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ---- soft‑max approximation of the max ----
        tau = max(1e-12, tau_factor * M * math.exp(-iteration / decay_rate))
        w, Ltau = _softmax_weights(conv, tau)

        # ---- gradient of the smooth max (sub‑gradient) ----
        rev = _weighted_rev(a, w, Nfft)

        # ---- Adam update (gradient of the objective 2·n·max / Σ²) ----
        coeff = 4.0 * n / (target_sum * target_sum)   # = 2.0
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- projection onto feasibility set (a≥0, Σa = target_sum) ----
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # ---- occasional jitter to break plateaus ----
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # ---- exact objective value (after normalisation it equals M) ----
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:8d}] new best = {best_score:.6f}")

    return best_seq, best_score
