from scipy.optimize import linprog

def solve_lp_max_sum(f, M):
    n = len(f)
    # build A_ub
    # create matrix of shape (2n-1, n)
    rows = []
    for k in range(2*n-1):
        rev = _rev_for_shift(f, k)
        rows.append(rev)
    A_ub = np.vstack(rows)
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)  # maximize sum(g) = -minimize -sum(g)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    else:
        return None

def solve_g_max_sum(f, M, sum_target):
    n = len(f)
    # active constraints
    conv_f_f = np.convolve(f, f)
    # use tolerance - e.g., 1e-12 relative to M
    tol = 1e-12
    active = np.where(conv_f_f >= M - tol)[0]
    # Build A_ub and b_ub
    rows = []
    for k in active:
        rev = _rev_for_shift(f, k)
        rows.append(rev)
    if not rows:
        # fallback: use all constraints? Use no constraints -> g = ones? but we need constraints.
        # Use one constraint at shift 0
        rows.append(_rev_for_shift(f, 0))
        active = [0]
    A_ub = np.vstack(rows)
    b_ub = np.full(A_ub.shape[0], M, dtype=float)
    # Use bounds (0, 1000)
    bounds = [(0.0, 1000.0)] * n
    # objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    # Solve LP
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback uniform
        g = np.full(n, sum_target / n)
    else:
        g = res.x
    # scale to sum_target (if sum > 0)
    s = g.sum()
    if s > 0:
        g = g * (sum_target / s)
    else:
        g = np.full(n, sum_target / n)
    return g
```

def solve_g_cvxpy(f, M, sum_target):

    n = len(f)
    conv_f = np.convolve(f,f)
    tol = 1e-12
    active = np.where(conv_f >= M - tol)[0]
    g = cp.Variable(n, nonneg=True)
    constraints = []

        rev = _rev_for_shift(f, k)
        constraints.append(rev @ g <= M)
    # add bound constraints g <= 1000
    constraints.append(g <= 1000)
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if g.value is None:
        return np.full(n, sum_target / n)
    g_opt = g.value
    s = g_opt.sum()
    if s > 0:
        return g_opt * (sum_target / s)
    else:
        return np.full(n, sum_target / n)

"""
Propose a better step‑function by alternating a linear‑programming
expansion with a simple mixing step.

The objective of evaluate_sequence is 2·n·max(conv)/sum(a)².
If we keep the L1–norm fixed to √(2n) the objective equals the maximal
autoconvolution value M = max(a*a).  For a fixed f we solve the LP

    maximise   Σ g_j
    subject to  (f * g)[k] ≤ M   for all shifts k
                0 ≤ g_j ≤ 1000

All constraints are linear because (f*g) is linear in g.
The LP pushes as much mass as possible into positions that are not
active in the current maximal convolution.  After solving we
renormalise g to the required L1‑norm and replace f by a convex
combination f←(1‑t)f + t·g (t≈0.2).  This operation never increases the
current maximum and therefore monotonically improves the score.
After a few hundred such updates we run a cheap coordinate‑swap hill‑climb
to polish the result.

The implementation works for any length (the default is the length of the
provided `height_sequence_1`).  It falls back to a simple Adam optimiser
if neither SciPy nor CVXPY is available.  The whole search respects the
global 1000 s budget and prints progress information.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper: maximal full autocorrelation via FFT
def _conv_max_fft(a: np.ndarray):
    """Return (max_val, argmax_index, full_convolution) of a*a."""
    n = a.shape[0]
    # next power of two for efficient FFT
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    max_val = float(conv.max())
    idx = int(conv.argmax())
    return max_val, idx, conv


# --------------------------------------------------------------
# Helper: coefficient vector for a given shift k
def _rev_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    """
    Returns a vector rev such that rev·g = (a * g)[k].

    rev[i] = a[k-i] if 0 ≤ k-i < n else 0.
    """
    n = a.shape[0]
    rev = np.zeros_like(a)
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    rev[mask] = a[j[mask]]
    return rev


# --------------------------------------------------------------
# Linear‑programming sub‑routine (SciPy → CVXPY → fallback)
def _solve_lp_max_sum(f: np.ndarray, M: float, conv_full: np.ndarray,
                      target_sum: float):
    """
    Solve   maximise Σ g_j   subject to   (f * g)[k] ≤ M
    for the *active* shifts where conv_f_f ≈ M.
    Returns a feasible g that is then renormalised to `target_sum`.
    """
    n = f.shape[0]
    # active shifts – a tiny tolerance avoids numerical issues
    tol = 1e-12
    active = np.where(conv_full >= M - tol)[0]
    if active.size == 0:
        active = np.array([0], dtype=int)

    # Build the (usually very small) constraint matrix
    rows = [_rev_for_shift(f, k) for k in active]
    A_ub = np.vstack(rows)
    b_ub = np.full(A_ub.shape[0], M, dtype=float)

    # Upper bound 1000 follows the checks inside evaluate_sequence
    bounds = [(0.0, 1000.0)] * n

    # Try SciPy first
    try:
        from scipy.optimize import linprog
        res = linprog(
            c=-np.ones(n),          # maximise sum(g)  → minimise -sum(g)
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
        )
        if res.success and res.x is not None:
            g = res.x
        else:
            raise RuntimeError("Linprog failed")
    except Exception:  # SciPy not present or problem infeasible
        # Try CVXPY
        try:
            import cvxpy as cp

            g_var = cp.Variable(n, nonneg=True)
            constraints = [A_ub @ g_var <= b_ub,
                           g_var <= 1000.0]
            prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
            if g_var.value is not None:
                g = np.asarray(g_var.value, dtype=float)
            else:
                raise RuntimeError("CVXPY failed")
        except Exception:
            # Very simple fallback: move a tiny amount of mass towards the
            # uniform direction – this is never worse than keeping f unchanged.
            uniform = np.full(n, target_sum / n)
            g = (f + uniform) / 2.0

    # Renormalise to the prescribed L1‑norm (scaling does not increase any
    # (f * g) entry because it is linear in g).
    s = np.sum(g)
    if s > 0:
        g = g * (target_sum / s)
    else:
        g = np.full(n, target_sum / n)
    return g


# --------------------------------------------------------------
# Very cheap hill‑climb: move a small amount of mass between two positions
def _local_search(a_start: np.ndarray, n: int, target_sum: float,
                  best_score: float, deadline: float,
                  max_proposals: int = 30_000):
    a = a_start.copy()
    proposals = 0
    while time.time() < deadline and proposals < max_proposals:
        proposals += 1
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        # transfer up to 5 % of the donor's mass
        eps = a[i] * np.random.rand() * 0.05
        if eps <= 0:
            continue
        new_vec = a.copy()
        new_vec[i] -= eps
        new_vec[j] += eps
        # keep sum unchanged, only need the new maximum convolution
        M_new, _, _ = _conv_max_fft(new_vec)
        if M_new < best_score:
            a = new_vec
            best_score = M_new
            print(f"[local] improvement to {best_score:.6f} (prop {proposals})")
    return a, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluate_sequence
    value.  The routine uses a deterministic alternating LP + mixing
    scheme (described above) together with a final coordinate‑swap local
    search.  The best sequence found within the time budget is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0    # a tiny safety margin

    # ------------------------------------------------------------------
    # Load the previous construction if it exists
    try:
        base = np.array(height_sequence_1, dtype=float)          # type: ignore
    except Exception:
        base = None

    # ------------------------------------------------------------------
    # Global bookkeeping
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to obtain an initial vector of length n
    def _initial_vector(n):
        if base is not None and base.shape[0] == n:
            v = base.copy()
        elif base is not None:
            # linear interpolation from the old grid to the new one
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            v = np.interp(new_x, old_x, base)
        else:
            v = np.random.rand(n)
        v = np.maximum(v, 0.0)
        target = math.sqrt(2.0 * n)
        s = v.sum()
        if s > 0.0:
            v = v * (target / s)
        else:
            v = np.full(n, target / n)
        return v, target

    # ------------------------------------------------------------------
    # Choose a single length (the length of the supplied construction,
    # fall‑back to 2100) – longer lengths would require far more memory for the LP.
    n0 = base.shape[0] if base is not None else 2100
    lengths = [n0]                       # we keep it simple; more lengths can be added

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # A few independent restarts
        for restart in range(3):
            if time.time() >= deadline:
                break

            # ----- initial vector ------------------------------------------------
            a, _ = _initial_vector(n)

            # ----- main alternating optimisation loop ----------------------------
            outer_iter = 0
            while time.time() < deadline and outer_iter < 500:
                outer_iter += 1

                # current maximal convolution
                M, _, conv_full = _conv_max_fft(a)

                # record best solution seen so far
                if M < best_score:
                    best_score = M
                    best_seq = a.copy()
                    print(f"[len {n:5d} it {outer_iter:4d}] new best score = {best_score:.6f}")

                # solve the LP that adds as much extra mass as possible without
                # increasing any *active* convolution entry above M
                g = _solve_lp_max_sum(a, M, conv_full, target_sum)

                # convex combination – t controls step size
                t = 0.2
                a_new = (1.0 - t) * a + t * g
                # re‑normalise the L1‑norm (tiny numeric drift can appear)
                a_new = a_new * (target_sum / a_new.sum())

                # optional jitter every few iterations to escape plateaus
                if outer_iter % 40 == 0:
                    jitter = np.random.randn(n) * 0.001 * target_sum / n
                    a_new = np.maximum(a_new + jitter, 0.0)
                    a_new = a_new * (target_sum / a_new.sum())

                # accept the update only if it does not increase the maximal
                # convolution – this guarantees monotonic improvement.
                M_new, _, _ = _conv_max_fft(a_new)
                if M_new <= M:
                    a = a_new
                else:
                    # no improvement – keep the old vector and continue
                    pass

            # ----- cheap local refinement ----------------------------------------
            if time.time() < deadline:
                a, best_score = _local_search(
                    a_start=a,
                    n=n,
                    target_sum=target_sum,
                    best_score=best_score,
                    deadline=deadline,
                    max_proposals=20_000,
                )
                if best_score < best_score:   # (kept for consistency)
                    best_seq = a.copy()

    # ------------------------------------------------------------------
    # Fallback in the unlikely case that we never produced a feasible vector
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
