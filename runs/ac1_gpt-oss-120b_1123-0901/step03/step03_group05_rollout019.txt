import numpy as np, time, math

def conv_full(a):
    n = a.shape[0]
    N = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[:2*n - 1]
    return conv

def conv_full_two(x, y):
    n = len(x) + len(y) - 1
    N = 1 << (n.bit_length())
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(X * Y, n=N)[:len(x) + len(y) - 1]
    return conv

def _adam_l2_optimize(a_init, n, target_sum, best_seq, best_score, deadline, max_iter=20000, lr0=0.1):
    a = a_init.copy()
    # Normalize sum to target
    total = a.sum()
    if total > 0:
        a = a * (target_sum / total)
    else:
        a = np.full(n, target_sum / n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = conv_full(a)           # length 2n-1
        M = conv.max()
        # compute loss gradient for L2 flattening
        S = a.sum()
        avg = S * S / (2 * n - 1)
        C = conv - avg

        # gradient: 4 * convolution of C with reversed a
        grad_full = conv_full_two(C, a[::-1])
        grad = 4.0 * grad_full[n-1 : n-1 + n]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        lr = lr0 / (1 + 1e-4 * it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project to nonnegative simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # occasional jitter to escape plateaus
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # compute current metric (M = max conv)
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:6d}] new best score = {best_score:.6f}")

    return best_seq, best_score

def _local_search(a_start, n, target_sum, best_score, deadline, max_proposals=20000):
    a = a_start.copy()
    while time.time() < deadline and max_proposals > 0:
        max_proposals -= 1
        # choose donor and receiver
        i = np.random.randint(n)
        if a[i] <= 1e-14:
            continue
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        eps = a[i] * np.random.rand() * 0.1  # up to 10%
        if eps <= 0:
            continue
        new_a = a.copy()
        new_a[i] -= eps
        new_a[j] += eps
        # recompute max
        conv = conv_full(new_a)
        M = conv.max()
        if M < best_score:
            a = new_a
            best_score = M
            print(f"[local] improvement to {best_score:.6f} (remaining {max_proposals})")
    return a, best_score

"""
Gradient‑flattening optimiser for the self‑convolution bound.

The objective 2·n·max(a*a)/(sum a)² is invariant under scaling.
We therefore fix  Σa = √(2·n)  and minimise the maximal entry of the
auto‑convolution.  The algorithm minimises the L₂ distance of the
convolution to its average value (which is the theoretical lower bound
≈2·n/(2·n−1)).  This drives the whole convolution curve towards a flat
shape, consequently reducing the peak.  An Adam optimiser is used on
the L₂ loss; after the gradient phase a cheap hill‑climb moves small
amounts of mass between coordinates to improve the max‑value further.
Several lengths (2100 up to 8000) and a few random restarts are tried
within the time budget.  The best sequence found is returned as a plain
Python list.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return conv


def _conv_full_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two arbitrary vectors using FFT."""
    n = len(x) + len(y) - 1
    N = 1 << (n.bit_length())                    # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(X * Y, n=N)[:n]
    return conv


# ------------------------------------------------------------
def _adam_l2_optimize(
    a_init: np.ndarray,
    n: int,
    target_sum: float,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    max_iter: int = 20000,
    lr0: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the L₂ distance of the self‑convolution
    to its average value (i.e. flattens the whole curve).
    """
    a = a_init.copy()
    # normalise to the required total sum
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    else:
        a = np.full_like(a, target_sum / n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # ----- convolution and loss data -----
        conv = _conv_full(a)               # length 2n‑1
        M = conv.max()
        S = a.sum()
        avg = S * S / (2 * n - 1)          # theoretical average of the convolution
        C = conv - avg                     # deviation from the flat target

        # ----- gradient of the L₂ loss -----
        # grad_i = 4 * Σ_k C_k * a_{k-i}
        grad_full = _conv_full_two(C, a[::-1])          # length 3n‑2
        grad = 4.0 * grad_full[n - 1 : n - 1 + n]       # keep n components

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto the feasible set -----
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # occasional jitter to escape plateaus
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # ----- current score (identical to max‑convolution for the fixed sum) -----
        cur_score = 2.0 * n * M / (S * S)   # should be ≈ M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:6d}] new best score = {best_score:.6f}")

    return best_seq, best_score


# ------------------------------------------------------------
def _local_search(
    a_start: np.ndarray,
    n: int,
    target_sum: float,
    best_score: float,
    deadline: float,
    max_proposals: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb: move a small amount of mass from one index to another
    and keep the move only if it reduces the maximal convolution entry.
    """
    a = a_start.copy()
    proposals = 0
    while time.time() < deadline and proposals < max_proposals:
        proposals += 1
        i = np.random.randint(n)
        if a[i] <= 1e-14:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        eps = a[i] * np.random.rand() * 0.07  # up to 7 % of the donor mass
        if eps <= 0.0:
            continue

        new_a = a.copy()
        new_a[i] -= eps
        new_a[j] += eps

        # renormalise (necessary because of rounding)
        new_a = new_a * (target_sum / new_a.sum())

        M = _conv_full(new_a).max()
        cur_score = 2.0 * n * M / (target_sum * target_sum)   # sum is exactly target_sum
        if cur_score < best_score:
            a = new_a
            best_score = cur_score
            print(f"[local] improvement to {best_score:.6f} (prop {proposals})")
    return a, best_score
