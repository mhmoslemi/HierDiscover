Propose a low‑score step‑function.

    quickly becomes the true max.

*   The learning‑rate is damped both by a standard exponential decay
    and by the current λ (large λ ⇒ smaller steps).

*   When λ becomes large (≈8) we switch to a *hard‑max* weight vector,
    i.e. a normalized indicator of the maximal convolution entries.
    This yields the exact sub‑gradient of the objective.

*   After the Adam phase we perform a short mirror‑descent (exponentiated‑gradient)
    refinement on the hard‑max surrogate together with a tiny random‑shake
    to escape plateaus.

*   Finally a cheap random‑move local search tries to nudgedly improve the
    best candidate found.

The routine runs for at most ``budget_s`` seconds (default 1000) and
returns the best sequence (plain Python list) discovered.

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length ``size``.
    ``size`` is chosen as the next power of two if not supplied.
    """
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max weights w_k = exp(lam·(conv_k‑max(conv))) / Σ_j exp(...)."""
    m = conv.max()
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:                     # safety – should not happen
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / s
    return w


def _hardmax_weights(conv: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Indicator weights for the (almost) maximal entries of ``conv``.
    All entries within ``eps`` of the maximum receive equal weight.
    """
    mx = conv.max()
    mask = (conv >= mx - eps).astype(conv.dtype)
    total = mask.sum()
    if total == 0:
        mask = np.ones_like(conv) / conv.size
    else:
        mask = mask / total
    return mask


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, size_corr: int = None) -> np.ndarray:
    """
    Gradient of  Σ_k w_k (a*a)[k] w.r.t. ``a``.
    The explicit formula is 2·(w ★ a) where ★ denotes convolution with
    the reversed copy of ``a``.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    # slice that corresponds to the gradient entries 0 … n‑1
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


def _interpolate_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """
    Linear interpolation of a sequence to a new length.
    The original sequence is treated as periodic on [0,1).
    """
    old_len = seq.size
    if old_len == new_len:
        return seq.copy()
    x_old = np.linspace(0, 1, old_len, endpoint=False)
    x_new = np.linspace(0, 1, new_len, endpoint=False)
    return np.interp(x_new, x_old, seq)

        # --------------------------------------------------------------
        if best_seq is not None:
            a = best_seq.copy()
        else:
            a = np.full(n, 1.0 / n)

        for ref in range(300):
            # recompute convolution and hard‑max weights
            conv = _full_convolution(a, size=conv_fft_size)
            w = _hardmax_weights(conv, eps=1e-15)

            grad_a = _gradient_from_weights(a, w, size_corr=size_corr)

            # step size decays with the refinement iteration
            eta = 0.12 / (ref + 1.0)

            # exponentiated‑gradient update (keeps a on the simplex)
            a = a * np.exp(-eta * grad_a)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a /= s

            if ref % 30 == 0:
                val = evaluate_sequence(a.tolist())
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[Refine {restart:03d}] step {ref:03d}  new best = {val:.6f}")

        # --------------------------------------------------------------
        # Very cheap random‑move local search (1000 attempts max)
        # --------------------------------------------------------------
        a = best_seq.copy() if best_seq is not None else a
        n = a.size
        for _ in range(800):
            i, j = np.random.randint(0, n, size=2)
            if i == j or a[i] < 1e-12:
                continue
            delta = min(a[i] * 0.02, 0.01)   # move at most 2 % of source mass
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = np.maximum(a_new, 0.0)
            a_new /= a_new.sum()

            val = evaluate_sequence(a_new.tolist())
            if val < best_val:
                best_val = val
                best_seq = a_new.copy()
                a = a_new
                print(f"[Local {restart:03d}] improved to {val:.6f}")

        # end of restart ------------------------------------------------

    # ----------------------------------------------------------------------
    # Fallback – should never be needed
    if best_seq is None:
        best_seq = np.full(3000, 1.0 / 3000)

    print(f"Final best objective = {best_val:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
