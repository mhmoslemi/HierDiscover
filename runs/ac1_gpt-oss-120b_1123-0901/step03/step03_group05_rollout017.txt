"""
Search for a non‑negative step‑function minimizing ``evaluate_sequence``.

The algorithm works on a fixed L1‑norm (sum = sqrt(2·n)), because the
objective then equals the maximal value of the autocorrelation.
For a given vector ``f`` we compute its maximal autocorrelation ``M``.
We then solve a linear program that maximises the L1‑norm of a new
vector ``g`` under the *linear* constraints
    (f * g)[k] <= M   for every shift k,
i.e. we keep the autocorrelation bound while allowing ``g`` to carry as
much total mass as possible.  After the LP we rescale ``g`` to the
required sum and mix it with ``f`` (convex combination with step size *t*).
Repeating the LP–mix step monotonically reduces the maximal
autocorrelation.  The procedure is run for several lengths (interpolated
from the known construction and random restarts) and a cheap stochastic
hill‑climb is applied at the end.

The whole search respects the 1000 s time budget.
"""
import time
import math
import numpy as np

# SciPy is used for the linear programming step.
try:
    from scipy.optimize import linprog
except Exception as e:  # pragma: no cover
    raise RuntimeError("SciPy is required for this optimiser.") from e


def _conv_max_fft(a: np.ndarray):
    """
    Full autocorrelation of ``a`` via FFT.
    Returns (max_value, index_of_max, full_convolution_array).
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())       # next power of two
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]
    k = int(conv.argmax())
    return float(conv.max()), k, conv


def _rev_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    """
    For a given shift index ``k``, returns the vector
    rev where rev[i] = a[k-i] if that index is within the support,
    otherwise 0.  This is the row of the convolution matrix that
    corresponds to shift ``k``.
    """
    n = a.shape[0]
    rev = np.zeros_like(a)
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    rev[mask] = a[j[mask]]
    return rev


def _local_search(
    a_start: np.ndarray,
    n: int,
    target_sum: float,
    best_score: float,
    deadline: float,
    max_proposals: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb: repeatedly move a small amount of mass from one
    index to another (keeping the total sum constant).  Accept a move only
    if the maximal autocorrelation value decreases.
    """
    a = a_start.copy()
    proposal = 0
    while time.time() < deadline and proposal < max_proposals:
        proposal += 1
        i = np.random.randint(n)               # donor
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(n - 1)           # receiver (different)
        if j >= i:
            j += 1

        eps = a[i] * np.random.rand() * 0.05   # up to 5 % of donor
        if eps <= 0:
            continue

        new_vec = a.copy()
        new_vec[i] -= eps
        new_vec[j] += eps

        M_new, _, _ = _conv_max_fft(new_vec)
        if M_new < best_score:
            a = new_vec
            best_score = M_new
            print(f"[local] improvement to {best_score:.6f} (proposal {proposal})")
    return a, best_score


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative vector) minimising the
    ``evaluate_sequence`` objective.  The method combines an LP‑based
    alternating update with a final stochastic hill‑climb, trying several
    lengths and random restarts within the allotted time.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # ------------------------------------------------------------------
    # Load the previously discovered construction if present
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        base = np.ones(2100, dtype=float)

    n0 = base.shape[0]
    target_sum0 = math.sqrt(2.0 * n0)
    base = base * (target_sum0 / base.sum())
    init_score = evaluate_sequence(base.tolist())
    best_seq = base.copy()
    best_score = init_score
    print(f"[init] length={n0}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore (larger lengths give more freedom)
    length_factors = [1.0, 1.2, 1.4, 1.6, 2.0]
    max_len = 5000
    lengths = sorted({int(n0 * f) for f in length_factors})
    lengths = [l for l in lengths if 300 <= l <= max_len]
    lengths = sorted(lengths, reverse=True)   # start with the biggest

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # number of restarts per length
        restarts = 3

        for r in range(restarts):
            if time.time() >= deadline:
                break

            # ------------------------------------------------------------------
            # Initialise: either interpolation of the old construction or random
            if r == 0:
                old_x = np.linspace(0.0, 1.0, n0)
                new_x = np.linspace(0.0, 1.0, n)
                a = np.interp(new_x, old_x, base)
                a = np.maximum(a, 0.0)
            else:
                a = np.random.rand(n)            # Dirichlet–like start
                a = np.maximum(a, 0.0)

            a = a * (target_sum / a.sum())        # enforce the L1 norm

            # ------------------------------------------------------------------
            # Core LP‑alternating optimisation
            max_outer_iters = 40
            t = 0.2                               # mixing step size
            no_improve = 0

            for outer in range(max_outer_iters):
                if time.time() >= deadline:
                    break

                M, _, conv_full = _conv_max_fft(a)
                # active constraints – shifts that achieve the maximal value
                tol = 1e-12
                active = np.where(conv_full >= M - tol)[0]
                if active.size == 0:
                    active = np.arange(conv_full.size)

                # Build the (small) constraint matrix for the active shifts
                rows = np.vstack([_rev_for_shift(a, k) for k in active])
                b_ub = np.full(rows.shape[0], M, dtype=np.float64)

                # Linear programme: maximise sum(g)  <=> minimise -sum(g)
                c = -np.ones(n, dtype=np.float64)
                bounds = [(0.0, 1000.0) for _ in range(n)]

                lp_res = linprog(
                    c,
                    A_ub=rows,
                    b_ub=b_ub,
                    bounds=bounds,
                    method='highs',
                    options={"presolve": True},
                )
                if not lp_res.success:
                    # If LP fails, abort this restart
                    print(f"[lp] failed (n={n}, outer={outer}): {lp_res.message}")
                    break

                g0 = lp_res.x
                sum_g0 = g0.sum()
                if sum_g0 <= 0:
                    break

                # Rescale to the required L1 norm
                g = g0 * (target_sum / sum_g0)

                # Evaluate the new candidate
                M_g, _, _ = _conv_max_fft(g)

                # Global best update
                if M_g < best_score:
                    best_score = M_g
                    best_seq = g.copy()
                    print(f"[len {n:5d}] NEW GLOBAL BEST = {best_score:.6f}")

                # Decide how to update the current iterate
                if M_g < M - 1e-12:
                    a_new = g
                    new_M = M_g
                else:
                    a_new = (1.0 - t) * a + t * g
                    a_new = a_new * (target_sum / a_new.sum())
                    new_M, _, _ = _conv_max_fft(a_new)

                # Accept if we made progress
                if new_M < M - 1e-12:
                    a = a_new
                    M = new_M
                    no_improve = 0
                else:
                    no_improve += 1
                    if no_improve >= 5:
                        t = max(0.05, t * 0.5)   # shrink step size if stuck
                        no_improve = 0

                # Small jitter every few iterations to escape plateaus
                if outer % 5 == 0:
                    jitter = np.random.randn(n) * 0.001 * target_sum / n
                    a = np.maximum(a + jitter, 0.0)
                    a = a * (target_sum / a.sum())

                # Early exit if we already beat the target bound
                if best_score < 1.503:
                    break

            # ------------------------------------------------------------------
            # After the LP loop run a cheap local hill‑climb on the best found
            # vector for this length (if time permits)
            if time.time() < deadline:
                best_seq, best_score = _local_search(
                    a_start=best_seq,
                    n=n,
                    target_sum=target_sum,
                    best_score=best_score,
                    deadline=deadline,
                    max_proposals=20_000,
                )
                print(f"[len {n:5d}] post‑local best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final cheap hill‑climb on the overall best (if any time left)
    if best_seq is not None and time.time() < deadline:
        n_best = best_seq.shape[0]
        best_seq, best_score = _local_search(
            a_start=best_seq,
            n=n_best,
            target_sum=math.sqrt(2.0 * n_best),
            best_score=best_score,
            deadline=deadline,
            max_proposals=30_000,
        )
        print(f"[final] final best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen, but keep a safe uniform vector
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
