def subgradient_refine(a, target_sum, deadline, max_steps=50000):
    n = len(a)
    Nfft = 1 << ((2 * n - 1).bit_length())
    coeff = 4.0 * n / (target_sum ** 2)
    step = 0.1 * target_sum / n
    for k in range(max_steps):
        if time.time() >= deadline:
            break
        # compute conv
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = conv.max()
        # find active indices
        mask = (conv >= M - 1e-12)
        w = mask.astype(float)
        w = w / w.sum()
        # weighted rev
        rev = _weighted_rev(a, w, Nfft)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const
        # simple gradient step
        a = a - step * grad
        # projection
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)
        # optional adaptive step reduction
        if (k+1) % 1000 == 0:
            step *= 0.9
        # update best maybe...
    return a

"""
Search for a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. Load the previously discovered construction ``height_sequence_1`` (if it
   exists) and use it as a base shape.
2. Consider several lengths (original, 1.5×, 2×, 3× and 4× up to 8000).
   For each length we generate a few simple initialisations
   (uniform, sine‑shaped, triangular, random) and also an interpolated
   copy of the base shape.
3. Every candidate is optimised with a **projected Adam** optimiser that
   minimises a smooth approximation of ``max(a∗a)``.
   * The smooth max is the soft‑max
     `Lτ = τ·log ∑exp(conv/τ)`.
   * The gradient of `Lτ` w.r.t. the vector is obtained by one FFT
     (weighted reverse convolution).
   * After each update the vector is projected onto the feasible set:
     non‑negative entries and the L1‑norm fixed to √(2 n).
   * Temperature τ starts as a fraction of the current maximum and
     decays exponentially.
4. When the temperature is very small we perform a few **sub‑gradient**
   steps on the exact max (weights are uniform on the active set of
   maximal convolution entries) – again with projection.
5. A short **stochastic hill‑climbing** phase moves a tiny amount of mass
   from one position to another; moves are kept only if they improve the
   exact objective.
6. The best vector seen (according to ``evaluate_sequence``) over all
   runs is returned.  The whole procedure stops before the 1000 s
   wall‑clock budget expires.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # not used in the final version but kept for safety

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """
    Compute the vector   rev_i = Σ_k w_k a_{k‑i}
    using FFTs (the linear map w → rev is the transpose of the
    convolution a∗·).
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _proj_feasible(a: np.ndarray, target_sum: float):
    """Project onto a≥0 and Σa = target_sum."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    else:
        a = np.full_like(a, target_sum / a.shape[0])
    return a


def _max_conv(a: np.ndarray, Nfft: int):
    """Return max_{k} (a∗a)[k] using a single FFT."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max()), conv


# ----------------------------------------------------------------------
# Initialisers for the search
# ----------------------------------------------------------------------


def init_uniform(n: int):
    """All ones – the scale will be fixed later."""
    return np.ones(n, dtype=float)


def init_random(n: int):
    """Uniform random numbers in [0,1)."""
    return np.random.rand(n)


def init_sine(n: int):
    """A half‑period sine wave, always non‑negative."""
    x = np.arange(n, dtype=float)
    a = np.sin(math.pi * (x + 0.5) / n)
    return a


def init_triangular(n: int):
    """Symmetric triangular shape, peak = 1."""
    i = np.arange(n, dtype=float)
    a = 1.0 - np.abs((i - (n - 1) / 2.0) / ((n + 1) / 2.0))
    a = np.maximum(a, 0.0)
    return a


def init_gaussian(n: int):
    """Gaussian bump, width chosen for a fairly smooth shape."""
    i = np.arange(n, dtype=float)
    sigma = n / 6.0
    a = np.exp(-0.5 * ((i - (n - 1) / 2.0) / sigma) ** 2)
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def adam_optimize(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Projected Adam optimiser on the smooth max (soft‑max) of the
    self‑convolution.  Returns the best vector seen and its exact
    max‑convolution value.
    """
    n = a0.shape[0]
    a = a0 * (target_sum / a0.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.05
    iteration = 0
    jitter_interval = 5000

    # temperature parameters
    tau_factor = 0.20            # initial τ = τ_factor·M
    decay_rate = 20000.0        # controls exponential decay of τ

    # FFT size (next power‑of‑two ≥ 2·n‑1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    best_a = a.copy()
    best_val = float("inf")

    while time.time() < deadline:
        # ----- exact convolution (needed for the current max) -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        if M < best_val:
            best_val = M
            best_a = a.copy()
            print(f"[adam n={n:5d} iter={iteration:8d}] new local best = {best_val:.6f}")

        # ----- soft‑max smoothing of the max -----
        tau = max(1e-12, tau_factor * M * math.exp(-iteration / decay_rate))
        w, Ltau = _softmax_weights(conv, tau)

        # ----- weighted reverse convolution (gradient of Lτ) -----
        rev = _weighted_rev(a, w, Nfft)

        # ----- Adam update (including the Lagrange term for the sum) -----
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto the feasible set -----
        a = _proj_feasible(a, target_sum)

        # ----- occasional jitter to shake off plateaus -----
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = _proj_feasible(a + jitter, target_sum)

        # limit the number of iterations to avoid infinite loops
        if iteration > 10_000_000:
            break

    return best_a, best_val


def subgradient_refine(a: np.ndarray, target_sum: float, deadline: float,
                       max_steps: int = 30_000):
    """
    Plain sub‑gradient descent on the exact max (a∗a) after the temperature
    has vanished.  Uses the same projection as the Adam phase.
    """
    n = a.shape[0]
    Nfft = 1 << ((2 * n - 1).bit_length())
    coeff = 4.0 * n / (target_sum * target_sum)
    step = 0.5 * target_sum / n        # initial step size

    for step_no in range(max_steps):
        if time.time() >= deadline:
            break

        M, conv = _max_conv(a, Nfft)
        # active set – all indices whose value is (almost) maximal
        mask = conv >= M - 1e-12
        w = mask.astype(float)
        w /= w.sum()                     # uniform weight on the active set

        rev = _weighted_rev(a, w, Nfft)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        a = a - step * grad
        a = _proj_feasible(a, target_sum)

        # geometric decay of the step size
        if (step_no + 1) % 1000 == 0:
            step *= 0.9

    return a


def hill_climb(a: np.ndarray, target_sum: float, deadline: float,
               max_moves: int = 20_000):
    """
    Very cheap stochastic hill‑climbing: repeatedly move a tiny amount of
    mass from one index to another and keep the move if it improves the
    exact objective.
    """
    n = a.shape[0]
    best = a.copy()
    best_val = _max_conv(best, 1 << ((2 * n - 1).bit_length()))[0]

    for _ in range(max_moves):
        if time.time() >= deadline:
            break

        i, j = np.random.choice(n, 2, replace=False)
        # propose a move of size delta (relative to the average entry)
        delta = (target_sum / n) * (np.random.randn() * 0.02)
        if delta > 0:
            delta = min(delta, best[i])             # cannot make i negative
            new_i = best[i] - delta
            new_j = best[j] + delta
        else:
            delta = min(-delta, best[j])
            new_i = best[i] + delta
            new_j = best[j] - delta

        cand = best.copy()
        cand[i] = new_i
        cand[j] = new_j
        cand = _proj_feasible(cand, target_sum)

        cand_val = _max_conv(cand, 1 << ((2 * n - 1).bit_length()))[0]
        if cand_val < best_val:
            best = cand
            best_val = cand_val

    return best
