def project_onto_simplex(v, target):
    # ensure v is array
    v = np.asarray(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
* Load the known construction (``height_sequence_1``) if it exists,
  otherwise start from a uniform vector.
* Search several lengths (original, 1.25×, 1.5× and 2×) – up to 8000 entries.
* For every length run a few *restarts*:
    – initialise either by interpolating the best known shape or by a random
      positive vector.
    – optimise the smooth approximation of the max self‑convolution
      (soft‑max temperature) using Adam.
    – after each gradient step project onto the feasible simplex
      ``{a≥0, Σa = √(2 n)}`` with the exact Euclidean projection (instead of a
      simple scaling).  This gives a more accurate handling of the sum‑constraint.
    – temperature τ starts as a fraction of the current max and decays
      exponentially; jitter is added occasionally to escape plateaus.
* The best sequence over *all* runs (measured with the exact objective)
  is returned.
* Execution respects a 1000 s wall‑clock limit.
"""

import time
import math
import numpy as np


def _softmax_weights(conv: np.ndarray, tau: float):
    """
    Stable soft‑max of the convolution values.
    Returns the weight vector w (summing to 1) and the smoothed maximum Lτ.
    """
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """
    Compute rev_i = Σ_k w_k a_{k-i} (out‑of‑range indices are ignored)
    using FFTs.
    """
    # forward FFT of a (zero‑padded)
    A_fft = np.fft.rfft(a, n=Nfft)
    # FFT of the reversed weight vector
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    # convolution of w_rev with a
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + len(a) - 1]          # keep the valid part
    # pick the required slice (see analysis notes)
    idx_start = 2 * len(a) - 2
    rev = conv_rev[idx_start - np.arange(len(a))]
    return rev


def _project_onto_simplex(v: np.ndarray, target: float):
    """
    Euclidean projection of v onto the simplex
        S = {x | x_i ≥ 0, Σ x_i = target }.
    Implements the algorithm of Duchi et al. (2008).
    """
    if target <= 0:
        return np.zeros_like(v)
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _optimize_one_run(
    a0: np.ndarray,
    n: int,
    global_best_seq: np.ndarray,
    global_best_score: float,
    deadline: float,
    max_iters: int,
):
    """
    Adam optimiser for a single length / initialisation.
    Returns updated global best sequence and score.
    """
    target_sum = math.sqrt(2.0 * n)
    a = a0.copy()
    a = _project_onto_simplex(a, target_sum)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.05
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule
    tau_factor = 0.30          # initial τ = τ_factor·M
    decay_rate = 30000.0       # slower decay for more stable convergence

    jitter_interval = 4000
    iteration = 0

    # FFT size (next power‑of‑two ≥ 2·n‑1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline and iteration < max_iters:
        # 1) exact self‑convolution via FFT
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # 2) smooth max via soft‑max
        tau = max(1e-12, tau_factor * M * math.exp(-iteration / decay_rate))
        w, Ltau = _softmax_weights(conv, tau)

        # 3) sub‑gradient of the smoothed max
        rev = _weighted_rev(a, w, Nfft)

        # 4) gradient of the normalised objective
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        # 5) Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # 6) projection onto the feasible simplex
        a = _project_onto_simplex(a, target_sum)

        # 7) occasional jitter to shake off plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = a + jitter
            a = _project_onto_simplex(a, target_sum)

        # 8) exact objective value (after normalisation it equals M)
        cur_score = M
        if cur_score < global_best_score:
            global_best_score = cur_score
            global_best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:8d}] new global best = {global_best_score:.6f}")

    return global_best_seq, global_best_score
