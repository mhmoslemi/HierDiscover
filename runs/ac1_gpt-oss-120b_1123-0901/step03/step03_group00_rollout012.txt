"""
Propose a step‑function (non‑negative vector) that minimizes

        F(a) = 2·n·max(conv(a,a)) / (sum(a)²)

where conv is the linear convolution (np.convolve).  

The optimizer follows a two‑phase scheme:

* Phase‑1: Adam on the logits of a soft‑max (guaranteeing a_i≥0,
  Σa_i=1).  The loss is a smooth‑max approximation
  φ_λ(conv) = (1/λ)·log Σ_k exp(λ·(conv_k – max(conv))) .
  λ is increased geometrically so that the surrogate converges to the
  true maximum.

* Phase‑2 (when λ is large): replace the smooth‑max by the exact
  sub‑gradient of the max (i.e. use a hard indicator on the maximal
  convolution indices).  This sharpens the descent near the optimum.

Several lengths (256 … 16384) are tried, each with a few random
restarts, warm‑starts from the best‑so‑far and a simple exponential
seed.  The best sequence found within the time budget (≈1000 s) is
returned as a plain Python list.  Small “shakes’’ are injected if an
iteration stalls, and a short local random‑move refinement is applied
at the very end.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a * a using FFT (full length 2·len(a)‑1)."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i + k – (n‑1)],  i=0…n‑1.
    Implements the linear cross‑correlation needed for the gradient.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


def _objective(a: np.ndarray) -> float:
    """
    Exact evaluation of the objective F(a) for a vector a.
    (The sum of a is assumed to be non‑zero.)
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_c = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (sum_a ** 2))


def _logits_from_probs(p: np.ndarray) -> np.ndarray:
    """Return stable logits for a probability vector."""
    return np.log(p + 1e-15)


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def _subgradient_weights(conv: np.ndarray, lam: float,
                         lam_switch: float, eps: float = 1e-12) -> np.ndarray:
    """
    Compute the weight vector w used in the gradient.
    For small lam – a smooth softmax; for large lam – hard indicator.
    """
    max_c = conv.max()
    if lam < lam_switch:
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:                     # numerical guard
            w = np.ones_like(conv) / conv.size
        else:
            w /= w_sum
    else:
        # hard indicator on (almost) maximal entries
        mask = conv >= max_c - eps
        cnt = mask.sum()
        if cnt == 0:                         # should not happen
            w = np.ones_like(conv) / conv.size
        else:
            w = np.zeros_like(conv, dtype=np.float64)
            w[mask] = 1.0 / cnt
    return w


def _optimize_length(n: int,
                     budget: float,
                     best_sofar: np.ndarray = None,
                     seed: int = 0) -> tuple[np.ndarray, float]:
    """
    Run several restarts of the Adam / sub‑gradient optimiser for a
    fixed length n.  Returns (best_seq, best_val) found for this n.
    """
    np.random.seed(seed)
    deadline = time.time() + budget

    # FFT sizes (powers of two) – pre‑computed for speed
    fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))   # (2n‑1) + n – 1

    best_val = float("inf")
    best_seq = None

    # Number of independent restarts – more for small n, less for big n
    restarts = 5 if n <= 2048 else 3

    for r in range(restarts):
        if time.time() > deadline:
            break

        # ------- initialise logits ---------------------------------
        if r == 0 and best_sofar is not None and best_sofar.shape[0] == n:
            init_a = best_sofar.copy()
            init_a /= init_a.sum()
            x = _logits_from_probs(init_a)
            print(f"[n={n} restart={r}] warm start from previous best")
        elif r == 1 and 'height_sequence_1' in globals():
            # use the historic construction (trim / pad)
            cand = np.array(height_sequence_1, dtype=np.float64)
            if cand.shape[0] >= n:
                cand = cand[:n]
            else:
                cand = np.pad(cand, (0, n - cand.shape[0]))
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s < 1e-6:
                cand = np.random.dirichlet(np.ones(n))
            else:
                cand /= s
            x = _logits_from_probs(cand)
            print(f"[n={n} restart={r}] start from height_sequence_1")
        elif r == 2:
            # exponential (geometric) seed
            decay = 0.999
            p = np.power(decay, np.arange(n, dtype=np.float64))
            p /= p.sum()
            x = _logits_from_probs(p)
            print(f"[n={n} restart={r}] geometric seed")
        else:
            # random Dirichlet start
            p = np.random.dirichlet(np.ones(n))
            x = _logits_from_probs(p)
            print(f"[n={n} restart={r}] random start")

        # Adam state
        m = np.zeros_like(x)
        v = np.zeros_like(x)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lr = 0.05

        lam = 0.5                # temperature for smooth‑max
        lam_factor = 1.02
        lam_switch = 200.0       # after this we use the hard indicator

        t = 0
        max_iters = 60000
        last_improve = time.time()
        # main optimisation loop
        while time.time() < deadline and t < max_iters:
            # ----- logits → probabilities (softmax) -----
            a = _softmax(x)                 # Σa = 1, a_i ≥ 0

            # ----- convolution -----
            conv = _full_convolution(a, fft_len=fft_len_conv)
            max_c = conv.max()

            # ----- weights for gradient (smooth‑max or hard) -----
            w = _subgradient_weights(conv, lam, lam_switch)

            # ----- gradient w.r.t a -----
            grad_a = 2.0 * _cross_correlation(w, a, fft_len=fft_len_corr)

            # ----- back‑prop through softmax (gradient w.r.t. logits) -----
            avg = np.dot(a, grad_a)
            grad_x = a * (grad_a - avg)

            # ----- Adam update -----
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

            # ----- temperature schedule -----
            lam = min(lam * lam_factor, 1e6)   # cap to avoid overflow

            # ----- exact objective (for bookkeeping) -----
            val = 2.0 * n * max_c / (a.sum() ** 2)   # a.sum()≈1
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[n={n} restart={r} iter={t:05d}] new best: {val:.6f}")

            # ----- stagnation handling -----
            if time.time() - last_improve > 5.0:
                # small random perturbation of logits
                x += (np.random.rand(n) - 0.5) * 1e-2
                last_improve = time.time()
                print(f"[n={n} restart={r} iter={t}] shake (stagnation)")

        # ----- optional short local refinement (random moves) -----
        if best_seq is not None and time.time() < deadline - 1.0:
            a = best_seq.copy()
            cur_val = best_val
            step_size = 0.02
            for _ in range(2000):
                i, j = np.random.choice(n, 2, replace=False)
                if a[i] < 1e-12:
                    continue
                delta = step_size * a[i]          # move a fraction of i to j
                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                new_val = _objective(a_new)
                if new_val < cur_val:
                    a = a_new
                    cur_val = new_val
                    best_seq = a_new.copy()
                    best_val = new_val
                step_size *= 0.9999   # slowly anneal
                if time.time() > deadline:
                    break

    return best_seq, best_val


def propose_candidate(seed: int = 0, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the given time budget, prints progress,
    and finally returns the best sequence found as a Python list.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    np.random.seed(seed)

    best_seq = None
    best_val = float("inf")

    # Explore several lengths (larger n offers more freedom)
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]

    # remaining budget split equally among lengths (roughly)
    ramaining_lengths = len(lengths)
    for n in lengths:
        if time.time() > deadline:
            break
        # allocate a slice of the remaining budget
        now = time.time()
        remaining = deadline - now
        # give at most 1/3 of the remaining time to the current length
        length_budget = min(remaining / (ramaining_lengths),
                            remaining * 0.33)
        ramaining_lengths -= 1

        print(f"\n=== Optimising length n={n} (budget {length_budget:.1f}s) ===")
        seq, val = _optimize_length(n, length_budget,
                                    best_sofar=best_seq,
                                    seed=seed + n)
        if seq is not None and val < best_val:
            best_val = val
            best_seq = seq
            print(f"*** New global best for n={n}: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded (very unlikely) return a uniform vector.
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = _objective(best_seq)
        print(f"Fallback sequence (uniform) with value {best_val:.6f}")

    # The evaluator enforces non‑negativity and caps values, but we keep the
    # sequence clean: remove numerical noise, ensure sum>0.01.
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total < 0.01:
        # renormalise to a tiny uniform level
        best_seq = np.full_like(best_seq, 1e-4)
    else:
        best_seq = best_seq / total * total   # keep exact sum (no change)

    print(f"\nFinal best value (approx.) : {best_val:.6f}")
    return best_seq.tolist()
