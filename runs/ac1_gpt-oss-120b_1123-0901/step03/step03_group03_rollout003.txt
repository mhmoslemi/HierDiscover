"""
Propose a non‑negative step‑function (vector) that minimises
``evaluate_sequence``.

Algorithm
---------
* The objective after normalising Σa = √(2 n) is simply
  max (a * a) (the peak of the self‑convolution).
* We perform a projected Adam optimisation on a *smooth* surrogate of
  the max, namely the log‑sum‑exp (soft‑max) of the convolution values.
* The gradient of the smooth objective is obtained analytically by a
  single FFT at each iteration (the weighted reverse‑convolution).
* After the Adam step we project onto the feasible set:
  a_i ≥ 0 and Σa_i = √(2 n) (scaling the vector).
* Multiple lengths are explored (original, 1.25×, 1.5×, 2×) and for each
  length several restarts are performed (interpolated base shape and
  random initialisations).  A time‑budgeted loop keeps the total wall‑clock
  under the given limit.
* The best vector seen (measured by the exact ``evaluate_sequence``) is
  returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are top‑level (no closures)
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, tau: float):
    """
    Stable soft‑max of the convolution values.
    Returns:
        w      – weight vector (sums to 1)
        Ltau   – the smooth approximation τ·log∑exp(conv/τ)
    """
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    # subtract the maximum for numerical stability
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """
    Compute the vector
        rev_i = Σ_k w_k a_{k‑i}
    (indices outside the support of ``a`` are ignored) using FFTs.
    """
    n = a.shape[0]
    # FFT of a (already zero‑padded to length Nfft)
    A_fft = np.fft.rfft(a, n=Nfft)
    # FFT of the reversed weight vector
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    # Convolution of w_rev with a
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # Keep only the valid part (length = len(w) + n – 1)
    conv_rev = conv_rev[: len(w) + n - 1]
    # The slice that corresponds to rev_i (i = 0…n‑1)
    idx_start = 2 * n - 2
    rev = conv_rev[idx_start - np.arange(n)]
    return rev


def _optimize_length(
    n: int,
    init_a: np.ndarray,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    max_iters: int,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the *smooth* maximal self‑convolution.
    The exact objective value (the peak of a*a) is used to update the
    global best.
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise and enforce the L1‑norm
    a = init_a.copy()
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.3                     # initial learning rate (will be decayed)
    decay_iter = 2e6              # decay speed of the learning rate
    jitter_interval = 3000        # how often to add a tiny random jitter
    jitter_scale = 1e-3           # relative amplitude of jitter

    # temperature schedule parameters
    init_tau_factor = 0.4         # start with a fairly smooth landscape
    tau_decay = 6e4               # slower exponential decay

    # FFT size (next power‑of‑two ≥ 2·n‑1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    for it in range(1, max_iters + 1):
        # Periodically check the time budget (cheap check)
        if (it & 1023) == 0 and time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # 1) exact self‑convolution via FFT (full length 2·n‑1)
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # --------------------------------------------------------------
        # 2) soft‑max smoothing of the peak
        tau = max(1e-12, init_tau_factor * M * math.exp(-it / tau_decay))
        w, Ltau = _softmax_weights(conv, tau)

        # --------------------------------------------------------------
        # 3) weighted reverse convolution – gradient of the smoothed max
        rev = _weighted_rev(a, w, Nfft)

        # --------------------------------------------------------------
        # 4) centre the gradient (removes the component that would change the sum)
        rev_centered = rev - rev.mean()

        # --------------------------------------------------------------
        # 5) Adam update (descent direction = rev_centered)
        m = beta1 * m + (1.0 - beta1) * rev_centered
        v = beta2 * v + (1.0 - beta2) * (rev_centered * rev_centered)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + it / decay_iter)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # 6) project onto the feasible set (a ≥ 0, Σa = √(2 n))
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            # fallback to uniform if everything collapsed to zero
            a = np.full_like(a, target_sum / n)

        # --------------------------------------------------------------
        # 7) occasional jitter to escape plateaus
        if (it % jitter_interval) == 0:
            jitter = np.random.randn(n) * (jitter_scale * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # --------------------------------------------------------------
        # 8) exact objective value (after normalisation it equals M)
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[len {n:5d} iter {it:9d}] new best = {best_score:.6f}"
            )

    return best_seq, best_score
