"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. Try a few simple parametric families (exponential, triangular,
   uniform) at several lengths (original length and a few multiples).
   The best of these coarse candidates is used as a warm start.

2. Refine the warm start with a smooth‐approximation of
   ``max(convolution)``.
   – The smoothed objective is
        fτ(a) = τ·log(∑k exp(convk/τ))   (τ → 0  ⇒  max).
   – Its gradient can be written as a convolution, which we evaluate
     efficiently with FFTs.
   – Adam is used for the optimisation, a symmetry projection
     ``a ← (a + a[::-1])/2`` is applied after each step and the vector
     is re‑normalised to keep the L1 norm fixed (scale‑invariance of the
     original objective means any fixed sum works; we use √(2·n)).
   – τ is decreased geometrically during the run (simulated annealing).

3. After the Adam phase a cheap stochastic local search swaps a tiny
   amount of mass between two random positions whenever it improves the
   true max‑convolution.

All operations are done with NumPy and use only a few FFTs per
iteration, so the method fits comfortably into the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a (no FFT scaling), length = 2·n‑1.
    Uses FFT with correct normalisation (numpy's inverse divide by N).
    """
    n = a.shape[0]
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N) * N      # undo the 1/N from irfft
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _grad_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a  (∑k w_k·conv_k) where w is a probability vector (sum=1)
    and conv_k = ∑i a_i a_{k-i}.  The gradient equals the convolution of
    a with w reversed.
    """
    n = a.shape[0]
    L = 2 * n - 1                      # length of w
    w_rev = w[::-1]
    M = _next_pow2(n + L - 1)          # size for a⋆w_rev
    A = np.fft.rfft(a, n=M)
    W = np.fft.rfft(w_rev, n=M)
    conv_rev = np.fft.irfft(A * W, n=M) * M
    conv_rev = conv_rev[: n + L - 1]   # length = 3·n‑2
    # gradient_i = conv_rev[(L‑1)‑i]
    idx = (L - 1) - np.arange(n)
    return conv_rev[idx]


# ----------------------------------------------------------------------
def _smooth_adam_optimize(init_a: np.ndarray,
                          n: int,
                          target_sum: float,
                          deadline: float,
                          max_iters: int = 200_000,
                          tau_start: float = 0.7,
                          tau_end: float = 1e-4) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth approximation τ·log‑sum‑exp of the
    convolution max.  Returns the best vector (according to the true max)
    seen before *deadline*.
    """
    a = init_a.copy()
    # normalise sum (scale‑invariant problem, but fixing a sum stabilises)
    if a.sum() > 0:
        a *= (target_sum / a.sum())
    else:
        a.fill(target_sum / n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.05

    # tau schedule – geometric decay
    tau = tau_start
    decay_factor = (tau_end / tau_start) ** (1.0 / max_iters)

    best_a = a.copy()
    best_val = _conv_full(a).max()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # full convolution (true, unscaled)
        conv = _conv_full(a)
        maxc = conv.max()

        # soft‑max weights
        # (conv - maxc) avoids overflow
        w_raw = np.exp((conv - maxc) / tau)
        w = w_raw / w_raw.sum()

        # gradient of the smooth objective
        grad = _grad_softmax(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = base_lr / (1.0 + 1e-4 * it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # enforce symmetry, non‑negativity and the L1 constraint
        a = (a + a[::-1]) * 0.5
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a *= (target_sum / s)
        else:
            a.fill(target_sum / n)

        # keep the best according to the *true* max‑convolution
        cur_val = conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[len {n:5d} it {it:7d}] new best max = {best_val:.6f}")

        # anneal τ
        tau = max(tau * decay_factor, tau_end)

        # occasional jitter to escape plateaus
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 5e-4 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= (target_sum / a.sum())

    return best_a, best_val


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  steps: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: move a tiny amount of mass from
    one index to another if it reduces the true max‑convolution.
    """
    n = a.shape[0]
    best = a.copy()
    best_val = _conv_full(best).max()

    for _ in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j or best[i] <= 0.0:
            continue

        delta = best[i] * 0.02      # move 2 % of the donor mass
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        # keep sum unchanged → no re‑normalisation needed
        val = _conv_full(cand).max()
        if val < best_val:
            best = cand
            best_val = val

    return best, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns the best vector found before the time budget expires.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0            # safety margin

    # ------------------------------------------------------------------
    # 1) base candidate – either the supplied construction or a simple
    #    uniform start.
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        print("[info] loaded previous construction, length =", base.shape[0])
    except Exception:
        base = np.full(2100, 1.0, dtype=float)
        print("[info] no previous construction – using uniform start")

    # standardise sum to √(2·n) (any fixed sum works because the metric is
    # scale‑invariant)
    n0 = base.shape[0]
    target_sum0 = math.sqrt(2.0 * n0)
    if base.sum() > 0:
        base *= (target_sum0 / base.sum())
    else:
        base.fill(target_sum0 / n0)

    best_seq = base.copy()
    best_val = _conv_full(best_seq).max()
    print(f"[init] length={n0}, initial max = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) decide which lengths to explore
    length_factors = [1.0, 1.25, 1.5, 2.0, 3.0, 4.0]
    lengths = sorted({max(300, min(12000, int(n0 * f))) for f in length_factors})
    print("[info] exploring lengths:", lengths)

    # ------------------------------------------------------------------
    # 3) main optimisation loop
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # a few restarts per length
        restarts = 3 if n <= 4000 else 2
        for r in range(restarts):
            if time.time() >= deadline:
                break

            # ----------------------------------------------------------------
            # initialise a starting vector
            if n == n0 and r == 0:
                init = base.copy()
            else:
                # random Dirichlet‑like start
                rand = np.random.exponential(scale=1.0, size=n)
                init = rand * (target_sum / rand.sum())

            # blend a little of the current global best if shapes match
            if best_seq.shape[0] == n:
                mix = 0.15
                init = mix * best_seq + (1 - mix) * init
                init = np.maximum(init, 0.0)
                init *= (target_sum / init.sum())

            print(f"[run] length={n}, restart={r+1}/{restarts}")

            # ----------------------------------------------------------------
            # allocate a slice of the remaining budget for this run
            time_left = deadline - time.time()
            max_time = min(180.0, time_left * 0.45)        # ≤180 s per run
            sub_deadline = time.time() + max_time

            # ----------------------------------------------------------------
            # smooth Adam optimisation
            opt_seq, opt_val = _smooth_adam_optimize(
                init, n, target_sum, sub_deadline,
                max_iters=150_000,
                tau_start=0.8,
                tau_end=1e-5
            )

            # ----------------------------------------------------------------
            # cheap local improvement
            if time.time() < deadline:
                ls_seq, ls_val = _local_search(
                    opt_seq, steps=30_000, deadline=deadline
                )
                if ls_val < opt_val:
                    opt_seq, opt_val = ls_seq, ls_val

            # ----------------------------------------------------------------
            # update the global best
            if opt_val < best_val:
                best_val = opt_val
                best_seq = opt_seq.copy()
                print(f"[global] new best max = {best_val:.6f} (len={n})")

    # ------------------------------------------------------------------
    # 4) final sanity check with the official evaluator
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[done] best max = {best_val:.6f}, official score = {final_score:.6f}, "
          f"length = {best_seq.shape[0]}")
    return best_seq.tolist()
