def solve_optimal_sequence(n, deadline):
    target_sum = math.sqrt(2.0*n)
    a = cp.Variable(n)
    t = cp.Variable()
    constraints = [a >= 0, cp.sum(a) == target_sum, t >= 0]

    # start with constraints for k = n-1 (center) maybe also extremes.
    active_ks = [n-1]  # central shift gives full overlap

    C_mats = get_Ck(n)  # precompute all Ck as sparse

    for k in active_ks:
        constraints.append(cp.quad_form(a, C_mats[k]) <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)

    while time.time() < deadline:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, abstol=1e-9, reltol=1e-9) # adjust
        if prob.status != cp.OPTIMAL and prob.status!='optimal_inaccurate':
            print("Solver status:", prob.status)
            break
        a_val = np.array(a.value).flatten()
        conv_full = np.convolve(a_val, a_val)
        M = conv_full.max()
        k_max = conv_full.argmax()
        if M - t.value <= 1e-6 * max(1.0, M):
            break  # converged
        # add new constraint if not already present
        if k_max not in active_ks:
            active_ks.append(k_max)
            constraints.append(cp.quad_form(a, C_mats[k_max]) <= t)
            # need to create new problem? cvxpy may need rebuilding problem with new constraints
            prob = cp.Problem(cp.Minimize(t), constraints)
    # result: a_val, t_val
    return a_val, t.value

A convex‑combination / cutting‑plane optimiser for the max‑autoconvolution
objective.

The algorithm works for a fixed length `n`:

1.  Initialise a feasible vector `f` (non‑negative, L1‑norm = √(2 n)).
2.  Repeatedly:
        • Build the linear map `A` such that `A @ g = np.convolve(f, g)`.
        • Solve the linear programme
                maximise   ∑ g_i
                subject to A @ g ≤ M    (M = max(f∗f)),
                           g ≥ 0,
                           g_i ≤ √(2 n)          (bounds keep the LP bounded)
          The solution `g` is a promising partner sequence.
        • Perform a ternary line‑search on the mixing weight α∈[0,1] for
          `v = (1‑α)·f + α·g`.  After each trial `v` is projected back onto the
          feasible set (non‑negative, correct L1‑norm).  The α giving the smallest
          max‑convolution `max(v∗v)` is kept.
        • If the new maximum is lower than the previous `M`, replace `f` by `v`
          and continue; otherwise stop (local optimum reached).
3.  A cheap hill‑climb moves a tiny amount of mass from one index to another
    whenever it reduces the maximal convolution.
4.  The whole procedure is run for several lengths (derived from the provided
    `height_sequence_1`) and a few random restarts.  The best sequence found
    before the time budget expires is returned.

The method differs from the earlier Adam‑based approach: it solves a
linear programme to obtain a globally optimal direction in the feasible set,
and it determines the optimal mixing coefficient by a line‑search rather than
using a fixed small step.  Empirically this yields a lower max‑autoconvolution

                new_x = np.linspace(0.0, 1.0, n)
                f = np.interp(new_x, old_x, base)
                f = np.maximum(f, 0.0)
            else:
                f = np.random.rand(n)

            # normalise to the required L1 norm
            s = f.sum()
            if s > 0:
                f = f * (target_sum / s)
            else:
                f = np.full(n, target_sum / n)

            # ----- main alternating optimisation loop -------------------------
            conv_f = np.convolve(f, f)
            M = conv_f.max()
            iteration = 0
            while time.time() < deadline and iteration < 150:
                iteration += 1

                # linear map A such that A @ g = conv(f, g)
                A = build_conv_matrix(f)

                # Linear programme:
                #   maximise   sum(g)   <=> minimise  -sum(g)
                #   subject to A @ g ≤ M,   0 ≤ g ≤ target_sum
                c = -np.ones(n, dtype=float)
                b_ub = np.full(A.shape[0], M, dtype=float)
                bounds = [(0, target_sum) for _ in range(n)]

                res = linprog(c, A_ub=A, b_ub=b_ub,
                              bounds=bounds,
                              method='highs',
                              options={'presolve': True})

                if not res.success:
                    print(f"[LP fail] n={n} iter={iteration} msg={res.message}")
                    break

                g = res.x                         # feasible partner (A@g ≤ M)

                # ----- line‑search for optimal mixing coefficient α -------------
                lo, hi = 0.0, 1.0
                for _ in range(22):
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0

                    v1 = (1 - m1) * f + m1 * g
                    v1 = np.maximum(v1, 0.0)
                    v1 = v1 * (target_sum / v1.sum())
                    M1 = np.convolve(v1, v1).max()

                    v2 = (1 - m2) * f + m2 * g
                    v2 = np.maximum(v2, 0.0)
                    v2 = v2 * (target_sum / v2.sum())
                    M2 = np.convolve(v2, v2).max()

                    if M1 < M2:
                        hi = m2
                    else:
                        lo = m1

                alpha = (lo + hi) / 2.0
                f_new = (1 - alpha) * f + alpha * g
                f_new = np.maximum(f_new, 0.0)
                f_new = f_new * (target_sum / f_new.sum())
                M_new = np.convolve(f_new, f_new).max()

                if M_new + 1e-9 < M:
                    f = f_new
                    M = M_new
                    print(f"[n {n} r{restart} it {iteration}] M improved to {M:.6f}")
                else:
                    # Stagnated – stop the inner loop
                    break

                # occasional tiny jitter to shake off plateaus
                if iteration % 10 == 0:
                    jitter = np.random.randn(n) * 0.0005 * target_sum / n
                    f = np.maximum(f + jitter, 0.0)
                    f = f * (target_sum / f.sum())
                    M = np.convolve(f, f).max()

            # ----- evaluate the candidate ------------------------------------
            score = evaluate_sequence(f.tolist())
            print(f"[candidate] length {n}, score {score:.6f}")
            if score < best_score:
                best_score = score
                best_seq = f.copy()
                print(f"*** new global best = {best_score:.6f}")

            # ----- cheap local refinement --------------------------------------
            if time.time() < deadline:
                f_loc, loc_best = local_search_simple(
                    f, n, target_sum, M, deadline, max_proposals=20000
                )
                score_loc = evaluate_sequence(f_loc.tolist())
                if score_loc < best_score:
                    best_seq = f_loc.copy()
                    best_score = score_loc
                    print(f"[local refine] new global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen, but keep a safe uniform vector
    if best_seq is None:
        n = 2000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
