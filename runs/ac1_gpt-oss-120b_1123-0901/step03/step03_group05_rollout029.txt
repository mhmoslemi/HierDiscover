def _adam_optimize_smooth(a_init, n, target_sum, best_seq, best_score, deadline,
                         max_iter=250000, jitter_interval=2000, alpha_start=0.2,
                         alpha_inc=0.0005, lr0=0.1):
    a = a_init.copy()
    a = a * (target_sum / a.sum())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha = alpha_start
    iteration = 0
    while time.time() < deadline and iteration < max_iter:
        # compute conv, max and gradient
        conv, M, grad = _smooth_gradient(a, alpha)
        # update best using actual max M (not smoothed)
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n}] new best M={best_score:.6f} iter={iteration}")
        # Adam update
        iteration += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        lr = lr0 / (1 + 1e-5 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)
        # jitter
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
        # increase alpha gradually to focus more on max
        alpha = alpha + alpha_inc
    return best_seq, best_score

"""
smooth_adam_opt – Gradient‑descent on a smooth approximation of the
max‑convolution objective.

The loss  L(a) = max_k (a∗a)[k]  is non‑smooth.  We replace it by the
log‑sum‑exp (LSE) smoothing
    f_α(a) = (1/α)·log Σ_k exp(α·(a∗a)[k]),
whose gradient is
    ∇f_α(a) = 2·Σ_k w_k·rev_k,
where w_k = exp(α·(a∗a)[k]) / Σ_j exp(α·(a∗a)[j]) are normalised
weights and rev_k is the reversed copy of a shifted by k
(i.e. rev_k[i] = a_{k‑i} when the index is in range, otherwise 0).
For a given length we run Adam on this smooth objective, slowly
increasing α so that the optimisation focusses more on the true
max‑convolution.  After each step we project onto the feasible set
(non‑negative and fixed ℓ₁‑norm).  The best sequence encountered
over several lengths and random restarts is returned.

The routine works entirely with FFT‑based convolutions (O(n log n))
so that thousands of iterations fit easily into the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full (zero‑padded) autocorrelation a*a using a real FFT."""
    n = a.shape[0]
    # next power of two ≥ 2·n‑1
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, alpha: float) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Compute a smooth approximation of the max‑convolution and its gradient.

    Returns
    -------
    conv : np.ndarray
        Full convolution a*a.
    M : float
        The exact max value (used for bookkeeping).
    grad : np.ndarray
        Gradient of the LSE‑smoothed objective.
    """
    n = a.shape[0]
    # ---------- convolution ----------
    conv = _full_convolution_fft(a)
    M = float(conv.max())

    # ---------- soft‑max weights ----------
    # shift by the maximum to keep exponentials in [0,1]
    w = np.exp(alpha * (conv - M))
    w_sum = w.sum()
    if w_sum == 0.0:                     # pathological, fall back to uniform weights
        w_norm = np.full_like(w, 1.0 / w.size)
    else:
        w_norm = w / w_sum

    # ---------- gradient ----------
    # rev_a is a reversed copy of a
    a_rev = a[::-1]

    # length needed for convolution of w_norm (len 2n‑1) with a_rev (len n)
    N = 1 << ((3 * n - 2).bit_length())
    W = np.fft.rfft(w_norm, n=N)
    AR = np.fft.rfft(a_rev, n=N)
    grad_full = np.fft.irfft(W * AR, n=N)
    # slice the centre part which corresponds to the gradient components
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return conv, M, grad

    """
    a = a_init.copy()
    a = a * (target_sum / a.sum())

    v = np.zeros_like(a)

    alpha = alpha_start
    it = 0

    while time.time() < deadline and it < max_iter:
        conv, M, grad = _smooth_gradient(a, alpha)

        # record improvement (using the exact max, not the smoothed one)
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(
                f"[len {n:5d} iter {it:7d}] new best max‑conv = {best_score:.6f}"
            )

        # -------- Adam update ----------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # -------- projection onto feasible set ----------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:                           # extremely unlikely
            a = np.full_like(a, target_sum / n)

        # -------- jitter (to escape plateaus) ----------
        if it % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # -------- slowly sharpen the surrogate ----------
        alpha = alpha + alpha_inc

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_search(
    a_start: np.ndarray,
    n: int,
    target_sum: float,
    best_score: float,
    deadline: float,
    max_proposals: int = 15_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climb: move a tiny amount of mass from one index to
    another only if the max‑convolution improves.
    """
    a = a_start.copy()
    proposal = 0
    while time.time() < deadline and proposal < max_proposals:
        proposal += 1
        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        eps = a[i] * np.random.rand() * 0.04   # up to 4 % transfer
        if eps <= 0.0:
            continue
        new_a = a.copy()
        new_a[i] -= eps
        new_a[j] += eps
        M_new = _full_convolution_fft(new_a).max()
        if M_new < best_score:
            a = new_a
            best_score = M_new
            print(f"[local] improvement → {best_score:.6f} (prop {proposal})")
    return a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative vector) that minimises the
    evaluate_sequence objective.  The method explores a few lengths,
    initialises each run either from the supplied SOTA construction or
    from a simple power‑law shape, then runs the smooth‑Adam optimiser
    described above.  A brief coordinate‑swap local search is applied at
    the end of each run.  The best sequence found before the time budget
    expires is returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Load the supplied construction if it exists, otherwise fall back
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        base = np.full(2100, 1.0, dtype=float)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Test the initial (possibly SOTA) sequence
    n0 = base.shape[0]
    target_sum0 = math.sqrt(2.0 * n0)
    base = base * (target_sum0 / base.sum())
    init_score = evaluate_sequence(base.tolist())
    best_seq = base.copy()
    best_score = init_score
    print(f"[init] length={n0}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore (keep them modest for speed)
    lengths = [2100, 3000, 4000, 6000]          # all ≤ 8000
    lengths = [L for L in lengths if L <= 8000 and L >= 300]

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # Number of random restarts per length
        restarts = 3
        for r in range(restarts):
            if time.time() >= deadline:
                break

            # ------- initialise --------------------------------------------------
            if r == 0:
                # Interpolate the current best onto the new grid
                old_x = np.linspace(0.0, 1.0, best_seq.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                a0 = np.interp(new_x, old_x, best_seq)
                a0 = np.maximum(a0, 0.0)
            elif r == 1:
                # Power‑law shape a_i ∝ (i+1)^β with β≈2 (good empirical start)
                beta = 2.0
                idx = np.arange(1, n + 1, dtype=float)
                a0 = idx ** beta
                a0 = np.maximum(a0, 0.0)
            else:
                # Pure Dirichlet (random positive vector)
                a0 = np.random.rand(n)

            a0 = a0 * (target_sum / a0.sum())

            # ------- optimisation ------------------------------------------------
            best_seq, best_score = _adam_optimize_smooth(
                a_init=a0,
                n=n,
                target_sum=target_sum,
                best_seq=best_seq,
                best_score=best_score,
                deadline=deadline,
                max_iter=30_000,
                jitter_interval=3000,
                alpha_start=0.5,
                alpha_inc=0.0008,
                lr0=0.08,
            )

            # ------- cheap local refinement ------------------------------------
            if time.time() < deadline:
                best_seq, best_score = _local_search(
                    a_start=best_seq,
                    n=n,
                    target_sum=target_sum,
                    best_score=best_score,
                    deadline=deadline,
                    max_proposals=12_000,
                )

    # ------------------------------------------------------------------
    # Safety fallback
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # return plain Python list as required by the evaluator
    return best_seq.tolist()
