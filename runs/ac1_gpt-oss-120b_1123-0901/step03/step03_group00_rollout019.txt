It uses a two‑phase Adam optimiser:

1. **Phase 1 – L2 flattening** – minimise the squared deviation of the
   self‑convolution from the uniform distribution (the ideal flat
   spectrum).  
2. **Phase 2 – smooth‑max** – tighten the maximum entry of the
   convolution using the same surrogate as the previous code.

The optimiser works on the probability simplex via a soft‑max
parameterisation, supports warm‑starts (including up‑sampling a shorter
best sequence to a longer one), and automatically adapts its schedule
to the remaining time budget.  Progress and the best objective value are
printed during the run.

```python
"""
Search for a non‑negative step‑function (list of heights) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

using a two‑phase Adam optimiser:
* Phase 1 – minimise the L2 distance of the self‑convolution to the flat
  distribution (mu = 1/(2n‑1)).  This quickly flattens the spectrum.
* Phase 2 – switch to a smooth‑max surrogate of the max‑operator (as in
  the previous solution) to polish the final maximum.

The code works for arbitrary lengths, warm‑starts from the current best
(or from the provided global `height_sequence_1`), and up‑samples a
shorter best sequence when moving to a longer length.  It respects a
global time budget (≈ 1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small utility: next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)‑1) using the FFT.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i = 0…n‑1 .
    """
    n = a.shape[0]
    rev_a = a[::-1]                         # reversed a
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _logits_from_probs(p: np.ndarray) -> np.ndarray:
    """Convert a probability vector (sums to 1) to unrestricted logits."""
    # Add a tiny constant to avoid log(0)
    return np.log(p + 1e-16)


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns a probability vector."""
    mx = np.max(x)
    ex = np.exp(x - mx)
    return ex / ex.sum()

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget in seconds (default 1000 s).  The function stops a
        little before the deadline to avoid truncation.

    Returns
    -------
    list[float]
        The best sequence found (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq = None
    best_val = float("inf")

    # Lengths – we start small and go larger, up‑sampling when possible.
    lengths = [1024, 2048, 4096, 8192, 16384]

    # Global hyper‑parameters (tuned for speed / stability)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05                     # base learning rate for Adam
    lam_factor = 1.03             # geometric increase of λ in phase‑2
    max_restarts = 3
    # Phase split – 40 % of the allotted iterations to the L2‑flattening phase
    phase1_frac = 0.4

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n--- Optimising length n = {n} ---")

        # FFT sizes (powers of two) for the two convolutions.
        fft_len_conv = _next_pow2(2 * n - 1)
        fft_len_corr = _next_pow2(3 * n)          # enough for (2n‑1) + n‑1

        # Target uniform value of the convolution (average = 1/(2n‑1))
        mu = 1.0 / (2 * n - 1)

        for restart in range(max_restarts):
            if time.time() > deadline:
                break

            # --------------------------------------------------------------
            # Initialise logits
            if restart == 0 and best_seq is not None:
                # Warm‑start from the current best (up‑sample if necessary)
                cand = np.asarray(best_seq, dtype=np.float64)
                if cand.shape[0] < n:

                    cand = np.repeat(cand, factor)
                    # Pad the remainder with zeros
                    if cand.shape[0] < n:
                        cand = np.pad(cand, (0, n - cand.shape[0]))
                elif cand.shape[0] > n:
                    cand = cand[:n]
                cand = np.maximum(cand, 0.0)
                cand_sum = cand.sum()
                if cand_sum < 1e-12:
                    cand = np.random.dirichlet(np.ones(n))
                else:
                    cand /= cand_sum
                x = _logits_from_probs(cand)
                print(f"[restart {restart}] warm start from previous best")
            elif 'height_sequence_1' in globals():
                # Use the provided construction (trim / pad)
                cand = np.asarray(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 0.0)
                cand_sum = cand.sum()
                if cand_sum < 1e-12:
                    cand = np.random.dirichlet(np.ones(n))
                else:
                    cand /= cand_sum
                x = _logits_from_probs(cand)
                print(f"[restart {restart}] start from height_sequence_1")
            else:
                # Pure random start (Gaussian logits)
                x = np.random.randn(n) * 0.1

            # Adam state
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)

            # Temperature for smooth‑max (used only in phase‑2)
            lam = 0.5

            t = 0
            max_iters = 25000                # safety cap per restart
            phase1_iters = int(max_iters * phase1_frac)

            last_improve = time.time()

            while time.time() < deadline and t < max_iters:
                # ------- logits → probabilities (simplex) -------
                a = _softmax(x)               # a_i ≥ 0, Σa_i = 1

                # ------- self‑convolution ----------------------
                conv = _full_convolution(a, fft_len=fft_len_conv)
                max_c = conv.max()

                # ------- exact objective (for tracking) -------
                val = 2.0 * n * max_c          # Σa = 1 ⇒ denominator = 1

                # ------- keep best result -------
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={restart} iter={t:05d}] new best: {val:.6f}")

                # ------- gradient computation -------------------
                if t < phase1_iters:
                    # Phase 1: minimise L2 deviation of conv from the flat target µ
                    diff = conv - mu                     # length 2n‑1, sums to 0
                    grad_a = 4.0 * _cross_correlation(diff, a, fft_len=fft_len_corr)
                else:
                    # Phase 2: smooth‑max surrogate (same as previous method)
                    w = np.exp(lam * (conv - max_c))
                    w /= w.sum()                          # Σ w = 1
                    grad_a = 2.0 * _cross_correlation(w, a, fft_len=fft_len_corr)

                    lam = min(lam * lam_factor, 1e6)

                avg = np.dot(a, grad_a)                 # ⟨a,∇φ⟩
                grad_x = a * (grad_a - avg)             # ∇_x φ

                # ------- Adam update ---------------------------
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)

                v_hat = v / (1.0 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # ------- stagnation handling -------------------
                if time.time() - last_improve > 5.0:
                    # Add a small random shake to escape plateaus
                    x += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()
                    print(f"[n={n} restart={restart} iter={t}] shake (stagnation)")

    if best_seq is None:
        # Very unlikely fallback – a uniform vector of moderate size
        n_fallback = 2000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Ensure exact normalisation to avoid tiny drift
    best_seq = best_seq / np.sum(best_seq)

    return best_seq.tolist()
```
