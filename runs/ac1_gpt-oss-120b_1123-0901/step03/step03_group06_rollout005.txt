"""
Propose a low‑peak step‑function using a multi‑scale Adam‑based
exponentiated‑gradient optimizer.

Algorithm
---------
* Represent the step heights a_i as a scaled probability vector
      a = S·p ,   S = sqrt(2·n) ,   ∑p_i = 1 ,   p_i ≥ 0 .
  The evaluation metric reduces to   score = 2·n·max(p∗p).

* For a fixed length n we minimise max(p∗p) by (approx.) sub‑gradient
  descent on the log‑probabilities (i.e. a multiplicative EG step).
  The sub‑gradient of the *soft* maximum
          M_β(p) = log_β ∑_k exp(β·(p∗p)_k)
  is   rev_i = Σ_k w_k·p_{k-i}
  where w_k ∝ exp(β·((p∗p)_k−max(p∗p))).
  The gradient of the objective (2·n·M_β) with respect to p is
          grad = 4·n·rev.

* Adam is used to adapt the step size for the log‑probabilities.
  After each update we project back onto the simplex,
  enforce symmetry (p_i = p_{n‑1‑i}) and renormalise.

* Lengths are processed in a multiscale schedule.
  The best probability vector from the previous scale is interpolated
  to the next scale and refined again.  After the schedule we perform
  a few random restarts at the best length to escape possible local
  minima.

* All heavy work (convolution and the required correlation) is done
  via a single FFT per iteration, so even n≈8000 runs comfortably
  within the 1000 s budget.

The function returns the best height vector (list of floats) found
before the time limit expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _optimize_adam_length(
    n: int,
    p_start: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam‑based exponentiated‑gradient optimisation for a fixed length ``n``.
    Returns the best probability vector ``p`` (∑p=1) and its score
    ``2·n·max(p∗p)``.
    """
    # ---- hyper‑parameters (tuned empirically) ----
    beta_start = 0.1
    beta_end = 30.0           # final soft‑max sharpness
    beta_len = 100_000        # iterations over which beta grows linearly
    jitter_interval = 5_000
    jitter_scale = 1e-3

    # Adam configuration
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # Initial state
    p = p_start.copy()
    logp = np.log(p + 1e-30)   # log‑space for EG
    N = _next_pow2(3 * n - 2)  # FFT size sufficient for all convolutions

    m = np.zeros_like(p)      # first moment
    v = np.zeros_like(p)      # second moment

    base_lr = 1.0 / (4.0 * n)  # base learning rate (same scale as original EG)
    best_local_score = float("inf")
    best_local_p = p.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ---------- full self‑convolution (p∗p) ----------
        # FFT of p (reused for both convolution and correlation)
        P = np.fft.rfft(p, n=N)
        conv = np.fft.irfft(P * P, n=N)[: 2 * n - 1]
        M = conv.max()

        cur_score = 2.0 * n * M
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_p = p.copy()
            print(f"[len {n:5d} iter {it:7d}] new local best = {cur_score:.6f}")

        # ---------- soft‑max weights ----------
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_len)
        w = np.exp(beta * (conv - M))               # 0 ≤ w ≤ 1, max w = 1

        # ---------- weighted correlation rev_i = Σ_k w_k·p_{k-i} ----------
        # Convolution of p with reversed w
        W_rev = np.fft.rfft(w[::-1], n=N)
        conv_rev = np.fft.irfft(P * W_rev, n=N)
        L = 2 * n - 1
        rev = conv_rev[L - n : L][::-1]              # length n

        # ---------- gradient (sub‑gradient of the smoothed max) ----------
        grad = 4.0 * n * rev                         # ∇_p (2·n·M_β)

        # ---------- Adam update of log‑probabilities ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = base_lr / (1.0 + it * 1e-4)             # slowly decaying step size
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        logp -= step

        # ---------- occasional jitter ----------
        if it % jitter_interval == 0:
            logp += np.random.randn(n) * jitter_scale

        # ---------- project back onto the probability simplex ----------
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)          # prevent overflow
        p = exp_shift / exp_shift.sum()

        # -------- enforce symmetry (optional but helpful) --------
        p = (p + p[::-1]) / 2.0
        p = np.maximum(p, 1e-30)                    # keep strictly positive
        p /= p.sum()

        # keep logp consistent with the symmetrised p
        logp = np.log(p + 1e-30)

    return best_local_p, best_local_score
