def max_subgradient(a, conv, fft_size=None):
    n = len(a)
    max_val = conv.max()
    max_mask = (conv == max_val).astype(np.float64)
    # average across max locations
    # Compute w = max_mask / max_mask.sum()
    w = max_mask / max_mask.sum()
    # compute convolution w * a_rev via FFT
    a_rev = a[::-1]
    size = fft_size if fft_size is not None else 1 << int(np.ceil(np.log2(len(w) + len(a_rev) - 1)))
    fw = np.fft.rfft(w, size)
    fa_rev = np.fft.rfft(a_rev, size)
    conv_w = np.fft.irfft(fw * fa_rev, size)
    grad = 2.0 * conv_w[n-1 : n-1 + n]  # slice
    return grad

"""
Propose a step‑function (non‑negative sequence) that minimizes

    F(a) = 2·n·max_k (a*a)[k] / (∑ a_i)²

The implementation combines three ideas:

1. **Analytic families** – a quick sweep over several classic window
   shapes (uniform, half‑sine, Hann, Gaussian, symmetric Beta, symmetric
   exponential) for a few different lengths (2000‑5000).  The best
   candidate from this cheap search seeds the subsequent optimisation.

2. **Projected sub‑gradient descent on the exact max‑convolution** –
   the objective is scale‑invariant, so we keep the vector on the
   probability simplex (∑a=1, a≥0).  The sub‑gradient of
   `max_k (a*a)[k]` is the average of the gradients of the active
   convolution indices and can be computed with a single FFT
   (correlation of the indicator mask with the reversed vector).

3. **Optional Adam refinements** – a few short Adam runs (log‑space
   parametrisation) are performed on the same sub‑gradient to escape
   shallow minima.

The whole search respects the 1000 s budget, constantly updates the
best‑found sequence and finally returns it as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length ``size``.
    Returns the vector of length 2·len(a)−1.
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _max_subgradient(a: np.ndarray, mask: np.ndarray, size_corr: int) -> np.ndarray:
    """
    Sub‑gradient of  max_k (a*a)[k]  w.r.t. a.
    ``mask`` has length 2·n−1 and contains 1 on the active indices,
    0 elsewhere (it is normalised to sum 1 internally).
    The gradient is 2·(mask ★ a) where ★ denotes correlation.
    """
    n = a.shape[0]
    mask_sum = mask.sum()
    if mask_sum == 0.0:                       # should not happen
        mask = np.ones_like(mask) / mask.size
    else:
        mask = mask / mask_sum

    a_rev = a[::-1]
    fw = np.fft.rfft(mask, size_corr)
    fa_rev = np.fft.rfft(a_rev, size_corr)
    corr = np.fft.irfft(fw * fa_rev, size_corr)
    # slice that corresponds to indices 0 … n‑1 of the original vector
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x≥0, ∑x=1}.
    Implementation of the algorithm from Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_val(a: np.ndarray) -> float:
    """
    Exact objective (scale invariant) for a normalised vector (∑a = 1):
          2·n·max_k (a*a)[k]
    """
    n = a.shape[0]
    conv = _full_convolution(a, size=_next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    The routine performs a cheap sweep over analytic families, then
    refines the best candidate with projected sub‑gradient descent and a
    few Adam restarts.  The returned value is a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # safety margin
    best_val = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # 1) cheap exploration over a handful of classic window families
    # ------------------------------------------------------------------
    n_candidates = [2000, 3000, 4000, 5000]

    for n in n_candidates:
        if time.time() >= deadline:
            break

        size_conv = _next_pow2(2 * n - 1)

        def _try(seq):
            nonlocal best_val, best_seq
            seq = np.maximum(seq, 0.0)
            if seq.sum() == 0.0:
                return
            seq = seq / seq.sum()
            conv = _full_convolution(seq, size=size_conv)
            val = float(2.0 * n * conv.max())
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Family] n={n:4d}  new best = {val:.6f}")

        # uniform
        _try(np.full(n, 1.0 / n))

        # half‑sine (strictly positive)
        i = np.arange(1, n + 1, dtype=np.float64)
        seq = np.sin(np.pi * i / (n + 1))
        _try(seq)

        # Hann window (0.5·(1‑cos(2πi/(n‑1))))
        i = np.arange(n, dtype=np.float64)
        seq = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1.0)))
        _try(seq)

        # Gaussian shapes, sigma relative to n
        for factor in (0.12, 0.15, 0.18, 0.22, 0.30):
            sigma = factor * n
            x = np.arange(n, dtype=np.float64)
            centre = (n - 1) / 2.0
            seq = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
            _try(seq)

        # Symmetric Beta distributions (parameters p=p)
        for p in (0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
            x = np.linspace(0.0, 1.0, n, dtype=np.float64)
            eps = 1e-12
            seq = np.power(x + eps, p - 1.0) * np.power(1.0 - x + eps, p - 1.0)
            seq = np.maximum(seq, eps)
            _try(seq)

        # Symmetric exponential (a_i = exp(-r·min(i, n‑1‑i)))
        for r in (0.15, 0.20, 0.25, 0.30):
            idx = np.arange(n, dtype=np.float64)
            seq = np.exp(-r * np.minimum(idx, n - 1 - idx))
            _try(seq)

    # ------------------------------------------------------------------
    # Helper for a single projected sub‑gradient refinement run
    # ------------------------------------------------------------------
    def _refine(a0: np.ndarray, max_iters: int, time_limit: float):
        """
        Refine a normalised vector using projected sub‑gradient descent.
        Returns the best vector found inside the time limit.
        """
        a = a0.copy()
        n = a.shape[0]
        size_conv = _next_pow2(2 * n - 1)
        size_corr = _next_pow2(3 * n)

        best_local_val = _objective_val(a)
        best_local_seq = a.copy()
        last_improve = time.time()

        for it in range(1, max_iters + 1):
            if time.time() >= time_limit:
                break

            conv = _full_convolution(a, size=size_conv)
            max_val = conv.max()
            # take all indices within a tiny tolerance of the maximum
            mask = (conv >= max_val - 1e-12).astype(np.float64)

            grad = _max_subgradient(a, mask, size_corr)

            # simple step‑size schedule
            lr = 0.25 / (1.0 + it * 1e-4)

            a = a - lr * grad
            a = _project_simplex(a)

            # occasional exact evaluation
            if it % 100 == 0:
                cur_val = _objective_val(a)
                if cur_val < best_local_val - 1e-9:
                    best_local_val = cur_val
                    best_local_seq = a.copy()
                    last_improve = time.time()
                    print(f"[Refine] it {it:05d}  new local best = {cur_val:.6f}")

            # shake if stuck for >1.2 s
            if time.time() - last_improve > 1.2:
                a = a + (np.random.rand(n) - 0.5) * 5e-4
                a = _project_simplex(a)
                last_improve = time.time()

        return best_local_seq, best_local_val

    # ------------------------------------------------------------------
    # 2) Refine the best analytic candidate (if any) with sub‑gradient descent
    # ------------------------------------------------------------------
    if best_seq is not None:
        # allocate a small portion of the remaining time for a long refinement
        remaining = deadline - time.time()
        if remaining > 5.0:
            max_iters = int(min(30000, remaining * 30))   # roughly 30 iters per second
            seq, val = _refine(best_seq, max_iters, deadline)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Global] post‑refine best = {val:.6f}")

    # ------------------------------------------------------------------
    # 3) A few short Adam‑based random restarts (log‑space) for extra exploration
    # ------------------------------------------------------------------
    def _adam_run(n, time_limit):
        """
        Adam optimiser on the smooth‑max surrogate, starting from a random
        distribution.  The final vector is projected onto the simplex.
        """
        size_conv = _next_pow2(2 * n - 1)
        size_corr = _next_pow2(3 * n)

        # ===== random initialisation =====
        mode = np.random.randint(0, 4)
        if mode == 0:
            a = np.random.dirichlet(np.ones(n))
        elif mode == 1:
            r = np.random.uniform(0.2, 0.95)
            a = np.power(r, np.arange(n, dtype=np.float64))
            a /= a.sum()
        elif mode == 2:
            i = np.arange(1, n + 1, dtype=np.float64)
            a = np.sin(np.pi * i / (n + 1))
            a = np.maximum(a, 0.0)
            a /= a.sum()
        else:
            sigma = np.random.uniform(0.12, 0.30) * n
            x = np.arange(n, dtype=np.float64)
            centre = (n - 1) / 2.0
            a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
            a /= a.sum()

        u = np.log(a + 1e-30)                # soft‑max parametrisation
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        beta1, beta2, eps = 0.9, 0.999, 1e-8
        base_lr = 0.10
        lam = 1.0
        lam_factor = 1.03

        it = 0
        best_local_val = float("inf")
        best_local_u = None
        last_improve = time.time()

        while time.time() < time_limit:
            it += 1
            # ----- probabilities -----
            u_shift = u - np.max(u)
            a = np.exp(u_shift)
            a /= a.sum()

            # ----- soft‑max over the convolution -----
            conv = _full_convolution(a, size=size_conv)
            w = np.exp(lam * (conv - conv.max()))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= w_sum

            # ----- gradient w.r.t. a (smooth‑max) -----
            # correlation w ★ a
            a_rev = a[::-1]
            fw = np.fft.rfft(w, size_corr)
            fa_rev = np.fft.rfft(a_rev, size_corr)
            corr = np.fft.irfft(fw * fa_rev, size_corr)
            grad_a = 2.0 * corr[n - 1 : n - 1 + n]

            # ----- convert to log‑space (soft‑max Jacobian) -----
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            # ----- Adam update -----
            m = beta1 * m + (1.0 - beta1) * grad_u
            v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = base_lr * (0.9998 ** it)
            u -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # ----- increase smoothing -----
            lam *= lam_factor

            # ----- evaluate exact objective every few steps -----
            if it % 30 == 0:
                cur_val = _objective_val(a)
                if cur_val < best_local_val:
                    best_local_val = cur_val
                    best_local_u = u.copy()
                    last_improve = time.time()
                    print(f"[Adam] n={n:4d} it={it:05d}  val={cur_val:.6f}")

            # ----- shake if stagnating -----
            if time.time() - last_improve > 1.5:
                u += (np.random.rand(n) - 0.5) * 5e-4
                m.fill(0.0)
                v.fill(0.0)
                last_improve = time.time()

            # safety stop
            if it >= 25000:
                break

        if best_local_u is not None:
            # reconstruct probability vector from the best logits
            u_shift = best_local_u - np.max(best_local_u)
            a_best = np.exp(u_shift)
            a_best /= a_best.sum()
            return a_best, best_local_val
        return None, float("inf")

    # run a handful of Adam restarts if time permits
    while time.time() < deadline - 2.0:
        n = np.random.choice(n_candidates)
        seq, val = _adam_run(int(n), deadline)
        if seq is not None and val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Adam] global new best = {val:.6f}")

    # ------------------------------------------------------------------
    # If nothing was found (should not happen) fall back to a uniform vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, 1.0 / n)

    print(f"Final best objective ≈ {best_val:.6f} (length {len(best_seq)})")
    # Convert to plain Python list for the outer harness.
    return best_seq.tolist()
