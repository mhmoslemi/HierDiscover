"""
Propose candidate step‑function heights.

Algorithm
---------
* Randomly pick a length n (≈800 … 3000) and initialise a shape
  a_i ∝ (i+1)^(-p)·exp(-i/τ) (power‑law + optional exponential tail).
* Optimise the shape with a scale‑invariant Adam optimiser that minimises a
  smooth‑max approximation of the full self‑convolution  a*a .
  – The smooth‑max uses a temperature τ that is exponentially decayed.
  – Gradients are obtained analytically as a cross‑correlation of the current
    sequence with the reversed soft‑max weight vector; both convolutions are
    evaluated by a real FFT (O(n log n)).
  – After each step the vector is projected onto the simplex
    Σa_i = √(2 n) (non‑negative, fixed sum), which makes the objective equal to
    the maximal convolution value.
* The optimisation for a given length runs for a few seconds (≈5 s) and
  returns the best local sequence.
* The global best sequence is kept.  Frequently a warm‑start is generated by
  interpolating the current global best to a new length, providing strong
  guidance for later restarts.
* Adam’s adaptive step sizes, a learning‑rate schedule and occasional random
  perturbations (to escape plateaus) are used.
* The whole search respects the overall time budget (default 1000 s) and
  prints progress messages.

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (non‑negative, Σx = s)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x_i ≥ 0, Σx_i = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power‑of‑2
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative artefacts caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Interpolate a sequence to a new length (linear interpolation)
# ----------------------------------------------------------------------
def _interpolate_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if old_len == new_len:
        return seq.copy()
    old_pos = np.linspace(0.0, 1.0, old_len)
    new_pos = np.linspace(0.0, 1.0, new_len)
    interp = np.interp(new_pos, old_pos, seq)
    return interp


# ----------------------------------------------------------------------
# Adam optimiser for a single length (smooth‑max objective)
# ----------------------------------------------------------------------
def _adam_optimize_one_length(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.15,
    min_tau_factor: float = 0.003,
    base_lr: float = 0.12,
    lr_decay: float = 0.9997,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam based minimisation of max(conv(a,a)) for a fixed length n.
    Returns the best sequence obtained and its evaluation score.
    """
    target_sum = np.sqrt(2.0 * n)          # Σ a_i = √(2 n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    best_seq = a.copy()
    conv = _full_convolution_fft(a, a)
    best_score = 2.0 * n * conv.max() / (target_sum ** 2)   # = conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    iteration = 1
    while time.time() < run_deadline:
        # --------------------------------------------------------------
        # 1) self‑convolution and current maximum
        # --------------------------------------------------------------
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()

        # --------------------------------------------------------------
        # 2) smooth‑max (soft‑max) weights
        # --------------------------------------------------------------
        tau = max(init_tau_factor * max_val * (0.97 ** iteration),
                  min_tau_factor * max_val)
        scaled = (conv - max_val) / tau                # ≤ 0
        w = np.exp(scaled)                             # in (0,1]
        w_sum = w.sum()
        if w_sum == 0.0:                               # should not happen
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # 3) gradient of the weighted sum Σ w_k * conv_k
        #    grad_i = 2 * Σ_j a_j * w_{i+j}
        #    (cross‑correlation, computed via FFT)
        # --------------------------------------------------------------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)          # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)                 # positions that give Σ_j a_j w_{i+j}
        grad = 2.0 * conv2[idx]

        # --------------------------------------------------------------
        # 4) Adam update (bias‑corrected) + projection onto simplex
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # Scale the step size so that the update magnitude is comparable to
        # the current peak of the convolution (prevents overly aggressive moves).
        max_grad = np.max(np.abs(m_hat)) + eps
        lr = base_lr * max_val / max_grad
        lr *= lr_decay ** iteration

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # --------------------------------------------------------------
        # 5) evaluate and possibly store as new best
        # --------------------------------------------------------------
        conv_new = _full_convolution_fft(a_new, a_new)
        score_new = 2.0 * n * conv_new.max() / (target_sum ** 2)   # = conv_new.max()

        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {best_score:.6f}")

        a = a_new
        iteration += 1

        # occasional random shake‑up to escape plateaus
        if np.random.rand() < 0.008:
            noise = np.random.randn(n) * (target_sum * 4e-4 / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)

    # Final evaluation with the supplied evaluator (guarantees identical rounding)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score
