"""
Improved step‑function search.

The algorithm repeatedly
  1. picks a problem size (larger sizes → lower theoretical bound);
  2. builds a simple “U‑shaped’’ seed (parameterised by an exponent);
  3. projects the seed onto the simplex Σa = √(2n) and enforces symmetry;
  4. runs a *smooth‑max* gradient descent (Adam + simplex projection) that
     minimises a differentiable surrogate of max(conv(a,a));
  5. optionally refines the result with a short simulated‑annealing phase.

The smooth‑max uses the soft‑max weights w_k = exp(α·c_k)/∑exp(α·c_k) where
c = conv(a,a).  Its gradient is
        ∂f/∂a_i = 2· Σ_k w_k·a_{k−i}
which is a cross‑correlation and is evaluated via a single FFT.
The schedule slowly raises α so that the optimisation first sees a
global shape and then focuses on the true max‑entry.

Larger `n` (up to ~8000) greatly lowers the attainable bound
(the uniform sequence gives a bound of 2; the average value of the
convolution is ≈1, so a bound near 1.2 is realistic).  The method is
fully vectorised, runs many iterations within the 1000 s budget and
stores the best candidate found.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex { x_i ≥ 0 , Σ x_i = s }.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex of radius ``s``."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# --------------------------------------------------------------------------- #
# Fast full convolution (real FFT) – returns a length 2*n‑1 array.
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    # next power of two for efficient FFT (at least 2*n-1)
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv

# --------------------------------------------------------------------------- #
# Objective value (max of the convolution) – for Σa = √(2n) this equals the
# full evaluation function.
# --------------------------------------------------------------------------- #
def _max_conv(a: np.ndarray) -> float:
    return float(_conv_full_fft(a).max())

# --------------------------------------------------------------------------- #
# Enforce symmetry (a_i = a_{n‑1‑i}).  Symmetry never hurts the bound and
# halves the effective search space.
# --------------------------------------------------------------------------- #
def _symmetrize(v: np.ndarray) -> np.ndarray:
    return (v + v[::-1]) * 0.5

# --------------------------------------------------------------------------- #
# Smooth‑max gradient descent (Adam + simplex projection).
# --------------------------------------------------------------------------- #
def smooth_max_grad_descent(a_init: np.ndarray,
                            target_sum: float,
                            max_time: float,
                            start_time: float):
    """Minimise max(conv(a,a)) using a smooth surrogate."""
    a = _project_onto_simplex(a_init.astype(float), target_sum)
    a = _symmetrize(a)                     # keep the vector symmetric
    n = a.shape[0]

    best_vec = a.copy()
    best_score = _max_conv(a)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # smoothing parameter schedule
    alpha = 5.0          # start very smooth
    alpha_max = 200.0    # final (sharp) value
    alpha_inc = 1.02

    while time.time() < start_time + max_time:
        # ---------- convolution of a with itself ----------
        conv = _conv_full_fft(a)

        # ---------- soft‑max weights ----------
        c_max = conv.max()
        # stable exponentiation
        exp_term = np.exp(alpha * (conv - c_max))
        w = exp_term / exp_term.sum()           # ∑ w = 1, w ≥ 0

        # ---------- gradient = 2 * (w ⋆ a) ----------
        # cross‑correlation w ⋆ a via FFT
        L = 1 << ((len(w) + n - 1).bit_length())
        W = np.fft.rfft(w, n=L)
        A = np.fft.rfft(a, n=L)
        corr = np.fft.irfft(W * np.conj(A), n=L)   # linear correlation
        # extract lags 0 … n‑1 (index offset = n‑1)
        grad = 2.0 * corr[n - 1:n - 1 + n]

        # ---------- Adam update ----------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate base (scaled to problem size)
        base_lr = 0.05
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)
        a_new = _symmetrize(a_new)             # keep symmetry

        # occasional random jitter to escape shallow plateaus
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            a_new = _symmetrize(a_new)

        a = a_new

        # ---------- bookkeeping ----------
        cur_score = _max_conv(a)
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{time.time():.2f}] GD new best = {best_score:.6f} (n={n})")

        # increase smoothing (sharpen towards true max)
        if alpha < alpha_max:
            alpha = min(alpha * alpha_inc, alpha_max)

    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Simulated annealing – same implementation as the previous baseline.
# --------------------------------------------------------------------------- #
def simulated_annealing(a_init: np.ndarray,
                        target_sum: float,
                        max_time: float,
                        start_time: float):
    """Random mass transfers between two positions with a classic SA schedule."""
    a = a_init.copy()
    n = a.shape[0]

    cur_score = _max_conv(a)
    best_vec = a.copy()
    best_score = cur_score

    T0 = max(0.01, cur_score * 0.05)   # initial temperature
    T_min = 1e-6

    while time.time() < start_time + max_time:
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1
        if a[i] <= 0.0:
            continue
        frac = np.random.rand() * 0.5
        delta = a[i] * frac

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = _project_onto_simplex(a_new, target_sum)
        a_new = _symmetrize(a_new)

        new_score = _max_conv(a_new)
        delta_score = new_score - cur_score

        elapsed = time.time() - start_time
        frac_time = max(0.0, 1.0 - elapsed / max_time)
        T = T_min + T0 * frac_time

        if delta_score <= 0.0 or np.random.rand() < np.exp(-delta_score / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_vec = a.copy()
                print(f"[{time.time():.2f}] SA new best = {best_score:.6f} (n={n})")

    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Main entry point required by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42,

    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin for the grader’s timeout handling
    deadline = start + budget_s - 0.5

    best_seq = None          # numpy array of the best candidate
    best_score = float("inf")

    # ------------------------------------------------------------------- #
    # Outer loop – a few restarts with varying lengths.
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        # Choose a length.  Larger n gives a lower theoretical bound.
        if best_seq is not None and np.random.rand() < 0.15:
            n = len(best_seq)
        else:
            n = np.random.randint(3000, 8001)   # 3k … 8k

        target_sum = np.sqrt(2.0 * n)   # force Σa = √(2n) → objective = max(conv)

        # --------------------------------------------------------------- #
        # Initialise a candidate.
        # --------------------------------------------------------------- #
        if "height_sequence_1" in globals() and np.random.rand() < 0.5:
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                # tile the existing sequence to reach the required length
                reps = n // src.size + 1
                init = np.tile(src, reps)[:n]
        else:
            # U‑shaped seed: a_i ∝ (i+1)^‑p + (n‑i)^‑p with random exponent p
            p = np.random.uniform(0.5, 4.0)
            idx = np.arange(n, dtype=float)
            weight = 1.0 / ((idx + 1) ** p) + 1.0 / ((n - idx) ** p)
            # add a tiny random component to avoid exact symmetry at start
            init = weight + 0.001 * np.random.rand(n)

        a0 = _project_onto_simplex(init, target_sum)
        a0 = _symmetrize(a0)

        # quick check – update global best
        cur_score = _max_conv(a0)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a0.copy()
            print(f"[{time.time():.2f}] Init best = {best_score:.6f} (n={n})")

        # --------------------------------------------------------------- #
        # Allocate time for this restart.
        # --------------------------------------------------------------- #
        time_left = deadline - time.time()
        if time_left <= 0:

        # about 70 % of the remaining budget goes to the smooth‑max GD,
        # the rest (if any) to a short SA polishing phase.
        gd_time = min(30.0, time_left * 0.7)
        a_gd, score_gd = smooth_max_grad_descent(a0, target_sum,
                                                 gd_time, time.time())
        if score_gd < best_score:
            best_score = score_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.2f}] GD best = {best_score:.6f} (n={n})")

        # optional short SA phase if we still have time
        time_left = deadline - time.time()
        if time_left > 2.0:

            a_sa, score_sa = simulated_annealing(best_seq, target_sum,
                                                 sa_time, time.time())
            if score_sa < best_score:
                best_score = score_sa
                best_seq = a_sa.copy()
                print(f"[{time.time():.2f}] SA after GD best = {best_score:.6f} (n={n})")

        # early exit if we already beat the target bound

            break

    # ------------------------------------------------------------------- #
    # Fallback – a uniform vector (should never be needed).
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 1500
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list as required by the grader.
    return best_seq.tolist()
