"""
Improved search for a low‑peak self‑convolution step function.

Algorithm Overview
------------------
* Work with the probability vector `p` (p_i ≥ 0, Σp_i = 1) and scaling
  a_i = sqrt(2·n)·p_i, because the objective becomes
      score = 2·n·max(p∗p) .
* Optimise `p` with an Adam‑style exponentiated‑gradient (mirror descent
  on the simplex).  The update is performed in log‑space to keep the
  simplex constraints automatically.
* A soft‑max approximation of the max‑operator is used:
      w_k = exp(β·(conv_k – M)),   M = max(conv) .
  The smoothing parameter β is increased linearly from a small value
  (0.1) to a large value (≈80) so the gradient starts smooth and ends
  almost exactly at the true maximum.
* The gradient of the smoothed max w.r.t. `p` is  rev_i = Σ_k w_k·p_{k−i},
  which is a correlation that can be computed by a single FFT.
* Multi‑scale schedule: start with a modest length, optimise, then
  up‑sample the current best solution to a larger length and continue.
  Several restarts (interpolated solution + random Dirichlet) are tried
  for each length.
* After the main optimisation a cheap local‑search (random perturbation
  + projection) is performed to squeeze a little more out of the best
  candidate.
* The whole routine respects the 1000 s time budget and prints progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """
    Full self‑convolution (p∗p) using a single FFT of size N.
    Returns a real array of length 2·len(p)−1.
    """
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    return conv[: 2 * p.shape[0] - 1]


def _weighted_rev_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k·p_{k−i}  (i = 0 … n‑1).

    This is a correlation of p with the weight vector w and is obtained
    via FFTs.
    """
    n = p.shape[0]
    L = w.shape[0]                 # = 2·n−1
    # FFT of p (size N already ≥ n+L−1)
    P = np.fft.rfft(p, n=N)
    # FFT of reversed weights
    W_rev = np.fft.rfft(w[::-1], n=N)
    # Convolution of p with reversed w
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    conv_rev = conv_rev[: n + L - 1]          # keep linear part
    # Extract rev = conv_rev[L‑n : L] and reverse to obtain correct order
    rev = conv_rev[L - n : L][::-1]
    return rev


# ----------------------------------------------------------------------
def _interpolate_p(p_old: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    if p_old.shape[0] == n_new:
        return p_old.copy()
    x_old = np.linspace(0.0, 1.0, p_old.shape[0])
    x_new = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x_new, x_old, p_old)
    p_new = np.maximum(p_new, 0.0)
    s = p_new.sum()
    if s == 0:
        p_new = np.full(n_new, 1.0 / n_new)
    else:
        p_new = p_new / s
    return p_new

def _optimize_adam_length(
    n: int,
    p_start: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:

    Returns the best probability vector (sum = 1) and its (raw) score
    ``2·n·max(p∗p)`` encountered during the run.
    """
    p = p_start.copy()
    # log‑space representation (mirror map for the simplex)
    logp = np.log(p + 1e-30)

    # Adam moments
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # schedule for the soft‑max temperature β
    beta_start, beta_end = 0.1, 80.0
    beta_len = max_iters // 2

    # FFT size sufficient for both convolution (2n‑1) and correlation
    N = _next_pow2(3 * n - 2)

    best_local_score = float("inf")
    best_local_p = p.copy()
    last_improve_iter = 0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- convolution and current objective ----
        conv = _full_convolution_fft(p, N)
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_local_score - 1e-12:
            best_local_score = cur_score
            best_local_p = p.copy()
            last_improve_iter = it
            print(f"[len {n:5d} iter {it:7d}] new local best = {cur_score:.6f}")

        # ---- soft‑max weights (smooth max) ----
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / beta_len)
        w = np.exp(beta * (conv - M))          # 0 < w_k ≤ 1

        # ---- correlation = sub‑gradient of smoothed max ----
        rev = _weighted_rev_fft(p, w, N)       # size n

        # ---- Adam update in log‑space (mirrored) ----
        # the true sub‑gradient of the (unsmoothed) max is rev;
        # we use rev directly.
        grad = rev
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (similar to the EG schedule)
        lr0 = 0.5
        lr = lr0 / (1.0 + it * 1e-4)

        logp -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 5000 == 0:
            logp += np.random.randn(n) * 0.001

        # ---- projection back onto the simplex ----
        # exponentiate after shifting for numerical stability
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)
        p = exp_shift / exp_shift.sum()

        # early exit if no improvement for a long stretch
        if it - last_improve_iter > 60_000:
            break

    return best_local_p, best_local_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load the previously discovered construction if available
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        # fallback: uniform vector of modest length
        base = np.full(500, 1.0)

    n0 = base.shape[0]
    S0 = math.sqrt(2.0 * n0)
    # Convert to probability vector (sum = 1)
    p_best = base / S0
    p_best = np.maximum(p_best, 0.0)
    if p_best.sum() == 0:
        p_best = np.full(n0, 1.0 / n0)
    else:
        p_best = p_best / p_best.sum()

    best_seq = (p_best * S0).tolist()
    best_score = evaluate_sequence(best_seq)
    best_n = n0
    print(f"[init] length={n0}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Multi‑scale schedule (lengths up to 8000)
    length_schedule = [500, 1000, 1500, 2100, 3000, 5000, 8000]
    length_schedule = [l for l in length_schedule if 300 <= l <= 8000]
    length_schedule = sorted(set(length_schedule))

    for n in length_schedule:
        if time.time() >= deadline:
            break

        # ---- prepare initial probability vectors for this length ----
        # 1) Interpolated from current global best
        p_interp = _interpolate_p(p_best, n)

        # 2) Random Dirichlet start (one extra restart)
        p_random = np.random.rand(n)
        p_random = p_random / p_random.sum()

        # List of initial points we will try for this length
        inits = [p_interp, p_random]

        for idx, p0 in enumerate(inits):
            if time.time() >= deadline:
                break

            print(f"[run] length={n}, restart={idx + 1}")

            # neighbour‑specific time slice (roughly half of remaining time)
            sub_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.5)

            p_opt, local_score = _optimize_adam_length(
                n=n,
                p_start=p0,
                deadline=sub_deadline,
                max_iters=200_000,
            )

            # ----- evaluate the candidate with the official scorer -----
            S = math.sqrt(2.0 * n)
            a_opt = p_opt * S
            cur_score = evaluate_sequence(a_opt.tolist())

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_opt.tolist()
                best_n = n
                p_best = p_opt.copy()
                print(f"[len {n}] *** improved global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final cheap local search (random perturbations) to polish the result
    # Only run if we still have time left.
    if time.time() < deadline:
        print("[local search] polishing best candidate")
        n = best_n
        S = math.sqrt(2.0 * n)
        p_current = np.array(best_seq) / S  # back to probability vector
        # Pre‑compute FFT size for final loops
        N = _next_pow2(3 * n - 2)
        # Adam‑like small‑step perturbations
        for it in range(1, 60_001):
            if time.time() >= deadline:
                break
            # propose a small random move in log‑space
            logp = np.log(p_current + 1e-30)
            logp += np.random.randn(n) * 0.002  # tiny Gaussian jitter
            # project back to simplex
            max_log = np.max(logp)
            exp_shift = np.exp(logp - max_log)
            p_proposed = exp_shift / exp_shift.sum()

            a_prop = p_proposed * S
            score_prop = evaluate_sequence(a_prop.tolist())
            # accept if better
            if score_prop < best_score:
                best_score = score_prop
                best_seq = a_prop.tolist()
                p_current = p_proposed
                print(f"[local] iter {it:5d} new best = {best_score:.6f}")

    print(f"[done] best score = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq
