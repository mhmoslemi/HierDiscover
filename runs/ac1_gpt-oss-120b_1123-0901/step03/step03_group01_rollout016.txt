        # smoothing

        # clamp to avoid overflow
        #rescaled = np.exp(scaled - np.max(scaled))

"""
Optimiser for the self‑convolution bound

The objective to minimise (after fixing Σa_i = √(2n)) is
    max_k (a * a)[k] .

The algorithm works in three layers:

1. **Outer loop** – many independent restarts with random problem sizes
   (n≈600…3000) and a variety of cheap initial shapes (uniform,
   U‑shaped, power‑law, exponential). The current best vector is kept
   and occasionally reused to seed new restarts.

2. **Smooth phase** – a smooth surrogate of the max‑norm is minimised:
       f(a) = (1/β)·log Σ_k exp(β·(a*a)[k]) .
   For large β this value converges to the true max.  An exponentiated‑
   gradient (multiplicative update) on the simplex is applied; the
   gradient of f is obtained analytically and evaluated by a fast FFT
   convolution.  β is increased slowly (annealing) so that the optimiser
   first sees a soft landscape and then focuses on the peak.

3. **Refinement phase** – a projected sub‑gradient descent (the method
   from the reference solution) cleans up the final few digits.

All operations are O(n log n) and fit comfortably into the 1000 s budget.
The best sequence found is returned as a plain Python list of non‑negative
heights.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: projection onto the simplex {x ≥ 0, Σx = s}.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast convolution via FFT (real input → real output).
# --------------------------------------------------------------------------- #
def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b using the real FFT."""
    n = a.shape[0] + b.shape[0] - 1
    n_fft = 1 << ((n - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    conv = np.fft.irfft(A * B, n=n_fft)[:n]
    return conv


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a * a (full convolution, length 2·n‑1)."""
    return _conv_fft(a, a)


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a (used as the objective after normalisation)."""
    return float(_conv_full_fft(a).max())


# --------------------------------------------------------------------------- #
# Smooth (log‑sum‑exp) optimisation with exponentiated gradient on the simplex.
# --------------------------------------------------------------------------- #
def _smooth_eg_optimize(a_init: np.ndarray,
                        target_sum: float,
                        max_time: float,
                        start_time: float) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth surrogate of max_k (a*a)[k] using
    exponentiated gradient (multiplicative) updates.
    """
    a = a_init.copy()
    n = a.shape[0]

    # beta controls the softness of the max; start small, anneal upwards.
    beta = 1.0
    beta_max = 30.0
    # learning‑rate schedule (decreases with iteration)
    eta0 = 0.5

    best_a = a.copy()
    best_score = _max_conv(a)

    it = 0
    while time.time() < start_time + max_time:
        # --------------------------------------------------------------- #
        # 1) compute convolution and the softmax over its entries
        # --------------------------------------------------------------- #
        conv = _conv_full_fft(a)                     # length 2n‑1
        m = conv.max()
        # softmax of β·conv, shifted by the maximum for numerical stability
        shifted = beta * (conv - m)
        # avoid overflow/underflow
        max_shift = np.max(shifted)
        exp_shifted = np.exp(shifted - max_shift)
        p = exp_shifted / exp_shifted.sum()          # probabilities, sum = 1

        # --------------------------------------------------------------- #
        # 2) gradient of the smooth objective:
        #    ∇f_i = 2·β· Σ_k p_k·a_{k‑i}
        #    which is a convolution of p with a, sampled appropriately.
        # --------------------------------------------------------------- #
        # full convolution p * a (length 3n‑2)
        conv_full = _conv_fft(p, a)                 # shape 3n‑2
        # extract the slice that corresponds to Σ_k p_k·a_{k‑i}
        grad = 2.0 * beta * conv_full[(n - 1):(n - 1 + n)]

        # --------------------------------------------------------------- #
        # 3) multiplicative update on the simplex (exponentiated gradient)
        # --------------------------------------------------------------- #
        eta = eta0 / np.sqrt(it + 1)                 # diminishing step size
        # guard against overflow in the exponent
        step = -eta * grad
        # clamp the exponent to a reasonable interval
        step = np.clip(step, -30, 30)
        a_new = a * np.exp(step)

        # renormalise to the required ℓ₁‑norm
        sum_a = a_new.sum()
        if sum_a == 0.0:                             # extremely unlikely
            a_new = np.full_like(a, target_sum / n)
        else:
            a_new = a_new * (target_sum / sum_a)

        a = a_new

        # --------------------------------------------------------------- #
        # 4) bookkeeping
        # --------------------------------------------------------------- #
        cur_score = _max_conv(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] EG new best = {best_score:.6f} (n={n})")

        # increase β slowly (focus more on the true max)
        if beta < beta_max:
            beta *= 1.015

        it += 1

    return best_a, best_score


# --------------------------------------------------------------------------- #
# Projected sub‑gradient descent (same idea as the reference implementation).
# --------------------------------------------------------------------------- #
def _subgradient_optimize(a_init: np.ndarray,
                          target_sum: float,
                          max_time: float,
                          start_time: float) -> tuple[np.ndarray, float]:
    """Refine a vector using a simple projected sub‑gradient method."""
    a = a_init.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = _max_conv(a)

    base_lr = 0.2
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    while time.time() < start_time + max_time:
        conv = _conv_full_fft(a)
        max_val = conv.max()
        cur_score = float(max_val)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] GD new best = {best_score:.6f} (n={n})")

        # Active constraints (within machine epsilon of the maximum)
        eps = 1e-12
        active = np.where(conv >= max_val - eps)[0]

        # Gradient of the sum of the active convolution entries
        grad = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]

        max_grad = grad.max()
        if max_grad <= 0.0:
            break

        eta = base_lr * max_val / max_grad
        momentum = beta_mom * momentum - eta * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # tiny random shake to avoid stagnation
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            base_lr *= 0.9

        a = a_new
        base_lr *= 0.99995   # mild decay

    return best_a, best_score


# --------------------------------------------------------------------------- #
# Candidate generation – several parametric families.
# --------------------------------------------------------------------------- #
def _make_initial_vector(n: int, kind: str, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n belonging to a simple
    parametric family.  The result is *not* normalised.
    """
    idx = np.arange(n, dtype=float)

    if kind == "uniform":
        return np.ones(n, dtype=float)

    if kind == "u_shape":
        # The classic 1/(i+1)+1/(n-i) used in the reference code.
        w = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
        return w

    if kind == "power":
        p = rng.uniform(0.1, 2.0)
        w = (idx + 1.0) ** (-p) + (n - idx) ** (-p)
        return w

    if kind == "exp":
        p = rng.uniform(0.1, 1.5)
        w = np.exp(-p * idx) + np.exp(-p * (n - 1 - idx))
        return w

    # default – slight random perturbation of a uniform vector
    w = np.ones(n, dtype=float) + rng.normal(scale=0.02, size=n)
    w[w < 0] = 0.0
    return w


# --------------------------------------------------------------------------- #
# Main entry point expected by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    Returns a plain Python list of non‑negative heights.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5      # safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")               # this stores max(conv) after normalisation

    # ------------------------------------------------------------------- #
    # Helper to evaluate a candidate (fast, no extra clipping)
    # ------------------------------------------------------------------- #
    def _evaluate(a: np.ndarray) -> float:
        return _max_conv(a)

    # ------------------------------------------------------------------- #
    # Outer search loop – many restarts.
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        # -------- choose a problem size -------------------------------- #
        if best_seq is not None and rng.random() < 0.2:
            n = len(best_seq)                     # occasionally reuse the current best length
        else:
            n = rng.integers(600, 3001)           # 600 … 3000  (larger n gives a better theoretical factor)

        target_sum = np.sqrt(2.0 * n)              # Σ a_i = √(2n)

        # -------- pick a parametric family for initialisation --------- #
        families = ["uniform", "u_shape", "power", "exp"]
        kind = rng.choice(families)

        # optionally seed from the best known construction (if it exists)
        if "height_sequence_1" in globals() and rng.random() < 0.4:
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                a0 = src[:n].copy()
            else:
                # pad the known sequence with a tiny random tail
                tail = rng.random(n - src.size) * 1e-3
                a0 = np.concatenate([src, tail])
        else:
            a0 = _make_initial_vector(n, kind, rng)

        # project onto the simplex (enforces Σ a_i = target_sum)
        a0 = _project_onto_simplex(a0.astype(float), target_sum)

        # quick evaluation of the raw start point
        cur_score = _evaluate(a0)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a0.copy()
            print(f"[{time.time():.2f}] Init best = {best_score:.6f} (n={n}, kind={kind})")

        # --------------------------------------------------------------- #
        # Allocate time for this restart (≈ 20 s split 10 s EG / 10 s GD)
        # --------------------------------------------------------------- #
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        phase_time = min(10.0, time_left * 0.5)

        # ---- Phase 1 : smooth exponentiated‑gradient optimisation ---- #
        a_eg, score_eg = _smooth_eg_optimize(a0, target_sum,
                                            phase_time, time.time())
        if score_eg < best_score:
            best_score = score_eg
            best_seq = a_eg.copy()
            print(f"[{time.time():.2f}] EG best = {best_score:.6f} (n={n})")

        # ---- Phase 2 : projected sub‑gradient refinement -------------- #
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        phase_time = min(10.0, time_left)

        a_gd, score_gd = _subgradient_optimize(a_eg, target_sum,
                                               phase_time, time.time())
        if score_gd < best_score:
            best_score = score_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.2f}] GD best = {best_score:.6f} (n={n})")

        # optional early exit if we already beat the target bound
        if best_score < 1.503:
            print(f"[{time.time():.2f}] Target bound reached (score={best_score:.6f})")
            break

    # ------------------------------------------------------------------- #
    # Fallback – uniform vector if nothing was found (should never happen).
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 1500
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list as required by the grader.
    return best_seq.tolist()
