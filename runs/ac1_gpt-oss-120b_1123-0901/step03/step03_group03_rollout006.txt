"""
Two‑phase Adam optimiser for the self‑convolution max problem.

* Phase 1 – smooth optimisation:
  The hard “max” in the objective is replaced by a log‑sum‑exp
  (soft‑max) with an annealed temperature τ.  The gradient of the
  smoothed objective is a weighted reverse convolution which can be
  obtained with a single FFT.  Adam (β₁=0.9, β₂=0.999) updates the
  coefficients and the iterate is projected onto the feasible set
  (a≥0, Σa = √(2 n)).

* Phase 2 – exact sub‑gradient descent:
  After τ has become very small the algorithm switches to a genuine
  sub‑gradient of the max (a single index of the convolution is
  active).  The same Adam machinery is used but the weight vector is a
  one‑hot vector located at the current maximal convolution index.

* Multiple restarts:
  For each chosen length we start from an interpolated copy of the
  previous best construction and run several Adam instances with
  different learning‑rates/temperature schedules.  The best sequence
  found over the whole run (measured by the exact objective
  ``evaluate_sequence``) is returned.

The code respects the 1000 s wall‑clock limit: the optimiser stops as
soon as the deadline is reached and always returns the best candidate
encountered so far.
"""

import time
import math
import numpy as np


def _softmax_weights(conv: np.ndarray, tau: float):
    """
    Stable soft‑max of the convolution values.

    Returns
    -------
    w : np.ndarray
        Weight vector (sums to 1) – the derivative of the smoothed max.
    Ltau : float
        The smoothed maximum τ·log∑exp(conv/τ).
    """
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, conv.max()

    maxc = conv.max()
    # subtract the maximum for numerical stability
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """
    Compute the weighted reverse convolution

        rev_i = Σ_k w_k · a_{k‑i}

    (indices outside 0…len(a)‑1 are ignored) using FFTs.
    """
    # FFT of a (zero‑padded)
    A_fft = np.fft.rfft(a, n=Nfft)
    # FFT of the reversed weight vector
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    # Convolution of w_rev with a
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]          # keep valid part
    # Extract the slice of length |a| that corresponds to the definition
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _project_onto_simplex(a: np.ndarray, target_sum: float):
    """
    Project a onto the feasible set {x≥0, Σx = target_sum}.
    Simple scaling works because we never need a sparse vertex;
    we just clip negatives and renormalise.
    """
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    else:
        # fallback to uniform if everything vanished
        a = np.full_like(a, target_sum / a.shape[0])
    return a


def _adam_step(a, grad, m, v, iteration, lr0):
    """
    One Adam update (bias‑corrected) for the variable `a`.
    """
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad * grad)

    m_hat = m / (1.0 - beta1 ** iteration)
    v_hat = v / (1.0 - beta2 ** iteration)

    lr = lr0 / (1.0 + 1e-4 * iteration)   # mild decay
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    return a_new, m, v


def _optimize_length_two_phase(
    n: int,
    init_a: np.ndarray,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    max_iters: int,
    lr0: float,
    tau_factor: float,
    decay_rate: float,
):
    """
    Adam optimiser with a soft‑max phase followed by a hard‑max phase.
    The exact objective (max self‑convolution) is used for bookkeeping.
    """
    target_sum = math.sqrt(2.0 * n)

    # start from the supplied vector and enforce the correct L1‑norm
    a = _project_onto_simplex(init_a.copy(), target_sum)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # FFT size (next power‑of‑two ≥ 2·n‑1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    phase_cut = max_iters // 2          # switch after half the budget
    jitter_interval = 4000

    iteration = 0
    while time.time() < deadline and iteration < max_iters:
        iteration += 1

        # ----- full self‑convolution  (via FFT) -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- choose smoothing / hard max -----
        if iteration <= phase_cut:
            # soft‑max smoothing
            tau = max(1e-12, tau_factor * M * math.exp(-iteration / decay_rate))
            w, _ = _softmax_weights(conv, tau)
        else:
            # exact max – one‑hot weight
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0

        # ----- weighted reverse convolution (gradient) -----
        rev = _weighted_rev(a, w, Nfft)          # shape (n,)
        grad = 2.0 * rev                         # ∇_a L (or ∇ max)

        # ----- Adam update -----
        a, m, v = _adam_step(a, grad, m, v, iteration, lr0)

        # ----- projection onto feasible set -----
        a = _project_onto_simplex(a, target_sum)

        # ----- occasional jitter to escape plateaus -----
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_onto_simplex(a, target_sum)

        # ----- keep the best exact solution seen so far -----
        cur_score = M                     # after normalisation objective = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:8d}] new best = {best_score:.6f}")

    return best_seq, best_score


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the given time budget.

    The search proceeds over a few different lengths (original size and
    several enlargements).  For each length several Adam runs are launched
    with different hyper‑parameters.  Each run consists of a soft‑max
    phase followed by a hard‑max phase.  The best sequence found overall
    is returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load the last known construction, otherwise fall back to a uniform vector
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        base = np.ones(2100, dtype=float)

    n0 = base.shape[0]
    target_sum0 = math.sqrt(2.0 * n0)
    base = np.maximum(base, 0.0)
    base = base * (target_sum0 / base.sum())

    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n0}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore (original + enlargements).  Upper bound keeps FFTs cheap.
    length_factors = [1.0, 1.25, 1.5, 2.0]
    lengths = sorted({int(n0 * f) for f in length_factors})
    lengths = [max(300, min(l, 12000)) for l in lengths]   # clamp to a reasonable range
    lengths = sorted(set(lengths))

    # ------------------------------------------------------------------
    # Hyper‑parameter configurations for the Adam runs.
    hp_grid = [
        {"lr0": 0.08, "tau_factor": 0.5,  "decay_rate": 20000.0},
        {"lr0": 0.04, "tau_factor": 0.25, "decay_rate": 30000.0},
        {"lr0": 0.02, "tau_factor": 0.10, "decay_rate": 50000.0},
    ]

    # ------------------------------------------------------------------
    # Main optimisation loop
    for n in lengths:
        if time.time() >= deadline:
            break

        # --- initialise a shape of the required length (linear interpolation) ---
        if n == n0:
            init_base = base.copy()
        else:
            old_x = np.linspace(0.0, 1.0, n0)
            new_x = np.linspace(0.0, 1.0, n)
            init_base = np.interp(new_x, old_x, base)
            init_base = np.maximum(init_base, 0.0)

        target_sum = math.sqrt(2.0 * n)
        init_base = init_base * (target_sum / init_base.sum())

        # --- run several Adam instances with different hyper‑parameters ---
        for hp in hp_grid:
            if time.time() >= deadline:
                break

            # allocate a rough iteration budget based on remaining time
            time_left = deadline - time.time()
            # Empirical cost per iteration grows roughly linearly with n
            est_iter_time = 1.8e-4 * (n / 2100.0)          # seconds per iteration (rough)
            max_iters = int(min(5_000_000, max(1_000, time_left / est_iter_time * 0.3)))
            max_iters = max(max_iters, 1_000)              # never less than a thousand

            print(f"[run] length={n}, hp={hp}, max_iters≈{max_iters}")

            best_seq, best_score = _optimize_length_two_phase(
                n=n,
                init_a=init_base,
                best_seq=best_seq,
                best_score=best_score,
                deadline=deadline,
                max_iters=max_iters,
                lr0=hp["lr0"],
                tau_factor=hp["tau_factor"],
                decay_rate=hp["decay_rate"],
            )

    # ------------------------------------------------------------------
    # Final sanity check (should never happen)
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # Return a plain Python list for the harness
    return best_seq.tolist()
