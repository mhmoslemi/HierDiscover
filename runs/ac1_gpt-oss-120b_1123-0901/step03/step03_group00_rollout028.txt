def _next_power_of_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << int(np.ceil(np.log2(x)))


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    n = a.shape[0]
    start = n - 1
    return corr[start:start + n]  # length n


def _fast_objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a)/(sum(a)²) using FFT.
    a is assumed non‑negative and already clipped to [0,1000].
    """
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:                     # safety – should never happen
        return np.inf
    return 2.0 * n * max_c / (sum_a * sum_a)
