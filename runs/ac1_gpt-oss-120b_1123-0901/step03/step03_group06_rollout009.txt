"""
Improved exponential‑gradient (mirror‑descent) optimiser for the
self‑convolution L∞ problem.

* Works with a probability vector p (Σp = 1).  The actual step heights
  are a = sqrt(2·n)·p, which makes the evaluate_sequence objective equal
  to 2·n·max(p∗p).

* Optimisation uses a smoothed max (log‑sum‑exp) with a β schedule that
  grows from 0.1 to a large value (200).  In the final phase we switch to
  the exact sub‑gradient of the max (indicator weights on the active
  indices).  The gradient of the smoothed objective is a correlation of
  p with the (normalised) weight vector; this correlation is obtained
  efficiently with a single FFT pair.

* A simple exponentiated‑gradient update (mirror descent on the simplex)
  is performed in log‑probability space, with a decaying step‑size.
  Periodic jitter helps escape plateaus.

* A multiscale schedule up‑samples the current best solution to larger
  lengths.  For each length we run a generous number of iterations;
  after every length the exact evaluate_sequence() is called and the
  best sequence is kept globally.

* The routine is fully deterministic given the NumPy RNG seed, but a
  few random restarts are performed at the smallest length to diversify
  the search.

The function ``propose_candidate`` respects the 1000 s time budget and
returns the best step‑function found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(p: np.ndarray, N: int) -> np.ndarray:
    """
    Full self‑convolution p∗p using a single FFT of size N.
    Returns the real array of length 2·len(p)‑1.
    """
    P = np.fft.rfft(p, n=N)
    conv = np.fft.irfft(P * P, n=N)
    n = p.shape[0]
    return conv[: 2 * n - 1]


def _correlation_fft(p: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k·p_{k‑i} for i = 0…n‑1.
    This is the correlation of p with w (w length = 2·n‑1) obtained via FFT.
    """
    n = p.shape[0]
    L = w.shape[0]                 # = 2·n‑1
    # FFT of p
    P = np.fft.rfft(p, n=N)
    # FFT of reversed weight vector
    W_rev = np.fft.rfft(w[::-1], n=N)
    # Convolution of p with reversed w
    conv_rev = np.fft.irfft(P * W_rev, n=N)
    # Keep the linear part (length n+L‑1)
    conv_rev = conv_rev[: n + L - 1]
    # Extract the needed slice and reverse
    rev = conv_rev[L - n : L][::-1]
    return rev


def _optimize_eg_length(
    n: int,
    p_start: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 0.1,
    beta_end: float = 200.0,
    jitter_interval: int = 5_000,
    jitter_scale: float = 0.005,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimisation for a fixed length ``n``.
    Returns the best probability vector ``p`` (sum = 1) and its
    objective value 2·n·max(p∗p) (i.e. the value that evaluate_sequence
    would return for the corresponding step heights).
    """
    p = p_start.copy()
    # work in log‑space to keep the simplex constraints
    logp = np.log(p + 1e-30)

    # FFT size that accommodates both the self‑convolution (2n‑1) and
    # the correlation with a weight vector (3n‑2).
    N = _next_pow2(3 * n - 2)

    best_local_score = float("inf")
    best_local_p = p.copy()

    # linear schedule for β during the first part of the run
    beta_len = int(0.8 * max_iters)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        # ----- convolution p∗p -----
        conv = _full_convolution_fft(p, N)          # length 2·n‑1
        M = conv.max()

        # current objective (no scaling needed – this is exactly what
        # evaluate_sequence reports for the corresponding heights)
        cur_score = 2.0 * n * M

        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_p = p.copy()
            print(
                f"[len {n:5d} iter {it:7d}] new local best = {cur_score:.6f}"
            )

        # ----- compute weight vector -----
        if it <= beta_len:
            # smooth max via log‑sum‑exp
            beta = beta_start + (beta_end - beta_start) * (it / beta_len)
            w = np.exp(beta * (conv - M))
            w_sum = w.sum()
            if w_sum == 0.0:               # numerical safety
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
        else:
            # exact sub‑gradient of the max: uniform over all active indices
            eps = 1e-12
            active = conv >= M - eps
            cnt = active.sum()
            if cnt == 0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = active.astype(float) / cnt

        # ----- correlation rev = Σ_k w_k·p_{k‑i} -----
        rev = _correlation_fft(p, w, N)

        # ----- gradient of the smoothed objective -----
        # objective = 2·n·Lβ  ⇒ ∇ = 4·n·rev
        grad = 4.0 * n * rev

        # ----- step size (decaying) -----
        eta0 = 1.0 / (4.0 * n)          # natural magnitude for EG
        eta = eta0 / (1.0 + it * 1e-4)   # simple decay

        # ----- EG update in log‑space -----
        logp -= eta * grad

        # occasional jitter to help escape plateaus
        if jitter_interval > 0 and it % jitter_interval == 0:
            logp += np.random.randn(n) * jitter_scale

        # ----- project back onto the simplex -----
        # exponentiate in a numerically stable way
        max_log = np.max(logp)
        exp_shift = np.exp(logp - max_log)
        p = exp_shift / exp_shift.sum()

    return best_local_p, best_local_score

        cur_score = evaluate_sequence(a_opt.tolist())

        print(
            f"[len {n}] local best = {best_local_score:.6f}, "
            f"exact score = {cur_score:.6f}"
        )
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_opt.copy()
            best_p = best_local_p.copy()
            print(f"[len {n}] improved global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final short refinement on the currently best length (if time permits)
    if time.time() < deadline:
        n = best_seq.shape[0]
        remaining_iters = 80_000
        p_refine, refine_score = _optimize_eg_length(
            n=n,
            p_start=best_p,
            deadline=deadline,
            max_iters=remaining_iters,
            beta_start=0.1,
            beta_end=300.0,
            jitter_interval=2_000,
            jitter_scale=0.003,
        )
        a_refine = p_refine * math.sqrt(2.0 * n)
        refine_score = evaluate_sequence(a_refine.tolist())
        if refine_score < best_score:
            best_score = refine_score
            best_seq = a_refine.copy()
            print(f"[refine] improved global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
