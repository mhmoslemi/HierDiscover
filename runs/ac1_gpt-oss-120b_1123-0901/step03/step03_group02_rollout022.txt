```
def adam_optimize(seq, max_iters, time_limit):
    n = len(seq)
    target_sum = np.sqrt(2.0 * n)
    a = seq.copy()
    # projection onto simplex
    a = project_onto_simplex(a, target_sum)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    lr_initial = 0.5
    iteration = 1
    while iteration <= max_iters and time.time() < time_limit:
        # compute convolution
        conv = full_convolution_fft(a, a)
        max_val = conv.max()
        # compute weights (sharp softmax)
        tau = max_val * max(0.01, 0.5**(iteration/50)) # degrade quickly
        # or top-k approach
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum > 0:
            w = w / w_sum
        # gradient: 2 * (a * w_rev) where '*' denotes convolution
        grad = 2.0 * full_convolution_fft(a, w[::-1])[2*n-2 - np.arange(n)]
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**iteration)
        v_hat = v/(1-beta2**iteration)
        step = lr_initial * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        # ensure nonnegative, project onto simplex
        a_new = np.maximum(a_new, 0.0)
        a_new = project_onto_simplex(a_new, target_sum)
        a = a_new
        iteration += 1
    return a
```

```
    g_scaled = g0 * (S / sum_g0)
    f_new = (1 - t) * f + t * g_scaled
    # ensure nonnegativity
    f_new = np.maximum(f_new, 0)
    return f_new
```

The goal is to minimise  

\[
F(a)=\frac{2\,n\;\max\bigl(a*a\bigr)}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0,
\]

where `a*a` is the ordinary (linear) convolution.  
If we keep the ℓ₁‑norm fixed to `√(2 n)` the factor `2 n / (∑a)²` becomes a constant and minimising `F` is equivalent to minimising the maximal convolution value.

The algorithm uses two complementary optimisation stages

1. **Adam‑based smooth‑max gradient descent** – fast, works on any length, keeps the
   vector on the simplex (non‑negative, fixed ℓ₁‑norm) and rapidly finds a good
   local minimum.

2. **Linear‑programming refinement** – given the current vector `f`,
   we solve  

   \[
   \max\;\sum_j g_j\quad\text{s.t.}\quad f*g\le M,\;g\ge0,
   \]

   where `M = max(f*f)`.  
   The solution `g` spreads the mass as much as possible while respecting the
   current bound.  After rescaling `g` to the required ℓ₁‑norm we blend it with
   the current vector (`f←(1‑t)f + t·g`).  Repeating a few times strictly
   decreases the objective.

The whole search is performed inside the 1000 s budget:

* many short restarts with random lengths and power‑law / exponential initial
  shapes,
* each restart first runs the Adam optimiser,
* if time permits a cheap LP refinement is applied,
* the best sequence over all runs is kept and finally returned.

All helper functions are top‑level (no closures) and only `numpy`, `cvxpy`
(and the provided `evaluate_sequence`) are used.

```python
"""
propose_candidate()

A hybrid optimiser for the step‑function problem.
1. Adam optimiser with a smooth‑max (soft‑max) approximation of max(conv).
   The iterate is always projected onto the simplex {x_i ≥ 0, Σx_i = √(2 n)}.
2. Optional LP refinement (if cvxpy is available):
   maximise the total mass of a new vector g under the current convolution bound.
   The optimiser blends the current vector with the LP solution.
Many random restarts with different lengths and power‑law / exponential
initialisations are performed within the allotted time budget.
The best sequence found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())      # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)        # eliminate tiny negatives
    return conv


# ----------------------------------------------------------------------
# Objective (identical to evaluate_sequence, but cheaper to call)
# ----------------------------------------------------------------------
def _score_of(seq: np.ndarray) -> float:
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    return float(2.0 * n * conv.max() / (np.sum(seq) ** 2))


# ----------------------------------------------------------------------
# Adam optimiser with smooth‑max weighting
# ----------------------------------------------------------------------
def _adam_optimize(init_seq: np.ndarray, time_limit: float) -> np.ndarray:
    """Adam optimiser that minimises the max‑convolution.

    The optimisation runs until the supplied time limit is reached.
    """
    a = init_seq.astype(np.float64)
    n = a.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # keep the iterate on the simplex from the beginning
    a = _project_onto_simplex(a, target_sum)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.6        # base learning rate
    it = 1

    # exponential decay for the soft‑max temperature
    def _temperature(max_val, it):
        # start relatively smooth, end very sharp
        return max(0.001 * max_val, max_val * (0.95 ** (it // 80)))

    while time.time() < time_limit:
        # --- self‑convolution & maximal value
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()

        # --- soft‑max weights
        tau = _temperature(max_val, it)
        shifted = (conv - max_val) / tau
        w = np.exp(shifted)                     # all entries ≤ 1
        w_sum = w.sum()
        if w_sum > 0.0:
            w /= w_sum
        else:                                   # fallback (should never happen)
            w = np.full_like(w, 1.0 / w.size)

        # --- gradient of Σ w_k conv_k
        # grad_i = 2 * Σ_j a_j w_{i+j}
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)      # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)             # positions giving Σ_j a_j w_{i+j}
        grad = 2.0 * conv2[idx]

        # --- Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)                     # enforce non‑negativity
        a = _project_onto_simplex(a, target_sum)   # enforce ℓ₁‑norm

        # stop early if the step became negligible
        if np.linalg.norm(step) < 1e-9:
            break
        it += 1

    return a


# ----------------------------------------------------------------------
# Build the convolution matrix A such that (f * g) = A @ g
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Return the dense matrix A with (A @ g)[k] = (f * g)[k] for a fixed f."""
    n = f.shape[0]
    K = 2 * n - 1
    A = np.zeros((K, n), dtype=np.float64)
    # column i contains the coefficient f[i] on rows i … i+n‑1
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


# ----------------------------------------------------------------------
# LP refinement (maximise total mass under the current convolution bound)
# ----------------------------------------------------------------------
def _lp_refine(seq: np.ndarray,
               time_limit: float,
               max_iters: int = 5,
               blend_t: float = 0.5) -> np.ndarray:
    """Perform a few LP‑based refinements of `seq`.

    Each iteration solves:
        maximise   Σ g_j
        subject to f * g ≤ M   (M = max(f * f)),   g ≥ 0
    The solution is rescaled to Σ g = √(2 n) and blended with `f`.
    """
    try:
        import cvxpy as cp
    except Exception:               # cvxpy not available → skip refinement
        return seq

    n = seq.shape[0]
    target_sum = np.sqrt(2.0 * n)
    a = seq.copy()

    for _ in range(max_iters):
        if time.time() > time_limit:
            break

        # current bound M
        M = float(np.max(np.convolve(a, a)))

        # convolution matrix for the current a
        A = _build_conv_matrix(a)

        # LP: maximise sum(g)  s.t. A @ g ≤ M
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M * np.ones(2 * n - 1)]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        # ECOS is a fast interior‑point solver for LPs
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
        except Exception:
            # any numerical trouble – abort the refinement
            break

        if g.value is None:
            break

        g0 = np.maximum(g.value, 0.0)
        s_g = g0.sum()
        if s_g <= 0.0:
            break

        # rescale to the required ℓ₁‑norm
        g_scaled = g0 * (target_sum / s_g)

        # convex combination (keeps the sum unchanged)
        a = (1.0 - blend_t) * a + blend_t * g_scaled
        a = np.maximum(a, 0.0)  # safety
    return a

        best_seq = np.full(n, target_sum / n, dtype=float)

    # Ensure we return a plain Python list (as required by the evaluator)
    final_score = _score_of(best_seq)
    print(f"[{time.time():.2f}] FINAL best score = {final_score:.6f}, length = {len(best_seq)}")
    return best_seq.tolist()
```
