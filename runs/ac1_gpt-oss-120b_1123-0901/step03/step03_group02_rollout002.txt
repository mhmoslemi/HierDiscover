def lp_refine_sequence(initial, target_sum, time_limit, max_iter=30, t=0.3):
    a = initial.copy()
    a = np.maximum(a, 0.0)
    # project onto simplex of sum = target_sum
    a = _project_onto_simplex(a, target_sum)
    # evaluate conv and M
    M = _full_convolution_fft(a, a).max()
    best_seq = a.copy()
    best_M = M
    iteration = 0
    while iteration < max_iter and time.time() < time_limit:
        n = len(a)
        # build matrix
        A = build_toeplitz_matrix(a)   # SciPy sparse
        # define variable
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M]
        # monotonic decreasing optional?
        # constraints += [g[i] >= g[i+1] for i in range(n-1)]
        obj = cp.Maximize(cp.sum(g))
        prob = cp.Problem(obj, constraints)
        prob.solve(solver='ECOS', warm_start=True, verbose=False, max_iters=200)
        if prob.status not in ["optimal", "optimal_inaccurate"]:
            # fallback to gradient approach
            break
        g0 = np.array(g.value).flatten()
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break
        scale = target_sum / sum_g0
        g_scaled = g0 * scale
        # update a
        a_new = (1 - t) * a + t * g_scaled
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)
        # compute new M
        M_new = _full_convolution_fft(a_new, a_new).max()
        # accept if M decreased
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_seq = a_new.copy()
        a = a_new
        M = M_new
        iteration += 1
    # return best_seq
    return best_seq

"""
Propose a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is a two‑stage optimiser:

1. **LP refinement** – for a fixed length `n` we repeatedly solve a
   linear programme that, given the current sequence `f`, finds a
   sequence `g` with maximal ℓ₁‑norm under the convolution bound
   `f * g ≤ M` (`M` is the current maximum self‑convolution).  The LP
   solution is scaled to satisfy the simplex constraint
   Σ g = √(2 n) and blended with `f`.  This step quickly reduces the
   peak of the convolution.

2. **Smooth‑max gradient descent** – after a few LP iterations we
   polish the result with the soft‑max sub‑gradient method from the
   previous state‑of‑the‑art code.  This fine‑tunes the sequence.

The optimiser runs many independent restarts with random lengths
(≈ 800‑2500) and random power‑law initialisations, occasionally warm‑
starting from the current globally best sequence.  The best sequence
found before the overall time budget expires is returned.

All helper functions are top‑level (no closures) and only standard
Python, NumPy, SciPy and CVXPY are used.
"""

import time
import numpy as np
import scipy.sparse as sp
import cvxpy as cp

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (sum = s, non‑negative)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – Full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power‑of‑2
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # remove tiny negative artefacts caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Helper – Build the (2n‑1)×n Toeplitz matrix for convolution with f
# ----------------------------------------------------------------------
def _build_toeplitz_matrix(f: np.ndarray) -> sp.csr_matrix:
    """Return the matrix A (shape (2n‑1, n)) such that A @ g = f * g."""
    n = f.shape[0]
    # Pre‑allocate flat arrays for rows, cols, data
    nnz = n * n                     # each column contributes n non‑zeros
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    data = np.empty(nnz, dtype=np.float64)

    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n, dtype=np.int32)
        cols[pos:pos + n] = i
        data[pos:pos + n] = f[i]
        pos += n
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# ----------------------------------------------------------------------
# LP refinement – reduce max self‑convolution by a few convex‑combination steps
# ----------------------------------------------------------------------
def _lp_refine_sequence(
    init: np.ndarray,
    target_sum: float,
    time_deadline: float,
    max_iter: int = 12,
    blend_factor: float = 0.32,
) -> np.ndarray:
    """
    Starting from `init` (already on the simplex), repeatedly solve
    the linear programme:

        maximise   Σ g_i
        subject to 0 ≤ g_i,
                    f * g ≤ M,

    where `M` is the current maximum of `f*f`.  The LP solution is
    scaled to satisfy Σ g_i = target_sum and blended with `f`.
    Returns the best sequence discovered within the allocated time.
    """
    a = np.maximum(init, 0.0)
    a = _project_onto_simplex(a, target_sum)

    best_seq = a.copy()
    best_M = _full_convolution_fft(a, a).max()
    iteration = 0

    while iteration < max_iter and time.time() < time_deadline:
        n = a.shape[0]

        # Build convolution matrix for the current a
        A = _build_toeplitz_matrix(a)

        # Linear programme
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= best_M]
        # (optional) enforce non‑increasing shape – experimentally useful
        # constraints += [g[i] >= g[i + 1] for i in range(n - 1)]

        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver="ECOS", warm_start=True, max_iters=250, verbose=False)
        except Exception:  # fallback to a more robust (but slower) solver
            prob.solve(solver="SCS", max_iters=1000, verbose=False)

        if prob.status not in ("optimal", "optimal_inaccurate"):
            # LP failed – give up this refinement
            break

        g0 = np.asarray(g.value).reshape(-1)
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break

        # Scale to satisfy the simplex constraint
        scale = target_sum / sum_g0
        g_scaled = g0 * scale

        # Convex combination step
        a_new = (1.0 - blend_factor) * a + blend_factor * g_scaled
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # New maximum self‑convolution
        M_new = _full_convolution_fft(a_new, a_new).max()

        # Keep the best sequence seen so far
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_seq = a_new.copy()

        a = a_new
        iteration += 1

    return best_seq


# ----------------------------------------------------------------------
# Soft‑max gradient descent (same idea as the previous state‑of‑the‑art)
# ----------------------------------------------------------------------
def _optimise_one_length_softmax(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 0.001,
    step_factor: float = 0.6,
    step_decay: float = 0.9994,
    momentum_beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """Fine‑tune `init` by smooth‑max sub‑gradient descent."""
    target_sum = np.sqrt(2.0 * n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    best_seq = a.copy()
    conv = _full_convolution_fft(a, a)
    best_score = 2.0 * n * conv.max() / (target_sum ** 2)

    velocity = np.zeros_like(a)
    iteration = 0

    while time.time() < run_deadline:
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()

        # smooth‑max weights
        tau = max(init_tau_factor * max_val * (0.95 ** iteration),
                  min_tau_factor * max_val)
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)                     # entries ≤ 1
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of Σ w_k conv_k  (cross‑correlation)
        w_rev = w[::-1]
        cc = _full_convolution_fft(a, w_rev)   # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)       # pick the required slice
        grad = 2.0 * cc[idx]

        # momentum
        if momentum_beta > 0.0:
            velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
            step_dir = velocity
        else:
            step_dir = grad

        # adaptive step size
        max_grad = np.max(np.abs(step_dir)) + 1e-12
        eta = step_factor * max_val / max_grad

        a_new = a - eta * step_dir
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # evaluate
        conv_new = _full_convolution_fft(a_new, a_new)
        score_new = 2.0 * n * conv_new.max() / (target_sum ** 2)

        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {score_new:.6f}")

        a = a_new
        step_factor *= step_decay
        iteration += 1

        # occasional random shake‑up
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (target_sum * 0.0005 / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            step_factor *= 0.9

    # final evaluation through the provided evaluator (ensures identical rounding)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score

        idx = np.arange(1, n + 1, dtype=float)
        init = idx ** (-p)
        if np.random.rand() < 0.35:
            tau = np.random.uniform(0.4, 2.0) * n
            init *= np.exp(-idx / tau)
        init += np.random.rand(n) * 1e-4   # tiny jitter
        return init

    # ------------------------------------------------------------------
    # Main restart loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 5.0:

        # Pick a problem size (larger early, smaller later)
        if remaining > 400:

    return best_seq.tolist()
