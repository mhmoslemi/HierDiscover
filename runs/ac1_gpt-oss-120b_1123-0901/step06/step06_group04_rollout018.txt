def generate_seeds(n, best_seq):
    seeds = []
    # 1 uniform
    seeds.append(np.full(n, 1.0/n))
    # 2 warm start
    if best_seq is not None and len(best_seq) == n:
        seeds.append(np.asarray(best_seq, dtype=np.float64))
    # 3 up/down sample from best of other size
    if best_seq is not None and len(best_seq) != n:
        old = np.asarray(best_seq, dtype=np.float64)
        x_old = np.linspace(0, 1, old.shape[0])
        x_new = np.linspace(0, 1, n)
        up = np.interp(x_new, x_old, old)
        if up.sum() > 0:
            up = up / up.sum()
        else:
            up = np.random.dirichlet(np.ones(n))
        seeds.append(up)
    # 4 random Dirichlet draws
    for _ in range(2):
        seeds.append(np.random.dirichlet(np.ones(n)))
    # 5 Beta-shaped distributions
    x = np.linspace(0, 1, n)
    for alpha in [0.3, 0.5, 0.8, 1.2, 2.0]:
        shape = np.power(x * (1 - x) + 1e-12, alpha - 1.0)
        shape = shape / shape.sum()
        seeds.append(shape)
    # 6 symmetric edge gaussian shapes
    i = np.arange(n)
    for gamma in [0.1, 0.3, 0.8]:
        shape = np.exp(-gamma * (i**2)) + np.exp(-gamma * ((n-1-i)**2))
        shape = shape / shape.sum()
        seeds.append(shape)
    # limit number of seeds based on n resources
    max_seeds = 12 if n <= 8192 else 8
    return seeds[:max_seeds]

def jitter_and_optimize(a_base, n, conv_fft_len, corr_fft_len, deadline, jitter_runs=5):
    best_a = a_base.copy()
    best_val = compute_value(best_a)
    for i in range(jitter_runs):
        if time.time() > deadline:
            break
        logits = np.log(np.maximum(best_a, 1e-12))
        logits += np.random.normal(scale=0.02, size=n)
        a0 = np.exp(logits - np.max(logits))
        a0 = a0 / a0.sum()
        a_adam, val_adam = _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline)
        a_ref, val_ref = _subgradient_refine(a_adam, n, conv_fft_len, corr_fft_len, deadline)
        if val_ref < best_val:
            best_val = val_ref
            best_a = a_ref.copy()
    return best_a, best_val

    supplied wall‑clock limit (default 1000 s).

The code is pure NumPy, uses only a few FFTs per iteration
(convolution and a cross‑correlation), and respects the “no‑closure”
and “no‑IO” restrictions.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a*a (zero‑padded) using an FFT length that is a power
    of two. Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]  for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Utility
# ----------------------------------------------------------------------


def _objective_value(a: np.ndarray) -> float:
    """
    Compute the exact evaluation function for a (assumes a is non‑negative).
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_c = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (sum_a ** 2))


# ----------------------------------------------------------------------
# Adam optimisation on a smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp surrogate of max(conv).
    Returns the best vector encountered and its exact objective value.
    """
    eps = 1e-16
    a = np.maximum(a0, eps)
    a /= a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.04

    lam = 0.5               # initial temperature
    lam_factor = 1.03

    # iteration budget grows for smaller vectors
    max_iters = int(80000 * 1024 / n) + 3000
    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # -------- soft‑max to stay on the simplex ----------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # -------- smooth‑max weights ----------
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0.0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # -------- gradient of smooth‑max ----------
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # -------- back‑prop through soft‑max ----------
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # -------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # -------- temperature annealing ----------
        lam = min(lam * lam_factor, 1e9)

        # -------- exact objective evaluation ----------
        val = 2.0 * n * max_c          # sum(a) == 1 by construction
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # -------- stagnation escape ----------
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact ℓ∞ sub‑gradient refinement (exponentiated‑gradient)
# ----------------------------------------------------------------------


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Precise sub‑gradient descent on the true max‑ℓ∞ term, using an
    exponentiated‑gradient update that respects non‑negativity and the simplex.
    """
    a = a0 / a0.sum()
    best_val = float("inf")
    best_a = a.copy()
    eta = 0.75                     # initial step size
    last_improve = time.time()

    for it in range(6000):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        # active set (ties are possible due to floating point)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # fallback to a fresh Dirichlet draw
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # objective
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # step‑size decay
        eta *= 0.9998

        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Seed generation (diverse starting points)
# ----------------------------------------------------------------------


def _generate_seeds(n: int, best_seq) -> list[np.ndarray]:
    """Return a list of candidate start vectors for length ``n``."""
    seeds = []

    # 1) uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) warm‑start from exact‑size best
    if best_seq is not None and len(best_seq) == n:
        seeds.append(np.asarray(best_seq, dtype=np.float64))

    # 3) up‑/down‑sample a different‑size best
    if best_seq is not None and len(best_seq) != n:
        old = np.asarray(best_seq, dtype=np.float64)
        x_old = np.linspace(0.0, 1.0, old.shape[0])
        x_new = np.linspace(0.0, 1.0, n)
        up = np.interp(x_new, x_old, old)
        if up.sum() > 0:
            up = up / up.sum()
        else:
            up = np.random.dirichlet(np.ones(n))
        seeds.append(up)

    # 4) historic global variable (if present)
    hseq = globals().get("height_sequence_1")
    if hseq is not None:
        h = np.asarray(hseq, dtype=np.float64)
        if h.shape[0] > n:
            h = h[:n]
        elif h.shape[0] < n:
            h = np.pad(h, (0, n - h.shape[0]))
        h = np.maximum(h, 0.0)
        if h.sum() > 0:
            h = h / h.sum()
        else:
            h = np.random.dirichlet(np.ones(n))
        seeds.append(h)

    # 5) a handful of random Dirichlet draws
    for _ in range(3):
        seeds.append(np.random.dirichlet(np.ones(n)))

    # 6) symmetric beta‑shaped (U‑ or mound‑shaped)
    x = np.linspace(0.0, 1.0, n)
    for alpha in [0.3, 0.5, 0.8, 1.2, 2.0]:
        shape = np.power(x * (1.0 - x) + 1e-12, alpha - 1.0)
        shape = (shape + shape[::-1]) * 0.5          # enforce symmetry
        if shape.sum() > 0:
            shape = shape / shape.sum()
            seeds.append(shape)

    # 7) edge‑Gaussian mixtures
    i = np.arange(n, dtype=np.float64)
    for gamma in [0.1, 0.3, 0.6, 1.0]:
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape = shape / shape.sum()
        seeds.append(shape)

    # 8) smooth sine‑squared bump
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    shape = shape / shape.sum()
    seeds.append(shape)

    # 9) cosine‑based mixture (center heavy)
    beta = 3.0
    shape = np.exp(beta * np.cos(2.0 * np.pi * np.arange(n) / n))
    shape = shape / shape.sum()
    seeds.append(shape)

    # ------------------------------------------------------------------
    # Trim to a reasonable number (larger n → fewer candidates)
    # ------------------------------------------------------------------
    max_cand = 12 if n <= 8192 else 8
    return seeds[:max_cand]


# ----------------------------------------------------------------------
# Local jitter + re‑optimisation around a current best vector
# ----------------------------------------------------------------------


def _jitter_and_refine(
    a_base: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    jitter_runs: int = 8,
) -> tuple[np.ndarray, float]:
    """
    Apply several small random perturbations to the logarithms of ``a_base``,
    re‑run the two‑phase optimiser and keep the best improvement.
    """
    best_a = a_base.copy()
    best_val = _objective_value(best_a)

    for j in range(jitter_runs):
        if time.time() > deadline:
            break

        # perturb logits (log‑space) – a small Gaussian noise
        logits = np.log(np.maximum(best_a, 1e-12))
        logits += np.random.normal(scale=0.025, size=n)
        a0 = np.exp(logits - np.max(logits))
        a0 = a0 / a0.sum()

        a_adam, val_adam = _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline)
        a_ref, val_ref = _subgradient_refine(a_adam, n, conv_fft_len, corr_fft_len, deadline)

        for cand_a, cand_val in [(a_adam, val_adam), (a_ref, val_ref)]:
            if cand_val < best_val:
                best_val = cand_val
                best_a = cand_a.copy()
                print(f"[n={n} jitter={j}] new local best {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the supplied
    ``evaluate_sequence`` objective.  The routine respects the given
    wall‑clock budget (seconds) and returns the best vector discovered
    as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    best_seq = None
    best_val = float("inf")

    # progression of lengths – start small, finish with the largest
    lengths = [2048, 4096, 8192, 16384, 32768, 65536]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

        # --------------------------------------------------------------
        # Build a heterogeneous pool of seeds
        # --------------------------------------------------------------
        seed_list = _generate_seeds(n, best_seq)

        # --------------------------------------------------------------
        # Optimise every seed
        # --------------------------------------------------------------
        for sid, seed_vec in enumerate(seed_list):
            if time.time() > deadline:
                break

            try:
                a_adam, val_adam = _adam_optimize(
                    seed_vec, n, conv_fft_len, corr_fft_len, deadline
                )
                a_ref, val_ref = _subgradient_refine(
                    a_adam, n, conv_fft_len, corr_fft_len, deadline
                )

                # keep the best of the two phases
                for cand_a, cand_val in [(a_adam, val_adam), (a_ref, val_ref)]:
                    if cand_val < best_val:
                        best_val = cand_val
                        best_seq = cand_a.copy()
                        print(
                            f"[n={n} seed={sid}] new global best {best_val:.6f}"
                        )
            except Exception as exc:  # pragma: no cover – defensive
                print(f"[n={n} seed={sid}] exception: {exc}")

        # --------------------------------------------------------------
        # Local jitter around the current best of this length
        # --------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            best_seq, best_val = _jitter_and_refine(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                jitter_runs=10,
            )

    # ------------------------------------------------------------------
    # Fallback safety net
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    print(
        f"\nFinished search – best value {best_val:.6f} with length {len(best_seq)}"
    )
    return best_seq.tolist()
