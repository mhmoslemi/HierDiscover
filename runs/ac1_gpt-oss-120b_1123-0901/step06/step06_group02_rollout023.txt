"""
Hybrid optimisation for step‑function bound.

- Phase 1: large set of random parametric shapes (exponential, power, beta,
  piecewise constant, spikes, mixtures) and normalisation to L1 = sqrt(2 n).
- Phase 2: Adam on a smooth log‑sum‑exp surrogate of the max‑convolution.
- Phase 3: Projected sub‑gradient descent on the exact max‑convolution.
- Phase 4: Simulated‑annealing hill‑climb with adaptive temperature and
  larger mass‑transfer moves.

All phases respect the time budget (default 1000 s) and keep a global best
solution which is returned as a plain list of floats.
"""

import time
import numpy as np

# fft convolution
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)
    return conv

# simplex projection
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _shape_exp(n: int) -> np.ndarray:
    tau = np.random.uniform(0.05, 2.5) * n
    idx = np.arange(n, dtype=float)
    a = np.exp(-idx / tau)
    return a

def _shape_pow(n: int) -> np.ndarray:
    p = np.random.uniform(0.1, 2.5)
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** (-p)

def _shape_beta(n: int) -> np.ndarray:

def _shape_mixed(n: int) -> np.ndarray:

# ---------- Adam on smoothed max ----------
def _adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 5e-5,
    alpha0: float = 0.03,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0
    while time.time() < deadline:
        # softmax mapping
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        p = exp_shift / exp_shift.sum()
        a = target_sum * p
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam n={n} new best max={cur_max:.6f}")
        # temperature for log‑sum‑exp smoothing
        tau = max(init_tau_factor * cur_max * (0.99 ** it), min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum
        # gradient of smoothed max w.r.t a:
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]          # shape (n,)
        # back‑prop to logits
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)
        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step
        # occasional jitter
        if it % 300 == 0 and np.random.rand() < 0.25:
            x += np.random.randn(n) * 0.001
        alpha *= 0.9995
    final_val = _full_convolution_fft(best_a, best_a).max()
    return best_a, float(final_val)

# ---------- Projected sub‑gradient descent ----------
def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step = 0.3 * target_sum / n  # initial step size, a bit larger
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())
        # gradient of conv_k w.r.t a
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]
        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)
        cand_val = _full_convolution_fft(cand, cand).max()
        if cand_val < best_val - 1e-12:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= 1.03
            if it % 200 == 0:
                print(f"[{time.time():.2f}] subgrad it={it} val={best_val:.6f}")
        else:
            step *= 0.70
            if np.random.rand() < 0.07:
                jitter = np.random.rand(n) * 1e-4
                a = _project_onto_simplex(a + jitter, target_sum)
        # early exit optional
        if best_val < 1.502:
            break
    return best_a, best_val

# ---------- Simulated‑annealing hill‑climb ----------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:

    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    # temperature schedule
    T0 = max(1e-3, best_val * 0.5)
    T_end = 1e-5
    start = time.time()

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))
        # pick source and destination

        if i == j or a[i] <= 0:
            continue
        delta = a[i] * np.random.rand() * 0.12  # up to 12 % of source mass
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_val = _full_convolution_fft(a_new, a_new).max()
        accept = False
        if new_val < best_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True
        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")
        if it % 2500 == 0:
            a = _project_onto_simplex(a, target_sum)
        if best_val < 1.502:

"""
Hybrid optimisation for the step‑function bound.

Workflow
--------
1. Diverse parametric random initialisations (exponential, power‑law,
   Beta, piecewise constant, spikes and mixtures).  Each candidate is
   normalised to Σa = √(2 n) and its peak convolution is recorded.
2. Adam optimiser on a smooth log‑sum‑exp surrogate of the max‑convolution.
3. Projected sub‑gradient descent that follows the exact worst convolution index.
4. Simulated‑annealing hill‑climb with an adaptive temperature schedule.

All stages respect the supplied time budget (default 1000 s).  The
best sequence found is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full linear convolution (real vectors)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)   # clip tiny negatives
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σw = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators – each returns a non‑negative vector of length n
# ----------------------------------------------------------------------
def _shape_exp(n: int) -> np.ndarray:
    tau = np.random.uniform(0.05, 2.5) * n
    idx = np.arange(n, dtype=float)
    a = np.exp(-idx / tau)
    return a


def _shape_pow(n: int) -> np.ndarray:
    p = np.random.uniform(0.1, 2.5)
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** (-p)
    return a


def _shape_beta(n: int) -> np.ndarray:
    a_par = np.random.uniform(0.2, 5.0)
    b_par = np.random.uniform(0.2, 5.0)
    x = (np.arange(n) + 0.5) / n
    a = (x ** (a_par - 1)) * ((1 - x) ** (b_par - 1))
    return a


def _shape_piecewise(n: int) -> np.ndarray:
    blocks = np.random.randint(3, 9)
    boundaries = np.sort(np.random.choice(n - 1, blocks - 1, replace=False)) + 1
    heights = np.random.rand(blocks) + 0.01
    a = np.empty(n, dtype=float)
    start = 0
    for end, h in zip(np.append(boundaries, n), heights):
        a[start:end] = h
        start = end
    return a


def _shape_spike(n: int) -> np.ndarray:
    m = np.random.randint(2, min(30, n // 10) + 1)
    a = np.zeros(n, dtype=float)
    pos = np.random.choice(n, m, replace=False)
    a[pos] = np.random.rand(m) + 0.01
    return a


def _shape_mixed(n: int) -> np.ndarray:
    num_comp = np.random.randint(2, 5)
    comps = []
    for _ in range(num_comp):
        t = np.random.choice(["exp", "pow", "beta"])
        if t == "exp":
            comps.append(_shape_exp(n))
        elif t == "pow":
            comps.append(_shape_pow(n))
        else:
            comps.append(_shape_beta(n))
    weights = np.random.dirichlet(np.ones(num_comp))
    a = sum(w * c for w, c in zip(weights, comps))
    return a

        a = np.maximum(a, 0.0)
    return a

def _adam_softmax(

                                         deadline - time.time() - 30.0)
        seq_local, val_local = _adam_softmax(
            n, x0, target_sum, run_deadline,
            init_tau_factor=0.15,
            min_tau_factor=3e-5,
            alpha0=0.025,
        )
        if val_local < best_val:
            best_val = val_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] after Adam best={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Targeted projected sub‑gradient descent (exact max)
    # --------------------------------------------------------------
    if time.time() < deadline - 80.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_deadline = deadline - 40.0
        seq_sub, val_sub = _subgradient_descent(
            best_seq.copy(),
            target_sum,
            sub_deadline,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑gradient best={best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Final simulated‑annealing hill‑climb (use remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        hill_deadline = deadline
        seq_hill, val_hill = _hillclimb(
            best_seq.copy(),
            target_sum,
            hill_deadline,
        )
        if val_hill < best_val:
            best_val = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] after hill‑climb best={best_val:.6f}")

    # Return as a plain Python list
    return best_seq.tolist()
