def _proj_simplex(v):
    v = np.asarray(v, dtype=float)
    if v.sum() == 0:
        # return uniform distribution
        return np.full_like(v, 1.0/v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]

def _subgradient_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=200000, eta0=0.6):
    a = np.maximum(a0, 0.0)
    s = a.sum()

        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s
    best_a = a.copy()
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # active set
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        a = a * np.exp(-eta * grad_a)
        # projection onto simplex
        a_sum = a.sum()
        if a_sum <= 0:
            a = np.random.dirichlet(np.ones(n))
        else:
            a = a / a_sum
        it += 1
    return best_a, best_val

    conv = _full_convolution(a, conv_fft_len)

        val = 2.0 * n * max_c

            cand.append(shape)
    # exponential decay
    for beta in [0.1, 0.2, 0.4, 0.8]:
        shape = np.exp(-beta * i) + np.exp(-beta * (n-1-i))
        shape /= shape.sum()
        cand.append(shape)
    return cand

def _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=200000):
    eps = 1e-15
    logits = np.log(a0 + eps)   # shape n
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.05
    best_a = a0.copy()
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        # softmax to ensure simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        # compute exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 1000 == 0:
                print(f"[Adam n={n} iter={it}] val={val:.6f}")
        # compute smooth max weights
        # lam may become large; limit to avoid overflow
        # compute diff = conv - max_c  (max zero)
        diff = conv - max_c
        # we can compute exp(lam*diff) safely
        # cap diff*lam at maybe 700 to avoid overflow
        scaled = lam * diff
        # clamp
        scaled = np.clip(scaled, -700, 700)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient of surrogate: 2 * cross_correlation(w, a)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # chain rule through softmax
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # update lam
        lam = min(lam * lam_factor, 1e6)
        # occasional noise
        if it % 5000 == 0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
        it += 1
    return best_a, best_val

assistantfinal```python
"""
Propose a step‑function (non‑negative vector) minimizing

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm works in several stages while respecting a time budget
(default 1000 s).

1. **Coarse‑to‑fine schedule** – start with a small length (256),
   optimise many diverse candidates, then linearly up‑sample the best
   solution to larger lengths (1024 → 4096 → 8192) and refine again.

2. **Diverse initial shapes** – random Dirichlet, edge‑concentrated
   power‑law, double‑Gaussian, sinusoidal half‑wave, cosine‑power,
   exponential tails and “spike” patterns.  A warm‑start from a
   previously found solution (global variable ``height_sequence_1``)
   is also used.

3. **Smooth‑max Adam** – optimises a log‑sum‑exp surrogate of the
   convolution peak.  The surrogate temperature is increased
   gradually so the optimisation eventually follows the true max.

4. **Exact‑max sub‑gradient** – exponentiated‑gradient updates on the
   simplex with a step size that decays as 1/√t.  The gradient is the
   sub‑gradient of the current max‑convolution and is obtained via an
   FFT‑based cross‑correlation.

5. **Frank–Wolfe refinement** – a cheap conditional gradient step
   that moves a fraction of the mass towards the index with the
   smallest sub‑gradient (the vertex that most reduces the peak).

6. **Final long sub‑gradient pass** – after the last up‑sampling a
   longer exponentiated‑gradient run is performed to push the value
   as low as possible.

All convolutions are performed with FFTs (O(n log n)).  The code
always returns a feasible sequence (non‑negative, finite numbers) and
clips it to the evaluator’s allowed range.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # uniform distribution if everything is non‑positive
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Construct a modest pool of diverse start vectors."""
    cand = []

    # warm start and a few rotated copies
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # simple sinusoidal half‑wave
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """‘Spike’ candidates with k equally spaced non‑zero entries."""
    out = []
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    return out


def _cosine_power_candidates(n: int) -> list[np.ndarray]:
    """Cosine‑power and exponential‑tail shapes."""
    out = []
    i = np.arange(n, dtype=np.float64)

    # cosine‑power (positive half‑wave)
    for p in (1, 2, 4):
        shape = np.abs(np.cos(np.pi * (i + 0.5) / n)) ** p
        shape = np.maximum(shape, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            out.append(shape)

    # exponential decay from the edges
    for beta in (0.05, 0.1, 0.2, 0.4):
        shape = np.exp(-beta * i) + np.exp(-beta * (n - 1 - i))
        shape /= shape.sum()
        out.append(shape)

    return out


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.1          # start temperature
    lam_factor = 1.05  # increase factor

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (sum(a)=1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 1000 == 0:
                print(f"[Adam n={n} iter={it:05d}] val={val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        diff = conv - max_c            # non‑positive vector
        scaled = lam * diff
        # clip to avoid overflow in exp
        scaled = np.clip(scaled, -700, 700)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to escape plateaus
        if it % 5000 == 0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient with an adaptive exponentiated‑gradient step.
    Returns best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad n={n} iter={it:05d}] val={val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # adaptive step size η = η0 / sqrt(t+1)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # projection back onto the simplex
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        it += 1

    return best_a, best_val


def _frank_wolfe_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Conditional‑gradient (Frank–Wolfe) refinement.
    Moves mass toward the index with the smallest sub‑gradient.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()

        # sub‑gradient of the current max
        active = np.isclose(conv, max_c, atol=1e-12)
        if not active.any():
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Frank–Wolfe vertex = unit vector at index of minimum gradient
        i_star = int(np.argmin(grad))

        # simple line‑search over a few candidate step sizes
        best_gamma = 0.0
        best_gamma_val = val
        for gamma in (0.1, 0.25, 0.5, 0.75, 1.0):
            a_new = (1.0 - gamma) * a
            a_new[i_star] += gamma
            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv_new.max()
            if val_new < best_gamma_val:
                best_gamma_val = val_new
                best_gamma = gamma

        if best_gamma == 0.0:
            # no improvement – stop early
            break

        # apply the best step
        a = (1.0 - best_gamma) * a
        a[i_star] += best_gamma
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # Size schedule – start small, then progressively refine.
    size_schedule = [256, 1024, 4096, 8192]

    for stage, n in enumerate(size_schedule):
        if time.time() >= deadline:
            break

        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        # warm start from the current best (interpolated) if available
        warm = None
        if best_seq is not None:
            old_n = len(best_seq)
            x_old = np.linspace(0, 1, old_n, endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            warm = np.interp(x_new, x_old, best_seq)

        # ------------------------------------------------------------------
        # Build a pool of starting vectors
        cand_pool = (
            _initial_candidates(n, warm)
            + _spike_candidates(n)
            + _cosine_power_candidates(n)
        )

        # To keep runtime reasonable, limit the number of candidates.
        max_cand = 50 if n <= 256 else 20 if n <= 1024 else 6
        if len(cand_pool) > max_cand:
            np.random.shuffle(cand_pool)
            cand_pool = cand_pool[:max_cand]

        # ------------------------------------------------------------------
        # Optimise each candidate with short runs
        for cid, cand in enumerate(cand_pool):
            if time.time() >= deadline:
                break

            a0 = np.array(cand, dtype=np.float64)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s < 1e-9:
                a0 = np.random.dirichlet(np.ones(n))
                s = a0.sum()
            a0 = a0 / s

            # Give each candidate up to ~5 s (or until the global deadline)
            local_deadline = min(deadline - 2.0, time.time() + 5.0)

            # Adam (smooth‑max) phase
            a_adam, val_adam = _adam_optimize(
                a0, n, conv_fft_len, corr_fft_len, local_deadline, max_iters=100_000
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[size {n} cand {cid}] Adam improved {val_adam:.6f}")

            # Sub‑gradient refinement (short)
            if time.time() < deadline - 2.0:
                sub_deadline = min(deadline - 2.0, time.time() + 5.0)
                a_sub, val_sub = _subgradient_optimize(
                    best_seq, n, conv_fft_len, corr_fft_len, sub_deadline,
                    max_iters=30_000, eta0=0.8
                )
                if val_sub < best_val:
                    best_val = val_sub
                    best_seq = a_sub.copy()
                    print(f"[size {n} cand {cid}] Subgrad improved {val_sub:.6f}")

            # Frank‑Wolfe refinement (very cheap)
            if time.time() < deadline - 2.0:
                fw_deadline = min(deadline - 2.0, time.time() + 3.0)
                a_fw, val_fw = _frank_wolfe_optimize(
                    best_seq, n, conv_fft_len, corr_fft_len, fw_deadline, max_iters=500
                )
                if val_fw < best_val:
                    best_val = val_fw
                    best_seq = a_fw.copy()
                    print(f"[size {n} cand {cid}] FW improved {val_fw:.6f}")

        # ------------------------------------------------------------------
        # After all candidates at this size, perform a longer sub‑gradient
        # pass on the current best (if time permits).
        if best_seq is not None and time.time() < deadline - 5.0:
            long_deadline = deadline - 1.0
            a_long, val_long = _subgradient_optimize(
                best_seq, n, conv_fft_len, corr_fft_len, long_deadline,
                max_iters=200_000, eta0=0.6
            )
            if val_long < best_val:
                best_val = val_long
                best_seq = a_long.copy()
                print(f"[size {n}] Long subgrad improved {val_long:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – ensure we always return a valid sequence
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator's accepted range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final check using the provided evaluator (optional but helpful)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluate_sequence value: {final_val:.6f}")
        best_val = final_val
    except Exception as exc:
        print(f"evaluate_sequence failed: {exc}")

    print(f"Returning best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
