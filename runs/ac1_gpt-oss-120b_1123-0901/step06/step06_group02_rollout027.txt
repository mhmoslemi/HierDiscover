def gradient_from_peaks(a, conv, delta_frac=0.001):
    n = a.shape[0]
    max_val = conv.max()
    thr = max_val - max(delta_frac * max_val, 1e-12)

def random_sequence(n, shape):
    # Return numpy array shape (n,)
    if shape == "exp":
        tau = np.random.uniform(0.1, 2.0) * n
        idx = np.arange(n).astype(float)
        a = np.exp(-idx / tau)
    elif shape == "pow":
        p = np.random.uniform(0.2, 1.5)
        idx = np.arange(1, n+1).astype(float)
        a = idx ** (-p)
    elif shape == "hann":
        i = np.arange(n).astype(float)
        a = 0.5 - 0.5 * np.cos(2*np.pi*i/(n-1))
    elif shape == "hamming":
        i = np.arange(n).astype(float)
        a = 0.54 - 0.46 * np.cos(2*np.pi*i/(n-1))
    elif shape == "gaussian":
        sigma = np.random.uniform(0.1, 0.4) * n
        mu = np.random.uniform(0.3, 0.7) * n
        i = np.arange(n).astype(float)
        a = np.exp(-0.5 * ((i-mu)/sigma)**2)
    elif shape == "linear":
        start = np.random.rand()
        end = np.random.rand()
        a = np.linspace(start, end, n)
    elif shape == "triangular":
        peak = np.random.randint(0, n)
        left = np.linspace(0, 1, peak+1)
        right = np.linspace(1, 0, n-peak)
        a = np.concatenate([left, right[1:]]) if peak>0 else np.concatenate([left, right])
        # For n=1 special case, fallback:
        if a.shape[0] != n:
            a = np.ones(n)
    else:
        a = np.random.rand(n)
    # Add small noise optionally
    if np.random.rand() < 0.2:
        noise = np.random.randn(n) * np.max(a) * 0.02
        a = np.maximum(a + noise, 0.0)
    return a

def adam_optimize(seq, target_sum, deadline):
    n = seq.shape[0]
    # initialize logits x = log(seq + eps)
    x = np.log(seq + 1e-12)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    best_a = seq.copy()
    best_val = _full_convolution_fft(seq, seq).max()
    beta1 = 0.9; beta2 = 0.999; eps = 1e-8
    alpha = 0.02
    init_tau_factor = 0.12
    min_tau_factor = 5e-6
    it = 0
    while time.time() < deadline:
        # softmax
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        p = exp_shift / np.sum(exp_shift)
        a = target_sum * p
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam new best max={cur_max:.6f}")
        # temperature tau
        tau = max(init_tau_factor * cur_max * (0.95**it), min_tau_factor * cur_max)
        # weights for softmax of conv
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient a: 2 * a * w_rev (via conv)
        w_rev = w[::-1]
        size = 1 << ((2*n - 2 + len(w_rev) - 1).bit_length())  # but we will compute conv2 properly.
        # Actually we can reuse method from earlier: compute conv2 = convolve(a, w_rev)
        size2 = 1 << ((n + w_rev.shape[0] - 1 - 1).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: n + w_rev.shape[0] - 1]
        idx = (w_rev.shape[0] - 1) - np.arange(n)
        grad_a = 2.0 * conv2[idx]
        # backprop to logits
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)
        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step
        # jitter
        if it % 200 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001
        # decay learning rate
        alpha *= 0.999
    return best_a, best_val

"""Hybrid optimizer for step-function bound (improved).

This version builds upon a previous hybrid scheme and adds:
  * A richer set of parametric random families for the initial
    population, including Gaussian and linear windows.
  * Adam optimisation on a temperature‑annealed soft‑max surrogate,
    with more aggressive decay of the temperature parameter.
  * Projected sub‑gradient descent that follows the exact worst
    convolution peaks, using a back‑tracking line‑search to adapt the
    step‑size.
  * A final hill‑climb that moves a larger fraction of mass and uses
    a Metropolis acceptance rule.
  * Combination of the best few candidates (convex mixtures) as an
    additional refinement step.

The routine respects a time budget (default 1000 s) and returns the
best non‑negative sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of two real vectors."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())  # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # Clip tiny negatives caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the full auto‑convolution of a."""
    return _full_convolution_fft(a, a).max()


def _record_candidate(
    seq: np.ndarray,
    best_seq: np.ndarray | None,
    best_score: float,
    top_seqs: list[tuple[float, np.ndarray]],
    max_top: int = 15,
) -> tuple[np.ndarray | None, float, list[tuple[float, np.ndarray]]]:
    """Evaluate seq, update the global best and a list of the best few candidates."""
    val = _conv_max(seq)
    if val < best_score:
        best_score = float(val)
        best_seq = seq.copy()
        print(f"[{time.time():.2f}] GLOBAL new best={best_score:.6f}  n={len(seq)}")
    # keep a sorted list of the best few
    top_seqs.append((val, seq.copy()))
    top_seqs.sort(key=lambda x: x[0])
    if len(top_seqs) > max_top:
        top_seqs.pop()
    return best_seq, best_score, top_seqs


# ----------------------------------------------------------------------
# Random parametric families for the initial population
# ----------------------------------------------------------------------
def _random_sequence(n: int, shape: str) -> np.ndarray:
    """Generate a non‑negative sequence of length n from a chosen parametric family."""
    if shape == "exp":
        tau = np.random.uniform(0.1, 2.0) * n
        idx = np.arange(n, dtype=float)
        a = np.exp(-idx / tau)
    elif shape == "pow":
        p = np.random.uniform(0.2, 1.5)
        idx = np.arange(1, n + 1, dtype=float)
        a = idx ** (-p)
    elif shape == "hann":
        i = np.arange(n, dtype=float)
        a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    elif shape == "hamming":
        i = np.arange(n, dtype=float)
        a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    elif shape == "gaussian":
        sigma = np.random.uniform(0.08, 0.3) * n
        mu = np.random.uniform(0.3, 0.7) * n
        i = np.arange(n, dtype=float)
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    elif shape == "linear":
        start = np.random.rand()
        end = np.random.rand()
        a = np.linspace(start, end, n)
    elif shape == "triangular":
        peak = np.random.randint(0, n)
        left = np.linspace(0, 1, peak + 1)
        right = np.linspace(1, 0, n - peak)
        a = np.concatenate([left, right[1:]]) if peak > 0 else np.concatenate([left, right])
    else:
        a = np.random.rand(n)
    # optional tiny noise
    if np.random.rand() < 0.2:
        noise = np.random.randn(n) * np.max(a) * 0.02
        a = np.maximum(a + noise, 0.0)
    return a


# ----------------------------------------------------------------------
# Adam optimiser on a smoothed max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 5e-6,
    alpha0: float = 0.02,
) -> tuple[np.ndarray, float]:
    """Adam optimisation of the soft‑max surrogate of max(conv)."""
    n = a0.shape[0]
    x = np.log(a0 + 1e-12)  # logits for soft‑max parametrisation
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    best_a = a0.copy()
    best_val = _conv_max(a0)
    it = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    alpha = alpha0

    while time.time() < deadline:
        # map logits to simplex
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        p = exp_shift / np.sum(exp_shift)
        a = target_sum * p

        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam new best max={cur_max:.6f}")

        # smoothed max (log‑sum‑exp) weights
        tau = max(init_tau_factor * cur_max * (0.95 ** it), min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a: 2 * a ⊛ w_rev
        w_rev = w[::-1]
        size2 = 1 << ((n + w_rev.shape[0] - 1 - 1).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: n + w_rev.shape[0] - 1]
        idx = (w_rev.shape[0] - 1) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # back‑prop to logits
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter
        if it % 200 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # decay learning rate slowly
        alpha *= 0.9995

    final_val = _conv_max(best_a)
    return best_a, float(final_val)


# ----------------------------------------------------------------------
# Gradient of the (near‑)max convolution peak(s)
# ----------------------------------------------------------------------
def _gradient_from_near_peaks(a: np.ndarray, conv: np.ndarray, delta_frac: float = 0.001) -> np.ndarray:
    """Return a sub‑gradient obtained from all convolution entries within delta of the max."""
    n = a.shape[0]
    max_val = conv.max()
    thr = max_val - max(delta_frac * max_val, 1e-12)
    mask = conv >= thr
    if not np.any(mask):
        mask[conv.argmax()] = True
    w = mask.astype(float)          # shape (2*n-1,)
    w_rev = w[::-1]

    size = 1 << ((3 * n - 2 - 1).bit_length())
    Fa = np.fft.rfft(a, n=size)
    Fw = np.fft.rfft(w_rev, n=size)
    conv2 = np.fft.irfft(Fa * Fw, n=size)[: 3 * n - 2]
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * conv2[idx]
    return grad


# ----------------------------------------------------------------------
# Targeted projected sub‑gradient descent with back‑tracking line‑search
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that follows the current worst peaks."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _conv_max(a)

    step = 0.3 * target_sum / n  # initial step size
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()

        grad = _gradient_from_near_peaks(a, conv, delta_frac=0.001)

        # try a projected step
        cand = _project_onto_simplex(a - step * grad, target_sum)
        cand_val = _conv_max(cand)

        if cand_val < cur_max - 1e-12:
            a = cand
            if cand_val < best_val:
                best_val = cand_val
                best_a = cand.copy()
                print(f"[{time.time():.2f}] subgrad it={it} new best={best_val:.6f}")
            step *= 1.07  # slightly increase after success
        else:
            step *= 0.55   # shrink on failure

        if step < 1e-9 * target_sum:
            break
        if best_val < 1.503:               # early stop if target already achieved
            break
    return best_a, best_val


# ----------------------------------------------------------------------
# Simple hill‑climb (simulated annealing) on the exact objective
# ----------------------------------------------------------------------
def _hill_climb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Move small amounts of mass between random positions with Metropolis acceptance."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _conv_max(a)

    # initial temperature proportional to current peak
    T0 = max(1e-4, best_val * 0.4)
    T_end = 1e-6
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue
        # transfer up to 12 % of mass at i
        delta = a[i] * np.random.rand() * 0.12
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _conv_max(a_new)

        if new_val < best_val:
            a = a_new
            best_a = a_new.copy()
            best_val = new_val
            print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                a = a_new

        # Renormalise every few thousand steps (numerical drift)
        if it % 3000 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break
    return best_a, best_val

# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the full auto‑convolution of a."""
    return _full_convolution_fft(a, a).max()


def _record_candidate(
    seq: np.ndarray,
    best_seq: np.ndarray | None,
    best_score: float,
    top_seqs: list[tuple[float, np.ndarray]],
    max_top: int = 15,
) -> tuple[np.ndarray | None, float, list[tuple[float, np.ndarray]]]:
    """Evaluate seq, update the global best and keep a short elite list."""
    val = _conv_max(seq)
    if val < best_score:
        best_score = float(val)
        best_seq = seq.copy()
        print(f"[{time.time():.2f}] GLOBAL new best={best_score:.6f}  n={len(seq)}")
    top_seqs.append((val, seq.copy()))
    top_seqs.sort(key=lambda x: x[0])
    if len(top_seqs) > max_top:
        top_seqs.pop()
    return best_seq, best_score, top_seqs


# ----------------------------------------------------------------------
# Random parametric families for the initial population
# ----------------------------------------------------------------------
def _random_sequence(n: int, shape: str) -> np.ndarray:
    """Generate a non‑negative sequence of length n from a chosen family."""
    if shape == "exp":
        tau = np.random.uniform(0.1, 2.0) * n
        idx = np.arange(n, dtype=float)
        a = np.exp(-idx / tau)
    elif shape == "pow":
        p = np.random.uniform(0.2, 1.5)
        idx = np.arange(1, n + 1, dtype=float)
        a = idx ** (-p)
    elif shape == "hann":
        i = np.arange(n, dtype=float)
        a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    elif shape == "hamming":
        i = np.arange(n, dtype=float)
        a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    elif shape == "gaussian":
        sigma = np.random.uniform(0.08, 0.3) * n
        mu = np.random.uniform(0.3, 0.7) * n
        i = np.arange(n, dtype=float)
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    elif shape == "linear":
        start = np.random.rand()
        end = np.random.rand()
        a = np.linspace(start, end, n)
    elif shape == "triangular":
        peak = np.random.randint(0, n)
        left = np.linspace(0, 1, peak + 1)
        right = np.linspace(1, 0, n - peak)
        a = np.concatenate([left, right[1:]]) if peak > 0 else np.concatenate([left, right])
    else:
        a = np.random.rand(n)
    if np.random.rand() < 0.2:
        noise = np.random.randn(n) * np.max(a) * 0.02
        a = np.maximum(a + noise, 0.0)
    return a


# ----------------------------------------------------------------------
# Adam optimiser on a smoothed max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 5e-6,
    alpha0: float = 0.02,
) -> tuple[np.ndarray, float]:
    """Adam optimisation of the soft‑max surrogate of max(conv)."""
    n = a0.shape[0]
    x = np.log(a0 + 1e-12)   # logits for soft‑max parametrisation
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    best_a = a0.copy()
    best_val = _conv_max(a0)
    it = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    alpha = alpha0

    while time.time() < deadline:
        # map logits to simplex
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        p = exp_shift / np.sum(exp_shift)
        a = target_sum * p

        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam new best max={cur_max:.6f}")

        # smoothed max (log‑sum‑exp) weights
        tau = max(init_tau_factor * cur_max * (0.95 ** it), min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a: 2 * a ⊛ w_rev
        w_rev = w[::-1]
        size2 = 1 << ((n + w_rev.shape[0] - 1 - 1).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: n + w_rev.shape[0] - 1]
        idx = (w_rev.shape[0] - 1) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # back‑prop to logits
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter
        if it % 200 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # decay learning rate slowly
        alpha *= 0.9995

    final_val = _conv_max(best_a)
    return best_a, float(final_val)


# ----------------------------------------------------------------------
# Gradient of the (near‑)max convolution peak(s)
# ----------------------------------------------------------------------
def _gradient_from_near_peaks(a: np.ndarray, conv: np.ndarray, delta_frac: float = 0.001) -> np.ndarray:
    """Return a sub‑gradient obtained from all convolution entries within delta of the max."""
    n = a.shape[0]
    max_val = conv.max()
    thr = max_val - max(delta_frac * max_val, 1e-12)
    mask = conv >= thr
    if not np.any(mask):
        mask[conv.argmax()] = True
    w = mask.astype(float)          # length 2*n‑1
    w_rev = w[::-1]

    size = 1 << ((3 * n - 2 - 1).bit_length())
    Fa = np.fft.rfft(a, n=size)
    Fw = np.fft.rfft(w_rev, n=size)
    conv2 = np.fft.irfft(Fa * Fw, n=size)[: 3 * n - 2]
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * conv2[idx]
    return grad


# ----------------------------------------------------------------------
# Targeted projected sub‑gradient descent with back‑tracking line‑search
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that follows the current worst peaks."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _conv_max(a)

    step = 0.3 * target_sum / n  # initial step size
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()

        grad = _gradient_from_near_peaks(a, conv, delta_frac=0.001)

        cand = _project_onto_simplex(a - step * grad, target_sum)
        cand_val = _conv_max(cand)

        if cand_val < cur_max - 1e-12:
            a = cand
            if cand_val < best_val:
                best_val = cand_val
                best_a = cand.copy()
                print(f"[{time.time():.2f}] subgrad it={it} new best={best_val:.6f}")
            step *= 1.07
        else:
            step *= 0.55

        if step < 1e-9 * target_sum:
            break
        if best_val < 1.503:               # early stop if target already reached
            break
    return best_a, best_val


# ----------------------------------------------------------------------
# Simple hill‑climb (simulated annealing) on the exact objective
# ----------------------------------------------------------------------
def _hill_climb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Move small amounts of mass between random positions with Metropolis acceptance."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _conv_max(a)

    # initial temperature proportional to current peak
    T0 = max(1e-4, best_val * 0.4)
    T_end = 1e-6
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue
        # transfer up to 12 % of mass at i
        delta = a[i] * np.random.rand() * 0.12
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _conv_max(a_new)

        if new_val < best_val:
            a = a_new
            best_a = a_new.copy()
            best_val = new_val
            print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                a = a_new

        if it % 3000 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break
    return best_a, best_val


# ----------------------------------------------------------------------
# Mix the best few candidates (convex combinations)
# ----------------------------------------------------------------------
def _mix_top_candidates(
    top_seqs: list[tuple[float, np.ndarray]],
    deadline: float,
) -> tuple[np.ndarray | None, float]:
    """Try convex averages of the best candidates that share the same length."""
    if len(top_seqs) < 2:
        return None, np.inf
    best_seq = None
    best_val = np.inf
    for _ in range(30):
        if time.time() >= deadline:
            break
        (v1, s1), (v2, s2) = np.random.choice(top_seqs, size=2, replace=False)
        if s1.shape[0] != s2.shape[0]:
            continue
        w = np.random.uniform(0.3, 0.7)
        cand = w * s1 + (1.0 - w) * s2
        val = _conv_max(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] mix new best={best_val:.6f}")

        if np.random.rand() < 0.25:
            pert = cand + np.random.randn(*cand.shape) * cand.max() * 0.005
            pert = np.maximum(pert, 0.0)
            pert = _project_onto_simplex(pert, np.sqrt(2.0 * cand.shape[0]))
            valp = _conv_max(pert)
            if valp < best_val:
                best_val = valp
                best_seq = pert.copy()
                print(f"[{time.time():.2f}] mix‑perturb new best={best_val:.6f}")
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """Hybrid search for a non‑negative step‑function with minimal evaluation score."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.10   # safety margin

    best_seq = None
    best_score = np.inf
    top_seqs = []  # elite list (score, sequence)

    # ------------------------------------------------------------------
    # Warm‑start from a previously saved construction if available
    # ------------------------------------------------------------------
    hs = globals().get("height_sequence_1")
    if hs is not None:
        hs_arr = np.asarray(hs, dtype=float)
        n0 = hs_arr.shape[0]
        target0 = np.sqrt(2.0 * n0)
        hs_norm = hs_arr * (target0 / (hs_arr.sum() + 1e-12))
        best_seq, best_score, top_seqs = _record_candidate(
            hs_norm, best_seq, best_score, top_seqs
        )

    # ------------------------------------------------------------------
    # Phase 1 – diverse random initialisations
    # ------------------------------------------------------------------
    phase1_end = deadline - 600.0  # reserve ~600 s for later phases
    shapes = ["exp", "pow", "hann", "hamming", "gaussian", "linear", "triangular"]
    while time.time() < phase1_end:
        remaining = phase1_end - time.time()
        if remaining > 300:
            n = np.random.randint(3000, 10000 + 1)
        elif remaining > 150:
            n = np.random.randint(2000, 8000 + 1)
        else:
            n = np.random.randint(1500, 5000 + 1)

        shape = np.random.choice(shapes)
        a0 = _random_sequence(n, shape)
        target = np.sqrt(2.0 * n)
        a0 = a0 * (target / (a0.sum() + 1e-12))
        best_seq, best_score, top_seqs = _record_candidate(
            a0, best_seq, best_score, top_seqs
        )

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on a smoothed max‑convolution
    # ------------------------------------------------------------------
    while time.time() < deadline - 350.0:
        if best_seq is None:
            break
        n = best_seq.shape[0]
        target = np.sqrt(2.0 * n)

        seed_seq = best_seq
        if top_seqs and np.random.rand() < 0.4:
            seed_seq = top_seqs[np.random.randint(0, len(top_seqs))][1]

        run_deadline = time.time() + min(22.0, deadline - time.time() - 30.0)
        seq_opt, val_opt = _adam_optimize(
            seed_seq,
            target,
            run_deadline,
            init_tau_factor=0.12,
            min_tau_factor=5e-6,
            alpha0=0.025,
        )
        best_seq, best_score, top_seqs = _record_candidate(
            seq_opt, best_seq, best_score, top_seqs
        )

    # ------------------------------------------------------------------
    # Phase 3 – Targeted sub‑gradient descent with line‑search
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 250.0:
        n = best_seq.shape[0]
        target = np.sqrt(2.0 * n)
        sub_deadline = deadline - 150.0   # keep time for later phases
        seq_sub, val_sub = _subgradient_descent(
            best_seq.copy(), target, sub_deadline
        )
        best_seq, best_score, top_seqs = _record_candidate(
            seq_sub, best_seq, best_score, top_seqs
        )

    # ------------------------------------------------------------------
    # Phase 4 – Final simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 50.0:
        n = best_seq.shape[0]
        target = np.sqrt(2.0 * n)
        hill_deadline = deadline - 5.0
        seq_hill, val_hill = _hill_climb(best_seq.copy(), target, hill_deadline)
        best_seq, best_score, top_seqs = _record_candidate(
            seq_hill, best_seq, best_score, top_seqs
        )

    # ------------------------------------------------------------------
    # Phase 5 – Convex mixtures of elite candidates
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        mix_seq, mix_val = _mix_top_candidates(top_seqs, deadline)
        if mix_seq is not None:
            best_seq, best_score, top_seqs = _record_candidate(
                mix_seq, best_seq, best_score, top_seqs
            )

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be used)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        target = np.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)
        best_score = _conv_max(best_seq)
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # Return a plain Python list as required by the evaluator
    return best_seq.tolist()
