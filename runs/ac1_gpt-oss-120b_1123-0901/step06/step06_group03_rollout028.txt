"""
Propose a step‑function (non‑negative vector) that minimises

        F(a) = 2·n·max(a∗a) / (∑a)²

The routine works in three stages:

1. **Diversified initialisation** – a pool of candidate vectors is built
   (uniform, power‑law, Laplace, Gaussian, sinusoid, Dirichlet draws and a
   warm‑start if present).  The best few are kept.

2. **Gradient‑based refinement** – each kept candidate is optimised with the
   same Adam + exponentiated‑gradient scheme that was used in the reference
   implementation, but on a *smaller* support (n≈2048) so many more
   iterations fit into the time budget.

3. **Linear‑programming “mass‑transfer’’ step** – given the current vector
   ``f`` we solve a linear programme that maximises the total mass of a new
   vector ``g`` while keeping the convolution ``f★g`` below the current
   worst‑case value ``M = max(f∗f)``.  Mixing ``g`` into ``f`` (with a few
   mixing ratios) strictly reduces the objective unless we are already at
   a (local) optimum.  Only a handful of LP solves are needed.

If time permits the final distribution is interpolated to a larger grid
(4096 points) and refined once more.

All numbers are forced to be non‑negative and are clipped to the evaluator’s
range [0, 1000].

The function returns a plain Python list that can be handed to
``evaluate_sequence``.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# optional imports – they are only needed for the LP refinement
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
except Exception:  # pragma: no cover
    _opt = None
    _sp = None


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = rfft(a, fft_len)
    conv = irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w★a needed for the gradient."""
    rev_a = a[::-1]
    fw = rfft(w, fft_len)
    fr = rfft(rev_a, fft_len)
    corr = irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective (sum(a) is assumed > 0)."""
    conv = _full_convolution(a, fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate (log‑sum‑exp) followed by a
    mirror‑descent sub‑gradient phase.  The routine updates and returns the
    best sequence/value found (global across all calls).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04               # a slightly smaller learning‑rate for stability
    lam = 0.5               # start temperature for the smooth‑max
    lam_factor = 1.018

    last_improve = time.time()
    it = 0

    # --------------------------- Adam phase ---------------------------
    while time.time() < deadline:
        # map logits → simplex by soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (useful for logging)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp formulation)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through the soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase temperature (approach the true max)
        lam = min(lam * lam_factor, 1e6)

        # occasional random “shake” if we have been stuck for a while
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        if it > 800_000:  # safety bound (larger for a smaller n)
            break

    # -------------------- Sub‑gradient phase (exact max) --------------------
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total

    eta0 = 0.6
    sub_last_improve = time.time()
    for sub_it in range(40_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[Subgrad n={n} it={sub_it:05d}] new best {val:.6f}")

        # active‑set mask (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (mirror‑descent on the simplex)
        eta = eta0 / np.sqrt(sub_it + 1)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                 # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 6.0:   # stop if no recent improvement
            break

    return global_best_seq, global_best_val

def _generate_candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse set of start vectors (including warm‑start)."""
    pool = []

    # warm‑start and its circular shifts
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # uniform
    pool.append(np.full(n, 1.0 / n))

    # power‑law edge‑concentrated shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)          # i·(n‑i+1)
    for alpha in np.linspace(-2.0, -0.2, 12):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # double‑exponential (two‑sided Laplace)
    i = np.arange(n, dtype=np.float64)
    for lam in np.linspace(0.1, 3.0, 12):
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        pool.append(shape)

    # Gaussian bumps (varying width)
    for sigma in np.linspace(n * 0.03, n * 0.4, 8):
        shape = np.exp(-0.5 * ((i - n / 2) ** 2) / (sigma ** 2))
        shape /= shape.sum()
        pool.append(shape)

    # half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # random Dirichlet draws
    for alpha in np.logspace(-2, 0, 10):
        shape = np.random.dirichlet(np.full(n, alpha))
        pool.append(shape)

    return pool

def _lp_optimize(f: np.ndarray, M: float, time_limit: float) -> np.ndarray | None:
    """
    Solve   max 1ᵀg   s.t.   f★g ≤ M  (component‑wise),   g ≥ 0.
    Returns the optimal g (or None on failure).
    """
    if _opt is None:          # SciPy not available – skip LP step
        return None

    n = f.shape[0]
    rows = []
    cols = []
    data = []

    # build the convolution matrix A where (A g)_k = (f★g)_k
    # row k = i + j  with i from f, j from g
    arange_n = np.arange(n, dtype=np.int32)
    for i, fi in enumerate(f):
        if fi == 0.0:
            continue                     # zero coefficient – no need to store
        rows.append(i + arange_n)        # length n
        cols.append(arange_n)            # 0 .. n‑1
        data.append(np.full(n, fi))

    rows = np.concatenate(rows) if rows else np.empty(0, dtype=np.int32)
    cols = np.concatenate(cols) if cols else np.empty(0, dtype=np.int32)
    data = np.concatenate(data) if data else np.empty(0, dtype=np.float64)

    A = _sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                     # maximise sum(g)  → minimise -sum(g)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # HiGHS respects the 'time_limit' option (seconds)
    options = {"presolve": True, "time_limit": max(0.1, time_limit)}
    try:
        res = _opt.linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0, None)] * n,
            method="highs",
            options=options,
        )
    except Exception as e:  # pragma: no cover
        print(f"[LP] linprog exception: {e}")
        return None

    if not res.success:
        print(f"[LP] linprog failed: {res.message}")
        return None
    return res.x

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    n = 2048
    conv_fft_len = _next_pow2(2 * n - 1)

    # warm‑start if the previous run stored a sequence
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        w = np.asarray(warm_start, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
        else:
            w = None
    else:
        w = None

    # ---------------------------------------------------------------
    # 2. generate a diverse pool and keep the most promising starts
    candidates = _generate_candidate_pool(n, w)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    for idx, cand in enumerate(candidates):
        a = np.maximum(cand, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        val = _objective(a, n, conv_fft_len)
        scored.append((val, a))
        if val < best_val:

            best_seq = a.copy()
        print(f"[screen {idx:03d}] obj={val:.6f}")

    scored.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in scored[:8]]   # keep a handful

    # ---------------------------------------------------------------

    for cid, a0 in enumerate(top_candidates):

        best_seq, best_val = _deep_optimize(

    return best_seq.tolist()
