            # arr is np array not normalized

            conv = np.fft.irfft(np.fft.rfft(a, conv_fft_len)**2, conv_fft_len)[:2*n-1]
            maxc = conv.max()

        def power_alpha(alpha):

            shape = np.power(prod, alpha, where=prod>0)
            return shape

        def beta_shape(p,q):

"""
Propose a step‑function (non‑negative vector) that minimises

        F(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm is a two‑phase search:

1. **Global parametric sweep** – several low‑dimensional families (power‑law,
   beta‑distribution, edge Gaussians, uniform/edge mixtures, etc.) are
   sampled many times for a given length `n`.  The best few candidates are
   kept.

2. **Local refinement** – each of the retained candidates is further
   optimised with a smooth‑max Adam phase followed by an exact ℓ∞
   sub‑gradient (exponentiated‑gradient) phase.  The improvement is
   accepted only if it lowers the exact objective.

The routine respects a wall‑clock budget (default 1000 s) and returns the
best sequence discovered as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using a power‑of‑two FFT.
    Returns an array of length 2*len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) required for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smooth‑max surrogate of max(conv).
    Returns the best vector (normalised to sum=1) and its exact objective value.
    """
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05

    lam = 1.0                # temperature for smooth‑max
    lam_factor = 1.02

    # iteration budget – generous for the final refinement
    max_iters = 5000

    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max to enforce simplex constraints
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0:
            w = w / w_sum
        else:
            w = np.full_like(w, 1.0 / w.size)

        # gradient of smooth‑max w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature annealing
        lam = min(lam * lam_factor, 1e8)

        # exact objective (sum(a)=1)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # occasional jitter if stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Exact ℓ∞ sub‑gradient refinement using an exponentiated‑gradient update.
    Returns the best vector (normalised) and its exact objective value.
    """
    a = a0 / a0.sum()
    best_val = float("inf")
    best_a = a.copy()
    eta = 0.45
    last_improve = time.time()

    for it in range(3000):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                         # fallback: random simplex point
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        eta *= 0.9995                         # slowly decay step size

        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Parameteric families – each returns an *unnormalised* non‑negative shape
# ----------------------------------------------------------------------


def _sample_power_law(n: int, rng: np.random.Generator) -> np.ndarray:
    """a_i ∝ (i·(n‑i+1))^α  with α∈[‑4,‑0.5] (edge‑concentrated)."""
    alpha = rng.uniform(-4.0, -0.5)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
    return shape


def _sample_beta_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Beta‑distribution shape on {0,…,n‑1}: x^{p‑1}(1‑x)^{q‑1}."""
    p = rng.uniform(0.1, 10.0)
    q = rng.uniform(0.1, 10.0)
    x = (np.arange(n) + 0.5) / n
    shape = np.power(x, p - 1.0) * np.power(1.0 - x, q - 1.0)
    return shape


def _sample_two_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sum of two symmetric Gaussians anchored at the ends."""
    sigma = rng.uniform(0.5, n / 10.0)
    i = np.arange(n, dtype=np.float64)
    left = np.exp(- (i ** 2) / (2.0 * sigma ** 2))
    right = np.exp(- ((n - 1 - i) ** 2) / (2.0 * sigma ** 2))
    shape = left + right
    return shape


def _sample_edge_mix(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of uniform and a sharp edge shape."""
    w = rng.uniform(0.0, 1.0)          # weight of edge component
    i = np.arange(n, dtype=np.float64)
    edge = np.exp(-0.5 * i ** 2) + np.exp(-0.5 * (n - 1 - i) ** 2)
    uniform = np.full(n, 1.0 / n)
    shape = (1.0 - w) * uniform + w * edge
    return shape


def _sample_gauss_uniform_mix(n: int, rng: np.random.Generator) -> np.ndarray:
    """(1‑w)·uniform + w·(Gaussian left + Gaussian right)."""
    w = rng.uniform(0.1, 0.9)
    sigma_frac = rng.uniform(0.001, 0.05)   # sigma as a fraction of n
    sigma = sigma_frac * n
    i = np.arange(n, dtype=np.float64)
    left = np.exp(- (i ** 2) / (2.0 * sigma ** 2))
    right = np.exp(- ((n - 1 - i) ** 2) / (2.0 * sigma ** 2))
    edge = left + right
    uniform = np.full(n, 1.0 / n)
    shape = (1.0 - w) * uniform + w * edge
    return shape


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the wall‑clock budget and returns the
    best sequence discovered as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # Lengths to explore – larger lengths give a finer search space.
    lengths = [16384, 32768, 65536]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = 1 << int(math.ceil(math.log2(2 * n - 1)))
        corr_fft_len = 1 << int(math.ceil(math.log2(3 * n - 2)))

        # -----------------------------------------------------------------
        # 1) Global parametric sweep
        # -----------------------------------------------------------------
        candidate_pool = []   # [(val, array), ...]

        families = [
            _sample_power_law,
            _sample_beta_shape,
            _sample_two_gaussian,
            _sample_edge_mix,
            _sample_gauss_uniform_mix,
        ]

        # Number of random draws per family – adapt to remaining time.
        draws_per_family = 250
        for sample_fn in families:
            for _ in range(draws_per_family):
                if time.time() > deadline:
                    break
                shape = sample_fn(n, rng)
                if shape.sum() == 0:
                    continue
                a = shape / shape.sum()
                # exact objective (sum(a)=1)
                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()
                val = 2.0 * n * max_c
                candidate_pool.append((val, a))

        # keep the best few candidates for local refinement
        candidate_pool.sort(key=lambda x: x[0])
        top_k = 8
        top_candidates = [cand for (_, cand) in candidate_pool[:top_k]]

        # -----------------------------------------------------------------
        # 2) Local refinement of the top candidates
        # -----------------------------------------------------------------
        for idx, cand in enumerate(top_candidates):
            if time.time() > deadline:
                break
            try:
                # Adam smooth‑max phase
                a_adam, val_adam = _adam_optimize(
                    cand, n, conv_fft_len, corr_fft_len, deadline
                )
                # Exact ℓ∞ sub‑gradient refinement
                a_ref, val_ref = _subgradient_refine(
                    a_adam, n, conv_fft_len, corr_fft_len, deadline
                )

                # Record the best of the two phases
                if val_ref < best_val:
                    best_val = val_ref
                    best_seq = a_ref.copy()
                    print(f"[n={n} cand={idx}] new global best {best_val:.6f}")
                elif val_adam < best_val:
                    best_val = val_adam
                    best_seq = a_adam.copy()
                    print(f"[n={n} cand={idx}] new global best (Adam) {best_val:.6f}")

            except Exception as exc:
                print(f"[n={n} cand={idx}] exception: {exc}")

        # -----------------------------------------------------------------
        # 3) Jitter around the current best of this length (optional)
        # -----------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            for jitter_id in range(4):
                if time.time() > deadline:
                    break
                logits = np.log(np.maximum(best_seq, 1e-12))
                logits += rng.normal(scale=0.02, size=n)
                a0 = np.exp(logits - logits.max())
                a0 = a0 / a0.sum()

                a_adam, val_adam = _adam_optimize(
                    a0, n, conv_fft_len, corr_fft_len, deadline
                )
                a_ref, val_ref = _subgradient_refine(
                    a_adam, n, conv_fft_len, corr_fft_len, deadline
                )
                if val_ref < best_val:
                    best_val = val_ref
                    best_seq = a_ref.copy()
                    print(f"[n={n} jitter={jitter_id}] new global best {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found
    # ----------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 16384
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = float("inf")
        print("Fallback to uniform sequence (no candidate survived).")

    print(f"\nBest value found: {best_val:.6f} with length {len(best_seq)}")
    return best_seq.tolist()
