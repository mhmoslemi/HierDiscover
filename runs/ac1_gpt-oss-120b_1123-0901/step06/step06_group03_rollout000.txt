"""
Propose a non‑negative step‑function minimizing

        F(a) = 2·n·max(a∗a) / (∑a)² .

Algorithm
---------

1. **Coarse scale (n≈1024)**
   * Build a very diverse start pool:
     – uniform, several symmetric/asymmetric Beta PDFs,
       power‑law edges, double‑exponential, Gaussian bumps,
       a sine wave and many random Dirichlet draws.
   * Normalise each to sum‑one and keep the best ~12 candidates
     using a fast FFT‑based evaluation of the exact objective.

2. **Refinement**
   * Each retained candidate is refined by two phases:
     – *Adam* on a smooth‑max surrogate (log‑sum‑exp) of the max‑convolution.
     – *Projected sub‑gradient* descent on the true max‑convolution,
       using Euclidean projection onto the simplex (Duchi et al.).
   * The optimiser runs as long as the global deadline permits
     and constantly updates the global best sequence/value.

3. **Multi‑scale up‑sampling**
   * The best sequence from the coarse scale is linearly interpolated
     to n=4096 and refined again (same two‑phase optimiser).
   * If time still remains, the solution is up‑sampled to n=8192
     and a final optimisation pass is performed.

All numbers are forced to be non‑negative and clipped to the evaluator’s
upper bound (1000).  The routine respects the supplied time budget and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of the objective (sum(a) is assumed > 0)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback (should not happen)
        theta = cssv[-1] / n
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _adam_phase(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation on a log‑sum‑exp surrogate of max(conv).
    Returns updated global best.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # work in the log‑domain → simplex via softmax
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.5          # initial smooth‑max temperature
    lam_factor = 1.02

    last_improve = time.time()
    it = 0
    max_iter = 300_000

    while time.time() < deadline and it < max_iter:
        # convert logits -> simplex vector a
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for bookkeeping)
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()
        val = 2.0 * n * M / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - M))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase temperature
        lam = min(lam * lam_factor, 1e6)

        # random shake if stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return global_best_seq, global_best_val


def _subgradient_phase(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray | None, float]:
    """
    Exact max‑convolution sub‑gradient descent with Euclidean simplex projection.
    """
    a = np.maximum(a0, 0.0)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total

    eta0 = 0.8
    it = 0
    max_iter = 120_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()
        val = 2.0 * n * M / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} iter={it:06d}] new best {val:.6f}")

        # active set mask (indices attaining the max)
        active = np.isclose(conv, M, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size schedule (mirror‑descent style)
        eta = eta0 / np.sqrt(it + 1)

        # gradient step and projection onto simplex
        a_next = a - eta * grad_a
        a = _project_onto_simplex(a_next)

        # occasional perturbation if stagnating
        if time.time() - last_improve > 8.0:
            # add tiny random jitter and re‑project
            jitter = (np.random.rand(n) - 0.5) * 1e-3
            a = _project_onto_simplex(a + jitter)
            last_improve = time.time()

        it += 1

    return global_best_seq, global_best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam → Sub‑gradient phases, updating the global best.
    """
    # Adam (smooth‑max)
    global_best_seq, global_best_val = _adam_phase(
        a0, n, conv_fft_len, corr_fft_len, deadline, global_best_seq, global_best_val
    )
    # Sub‑gradient (exact max)
    global_best_seq, global_best_val = _subgradient_phase(
        global_best_seq if global_best_seq is not None else a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    return global_best_seq, global_best_val

def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Symmetric or asymmetric Beta‑PDF discretised on n points (excluding 0 and 1).
    """
    x = (np.arange(1, n + 1) - 0.5) / n  # points in (0,1)
    # avoid overflow for extreme parameters
    shape = np.exp((alpha - 1) * np.log(x + 1e-16) + (beta - 1) * np.log(1 - x + 1e-16))
    if shape.sum() == 0:
        shape = np.ones_like(shape)
    shape /= shape.sum()
    return shape


def _generate_candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse set of starting vectors."""
    pool = []

    # warm‑start and its rotations
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # 1. Uniform baseline
    pool.append(np.full(n, 1.0 / n))

    # 2. Symmetric Beta families (alpha = beta)
    for a in np.linspace(0.3, 8.0, 12):
        pool.append(_beta_shape(n, a, a))

    # 3. Asymmetric Beta families
    alphas = np.linspace(0.3, 8.0, 6)
    betas  = np.linspace(0.3, 8.0, 6)
    for a in alphas:
        for b in betas:
            if abs(a - b) > 0.5:          # skip near‑symmetric duplicates
                pool.append(_beta_shape(n, a, b))

    # 4. Power‑law edge shapes (i·(n‑i+1))^α
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 14):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # 5. Double‑exponential (two‑sided Laplace) shapes
    i = np.arange(n, dtype=np.float64)
    for lam in np.linspace(0.1, 3.0, 13):
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        pool.append(shape)

    # 6. Gaussian bumps of varying width
    for sigma in np.linspace(n * 0.03, n * 0.4, 9):
        shape = np.exp(-0.5 * ((i - n / 2) ** 2) / (sigma ** 2))
        shape /= shape.sum()
        pool.append(shape)

    # 7. Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # 8. Random Dirichlet draws with varying concentration
    for alpha in np.logspace(-2, 0, 13):
        shape = np.random.dirichlet(np.full(n, alpha))
        pool.append(shape)

    return pool


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point.  Searches for a step‑function that minimises the
    evaluation metric.  The routine respects the supplied time budget
    (default 1000 s) and returns the best sequence found as a plain list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Stage 1: coarse resolution (n≈1024) to explore many shapes quickly
    # ------------------------------------------------------------------
    n0 = 1024
    conv_fft_len0 = _next_pow2(2 * n0 - 1)
    corr_fft_len0 = _next_pow2(3 * n0 - 2)

    # Warm start from a previously found construction if present
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        w = np.asarray(warm_start, dtype=np.float64)
        if w.shape[0] > n0:
            w = w[:n0]
        elif w.shape[0] < n0:
            w = np.pad(w, (0, n0 - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
        else:
            w = None
    else:
        w = None

    # Build a large and diverse candidate pool
    candidates = _generate_candidate_pool(n0, w)

    # Quick evaluation of all candidates; keep the best few
    best_val = float("inf")
    best_seq = None
    scored = []   # list of (value, normalised vector)

    print("[stage 1] evaluating candidate pool (size=%d)" % len(candidates))
    for idx, cand in enumerate(candidates):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n0))
            s = a.sum()
        a /= s
        val = _objective(a, n0, conv_fft_len0)
        scored.append((val, a))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        if idx % 20 == 0:
            print(f"    screened {idx}/{len(candidates)} – cur best {best_val:.6f}")

    # Keep the top 12 candidates for deep optimisation
    scored.sort(key=lambda x: x[0])
    top_vectors = [vec for _, vec in scored[:12]]

    # ------------------------------------------------------------------
    # Stage 2: deep optimisation on the coarse grid
    # ------------------------------------------------------------------
    print("[stage 2] deep optimisation on n=%d (up to %d candidates)" % (n0, len(top_vectors)))
    for cid, a0 in enumerate(top_vectors):
        if time.time() >= deadline:
            break
        print(f"[optim start {cid}] launching deep optimisation (n={n0})")
        best_seq, best_val = _deep_optimize(
            a0,
            n0,
            conv_fft_len0,
            corr_fft_len0,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3: up‑sample to n=4096 and refine again
    # ------------------------------------------------------------------
    if time.time() + 6 < deadline:
        n1 = 4096
        conv_fft_len1 = _next_pow2(2 * n1 - 1)
        corr_fft_len1 = _next_pow2(3 * n1 - 2)

        # Upsample by linear interpolation
        x_old = np.linspace(0, 1, n0, endpoint=False)
        x_new = np.linspace(0, 1, n1, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("[stage 3] optimisation on n=4096")
        best_seq, best_val = _deep_optimize(
            a_big,
            n1,
            conv_fft_len1,
            corr_fft_len1,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 4: if time permits, go to n=8192 for final polishing
    # ------------------------------------------------------------------
    if time.time() + 12 < deadline:
        n2 = 8192
        conv_fft_len2 = _next_pow2(2 * n2 - 1)
        corr_fft_len2 = _next_pow2(3 * n2 - 2)

        # Interpolate best_seq to the larger grid
        if best_seq.shape[0] != n2:
            x_old = np.linspace(0, 1, best_seq.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n2, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = best_seq.copy()
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("[stage 4] final optimisation on n=8192")
        best_seq, best_val = _deep_optimize(
            a_big,
            n2,
            conv_fft_len2,
            corr_fft_len2,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Finalisation
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: simple uniform vector
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip according to evaluator's limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print("[finished] best objective found = %.6f" % best_val)
    return best_seq.tolist()
