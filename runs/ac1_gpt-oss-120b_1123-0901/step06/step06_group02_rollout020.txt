"""Hybrid multi‑restart optimizer for the step‑function maximal convolution bound.

The algorithm proceeds in four stages:
1. Warm‑start from any existing construction.
2. Generate many random candidates of varying length and shape; keep the best K.
3. For each of the top candidates run a short Adam optimisation on a
   log‑sum‑exp surrogate, followed by an exact active‑set sub‑gradient
   descent and a simulated‑annealing hill‑climb.
4. Return the best sequence found (as a plain Python list)."""

import time
import math
import heapq
import numpy as np

def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors via FFT (non‑negative result)."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)   # clip tiny negatives
    return conv

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {w ≥ 0, Σ w = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.10,
    min_tau_factor: float = 1e-4,
    alpha0: float = 0.02,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the log‑sum‑exp surrogate of max(conv(a,a))."""
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # soft‑max mapping to simplex
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        p = exp_shift / np.sum(exp_shift)
        a = target_sum * p

        # exact max (for logging and termination)
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam n={n} new best max={cur_max:.6f}")

        # smoothed max weights (log‑sum‑exp)
        tau = max(init_tau_factor * cur_max * (0.99 ** it), min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t a of the surrogate ∑ w_k conv_k
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # chain rule to logits
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape flat plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slowly decay learning rate
        alpha *= 0.9995

    final_val = _full_convolution_fft(best_a, best_a).max() if best_a is not None else np.inf
    return best_a, float(final_val)

def _multiconstraint_subgradient(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    active_tol: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent that follows the average gradient of all
    constraints whose convolution value is within ``active_tol`` relative to the
    current maximum.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    # initial step proportional to average coefficient size
    step = 0.3 * target_sum / n

    while time.time() < deadline:
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        eps = max_val * active_tol
        active = np.where(conv >= max_val - eps)[0]

        # construct gradient = average of gradients of active constraints
        grad = np.zeros_like(a)
        for k in active:
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        if active.size > 0:
            grad /= active.size

        # projected step
        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)
        cand_val = _full_convolution_fft(cand, cand).max()

        if cand_val < best_val - 1e-12:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= 1.02          # modest acceleration after success
        else:
            step *= 0.70          # shrink step on failure
            if np.random.rand() < 0.07:
                # small jitter to shake off stagnation
                jitter = np.random.rand(n) * 1e-4
                a = _project_onto_simplex(a + jitter, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val

def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing hill‑climb moving mass between two positions."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # temperature schedule
    T0 = max(1e-3, best_val * 0.4)
    T_end = 1e-5
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # occasionally allow a larger jump
        if np.random.rand() < 0.3:
            delta = a[i] * np.random.rand() * 0.20   # up to 20 % of source mass
        else:
            delta = a[i] * np.random.rand() * 0.08   # up to 8 %

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()
        accept = False
        if new_val < best_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")

        # re‑project every few thousand steps to combat drift
        if it % 2500 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val

"""
Hybrid multi‑restart optimizer for the step‑function maximal‑convolution bound.

The algorithm works in four stages:

1. Warm‑start from a previously discovered construction (if available).
2. Generate a large pool of random candidates of varied lengths and shapes,
   keeping the K best (K≈10) on a heap.
3. For each of those top candidates run a short Adam optimisation on a
   log‑sum‑exp surrogate, followed by an exact active‑set sub‑gradient descent
   and a simulated‑annealing hill‑climb.  The best sequence found so far is
   updated after every sub‑stage.
4. Return the best sequence as a plain Python list.

All helper functions are top‑level (no closures) and respect the overall
time budget.  The objective is scale‑invariant, therefore we keep each
candidate normalised so that Σ a_i = √(2 n); the evaluation value then equals
the maximal convolution value.
"""

import time
import math
import heapq
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution (real FFT) – returns non‑negative values
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())               # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)                  # clip tiny negatives
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]          # sorted descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)     # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1       # zero‑based index of last positive entry
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the log‑sum‑exp surrogate of max(conv(a,a))
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.10,
    min_tau_factor: float = 1e-4,
    alpha0: float = 0.02,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser for the smooth approximation of max(conv(a,a))."""
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # ---- map logits to the simplex (soft‑max) ----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)                # stable
        p = exp_shift / np.sum(exp_shift)            # Σ p = 1
        a = target_sum * p

        # ---- exact max (used for logging) ----
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam n={n} new best max={cur_max:.6f}")

        # ---- smoothed max weights (log‑sum‑exp) ----
        tau = max(init_tau_factor * cur_max * (0.99 ** it),
                  min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient w.r.t. a of the surrogate Σ w_k conv_k ----
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # ---- back‑prop through the soft‑max ----
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ---- Adam update ----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slowly decay learning rate
        alpha *= 0.9995

    final_val = _full_convolution_fft(best_a, best_a).max() if best_a is not None else np.inf
    return best_a, float(final_val)


# ----------------------------------------------------------------------
# Sub‑gradient descent that follows the average gradient of all active
# constraints (i.e. all convolution entries close to the current maximum)
# ----------------------------------------------------------------------
def _multiconstraint_subgradient(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    active_tol: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent using the active set of binding constraints."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step = 0.3 * target_sum / n   # initial step size

    while time.time() < deadline:
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        eps = max_val * active_tol
        active = np.where(conv >= max_val - eps)[0]

        # gradient = average of gradients of the active constraints
        grad = np.zeros_like(a)
        for k in active:
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1, dtype=int)
            grad[idx] += 2.0 * a[k - idx]
        if active.size > 0:
            grad /= active.size

        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)
        cand_val = _full_convolution_fft(cand, cand).max()

        if cand_val < best_val - 1e-12:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= 1.02                     # slight acceleration after success
        else:
            step *= 0.70                     # shrink step on failure
            if np.random.rand() < 0.07:      # jitter to avoid stagnation
                jitter = np.random.rand(n) * 1e-4
                a = _project_onto_simplex(a + jitter, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (simulated annealing) on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Move tiny amounts of mass between positions with a Metropolis acceptance rule."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # temperature schedule
    T0 = max(1e-3, best_val * 0.4)
    T_end = 1e-5
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # occasionally allow a larger transfer
        if np.random.rand() < 0.3:
            delta = a[i] * np.random.rand() * 0.20   # up to 20 % of source mass
        else:
            delta = a[i] * np.random.rand() * 0.08   # up to 8 %

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()
        accept = False
        if new_val < best_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")

        # re‑normalise every few thousand steps to mitigate drift
        if it % 2500 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val
