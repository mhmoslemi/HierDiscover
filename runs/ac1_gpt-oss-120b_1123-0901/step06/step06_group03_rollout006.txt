"""
Propose a step‑function (non‑negative vector) that minimizes

        F(a) = 2·n·max(a∗a) / (∑a)²

using a multi‑stage stochastic‑gradient search with an annealed
p‑norm surrogate for the max‑convolution.  The pipeline is:

1.  Build a diverse pool of starting vectors (uniform, power‑law,
    double‑exponential, Gaussian, beta‑densities, sinusoids,
    random Dirichlet draws and an optional warm‑start).

2.  For the best few candidates run an Adam optimiser on the smooth
    surrogate   S_p(a)= (∑ (a∗a)_i^p)^{1/p}   with p increasing from
    2 to ≈5 000.  Gradients are obtained via FFT based cross‑correlation
    and the parameters are exponentiated (soft‑max) to stay on the
    probability simplex.

3.  Refine the best solution with an exact sub‑gradient mirror‑descent
    on the true max‑convolution.

4.  If time permits, up‑sample the current best distribution to a
    larger grid (8 192 → 16 384 points) and repeat steps 2‑3.  Linear
    interpolation is used for up‑sampling.

All operations respect the 1000 s budget; the function returns the
best sequence found as a plain Python list.
"""

import time
import numpy as np
from typing import List, Tuple, Optional

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)‑1 using real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of the problem objective (∑a is assumed >0)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

def _generate_candidate_pool(n: int, warm: Optional[np.ndarray]) -> List[np.ndarray]:
    """
    Return a list of diverse starting vectors of length n.
    Includes several parametric families and random Dirichlet draws.
    """
    pool: List[np.ndarray] = []

    # ------------------------------------------------------------------
    # 1️⃣ Warm‑start and simple rotations (if available)
    # ------------------------------------------------------------------
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # ------------------------------------------------------------------
    # 2️⃣ Uniform distribution (baseline)
    # ------------------------------------------------------------------
    pool.append(np.full(n, 1.0 / n))

    # ------------------------------------------------------------------
    # 3️⃣ Power‑law edge‑concentrated shapes: (i·(n‑i+1))^α
    # ------------------------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)                     # i·(n‑i+1)
    for α in np.linspace(-2.0, -0.2, 12):
        shape = np.where(prod > 0, prod ** α, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # ------------------------------------------------------------------
    # 4️⃣ Two‑sided Laplace (double‑exponential) families
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)
    for lam in np.linspace(0.1, 3.0, 12):
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        pool.append(shape)

    # ------------------------------------------------------------------
    # 5️⃣ Gaussian bumps centred at the middle (σ sweep)
    # ------------------------------------------------------------------
    for sigma in np.linspace(n * 0.03, n * 0.4, 9):
        shape = np.exp(-0.5 * ((i - n / 2) ** 2) / (sigma ** 2))
        shape /= shape.sum()
        pool.append(shape)

    # ------------------------------------------------------------------
    # 6️⃣ Beta‑distribution‑like shapes (discretised Beta pdf)
    # ------------------------------------------------------------------
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    for a in (0.5, 1.0, 2.0, 5.0):
        for b in (0.5, 1.0, 2.0, 5.0):
            shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
            if shape.sum() > 0:
                shape /= shape.sum()
                pool.append(shape)

    # ------------------------------------------------------------------
    # 7️⃣ Single half‑wave sinusoid
    # ------------------------------------------------------------------
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # ------------------------------------------------------------------
    # 8️⃣ Random Dirichlet draws (various concentration parameters)
    # ------------------------------------------------------------------
    for α in np.logspace(-2, 0, 10):
        shape = np.random.dirichlet(np.full(n, α))
        pool.append(shape)

    return pool

def _deep_optimize_power_surrogate(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: Optional[np.ndarray],
    global_best_val: float,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation on an annealed p‑norm surrogate for the max‑convolution,
    followed by a short exact sub‑gradient phase.
    """
    eps = 1e-15
    # use logits (soft‑max) to stay on the simplex
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    # annealing schedule for the power p
    p = 2.0
    p_factor = 1.07
    p_max = 5000.0

    last_improve = time.time()
    it = 0

    # --------------------------------------------------------------
    # Adam on the surrogate
    # --------------------------------------------------------------
    while time.time() < deadline:
        # ----- build the probability vector from logits -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()          # ∑a = 1

        # ----- exact objective (useful for logging) -----
        conv = _full_convolution(a, conv_fft_len)
        cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_val < global_best_val:
            global_best_val = cur_val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:06d}] new best {cur_val:.6f}")

        # ----- compute weights for the p‑norm surrogate -----
        #   w_i ∝ (conv_i)^{p‑1}
        #   use log‑space to avoid overflow for large p
        log_conv = np.log(conv + eps)
        if p > 2.0:
            # stability: subtract max log
            exponent = (p - 1.0) * (log_conv - log_conv.max())
            w = np.exp(exponent)
        else:
            w = np.maximum(conv, eps) ** (p - 1.0)

        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate (chain rule through the soft‑max) -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)   # ∂/∂a of Σ w_i conv_i
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- anneal p -----
        p = min(p * p_factor, p_max)

        # ----- occasional random shake if stagnating -----
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        if it > 800_000:          # safety bound
            break

    # --------------------------------------------------------------
    # Exact sub‑gradient refinement on the true max‑convolution
    # --------------------------------------------------------------
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total

    eta0 = 0.7
    sub_last_improve = time.time()
    for sub_it in range(30_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[n={n} Subgrad iter={sub_it:05d}] new best {val:.6f}")

        # active set mask (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(sub_it + 1)       # diminishing step‑size
        a = a * np.exp(-eta * grad_a)          # mirror‑descent (multiplicative)
        s = a.sum()
        if s < 1e-12:                          # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 6.0:
            break

    return global_best_seq, global_best_val

        corr_fft_len_big2 = _next_pow2(3 * n_big2 - 2)

        if best_seq is not None:
            # choose size of source vector for interpolation
            src = best_seq
            src_n = src.shape[0]
            x_old = np.linspace(0.0, 1.0, src_n, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n_big2, endpoint=False)
            a_big2 = np.interp(x_new, x_old, src)
            a_big2 = np.maximum(a_big2, 0.0)
            a_big2 /= a_big2.sum()
        else:
            a_big2 = np.random.dirichlet(np.ones(n_big2))

        print("[big2] optimisation on enlarged grid (n=16384)")
        best_seq, best_val = _deep_optimize_power_surrogate(
            a_big2, n_big2, conv_fft_len_big2, corr_fft_len_big2,
            deadline, best_seq, best_val
        )

    # --------------------------------------------------------------
    # Finalisation – ensure feasibility for the evaluator
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[final] best objective ≈ {best_val:.6f} with length {best_seq.shape[0]}")
    return best_seq.tolist()
