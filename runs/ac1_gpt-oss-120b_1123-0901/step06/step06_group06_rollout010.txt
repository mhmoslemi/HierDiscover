def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_val = float('inf')
    # Sequence length schedule for progressive refinement
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]
    # But we only go as far as time permits.
    for n in lengths:
        if time.time() > deadline:
            break
        # compute FFT lengths
        conv_len = 1 << int(np.ceil(np.log2(2*n-1)))
        corr_len = 1 << int(np.ceil(np.log2(2*n-1 + n-1)))
        # Build initial candidates
        candidates = []
        # 1: upscaled previous best if any
        if best_seq is not None:
            # upscale via linear interpolation
            old_n = len(best_seq)
            # simple repeat or interpolation
            a_up = np.interp(np.linspace(0, old_n-1, n), np.arange(old_n), best_seq)
            a_up = np.maximum(a_up, 0.0)
            if a_up.sum() < 1e-12:
                a_up = np.random.dirichlet(np.ones(n))
            else:
                a_up = a_up / a_up.sum()
            candidates.append(a_up)
        # 2-... random candidates
        for _ in range(5):
            cand = np.random.dirichlet(np.ones(n))
            candidates.append(cand)
        # Additional shapes: exponentials, gaussian
        for gamma in [0.0005,0.001,0.002,0.004]:
            i = np.arange(n)
            shape = np.exp(-gamma*i) + np.exp(-gamma*(n-1-i))
            shape = shape / shape.sum()
            candidates.append(shape)
        # maybe power law
        idx = np.arange(1,n+1)
        prod = idx * (n - idx + 1)
        for alpha in [-1.0, -0.8, -0.5, -0.3]:
            shape = np.power(prod, alpha, where=prod>0, out=np.zeros_like(prod))
            if shape.sum()>0:
                shape = shape / shape.sum()
                candidates.append(shape)

        # For each candidate perform optimization
        for cid, a0 in enumerate(candidates):
            if time.time() > deadline:
                break
            a = np.copy(a0)
            # ensure nonzero sum
            if a.sum() == 0:
                a = np.random.dirichlet(np.ones(n))
            # iteration schedule: first smoothed (Adam) for some steps, then EG
            # Determine number of steps based on n and remaining time.
            # simple: let iters_total = int(20000 * 1024 / n) maybe.
            iters_total = int(20000 * 1024 / max(n,1))
            # Split: early smoothed stage [0:0.2*iters_total], later EG [0.2*iters_total:]
            smoothed_iters = max(1, iters_total // 5)
            eg_iters = iters_total - smoothed_iters

            # Smoothed Adam phase
            logits = np.log(a + 1e-15)
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.07 * np.sqrt(2048.0 / n)
            lam = 0.1
            lam_factor = 1.02
            last_improve = time.time()
            for t in range(smoothed_iters):
                # compute softmax a
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()
                # compute conv
                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                # smooth weights
                w = np.exp(lam * (conv - max_c))
                sum_w = w.sum()
                if sum_w == 0:
                    w = np.full_like(w, 1.0 / w.shape[0])
                else:
                    w /= sum_w
                # gradient
                grad = 2.0 * _corr_fft(w, a, corr_len)
                avg = np.dot(a, grad)
                grad_log = a * (grad - avg)
                # Adam update
                m = beta1 * m + (1 - beta1) * grad_log
                v = beta2 * v + (1 - beta2) * (grad_log ** 2)
                m_hat = m / (1 - beta1 ** (t+1))
                v_hat = v / (1 - beta2 ** (t+1))
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
                lam = min(lam * lam_factor, 1e6)
                # evaluate objective
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cid} Adam iter={t}] new best {val:.6f}")
                # stagnation
                if time.time() - last_improve > 5.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-3
                    last_improve = time.time()
                if time.time() > deadline:
                    break

            # Exponentiated gradient (exact max) phase
            a = a / a.sum()
            eta = 0.3 + 0.2 * np.random.rand()  # randomize step size a bit
            sub_last_improve = time.time()
            for sub_it in range(eg_iters):
                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                # find active mask
                # Use tolerance relative to max
                active = conv >= max_c - 1e-12
                # allocate uniform weight among active entries
                w = np.zeros_like(conv)
                if active.any():
                    w[active] = 1.0 / active.sum()
                grad = 2.0 * _corr_fft(w, a, corr_len)
                a = a * np.exp(-eta * grad)
                s = a.sum()
                if s < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s
                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last_improve = time.time()
                    print(f"[n={n} cand={cid} EG iter={sub_it}] new best {val:.6f}")
                if time.time() - sub_last_improve > 8.0:
                    break
                if time.time() > deadline:
                    break

        # Fine-tune on best for this n
        if best_seq is not None and len(best_seq) == n:
            a = best_seq.copy()
            logits = np.log(a + 1e-15)
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.04 * np.sqrt(2048.0/n)
            lam = 0.3
            lam_factor = 1.015
            last_improve = time.time()
            fine_iters = int(40000 * 1024 / n) + 5000
            for t in range(fine_iters):
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()
                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)
                grad = 2.0 * _corr_fft(w, a, corr_len)
                avg = np.dot(a, grad)
                grad_log = a * (grad - avg)
                m = beta1 * m + (1 - beta1) * grad_log
                v = beta2 * v + (1 - beta2) * (grad_log ** 2)
                m_hat = m / (1 - beta1 ** (t+1))
                v_hat = v / (1 - beta2 ** (t+1))
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
                lam = min(lam * lam_factor, 1e6)
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} fine iter={t}] new best {val:.6f}")
                if time.time() - last_improve > 10.0:
                    logits += (np.random.rand(n) -0.5)*5e-3
                    last_improve = time.time()
                if time.time() > deadline:
                    break

    # final fallback
    if best_seq is None:
        n0 = 1024
        best_seq = np.full(n0, 1.0/n0)
        best_val = evaluate_sequence(best_seq.tolist())
        print("Fallback uniform sequence used, val:", best_val)

    # Save best as global variable for next run
    globals()['height_sequence_1'] = best_seq.tolist()
    print("Best value:", best_val, "Length:", len(best_seq))
    return best_seq.tolist()

def lp_improve(f, n, conv_len, eps=1e-8):
    conv = _conv_fft(f, conv_len)
    M = conv.max()
    active = np.where(conv >= M - eps)[0]
    if active.size == 0:
        return None
    # construct matrix A: shape (len(active), n)
    A = np.zeros((active.size, n))
    for idx, k in enumerate(active):
        # for each i where 0 <= i < n and 0 <= k - i < n
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # In row idx, entries at column i from i_min..i_max are f[k - i]
        # This is reversed slice of f.
        A[idx, i_min:i_max+1] = f[k - np.arange(i_min, i_max+1)]
    # Build LP: maximize sum b
    # Using linprog: minimize -sum(b)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    # linprog expects A_ub x <= b_ub
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=M * np.ones(active.size),
                                 bounds=bounds, method='highs', options={'presolve': True})
    if not res.success or res.x is None:
        return None
    b = res.x
    # Ensure sum >0
    if b.sum() <= 0:
        return None
    # Normalize to sum=1
    b = b / b.sum()
    return b
```

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, conv_len: int) -> float:
    """Exact objective (sum(a) is assumed to be 1)."""
    conv = _conv_fft(a, conv_len)
    return 2.0 * n * conv.max()

    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # learning‑rate scales mildly with size
    lr = 0.08 * math.sqrt(2048.0 / n)

    lam = 0.2          # temperature (smooth‑max)
    lam_factor = 1.018

    last_improve = time.time()
    for t in range(1, max_iters + 1):
        # ----- simplex mapping -----
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ----- smooth max -----
        conv = _conv_fft(a, conv_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0.0 else np.full_like(w, 1.0 / w.size)

        # ----- gradient of surrogate -----
        grad_a = 2.0 * _corr_fft(w, a, corr_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature cool‑down → true max
        lam = min(lam * lam_factor, 1e6)

        # ----- check improvement -----
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t:05d}] new best {val:.6f}")

        # stagnation → gentle random shake
        if time.time() - last_improve > 7.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break
    return best_val, best_seq, a


def _eg_phase(a0: np.ndarray,
              n: int,
              conv_len: int,
              corr_len: int,
              max_iters: int,
              deadline: float,
              best_val: float,
              best_seq: np.ndarray):
    """
    Exact exponentiated‑gradient (sub‑gradient) phase.
    Returns possibly updated (best_val, best_seq, a).
    """
    a = a0 / a0.sum()
    eta = 0.55                # base step‑size
    sub_last_improve = time.time()
    for it in range(max_iters):
        conv = _conv_fft(a, conv_len)
        max_c = conv.max()
        # indices attaining the max (tiny tolerance)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _corr_fft(w, a, corr_len)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _conv_fft(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[n={n} EG iter={it}] new best {val:.6f}")

        if time.time() - sub_last_improve > 8.0:
            break
        if time.time() > deadline:
            break
    return best_val, best_seq, a


def _fine_tune(a0: np.ndarray,
               n: int,
               conv_len: int,
               corr_len: int,
               max_iters: int,
               deadline: float,
               best_val: float,
               best_seq: np.ndarray):
    """
    A longer Adam run (temperature → true max) used on the globally
    best sequence for the current length.
    """
    a = a0.copy()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.045 * math.sqrt(2048.0 / n)

    lam = 0.4
    lam_factor = 1.012
    last_improve = time.time()
    for t in range(1, max_iters + 1):
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        conv = _conv_fft(a, conv_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

        grad_a = 2.0 * _corr_fft(w, a, corr_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        lam = min(lam * lam_factor, 1e6)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} fine‑tune iter={t:05d}] new best {val:.6f}")

        if time.time() - last_improve > 10.0:
            logits += (np.random.rand(n) - 0.5) * 5e-3
            last_improve = time.time()

        if time.time() > deadline:
            break
    return best_val, best_seq

def _lp_refine(f: np.ndarray,
               n: int,
               conv_len: int,
               eps: float = 1e-6) -> np.ndarray | None:
    """
    Linear‑programming refinement step.
    Build a tiny LP using only those convolution indices where
    f*f attains (or almost attains) its maximum.  Maximise Σ b_j
    under the constraints (f * b)_k ≤ M for the active indices.
    Returns the normalised solution ``b`` or ``None`` on failure.
    """
    conv = _conv_fft(f, conv_len)
    M = conv.max()
    # active set – at most a few dozen entries
    active = np.where(conv >= M - eps)[0]
    if active.size == 0:
        return None

    # Construct a dense matrix A with shape (len(active), n)
    A = np.zeros((active.size, n), dtype=np.float64)
    for row, k in enumerate(active):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        cols = np.arange(i_min, i_max + 1, dtype=np.int64)
        A[row, cols] = f[k - cols]

    c = -np.ones(n)                     # maximise Σb  → minimise -Σb
    bounds = [(0, None)] * n
    ub = M * np.ones(active.shape[0], dtype=np.float64)

    # Use the high‑performance "highs" method from SciPy
    res = optimize.linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                           method='highs', options={'presolve': True})
    if not res.success or res.x is None:
        return None
    b = res.x
    if b.sum() < 1e-12:
        return None
    return b / b.sum()


def _mix_sequences(f: np.ndarray, g: np.ndarray, t: float = 0.2) -> np.ndarray:
    """Convex blend f and g, then re‑normalise to sum = 1."""
    a = (1.0 - t) * f + t * g
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones_like(a))
        s = a.sum()
    return a / s

def _random_gaussian_mix(n: int, n_comp: int = 3) -> np.ndarray:
    """Mixture of a few Gaussians with random centres and widths."""
    i = np.arange(n, dtype=np.float64)
    weights = np.random.dirichlet(np.ones(n_comp))
    mix = np.zeros(n, dtype=np.float64)
    for w in weights:
        mu = np.random.uniform(0, n - 1)
        sigma = np.random.uniform(n / 20.0, n / 4.0)
        mix += w * np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    mix = np.maximum(mix, 0.0)
    if mix.sum() == 0:
        mix = np.random.dirichlet(np.ones(n))
    else:
        mix /= mix.sum()
    return mix


def _build_candidates(n: int) -> list[np.ndarray]:
    """Diversified initial pool for a given length."""
    cand = []

    # 0) warm‑start from the global variable if present
    if "height_sequence_1" in globals():
        h = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if h.shape[0] > n:
            h = h[:n]
        else:
            h = np.pad(h, (0, n - h.shape[0]))
        h = np.maximum(h, 0.0)
        if h.sum() > 0:
            h = h / h.sum()
        else:
            h = np.random.dirichlet(np.ones(n))
        cand.append(h)

    # 1) plain Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # 2) spiky Dirichlet (concentration < 1)
    for conc in (0.2, 0.4, 0.6):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 3) over‑dispersed Dirichlet (concentration > 1)
    for conc in (1.5, 2.0, 3.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 4) two‑sided exponential decay
    for gamma in (0.0008, 0.0015, 0.003, 0.006):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape = shape / shape.sum()
        cand.append(shape)

    # 5) smooth tent (linear rise then fall)
    left = np.linspace(0, 1, n // 2, endpoint=False)
    right = left[::-1]
    tent = np.concatenate([left, right])
    if tent.shape[0] < n:
        tent = np.concatenate([tent, [0.0]])
    tent = np.maximum(tent, 0)
    tent = tent / tent.sum()
    cand.append(tent)

    # 6) cosine bump
    i = np.arange(n, dtype=np.float64)
    coswin = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    coswin = coswin / coswin.sum()
    cand.append(coswin)

    # 7) random sparse pattern (≈ √n spikes)
    m_spikes = max(2, int(np.sqrt(2 * n)))
    sparse = np.zeros(n, dtype=np.float64)
    idxs = np.random.choice(n, size=m_spikes, replace=False)
    sparse[idxs] = np.random.rand(m_spikes)
    if sparse.sum() > 0:
        sparse = sparse / sparse.sum()
        cand.append(sparse)

    # 8) mixture of a few Gaussians
    cand.append(_random_gaussian_mix(n, n_comp=3))
    cand.append(_random_gaussian_mix(n, n_comp=5))

    # 9) power‑law edge‑concentrated shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 5):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape = shape / shape.sum()
            cand.append(shape)

    return cand

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence that minimises the evaluation
    function.  The routine respects the time budget and returns the best
    sequence found (as a plain Python list).  The best sequence is also
    stored in the global variable ``height_sequence_1`` for later runs.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_val = float("inf")

    # Length schedule – starts small, doubles each step.
    lengths = [256, 512, 1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_len = _next_pow2(2 * n - 1)
        corr_len = _next_pow2(3 * n - 2)

        # ------------------------------------------------------------------
        # Build a pool of starting points.
        candidates = _build_candidates(n)

        for cid, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # Normalise to a simplex; work in the log‑space (soft‑max).
            a0 = np.maximum(a0, 0.0)
            if a0.sum() == 0:
                a0 = np.random.dirichlet(np.ones(n))
            a0 = a0 / a0.sum()

            # Adam (smooth‑max) phase
            # Iteration budget shrinks as n grows.
            max_iters_adam = max(2000, int(30000 * 1024 / n))
            best_val, best_seq, a_mid = _adam_phase(
                a0, n, conv_len, corr_len,
                max_iters=max_iters_adam,
                deadline=deadline,
                best_val=best_val,
                best_seq=best_seq)

            # Exact sub‑gradient (exponentiated‑gradient) phase
            max_iters_eg = max(2000, int(12000 * 1024 / n))
            best_val, best_seq, a_mid = _eg_phase(
                a_mid, n, conv_len, corr_len,
                max_iters=max_iters_eg,
                deadline=deadline,
                best_val=best_val,
                best_seq=best_seq)

        # --------------------------------------------------------------
        # LP refinement on the current global best for this length.
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] LP refinement (if feasible)…")
            g = _lp_refine(best_seq, n, conv_len, eps=1e-6)
            if g is not None:
                # mix a little bit and fine‑tune again
                mixed = _mix_sequences(best_seq, g, t=0.2)
                best_val, best_seq, _ = _adam_phase(
                    mixed, n, conv_len, corr_len,
                    max_iters=1000,
                    deadline=deadline,
                    best_val=best_val,
                    best_seq=best_seq)
                best_val, best_seq, _ = _eg_phase(
                    best_seq, n, conv_len, corr_len,
                    max_iters=2000,
                    deadline=deadline,
                    best_val=best_val,
                    best_seq=best_seq)

        # --------------------------------------------------------------
        # Fine‑tune the (possibly) updated global best.
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] final fine‑tune (val={best_val:.6f})")
            max_fine = int(60000 * 1024 / n) + 5000
            best_val, best_seq = _fine_tune(best_seq, n, conv_len, corr_len,
                                            max_iters=max_fine,
                                            deadline=deadline,
                                            best_val=best_val,
                                            best_seq=best_seq)

        # Prepare for the next size – upscale the current best (if any).
        if best_seq is not None and len(best_seq) == n:
            # linear interpolation to the next length (if it exists)
            next_n_idx = lengths.index(n) + 1
            if next_n_idx < len(lengths):
                nxt = lengths[next_n_idx]
                xs = np.linspace(0, 1, n)
                xs_nxt = np.linspace(0, 1, nxt)
                best_seq = np.interp(xs_nxt, xs, best_seq)
                best_seq = np.maximum(best_seq, 0.0)
                best_seq = best_seq / best_seq.sum()
                # keep it in the global variable for the following iteration
                globals()['height_sequence_1'] = best_seq.tolist()

    # ------------------------------------------------------------------
    # Fallback – uniform vector (should never be needed).
    if best_seq is None:
        n0 = 1024
        best_seq = np.full(n0, 1.0 / n0, dtype=np.float64)
        print("Fallback uniform sequence used.")

    # Store the final answer globally.
    globals()['height_sequence_1'] = best_seq.tolist()
    print(f"\n*** Finished: best value = {best_val:.6f}, length = {len(best_seq)} ***")
    return best_seq.tolist()
