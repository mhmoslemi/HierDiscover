"""
Propose a low‑evaluation step function.

Algorithm
---------
1. **Diverse warm‑starts** – generate many plausible vectors
   (uniform, random Dirichlet, edge‑exponential/power‑law families,
   sparse random support, and the previously best sequence if present).
   Keep the top‑K by a cheap FFT‑based objective.

2. **Smooth‑max Adam** – for each start run Adam on the
   log‑sum‑exp surrogate of the max‑convolution while keeping the
   vector on the simplex via a soft‑max.  The temperature (λ) grows
   quickly, the learning rate decays, and occasional random “shakes”
   are added when progress stalls.

3. **Exact‑max exponentiated‑gradient** – after Adam switch to the
   true max‑convolution, using the active set as a uniform weight.
   Perform many exponentiated‑gradient steps with a decreasing step‑size.

4. **Linear‑program refinement** – using the current vector, collect the
   indices where the convolution is near its maximum, solve a tiny LP
   (min t s.t. f∗g ≤ t on these indices, Σg=1, g≥0) and mix the result
   with the current vector.  Repeat a few times.

5. **Scale up** – if enough time remains, linearly interpolate the best
   4096‑vector to 8192 entries and repeat steps 2‑4 on the finer grid.

All steps respect the 1000 s budget and keep the sequence non‑negative.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the objective (sum(a) is assumed ≈ 1)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp (smooth‑max) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr0 = 0.07                # a slightly larger start learning rate
    lr = lr0

    # temperature schedule – reach high λ quickly
    lam = 0.8
    lam_factor = 1.025
    lam_cap = 1e6

    best_seq = global_best_seq
    best_val = global_best_val
    last_improve = time.time()
    it = 0
    max_iter = 350_000

    while time.time() < deadline and it < max_iter:
        # simplex via soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # surrogate evaluation
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        lr = lr0 / (1.0 + it * 5e-7)   # tiny decay to stabilize late stage
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # random shake when stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient steps on the true max‑convolution."""
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()

    eta0 = 0.7
    eta = eta0
    best_seq = global_best_seq
    best_val = global_best_val
    last_improve = time.time()
    max_iter = 30_000
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it:05d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # simple schedule for η
        if (it + 1) % 5000 == 0:
            eta *= 0.6

        # early stop if no improvement for a while
        if time.time() - last_improve > 6.0:
            break

    return best_seq, best_val


def _lp_refinement(
    a: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Tiny linear‑program refinement:
    minimise t such that (a ∗ g)[k] ≤ t for the K largest convolution entries,
    Σ g = 1, g ≥ 0.
    The solution g is mixed with a (0.5/0.5) and kept if it improves the bound.
    """
    # compute current convolution
    conv = _full_convolution(a, conv_fft_len)
    K = min(30, conv.size)
    # indices of the largest values (potentially active constraints)
    top_idx = np.argpartition(conv, -K)[-K:]

    # Build sparse A_ub for constraints conv_f(g) ≤ t
    rows = []
    cols = []
    data = []
    for row, k in enumerate(top_idx):
        # coefficients for g_j: a_{k-j}
        start_j = max(0, k - (n - 1))
        end_j = min(n - 1, k)
        for j in range(start_j, end_j + 1):
            rows.append(row)
            cols.append(j)
            data.append(a[k - j])
        # coefficient for t (last variable) is -1
        rows.append(row)
        cols.append(n)          # index of t
        data.append(-1.0)

    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(K, n + 1))
    b_ub = np.zeros(K)

    # Equality Σ g = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]  # t is non‑negative
    c = np.zeros(n + 1)
    c[n] = 1.0                               # minimise t

    remaining = deadline - time.time()
    if remaining <= 0.0:
        return a, global_best_val

    # solve LP
    res = scipy.optimize.linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=bounds,
        method="highs",
        options={"time_limit": max(0.0, remaining)},
    )
    if not res.success:
        return a, global_best_val

    g = res.x[:n]
    # mix old and new vectors
    new_a = 0.5 * a + 0.5 * g
    new_a = np.maximum(new_a, 0.0)
    new_a /= new_a.sum()

    new_val = _objective(new_a, n, conv_fft_len)
    if new_val < global_best_val:
        print(f"[LP refine n={n}] improvement {new_val:.6f}")
        global_best_val = new_val
        global_best_seq = new_a.copy()
    return new_a, global_best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """Run Adam → sub‑gradient → optional LP refinements."""
    # Adam phase
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    # Sub‑gradient phase
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    # LP refinements (up to three passes, time‑aware)
    for rep in range(3):
        if time.time() >= deadline:
            break
        best_seq, best_val = _lp_refinement(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            best_seq,
            best_val,
        )
    return best_seq, best_val


def _generate_parametric_candidates(
    n: int, warm_start: np.ndarray | None
) -> list[np.ndarray]:
    """Produce a rich collection of plausible vectors."""
    candidates = []

    # warm‑start and cyclic shifts
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # random Dirichlet with several concentrations
    for conc in (0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # sparse support candidates – a few random spikes
    for k in (2, 3, 5, 10, 20):
        for _ in range(3):
            g = np.zeros(n)
            idx = np.random.choice(n, size=k, replace=False)
            g[idx] = np.random.rand(k)
            g = np.maximum(g, 0.0)
            g_sum = g.sum()
            if g_sum < 0.01:
                g = np.random.dirichlet(np.ones(n))
            else:
                g /= g_sum
            candidates.append(g)

    # edge‑exponential families (with optional uniform mix)
    i = np.arange(n, dtype=np.float64)
    for alpha in np.logspace(-2, 1.0, num=12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # edge‑power‑law families
    for p in np.linspace(0.2, 5.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.5, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # two‑Gaussian edges
    for gamma in (0.05, 0.2, 0.8, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # sinusoidal half‑wave
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    return candidates


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Entry point – searches for a sequence with minimal evaluation score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Base grid size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start from previously stored best (if any)
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – generate cheap candidates and keep the best few
    all_candidates = _generate_parametric_candidates(n, warm_start)

    cand_vals = []
    for idx, cand in enumerate(all_candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    cand_vals.sort(key=lambda x: x[0])
    top_k = 12
    top_candidates = [c for _, c in cand_vals[:top_k]]

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation of each start
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation start #{start_id + 1}/{len(top_candidates)} (n={n}) ---")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – upscale if enough time remains
    time_left = deadline - time.time()
    if time_left > 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation to finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big
            conv_fft_len = conv_fft_len_big
            corr_fft_len = corr_fft_len_big

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Final clipping to the evaluator's allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search finished: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]}) ***")
    return best_seq.tolist()
