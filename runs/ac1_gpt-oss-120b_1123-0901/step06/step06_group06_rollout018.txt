1. def _conv_fft(a, fft_len):
   ft = np.fft.rfft(a, fft_len)
   conv = np.fft.irfft(ft * ft, fft_len)
   return conv[:2*a.shape[0]-1]

2. def _corr_fft(w, a, fft_len):
   rev_a = a[::-1]
   fw = np.fft.rfft(w, fft_len)
   fr = np.fft.rfft(rev_a, fft_len)
   corr = np.fft.irfft(fw * fr, fft_len)
   start = a.shape[0] - 1
   return corr[start:start + a.shape[0]]

3. def _objective(a, n, conv_len):
   conv = _conv_fft(a, conv_len)

   if sum_a < 1e-12:

def adam_optimize(a_init, n, conv_len_fft, corr_len_fft, max_iters, lr, lam_start, lam_factor, deadline):
    a = a_init.copy()
    # Convert to logits for simplex
    a = a / a.sum()
    logits = np.log(a + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a.copy()
    best_val = _objective(a, n, conv_len_fft)
    lam = lam_start
    last_improve_time = time.time()
    for t in range(1, max_iters+1):
        # compute softmax
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()
        # conv
        conv = _conv_fft(a, conv_len_fft)
        max_c = conv.max()
        # smooth max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        grad_a = 2.0 * _corr_fft(w, a, corr_len_fft)
        # backprop through softmax
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # update lambda
        lam = min(lam * lam_factor, 1e6)
        # compute true objective
        val = 2.0 * n * max_c  # assuming sum=1.
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve_time = time.time()
        # stagnation: if time since last improve > some seconds, add random noise
        if time.time() - last_improve_time > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve_time = time.time()
        if time.time() > deadline:
            break
    return best_a, best_val

def exponentiated_gradient_optimize(a_init, n, conv_len_fft, corr_len_fft, max_iters, eta, deadline):
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    conv = _conv_fft(a, conv_len_fft)
    best_val = 2.0 * n * conv.max()
    last_improve = time.time()
    for it in range(max_iters):
        conv = _conv_fft(a, conv_len_fft)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        # if any active consider uniform over them
        if np.any(active):
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad_a = 2.0 * _corr_fft(w, a, corr_len_fft)
        a = a * np.exp(-eta * grad_a)
        # renormalize
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        conv = _conv_fft(a, conv_len_fft)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
        if time.time() - last_improve > 8.0:
            break
        if time.time() > deadline:
            break
    return best_a, best_val

    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"=== Starting search for length {n} ===")
        # compute FFT lengths
        conv_len = 1 << int(np.ceil(np.log2(2*n-1)))
        corr_len = 1 << int(np.ceil(np.log2(3*n-2)))  # w length + a length -1
        # Build candidate list
        candidates = []
        # Warm start from global var if exists
        if 'height_sequence_1' in globals():

            h = np.maximum(h, 0.0)

            candidates.append(cand)

        for _ in range(30):

        # Random sparse patterns

        for idx, cand in enumerate(candidates):

            remaining = deadline - time.time()

            # parameters for Adam

                                                deadline)
            if refined_val < best_val:
                best_val = refined_val
                best_seq = refined.copy()

            # exponentiated gradient finish

            refined2, refined2_val = exponentiated_gradient_optimize(best_seq, n,
                                                                   conv_len, corr_len,
                                                                   max_iters=8000,
                                                                   eta=0.55,
                                                                   deadline=deadline)
            if refined2_val < best_val:
                best_val = refined2_val

"""
Propose a non‑negative step‑function (list of heights) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

where a*a is the linear convolution (numpy.convolve).

Strategy
--------
* For several lengths (1024, 2048, 4096, 8192) build a **large pool of
  diversified candidate shapes**:
  – warm‑start from the previous global solution (height_sequence_1);
  – Dirichlet draws with different concentration parameters;
  – random Beta‑distribution shapes (α,β drawn on a log‑scale);
  – two‑sided exponentials;
  – power‑law families;
  – sparse random spikes.
  The candidates are normalised to sum = 1 (the objective is scale‑invariant).

* **Fast evaluation** of the objective uses FFT‑based convolution
  (`_conv_fft`).  The best candidate seen so far (globally across
  all lengths) is remembered.

* The current best shape for the *current length* is refined with:
  – an Adam optimiser on a smooth‑max surrogate of the true max;
  – a few hundred iterations of an exponentiated‑gradient (sub‑gradient)
    phase that works directly on the true max.
  Both phases respect the remaining time budget.

* The whole search respects a total time budget (default 1000 s).  If the
  budget expires the routine simply returns the best sequence seen so far.

The returned object is a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based linear convolution (a*a) and the cross‑correlation needed for
# the gradient of the smooth‑max surrogate.
def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact evaluation of the objective (no clamping)."""
    conv = _conv_fft(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Parametric families for the initial pool
def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) shape on {0,…,n‑1} (shifted by 0.5 to avoid zero)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # safe log‑power handling
    term1 = np.exp((alpha - 1.0) * np.log(x, where=x > 0, out=np.zeros_like(x)))
    term2 = np.exp((beta - 1.0) * np.log(1.0 - x, where=(1.0 - x) > 0,
                    out=np.zeros_like(x)))
    shape = term1 * term2
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s < 1e-12:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape = shape / s
    return shape


def _exp_shape(n: int, gamma: float) -> np.ndarray:
    """Two‑sided exponential decay."""
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s < 1e-12:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape = shape / s
    return shape


def _powerlaw_shape(n: int, alpha: float) -> np.ndarray:
    """Edge‑concentrated power‑law: (i·(n‑i+1))^α."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
    s = shape.sum()
    if s < 1e-12:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape = shape / s
    return shape


def _sparse_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random sparse pattern with ≈√(2n) spikes."""
    m = max(1, int(np.sqrt(2 * n)))
    sparse = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=m, replace=False)
    sparse[idx] = rng.random(m) + 0.1  # avoid exact zeros on selected spots
    s = sparse.sum()
    if s < 1e-12:
        sparse = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        sparse = sparse / s
    return sparse


# ----------------------------------------------------------------------
def _build_candidate_pool(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse list of initial shapes (all normalised to sum = 1)."""
    cand = []

    # 1) warm‑start from the previous global solution if it exists
    if "height_sequence_1" in globals():
        h = np.asarray(height_sequence_1, dtype=np.float64)
        if h.shape[0] > n:
            h = h[:n]
        else:
            h = np.pad(h, (0, n - h.shape[0]), mode="constant")
        h = np.maximum(h, 0.0)
        if h.sum() > 0:
            h = h / h.sum()
            cand.append(h)

    # 2) Dirichlet draws (different concentrations)
    for conc in (0.5, 1.0, 2.0, 5.0):
        cand.append(rng.dirichlet(np.full(n, conc)))

    # 3) Random Beta shapes (α and β on a log‑scale)
    for _ in range(40):
        alpha = 10 ** rng.uniform(-2.0, 1.0)   # 0.01 … 10
        beta = 10 ** rng.uniform(-2.0, 1.0)
        shape = _beta_shape(n, alpha, beta)
        # add a random circular shift – linear convolution is not shift‑invariant
        shift = rng.integers(n)
        shape = np.roll(shape, shift)
        cand.append(shape)

    # 4) Two‑sided exponentials
    for gamma in rng.uniform(0.0003, 0.01, size=20):
        cand.append(_exp_shape(n, float(gamma)))

    # 5) Power‑law family
    for alpha in np.linspace(-2.0, -0.1, 15):
        cand.append(_powerlaw_shape(n, alpha))

    # 6) Random sparse patterns
    for _ in range(15):
        cand.append(_sparse_shape(n, rng))

    return cand


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   max_iters: int,
                   lr: float,
                   lam_start: float,
                   lam_factor: float,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate.
    Returns (best_sequence, best_value).
    """
    a = a0.copy()
    a = a / a.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    best_a = a.copy()
    conv = _conv_fft(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)

    lam = lam_start
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # soft‑max → simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # convolution & smooth‑max weights
        conv = _conv_fft(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (linear in w)
        grad_a = 2.0 * _corr_fft(w, a, corr_fft_len)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule (approach true max)
        lam = min(lam * lam_factor, 1e6)

        # exact objective (sum(a) == 1 by construction)
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # random shake if we are stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val


def _expgrad_optimize(a0: np.ndarray,
                      n: int,
                      conv_fft_len: int,
                      corr_fft_len: int,
                      max_iters: int,
                      eta: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (sub‑gradient) phase using the true max.
    Returns (best_sequence, best_value).
    """
    a = a0.copy()
    a = a / a.sum()
    best_a = a.copy()
    conv = _conv_fft(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    last_improve = time.time()

    for it in range(max_iters):
        conv = _conv_fft(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _corr_fft(w, a, corr_fft_len)
        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against collapse
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _conv_fft(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        if time.time() - last_improve > 8.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000.0) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq = None
    best_val = float("inf")

    # Lengths to explore – larger n gives a potentially lower bound,
    # but each iteration is more expensive.
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Exploring length n={n} ===")
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))  # (2n‑1) + n‑1

        # ------------------------------------------------------------------
        # 1) generate a pool of candidate shapes
        candidates = _build_candidate_pool(n, rng)

        # 2) quick evaluation of every candidate
        for cid, cand in enumerate(candidates):
            if time.time() > deadline:
                break
            cand = np.maximum(cand, 0.0)
            if cand.sum() < 1e-12:
                continue
            cand = cand / cand.sum()
            val = _objective(cand, n, conv_fft_len)
            if np.isfinite(val) and val < best_val:
                best_val = val
                best_seq = cand.copy()
                print(f"[n={n} cand={cid}] initial val {val:.6f}  <-- new best")
        # ------------------------------------------------------------------
        # 3) if we have a candidate of the current length, run a long
        #    optimisation (Adam → exponentiated‑gradient)
        if best_seq is not None and best_seq.shape[0] == n:
            # Adam parameters
            max_iters = int(50000 * 1024.0 / n) + 5000
            lr = 0.07 * np.sqrt(2048.0 / n)
            lam_start = 0.2
            lam_factor = 1.020

            print(f"[n={n}] Adam refinement (max_iters={max_iters})")
            refined_a, refined_val = _adam_optimize(best_seq,
                                                    n,
                                                    conv_fft_len,
                                                    corr_fft_len,
                                                    max_iters,
                                                    lr,
                                                    lam_start,
                                                    lam_factor,
                                                    deadline)
            if refined_val < best_val:
                best_val = refined_val
                best_seq = refined_a.copy()
                print(f"[n={n}] Adam improved best to {best_val:.6f}")

            # Exponentiated‑gradient fine‑tune
            print(f"[n={n}] Exponentiated‑gradient fine‑tune")
            refined2_a, refined2_val = _expgrad_optimize(best_seq,
                                                         n,
                                                         conv_fft_len,
                                                         corr_fft_len,
                                                         max_iters=8000,
                                                         eta=0.55,
                                                         deadline=deadline)
            if refined2_val < best_val:
                best_val = refined2_val
                best_seq = refined2_a.copy()
                print(f"[n={n}] EG improved best to {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – uniform vector (should never be needed but keeps things safe).
    if best_seq is None:
        n = 1024
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n, 1 << int(np.ceil(np.log2(2 * n - 1))))

    print(f"\n=== Finished ===\nBest value found: {best_val:.6f} (length {best_seq.shape[0]})")
    return best_seq.tolist()
