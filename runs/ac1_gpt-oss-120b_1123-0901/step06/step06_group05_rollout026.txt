"""
Propose a step‑function (non‑negative heights) that minimises

        F(a) = 2·n·max(a*a)/(sum(a)²)

The implementation builds on the previous search but introduces several
enhancements:

* A *projected sub‑gradient* method (instead of the exponentiated‑gradient
  update) which directly projects onto the probability simplex and often
  converges faster.
* A simple Nesterov‑style momentum for the Adam phase.
* A hierarchical multi‑scale optimisation: start with a modest length,
  optimise, then up‑sample the best shape and continue at larger lengths.
* More aggressive hyper‑parameters (more iterations for small n, a
  temperature schedule that reaches the true max sooner) and an occasional
  random jitter when progress stalls.
* A final cheap stochastic hill‑climb to squeeze any remaining improvement.

All helper functions are top‑level; the returned value is a plain Python
list, ready for ``evaluate_sequence``.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation needed for the sub‑gradient of ``max(a*a)``.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]
    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative vector to ``new_n`` points by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:                     # protect against a zero vector
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Adam optimisation on the smooth‑max surrogate of the objective.
    Uses a Nesterov‑style momentum for a smoother trajectory.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # optimisation variables live in the log‑softmax (logits) space
    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.07                     # a tad larger than the previous version
    lam = 0.2                     # start temperature – very smooth
    lam_factor = 1.025            # increase temperature slowly

    best_a = a0.copy()
    best_val = float('inf')
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # ----- soft‑max → point on the simplex -----
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ----- exact objective (sum(a)=1) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t}] new best {val:.6f}")

        # ----- smooth‑max weights -----
        w = np.exp(lam * (conv - max_c))   # shifted for stability
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

        # ----- gradient of smooth‑max -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- back‑prop through the soft‑max (logits) -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update on logits (with Nesterov look‑ahead) -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits += (lr * m_hat) / (np.sqrt(v_hat) + eps)   # note the ‘+’ for Nesterov

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, 1e5)

        # ----- jitter if stuck -----
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val

def _run_proj_subgrad(a0: np.ndarray, n: int, deadline: float, max_iters: int):

    The update step is a simple gradient step followed by Euclidean projection

    """

    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    last_improve = time.time()
    eta0 = 0.8                     # initial step size (larger than before)

    for it in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} ProjSubgrad iter={it}] new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # sub‑gradient of max‑convolution
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        # gradient step + projection onto simplex
        a = _project_simplex(a - eta * grad_a)

        # stagnation detection
        if time.time() - last_improve > 10.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _local_hill_climb(a0: np.ndarray, n: int, deadline: float,
                     max_trials: int = 200_000):
    """
    Very cheap stochastic local search.
    Moves a small amount of mass ε from one index to another and keeps
    the move if it reduces the exact objective.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    best_a = a0.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    no_improve = 0
    trials = 0
    while time.time() < deadline and trials < max_trials and no_improve < 5_000:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0.0, min(0.05, best_a[i]))
        if eps == 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = cand
            no_improve = 0
            print(f"[n={n} HillClimb] improved to {val:.6f}")
        else:
            no_improve += 1
        trials += 1

    return best_a, best_val

def _generate_initial_candidates(n: int, prev_best):
    """
    Produce a heterogeneous pool of starting points for length ``n``.
    ``prev_best`` (if not ``None``) is up‑sampled and used as a warm‑start.
    """
    candidates = []

    # 1. warm‑start from the best sequence found so far (upsampled)
    if prev_best is not None:
        candidates.append(_upsample(prev_best, n))

    # 2. historic solution if a global variable exists
    hseq = globals().get('height_sequence_1')
    if isinstance(hseq, (list, np.ndarray)):
        h = np.asarray(hseq, dtype=np.float64)
        candidates.append(_upsample(h, n))

    # 3. random Dirichlet families (different concentrations)
    candidates.append(np.random.dirichlet(np.ones(n)))
    for conc in (0.01, 0.1, 0.3):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # 4. edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 5):
        shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # 5. symmetric exponentials
    for gamma in (0.0008, 0.0015, 0.003):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # 6. Hann window
    i = np.arange(n, dtype=np.float64)
    hann = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    hann /= hann.sum()
    candidates.append(hann)

    # 7. half‑sine
    sine = np.sin(np.pi * i / (n - 1))
    sine /= sine.sum()
    candidates.append(sine)

    # 8. Gaussian‑like bumps
    mu = (n - 1) / 2.0
    for sigma in (n / 12, n / 8, n / 5):
        shape = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        shape /= shape.sum()
        candidates.append(shape)

    # Normalise and discard degenerate entries
    cleaned = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 1e-12:
            cleaned.append(c / s)
    return cleaned

def propose_candidate(seed: int = 0, budget_s: float = 1000):
    """
    Search for a non‑negative step‑function (list of heights) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

    The strategy:
      1. start with a modest length (2048) and a pool of diverse seeds;
      2. run Adam on a smooth‑max surrogate;
      3. refine with projected sub‑gradient descent on the exact max;
      4. up‑sample the best shape and repeat for larger lengths
         (4096 → 8192 → 16384);
      5. finish with a cheap stochastic hill‑climb.
    The whole search respects the supplied time budget.
    Returns the best sequence encountered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float('inf')

    # length schedule – we grow the problem gradually
    base_n = 2048
    lengths = [base_n, base_n * 2, base_n * 4, base_n * 8]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        # Build a small but heterogeneous candidate pool
        candidates = _generate_initial_candidates(n, best_seq)

        # iteration budgets – roughly proportional to 1/n
        max_adam = int(20000 * base_n / n) + 6_000
        max_sub = int(6_000 * base_n / n) + 1_500

        for cid, init in enumerate(candidates):
            if time.time() > deadline:
                break

            # ---- Adam on smooth‑max ----
            a_adam, val_adam = _run_adam(init, n, deadline, max_adam)
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n} cand={cid}] Adam set new global best {val_adam:.6f}")

            # ---- Projected sub‑gradient descent ----
            a_sub, val_sub = _run_proj_subgrad(a_adam, n, deadline, max_sub)
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[n={n} cand={cid}] ProjSubgrad set new global best {val_sub:.6f}")

            # ---- Fine‑tune the current global best for this length ----
            if best_seq is not None and len(best_seq) == n:
                a_fine, val_fine = _run_adam(best_seq, n, deadline,
                                             int(0.5 * max_adam))
                if val_fine < best_val:
                    best_val = val_fine
                    best_seq = a_fine.copy()
                    print(f"[n={n}] Fine‑tune improved global best to {val_fine:.6f}")

    # ------------------------------------------------------------------
    # Final cheap local search to squeeze out any remaining improvement
    if best_seq is not None and time.time() < deadline:
        print(f"\n=== Final local hill‑climb (n={len(best_seq)}) ===")
        a_final, val_final = _local_hill_climb(best_seq, len(best_seq), deadline)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final
            print(f"[final] hill‑climb improved global best to {val_final:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform if nothing was produced (should never happen)
    if best_seq is None:
        n_fallback = base_n
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    return best_seq.tolist()
