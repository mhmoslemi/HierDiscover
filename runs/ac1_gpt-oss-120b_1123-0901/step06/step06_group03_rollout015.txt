```
def _generate_candidate_pool_custom(n, warm):
    pool = []
    # warm etc same as earlier
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n//4, n//2, 3*n//4):
            pool.append(np.roll(warm, shift))
    # uniform
    pool.append(np.full(n,1.0/n))
    i = np.arange(n)
    # half sinusoid
    shape = np.sin(np.pi*(i+0.5)/n)
    shape = np.maximum(shape,0)
    shape /= shape.sum()
    pool.append(shape)
    # Beta shapes
    xs = (i+0.5)/n
    alphas = np.logspace(-1, 1, 9)  # 0.1 to 10
    betas = np.logspace(-1,1,9)
    for a in alphas:
        for b in betas:
            shape = (xs**(a-1))*((1-xs)**(b-1))
            if np.any(np.isnan(shape)) or shape.sum() ==0:
                continue
            shape = shape/shape.sum()
            pool.append(shape)
    # double exponential
    for lam in np.linspace(0.1, 3.0, 12):
        shape = np.exp(-lam*i) + np.exp(-lam*(n-1-i))
        shape = shape/shape.sum()
        pool.append(shape)
    # power law edges originally
    idx = np.arange(1, n+1, dtype=np.float64)
    prod = idx * (n - idx + 1)
    for alpha in np.linspace(-2.0, -0.2, 10):
        shape = np.where(prod>0, prod**alpha, 0)
        if shape.sum()>0:
            shape /= shape.sum()
            pool.append(shape)
    # Gaussian
    for sigma in np.linspace(n*0.03, n*0.4, 8):
        shape = np.exp(-0.5*((i - n/2)**2)/sigma**2)
        shape /= shape.sum()
        pool.append(shape)
    # Random Dirichlet
    for a in np.logspace(-2,0,10):
        shape = np.random.dirichlet(np.full(n, a))
        pool.append(shape)
    # Cosine bump
    shape = np.cos(np.pi*(i - n/2)/n)
    shape = np.maximum(shape,0)
    shape = shape/shape.sum()
    pool.append(shape)
    return pool
```

The implementation below searches for a non‑negative step‑function (a list of heights) that minimizes  

\[
F(a)=\frac{2\,n\;\max(a*a)}{(\sum a)^2}\;,
\]

where `a*a` denotes the linear convolution of `a` with itself.  
The algorithm consists of three stages:

1. **Rich candidate pool** – a large collection of very different
   shapes (uniform, sinusoid, beta‑distributions, double‑exponential,
   Gaussian, power–law edges, random Dirichlet draws and a warm‑start
   if one is present) is built and the best few are kept.

2. **Accelerated mirror‑descent optimisation** – each retained candidate
   is refined by  
   * an **Adam** optimiser on a smooth‑max surrogate of the
     `max`‑convolution (the temperature is raised quickly, learning‑rate
     decays and occasional random “shakes’’ avoid stagnation);
   * an **exact sub‑gradient mirror‑descent** phase (exponentiated‑gradient
     updates on the simplex) that works directly on the true maximum.
   The optimiser respects the global time budget and updates a global
   best solution after every improvement.

3. **Upsampling & final polishing** – when time permits the current best
   distribution is linearly interpolated to a larger support
   (`8192` points) and the whole optimisation is repeated, which yields
   a tighter bound.

All numbers are forced to be non‑negative and limited to `1000.0`
(the evaluator’s limits).  The routine respects the 1000 s budget,
prints progress information and finally returns the best sequence
as a plain Python list.

```python
"""
Propose a step‑function minimizing 2·n·max(a*a)/(∑a)².

Algorithm
---------
1. Build a large, diverse pool of candidate vectors (uniform,
   sinusoid, beta‑distribution, double‑exponential, Gaussian,
   power‑law edges, random Dirichlet, warm‑start and its rotations).
2. Evaluate every candidate with a fast FFT‑based objective and keep the
   best few.
3. For each of those candidates run a two‑phase optimiser:
   * Adam on a smooth‑max surrogate (log‑sum‑exp) with a fast‑growing
     temperature, adaptive learning‑rate and occasional random perturbation.
   * Exact mirror‑descent (exponentiated‑gradient) on the true max‑convolution.
   The optimiser updates a global best solution whenever a lower
   objective value is found.
4. If time permits, up‑sample the best vector to a larger grid
   (8192 points) and repeat step 3.
5. Return the best sequence (clipped to the evaluator’s domain).

The code uses only numpy and the supplied evaluate_sequence().
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than *x*."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective (assumes sum(a)>0)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _generate_candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a diverse pool of start vectors for length *n*.
    Includes a warm‑start (if any), uniform, sinusoid, beta‑families,
    double‑exponential, Gaussian, power‑law edges, cosine bump and
    random Dirichlet draws.
    """
    pool: list[np.ndarray] = []

    # warm‑start and its rotations
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # uniform baseline
    pool.append(np.full(n, 1.0 / n))

    i = np.arange(n, dtype=np.float64)
    xs = (i + 0.5) / n                 # points in (0,1)

    # single half‑wave sinusoid
    s = np.sin(np.pi * (i + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    pool.append(s)

    # cosine bump (non‑negative half‑cosine)
    c = np.cos(np.pi * (i - n / 2) / n)
    c = np.maximum(c, 0.0)
    c /= c.sum()
    pool.append(c)

    # beta‑distribution family (shape parameters α,β > 0)
    alphas = np.logspace(-1, 1, 9)   # 0.1 … 10
    betas = np.logspace(-1, 1, 9)
    for a in alphas:
        for b in betas:
            shape = (xs ** (a - 1)) * ((1.0 - xs) ** (b - 1))
            if np.any(np.isnan(shape)):
                continue
            ssum = shape.sum()
            if ssum == 0:
                continue
            shape /= ssum
            pool.append(shape)

    # double‑exponential (two‑sided Laplace) families
    for lam in np.linspace(0.1, 3.0, 12):
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        shape /= shape.sum()
        pool.append(shape)

    # Gaussian‑type bumps centred in the middle (width sweep)
    for sigma in np.linspace(n * 0.03, n * 0.4, 8):
        shape = np.exp(-0.5 * ((i - n / 2) ** 2) / (sigma ** 2))
        shape /= shape.sum()
        pool.append(shape)

    # power‑law edge‑concentrated shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)                # i·(n‑i+1)
    for alpha in np.linspace(-2.0, -0.2, 10):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # random Dirichlet draws with a range of concentration parameters
    for conc in np.logspace(-2, 0, 10):
        shape = np.random.dirichlet(np.full(n, conc))
        pool.append(shape)

    return pool

def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Two‑phase optimiser:

    • Adam on the smooth‑max surrogate (log‑sum‑exp) – fast temperature
      increase, adaptive step size and occasional random shake.
    • Exact mirror‑descent (exponentiated‑gradient) on the true max‑convolution.
    The routine updates and returns the best sequence/value found.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    # Adam hyper‑parameters
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04                                     # a bit larger than before
    lam = 0.5                                     # temperature for smooth‑max
    lam_factor = 1.05                            # grow faster

    last_improve = time.time()
    it = 0
    max_adam_iter = 800_000

    # -------------------------------------------------------------------
    # 1) Adam on smooth‑max surrogate
    # -------------------------------------------------------------------
    while time.time() < deadline and it < max_adam_iter:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # current exact objective (useful for logging)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp formulation)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through the soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase the temperature (approach the true max)
        lam = min(lam * lam_factor, 1e9)

        # occasional random “shake’’ when we have been stagnant
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    # -------------------------------------------------------------------
    # 2) Exact sub‑gradient mirror‑descent
    # -------------------------------------------------------------------
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    total = a.sum()
    a = a / total if total > 0 else np.full(n, 1.0 / n)

    eta0 = 0.8                     # slightly larger initial step size
    sub_last_improve = time.time()
    max_sub_iter = 150_000

    for sub_it in range(max_sub_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[Subgrad n={n} it={sub_it:06d}] new best {val:.6f}")

        # active‑set mask (all indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (mirror‑descent on the simplex)
        eta = eta0 / np.sqrt(sub_it + 1)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 6.0:
            # no improvement for a while – stop
            break

    # -------------------------------------------------------------------
    # 3) Small random perturbation phase (pure exploration)
    # -------------------------------------------------------------------
    if time.time() < deadline:
        a_ref = global_best_seq if global_best_seq is not None else a
        a_ref = np.maximum(a_ref, 0.0)
        a_ref /= a_ref.sum()
        eps_log = 1e-12
        logits_ref = np.log(a_ref + eps_log)

        for rnd_it in range(5_000):
            if time.time() >= deadline:
                break
            # gradually shrinking Gaussian noise on logits
            scale = 0.02 / np.sqrt(rnd_it + 1)
            noisy_logits = logits_ref + (np.random.rand(n) - 0.5) * scale
            mx = noisy_logits.max()
            exp_log = np.exp(noisy_logits - mx)
            a_try = exp_log / exp_log.sum()
            val = _objective(a_try, n, conv_fft_len)
            if val < global_best_val:
                global_best_val = val
                global_best_seq = a_try.copy()
                print(f"[Random n={n} it={rnd_it:04d}] new best {val:.6f}")

    return global_best_seq, global_best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    metric.  The routine respects the supplied time‐budget (default 1000 s)
    and returns the best sequence encountered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Primary support size (medium resolution – cheap enough for many iter.)
    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Warm start from a previously found construction, if available
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        w = np.asarray(warm_start, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]), constant_values=0.0)
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
        else:
            w = None
    else:
        w = None

    # ------------------------------------------------------------------
    # 1) Build and screen a diverse pool of start vectors
    # ------------------------------------------------------------------
    candidates = _generate_candidate_pool(n, w)

    best_seq: np.ndarray | None = None
    best_val = float("inf")
    scored: list[tuple[float, np.ndarray]] = []

    for idx, cand in enumerate(candidates):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        val = _objective(a, n, conv_fft_len)
        scored.append((val, a))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        print(f"[screen {idx:03d}] obj={val:.6f}")

    # keep the most promising starts (up to eight)
    scored.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in scored[:8]]

    # ------------------------------------------------------------------
    # 2) Deep optimisation passes on the promising candidates
    # ------------------------------------------------------------------
    for cid, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[optim start {cid}] launching deep optimisation")
        best_seq, best_val = _deep_optimize(
            a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
        )

    # ------------------------------------------------------------------
    # 3) Optional up‑sampling to a larger grid (if time permits)
    # ------------------------------------------------------------------
    if time.time() + 15 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_big, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("[big] starting optimisation on enlarged grid")
        best_seq, best_val = _deep_optimize(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big,
            deadline, best_seq, best_val
        )

    # ------------------------------------------------------------------
    # Finalisation
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[final] best objective = {best_val:.6f}, length = {best_seq.size}")
    return best_seq.tolist()
```
