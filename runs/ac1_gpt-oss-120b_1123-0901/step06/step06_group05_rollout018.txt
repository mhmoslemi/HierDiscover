def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Returns a vector of length len(a) equal to:

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical safety – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Renormalise just in case of tiny numerical drift
    w_sum = w.sum()
    if w_sum <= 0:
        w = np.full_like(v, 1.0 / len(v))
    else:
        w /= w_sum
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative vector to ``new_n`` points by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:                     # protect against zero vector
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s

        h = np.asarray(hseq, dtype=np.float64)
        candidates.append(_upsample(h, n))

    # Random Dirichlet families (different concentration parameters)
    candidates.append(np.random.dirichlet(np.ones(n)))
    for conc in (0.2, 0.5, 0.8):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # Power‑law edge‑concentrated shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.1, 7):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # Symmetric exponential decays
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # Hann window
    i = np.arange(n, dtype=np.float64)
    hann = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    hann /= hann.sum()
    candidates.append(hann)

    # Half‑sine shape
    sine = np.sin(np.pi * i / (n - 1))
    sine = np.maximum(sine, 0)
    sine /= sine.sum()
    candidates.append(sine)

    # Gaussian‑like bumps (several widths)
    mu = (n - 1) / 2.0
    for sigma in (n / 10, n / 6, n / 4):
        shape = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        shape /= shape.sum()
        candidates.append(shape)

    # Monotone decreasing exponentials (asymmetric)
    for gamma in (0.001, 0.002, 0.004):
        shape = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape /= shape.sum()
        candidates.append(shape)

    # Randomly shuffled Dirichlet (introduces asymmetry)
    for _ in range(2):
        shape = np.random.dirichlet(np.ones(n))
        np.random.shuffle(shape)
        shape /= shape.sum()
        candidates.append(shape)

    # Clean up (avoid zero vectors)
    clean = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        clean.append(c / s)
    return clean

def _run_adam_smooth(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Adam optimiser on the smooth‑max surrogate of the objective.
    Returns the best sequence (as a NumPy array) and its exact objective value.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Parameterise a via soft‑max logits to enforce the simplex constraint
    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.30               # start temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float('inf')
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # Soft‑max mapping (exact projection on the simplex)
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # Exact objective (the denominator is 1 because Σa = 1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t}] new best {val:.6f}")

        # Smooth‑max weights (soft‑max over the convolution vector)
        w = np.exp(lam * (conv - max_c))          # shift by max for stability
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient of the smooth surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Back‑prop through the soft‑max (logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # Temperature schedule (anneal towards true max)
        lam = min(lam * lam_factor, 1e4)

        # Occasionally jitter the logits if stuck for >30 s
        if time.time() - last_improve > 30.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val

def _run_subgrad_simple(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Simple projected sub‑gradient descent on the true max‑convolution.
    Uses the active‑set sub‑gradient and projects onto the probability simplex.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    last_improve = time.time()
    eta0 = 0.55  # initial step‑size multiplier

    for it in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} SubgradSimple iter={it}] new best {val:.6f}")

        # Build a uniform weight over all maximising indices
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # Sub‑gradient of the max‑convolution
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Decaying step‑size (classical SG rule)
        eta = eta0 / np.sqrt(it)

        # Gradient step (move opposite to the sub‑gradient)
        a_tmp = a - eta * grad

        # Project back onto the simplex
        a = _project_simplex(a_tmp)

        # Early exit if nothing improved for a while
        if time.time() - last_improve > 30.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _local_hill_climb(a0: np.ndarray, n: int, deadline: float,
                     max_trials: int = 200_000):
    """
    Very cheap stochastic local search that moves a small mass ε from one
    index to another and keeps the move if it strictly reduces the exact
    objective.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    best_a = a0.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    no_improve = 0
    trials = 0
    while time.time() < deadline and trials < max_trials and no_improve < 5_000:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0.0, min(0.05, best_a[i]))
        if eps == 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = cand
            no_improve = 0
            print(f"[n={n} HillClimb] improved to {val:.6f}")
        else:
            no_improve += 1
        trials += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the evaluation function defined in the problem statement.
    The routine respects the given time budget (seconds) and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_val = float('inf')

    # ------------------------------------------------------------------
    # Stage A – medium length (the sweet spot for quality vs speed)
    # ------------------------------------------------------------------
    n_mid = 4096
    print(f"\n=== Stage A – optimise n={n_mid} ===")
    candidates = _generate_initial_candidates(n_mid, None)

    # Allocate iteration budgets proportionally to length
    max_adam = int(30000 * 2048 / n_mid) + 10_000   # a bit more than before
    max_subgrad = 12_000

    for cid, init in enumerate(candidates):
        if time.time() > deadline:
            break

        # ---- Adam (smooth‑max) ----
        a_adam, val_adam = _run_adam_smooth(init, n_mid, deadline, max_adam)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[n={n_mid} cand={cid}] Adam new global best {val_adam:.6f}")

        # ---- Sub‑gradient (projected) ----
        a_sub, val_sub = _run_subgrad_simple(a_adam, n_mid, deadline, max_subgrad)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[n={n_mid} cand={cid}] SubgradSimple new global best {val_sub:.6f}")

        # ---- Short fine‑tune on the current global best for this length ----
        if best_seq is not None and len(best_seq) == n_mid:
            a_fine, val_fine = _run_adam_smooth(best_seq, n_mid, deadline,
                                                int(0.5 * max_adam))
            if val_fine < best_val:
                best_val = val_fine
                best_seq = a_fine.copy()
                print(f"[n={n_mid}] Fine‑tune improved global best to {val_fine:.6f}")

    # ------------------------------------------------------------------
    # Stage B – upscale to larger lengths if time permits
    # ------------------------------------------------------------------
    for n_big in (8192, 16384):
        if time.time() > deadline:
            break
        print(f"\n=== Stage B – upscale to n={n_big} ===")
        # Warm‑start from the current best (upsampled)
        if best_seq is None:
            a0 = np.random.dirichlet(np.ones(n_big))
        else:
            a0 = _upsample(best_seq, n_big)

        max_adam_big = int(25000 * 2048 / n_big) + 8_000
        max_subgrad_big = 8_000

        # Adam on upsampled shape
        a_adam_big, val_adam_big = _run_adam_smooth(a0, n_big, deadline, max_adam_big)
        if val_adam_big < best_val:
            best_val = val_adam_big
            best_seq = a_adam_big.copy()
            print(f"[n={n_big}] Adam upsampled new global best {val_adam_big:.6f}")

        # Sub‑gradient refinement
        a_sub_big, val_sub_big = _run_subgrad_simple(a_adam_big, n_big, deadline, max_subgrad_big)
        if val_sub_big < best_val:
            best_val = val_sub_big
            best_seq = a_sub_big.copy()
            print(f"[n={n_big}] SubgradSimple upsampled new best {val_sub_big:.6f}")

        # Final short fine‑tune
        a_fine_big, val_fine_big = _run_adam_smooth(best_seq, n_big, deadline,
                                                    int(0.5 * max_adam_big))
        if val_fine_big < best_val:
            best_val = val_fine_big
            best_seq = a_fine_big.copy()
            print(f"[n={n_big}] Fine‑tune upsampled improved best {val_fine_big:.6f}")

    # ------------------------------------------------------------------
    # Stage C – final cheap stochastic hill‑climb on the overall best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print(f"\n=== Final local hill‑climb (n={len(best_seq)}) ===")
        a_final, val_final = _local_hill_climb(best_seq, len(best_seq), deadline)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final
            print(f"[final] hill‑climb improved global best to {val_final:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
