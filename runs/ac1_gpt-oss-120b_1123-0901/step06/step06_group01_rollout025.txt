def normalize(a):

    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        return a
    return a * target / a_sum

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

The search combines three phases:

1. **Broad exploration** – generate many candidate shapes (Gaussian,
   Beta, exponential, Dirichlet, sparse spikes, classic windows) for a
   small set of lengths (including considerably larger than the previous
   best).  Each candidate is evaluated with a fast FFT based exact
   objective and the best few are kept.

2. **Intensive refinement** – the globally best candidate is refined by
   projected gradient descent on the exact max‑convolution (single
   active index sub‑gradient).  The step‑size decays as 1/√t and the
   iterate is projected onto the simplex Σa_i = √(2n).  This phase uses
   the remaining time budget.

3. **Cheap stochastic polishing** – a simple simulated‑annealing
   mass‑transfer hill‑climber makes final fine‑grained improvements.

The whole routine respects the 1000 s budget and prints progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers -------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _normalize(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Clamp to ≥0 and scale to the given L1 sum."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full_like(a, target_sum / a.shape[0])
    return a * (target_sum / s)


def _compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum² (no clipping, a ≥0)."""
    n = a.shape[0]
    total = a.sum()
    Nfft = _next_pow2(2 * n - 1)
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
    max_conv = float(conv.max())
    return 2.0 * n * max_conv / (total * total)


# ----------------------------------------------------------------------
# Shape generators ------------------------------------------------------


def gen_gaussian(n: int, sigma: float) -> np.ndarray:
    """Symmetric Gaussian centred in the middle."""
    idx = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
    return _normalize(a, math.sqrt(2.0 * n))


def gen_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the index grid."""
    x = np.linspace(0.0, 1.0, n)
    # avoid 0^neg and 1^neg by adding eps
    eps = 1e-12
    a = (np.power(x + eps, alpha - 1.0) *
         np.power(1.0 - x + eps, alpha - 1.0))
    return _normalize(a, math.sqrt(2.0 * n))


def gen_exponential(n: int, rate: float) -> np.ndarray:
    """Symmetric exponential decay from the centre."""
    idx = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(idx - mu))
    return _normalize(a, math.sqrt(2.0 * n))


# ----------------------------------------------------------------------
# Optimisers -----------------------------------------------------------


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected Gradient Descent on the exact max‑convolution.
    Uses the single index active set (argmax) for a cheap sub‑gradient.
    Returns the best vector observed and its exact objective value.
    """
    target = math.sqrt(2.0 * n)

    # start on the simplex
    a = _normalize(a0, target)

    Nfft = _next_pow2(2 * n - 1)
    step0 = 0.001 * target          # base step size ~0.1 % of total mass

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # convolution and current max
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD len {n:5d} iter {it:8d}] new best max = {best_score:.6f}")

        # active index (hard max)
        k = int(conv.argmax())
        # compute reverse‑gradient for this shift
        rev = np.zeros_like(a)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                rev[i] = a[j]
        g = 2.0 * rev                     # ∇ max (a*a)

        # gradient step + projection
        lr = step0 / math.sqrt(it)
        a_tmp = a - lr * g
        a = _project_onto_simplex(a_tmp, target)

        # occasional jitter to escape plateaux
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.005 * target / n, size=n)
            a = _project_onto_simplex(a + jitter, target)

    # finally return the best observed (converted back to exact score)
    final_score = _compute_score(best_seq)
    return best_seq, final_score


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on the simplex:
    repeatedly move a small fraction of mass from i to j.
    Accept improvements always; accept worsening moves with a decaying
    simulated‑annealing probability.
    """
    cur_seq = a.copy()
    cur_score = best_score
    target_sum = cur_seq.sum()
    Nfft = _next_pow2(2 * n - 1)

    max_steps = 300_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        # transfer 1‑5 % of the source mass
        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # new max‑convolution
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = 2.0 * n * float(conv.max()) / (target_sum * target_sum)

        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score

    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (Python list) found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # small safety margin

    # ------------------------------------------------------------------
    # Load any previously discovered construction (optional)
    base = None
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Choose a set of lengths to explore (including much larger ones)
    if base is not None:
        base_len = int(base.shape[0])
    else:
        base_len = 2100

    length_factors = [0.8, 0.9, 1.0, 1.2, 1.5, 2.0, 3.0]   # up to ~3 × base
    lengths = sorted({
        max(300, min(int(base_len * f), 8000)) for f in length_factors
    })

    # Keep the globally best few candidates (seq, score)
    global_top = []          # list of (seq, score), sorted ascending
    top_k = 5                # how many candidates to keep for later refinement

    # ------------------------------------------------------------------
    # Phase 1 – broad exploration
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        candidates = []

        # 1) classic windows (already useful)
        hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
        candidates.append(_normalize(hann, target_sum))

        black = (0.42
                 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
                 + 0.08 * np.cos(4.0 * np.pi * np.arange(n) / (n - 1)))
        candidates.append(_normalize(black, target_sum))

        # 2) Gaussian family
        for frac in [0.12, 0.18, 0.24, 0.30, 0.40]:
            sigma = frac * n
            candidates.append(gen_gaussian(n, sigma))

        # 3) Symmetric Beta densities
        for a in [0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0]:
            candidates.append(gen_beta(n, a))

        # 4) Symmetric exponential decays
        for rate in [0.001, 0.0025, 0.005, 0.008, 0.012]:
            candidates.append(gen_exponential(n, rate))

        # 5) Random Dirichlet draws (different concentrations)
        for conc in [0.1, 0.3, 0.6, 1.0, 2.0]:
            g = rng.gamma(shape=conc, scale=1.0, size=n)
            candidates.append(_normalize(g, target_sum))

        # 6) Sparse random spikes (few large values)
        for k in [5, 10, 20, 30]:
            idx = rng.choice(n, size=k, replace=False)
            vals = rng.random(k)
            vals = vals / vals.sum() * target_sum
            sparse = np.zeros(n)
            sparse[idx] = vals
            candidates.append(sparse)

        # 7) Interpolation of a previous best (if available)
        if base is not None and base.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
            candidates.append(_normalize(interp, target_sum))

        # ------------------------------------------------------------------
        # evaluate all candidates for this length
        for cand in candidates:
            if time.time() >= deadline:
                break
            score = _compute_score(cand)
            # keep top_k globally
            if len(global_top) < top_k or score < global_top[-1][1]:
                global_top.append((cand.copy(), score))
                global_top.sort(key=lambda x: x[1])
                if len(global_top) > top_k:
                    global_top.pop()
                print(f"[explore] n={n:5d} score={score:.6f}  (global best={global_top[0][1]:.6f})")

    # ------------------------------------------------------------------
    # Phase 2 – intensive PGD refinement on the best few candidates
    if global_top:
        # start from the best global candidate
        best_seq, best_score = global_top[0]
        n_best = best_seq.shape[0]
        # allocate most of the remaining budget to a long PGD run
        remaining = deadline - time.time()
        # leave around 30 s for the final stochastic polishing
        pgd_deadline = time.time() + max(5.0, remaining - 30.0)
        # a generous iteration budget (the loop stops earlier if time runs out)
        max_iters = 10_000_000
        print(f"[refine] starting long PGD on length {n_best} (budget ≈ {int(pgd_deadline - time.time())} s)")
        best_seq, best_score = _pgd_optimize(
            a0=best_seq,
            n=n_best,
            deadline=pgd_deadline,
            max_iters=max_iters,
            best_seq=best_seq,
            best_score=best_score,
            rng=rng,
        )
        # The PGD routine returns the best exact max‑convolution; convert to full score
        best_score = _compute_score(best_seq)
        print(f"[refine] after PGD best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap stochastic polishing
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        refine_deadline = time.time() + min(remaining, 30.0)
        print(f"[polish] final stochastic hill‑climb, time budget {int(refine_deadline - time.time())} s")
        best_seq, best_score = _local_refine(
            a=best_seq,
            n=best_seq.shape[0],
            deadline=refine_deadline,
            best_score=best_score,
            rng=rng,
        )
        best_score = _compute_score(best_seq)
        print(f"[polish] final score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen
    if best_seq is None:
        n = max(300, min(base_len, 2000))
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n)
        best_score = _compute_score(best_seq)
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
