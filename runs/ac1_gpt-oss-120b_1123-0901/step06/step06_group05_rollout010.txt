    corr_fft_len = _next_pow2(3*n-2)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lam = lam_start
    best_a = a0.copy()
    best_val = float('inf')
    last_improve = time.time()
    for t in range(1, max_iters+1):
        # compute softmax mapping
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()
        # compute conv and objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # smooth max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()

        # gradient
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # backprop through softmax
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)

            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
        if time.time() > deadline:
            break

"""
Propose a step‑function (list of non‑negative heights) that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)²) .

The algorithm works in three stages for each chosen length n:

1.   Generate a modest, diverse pool of seed shapes (arcsine‑type,
     power‑law, Gaussians, exponentials, Hann, sine, a few random Dirichlet
     draws and an up‑sampled best‑so‑far).

2.   **Quick‑run** – each seed is refined with a short Adam optimisation
     (smooth‑max surrogate).  The best *k* seeds are kept.

3.   **Deep‑run** – the selected seeds undergo a long Adam phase,
     an exponentiated‑gradient sub‑gradient phase on the true max,
     and a cheap stochastic hill‑climb.  The global best across all
     lengths is kept.

The code respects the supplied time budget (default 1000 s) and
returns a plain Python list.  All heavy computation (convolutions) is
performed with FFTs for speed.  The design deliberately reduces the
number of expensive long runs by pre‑filtering candidates, which
typically yields a lower final bound than the previous monolithic
approach.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   (n = len(a)).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative vector to ``new_n`` points by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:                     # protect against a zero vector
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s

def _generate_initial_candidates(n: int, prev_best):
    """
    Produce a modest but diverse pool of seed shapes for length ``n``.
    The pool contains:
        * up‑sampled previous best,
        * a few random Dirichlet draws,
        * arcsine‑type power‑law shapes (α in [-1.8, -0.2]),
        * symmetric exponentials,
        * Hann and half‑sine windows,
        * a few Gaussian bumps.
    All candidates are normalised to sum to one.
    """
    cand = []

    if prev_best is not None:
        cand.append(_upsample(prev_best, n))

    # random Dirichlet draws (different concentration parameters)
    cand.append(np.random.dirichlet(np.ones(n)))
    for conc in (0.2, 0.5, 0.8):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)                     # = (idx)*(n-idx+1) for 1‑based idx

    # power‑law / arcsine family
    for alpha in np.linspace(-1.8, -0.2, 9):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # symmetric two‑sided exponentials
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        cand.append(shape)

    # Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    hann /= hann.sum()
    cand.append(hann)

    # half‑sine
    sine = np.sin(np.pi * i / (n - 1))
    sine = np.maximum(sine, 0)
    sine /= sine.sum()
    cand.append(sine)

    # centred Gaussians (several widths)
    mu = (n - 1) / 2.0
    for sigma in (n / 10, n / 6, n / 4):
        shape = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # normalise and drop pathological entries
    clean = []
    for c in cand:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 1e-12:
            clean.append(c / s)
    return clean

def _run_adam(a0: np.ndarray, n: int, deadline: float,
              max_iters: int, lr: float = 0.05,
              lam_start: float = 0.30, lam_factor: float = 1.02):
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best sequence seen (as a probability vector) and its exact value.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = float('inf')
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # exact objective (sum(a)=1 ⇒ denominator drops out)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={t}] new best {val:.6f}")

        # smooth‑max weights (soft‑max on the convolution vector)
        w = np.exp(lam * (conv - max_c))          # shift by max for stability
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # back‑prop through the soft‑max (logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e4)

        # occasional random shake if we are stuck
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val

def _run_subgrad(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Exponentiated‑gradient sub‑gradient phase on the true max‑convolution.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    last_improve = time.time()
    eta0 = 0.55                      # initial step size

    for it in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it}] new best {val:.6f}")

        # active set where the convolution actually attains the maximum
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (classical EG schedule)
        eta = eta0 / np.sqrt(it)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                        # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val

def _local_hill_climb(a0: np.ndarray, n: int, deadline: float,
                     max_trials: int = 200_000):
    """
    Very cheap stochastic local search.
    Repeatedly moves a small mass ε from one index to another and keeps
    the move if it reduces the exact objective.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    best_a = a0.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    no_improve = 0
    trials = 0
    while time.time() < deadline and trials < max_trials and no_improve < 5_000:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0.0, min(0.05, best_a[i]))
        if eps == 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = cand
            no_improve = 0
            print(f"[HillClimb n={n}] improved to {val:.6f}")
        else:
            no_improve += 1
        trials += 1

    return best_a, best_val

    """
    Search for a non‑negative step‑function (list of heights) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

    The search consists of:
        * a short, cheap Adam run for many diverse seeds,
        * a deeper optimisation for the most promising seeds,
        * a final stochastic hill‑climb.
    The routine respects the provided time budget and returns the best
    sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None          # stored as a numpy array summing to 1
    best_val = float('inf')

    # Lengths to explore – start from a modest size, then use larger ones
    # if time permits.
    lengths = [4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # --------------------- generate seeds ---------------------
        seeds = _generate_initial_candidates(n, best_seq)

        # ------------------- Phase 1: quick filter ----------------
        quick_iters = max(2000, int(5000 * 2048 / n))
        top_k = 5
        scored = []   # (value, sequence)

        for idx, seed_vec in enumerate(seeds):
            if time.time() > deadline:
                break
            a_q, _ = _run_adam(seed_vec, n, deadline, quick_iters,
                               lr=0.10, lam_start=0.5, lam_factor=1.01)
            # exact objective (sum(a)=1)
            conv = _full_convolution(a_q, _next_pow2(2 * n - 1))
            val = 2.0 * n * conv.max()
            scored.append((val, a_q))
            if val < best_val:
                best_val = val
                best_seq = a_q.copy()
                print(f"[Phase 1 seed {idx}] new global best {val:.6f}")

        # retain the best few for deep optimisation
        scored.sort(key=lambda x: x[0])
        elite = [seq for _, seq in scored[:top_k]]

        # ------------------- Phase 2: deep optimisation ---------------
        for elite_idx, start_seq in enumerate(elite):
            if time.time() > deadline:
                break

            # long Adam run
            long_iters = max(20000, int(40000 * 2048 / n))
            a_adam, val_adam = _run_adam(start_seq, n, deadline,
                                         long_iters, lr=0.05,
                                         lam_start=0.30, lam_factor=1.02)
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[Phase 2 elite {elite_idx}] Adam improved best to {val_adam:.6f}")

            # sub‑gradient refinement
            sub_iters = max(10000, int(15000 * 2048 / n))
            a_sub, val_sub = _run_subgrad(a_adam, n, deadline, sub_iters)
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Phase 2 elite {elite_idx}] Subgrad improved best to {val_sub:.6f}")

            # short fine‑tune on the current global best of this length
            if best_seq is not None and len(best_seq) == n:
                a_fine, val_fine = _run_adam(best_seq, n, deadline,
                                             int(0.5 * long_iters),
                                             lr=0.03, lam_start=0.25, lam_factor=1.015)
                if val_fine < best_val:
                    best_val = val_fine
                    best_seq = a_fine.copy()
                    print(f"[Phase 2 elite {elite_idx}] Fine‑tune improved best to {val_fine:.6f}")

    # ------------------------------------------------------------------
    # Final cheap hill‑climb on the overall best candidate
    if best_seq is not None and time.time() < deadline:
        print(f"\n=== Final hill‑climb (n={len(best_seq)}) ===")
        a_final, val_final = _local_hill_climb(best_seq, len(best_seq), deadline,
                                               max_trials=250_000)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final
            print(f"[final] hill‑climb improved global best to {val_final:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = 2.0 * n_fallback * (_full_convolution(best_seq,
                                 _next_pow2(2 * n_fallback - 1)).max())
        print("Fallback uniform sequence used.")

    print(f"\nBest value obtained: {best_val:.6f}")
    return best_seq.tolist()
