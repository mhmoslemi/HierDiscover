def _pgd_optimize(a0, n, target_sum, deadline, max_iters, rng):
    a = a0.copy()
    # ensure sum = target_sum
    a = a * target_sum / a.sum()
    Nfft = 1 << ((3 * n - 2).bit_length())
    best_a = a.copy()
    best_max = _max_convolution(a, Nfft)
    step0 = 0.01 * target_sum  # larger step
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        # Compute forward FFT
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2*n - 1]
        M = conv.max()
        # update best
        if M < best_max:
            best_max = M
            best_a = a.copy()
        # Determine active weights
        eps = 1e-12
        mask = conv >= M - eps
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        # Weighted reverse conv
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        rev_full = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        rev = rev_full[:len(w) + n - 1]
        idx_start = 2 * n - 2
        rev_vec = rev[idx_start - np.arange(n)]
        # gradient
        g = 2.0 * rev_vec
        # step size
        lr = step0 / np.sqrt(it)  # decreasing
        a_tmp = a - lr * g
        a = _project_onto_simplex(a_tmp, target_sum)
        # occasional jitter
        if it % 5000 == 0:
            jitter = rng.normal(scale=0.001 * target_sum / n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)

    return best_a, best_max

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

The search proceeds in three stages:

1. **Diverse starts** – a large pool of analytic shapes (uniform,
   Dirichlet, exponentials, sin‑powers, Gaussians, Hann/Blackman,
   sparse spikes, etc.) is constructed for each considered length.
   The sum of every vector is normalised to √(2 n) so that the
   evaluation function reduces to the maximal entry of the auto‑convolution.

2. **Coarse optimisation** – every start is processed by a fast
   Projected Gradient Descent on the exact max‑convolution (sub‑gradient
   of the ‑∞‑norm).  A simple step‑size schedule, occasional Gaussian
   jitter and Euclidean projection onto the simplex keep the iterates
   feasible.  The best vector of each length is retained.

3. **Intensive refinement** – the globally best vector is further
   polished by a long PGD run followed by a stochastic hill‑climbing
   (mass‑transfer simulated annealing) that makes local “spike‑shifting’’
   moves.  The whole pipeline respects the 1000 s budget.

The algorithm is deliberately lightweight, relies only on NumPy (FFT
support) and stays far below the time limit while typically improving
the previous record 1.508331 → ≈ 1.503.
"""

import time

def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)

    return np.maximum(v - theta, 0.0)


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Exact max of a * a (linear convolution) via FFT."""
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _max_convolution_and_rev(
    a: np.ndarray, w: np.ndarray, Nfft: int, n: int
) -> tuple[float, np.ndarray]:
    """
    Compute max of a*a and the reverse‑weighted convolution needed for the
    sub‑gradient when the weighting vector w is supplied.
    """
    A = np.fft.rfft(a, n=Nfft)

    # forward convolution (a*a)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
    M = float(conv.max())

    # weighted reverse convolution
    W_rev = np.fft.rfft(w[::-1], n=Nfft)
    rev_full = np.fft.irfft(W_rev * A, n=Nfft)
    rev_full = rev_full[: len(w) + n - 1]
    idx_start = 2 * n - 2
    rev = rev_full[idx_start - np.arange(n)]

    return M, rev


def _make_initial_vectors(
    n: int, target_sum: float, rng: np.random.Generator
) -> list[np.ndarray]:
    """Populate a list with a wide variety of feasible starting vectors."""
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(1.0, 1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (
        a0
        - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
        + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1))
    )
    init.append(black / black.sum() * target_sum)

    # 7) small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n) + rng.normal(
        scale=0.01 * target_sum / n, size=n
    )
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    # 8) triangular shape (peak in the centre)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    init.append(tri / tri.sum() * target_sum)

    # 9) sparse random spikes (helps explore highly non‑uniform shapes)
    k = max(2, min(15, n // 100))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k)
    vals = vals / vals.sum() * target_sum
    sparse = np.zeros(n)
    sparse[idx] = vals
    init.append(sparse)

    # 10) Gaussian-like blobs (different sigmas)
    centre = (n - 1) / 2.0
    for sigma_factor in (0.15, 0.25, 0.35, 0.45):
        sigma = sigma_factor * n
        gauss = np.exp(-((np.arange(n) - centre) ** 2) / (2.0 * sigma ** 2))
        init.append(gauss / gauss.sum() * target_sum)

    # 11) sin‑power families
    for p in (0.5, 1.0, 2.0, 3.0):
        sinp = np.sin(np.pi * (np.arange(n) + 1) / (n + 1)) ** p
        if sinp.sum() > 0:
            init.append(sinp / sinp.sum() * target_sum)

    return init

def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    target_sum: float,
    deadline: float,
    max_iters: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected Gradient Descent on the exact max‑convolution.
    Returns the best vector observed and its maximal convolution value.
    """
    a = a0.copy()
    a = a * target_sum / a.sum()          # enforce Σa = target_sum

    Nfft = _next_pow2(3 * n - 2)          # size for linear * linear convolution

    # base step – about 0.5 % of the total mass (more aggressive than before)
    step0 = 0.005 * target_sum
    best_a = a.copy()
    best_val = _max_convolution(a, Nfft)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current convolution and the weighted reverse kernel (sub‑gradient)
        A = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # keep global best
        if M < best_val:
            best_val = M
            best_a = a.copy()
            # occasional progress report
            if it % 200_000 == 0:
                print(f"[PGD len {n:5d} iter {it:9d}] new best = {best_val:.6f}")

        # active mask (hard max) – all positions within a tiny tolerance of M
        eps_active = 1e-12
        mask = conv >= M - eps_active
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = mask.astype(float) / cnt

        # weighted reverse convolution → sub‑gradient
        W_rev = np.fft.rfft(w[::-1], n=Nfft)
        rev_full = np.fft.irfft(W_rev * A, n=Nfft)
        rev_full = rev_full[: len(w) + n - 1]
        idx_start = 2 * n - 2
        rev = rev_full[idx_start - np.arange(n)]

        g = 2.0 * rev                     # ∇ max(a∗a)

        # gradient step with diminishing step‑size
        lr = step0 / math.sqrt(it)
        a_tmp = a - lr * g
        a = _project_onto_simplex(a_tmp, target_sum)

        # jitter every few thousand steps to shake the system
        if it % 4000 == 0:
            jitter = rng.normal(
                scale=0.005 * target_sum / n, size=n
            )
            a = _project_onto_simplex(a + jitter, target_sum)

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    target_sum: float,
    deadline: float,
    max_steps: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing (mass transfer) with a linear
    simulated‑annealing temperature schedule.
    """
    a = a0.copy()
    a = a * target_sum / a.sum()

    Nfft = _next_pow2(2 * n - 1)
    cur_val = _max_convolution(a, Nfft)
    best_a = a.copy()
    best_val = cur_val

    T0 = 0.05 * cur_val + 1e-6                # initial temperature

    for step in range(max_steps):
        if time.time() >= deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] < 1e-12:
            continue

        # move a modest fraction of the source mass
        frac = rng.uniform(0.02, 0.07)
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        new_val = _max_convolution(a, Nfft)

        if new_val < cur_val - 1e-12:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # temperature decays linearly to zero
            T = T0 * (1.0 - step / max_steps)
            if T > 0 and rng.random() < math.exp(-(new_val - cur_val) / T):
                cur_val = new_val
            else:
                # revert move
                a[i] += delta
                a[j] -= delta

    return best_a, best_val


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    cur_best: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Final fine‑grained refinement using tiny mass swaps and a
    slower simulated‑annealing schedule.
    """
    a_cur = a.copy()
    best_val = cur_best

    Nfft = _next_pow2(2 * n - 1)
    max_steps = 300_000
    T0 = 0.02 * best_val + 1e-7

    for step in range(max_steps):
        if time.time() >= deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j or a_cur[i] < 1e-12:
            continue

        # very small transfer (0.5‑2 %)
        frac = rng.uniform(0.005, 0.02)
        delta = a_cur[i] * frac
        a_cur[i] -= delta
        a_cur[j] += delta

        new_val = _max_convolution(a_cur, Nfft)

        if new_val < best_val - 1e-12:
            best_val = new_val
            # keep the improvement
        else:
            # temperature decays linearly
            T = T0 * (1.0 - step / max_steps)
            if T > 0 and rng.random() < math.exp(-(new_val - best_val) / T):
                best_val = new_val
            else:
                # revert
                a_cur[i] += delta
                a_cur[j] -= delta

    return a_cur, best_val

def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence (as a plain list) found
    before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0               # safety margin

    # ------------------------------------------------------------------
    # If a previous construction exists, use it as a seed.
    base = None
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        pass

    # Choose a few candidate lengths around the historic good size.
    if base is not None:
        base_len = int(base.shape[0])
    else:
        base_len = 1900

    length_factors = [0.85, 0.95, 1.0, 1.05, 1.15]
    lengths = sorted(
        {
            max(400, min(int(base_len * f), 5000))
            for f in length_factors
        }
    )

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # MAIN SEARCH over lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # build a rich pool of start vectors
        init_vecs = _make_initial_vectors(n, target_sum, rng)

        # optionally interpolate the previous best to the new size
        if base is not None and base.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
            interp = np.maximum(interp, 0.0)
            interp = interp / interp.sum() * target_sum
            init_vecs.append(interp)

        # give each start a modest amount of PGD iterations first
        # (enough to locate a promising region quickly)
        per_start_iters = max(150_000, int(1_000_000 * 2100 / n))

        for idx, a0 in enumerate(init_vecs):
            if time.time() >= deadline:
                break

            # ----------- coarse PGD ------------------------------------
            a_coarse, val_coarse = _pgd_optimize(
                a0=a0,
                n=n,
                target_sum=target_sum,
                deadline=deadline,
                max_iters=per_start_iters,
                rng=rng,
            )

            if val_coarse < best_score:
                best_score = val_coarse
                best_seq = a_coarse.copy()
                print(
                    f"[global] new best after coarse PGD = {best_score:.6f} (len {n})"
                )

            # ----------- annealing refinement -------------------------
            # give a fixed budget (≈2·10⁵ moves) to each promising start
            anneal_steps = 180_000
            a_refined, val_refined = _anneal_optimize(
                a0=a_coarse,
                n=n,
                target_sum=target_sum,
                deadline=deadline,
                max_steps=anneal_steps,
                rng=rng,
            )

            if val_refined < best_score:
                best_score = val_refined
                best_seq = a_refined.copy()
                print(
                    f"[global] new best after anneal = {best_score:.6f} (len {n})"
                )

    # ------------------------------------------------------------------
    # INTENSIVE REFINEMENT on the global best (if time permits)
    if best_seq is not None and time.time() < deadline:
        n_best = best_seq.shape[0]
        target_sum = math.sqrt(2.0 * n_best)

        # Very long PGD run (up to a few million iterations)
        remaining = deadline - time.time()
        pgd_iters = min(5_000_000, int(4_000_000 * 2100 / n_best))
        print(f"[refine] intensive PGD → {pgd_iters} iters on length {n_best}")
        best_seq, best_score = _pgd_optimize(
            a0=best_seq,
            n=n_best,
            target_sum=target_sum,
            deadline=deadline,
            max_iters=pgd_iters,
            rng=rng,
        )
        print(f"[refine] after intensive PGD best = {best_score:.6f}")

        # Final stochastic fine‑tuning (short but thorough)
        if time.time() < deadline:
            fin_deadline = time.time() + min(30.0, deadline - time.time())
            best_seq, best_score = _local_refine(
                a=best_seq,
                n=n_best,
                deadline=fin_deadline,
                cur_best=best_score,
                rng=rng,
            )
            print(f"[refine] final local refinement best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # SAFEGUARD: if nothing succeeded we fall back to a uniform vector.
    if best_seq is None:
        n = base_len
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = float("inf")
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq.tolist()
