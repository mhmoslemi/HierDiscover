"""
Propose a step‐function (non‑negative vector) with a low evaluation score.

Algorithm
---------
1.  **Multi‑scale hierarchy** – start on a tiny problem (n≈1024) where many
    candidates can be evaluated quickly, then up‑sample the best candidate to
    larger lengths (4096 → 8192) and continue refining.
2.  **Random parametric families** – a cheap global search over a 2‑parameter
    “Beta‑shape’’ a_i ∝ (i+1)^α·(n‑i)^β (α,β ∈ [‑4, 4]) and a few Dirichlet
    samples.  The best shape is kept as a warm‑start.
3.  **Smooth‑max Adam** – minimise a smoothed version of the objective
    `max(conv)` using a log‑sum‑exp surrogate.  The temperature (λ) grows
    slowly (factor 1.01) so that the optimisation stays stable.
4.  **Exact‑max sub‑gradient (exponentiated‑gradient)** – after Adam we switch
    to the true non‑smooth objective.  The learning‑rate η decays as
    η_t = η0 / sqrt(t+1) to guarantee convergence.
5.  **Targeted local hill‑climbing** – repeatedly move a random fraction of
    mass from one index to another; a move is kept if it reduces the exact
    objective.
6.  **Final up‑sampling** – if time permits the current best vector is linearly
    interpolated to a finer grid (8192) and the whole pipeline (Adam → sub‑gradient
    → local search) is repeated.

All heavy work uses FFT‑based convolution (`_full_convolution`) which is
O(n log n).  The routine respects the supplied time budget and always returns
a valid sequence (non‑negative, finite) before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # keep only the linear part
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a).  Returns a vector of length len(a):
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0 … n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_val(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv(a,a)).  Assumes Σa = 1."""
    conv = _full_convolution(a, fft_len)
    return 2.0 * n * conv.max()


def _smooth_max_adam(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed‑max surrogate.
    Returns the best vector (Σa=1) encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)            # initialise logits = log(a)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04

    # temperature λ – start low and grow slowly
    lam = 0.5
    lam_factor = 1.01

    best_a = a0.copy()
    best_val = _objective_val(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:05d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp trick)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # gradient of surrogate (chain rule)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # convert to gradient w.r.t. logits (softmax Jacobian)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e5)

        # occasional shake‑up if improvement stalls
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _exp_gradient_subgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the exact max objective.
    Anneals the step‑size η_t = η0 / sqrt(t+1).
    """
    a = np.maximum(a0, 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        a_sum = a.sum()
    a /= a_sum

    best_a = a.copy()
    best_val = _objective_val(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:05d}] new best {val:.6f}")

        # active set of indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)      # multiplicative update
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            a_sum = a.sum()
        a /= a_sum

        it += 1

    return best_a, best_val


def _local_random_hill_climb(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one index
    to another and keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    for it in range(max_moves):
        if time.time() >= deadline:
            break
        i = np.random.randint(n)
        if a[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # move up to 40 % of the source mass
        delta = a[i] * (np.random.rand() * 0.4)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 3000 == 0:
                print(f"[Local move {it}] improved to {val2:.6f}")

    return best_a, best_val


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Generate a symmetric ‘Beta‑like’ shape:
        a_i ∝ (i+1)^α · (n‑i)^β .
    The result is normalised to sum = 1.
    """
    idx = np.arange(n, dtype=np.float64) + 1.0
    rev = n - np.arange(n, dtype=np.float64)
    # avoid overflow/underflow for extreme exponents
    shape = np.where((idx > 0) & (rev > 0),
                     np.exp(alpha * np.log(idx) + beta * np.log(rev)),
                     0.0)
    if shape.sum() == 0.0:
        return None
    shape /= shape.sum()
    return shape


def _parameter_search_small(
    n: int,
    time_limit: float,
    max_samples: int = 500,
) -> np.ndarray:
    """
    Random search over simple parametric families (Beta‑shape, Dirichlet)
    on a modest size n.  Returns the best candidate found (normalised).
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    best_a = np.full(n, 1.0 / n, dtype=np.float64)
    best_val = _objective_val(best_a, n, conv_fft_len)

    start = time.time()
    samples = 0
    while time.time() < time_limit and samples < max_samples:
        # ---- Beta shape -------------------------------------------------
        if np.random.rand() < 0.7:
            alpha = np.random.uniform(-4.0, 4.0)
            beta = np.random.uniform(-4.0, 4.0)
            a = _beta_shape(n, alpha, beta)
            if a is None:
                continue
        else:
            # ---- Dirichlet ------------------------------------------------
            concentration = np.random.uniform(0.1, 5.0)
            a = np.random.dirichlet(np.full(n, concentration))

        # make sure we respect the evaluator's clipping range
        a = np.clip(a, 0.0, 1000.0)
        a /= a.sum()

        val = _objective_val(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[ParamSearch] new best {val:.6f} (α={alpha:.2f},β={beta:.2f})")
        samples += 1

    print(f"[ParamSearch] finished with best {best_val:.6f} after {samples} samples")
    return best_a


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

        best_a, n_mid, conv_fft_mid, corr_fft_mid, sub_deadline,
        max_iters=200_000, eta0=0.8,
    )

        best_a = a_sub.copy()
        print(f"[Stage2] Sub‑grad improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 3. Cheap local random search to polish
    if time.time() < deadline - 5.0:
        a_loc, val_loc = _local_random_hill_climb(
            best_a, n_mid, conv_fft_mid, deadline, max_moves=40_000
        )
        if val_loc < best_val:
            best_val = val_loc
            best_a = a_loc.copy()
            print(f"[Stage2] Local search improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # 4. Upscale to a finer grid (n=8192) if we still have time
    if time.time() + 20.0 < deadline:
        n_big = 8192
        conv_fft_big = _next_pow2(2 * n_big - 1)
        corr_fft_big = _next_pow2(3 * n_big - 2)

        x_mid = np.linspace(0.0, 1.0, n_mid, endpoint=False)
        x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_big, x_mid, best_a)
        a_big = np.clip(a_big, 0.0, 1000.0)
        a_big /= a_big.sum()

        # Adam on the bigger grid (shorter budget)
        big_deadline = deadline - 5.0
        a_big_adam, val_big_adam = _smooth_max_adam(
            a_big, n_big, conv_fft_big, corr_fft_big, big_deadline,
            max_iters=120_000,
        )
        # Sub‑gradient refinement
        a_big_sub, val_big_sub = _exp_gradient_subgrad(
            a_big_adam, n_big, conv_fft_big, corr_fft_big,
            deadline, max_iters=100_000, eta0=0.6,
        )
        if val_big_sub < best_val:
            best_val = val_big_sub
            best_a = a_big_sub.copy()
            print(f"[Stage3] Upscaled refinement best {val_big_sub:.6f}")

    # ------------------------------------------------------------------
    # 5. Final safe‑guard: ensure a valid output
    if best_a is None or best_a.size == 0:
        best_a = np.full(4096, 1.0 / 4096, dtype=np.float64)

    best_a = np.clip(best_a, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
