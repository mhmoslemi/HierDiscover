```
def subgrad_opt(a0, n, fft_len, deadline):
    a = a0.copy()
    best = a.copy(); best_val = inf
    t=0
    while time < deadline:
        conv = convolution(a)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val: update best.
        # compute gradient using active set like before
        active = np.isclose(conv, max_c, atol=1e-12)
        # if no active, break
        w = np.zeros_like(conv)
        w[active] = 1.0/active.sum()
        grad = 2 * cross_correlation(w, a)
        # step size schedule
        eta = 0.5 / np.sqrt(t+1)  # decreasing
        a = a * np.exp(-eta * grad)
        a = a / a.sum()
        t+=1
```

def _subgradient_opt2(a0, n, fft_len, corr_fft_len, deadline, max_iters=500000):
    a = a0.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s
    best_a = a.copy()
    best_val = float('inf')
    best_time = time.time()
    stall_counter = 0
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall_counter = 0
        else:
            stall_counter += 1

        # compute gradient for active max
        active = np.isclose(conv, max_c, atol=1e-12)
        # if too many active, sample subset maybe
        w = np.zeros_like(conv)
        act_sum = active.sum()
        if act_sum > 0:
            w[active] = 1.0 / act_sum

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size schedule
        eta = 0.6 / np.sqrt(it + 1)  # decaying

        a = a * np.exp(-eta * grad)
        # renormalize
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # occasional random perturbation if stalling
        if stall_counter > 5000:
            # add small random noise
            a = np.maximum(a + 0.01 * (np.random.rand(n) - 0.5), 0)
            a = a / a.sum()
            stall_counter = 0
    return best_a, best_val

def _gradient_guided_search(a0, n, fft_len, deadline, max_moves=None):
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    # schedule step fraction decaying
    while time.time() < deadline:
        if max_moves is not None and it >= max_moves:
            break
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        act_sum = active.sum()
        if act_sum>0:
            w[active] = 1.0/act_sum

        grad = 2.0 * _cross_correlation(w, a, _next_pow2(3*n-2))
        # source indices where grad larger than avg
        avg = np.dot(a, grad)
        # typical update direction: move away from large grad values
        # choose source i with probability proportional to (grad_i - avg)^+
        scores_src = grad - avg
        scores_src = np.clip(scores_src, 0, None)
        sum_src = scores_src.sum()
        if sum_src == 0:
            # fallback random
            i = np.random.randint(n)
        else:
            probs_src = scores_src / sum_src
            i = np.random.choice(n, p=probs_src)

        # dest j where grad_j < avg
        scores_dest = avg - grad
        scores_dest = np.clip(scores_dest, 0, None)
        sum_dest = scores_dest.sum()
        if sum_dest == 0:
            j = np.random.randint(n)
        else:
            probs_dest = scores_dest / sum_dest
            j = np.random.choice(n, p=probs_dest)

        if i == j:
            continue
        # move fraction delta of mass from i to j
        # step size decays with iteration
        step = 0.05 / (np.sqrt(it+1))
        delta = min(a[i] * step, a[i])
        if delta < 1e-12:
            it += 1
            continue
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2  # keep improvement
        # else maybe accept with small probability (simulated annealing)
        else:
            # temperature schedule
            T = max(0.01, (deadline - time.time()) / (deadline - start_time))
            # accept if val2 is not too much worse
            if np.random.rand() < np.exp(-(val2 - best_val) / (T+1e-9)):
                a = a2

        it += 1
    return best_a, best_val

"""
Propose a step‑function (non‑negative vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm is a hybrid of several fast optimisation phases:

1.  A diverse pool of warm‑started candidates (uniform, Dirichlet,
    edge‑concentrated power‑law, double‑exponential, sinusoidal, spike
    patterns).  The pool is also seeded with the previous best solution
    if it exists.

2.  For every candidate we run a short Adam optimisation on the smooth
    “log‑sum‑exp’’ surrogate of the max‑convolution.  The parameters are
    kept on the simplex via a soft‑max.

3.  The best candidate is then refined with a long exponentiated‑gradient
    (multiplicative) sub‑gradient method.  The step size decays (∝1/√t)
    and the routine injects a tiny random shake when progress stalls.

4.  A cheap hill‑climbing random‑move phase (up to a few‑hundred‑thousand
    moves) further decreases the objective.

5.  If time permits the solution is interpolated to a finer grid
    (8192 → 16384 points) and the Adam + sub‑gradient refinements are
    repeated on the larger problem, exploiting the extra degrees of
    freedom to lower the maximum convolution.

All phases respect the global time budget (default 1000 s) and the
routine always returns the best sequence found.  The implementation
uses only NumPy and works on a single CPU core (the limit is enforced
by the caller)."""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Generate a modest set of diverse start vectors."""
    cand = []

    # warm start + a few rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-3.0, -0.2, 13):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # symmetric double‑exponential shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in np.geomspace(0.2, 5.0, 8):
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        cand.append(shape)

    # positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """A few “spike’’ candidates with k equally spaced non‑zero entries."""
    out = []
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    return out


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector (sum≈1) encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.08                     # a slightly larger learning rate
    lam = 0.5                     # start temperature
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (sum(a)=1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional tiny random shake if we seem to stall
        if it % 7_000 == 0 and time.time() + 5 > deadline:
            logits += (np.random.rand(n) - 0.5) * 5e-3

        it += 1

    return best_a, best_val


def _subgradient_opt2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient phase with a decaying step size.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall = 0
            if it % 5_000 == 0:
                print(f"[Subgrad iter={it:07d}] new best {val:.6f}")
        else:
            stall += 1

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update and renormalisation
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # random perturbation if we have been stuck for a while
        if stall > 10_000:
            a = np.maximum(a + 0.02 * (np.random.rand(n) - 0.5), 0.0)
            a /= a.sum()
            stall = 0

        it += 1
    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one index
    to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        # move up to 40 % of the source mass
        delta = a[i] * (np.random.rand() * 0.4)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it}] improved to {val2:.6f}")
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – base resolution (n = 4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # warm start from a previous solution, if present
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    candidates = _initial_candidates(n, warm) + _spike_candidates(n)

    # ------------------------------------------------------------------
    # Optimise each candidate briefly (Adam + short sub‑gradient)
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        s = a0.sum()
        if s < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / s

        # give each candidate a few seconds of Adam time
        cand_deadline = min(deadline - 5.0, time.time() + 12.0)
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, cand_deadline, max_iters=120_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()

    # ------------------------------------------------------------------

            best_seq, n, conv_fft_len, deadline, max_moves=150_000

    # ------------------------------------------------------------------
    # Phase 4 – upscale to 8192 if time permits
    if time.time() + 30 < deadline:
        n_big = 8192

        if best_seq is not None:

        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # short Adam on the larger grid

        a_big_adam, val_big_adam = _adam_optimize(

        )
        if val_big_adam < best_val:
