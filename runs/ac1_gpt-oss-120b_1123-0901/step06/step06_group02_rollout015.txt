def linear_program_step(f, target_sum, deadline):
    n = len(f)
    M = _full_convolution_fft(f, f).max()
    # Build A_ub matrix for f * g <= M
    # conv index range 0..2n-2
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        # indices i where 0 <= i < n and 0 <= k-i < n
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # g_j where j = k - i
        # For each i in [i_start, i_end], f_i contributes to g_{k - i}
        # So for each j in appropriate range, coefficient = f_{k - j}
        # But easier: iterate over i
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    # Build sparse matrix
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # because f_i >=0, A entries nonnegative
    # constraints: A @ g <= M
    b = np.full(2*n-1, M)
    # objective: maximize sum(g) -> min -sum(g)
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]

    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                 method='highs', options={'time_limit': max(5, deadline - time.time())})
    if res.success:
        g = res.x
        # res may give sum less than target_sum, but we can scale up to target_sum while staying within constraints:
        scaling = target_sum / g.sum()
        g_scaled = g * scaling
        # However after scaling, constraints may be violated; but if scaling close to 1 if sum near target or we can just combine after scaling with f.
        return g_scaled
    else:
        return None

"""
Hybrid multi‑phase optimizer for the non‑negative step‑function bound.

The objective after normalising Σ a_i = √(2 n) is simply
    max(conv(a,a)) .
We minimise this quantity by a combination of:

1. Diverse random‐parameter initialisations (exponential, power‑law,
   Hann, Hamming, Kaiser, Blackman, Tukey windows, optionally symmetrised).

2. Adam optimiser on a smoothed max (log‑sum‑exp) surrogate
   using the soft‑max parametrisation a = S·softmax(x).

3. Exact sub‑gradient descent that always follows the current worst
   convolution index (projected onto the simplex).

4. A short linear‑programming “active‑set” step: for a fixed f,
   maximise Σ g_i subject to the linear constraints (f * g)_k ≤ M.
   The obtained g is mixed with the current best and projected back.

5. Simulated‑annealing hill‑climb moving tiny masses between positions.

All phases respect a single global time budget (default 1000 s) and
share a global best solution.  The code works with NumPy and SciPy only
(and no closures or lambdas, as required by the execution sandbox).
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of real vectors x and y."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # Clip tiny negatives caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]          # sorted descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)     # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1        # zero‑based index of last positive entry
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the log‑it (soft‑max) surrogate
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 5e-6,
    alpha0: float = 0.03,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed‑max surrogate.

    Returns (best_sequence, best_max_conv) where best_sequence already
    satisfies the simplex constraint Σ a_i = target_sum.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    # main loop
    while time.time() < deadline:
        # map logits to the simplex
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)               # stable soft‑max
        p = exp_shift / np.sum(exp_shift)           # Σ p = 1
        a = target_sum * p

        # exact max of the convolution (used for logging & tau)
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam n={n} new best max={cur_max:.6f}")

        # smoothed max weights (log‑sum‑exp)
        tau = max(init_tau_factor * cur_max * (0.99 ** it),
                  min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a : 2·a⊛w_rev
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        # pick the n entries that correspond to derivative
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # back‑prop to logits (chain rule)
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slowly decay learning rate
        alpha *= 0.9995

    # final exact evaluation of the best vector
    if best_a is None:
        best_a = target_sum * np.exp(x0) / np.sum(np.exp(x0))
        best_val = _full_convolution_fft(best_a, best_a).max()
    else:
        best_val = _full_convolution_fft(best_a, best_a).max()
    return best_a, float(best_val)


# ----------------------------------------------------------------------
# Targeted projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _targeted_subgradient(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent that follows the current worst convolution index.
    Returns a (sequence, max_convolution) pair.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    # initial step size proportional to average mass
    step = 0.2 * target_sum / n

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())            # index of current worst peak

        # gradient of conv_k w.r.t. a
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # projected step
        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)

        cand_val = _full_convolution_fft(cand, cand).max()
        if cand_val < best_val - 1e-12:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= 1.02                       # modestly increase after success
            if it % 200 == 0:
                print(f"[{time.time():.2f}] subgrad it={it} val={best_val:.6f}")
        else:
            step *= 0.70                       # reduce step on failure
            # occasional jitter to avoid stagnation
            if np.random.rand() < 0.07:
                jitter = np.random.rand(n) * 1e-4
                a = _project_onto_simplex(a + jitter, target_sum)

        if best_val < 1.503:
            break
    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (simulated annealing) on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Move tiny amounts of mass between random positions with a Metropolis rule.
    Returns (best_sequence, best_max_conv).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # temperature schedule
    T0 = max(1e-3, best_val * 0.4)
    T_end = 1e-5
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        delta = a[i] * np.random.rand() * 0.08  # move ≤8 % of source mass
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()
        accept = False
        if new_val < best_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")

        # re‑normalise every few thousand steps to avoid drift
        if it % 2500 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (fixed f, maximise Σ g_i subject to f*g≤M)
# ----------------------------------------------------------------------
def _linear_program_step(
    f: np.ndarray,
    target_sum: float,
    deadline: float,
) -> np.ndarray | None:
    """
    Solve   max Σ g_i   s.t.   f * g ≤ M   and   g ≥ 0.
    Returns the optimal g (un‑scaled) or None on failure.
    """
    n = f.shape[0]
    M = _full_convolution_fft(f, f).max()

    # Build sparse matrix A where A[k, j] = f[i] with i + j = k
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=int)
        rows.extend([k] * len(i_vals))
        cols.extend((k - i_vals).tolist())
        data.extend(f[i_vals].tolist())

    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)

    c = -np.ones(n)                     # maximise Σ g_i  → minimise -Σ g_i
    bounds = [(0, None) for _ in range(n)]

    # time limit for the LP (a few seconds)
    lp_time = max(5.0, deadline - time.time())
    try:
        res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"time_limit": lp_time},
        )
    except Exception:
        return None

    if not res.success or res.x is None:
        return None

    g = np.maximum(res.x, 0.0)
    # make sure the returned g has the correct total mass (we will re‑scale later)
    return g


# ----------------------------------------------------------------------
# Parameterised random initialiser (various window shapes)
# ----------------------------------------------------------------------
def _random_initial_sequence(n: int) -> np.ndarray:
    """Generate a random non‑negative symmetric or asymmetric shape."""
    target_sum = np.sqrt(2.0 * n)

    shape = np.random.choice(
        ["exp", "pow", "hann", "hamming", "kaiser", "blackman", "tukey"]
    )
    if shape == "exp":
        tau = np.random.uniform(0.2, 3.0) * n
        a = np.exp(-np.arange(n, dtype=float) / tau)
    elif shape == "pow":
        p = np.random.uniform(0.1, 1.8)
        a = (np.arange(1, n + 1, dtype=float)) ** (-p)
    elif shape == "hann":
        a = np.hanning(n)
    elif shape == "hamming":
        a = np.hamming(n)
    elif shape == "kaiser":
        beta = np.random.uniform(0.5, 8.0)
        a = np.kaiser(n, beta)
    elif shape == "blackman":
        a = np.blackman(n)
    else:   # tukey
        alpha = np.random.uniform(0.1, 0.9)
        i = np.arange(n, dtype=float)
        a = np.ones(n)
        # first cosine taper
        mask1 = i < alpha * (n - 1) / 2
        a[mask1] = 0.5 * (1 + np.cos(np.pi * (2 * i[mask1] / (alpha * (n - 1)) - 1)))
        # second cosine taper
        mask2 = i > (n - 1) * (1 - alpha / 2)
        a[mask2] = 0.5 * (1 + np.cos(np.pi * (2 * i[mask2] / (alpha * (n - 1)) - 2 / alpha + 1)))

    # Optional symmetrisation
    if np.random.rand() < 0.7:
        a = (a + a[::-1]) / 2.0

    # Optional tiny Gaussian noise
    if np.random.rand() < 0.3:
        a = a + np.random.randn(n) * np.max(a) * 0.03
        a = np.clip(a, 0.0, None)

    # Normalise to the required L1 sum
    a = a * (target_sum / a.sum())
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid optimiser for the step‑function bound.

    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    best_seq = None               # numpy array
    best_score = np.inf           # best max‑convolution seen so far

    def _record(seq: np.ndarray) -> float:
        """Evaluate a sequence (exact max‑conv) and possibly update global best."""
        nonlocal best_seq, best_score
        cur = _full_convolution_fft(seq, seq).max()
        if cur < best_score:
            best_score = float(cur)
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] GLOBAL new best={best_score:.6f}  n={len(seq)}")
        return cur

    # ------------------------------------------------------------------
    # Warm‑start from any provided construction (height_sequence_1)
    # ------------------------------------------------------------------
    try:
        hs = globals().get("height_sequence_1")
        if hs is not None:
            arr = np.asarray(hs, dtype=float)
            n0 = arr.shape[0]
            target0 = np.sqrt(2.0 * n0)
            arr = arr * (target0 / arr.sum())
            _record(arr)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – diverse parametric random initialisations
    # ------------------------------------------------------------------
    while time.time() < deadline - 500.0:
        remaining = deadline - time.time()
        # adapt length to remaining time (larger early, smaller later)
        if remaining > 800:
            n = np.random.randint(2500, 8000 + 1)
        elif remaining > 600:
            n = np.random.randint(2000, 6000 + 1)
        elif remaining > 400:
            n = np.random.randint(1500, 4000 + 1)
        else:
            n = np.random.randint(1200, 3000 + 1)

        a0 = _random_initial_sequence(n)
        _record(a0)

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimiser on smoothed max‑convolution
    # ------------------------------------------------------------------
    while best_seq is not None and time.time() < deadline - 300.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        x0 = np.log(best_seq + 1e-12)                     # initialise logits
        run_deadline = time.time() + min(40.0, deadline - time.time() - 30.0)

        seq_adam, val_adam = _optimise_adam_softmax(
            n,
            x0,
            target_sum,
            run_deadline,
            init_tau_factor=0.15,
            min_tau_factor=1e-6,
            alpha0=0.03,
        )
        if val_adam < best_score:
            best_seq = seq_adam
            best_score = val_adam
            print(f"[{time.time():.2f}] After Adam refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Targeted projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    while best_seq is not None and time.time() < deadline - 180.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_deadline = time.time() + min(30.0, deadline - time.time() - 30.0)

        seq_sub, val_sub = _targeted_subgradient(
            best_seq.copy(),
            target_sum,
            sub_deadline,
        )
        if val_sub < best_score:
            best_seq = seq_sub
            best_score = val_sub
            print(f"[{time.time():.2f}] After sub‑gradient best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Linear‑programming active‑set refinement
    # ------------------------------------------------------------------
    while best_seq is not None and time.time() < deadline - 120.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)

        # solve the LP for the current best sequence
        g = _linear_program_step(best_seq, target_sum, deadline)
        if g is None:
            break

        # mix the LP solution with the current best (convex combination)
        t_mix = 0.5
        cand = (1.0 - t_mix) * best_seq + t_mix * g
        cand = _project_onto_simplex(cand, target_sum)
        cand_val = _full_convolution_fft(cand, cand).max()
        if cand_val < best_score:
            best_seq = cand
            best_score = cand_val
            print(f"[{time.time():.2f}] After LP combine best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Final simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    while best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)

        seq_hill, val_hill = _hillclimb(
            best_seq.copy(),
            target_sum,
            deadline,
        )
        if val_hill < best_score:
            best_seq = seq_hill
            best_score = val_hill
            print(f"[{time.time():.2f}] After hill‑climb best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = _full_convolution_fft(best_seq, best_seq).max()
        print(f"[{time.time():.2f}] Fallback uniform score={best_score:.6f}")

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
