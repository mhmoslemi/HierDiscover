"""
Propose a non‑negative step‑function (list of heights) with a low value of

        F(a) = 2·n·max(a*a) / (sum(a)²)

The search works in two stages:

1. **Quick sweep** – short Adam runs on a handful of diversified
   initial shapes for moderate lengths (1024, 2048, 4096, 8192).  The
   best sequence found is kept as a warm‑start.

2. **Deep refinement** – a long Adam + exact sub‑gradient (exponentiated‑
   gradient) optimisation on a larger length (up to 16384).  The
   optimisation stays on the probability simplex by working in the
   log‑softmax (“logits”) domain and uses FFT‑based convolutions for
   speed.  Iterations stop as soon as the time budget (default 1000 s)
   is exhausted, and the best sequence encountered is returned.

All heavy work (convolution and the required cross‑correlation) is
performed with real FFTs (O(n log n)).  The function prints occasional
progress reports and finally returns a plain Python list.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a using an FFT of length ``fft_len``.
    Returns only the first 2·len(a)‑1 entries.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Result length = len(a) and is given by

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, conv_len: int) -> float:
    """Exact objective (no clamping) for a vector that already sums to 1."""
    conv = _conv_fft(a, conv_len)
    return float(2.0 * n * conv.max())


def _initial_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine shape: a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    return arcs / arcs.sum()


def _initial_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    gauss = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return gauss / gauss.sum()


def _build_initial_candidates(n: int) -> list[np.ndarray]:
    """
    Produce a diversified list of starting points for the optimisation.
    The vectors are normalised to sum to 1.
    """
    cand = []

    # optionally warm‑start from a previous solution
    global height_sequence_1
    if "height_sequence_1" in globals():
        h = np.asarray(height_sequence_1, dtype=np.float64)
        if h.shape[0] > n:
            h = h[:n]
        else:
            h = np.pad(h, (0, n - h.shape[0]))
        h = np.maximum(h, 0.0)
        if h.sum() < 1e-12:
            h = np.random.dirichlet(np.ones(n))
        else:
            h = h / h.sum()
        cand.append(h)

    # 1) Dirichlet (uniform)
    cand.append(np.random.dirichlet(np.ones(n)))

    # 2) spiky Dirichlet
    for conc in (0.2, 0.4, 0.6):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 3) over‑dispersed Dirichlet
    for conc in (1.5, 2.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 4) arcsine
    cand.append(_initial_arcsine(n))

    # 5) Gaussian with a few sigmas
    for sigma in (n / 12.0, n / 8.0, n / 5.0):
        cand.append(_initial_gaussian(n, sigma))

    # 6) power‑law (edge‑concentrated)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 5):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # 7) simple tent
    left = np.linspace(0, 1, n // 2, endpoint=False)
    right = left[::-1]
    tent = np.concatenate([left, right])
    if tent.shape[0] < n:   # odd length
        tent = np.concatenate([tent, [0.0]])
    tent = np.maximum(tent, 0)
    cand.append(tent / tent.sum())

    return cand

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence (step‑function) that minimises
    the evaluation function defined in the statement.  The routine respects
    a time budget (default 1000 s) and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1. QUICK SWEEP over moderate lengths
    quick_lengths = [1024, 2048, 4096, 8192]

    for n in quick_lengths:
        if time.time() > deadline:
            break
        print(f"\n--- Quick sweep, length n={n} ---")
        conv_len_fft = _next_pow2(2 * n - 1)
        corr_len_fft = _next_pow2(2 * n - 1 + n - 1)

        # a handful of diversified starts
        init_cands = _build_initial_candidates(n)[:10]

        for cid, a0 in enumerate(init_cands):
            if time.time() > deadline:
                break
            a = a0 / a0.sum()               # simplex
            # very short Adam run – enough to escape obvious bad spots
            logits = np.log(a + 1e-15)
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.06 * np.sqrt(2048.0 / n)   # a bit more aggressive for quick sweep
            lam = 0.1
            lam_factor = 1.02
            max_iters = 3000

            for t in range(1, max_iters + 1):
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                conv = _conv_fft(a, conv_len_fft)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                grad_a = 2.0 * _corr_fft(w, a, corr_len_fft)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                lam = min(lam * lam_factor, 1e4)

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[quick n={n} cid={cid} it={t:04d}] new best {val:.6f}")

                if time.time() > deadline:
                    break

            # ------------------------------------------------------------
            # short exact sub‑gradient (exponentiated‑gradient) phase
            a = a / a.sum()
            eta = 0.5
            sub_iters = 1500
            for sub_it in range(sub_iters):
                conv = _conv_fft(a, conv_len_fft)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)
                grad_a = 2.0 * _corr_fft(w, a, corr_len_fft)
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[quick sub n={n} cid={cid} it={sub_it}] new best {val:.6f}")

                if time.time() > deadline:
                    break

    # ------------------------------------------------------------------
    # 2. DEEP REFINEMENT on a larger length (pick the longest tried or 16384)
    if best_seq is None:
        # safeguard – should never happen
        best_seq = np.full(1024, 1.0 / 1024.0, dtype=np.float64)

    # decide target length – try to go one step larger than the best so far
    target_n = max(best_seq.shape[0] * 2, 16384)
    # cap at a size that comfortably fits into the remaining budget
    if target_n > 65536:
        target_n = 65536

    print(f"\n=== Deep refinement, length n={target_n} ===")
    conv_len_fft = _next_pow2(2 * target_n - 1)
    corr_len_fft = _next_pow2(2 * target_n - 1 + target_n - 1)

    # Start from a good analytical shape (arcsine) but blend with the
    # best sequence found so far if it has the same length.
    if best_seq.shape[0] == target_n:
        a = best_seq.copy()
    else:
        a = _initial_arcsine(target_n)

    a = np.maximum(a, 0.0)
    a /= a.sum()        # keep it on the simplex

    logits = np.log(a + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    lr = 0.07 * np.sqrt(2048.0 / target_n)
    lam = 0.05
    lam_factor = 1.008

    # Adam phase – runs as long as we have time left
    iteration = 0
    last_improve = time.time()
    while time.time() < deadline:
        iteration += 1

        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        conv = _conv_fft(a, conv_len_fft)
        max_c = conv.max()

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _corr_fft(w, a, corr_len_fft)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        lam = min(lam * lam_factor, 5000.0)   # keep λ in a safe range

        val = 2.0 * target_n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={target_n} it={iteration}] new best {val:.6f}")

        # occasional gentle shake if stuck
        if time.time() - last_improve > 8.0:
            logits += (np.random.rand(target_n) - 0.5) * 5e-3
            last_improve = time.time()

        # stop the Adam loop if we have less than 0.2 s for the exact phase
        if time.time() > deadline - 0.2:
            break

    # ------------------------------------------------------------
    # Exact sub‑gradient (exponentiated‑gradient) phase – runs until deadline
    eta = 0.55
    sub_last_improve = time.time()
    sub_iter = 0
    while time.time() < deadline:
        sub_iter += 1
        conv = _conv_fft(a, conv_len_fft)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _corr_fft(w, a, corr_len_fft)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(target_n))
            s = a.sum()
        a = a / s

        val = 2.0 * target_n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[Subgrad n={target_n} it={sub_iter}] new best {val:.6f}")

        if time.time() - sub_last_improve > 10.0:
            # shake the simplex a little when progress stalls
            a += (np.random.rand(target_n) - 0.5) * 1e-3
            a = np.maximum(a, 0.0)
            a /= a.sum()
            sub_last_improve = time.time()

    # ------------------------------------------------------------------
    # Store the result into the global name for possible reuse by future runs
    global height_sequence_1
    height_sequence_1 = best_seq.tolist()

    # Very final safety: ensure non‑negativity and reasonable magnitude
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    return best_seq.tolist()
