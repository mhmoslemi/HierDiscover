def full_convolution_fft(x):
    n = x.shape[0]
    size = 1 << ((2*n - 1 - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(fx * fx, n=size)[:2*n-1]
    np.maximum(conv, 0, out=conv)
    return conv

def improve_with_lp(a, target_sum, deadline, max_iters=5):
    best = a.copy()
    best_val = max_convolution(a)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        M = max_convolution(best)
        # Solve LP
        n = best.shape[0]
        # Build constraint matrix A_ub
        rows = 2*n - 1
        A = np.zeros((rows, n), dtype=float)
        # fill using shifted a
        for j in range(n):
            A[j:j+n, j] = best
        # bounds
        bounds = [(0, None)] * n
        c = -np.ones(n)  # maximize sum(g)
        res = linprog(c, A_ub=A, b_ub=np.full(rows, M), bounds=bounds, method='highs')
        if not res.success:
            break
        g0 = res.x
        if g0.sum() <= 0:
            break
        # scale to target sum
        g = g0 * (target_sum / g0.sum())
        # maybe also try mixture (0.5)
        # Evaluate direct
        val_g = max_convolution(g)
        if val_g < best_val:
            best = g.copy()
            best_val = val_g
            # continue on new best
            continue
        # else try mixing
        blended = 0.5*best + 0.5*g
        blended_val = max_convolution(blended)
        if blended_val < best_val:
            best = blended.copy()
            best_val = blended_val
            continue
        # else break maybe
    return best, best_val
```

"""
Hybrid optimizer for the step‑function bound (improved).

The objective 2·n·max(conv(a,a))/sum(a)² is scale‑invariant,
so we fix Σ aᵢ = √(2 n) and minimise max(conv(a,a)).
The search consists of four stages:

1.  Diverse random parametrisations (exponential, power‑law, Gaussian,
    Hann/Hamming, inverse‑sqrt, sine‑window) with lengths in [1500,4000].
    The best candidate found is kept.

2.  Adam optimisation on a smoothed max‑convolution (log‑sum‑exp surrogate)
    using the soft‑max parametrisation.  Several restarts are performed.

3.  Exact active‑set projected sub‑gradient descent.  Instead of the
    single worst index we use all indices whose convolution value is
    within a small fraction of the current maximum, which yields a
    stronger descent direction.

4.  Linear‑programming “mass‑inflation’’ step: for the current sequence a
    we maximise Σ gⱼ under the linear constraints (a∗g)ₖ ≤ M (M = max(conv(a,a))).
    The optimal g is rescaled to the required ℓ₁‑norm and mixed with a.
    A few LP iterations can substantially lower the peak.

5.  Final simulated‑annealing hill‑climb that transfers tiny amounts of
    mass between random positions and accepts moves with a Metropolis rule.

All stages respect the overall time budget (default 1000 s) and share a
global best solution.  If nothing is found a uniform construction is
returned as a safe fallback.
"""

import time
import numpy as np

# optional LP support
try:
    from scipy.optimize import linprog
    _HAS_LP = True
except Exception:  # pragma: no cover
    _HAS_LP = False


# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of two real vectors."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # Clip tiny negatives caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]          # sorted descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)     # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1        # zero‑based index of last positive entry
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: compute max of self‑convolution (the objective after normalisation)
# ----------------------------------------------------------------------
def _max_conv(a: np.ndarray) -> float:
    """Return max(conv(a,a))."""
    return _full_convolution_fft(a, a).max()


# ----------------------------------------------------------------------
# Adam optimiser on the log‑it (soft‑max) parametrisation
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 5e-5,
    alpha0: float = 0.025,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser for the smoothed‑max surrogate."""
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # ----- map logits to the simplex (a = target_sum * softmax(x)) -----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)       # stable soft‑max
        p = exp_shift / np.sum(exp_shift)   # Σ p = 1
        a = target_sum * p

        # ----- exact max of the convolution (used for logging) -----
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam n={n} new best max={cur_max:.6f}")

        # ----- smoothed max (log‑sum‑exp) weights -----
        tau = max(init_tau_factor * cur_max * (0.99 ** it), min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t. a: 2·a⊛w_rev  -----
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # ----- back‑prop to logits (chain rule) -----
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slow decay of learning rate
        alpha *= 0.9995

    # final exact evaluation (max‑convolution) of the best vector
    final_val = _full_convolution_fft(best_a, best_a).max()
    return best_a, float(final_val)


# ----------------------------------------------------------------------
# Targeted projected sub‑gradient descent using an active set of constraints
# ----------------------------------------------------------------------
def _targeted_active_subgradient(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    eps_frac: float = 1e-3,
    init_step_factor: float = 0.2,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that follows all near‑active indices."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _max_conv(a)
    step = init_step_factor * target_sum / n

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        # indices that are within eps_frac of the max
        active = np.where(conv >= max_val - eps_frac * max_val)[0]

        # compute gradient as sum of gradients of the active peaks
        grad = np.zeros_like(a)
        for k in active:
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            js = np.arange(j_start, j_end + 1, dtype=int)
            grad[js] += 2.0 * a[k - js]

        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)
        cand_val = _max_conv(cand)

        if cand_val < best_val - 1e-12:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= 1.02
            if it % 200 == 0:
                print(f"[{time.time():.2f}] subgrad it={it} val={best_val:.6f}")
        else:
            step *= 0.70
            # occasional jitter
            if np.random.rand() < 0.07:
                jitter = np.random.rand(n) * 1e-4
                a = _project_onto_simplex(a + jitter, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming “mass‑inflation’’ step
# ----------------------------------------------------------------------
def _lp_mass_inflation(
    a: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 5,
) -> tuple[np.ndarray, float]:
    """Solve LP to enlarge total mass under the current max‑conv bound."""
    if not _HAS_LP:
        return a, _max_conv(a)

    n = a.shape[0]
    best = a.copy()
    best_val = _max_conv(best)

    # Pre‑allocate constraint matrix (dense, but still modest for n≈3000)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = a  # shifted copy of a

    # Lower and upper bounds for variables g_j >= 0
    bounds = [(0, None)] * n
    c = -np.ones(n)                     # maximise Σ g_j  <=> minimise -Σ g_j

    for it in range(max_iters):
        if time.time() > deadline:
            break
        M = _max_conv(best)

        # Solve LP: maximize Σ g_j  subject to (a ∗ g)_k ≤ M
        res = linprog(
            c,
            A_ub=A,
            b_ub=np.full(rows, M, dtype=float),
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": max(1.0, (deadline - time.time()))},
        )
        if not res.success:
            # LP failed – abort the LP phase
            break

        g0 = res.x
        if g0.sum() == 0.0:
            break

        # Rescale to required ℓ₁‑norm
        g = g0 * (target_sum / g0.sum())

        # Evaluate candidate and try mixing
        val_g = _max_conv(g)
        if val_g < best_val:
            best = g.copy()
            best_val = val_g
            continue

        # mix half‑and‑half (simple relaxation)
        mixed = 0.5 * best + 0.5 * g
        val_mixed = _max_conv(mixed)
        if val_mixed < best_val:
            best = mixed.copy()
            best_val = val_mixed
            continue

        # No improvement – stop early
        break

    return best, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (simulated annealing) on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Move tiny amounts of mass between positions with a Metropolis rule."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _max_conv(a)

    T0 = max(1e-3, best_val * 0.4)   # initial temperature
    T_end = 1e-5
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        delta = a[i] * np.random.rand() * 0.08   # move up to 8 % of source mass
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _max_conv(a_new)
        accept = False
        if new_val < best_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")

        if it % 2500 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val
