def simplex_projection(v):
    """Project vector v onto the probability simplex (non-negative, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def run_adam(a0, n, conv_len, corr_len, deadline, max_iters=0):
    # if max_iters==0 compute default based on n
    if max_iters == 0:
        max_iters = int(40000 * 1024 / n) + 2000
    a = a0.copy()
    logits = np.log(a + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.07 * np.sqrt(2048.0 / n)
    lam = 0.15
    lam_factor = 1.02
    best_a = a.copy()
    best_val = compute_objective(a, n, conv_len)
    last_improve = time.time()
    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # softmax
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()
        conv = conv_fft(a, conv_len)
        max_c = conv.max()
        # smooth max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        grad_a = 2.0 * corr_fft(w, a, corr_len)
        # backprop through softmax
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # temperature schedule
        lam = min(lam * lam_factor, 1e6)
        # evaluate
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t}] new local best {val:.6f}")
        # stagnation: random jitter
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

"""
Propose a non‑negative step‑function (list of heights) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

The search builds a diverse pool of initial shapes, runs a two‑stage
optimiser (Adam on a smooth‑max surrogate followed by an exact
exponentiated‑gradient sub‑gradient phase) and repeatedly fine‑tunes the
current global best.  After each length we up‑sample the best sequence
and use it as a warm start for the next (larger) length.  Finally a
small random‑perturbation local‑search is performed.  The routine respects
a user‑provided time budget (default 1000 s) and returns the best
sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based linear convolution and the cross‑correlation needed for gradients.
def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]
    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, conv_len: int) -> float:
    """Exact evaluation of the objective (sum(a) is assumed to be 1)."""
    conv = _conv_fft(a, conv_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _simplex_proj(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
def _build_initial_candidates(n: int) -> list[np.ndarray]:
    """Create a diversified list of starting points (all normalised to sum = 1)."""
    cand = []

    # simple deterministic patterns
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))          # uniform

    i = np.arange(n, dtype=np.float64)

    # arcsine (inverse‑sqrt edges)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    cand.append(arcs / arcs.sum())

    # cosine window (single bell)
    coswin = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    cand.append(coswin / coswin.sum())

    # half‑sine window
    sinwin = np.sin(np.pi * i / (n - 1))
    cand.append(sinwin / sinwin.sum())

    # Gaussian
    sigma = n / 8.0
    gauss = np.exp(-0.5 * ((i - (n - 1) / 2) / sigma) ** 2)
    cand.append(gauss / gauss.sum())

    # two‑sided exponential decay
    for gamma in (0.001, 0.003, 0.006):
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        cand.append(shape / shape.sum())

    # smooth tent (linear rise then fall)
    left = np.linspace(0, 1, n // 2, endpoint=False)
    right = left[::-1]
    tent = np.concatenate([left, right])
    if tent.shape[0] < n:                                 # odd length
        tent = np.concatenate([tent, [0.0]])
    cand.append(tent / tent.sum())

    # power‑law edge concentrated shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 7):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # random Dirichlet families
    cand.append(np.random.dirichlet(np.ones(n)))
    for conc in (0.2, 0.4, 0.6, 0.8):
        cand.append(np.random.dirichlet(np.full(n, conc)))
    for conc in (1.5, 2.0, 3.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # random sparse “spike’’ pattern (≈ √n spikes)
    m_spikes = max(2, int(np.sqrt(2 * n)))
    sparse = np.zeros(n, dtype=np.float64)
    indices = np.random.choice(n, size=m_spikes, replace=False)
    sparse[indices] = 1.0
    if sparse.sum() > 0:
        cand.append(sparse / sparse.sum())

    # logistic bump (symmetric)
    for k in (0.1, 0.2, 0.4):
        mid = (n - 1) / 2.0
        sig = 1.0 / (1.0 + np.exp(-k * (i - mid)))
        shape = sig * (1 - sig)
        cand.append(shape / shape.sum())

    # polynomial bump (i·(n‑1‑i))^p
    for p in (1.0, 2.0, 3.0):
        shape = np.power(i * (n - 1 - i) + 1e-12, p)
        cand.append(shape / shape.sum())

    # rectangular window covering the middle 60 %
    rect = np.zeros(n, dtype=np.float64)
    a0 = n // 5
    rect[a0:-a0] = 1.0
    cand.append(rect / rect.sum())

    # ensure non‑negative & normalised
    out = []
    for a in cand:
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-12:
            out.append(a / s)
    return out


# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray, n: int, conv_len: int, corr_len: int,
              deadline: float, max_iters: int = 0) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max surrogate of the max convolution.
    Returns the best sequence seen and its objective value.
    """
    if max_iters == 0:
        max_iters = int(40000 * 1024 / n) + 2000

    a = a0.copy()
    logits = np.log(a + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.07 * np.sqrt(2048.0 / n)
    lam = 0.15
    lam_factor = 1.02

    best_a = a.copy()
    best_val = _objective(a, n, conv_len)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max – guarantees a lies on the simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # convolution & smooth‑max weights
        conv = _conv_fft(a, conv_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate (linear in w)
        grad_a = 2.0 * _corr_fft(w, a, corr_len)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule – gradually focus on the true max
        lam = min(lam * lam_factor, 1e6)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t:05d}] new local best {val:.6f}")

        # if we have not improved for a while, shake the logits a bit
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
    return best_a, best_val


# ----------------------------------------------------------------------
def _run_exponentiated_gradient(a0: np.ndarray, n: int,
                               conv_len: int, corr_len: int,
                               deadline: float,
                               max_iters: int = 6000,
                               best_val_sofar: float = float('inf')) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient phase using the exponentiated‑gradient update.
    """
    a = a0.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a, n, conv_len)
    if best_val_sofar < best_val:
        best_val = best_val_sofar
        best_a = a.copy()

    eta = 0.55
    last_improve = time.time()

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, conv_len)
        max_c = conv.max()

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _corr_fft(w, a, corr_len)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                                 # protect against collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        val = 2.0 * n * np.max(_conv_fft(a, conv_len))
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it:04d}] new local best {val:.6f}")

        if time.time() - last_improve > 7.0:
            break
    return best_a, best_val


# ----------------------------------------------------------------------
def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a sequence to a larger length."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence that yields a low evaluation score.
    Respects the time budget (default 1000 s) and returns the best
    sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    best_seq = None
    best_val = float("inf")
    prev_seq = None                             # up‑sampled warm‑start for next length

    # length schedule – we go up to 16384 (still fast with FFT)
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        conv_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

        # ------------------------------------------------------------------
        # build a diversified list of initial candidates
        candidates = _build_initial_candidates(n)

        # add the up‑sampled version of the best sequence from the previous length
        if prev_seq is not None:
            up = _upsample_sequence(prev_seq, n)
            up = np.maximum(up, 0.0)
            s = up.sum()
            if s > 1e-12:
                candidates.append(up / s)

        # ------------------------------------------------------------------
        # optimise each candidate
        for cid, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # make sure we start from a proper simplex point
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s < 1e-12:
                a0 = np.random.dirichlet(np.ones(n))
                s = a0.sum()
            a0 = a0 / s

            # first the Adam phase
            a_adam, val_adam = _run_adam(a0, n, conv_len, corr_len, deadline)

            # then the exact sub‑gradient (exponentiated‑gradient) phase
            a_opt, val_opt = _run_exponentiated_gradient(
                a_adam, n, conv_len, corr_len, deadline,
                best_val_sofar=val_adam
            )

            # keep the best across all candidates so far
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[n={n} cand={cid}] GLOBAL NEW BEST {val_opt:.6f}")

        # ------------------------------------------------------------------
        # fine‑tune the global best for this length (if we have one of right size)
        if best_seq is not None and best_seq.shape[0] == n:
            print(f"[n={n}] fine‑tuning the current global best (val={best_val:.6f})")
            a_fine, val_fine = _run_adam(best_seq, n, conv_len, corr_len, deadline)
            if val_fine < best_val:
                best_val = val_fine
                best_seq = a_fine.copy()
                print(f"[n={n}] fine‑tune improved global best to {val_fine:.6f}")

        # remember this length's best for up‑sampling to the next length
        prev_seq = best_seq.copy() if best_seq is not None and best_seq.shape[0] == n else prev_seq

    # ----------------------------------------------------------------------
    # final perturbation‑based local search (helps escape shallow plateaus)
    if best_seq is not None:
        n = best_seq.shape[0]
        conv_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_len = 1 << int(np.ceil(np.log2(3 * n - 2)))
        for trial in range(15):
            if time.time() > deadline:
                break
            noise = np.random.randn(n) * 0.01
            pert = best_seq + noise
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s < 1e-12:
                continue
            pert = pert / s
            a_opt, val_opt = _run_adam(pert, n, conv_len, corr_len, deadline)
            a_opt, val_opt = _run_exponentiated_gradient(
                a_opt, n, conv_len, corr_len, deadline,
                best_val_sofar=val_opt
            )
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[perturb {trial}] GLOBAL NEW BEST {val_opt:.6f}")

    # ----------------------------------------------------------------------
    # fallback – a simple uniform vector (should never be needed)
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = _objective(best_seq, fallback_n,
                              1 << int(np.ceil(np.log2(2 * fallback_n - 1))))

    # expose the best sequence for possible warm‑starts in future calls
    global height_sequence_1
    height_sequence_1 = best_seq.tolist()

    print(f"\nBest value found: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
