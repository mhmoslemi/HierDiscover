def _generate_candidate_pool(n, warm):
    pool = []

    if warm: add warm and rotations.

    # Uniform baseline
    pool.append(np.full(n, 1.0 / n))

    # Symmetric power law (like previous)

    idx = np.arange(1, n+1)  # 1..n
    prod = idx * (n - idx + 1)
    for alpha in np.linspace(-2.5, -0.2, 20):
        shape = np.where(prod > 0, prod.astype(float) ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # Asymmetric exponential
    for lam_left in np.linspace(0.1, 3.0, 8):
        for lam_right in np.linspace(0.1, 3.0, 8):
            i = np.arange(n, dtype=float)
            shape = np.exp(-lam_left * i) + np.exp(-lam_right * (n-1-i))
            shape /= shape.sum()
            pool.append(shape)

    # Asymmetric power law sum shape
    xs = np.arange(n, dtype=float) + 1.0
    for alpha in np.linspace(0.2, 3.0, 7):
        for beta in np.linspace(0.2, 3.0, 7):
            shape = (xs ** (-alpha)) + ((n - xs + 1) ** (-beta))
            # Possibly large numbers; might overflow; but for moderate alpha,beta fine.
            shape = np.maximum(shape, 0.0)
            s = shape.sum()
            if s > 0:
                shape /= s
                pool.append(shape)

    # Beta distribution family
    for alpha in np.linspace(0.2, 5.0, 6):
        for beta in np.linspace(0.2, 5.0, 6):
            x = (np.arange(n) + 0.5) / n
            shape = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
            shape = np.nan_to_num(shape, nan=0.0, posinf=0.0, neginf=0.0)
            s = shape.sum()
            if s > 0:
                shape /= s
                pool.append(shape)

    # Random Dirichlet draws
    for alpha in np.logspace(-2, 0, 10):
        shape = np.random.dirichlet(np.full(n, alpha))
        pool.append(shape)

    # Add some random perturbations of uniform
    for _ in range(5):
        shape = np.full(n, 1.0 / n) + 0.01*np.random.randn(n)
        shape = np.maximum(shape, 0)
        s = shape.sum()
        if s > 0:
            shape /= s
            pool.append(shape)

    return pool

"""
Propose a (near‑optimal) non‑negative step‑function for the objective

        F(a) = 2·n·max(a∗a) / (∑a)² .

The method consists of three phases:

1. **Rich candidate pool** – a large, diverse set of probability vectors is
   built (uniform, symmetric/asymmetric power‑law, asymmetric double‑exponential,
   asymmetric Beta pdfs, random Dirichlet draws, warm‑starts and simple
   perturbations).  Every vector is normalised to unit sum and its exact
   objective is evaluated with an FFT‑based convolution.  The best few
   candidates are kept.

2. **Deep local optimisation** – each kept candidate is refined by
   *Adam* on a smooth‑max (log‑sum‑exp) surrogate of the max‑convolution,
   followed by an *exact sub‑gradient* phase (exponentiated‑gradient mirror
   descent on the simplex).  Hyper‑parameters (learning‑rate, temperature
   schedule, step‑size) are tuned for fast convergence and the routine is
   repeatedly stopped/re‑started with a small random “shake” when progress
   stalls.

3. **Upsampling & final refinement** – the best distribution found on a
   modest grid (n≈2048) is interpolated to a larger grid (4096, then 8192 if
   time permits) and the deep optimisation is repeated, which typically
   yields a lower objective value.

All numbers are forced to be non‑negative and ≤ 1000 (the evaluator clips
them automatically).  The function respects the supplied time budget
(`budget_s`, default 1000 s) and returns the best sequence as a plain Python
list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns an array of length len(a) (the part that aligns with a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective (sum(a) is assumed > 0)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑sum‑exp surrogate of max(a*a).
    Returns the best sequence/value seen (global across all calls).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                 # logits → probabilities via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.10                                 # a bit larger than the previous 0.05

    lam = 0.2                                 # temperature (small → smooth)
    lam_factor = 1.015                        # increase slowly, approach the true max

    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        # ----- simplex projection via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (useful for logging)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        # keep the globally best solution
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        # w_i ∝ exp(lam * (conv_i - max_c))
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate w.r.t. a (before the soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain‑rule through the soft‑max (mirrored gradient on logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase temperature – we converge to the true max
        lam = min(lam * lam_factor, 1e5)

        # occasional random shake when we have been stagnant
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        if it >= 120_000:                    # safety bound
            break

    return global_best_seq, global_best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent sub‑gradient on the exact max‑convolution.
    Works on the probability simplex (exponentiated‑gradient update).
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total

    eta0 = 0.7                               # start step‑size
    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        # record global best
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # exact gradient of the max‑convolution
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step size (mirror‑descent on the simplex)
        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        # re‑normalise to the simplex
        s = a.sum()
        if s < 1e-12:                         # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 8.0:   # stop if no recent improvement
            break
        it += 1
        if it >= 200_000:                     # safety bound
            break

    return global_best_seq, global_best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam → sub‑gradient on a single start vector.
    Returns the best sequence/value encountered (global across all calls).
    """
    # ---- Adam on smooth‑max ----
    best_seq, best_val = _adam_optimize(
        a0, n, conv_fft_len, corr_fft_len, deadline, global_best_seq, global_best_val
    )
    # ---- Exact sub‑gradient ----
    best_seq, best_val = _subgradient_optimize(
        best_seq, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
    )
    return best_seq, best_val

def _generate_candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of probability vectors (sum = 1, non‑negative)."""
    pool: list[np.ndarray] = []

    # Warm‑start and simple rotations
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Uniform baseline
    pool.append(np.full(n, 1.0 / n))

    # --------------------------------------------------------------
    # 1. Symmetric power‑law (edges heavy)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1)
    for alpha in np.linspace(-2.5, -0.2, 20):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # --------------------------------------------------------------
    # 2. Asymmetric double‑exponential (two‑sided Laplace with different rates)
    i = np.arange(n, dtype=np.float64)
    for lam_l in np.linspace(0.1, 3.0, 8):
        for lam_r in np.linspace(0.1, 3.0, 8):
            shape = np.exp(-lam_l * i) + np.exp(-lam_r * (n - 1 - i))
            shape = np.maximum(shape, 0.0)
            shape /= shape.sum()
            pool.append(shape)

    # --------------------------------------------------------------
    # 3. Asymmetric power‑law sum shape: (i+1)^-α + (n-i)^-β
    xs = np.arange(n, dtype=np.float64) + 1.0
    for alpha in np.linspace(0.2, 3.0, 7):
        for beta in np.linspace(0.2, 3.0, 7):
            shape = (xs ** (-alpha)) + ((n - xs + 1) ** (-beta))
            shape = np.maximum(shape, 0.0)
            s = shape.sum()
            if s > 0:
                shape /= s
                pool.append(shape)

    # --------------------------------------------------------------
    # 4. Asymmetric Beta PDFs (discretised)
    x = (np.arange(n) + 0.5) / n  # mid‑point of each bin
    for a in np.linspace(0.2, 5.0, 6):
        for b in np.linspace(0.2, 5.0, 6):
            shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
            # Numerical safety
            shape = np.nan_to_num(shape, nan=0.0, posinf=0.0, neginf=0.0)
            s = shape.sum()
            if s > 0:
                shape /= s
                pool.append(shape)

    # --------------------------------------------------------------
    # 5. Random Dirichlet draws (various concentration parameters)
    for conc in np.logspace(-2, 0, 12):
        shape = np.random.dirichlet(np.full(n, conc))
        pool.append(shape)

    # --------------------------------------------------------------
    # 6. Small random perturbations of the uniform vector
    for _ in range(5):
        shape = np.full(n, 1.0 / n) + 0.02 * np.random.randn(n)
        shape = np.maximum(shape, 0.0)
        s = shape.sum()
        if s > 0:
            shape /= s
            pool.append(shape)

    return pool

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation metric.
    The routine respects the supplied time‑budget (default 1000 s) and returns
    the best sequence encountered as a plain Python list.
    """
    np.random.seed(seed)
    # Small safety margin to avoid overrunning the time limit
    deadline = time.time() + budget_s - 0.5

    # ------------------------------------------------------------------
    # Phase 1 – work on a modest grid first (fast iterations)
    n = 2048
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Warm‑start from a previously discovered construction, if any
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        w = np.asarray(warm_start, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        else:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
        else:
            w = None
    else:
        w = None

    # ------------------------------------------------------------------
    # Build candidates
    candidates = _generate_candidate_pool(n, w)

    best_seq: np.ndarray | None = None
    best_val = float("inf")
    scored: list[tuple[float, np.ndarray]] = []

    print(f"[setup] generated {len(candidates)} raw candidates (n={n})")

    for idx, cand in enumerate(candidates):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:                         # avoid pathological zero‑sum vectors
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        val = _objective(a, n, conv_fft_len)
        scored.append((val, a))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        if idx % 25 == 0:
            print(f"[screen {idx:04d}/{len(candidates)}] obj={val:.6f}")
        # early exit if we run out of time
        if time.time() >= deadline - 30:     # keep a few seconds for optimisation
            break

    # keep the most promising starts (up to 10)
    scored.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in scored[:10]]

    # ------------------------------------------------------------------
    # Phase 2 – deep optimisation on the selected candidates
    for cid, a0 in enumerate(top_candidates):
        if time.time() >= deadline - 5:
            break
        print(f"[optim start {cid}] launching deep optimisation (n={n})")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )
        print(f"[optim end {cid}] current global best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – up‑sample to larger supports if time permits
    # First upscale to 4096, then (if still time) to 8192.
    for n_big in (4096, 8192):
        if time.time() + 12 >= deadline:      # need a few seconds for the next phase
            break
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # Interpolate the current best sequence onto the new grid
        if best_seq is not None:
            x_old = np.linspace(0.0, 1.0, n, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.full(n_big, 1.0 / n_big)

        print(f"[upsample] starting optimisation on n={n_big}")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        print(f"[upsample] best after n={n_big} is {best_val:.6f}")

        # prepare for possible next up‑sample
        n = n_big
        conv_fft_len = conv_fft_len_big
        corr_fft_len = corr_fft_len_big

    # ------------------------------------------------------------------
    # Finalise – ensure compliance with the evaluator's clipping rules
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to [0, 1000] (the evaluator does this automatically, but we do it
    # explicitly to avoid accidental negative or huge values)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[final] returning sequence of length {best_seq.size}, best value {best_val:.6f}")
    return best_seq.tolist()
