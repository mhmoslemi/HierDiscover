def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros case?
        theta = 0
    else:
        rho = ind[cond][-1] - 1  # zero-indexed
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

Algorithm
---------

* **Stage 0 – utilities** – power‑of‑two utilities, FFT‑based convolution,
  cross‑correlation, and a fast Euclidean projection onto the probability
  simplex (sum = 1, a_i ≥ 0).

* **Stage 1 – diverse warm‑starts** – we try a handful of sensible seeds:
  previous best (if any), a random Dirichlet, a symmetric power‑law edge,
  a half‑wave sinusoid and the uniform vector.  All are normalised to sum = 1.

* **Stage 2 – L2‑flattening** – we minimise the squared deviation of the
  full convolution from the flat target 1/(2 n‑1).  The gradient of the
  L2‑loss is 4·(diff ★ a) where `★` is linear cross‑correlation.
  After each gradient step we project back onto the simplex.
  This quickly drives the autocorrelation toward a constant and gives a
  good reduction of the maximal entry.

* **Stage 3 – exact max‑subgradient refinement** – starting from the best
  vector obtained in Stage 2 we run a classic sub‑gradient method for the
  true objective `max(a∗a)`.  For the active shift(s) the sub‑gradient is
  simply a (weighted) shifted copy of `a`, which we compute by a single
  FFT‑based cross‑correlation.  A diminishing step‑size and simplex projection
  are used.

* **Stage 4 – cheap random local search** – we repeatedly move a random
  fraction of mass from one coordinate to another and keep the move if the
  exact objective improves.

* **Stage 5 – optional up‑sampling** – if time permits we linearly
  interpolate the current best sequence to length 2·n, repeat a short
  L2‑flattening pass and keep the better of the two.

All phases respect the global time budget (default 1000 s) and the function
always returns the best sequence found.  The code relies only on NumPy and
no external files or network access.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a).
    Returns a vector of length len(a):
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    Implementation follows the O(n) algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)           # 1‑based indexing
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]                 # last true index (1‑based)
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a low‑value non‑negative step‑function within the given time."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # problem size (the “standard” length used by the reference solution)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0 – initialise a pool of starting vectors
    candidates = []

    # warm start (if a previous solution is present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] != n:
            if w.shape[0] > n:
                w = w[:n]
            else:
                w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        s = w.sum()
        if s >= 1e-12:
            w /= s
            candidates.append(w)

    # random Dirichlet
    candidates.append(np.random.dirichlet(np.ones(n)))

    # symmetric power‑law edge‑concentrated shape (α = ‑1)
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    shape = np.where(prod > 0, prod ** -1.0, 0.0)
    if shape.sum() > 0:
        shape /= shape.sum()
        candidates.append(shape)

    # half‑wave sinusoid (positive part only)
    sin = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    sin = np.maximum(sin, 0.0)
    sin /= sin.sum()
    candidates.append(sin)

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # ------------------------------------------------------------------
    # 1 – short L2‑flattening runs for each seed
    target_const = 1.0 / (2 * n - 1)        # flat convolution value when Σa = 1

    for cid, a0 in enumerate(candidates):
        if time.time() > deadline:
            break

        a = np.maximum(a0, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
        else:
            a = a / s

        eta0 = 0.9
        max_iters = 20000
        it = 0
        while it < max_iters and time.time() < deadline:
            conv = _full_convolution(a, conv_fft_len)         # length 2n‑1
            max_c = conv.max()
            val = 2.0 * n * max_c   # Σa = 1  → denominator = 1
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[seed {cid}] new best {val:.6f} (iter {it})")

            diff = conv - target_const
            grad = 4.0 * _cross_correlation(diff, a, corr_fft_len)

            gnorm = np.linalg.norm(grad)
            if gnorm > 0:
                step = eta0 / np.sqrt(it + 1)
                a = a - step * (grad / gnorm)
                a = _project_simplex(a)
            it += 1

    # ------------------------------------------------------------------
    # 2 – exact max‑subgradient refinement (projected gradient version)
    if best_seq is None:
        best_seq = np.random.dirichlet(np.ones(n))

    a = best_seq.copy()
    eta = 0.6
    it = 0
    while time.time() < deadline - 5.0:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[subgrad] new best {val:.6f} (iter {it})")

        # active set (indices attaining the maximum, tolerance to avoid numerical noise)
        active = np.isclose(conv, max_c, atol=1e-12)
        if not active.any():
            break
        # uniform weight on the active shifts
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        # sub‑gradient = 2·(w ★ a)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = eta / np.sqrt(it + 1)
        a = a - step * grad
        a = _project_simplex(a)
        it += 1

    # ------------------------------------------------------------------
    # 3 – cheap random local hill‑climbing
    if time.time() < deadline - 5.0:
        a = best_seq.copy()
        conv = _full_convolution(a, conv_fft_len)
        best_local_val = 2.0 * n * conv.max()
        best_local_a = a.copy()
        max_moves = 30000
        move_it = 0
        while time.time() < deadline and move_it < max_moves:
            i = np.random.randint(n)
            if a[i] < 1e-14:
                move_it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                move_it += 1
                continue
            # transfer a random fraction (up to 30 %) of the mass at i to j
            delta = a[i] * (np.random.rand() * 0.3)
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta
            conv2 = _full_convolution(a2, conv_fft_len)
            val2 = 2.0 * n * conv2.max()
            if val2 < best_local_val:
                best_local_val = val2
                best_local_a = a2.copy()
                a = a2
                if move_it % 2000 == 0:
                    print(f"[local] improved to {val2:.6f} (move {move_it})")
            move_it += 1
        if best_local_val < best_val:
            best_val = best_local_val
            best_seq = best_local_a.copy()
            print(f"[local] global best improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4 – optional up‑sampling to 2·n for a final refinement
    if time.time() + 30.0 < deadline:
        n_big = n * 2
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)
        # linear interpolation
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        target_const_big = 1.0 / (2 * n_big - 1)
        eta0 = 0.8
        it_big = 0
        max_iters_big = 50000
        while it_big < max_iters_big and time.time() < deadline:
            conv = _full_convolution(a_big, conv_fft_len_big)
            max_c = conv.max()
            val = 2.0 * n_big * max_c
            if val < best_val:
                best_val = val
                best_seq = a_big.copy()
                print(f"[big] new best {val:.6f} (iter {it_big})")
            diff = conv - target_const_big
            grad = 4.0 * _cross_correlation(diff, a_big, corr_fft_len_big)
            gnorm = np.linalg.norm(grad)
            if gnorm > 0:
                step = eta0 / np.sqrt(it_big + 1)
                a_big = a_big - step * (grad / gnorm)
                a_big = _project_simplex(a_big)
            it_big += 1

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return value
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # Final clipping to the evaluator's allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
