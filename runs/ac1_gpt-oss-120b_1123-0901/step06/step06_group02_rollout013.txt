def generate_initial_sequence(n, rng):
    shape_type = rng.choice(['symmetric_exp', 'symmetric_pow', 'concat_two_exp', 'hann', 'hamming', 'dirichlet'])
    if shape_type == 'symmetric_exp':
        tau = rng.uniform(0.05, 0.5) * n   # (tau small if decays fast)
        idx = np.arange(n)
        a = np.exp(-np.minimum(idx, n-1-idx) / tau)
    elif shape_type == 'symmetric_pow':
        p = rng.uniform(0.2, 2.0)
        idx = np.arange(1, n+1)
        a = (np.minimum(idx, n+1-idx) ** -p)
    elif shape_type == 'concat_two_exp':
        # numbers for each half
        tau1 = rng.uniform(0.05, 0.5) * (n//2)
        tau2 = rng.uniform(0.05, 0.5) * (n//2)
        left = np.exp(-np.arange(n//2) / tau1)
        right = np.exp(-np.arange(n-n//2) / tau2)

    elif shape_type == 'hann':
        i = np.arange(n)

"""
Hybrid multi‑restart optimizer for the step‑function bound.

The evaluation objective is scale‑invariant; we fix Σ a_i = √(2 n) and
minimise max(conv(a,a)).  The optimizer works in four layers:

1.  Warm‑start from any pre‑computed global sequence.
2.  A fast random‑restart loop: for a random length n we draw an
    initial shape from several parametric families (symmetric exponential,
    symmetric power‑law, Hann/Hamming windows, Dirichlet draws,
    sparse spikes).  Each candidate is refined by a simulated‑annealing
    hill‑climb that moves a variable fraction of mass between two
    positions.  The move size shrinks with the remaining time to allow
    both exploration and fine tuning.
3.  After the restart phase we run a longer Adam optimisation on a
    soft‑max parametrisation (log‑sum‑exp surrogate of the maximum).
4.  A final exact projected sub‑gradient descent followed by a short
    hill‑climb gives the final polish.

All stages respect the overall time budget (default 1000 s) and keep
track of the globally best sequence seen.  The returned value is a plain
Python list of non‑negative floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used everywhere)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of real vectors x and y."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # Clip tiny negatives caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]          # sorted descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)     # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1        # zero‑based index of last positive entry
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the log‑it (soft‑max) parametrisation
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 5e-5,
    alpha0: float = 0.025,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser for the smoothed‑max surrogate."""
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # map logits to simplex (soft‑max)
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)           # stable
        p = exp_shift / np.sum(exp_shift)       # Σ p = 1
        a = target_sum * p

        # exact max of the convolution (for bookkeeping)
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam n={n}  new best max={cur_max:.6f}")

        # log‑sum‑exp smoothing weights
        tau = max(init_tau_factor * cur_max * (0.99 ** it), min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a: 2·a ⊛ rev(w)
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # back‑prop to logits
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slow decay of learning rate
        alpha *= 0.9995

    final_val = _full_convolution_fft(best_a, best_a).max()
    return best_a, float(final_val)


# ----------------------------------------------------------------------
# Targeted projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _targeted_subgradient(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent that follows the current worst index."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step = 0.2 * target_sum / n          # initial step size

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())            # worst index
        # gradient of conv_k w.r.t. a
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # projected step
        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)

        cand_val = _full_convolution_fft(cand, cand).max()
        if cand_val < best_val - 1e-12:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= 1.02                       # slightly increase step after success
            if it % 200 == 0:
                print(f"[{time.time():.2f}] subgrad it={it}  val={best_val:.6f}")
        else:
            step *= 0.70                       # reduce step on failure
            # occasional jitter
            if np.random.rand() < 0.07:
                jitter = np.random.rand(n) * 1e-4
                a = _project_onto_simplex(a + jitter, target_sum)

        # early exit if already very good
        if best_val < 1.503:
            break
    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Move tiny amounts of mass between random positions with a Metropolis rule."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # temperature schedule: start at a fraction of the current peak
    T0 = max(1e-3, best_val * 0.4)
    T_end = 1e-5
    start = time.time()

    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        # move size shrinks with the remaining temperature
        max_move_frac = 0.5 * (T / T0) + 0.05   # from 0.55 down to 0.05

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue
        delta = a[i] * np.random.rand() * max_move_frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()
        accept = False
        if new_val < best_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it}  new best={best_val:.6f}")

        # occasional re‑normalisation to avoid drift
        if it % 2500 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:      # early stop when target is reached
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Helpers for generating diverse initial shapes
# ----------------------------------------------------------------------
def _make_initial_sequence(n: int) -> np.ndarray:
    """Return a random non‑negative sequence of length n, normalised to Σ=√(2n)."""
    S = np.sqrt(2.0 * n)

    # choose a parametric family
    family = np.random.choice([
        "sym_exp",          # symmetric exponential decay from both ends
        "sym_pow",          # symmetric power‑law (≈1/√i)
        "hann", "hamming", # classical windows (peak in centre – kept for completeness)
        "dirichlet",        # random simplex point
        "sparse",           # few random spikes
    ])

    if family == "sym_exp":
        # τ proportional to n, chosen uniformly in a modest range
        tau = np.random.uniform(0.05, 0.5) * n
        idx = np.arange(n)
        a = np.exp(-np.minimum(idx, n - 1 - idx) / tau)

    elif family == "sym_pow":
        p = np.random.uniform(0.2, 1.2)      # p≈0.5 gives 1/√i behaviour
        idx = np.arange(1, n + 1)
        a = (np.minimum(idx, n + 1 - idx) ** (-p))

    elif family == "hann":
        i = np.arange(n, dtype=float)
        a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))

    elif family == "hamming":
        i = np.arange(n, dtype=float)
        a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))

    elif family == "dirichlet":
        a = np.random.dirichlet(np.ones(n))

    else:   # "sparse"
        # pick a random number of spikes (5 %–20 % of n)
        k = np.random.randint(max(2, n // 20), max(2, n // 5) + 1)
        a = np.zeros(n, dtype=float)
        pos = np.random.choice(n, size=k, replace=False)
        a[pos] = np.random.rand(k) + 0.1   # avoid exact zeros for stability

    # optional tiny noise to break symmetry
    if np.random.rand() < 0.3:
        noise = np.random.randn(n) * a.max() * 0.03
        a = np.maximum(a + noise, 0.0)

    # normalise to the required L1 sum
    a = a * (S / a.sum())
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid optimiser for the step‑function bound.

    The routine respects the supplied time budget and returns the best
    sequence (as a plain Python list) found so far.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.10   # safety margin

    best_seq = None
    best_score = np.inf

    def _record(seq: np.ndarray):
        """Evaluate `seq` (exact max‑convolution) and keep the global best."""
        nonlocal best_seq, best_score
        val = _full_convolution_fft(seq, seq).max()
        if val < best_score:
            best_score = float(val)
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] GLOBAL new best={best_score:.6f}  n={len(seq)}")
        return val

    # ------------------------------------------------------------------
    # Warm‑start from any global construction that may already exist
    # ------------------------------------------------------------------
    try:
        hs = globals().get("height_sequence_1")
        if hs is not None:
            hs_arr = np.asarray(hs, dtype=float)
            if hs_arr.ndim == 1 and hs_arr.size > 0:
                n0 = hs_arr.shape[0]
                target0 = np.sqrt(2.0 * n0)
                hs_norm = hs_arr * (target0 / hs_arr.sum())
                _record(hs_norm)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – many fast random restarts with hill‑climbing
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        if now >= deadline - 12.0:   # reserve ~12 s for the final refinement phases
            break

        # pick a length – longer early on, shorter later
        remaining = deadline - now
        if remaining > 600:
            n = np.random.randint(4000, 8000 + 1)
        elif remaining > 300:
            n = np.random.randint(3000, 6000 + 1)
        else:
            n = np.random.randint(2000, 4500 + 1)

        # generate a diverse starting point
        a0 = _make_initial_sequence(n)

        # allocate a short budget for this hill‑climb (1–3 s)
        per_iter_budget = min(2.5, deadline - now - 6.0)
        sub_deadline = time.time() + per_iter_budget

        # aggressive hill‑climb
        seq_hc, val_hc = _hillclimb(a0, np.sqrt(2.0 * n), sub_deadline)
        _record(seq_hc)

        # if the hill‑climb produced an improvement, polish it with a quick sub‑gradient pass
        if val_hc < best_score + 0.001 and (deadline - time.time()) > 3.0:
            sub_deadline2 = time.time() + min(1.5, deadline - time.time() - 2.0)
            seq_sg, val_sg = _targeted_subgradient(seq_hc.copy(),
                                                   np.sqrt(2.0 * n),
                                                   sub_deadline2)
            _record(seq_sg)

    # ------------------------------------------------------------------
    # Phase 2 – Adam on a smoothed surrogate (longer run)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 8.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        x0 = np.log(best_seq + 1e-12)          # soft‑max logits
        run_deadline = time.time() + min(20.0, deadline - time.time() - 5.0)
        seq_ad, val_ad = _optimise_adam_softmax(
            n, x0, target_sum, run_deadline,
            init_tau_factor=0.14,
            min_tau_factor=3e-5,
            alpha0=0.027,
        )
        _record(seq_ad)

    # ------------------------------------------------------------------
    # Phase 3 – Exact sub‑gradient descent (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_deadline = deadline - 2.5
        seq_sg2, val_sg2 = _targeted_subgradient(best_seq.copy(),
                                                 target_sum,
                                                 sub_deadline)
        _record(seq_sg2)

    # ------------------------------------------------------------------
    # Phase 4 – Final short hill‑climb for fine‑tuning
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        seq_final, val_final = _hillclimb(best_seq.copy(),
                                          target_sum,
                                          deadline)
        _record(seq_final)

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = _full_convolution_fft(best_seq, best_seq).max()
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # Return a plain Python list as required by the evaluator
    return best_seq.tolist()
