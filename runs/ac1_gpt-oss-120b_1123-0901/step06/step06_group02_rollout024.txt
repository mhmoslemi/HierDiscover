"""
Hybrid multi‑restart optimizer for the step‑function bound
=======================================================

The objective 2·n·max(conv(a,a))/sum(a)² is scale‑invariant, so we
normalise each candidate to Σ aᵢ = √(2 n) and simply minimise
max(conv(a,a)).

The search consists of many short independent runs (random initial
shape → brief Adam → sub‑gradient → simulated‑annealing hill‑climb).
Each run explores a different length and a different family of
shapes (exponential, power‑law, Hann/Hamming windows, Dirichlet,
sorted random, linear ramp).  The best sequence seen so far is kept
globally and refined in a final long sub‑gradient + hill‑climb phase.

All phases respect the global time budget (default 1000 s) and
periodically print progress.  The routine returns a plain Python list
as required by the evaluator.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of real vectors x and y."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # Clip tiny negatives caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]          # sorted descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)     # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1        # zero‑based index of last positive entry
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: maximum entry of a*a convolution (exact objective)
# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray) -> float:
    return float(_full_convolution_fft(a, a).max())


# ----------------------------------------------------------------------
# Random initialisers (various parametric families)
# ----------------------------------------------------------------------
def _random_initial_sequence(n: int) -> np.ndarray:
    """Return a random non‑negative sequence of length n (not normalised)."""
    shape = np.random.choice(
        ["exp", "pow", "hann", "hamming", "dirichlet", "sorted", "linear"]
    )
    if shape == "exp":
        tau = np.random.uniform(0.1, 2.5) * n
        idx = np.arange(n, dtype=float)
        a = np.exp(-idx / tau)
    elif shape == "pow":
        p = np.random.uniform(0.2, 1.7)
        idx = np.arange(1, n + 1, dtype=float)
        a = idx ** (-p)
    elif shape == "hann":
        i = np.arange(n, dtype=float)
        a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    elif shape == "hamming":
        i = np.arange(n, dtype=float)
        a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    elif shape == "dirichlet":
        a = np.random.dirichlet(np.ones(n))
    elif shape == "sorted":
        a = np.sort(np.random.rand(n))[::-1] + 1e-6   # decreasing random
    else:  # linear ramp (decreasing)
        a = np.linspace(1.0, 0.01, n)

    # Occasionally add a tiny Gaussian perturbation
    if np.random.rand() < 0.25:
        noise = np.random.randn(n) * np.max(a) * 0.02
        a = np.maximum(a + noise, 0.0)

    return a.astype(float)


# ----------------------------------------------------------------------
# Adam optimiser on the log‑sum‑exp surrogate of the max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, target_sum: float, deadline: float):
    """
    Adam optimiser for the smoothed‑max surrogate.
    Returns the best coefficient vector (on the simplex) and its exact
    max‑convolution value.
    """
    x = np.log(a0 + 1e-12)               # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = 0.03                         # initial learning rate

    while time.time() < deadline:
        # map logits to simplex via soft‑max
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        p = exp_shift / np.sum(exp_shift)          # Σ p = 1
        a = target_sum * p

        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam new best max={cur_max:.6f}")

        # smoothed max (log‑sum‑exp) weights
        tau = max(0.08 * cur_max * (0.99 ** it), 2e-5 * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient wrt a: 2·a⊛w_rev
        w_rev = w[::-1]
        size2 = 1 << ((2 * a.shape[0] - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * a.shape[0] - 2]
        idx = (2 * a.shape[0] - 2) - np.arange(a.shape[0])
        grad_a = 2.0 * conv2[idx]

        # back‑prop to logits
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # Adam update
        it += 1
        m = 0.9 * m + 0.1 * grad_x
        v = 0.999 * v + 0.001 * (grad_x ** 2)

        m_hat = m / (1.0 - 0.9 ** it)
        v_hat = v / (1.0 - 0.999 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + 1e-8)
        x -= step

        # occasional jitter to shake out of plateaus
        if it % 300 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(x.size) * 0.001

        # decay learning rate gently
        alpha *= 0.9995

    final_val = _conv_max(best_a) if best_a is not None else np.inf
    return best_a, float(final_val)


# ----------------------------------------------------------------------
# Targeted projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray, target_sum: float, deadline: float):
    """Sub‑gradient descent that follows the current worst anti‑diagonal."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _conv_max(a)

    # start with a relatively aggressive step size
    step = 0.5 * target_sum / n

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())                     # worst index
        # gradient of conv_k w.r.t. a
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # projected step
        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)
        cand_val = _conv_max(cand)

        if cand_val < best_val - 1e-12:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= 1.02                    # modestly increase after success
            if it % 200 == 0:
                print(f"[{time.time():.2f}] subgrad it={it}  val={best_val:.6f}")
        else:
            step *= 0.70                    # shrink on failure
            # occasional tiny jitter to avoid stagnation
            if np.random.rand() < 0.07:
                jitter = np.random.rand(n) * 1e-4
                a = _project_onto_simplex(a + jitter, target_sum)

        if best_val < 1.503:
            break
    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (simulated annealing) on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_optimize(a0: np.ndarray, target_sum: float, deadline: float):
    """Move tiny amounts of mass between positions with a Metropolis rule."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _conv_max(a)

    # initial temperature proportional to current peak
    T0 = max(1e-3, best_val * 0.6)
    T_end = 1e-5
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # delta fraction decreases from ~30 % to ~2 % of source mass
        max_frac = 0.30 * (1.0 - fraction) + 0.02 * fraction
        delta = a[i] * np.random.rand() * max_frac

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _conv_max(a_new)

        accept = False
        if new_val < best_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it}  new best={best_val:.6f}")

        # full renormalisation every few thousand steps
        if it % 2500 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break
    return best_a, best_val

                                                 target_sum, hill_deadline)
        if val_hill < best_score:
            best_score = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] after hillclimb iter={iteration}  val={val_hill:.6f}")

        # stop early if target reached
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Phase 1 – final deep refinement on the global best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)

        # long sub‑gradient descent
        sub_deadline = time.time() + min(60.0, deadline - time.time() - 5.0)
        seq_sub, val_sub = _subgradient_optimize(best_seq.copy(), target_sum, sub_deadline)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] final subgrad best={best_score:.6f}")

        # final hill‑climb (uses whatever time remains)
        if time.time() < deadline - 1.0:
            hill_deadline = deadline
            seq_hill, val_hill = _hillclimb_optimize(best_seq.copy(), target_sum, hill_deadline)
            if val_hill < best_score:
                best_score = val_hill
                best_seq = seq_hill.copy()
                print(f"[{time.time():.2f}] final hillclimb best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be used)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = _conv_max(best_seq)
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    print(f"[{time.time():.2f}] DONE best score={best_score:.6f} length={len(best_seq)}")
    return best_seq.tolist()
