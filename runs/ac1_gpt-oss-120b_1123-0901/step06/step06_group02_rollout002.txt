"""
Hybrid multi‑start optimiser for the step‑function bound.

The search is organised in four stages:

1.  Randomised parametric initialisations across a variety of lengths
    (exponential, power‑law, linear, sinusoidal, log, sigmoid, Gaussian …).
    Every candidate is normalised to Σa = √(2 n) and its exact max‑convolution
    value (the objective after normalisation) is recorded.

2.  For the best few candidates a soft‑max Adam optimiser refines the shape.
    The logits are optimised on a log‑sum‑exp surrogate of the max‑convolution,
    while the simplex constraint is enforced by a soft‑max mapping.

3.  Exact projected sub‑gradient descent with a simple back‑tracking line
    search follows the true worst convolution index.  The step size is
    increased on successful moves and reduced otherwise.

4.  A final simulated‑annealing hill‑climb moves tiny amounts of mass between
    random positions, accepting moves with a Metropolis rule.  This cleans up
    any remaining local irregularities.

All stages respect the global time budget (default 1000 s).  The routine
keeps the globally best sequence seen and returns it as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors x and y (no padding artefacts)."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)            # clip tiny negatives
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (s ≥ 0)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]          # descending sort
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)     # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1       # zero‑based index of last positive entry
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: evaluate the exact max‑convolution (objective after normalisation)
# ----------------------------------------------------------------------
def _max_conv(a: np.ndarray) -> float:
    """Return max(conv(a,a)) – the objective when Σa = √(2 n)."""
    return float(_full_convolution_fft(a, a).max())


# ----------------------------------------------------------------------
# Random parametric families for initialisation
# ----------------------------------------------------------------------
def _random_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a non‑negative sequence of length n from a random parametric family."""
    shape = rng.choice(
        ["exp", "pow", "hann", "hamming", "linear", "log", "sigmoid", "gauss", "beta"]
    )
    i = np.arange(n, dtype=float)

    if shape == "exp":
        tau = rng.uniform(0.1, 2.5) * n
        a = np.exp(-i / tau)

    elif shape == "pow":
        p = rng.uniform(0.2, 1.7)
        idx = i + 1.0
        a = idx ** (-p)

    elif shape == "hann":
        a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))

    elif shape == "hamming":
        a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))

    elif shape == "linear":
        a = (n - i) / n

    elif shape == "log":
        p = rng.uniform(0.5, 2.0)
        a = 1.0 / (np.log(i + 2.0) ** p + 1e-12)

    elif shape == "sigmoid":
        centre = rng.uniform(0.3, 0.7) * n
        scale = rng.uniform(0.05, 0.2) * n
        a = 1.0 / (1.0 + np.exp((i - centre) / scale))

    elif shape == "gauss":
        mu = rng.uniform(0.3, 0.7) * n
        sigma = rng.uniform(0.05, 0.2) * n
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)

    else:  # "beta"
        alpha = rng.uniform(0.5, 5.0)
        beta = rng.uniform(0.5, 5.0)
        a = rng.beta(alpha, beta, size=n)

    # optional tiny noise to escape symmetry
    if rng.random() < 0.2:
        noise = rng.standard_normal(n) * a.max() * 0.01
        a = np.maximum(a + noise, 0.0)

    return a


# ----------------------------------------------------------------------
# Adam optimiser on the log‑sum‑exp (soft‑max) surrogate
# ----------------------------------------------------------------------
def _adam_softmax(
    n: int,
    logits0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 5e-5,
    alpha0: float = 0.04,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Run Adam on the log‑sum‑exp surrogate and return best vector and its max‑conv."""
    x = logits0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    # pre‑allocate FFT size for gradient (2 n‑2 is the maximum index of conv)
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())

    while time.time() < deadline:
        # map logits → simplex via soft‑max
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        p = exp_shift / exp_shift.sum()
        a = target_sum * p

        # exact max‑convolution for logging / tau schedule
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam n={n} new best max={cur_max:.6f}")

        # smoothed max via log‑sum‑exp weights
        tau = max(init_tau_factor * cur_max * (0.99 ** it), min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a: 2 * a ⊛ w_rev (convolution)
        w_rev = w[::-1]
        Fa = np.fft.rfft(a, n=fft_len)
        Fw = np.fft.rfft(w_rev, n=fft_len)
        conv2 = np.fft.irfft(Fa * Fw, n=fft_len)[: conv_len]
        # pick the slice that corresponds to gradient entries
        idx = (conv_len - 1) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # back‑prop to logits (soft‑max Jacobian)
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.25:
            x += np.random.randn(n) * 0.001

        # decay learning rate gently
        alpha *= 0.9995

    # final evaluation of the best vector found
    final_val = _max_conv(best_a)
    return best_a, final_val


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent with back‑tracking line search
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_step_factor: float = 0.3,
    increase_factor: float = 1.12,
    decrease_factor: float = 0.55,
    min_step: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max(conv) with simple line‑search."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _max_conv(a)
    step = init_step_factor * target_sum / n

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())                # index of current worst peak

        # gradient of conv_k (sub‑gradient of the max) w.r.t. a
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = a[k - js]                  # derivative = a_rev slice

        # back‑tracking line search
        accepted = False
        while step > min_step:
            cand = a - step * grad
            cand = _project_onto_simplex(cand, target_sum)
            cand_val = _max_conv(cand)

            if cand_val < best_val - 1e-12:   # strict improvement
                accepted = True
                break
            else:
                step *= decrease_factor

        if accepted:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= increase_factor
            if it % 150 == 0:
                print(f"[{time.time():.2f}] subgrad it={it} val={best_val:.6f}")
        else:
            # no successful step – inject a small random jitter
            jitter = np.random.rand(n) * 1e-4
            a = _project_onto_simplex(a + jitter, target_sum)

        # early exit if we have already crossed the target bound
        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_temp_factor: float = 0.35,
    final_temp: float = 1e-5,
    max_move_frac: float = 0.08,
) -> tuple[np.ndarray, float]:
    """Annealed hill‑climb moving tiny mass chunks between two positions."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _max_conv(a)

    T0 = max(1e-3, best_val * init_temp_factor)
    start = time.time()

    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(final_temp, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        delta = a[i] * np.random.rand() * max_move_frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _max_conv(a_new)

        accept = False
        if new_val < best_val:
            accept = True
        elif np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
            accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")

        if it % 2500 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid optimiser returning a non‑negative step‑function with minimal
    evaluate_sequence() score.  The routine respects the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.10   # small safety margin

    best_seq = None
    best_score = np.inf   # stores max‑conv (objective after normalisation)

    # ------------------------------------------------------------------
    # Helper to record a candidate (stores the exact max‑conv)
    # ------------------------------------------------------------------
    def _record(seq: np.ndarray):
        nonlocal best_seq, best_score
        score = _max_conv(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] GLOBAL new best={best_score:.6f} n={len(seq)}")
        return score

    # ------------------------------------------------------------------
    # Warm‑start from pre‑computed global variable if available
    # ------------------------------------------------------------------
    try:
        hs = globals().get("height_sequence_1")
        if hs is not None:
            hs_arr = np.asarray(hs, dtype=float)
            n0 = hs_arr.shape[0]
            target0 = np.sqrt(2.0 * n0)
            hs_norm = hs_arr * (target0 / hs_arr.sum())
            _record(hs_norm)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – diverse random initialisations (different lengths)
    # ------------------------------------------------------------------
    # allocate ~30 % of the total budget for this exploratory phase
    phase1_end = deadline - 0.30 * (deadline - start_time)
    while time.time() < phase1_end:
        # choose a length – longer lengths become more probable later
        remaining = deadline - time.time()
        if remaining > 600:
            n = rng.integers(2500, 8000 + 1)
        elif remaining > 300:
            n = rng.integers(2000, 6000 + 1)
        else:
            n = rng.integers(1500, 4000 + 1)

        target_sum = np.sqrt(2.0 * n)
        a0 = _random_initial_sequence(n, rng)
        a0 = a0 * (target_sum / a0.sum())
        _record(a0)

    # ------------------------------------------------------------------
    # Keep the top few candidates for intensive refinement
    # ------------------------------------------------------------------
    # We collect up to 5 best sequences seen so far.
    # (If less than 5 were seen we just use whatever we have.)
    # The collection is built by scanning the global best and
    # re‑evaluating a few recent random candidates.
    top_k = 5
    candidates = []
    if best_seq is not None:
        candidates.append(best_seq.copy())
    # add a few more by random re‑initialisation around the best
    for _ in range(top_k - len(candidates)):
        if best_seq is None:
            break
        n = best_seq.shape[0]
        # perturb the best a bit to obtain a new start
        perturb = rng.normal(scale=0.02, size=n)
        a_new = best_seq + perturb * best_seq.max()
        a_new = np.clip(a_new, 0.0, None)
        if a_new.sum() == 0.0:
            continue
        a_new = a_new * (np.sqrt(2.0 * n) / a_new.sum())
        candidates.append(a_new)

    # ------------------------------------------------------------------
    # Phase 2 – Adam refinement on the log‑sum‑exp surrogate
    # ------------------------------------------------------------------
    phase2_end = deadline - 0.20 * (deadline - start_time)
    for seq in candidates:
        if time.time() >= phase2_end:
            break
        n = seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        logits0 = np.log(seq + 1e-12)   # initialise logits from current vector
        run_deadline = time.time() + min(20.0, phase2_end - time.time() - 0.5)
        refined_seq, refined_val = _adam_softmax(
            n,
            logits0,
            target_sum,
            run_deadline,
            init_tau_factor=0.12,
            min_tau_factor=5e-5,
            alpha0=0.04,
        )
        if refined_val < best_score:
            best_score = refined_val
            best_seq = refined_seq.copy()
            print(f"[{time.time():.2f}] after Adam refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exact sub‑gradient descent with line search
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_deadline = deadline - 15.0
        seq_sub, val_sub = _subgradient_descent(
            best_seq.copy(),
            target_sum,
            sub_deadline,
            init_step_factor=0.25,
            increase_factor=1.10,
            decrease_factor=0.60,
        )
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑gradient best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Final simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        hill_deadline = deadline
        seq_hill, val_hill = _hillclimb(
            best_seq.copy(),
            target_sum,
            hill_deadline,
            init_temp_factor=0.35,
            final_temp=1e-5,
        )
        if val_hill < best_score:
            best_score = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] after hill‑climb best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = _max_conv(best_seq)
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
