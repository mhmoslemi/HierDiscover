"""
Propose a non‑negative step‑function that (empirically) improves the
upper bound of

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The method combines several ideas that proved useful in earlier
experiments and adds a few new ingredients:

* A richer pool of structured initial vectors (edge‑concentrated
  power‑laws, double‑Gaussian edges, sine‑half‑waves, triangles,
  uniform‑plus‑spike mixtures and a possible warm‑start).

* Adam optimisation on a smooth‑max surrogate (log‑sum‑exp) with a
  gradually increasing temperature.  The learning‑rate schedule and
  temperature factor have been tweaked to obtain a steadier descent.

* Exact‑max sub‑gradient optimisation (exponentiated‑gradient
  updates) for a long refinement phase.

* A cheap simulated‑annealing hill‑climber that can escape shallow
  local minima after the deterministic phases.

* If time permits, a linear interpolation to a finer grid (8192
  points) followed by a short re‑optimisation.

All heavy work is performed with FFT‑based convolution
(`_full_convolution`) and the cross‑correlation needed for gradients
(`_cross_correlation`).  The routine respects the supplied time
budget (default 1000 s) and always returns a valid sequence.
"""

import time
import math

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]

def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """Return 2·n·max(a∗a) (sum(a) is assumed = 1)."""
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max()

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns best vector (sum≈1) seen and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                       # a bit larger than before
    lam = 0.5                       # temperature start
    lam_factor = 1.04                # slower increase – smoother descent

    best_a = a0.copy()
    best_val = _objective(best_a, conv_fft_len)

    it = 0
    while it < max_iters and time.time() < deadline:
        # Soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # Exact objective (sum(a)≈1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:05d}] new best {val:.6f}")

        # Smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient of surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # Temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # Small random jitter when we are close to the deadline
        if it % 5000 == 0 and time.time() > deadline - 8:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 200_000,
                          eta: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s

    best_a = a.copy()
    best_val = _objective(best_a, conv_fft_len)

    it = 0
    while it < max_iters and time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it:05d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _annealing_optimize(a0: np.ndarray,
                        n: int,
                        conv_fft_len: int,
                        deadline: float,
                        max_moves: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Cheap simulated‑annealing hill‑climber.
    Proposes random mass transfers and accepts according to a temperature
    schedule that decays from 1 % of the current objective to 0.01 %.
    """
    cur = a0.copy()
    cur_val = _objective(cur, conv_fft_len)

    best = cur.copy()
    best_val = cur_val

    # temperature relative to the current objective
    start_temp = max(cur_val * 0.01, 1e-4)
    end_temp = max(cur_val * 0.0001, 1e-6)

    for move in range(max_moves):
        if time.time() >= deadline:
            break

        # pick source and destination
        i = np.random.randint(n)
        if cur[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # move a random fraction (up to 40 %) of the source mass
        delta = cur[i] * (np.random.rand() * 0.4)
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        val = _objective(cand, conv_fft_len)
        delta_val = val - cur_val

        # linear temperature decay
        t = start_temp + (end_temp - start_temp) * (move / max_moves)

        if delta_val < 0 or np.random.rand() < math.exp(-delta_val / (t + 1e-15)):
            cur = cand
            cur_val = val
            if val < best_val:
                best_val = val
                best = cand.copy()
                if move % 3000 == 0:
                    print(f"[Anneal move={move}] new best {val:.6f}")

    return best, best_val


# --------------------------------------------------------------
# Upscaling utility
# --------------------------------------------------------------
def _upscale(a: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation from len(a) to n_big, followed by renormalisation."""
    n_old = a.shape[0]
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_big, endpoint=False)
    big = np.interp(x_new, x_old, a)
    big = np.maximum(big, 0.0)
    big /= big.sum()
    return big

# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function (non‑negative vector) with a low
    evaluation score.  The routine respects the given time budget
    and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Base problem size (the previous state‑of‑the‑art used 4096)
    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if a previous solution is present)
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.ndim != 1:
            warm = None
        else:
            if warm.shape[0] > n:
                warm = warm[:n]
            elif warm.shape[0] < n:
                warm = np.pad(warm, (0, n - warm.shape[0]))
            warm = np.maximum(warm, 0.0)
            if warm.sum() >= 0.01:
                warm = warm / warm.sum()
            else:
                warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a pool of diverse initial candidates
    # ------------------------------------------------------------------
    candidates = _initial_pool(n, warm)

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optimise each candidate (short runs) – keep the global best
    # ------------------------------------------------------------------
    remaining = len(candidates)
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break

        # Normalise candidate onto the simplex
        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / a0.sum()

        # Allocate a slice of the remaining budget for this candidate
        slice_deadline = min(deadline - 5.0,
                             time.time() + max(5.0, (deadline - time.time()) / (remaining + 1)))
        remaining -= 1

        # --------------------------------------------------------------
        # Adam phase (smooth‑max)
        # --------------------------------------------------------------
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, slice_deadline, max_iters=120_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Cand {cid}] Adam best {val_adam:.6f}")

        # --------------------------------------------------------------
        # Brief sub‑gradient refinement
        # --------------------------------------------------------------
        if time.time() < deadline:
            a_sub, val_sub = _subgradient_optimize(
                best_seq, n, conv_fft_len, corr_fft_len, deadline, max_iters=40_000
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Cand {cid}] Subgrad best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Long sub‑gradient refinement on the overall best candidate
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        a_ref, val_ref = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=250_000,
            eta=0.6,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] Subgrad long best {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing local search (cheap but often effective)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        a_ann, val_ann = _annealing_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=120_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved best {val_ann:.6f}")

    # ------------------------------------------------------------------
    # If time permits, upscale to a finer grid and repeat a short
    # optimisation cycle – this can capture higher‑frequency structure.
    # ------------------------------------------------------------------
    if time.time() + 20 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            a_big = _upscale(best_seq, n_big)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        # Adam on the larger grid (short run)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=80_000,
        )
        # Sub‑gradient refinement
        a_big_ref, val_big_ref = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=80_000,
            eta=0.6,
        )
        if val_big_ref < best_val:
            best_val = val_big_ref
            best_seq = a_big_ref.copy()
            print(f"[Big] refined best {val_big_ref:.6f}")

    # ------------------------------------------------------------------
    # Fallback – ensure something valid is returned
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clamp to the evaluator's accepted range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
