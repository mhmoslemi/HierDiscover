def propose_candidate(seed=0, budget_s=1000):

    np.random.seed(seed)

    start = time.time()

    deadline = start + budget_s - 0.5

    best_seq = None

    best_val = float('inf')

    # lengths list maybe incremental.

    lengths = [1024, 2048, 4096, 8192] # can adjust based on time.

    for n in lengths:

        if time.time() > deadline: break

        # compute FFT length as next pow2 of 2*n.

        conv_len = 1 << int(np.ceil(np.log2(2*n -1)))

        corr_len = 1 << int(np.ceil(np.log2(2*n -1 + n -1)))

        # generate candidates

        candidates = generate_candidates(n)

        for idx, a0 in enumerate(candidates):
            if time.time() > deadline: break

            a = a0.copy()
            # normalization sum to 1
            s = a.sum()
            if s <= 0: continue
            a = a / s

            # Use optimizer: first Adam

            a_opt, val = run_optimizer(a, n, conv_len, corr_len, deadline)

            if val < best_val:
                best_val = val
                best_seq = a_opt.copy()
                print(f"New best: {val:.6f} at n={n} cand={idx}")

    # final fallback uniform if nothing.

    if best_seq is None:
        best_seq = np.full(1024, 1/1024.0)

    return best_seq.tolist()

def run_optimizer(a, n, conv_len, corr_len, deadline):
    # Phase1: Adam
    logits = np.log(a + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1 * np.sqrt(2048.0 / n)  # may adjust

    lam = 0.5
    lam_factor = 1.015 # slower

    max_iters = int(20000 * 1024 / n) + 2000

    best_a = a.copy()
    best_val = float('inf')
    last_improve = time.time()

    for t in range(1, max_iters+1):
        # compute softmax
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        conv = _conv_fft(a, conv_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _corr_fft(w, a, corr_len)

        # backprop through softmax
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        lam = min(lam * lam_factor, 1e6)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # if stagnation
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    # Phase2: exponentiated gradient on exact max
    a = best_a.copy()
    eta = 0.5
    sub_iters = int(3000 * 1024 / n) + 1000
    last_improve = time.time()
    for it in range(sub_iters):
        conv = _conv_fft(a, conv_len)
        max_c = conv.max()
        # w indicator for max indices
        active = np.isclose(conv, max_c, atol=1e-12)  # maybe >0
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _corr_fft(w, a, corr_len)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s <= 0:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        if time.time() - last_improve > 8.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val

"""
Simple yet powerful search for a low‑value step‑function.

The algorithm mixes a large, diverse set of *parameterised* initial shapes
(edge‑heavy, asymmetric beta‑laws, mixtures of Gaussians, random sparse
patterns, …), then refines every candidate with two fast optimisation
phases:

1. Adam on a smooth‑max surrogate of the true objective.
2. An exact sub‑gradient (exponentiated‑gradient) phase focused on the
   active convolution indices.

Both phases run entirely on the frequency domain (FFT‑based
convolutions) and respect a global time budget (default 1000 s).  The
best sequence found (as a plain Python list) is returned.

The code is deliberately **stateless** – all helpers are top‑level
functions, no closures or lambdas – and can be executed with at most two
CPU cores.  It uses only the standard library and NumPy (which is
already available in the evaluation environment).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (a*a) and the corresponding cross‑correlation
# needed for gradients.
def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact objective value from a convolution array."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _adam_phase(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    max_iters: int = 10000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smooth‑max surrogate of the true max.
    Returns the best vector seen and its corresponding objective.
    """
    a = a0.copy()
    # work in log‑soft‑max (logits) so the simplex is automatically respected
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    # learning‑rate scales mildly with problem size
    lr = 0.09 * np.sqrt(2048.0 / n)

    # temperature (smooth‑max) schedule
    lam = 0.4
    lam_factor = 1.016

    best_a = a.copy()
    best_val = float("inf")
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # ----------------------------------- soft‑max (simplex)
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ----------------------------------- smooth‑max of the convolution
        conv = _conv_fft(a, conv_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----------------------------------- gradient of the surrogate
        grad_a = 2.0 * _corr_fft(w, a, corr_len)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature growth
        lam = min(lam * lam_factor, 1e6)

        # current objective (exact, because sum(a)==1)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # occasional gentle shake if we are stuck
        if time.time() - last_improve > 7.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _subgradient_phase(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Exact max‑subgradient (exponentiated‑gradient) phase.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    last_improve = time.time()
    eta = 0.55  # step‑size for exponentiated gradient

    for it in range(max_iters):
        conv = _conv_fft(a, conv_len)
        max_c = conv.max()

        # weights that pick the active (max) constraints
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _corr_fft(w, a, corr_len)

        # exponentiated update on the simplex
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # early abort if no progress for a while
        if time.time() - last_improve > 8.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _refine_sequence(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    One extra fine‑tuning pass (Adam → sub‑gradient) for the current best.
    """
    # Adam with a tighter schedule
    a_mid, val_mid = _adam_phase(
        a0,
        n,
        conv_len,
        corr_len,
        deadline,
        max_iters=int(15000 * 1024 / n) + 3000,
    )
    # Sub‑gradient to polish the final max
    a_fin, val_fin = _subgradient_phase(
        a_mid,
        n,
        conv_len,
        corr_len,
        deadline,
        max_iters=int(3000 * 1024 / n) + 1000,
    )
    # Return whichever is better
    if val_mid < val_fin:
        return a_mid, val_mid
    else:
        return a_fin, val_fin

def _build_initial_candidates(n: int) -> list[np.ndarray]:
    """
    Produce a diversified list of starting points.
    All candidates are normalised to sum = 1.
    """
    cand = []

    # 1) warm‑start from a previously stored solution (if any)
    global height_sequence_1
    if "height_sequence_1" in globals():
        h = np.asarray(height_sequence_1, dtype=np.float64)
        if h.shape[0] > n:
            h = h[:n]
        else:
            h = np.pad(h, (0, n - h.shape[0]))
        h = np.maximum(h, 0.0)
        if h.sum() > 1e-8:
            cand.append(h / h.sum())

    # 2) pure Dirichlet (conc = 1)
    cand.append(np.random.dirichlet(np.ones(n)))

    # 3) spiky Dirichlet (conc < 1)
    for conc in (0.2, 0.4, 0.6, 0.8):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 4) over‑dispersed Dirichlet (conc > 1)
    for conc in (1.5, 2.0, 3.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 5) asymmetric beta‑type power‑law: a_i ∝ (i+1)^α·(n-i)^β
    for _ in range(8):
        α = np.random.uniform(-3.0, 1.0)
        β = np.random.uniform(-3.0, 1.0)
        i = np.arange(n, dtype=np.float64)
        shape = np.power(i + 1, α, where=True) * np.power(n - i, β, where=True)
        shape = np.where(np.isfinite(shape), shape, 0.0)
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # 6) one‑sided exponential decays (different rates)
    for γ in (0.001, 0.003, 0.007, 0.015):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-γ * i)
        cand.append(shape / shape.sum())

    # 7) asymmetric Gaussian bumps
    for _ in range(6):
        μ = np.random.uniform(0, n - 1)
        σ = np.random.uniform(n / 30.0, n / 6.0)
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-0.5 * ((i - μ) / σ) ** 2)
        cand.append(shape / shape.sum())

    # 8) mixture of two edge‑centred Gaussians
    for _ in range(4):
        σ = np.random.uniform(n / 25.0, n / 10.0)
        i = np.arange(n, dtype=np.float64)
        left = np.exp(-0.5 * (i / σ) ** 2)
        right = np.exp(-0.5 * ((i - (n - 1)) / σ) ** 2)
        mix = 0.5 * left + 0.5 * right
        cand.append(mix / mix.sum())

    # 9) random sparse patterns (≈ √n spikes)
    m_spikes = max(2, int(np.sqrt(2 * n)))
    for _ in range(3):
        sparse = np.zeros(n, dtype=np.float64)
        idx = np.random.choice(n, size=m_spikes, replace=False)
        sparse[idx] = 1.0
        cand.append(sparse / sparse.sum())

    # 10) deterministic families already used in the previous code
    i = np.arange(n, dtype=np.float64)

    # tent
    left = np.linspace(0, 1, n // 2, endpoint=False)
    right = left[::-1]
    tent = np.concatenate([left, right])
    if tent.shape[0] < n:
        tent = np.concatenate([tent, [0.0]])
    tent = tent / tent.sum()
    cand.append(tent)

    # cosine window
    coswin = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    caset = coswin / coswin.sum()
    cand.append(caset)

    # half‑sine window
    sinwin = np.sin(np.pi * i / (n - 1))
    sinset = sinwin / sinwin.sum()
    cand.append(sinset)

    # discrete Gaussian (σ ~ n/8)
    σ = n / 8.0
    gauss = np.exp(-0.5 * ((i - (n - 1) / 2) / σ) ** 2)
    gauss = gauss / gauss.sum()
    cand.append(gauss)

    # truncated arcsine (inverse‑sqrt edges)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcs = arcs / arcs.sum()
    cand.append(arcs)

    # keep only valid, normalised candidates
    filtered = []
    for c in cand:
        if np.all(np.isfinite(c)) and np.all(c >= 0) and c.sum() > 0:
            filtered.append(c / c.sum())
    return filtered


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    best_seq = None
    best_val = float("inf")

    # Lengths – start with moderate sizes, go larger if time permits
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        # FFT lengths (next power of two)
        conv_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_len = 1 << int(np.ceil(np.log2(2 * n - 1 + n - 1)))

        # ------------------------------------------------------------------
        # Build a rich pool of initial candidates
        candidates = _build_initial_candidates(n)

        # ------------------------------------------------------------------
        # Optimise every candidate (Adam → sub‑gradient)
        for cid, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # Normalise to a probability vector (sum = 1)
            a0 = a0 / a0.sum()
            # ------- first phase: Adam on smooth‑max --------------------
            a_mid, val_mid = _adam_phase(
                a0, n, conv_len, corr_len, deadline,
                max_iters=int(40000 * 1024 / n) + 3000,
            )
            # ------- second phase: exact sub‑gradient --------------------
            a_opt, val_opt = _subgradient_phase(
                a_mid, n, conv_len, corr_len, deadline,
                max_iters=int(5000 * 1024 / n) + 1500,
            )

            # Keep the best from the two phases
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[n={n} cand={cid}] new best {best_val:.6f}")

        # ------------------------------------------------------------------
        # Fine‑tune the globally best sequence for this length (if any)
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] fine‑tuning the global best (val={best_val:.6f})")
            a_fine, val_fine = _refine_sequence(
                best_seq, n, conv_len, corr_len, deadline,
            )
            if val_fine < best_val:
                best_val = val_fine
                best_seq = a_fine.copy()
                print(f"[n={n}] after fine‑tune new best {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – a simple uniform vector (should never be needed).
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Ensure the result is a plain list as required by the external evaluator.
    return best_seq.tolist()
