"""
Propose a low‑objective step‑function.

The search proceeds in three phases:

1. **Rich candidate generation** – a large pool of diverse
   parametric shapes (edge‑exponential, edge‑power‑law,
   two‑edge Gaussians, block‑edge mixtures, random sparse vectors,
   Dirichlet samples, warm‑starts) is built and quickly evaluated
   with an FFT based convolution.  The best few (≈12) survive.

2. **Deep optimisation** – each survivor is refined by
   *Adam on a smooth‑max surrogate* (log‑sum‑exp) while the vector
   stays on the simplex via a soft‑max parametrisation, followed by
   an *exponentiated‑gradient sub‑gradient* phase that works with the
   true max‑convolution active set.  Hyper‑parameters (learning‑rate,
   temperature schedule, sub‑gradient step‑size) are varied between
   runs to escape local minima.

3. **Upsampling (optional)** – if time permits the best 4096‑vector
   is interpolated to a 8192‑grid and a short optimisation pass is
   performed there.

All operations respect the 1000 s budget; the routine prints the
current best objective and finally returns the best sequence as a
plain Python list (clipped to the evaluator’s allowed range).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the objective (sum(a) is assumed ≈ 1)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


def _softmax(logits: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping logits → probability simplex."""
    mx = logits.max()
    exp_shift = np.exp(logits - mx)
    return exp_shift / exp_shift.sum()


def _adam_optimize(
    logits0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
    *,
    lr: float = 0.05,
    init_temp: float = 0.01,
    temp_factor: float = 1.02,
    max_iter: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector/value observed.
    """
    eps = 1e-15
    logits = logits0.copy()
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = init_temp
    lam_cap = 1e6
    best_seq, best_val = global_best_seq, global_best_val

    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        # simplex projection via soft‑max
        a = _softmax(logits)

        # convolution, true objective (used for reporting)
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_len)
        # chain‑rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * temp_factor, lam_cap)

        # shake if stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
    *,
    eta: float = 0.6,
    max_iter: int = 20_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient refinement on the exact max‑convolution.
    """
    a = np.maximum(a0, 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum

    best_seq, best_val = global_best_seq, global_best_val
    last_improve = time.time()

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} iter={it:05d}] new best {val:.6f}")

        # active set (indices achieving the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 5.0:
            break

    return best_seq, best_val


def _optimise_start(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam → sub‑gradient on a single start vector.
    Returns the (possibly) updated global best.
    """
    # Adam phase (log‑softmax parametrisation)
    logits0 = np.log(a0 + 1e-15)
    best_seq, best_val = _adam_optimize(
        logits0,
        n,
        conv_len,
        corr_len,
        deadline,
        global_best_seq,
        global_best_val,
        # a few random‑ish hyper‑parameter draws to enhance diversity
        lr=0.04 + 0.02 * np.random.rand(),
        init_temp=0.005 + 0.02 * np.random.rand(),
        temp_factor=1.018 + 0.004 * np.random.rand(),
    )
    # Sub‑gradient phase
    best_seq, best_val = _subgradient_optimize(
        best_seq if best_seq is not None else a0,
        n,
        conv_len,
        corr_len,
        deadline,
        best_seq,
        best_val,
        eta=0.5 + 0.2 * np.random.rand(),
    )
    return best_seq, best_val


def _generate_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Create a large, diverse pool of plausible shapes."""
    candidates = []

    # --- warm‑start and rotations ------------------------------------------------
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # --- trivial baseline -------------------------------------------------------
    candidates.append(np.full(n, 1.0 / n))

    # --- random Dirichlet vectors (various concentrations) -----------------------
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # --- edge‑exponential families (symmetric) ---------------------------------
    i = np.arange(n, dtype=np.float64)
    for alpha in np.logspace(-2, 1.0, num=12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # --- asymmetric edge‑exponential mixtures ------------------------------------
    for _ in range(12):
        al = 10 ** np.random.uniform(-2, 1)
        ar = 10 ** np.random.uniform(-2, 1)
        edge = np.exp(-al * i) + np.exp(-ar * (n - 1 - i))
        edge /= edge.sum()
        w = np.random.rand()
        shape = w * edge + (1.0 - w) / n
        candidates.append(shape)

    # --- edge‑power‑law families -------------------------------------------------
    for p in np.linspace(0.2, 5.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.5, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # --- two‑Gaussian edge shapes -----------------------------------------------
    for gamma in (0.05, 0.1, 0.2, 0.4, 0.8, 1.5):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # --- block‑edge (constant plateaus at both ends) ----------------------------
    for frac in np.linspace(0.02, 0.25, num=6):
        w = int(frac * n)
        v = np.zeros(n)
        v[:w] = 1.0
        v[-w:] = 1.0
        if v.sum() > 0:
            v /= v.sum()
            candidates.append(v)

    # --- random sparse vectors ---------------------------------------------------
    for _ in range(20):
        M = np.random.randint(32, 1024)          # number of non‑zeros
        idx = np.random.choice(n, M, replace=False)
        vals = np.random.rand(M) + 0.1           # avoid tiny entries
        v = np.zeros(n)
        v[idx] = vals
        v /= v.sum()
        candidates.append(v)

    # --- sinusoidal half‑wave (already used before) -----------------------------
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    return candidates


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point.
    Searches for a step‑function that minimises
        F(a) = 2·n·max(a∗a) / (sum(a)²)
    under the evaluator's safety checks.
    Returns the best sequence found (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ----------------------------------------------------------------------
    # Work initially with n = 4096 (a convenient size for FFTs)
    n = 4096
    conv_len = _next_pow2(2 * n - 1)
    corr_len = _next_pow2(3 * n - 2)

    # ----------------------------------------------------------------------
    # Retrieve a possible warm‑start from previous runs
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None

    # ----------------------------------------------------------------------
    # 1) Fast generation & pre‑selection of diverse candidates
    all_candidates = _generate_candidates(n, warm_start)

    cand_vals = []
    for idx, cand in enumerate(all_candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:                     # degenerate – replace
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_len)
        cand_vals.append((val, cand))

    # keep the most promising dozen
    cand_vals.sort(key=lambda x: x[0])
    top_k = 12
    top_candidates = [c for _, c in cand_vals[:top_k]]

    # ----------------------------------------------------------------------
    # 2) Deep optimisation on the selected starts
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation on start #{start_id + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _optimise_start(
            a0,
            n,
            conv_len,
            corr_len,
            deadline,
            best_seq,
            best_val,
        )

    # ----------------------------------------------------------------------
    # 3) If time permits, try additional random restarts (shorter budget)
    while time.time() < deadline - 10.0:   # keep a few seconds buffer
        a0 = np.random.dirichlet(np.ones(n))
        best_seq, best_val = _optimise_start(
            a0,
            n,
            conv_len,
            corr_len,
            deadline,
            best_seq,
            best_val,
        )
        # occasional break to avoid endless loop
        if best_val < 1.50:            # already excellent – stop early
            break

    # ----------------------------------------------------------------------
    # 4) Optional up‑sampling to n = 8192 (if enough time left)
    time_left = deadline - time.time()
    if time_left > 30.0 and best_seq is not None:
        n_big = 8192
        conv_len_big = _next_pow2(2 * n_big - 1)
        corr_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best vector
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        if a_big.sum() < 0.01:
            a_big = np.random.dirichlet(np.ones(n_big))
        a_big = a_big / a_big.sum()

        print("--- Upsampling optimisation (n=8192) ---")
        best_seq, best_val = _optimise_start(
            a_big,
            n_big,
            conv_len_big,
            corr_len_big,
            deadline,
            best_seq,
            best_val,
        )
        n = n_big

    # ----------------------------------------------------------------------
    # Fallback – if nothing succeeded, return a uniform vector.
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final sanity check (optional – can be removed to save time)
    # val = evaluate_sequence(best_seq.tolist())
    # print(f"Final reported objective = {val:.6f}")

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
