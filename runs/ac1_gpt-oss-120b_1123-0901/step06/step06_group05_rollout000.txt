"""
Propose a high‑quality step‑function (non‑negative heights) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)²)

The search runs within a given time budget (default 1000 s) and returns the
best sequence found as a plain Python list.

Algorithm
---------
1. **Multi‑scale exploration** – lengths 4096, 8192, 16384, 32768 are tried
   (larger n gives more expressive power).  For each length a diverse pool of
   seed shapes is built:
   * warm‑start from the current global best (upsampled)
   * historic solution (global variable ``height_sequence_1``)
   * several random Dirichlet draws (different concentration parameters)
   * random “block‑wise” shapes (few constant blocks)
   * classic analytic windows (power‑law, symmetric exponential,
     Hann, half‑sine, Gaussian)
2. **Smooth‑max Adam optimisation** – a soft‑max surrogate of the max is
   minimised on the probability simplex using Adam.  Several hyper‑parameter
   schedules (learning‑rate, temperature start/increase) are tried for each
   seed.
3. **Exact sub‑gradient refinement** – exponentiated‑gradient updates on the
   true max‑convolution are applied to the Adam result.
4. **Simulated annealing** – a cheap Monte‑Carlo search moves a small amount
   of mass between two indices and accepts worsening moves with a decaying
   temperature.
5. **Final stochastic hill‑climb** – cheap local moves (ε‑transfer
   between two positions) are performed until no improvement for a while.
6. The best sequence (normalised to sum = 1) is returned.

All heavy work uses FFT‑based convolutions, which are O(n log n) and
fast enough for the allocated budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution of a with itself (a*a) using FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_corr(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]
    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

# Candidate generation

def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative vector to ``new_n`` points by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:                     # protect against a zero vector
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s


def _random_block_shape(n: int, blocks: int = 8) -> np.ndarray:
    """
    Random piecewise‑constant shape with ``blocks`` constant sections.
    The total mass is 1.
    """
    # Randomly allocate how many indices each block gets
    sizes = np.random.multinomial(n, np.full(blocks, 1 / blocks))
    # Random masses for each block (Dirichlet)
    masses = np.random.dirichlet(np.ones(blocks))
    seq = np.empty(n, dtype=np.float64)
    pos = 0
    for sz, m in zip(sizes, masses):
        if sz == 0:
            continue
        seq[pos: pos + sz] = m / sz
        pos += sz
    # Fill any residual zeros (unlikely) with tiny uniform mass
    if pos < n:
        seq[pos:] = 1e-12
    return seq / seq.sum()


def _generate_initial_candidates(n: int, prev_best):
    """
    Produce a heterogeneous pool of starting points for a given length ``n``.
    ``prev_best`` (if not ``None``) is up‑sampled and used as a warm‑start.
    Returns a list of probability vectors (np.ndarray) that sum to 1.
    """
    candidates = []

    # 1) warm‑start from the current global best
    if prev_best is not None:
        candidates.append(_upsample(prev_best, n))

    # 2) historic solution (if present)
    hseq = globals().get('height_sequence_1')
    if isinstance(hseq, (list, np.ndarray)):
        h = np.asarray(hseq, dtype=np.float64)
        candidates.append(_upsample(h, n))

    # 3) random Dirichlet draws (different concentration levels)
    candidates.append(np.random.dirichlet(np.ones(n)))
    for conc in (0.2, 0.5, 0.8):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # 4) random block‑wise shapes
    for blocks in (4, 8, 12):
        candidates.append(_random_block_shape(n, blocks))

    # 5) edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.1, 7):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # 6) symmetric two‑sided exponentials
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # 7) Hann window (raised‑cosine)
    i = np.arange(n, dtype=np.float64)
    hann = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    hann /= hann.sum()
    candidates.append(hann)

    # 8) half‑sine shape
    sine = np.sin(np.pi * i / (n - 1))
    sine = np.maximum(sine, 0)
    sine /= sine.sum()
    candidates.append(sine)

    # 9) centred Gaussian‑like bumps (several widths)
    mu = (n - 1) / 2.0
    for sigma in (n / 10, n / 6, n / 4):
        shape = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        shape /= shape.sum()
        candidates.append(shape)

    # 10) very sparse spikes (few non‑zero entries)
    for k in (2, 3, 5):
        sparse = np.zeros(n, dtype=np.float64)
        idxs = np.random.choice(n, k, replace=False)
        vals = np.random.rand(k)
        sparse[idxs] = vals
        if sparse.sum() > 0:
            sparse /= sparse.sum()
            candidates.append(sparse)

    # Clean up: ensure non‑negative, normalised, non‑degenerate
    clean = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        clean.append(c / s)
    return clean

# Adam optimiser on smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int,
                   lr: float = 0.05,
                   lam_start: float = 0.1,
                   lam_factor: float = 1.02) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max surrogate of the objective.
    Returns (best_vector, best_value) where value is the exact
    evaluate_sequence objective (i.e. 2·n·max(conv)).
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # log‑softmax parametrisation
    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = None
    best_val = float('inf')
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # map logits → probability simplex (softmax)
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # exact objective (since sum(a)=1, denominator disappears)
        conv = _conv_self(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t}] new best {val:.6f}")

        # smooth‑max weights (soft‑max on the convolution entries)
        w = np.exp(lam * (conv - max_c))          # shifts by max_c for stability
        w_sum = w.sum()
        if w_sum != 0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.size)

        # gradient of surrogate wrt a
        grad_a = 2.0 * _cross_corr(w, a, corr_fft_len)

        # back‑prop through soft‑max (logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # schedule temperature
        lam = min(lam * lam_factor, 1e4)

        # occasional shake if stuck
        if time.time() - last_improve > 8.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    if best_a is None:                # fallback (should never happen)
        best_a = a0.copy()
        best_val = 2.0 * n * _conv_self(best_a, conv_fft_len).max()
    return best_a, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient sub‑gradient phase (exact max)
# ----------------------------------------------------------------------
def _subgrad_optimize(a0: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int,
                      eta0: float = 0.55) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient optimisation on the true max‑convolution.
    Returns (best_vector, best_value).
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_self(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it}] new best {val:.6f}")

        # active set (where convolution attains the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_corr(w, a, corr_fft_len)

        # decaying step size (classical EG schedule)
        eta = eta0 / np.sqrt(it)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:        # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:   # stagnation ⇒ stop early
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing on the exact objective
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         max_steps: int = 12000,
                         init_temp: float = 0.02,
                         decay: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Simple Monte‑Carlo annealing: move a small amount of mass ε from i to j.
    Accept worsening moves with probability exp(-(Δ)/T).  Returns the best
    sequence encountered.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    a = a0.copy()
    conv = _conv_self(a, conv_fft_len)
    best_a = a.copy()
    best_val = 2.0 * n * conv.max()
    T = init_temp
    steps = 0

    while time.time() < deadline and steps < max_steps:
        i, j = np.random.choice(n, 2, replace=False)
        # epsilon proportional to the source mass (max 5 % of its current value)
        eps = np.random.uniform(0.0, min(0.05, a[i]))
        if eps <= 0.0:
            steps += 1
            continue

        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps

        # recompute convolution exactly (still cheap thanks to FFT)
        conv_cand = _conv_self(cand, conv_fft_len)
        val_cand = 2.0 * n * conv_cand.max()

        delta = val_cand - best_val
        if delta < 0 or np.exp(-delta / max(T, 1e-12)) > np.random.rand():
            # accept
            a = cand
            conv = conv_cand
            if val_cand < best_val:
                best_val = val_cand
                best_a = cand.copy()
                # optional progress report
                print(f"[n={n} SA step={steps}] improved to {best_val:.6f}")
                # reset temperature a bit when we improve
                T = init_temp
        else:
            # reject – keep current a, conv unchanged
            pass

        T *= decay
        steps += 1

        # occasional temperature boost if we get stuck too long
        if steps % 2000 == 0 and time.time() - deadline > -5.0:
            T = init_temp

    return best_a, best_val


# ----------------------------------------------------------------------
# Cheap stochastic hill‑climb (pairwise mass transfers)
# ----------------------------------------------------------------------
def _local_hill_climb(a0: np.ndarray, n: int, deadline: float,
                     max_trials: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climb.
    Repeatedly moves a small mass ε from one index to another and keeps the
    move if it reduces the exact objective.
    """
    conv_fft_len = _next_pow2(2 * n - 1)

    best_a = a0.copy()
    conv = _conv_self(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    no_improve = 0
    trials = 0
    while time.time() < deadline and trials < max_trials and no_improve < 5_000:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0.0, min(0.05, best_a[i]))
        if eps == 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        conv_cand = _conv_self(cand, conv_fft_len)
        val_cand = 2.0 * n * conv_cand.max()
        if val_cand < best_val:
            best_val = val_cand
            best_a = cand
            no_improve = 0
            print(f"[n={n} HillClimb] improved to {val_cand:.6f}")
        else:
            no_improve += 1
        trials += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the provided evaluation function within the given time budget.
    The returned list is a plain Python list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None          # best probability vector (sum == 1)
    best_val = float('inf')

    # Length schedule – increasing expressive power
    lengths = [4096, 8192, 16384, 32768]

    # Hyper‑parameter families for Adam (lr, lam_start, lam_factor, max_iters)
    adam_params = [
        (0.05, 0.10, 1.015, 22000),
        (0.05, 0.30, 1.020, 20000),
        (0.03, 0.10, 1.010, 25000),
        (0.07, 0.10, 1.020, 18000),
    ]

    # Sub‑gradient (exact max) iteration budget – scaled with n
    subgrad_iters_factor = 0.0015   # approx 0.0015·n iterations (capped later)

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # ------------------------------------------------------------------
        # 1) generate diverse seeds
        # ------------------------------------------------------------------
        candidates = _generate_initial_candidates(n, best_seq)

        # ------------------------------------------------------------------
        # 2) For each seed, run several Adam variants → sub‑gradient → SA
        # ------------------------------------------------------------------
        for cid, init in enumerate(candidates):
            if time.time() > deadline:
                break

            # ---- Adam phase (multiple hyper‑parameter sets) ----
            for (lr, lam_start, lam_factor, max_iters) in adam_params:
                if time.time() > deadline:
                    break
                a_adam, val_adam = _adam_optimize(init, n, deadline,
                                                  max_iters,
                                                  lr=lr,
                                                  lam_start=lam_start,
                                                  lam_factor=lam_factor)
                if val_adam < best_val:
                    best_val = val_adam
                    best_seq = a_adam.copy()
                    print(f"[n={n} cand={cid}] Adam new global best {val_adam:.6f}")

                # ---- Sub‑gradient refinement (exact max) ----
                max_subgrad = int(max(3000, subgrad_iters_factor * n))
                a_sub, val_sub = _subgrad_optimize(a_adam, n, deadline, max_subgrad)
                if val_sub < best_val:
                    best_val = val_sub
                    best_seq = a_sub.copy()
                    print(f"[n={n} cand={cid}] Subgrad new global best {val_sub:.6f}")

                # ---- Simulated annealing (cheap) ----
                a_sa, val_sa = _simulated_annealing(a_sub, n, deadline,
                                                    max_steps=8000,
                                                    init_temp=0.02,
                                                    decay=0.9995)
                if val_sa < best_val:
                    best_val = val_sa
                    best_seq = a_sa.copy()
                    print(f"[n={n} cand={cid}] SA new global best {val_sa:.6f}")

        # ------------------------------------------------------------------
        # 3) Global fine‑tune on the current length (if we have a candidate)
        # ------------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            a_fine, val_fine = _adam_optimize(best_seq, n, deadline,
                                              max_iters=int(0.5 * max_iters),
                                              lr=0.04,
                                              lam_start=0.12,
                                              lam_factor=1.015)
            if val_fine < best_val:
                best_val = val_fine
                best_seq = a_fine.copy()
                print(f"[n={n}] Fine‑tune improved global best to {val_fine:.6f}")

    # ----------------------------------------------------------------------
    # Final cheap hill‑climb to squeeze any remaining gain
    # ----------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print(f"\n=== Final local hill‑climb (n={len(best_seq)}) ===")
        a_final, val_final = _local_hill_climb(best_seq, len(best_seq), deadline)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final
            print(f"[final] hill‑climb improved global best to {val_final:.6f}")

    # ----------------------------------------------------------------------
    # Safety fall‑back (should never trigger)
    # ----------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 4096
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Convert to plain Python list as required by the external evaluator
    return best_seq.tolist()
