"""
Propose a step‑function (non‑negative vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

Algorithm
---------
* Progressive refinement: start on a modest size (≈1 k), find a good
  shape, then interpolate to larger grids (2 k, 4 k, 8 k) and continue
  optimisation.  This gives the optimiser many degrees of freedom while
  keeping the total runtime within the 1000 s budget.
* Diverse warm‑starts: a mixture of deterministic edge‑concentrated
  families (exponential, uniform blocks, double‑Gaussian) and random
  Dirichlet draws.  The warm‑start from the previous run (global variable
  ``height_sequence_1``) is also used.
* Adam optimisation on a smooth‑max surrogate (log‑sum‑exp).  The
  temperature (inverse “lam”) is increased slowly so the surrogate
  approaches the true maximum only after many iterations.
* Exact‑max sub‑gradient phase (exponentiated‑gradient) that refines the
  solution on the original objective.
* A cheap random‑local‑search that swaps a fraction of mass between two
  coordinates whenever it improves the objective.
* All convolutions are performed with real FFTs (O(n log n)).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a):
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _normalize(v: np.ndarray) -> np.ndarray:
    """Clamp to ≥0, enforce sum=1 (if sum too small, replace by Dirichlet)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 0.01:
        v = np.random.dirichlet(np.ones(v.size))
        s = v.sum()
    return v / s


def _candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest pool of diverse start vectors for a given size."""
    cand = []

    # warm‑start (and a few rotations) if supplied
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # edge‑uniform blocks of different widths
    for width in (32, 64, 128, 256, 512, 1024):
        a = np.zeros(n, dtype=np.float64)
        if width * 2 <= n:
            a[:width] = 0.5 / width
            a[-width:] = 0.5 / width
        cand.append(a)

    # exponential edges
    for rate in (0.001, 0.003, 0.007, 0.015, 0.03, 0.06):
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-rate * x) + np.exp(-rate * (n - 1 - x))
        cand.append(a)

    # double Gaussian (centered at each side)
    x = np.arange(n, dtype=np.float64)
    for sigma in (0.05, 0.10, 0.20, 0.40):
        a = np.exp(-0.5 * ((x / (sigma * n)) ** 2))
        a += np.exp(-0.5 * (((n - 1 - x) / (sigma * n)) ** 2))
        cand.append(a)

    # half‑sine (positive half‑wave)
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a = np.maximum(a, 0.0)
    cand.append(a)

    # linear ramp (triangular) – peaks in the centre (included for completeness)
    i = np.arange(n, dtype=np.float64)
    a = np.minimum(i + 1, n - i)
    cand.append(a)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best simplex‑vector encountered (sum≈1) and its exact value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # parametrisation on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = 0.5
    lam_factor = 1.015                 # slower temperature schedule

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # ----- soft‑max → simplex -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- exact objective (max‑convolution) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        w = np.exp(lam * (conv - max_c))      # shift by max_c for stability
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)   # ∇_a S
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)                           # ∇_logits S

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule (focus more on true max) -----
        lam = min(lam * lam_factor, 1e6)

        # occasional tiny random perturbation to escape plateaus
        if it % 5000 == 0 and time.time() - deadline > -5:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    eta: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase with exponentiated‑gradient updates.
    Returns best vector seen and its exact objective.
    """
    a = _normalize(a0)
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad n={n} it={it:06d}] new best {val:.6f}")

        # active set – indices attaining the maximum
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient step (multiplicative)
        a = a * np.exp(-eta * grad_a)

        # project back onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
        else:
            a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one index
    to another; keep the move if it improves the exact objective.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # transfer up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 3000 == 0:
                print(f"[Local n={n} move={it}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence it discovered (as a Python list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a tiny safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Progressive grid sizes – start small, finish on a fine grid
    grid_sizes = [1024, 2048, 4096, 8192]   # 8192 is the largest we attempt

    # warm‑start from a previous run (if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        warm = np.maximum(warm, 0.0)

    for n in grid_sizes:
        if time.time() >= deadline - 5:
            break

        print(f"\n=== Optimisation on length {n} ===")

        # FFT sizes needed for convolutions & correlations
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        # ------------------------------------------------------------------
        # Build a pool of start vectors (use the best sequence from the
        # previous resolution as a warm‑start if it exists)
        warm_n = None
        if warm is not None:
            if warm.shape[0] > n:
                warm_n = warm[:n]
            elif warm.shape[0] < n:
                warm_n = np.pad(warm, (0, n - warm.shape[0]))
            else:
                warm_n = warm.copy()
        candidates = _candidate_pool(n, warm_n)

        # ------------------------------------------------------------------
        # Short Adam runs on every candidate (quick exploration)
        for cid, cand in enumerate(candidates):
            if time.time() >= deadline - 5:
                break
            a0 = _normalize(cand)

            # reserve a few seconds per candidate; shrink if we are running out of time
            cand_deadline = min(deadline - 5, time.time() + 12.0)

            a_adam, val_adam = _adam_optimize(
                a0, n, conv_fft_len, corr_fft_len, cand_deadline, max_iters=120_000, lr=0.05
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[cand {cid}] Adam improves global best to {val_adam:.6f}")

        # ------------------------------------------------------------------
        # A longer exact‑max sub‑gradient refinement on the current best
        if best_seq is not None and time.time() < deadline - 5:
            a_ref, val_ref = _subgradient_optimize(
                best_seq, n, conv_fft_len, corr_fft_len, deadline, max_iters=250_000, eta=0.55
            )
            if val_ref < best_val:
                best_val = val_ref
                best_seq = a_ref.copy()
                print(f"[Refine n={n}] Subgrad best {val_ref:.6f}")

        # ------------------------------------------------------------------
        # Cheap random local search for a final polish
        if best_seq is not None and time.time() < deadline - 5:
            a_loc, val_loc = _local_random_search(
                best_seq, n, conv_fft_len, deadline, max_moves=30_000
            )
            if val_loc < best_val:
                best_val = val_loc
                best_seq = a_loc.copy()
                print(f"[Local n={n}] improved best {val_loc:.6f}")

        # ------------------------------------------------------------------
        # Prepare warm‑start for the next (larger) grid: linear interpolation
        if best_seq is not None and n < grid_sizes[-1]:
            n_next = grid_sizes[grid_sizes.index(n) + 1]
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_next, endpoint=False)
            warm = np.interp(x_new, x_old, best_seq)
            warm = np.maximum(warm, 0.0)
            warm = warm / warm.sum()
        else:
            warm = None   # no further up‑sampling

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed)
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the limits accepted by the evaluator
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final verification (optional – the evaluator returns the same value)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"\nFinal evaluated value {final_val:.6f} (length {len(best_seq)})")
    except Exception as exc:   # evaluate_sequence might not be present in some test harnesses
        final_val = None
        print(f"\nFinal value could not be evaluated: {exc}")

    return best_seq.tolist()
