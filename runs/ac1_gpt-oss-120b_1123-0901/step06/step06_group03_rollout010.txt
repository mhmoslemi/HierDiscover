def cross_corr(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)

    start = a.shape[0] - 1

def _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val):
    eps = 1e-12
    logits = np.log(a0 + eps) 
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.05
    lam = 0.5   # start low
    lam_factor = 1.02
    it = 0
    last_improve = time.time()
    while time.time() < deadline:
        # compute a
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        # compute conv, obj
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={it}] new best {val:.6f}")
        # smooth max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log * grad_log)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # temperature schedule
        lam = min(lam * lam_factor, 1e6)
        # random shake if stagnant
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
        it += 1
        if it >= 300_000:
            break
    return best_seq, best_val

def _subgradient_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val):
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0/n)
    else:
        a = a / s
    eta0 = 0.6
    sub_last_improve = time.time()
    for it in range(50000):
        if time.time() >= deadline:
            break
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[Subgrad n={n} iter={it}] new best {val:.6f}")
        # active set mask
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        # stagnation stop
        if time.time() - sub_last_improve > 8.0:
            break

"""
Propose a step‑function that (approximately) minimizes  

        F(a) = 2·n·max(a∗a) / (∑ a)² .

Algorithm (single‑thread, works under the 1000 s budget)

1. **Initial candidate pool** – for a modest support size (starting at 1024)
   a diverse set of non‑negative vectors is created:
   * warm‑start (if a previous best is available) and its cyclic shifts,
   * uniform, cosine, Hamming, Gaussian bumps, edge‑concentrated power‑law,
     double‑Gaussian at the ends, random Dirichlet draws, sparse “k‑spike’’ vectors.
   All candidates are normalised to unit sum.

2. **Screening** – each candidate is evaluated with a fast FFT based exact
   objective; the best few (≤ 8) are kept for refinement.

3. **Deep optimisation** – every kept candidate is refined by two phases
   * **Adam on a smooth‑max surrogate** (log‑sum‑exp of the convolution);
     the optimisation works in the simplex via a soft‑max over the logits.
   * **Exact sub‑gradient / mirror‑descent** on the true max‑convolution.
   Both phases respect the global deadline and constantly update the
   globally best sequence/value.

4. **Resolution doubling** – the current best distribution is linearly
   interpolated to a larger support (×2) and steps 2‑3 are repeated.
   The support size progresses 1024 → 2048 → 4096 → 8192 → 16384 (as long
   as time permits).

5. **Finalisation** – the best sequence found is clipped to the evaluator’s
   allowed range [0, 1000] and returned as a plain Python list.

The routine prints progress information and never uses closures or lambdas,
so it complies with the execution sandbox.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact evaluation of F(a).  Assumes sum(a) > 0."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑sum‑exp surrogate of max(a∗a).
    The iterate lives in the probability simplex via a soft‑max of logits.
    """
    eps = 1e-12
    logits = np.log(a0 + eps)               # start in log‑space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2 = 0.9, 0.999
    lr = 0.05
    lam = 0.5           # temperature for the smooth‑max
    lam_factor = 1.018

    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        # ----- map logits → simplex (soft‑max) -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- exact objective (used for logging) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n:5d} it={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate (chain rule through soft‑max) -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, 1e6)

        # ----- occasional random shake if we have been stagnant -----
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        if it >= 300_000:               # safety bound
            break

    return best_seq, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact maximum of the convolution.
    Uses an exponentiated‑gradient update (multiplicative weights).
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    eta0 = 0.6
    sub_last_improve = time.time()

    for it in range(50_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[Subgrad n={n:5d} it={it:05d}] new best {val:.6f}")

        # ----- active set (indices attaining the maximum) -----
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # ----- sub‑gradient of max‑convolution -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- multiplicative‑weights (mirror descent on simplex) -----
        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 6.0:   # stagnation stop
            break

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam → sub‑gradient phases and return the best sequence/value
    found (global across all calls).
    """
    best_seq, best_val = _adam_optimize(
        a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
    )
    best_seq, best_val = _subgradient_optimize(
        best_seq, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
    )
    return best_seq, best_val

    # Hamming window (non‑negative, smooth)
    shape = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # Gaussian bumps centred in the middle (varying width)
    i = np.arange(n, dtype=np.float64)
    for sigma in np.linspace(n * 0.04, n * 0.25, 5):
        shape = np.exp(-0.5 * ((i - n / 2) ** 2) / (sigma ** 2))
        shape /= shape.sum()
        pool.append(shape)

    # Two symmetric edge Gaussians (small σ)
    for sigma in np.linspace(n * 0.03, n * 0.12, 4):
        left = np.exp(-0.5 * (i ** 2) / (sigma ** 2))
        right = np.exp(-0.5 * ((i - (n - 1)) ** 2) / (sigma ** 2))
        shape = left + right
        shape /= shape.sum()
        pool.append(shape)

    # Power‑law edge‑concentrated family (i·(n‑i+1))^α
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)                # i·(n‑i+1)
    for alpha in np.linspace(-2.0, -0.2, 8):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # Random Dirichlet draws (different concentration parameters)
    for alpha in np.logspace(-2, 0, 12):
        shape = np.random.dirichlet(np.full(n, alpha))
        pool.append(shape)

    # Sparse “k‑spike’’ vectors – useful for exploring extreme shapes
    for k in (2, 3, 5, 8):
        idxs = np.random.choice(n, size=k, replace=False)
        shape = np.zeros(n)
        shape[idxs] = 1.0 / k
        pool.append(shape)

    return pool


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function minimising the supplied evaluation
    metric.  The routine respects the given time budget (default 1000 s) and
    returns the best sequence encountered as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    #  Multi‑resolution loop: 1024 → 2048 → 4096 → 8192 → 16384
    # ------------------------------------------------------------------
    n = 1024
    max_n = 16384
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    stage = 0

    while True:
        if time.time() >= deadline:
            break
        stage += 1
        print(f"\n=== Stage {stage}: n = {n} ===")

        # ----- warm‑start preparation (interpolated from previous best) -----
        if best_seq is not None:
            x_old = np.linspace(0.0, 1.0, best_seq.shape[0], endpoint=False)
            x_new = np.linspace(0.0, 1.0, n, endpoint=False)
            warm = np.interp(x_new, x_old, best_seq)
            warm = np.maximum(warm, 0.0)
            warm /= warm.sum()
        else:
            warm = None

        # ----- build and screen a diverse pool of candidates -----
        candidates = _generate_initial_pool(n, warm)
        conv_fft_len = _next_pow2(2 * n - 1)
        scored: list[tuple[float, np.ndarray]] = []

        for idx, cand in enumerate(candidates):
            a = np.maximum(cand, 0.0)
            s = a.sum()
            if s < 1e-12:                           # avoid degenerate vectors
                a = np.random.dirichlet(np.ones(n))
                s = a.sum()
            a = a / s
            val = _objective(a, n, conv_fft_len)
            scored.append((val, a))
            if val < best_val:
                best_val = val
                best_seq = a.copy()
            print(f"[screen {idx:03d}] obj = {val:.6f}")

        # keep the most promising starts (up to eight)
        scored.sort(key=lambda x: x[0])
        top_candidates = [vec for _, vec in scored[:8]]

        # ----- deep optimisation passes on the top candidates -----
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        for cid, a0 in enumerate(top_candidates):
            if time.time() >= deadline:
                break
            print(f"[optim start {cid}] launching deep optimisation (n={n})")
            best_seq, best_val = _deep_optimize(
                a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
            )

        # ----- decide whether to increase the resolution -----
        if n >= max_n:
            break
        # give a little breathing room before the next up‑sampling
        if time.time() + 5.0 >= deadline:
            break

        # double the support size (linear interpolation of the current best)
        n = min(max_n, n * 2)
        if best_seq is not None:
            old_n = best_seq.shape[0]
            x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n, endpoint=False)
            best_seq = np.interp(x_new, x_old, best_seq)
            best_seq = np.maximum(best_seq, 0.0)
            best_seq /= best_seq.sum()
        else:
            # should never happen, but keep the code robust
            best_seq = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    #  Finalisation
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(1024, 1.0 / 1024, dtype=np.float64)

    # Clip to evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"\nFinished. Best objective ≈ {best_val:.6f} with length {best_seq.shape[0]}")
    return best_seq.tolist()
