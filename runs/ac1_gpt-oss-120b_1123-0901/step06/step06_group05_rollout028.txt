    corr_len = _next_pow2(3*n-2)
    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1,beta2,eps = 0.9,0.999,1e-8
    lr = 0.05

    best_a = a0.copy()
    best_loss = float('inf')
    best_val = float('inf')
    last_improve = time.time()

    for t in range(1, max_iters+1):
        # map
        exp_logits = np.exp(logits - np.max(logits))
        a = exp_logits / exp_logits.sum()
        # compute conv
        conv = _full_convolution(a, conv_len)
        avg = conv.mean()  # or (np.sum(a) ** 2) / (2*n - 1)
        diff = conv - avg
        loss = 0.5 * np.sum(diff**2)

        if loss < best_loss:
            best_loss = loss
            best_a = a.copy()
            # compute objective
            val = 2.0 * n * conv.max()
            if val < best_val:

        # gradient
        grad_a = 2.0 * _cross_correlation(diff, a, corr_len)

        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if time.time() > deadline:
            break

    # Return best_a and best_val
    return best_a,best_val

    """
    Multi‑stage optimisation:
    1. Small‑scale search (n≈2k) to learn a good “seed” shape.
    2. Progressive up‑sampling (8k → 32k → 64k) while

       – then tightening the true max‑convolution with a smooth‑max Adam

    sequence (list of non‑negative floats) found.

    deadline = time.time() + budget_s - 0.5

    best_val = float('inf')

def _beta_shape(n, alpha, beta):
    # avoid zero at ends for alpha,beta<1 by adding epsilon
    x = np.linspace(0.0, 1.0, n)
    # shift slightly to avoid exact 0/1 when α,beta<1
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    shape = np.power(x, alpha - 1.0) * np.power(1 - x, beta - 1.0)
    # If shape sum zero (unlikely) fallback to uniform
    s = shape.sum()
    if s <= 0:
        shape = np.ones_like(shape)
        s = shape.sum()
    return shape / s

2. **Progressive up‑sampling.**  The best sequence of the previous stage
   is linearly interpolated to a larger size (8 k, 32 k…) and the same
   three‑step refinement is performed again.  Because the shape is
   already good, only a few thousand iterations suffice at the larger
   sizes.

3. **Final cheap hill‑climb** that swaps tiny masses between indices to
   squeeze any remaining improvement.

All optimisation steps respect the global time budget (default 1000 s)
and the function returns a plain Python list suitable for the external
evaluator.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative vector to ``new_n`` points by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:                     # protect against a zero vector
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric Beta(α,β) density discretised on n points."""
    x = np.linspace(0.0, 1.0, n)
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    shape = np.power(x, alpha - 1.0) * np.power(1 - x, beta - 1.0)
    s = shape.sum()
    if s <= 0:
        shape = np.ones_like(shape)
        s = shape.sum()
    return shape / s


def _generate_initial_candidates(n: int, prev_best):
    """
    Produce a heterogeneous pool of starting points for length ``n``.
    ``prev_best`` (if not ``None``) is up‑sampled and used as a warm‑start.
    """
    candidates = []

    # 1. warm‑start from the best sequence found so far (upsampled)
    if prev_best is not None:
        candidates.append(_upsample(prev_best, n))

    # 2. historic solution if the global variable exists
    hseq = globals().get('height_sequence_1')
    if isinstance(hseq, (list, np.ndarray)):
        h = np.asarray(hseq, dtype=np.float64)
        candidates.append(_upsample(h, n))

    # 3. random Dirichlet families (different concentrations)
    candidates.append(np.random.dirichlet(np.ones(n)))
    for conc in (0.2, 0.5, 0.8):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # 4. edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.1, 7):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # 5. symmetric two‑sided exponentials
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # 6. Hann window (raised‑cosine)
    i = np.arange(n, dtype=np.float64)
    hann = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    hann /= hann.sum()
    candidates.append(hann)

    # 7. half‑sine shape
    sine = np.sin(np.pi * i / (n - 1))
    sine = np.maximum(sine, 0)
    sine /= sine.sum()
    candidates.append(sine)

    # 8. centred Gaussian‑like bumps with several widths
    mu = (n - 1) / 2.0
    for sigma in (n / 10, n / 6, n / 4):
        shape = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        shape /= shape.sum()
        candidates.append(shape)

    # 9. Beta‑distribution shapes (a few useful parameter pairs)
    for a, b in ((0.5, 0.5), (0.3, 0.3), (0.7, 0.7),
                 (1.5, 1.5), (2.0, 2.0)):
        candidates.append(_beta_shape(n, a, b))

    # Normalise and discard pathological entries
    clean = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        clean.append(c / s)
    return clean


# ----------------------------------------------------------------------
def _run_flatten_adam(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Adam optimiser that minimises the variance of the convolution
    (i.e. forces the whole convolution toward a constant).
    Returns the best sequence seen and its exact objective value.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05

    best_a = a0.copy()
    best_val = float('inf')
    best_loss = float('inf')
    last_improve = time.time()

    # the average value of a convolution when sum(a)=1
    avg_conv = 1.0 / (2 * n - 1)

    for t in range(1, max_iters + 1):
        # ---- simplex projection via soft‑max ----
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ---- convolution and loss ----
        conv = _full_convolution(a, conv_fft_len)
        diff = conv - avg_conv                      # centred vector
        loss = 0.5 * np.sum(diff * diff)

        # exact objective (sum(a)=1 ⇒ denominator drops out)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if loss < best_loss or val < best_val:
            best_loss = loss
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Flatten iter={t}] loss={loss:.2e}  obj={val:.6f}")

        # ---- gradient of the variance loss ----
        grad_a = 2.0 * _cross_correlation(diff, a, corr_fft_len)

        # ---- back‑prop through the soft‑max (logits) ----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- occasional random jitter if stuck ----
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Adam optimiser on the smooth‑max surrogate of the original objective.
    Returns the best sequence seen and its exact objective value.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.30                # temperature for the smooth‑max (soft max)
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float('inf')
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # ---- simplex via soft‑max ----
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ---- exact objective (sum(a)=1) ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t}] new best {val:.6f}")

        # ---- smooth‑max (log‑sum‑exp) weights ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

        # ---- gradient of the surrogate ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- back‑prop through the soft‑max (logits) ----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, 1e4)

        # ---- occasional random shake if stuck ----
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _run_subgrad(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Exponentiated‑gradient sub‑gradient phase on the true max‑convolution.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    last_improve = time.time()
    eta0 = 0.55

    for it in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _local_hill_climb(a0: np.ndarray, n: int, deadline: float,
                     max_trials: int = 200_000):
    """
    Very cheap stochastic local search: move a small ε of mass from one
    index to another if it reduces the exact objective.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    best_a = a0.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    no_improve = 0
    trials = 0
    while time.time() < deadline and trials < max_trials and no_improve < 5_000:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0.0, min(0.05, best_a[i]))
        if eps == 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = cand
            no_improve = 0
            print(f"[n={n} HillClimb] improved to {val:.6f}")
        else:
            no_improve += 1
        trials += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises   F(a) = 2·n·max(a*a) / (sum(a)²).

    The search proceeds through three size‑levels (2048 → 8192 → 32768),
    applying a variance‑reduction Adam phase, a smooth‑max Adam phase,
    a short exponentiated‑gradient sub‑gradient phase and finishes with a
    cheap hill‑climb.  The best sequence encountered is returned as a plain
    Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float('inf')
    prev_best = None

    # length schedule – larger n gives more expressive power at higher cost
    lengths = [2048, 8192, 32768]   # add 65536 if time remains after the loop

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        candidates = _generate_initial_candidates(n, prev_best)

        # iteration budgets (decrease with n)
        max_flat = int(8000 * 2048 / n) + 2000     # flattening phase
        max_adam = int(15000 * 2048 / n) + 4000    # smooth‑max phase
        max_sub = int(4000 * 2048 / n) + 800       # sub‑gradient phase

        for cid, init in enumerate(candidates):
            if time.time() > deadline:
                break

            # 1️⃣  flatten the convolution (variance reduction)
            a_flat, val_flat = _run_flatten_adam(init, n, deadline, max_flat)

            # 2️⃣  smooth‑max Adam refinement
            a_adam, val_adam = _run_adam(a_flat, n, deadline, max_adam)
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n} cand={cid}] Adam set new global best {val_adam:.6f}")

            # 3️⃣  exponentiated‑gradient sub‑gradient phase
            a_sub, val_sub = _run_subgrad(a_adam, n, deadline, max_sub)
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[n={n} cand={cid}] Subgrad set new global best {val_sub:.6f}")

            # 4️⃣  one short fine‑tune on the current global best for this n
            if best_seq is not None and len(best_seq) == n:
                a_fine, val_fine = _run_adam(best_seq, n, deadline,
                                             max_adam // 2)
                if val_fine < best_val:
                    best_val = val_fine
                    best_seq = a_fine.copy()
                    print(f"[n={n}] Fine‑tune improved global best to {val_fine:.6f}")

        # remember the best of this stage for the next up‑sampling step
        if best_seq is not None and len(best_seq) == n:
            prev_best = best_seq.copy()
        else:
            prev_best = None

    # ------------------------------------------------------------------
    # Final cheap hill‑climb to squeeze any remaining improvement
    if best_seq is not None and time.time() < deadline:
        print(f"\n=== Final local hill‑climb (n={len(best_seq)}) ===")
        a_final, val_final = _local_hill_climb(best_seq, len(best_seq), deadline)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final
            print(f"[final] hill‑climb improved global best to {val_final:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform if nothing was produced (should never happen)
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Return a plain Python list, as required by the external evaluator
    return best_seq.tolist()
