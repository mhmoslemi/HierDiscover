"""
Propose a step‑function (non‑negative vector) that minimises

    F(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm combines three ideas:

1. **Adam on a smooth‑max surrogate** – fast global search on several
   seeds (uniform, previous best, jittered versions, random Dirichlet).

2. **Projected‑gradient descent (PGD)** on the exact ℓ∞‑norm.
   The sub‑gradient of the maximum convolution entry is obtained by
   cross‑correlation; updates are performed on the probability simplex
   using Euclidean projection.

3. **Fine‑grained local mass transfers** – after PGD we perform a short
   stochastic “mass‑shifting” phase that moves tiny amounts of probability
   between coordinates while preserving the simplex constraint.

All steps respect the wall‑clock budget (default 1000 s) and keep the
best sequence found.  The code works for any length; if a previous best
`height_sequence_1` is present it is used as the starting point.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a*a using a power‑of‑two FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) required for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]  for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i >= 0, Σ_i x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all values are too small – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimisation on a smooth‑max surrogate (same spirit as the
# reference implementation, but with a slightly larger iteration budget)
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate of max(conv).
    Returns (best_a, best_val) where best_val = 2*n*max(conv) for the
    best iterate discovered before the deadline.
    """
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05

    lam = 1.0               # smooth‑max temperature
    lam_factor = 1.02

    # a bit more iterations for large n (the old code used ~3k for n=32k)
    max_iters = int(80000 * 1024 / n) + 4000

    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0:
            w = w / w_sum
        else:
            w = np.full_like(w, 1.0 / w.size)

        # gradient of the smooth‑max w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # back‑prop through soft‑max (logits gradient)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature annealing
        lam = min(lam * lam_factor, 1e8)

        # exact objective (sum a = 1)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # occasional jitter to escape stagnation
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected‑gradient descent on the true ℓ∞‑norm
# ----------------------------------------------------------------------


def _pgd_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on max(conv) with Euclidean projection onto the simplex.
    Returns (best_a, best_val).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() == 0:
        a = np.random.dirichlet(np.ones(n))
    else:
        a = a / a.sum()

    # current best
    conv = _full_convolution(a, fft_len=conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    # parameters
    eta0 = 0.5                      # initial step size
    max_iters = 8000                # generous budget – will stop early if time runs out
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # convolution and active set
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # sub‑gradient of the max‑norm
        grad = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # diminishing step size
        eta = eta0 / np.sqrt(t)

        # projected update
        a_new = _proj_simplex(a - eta * grad)

        # evaluate new point
        conv_new = _full_convolution(a_new, fft_len=conv_fft_len)
        val_new = 2.0 * n * conv_new.max()

        # keep the best ever seen
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            last_improve = time.time()

        # accept the update only if it improves the current objective;
        # otherwise keep the old a (helps avoid wandering)
        if val_new < 2.0 * n * max_c:
            a = a_new
        else:
            # occasional small random jitter to shake out of plateaus
            if time.time() - last_improve > 6.0:
                jitter = np.random.normal(scale=0.005, size=n)
                logits = np.log(np.maximum(a, 1e-12)) + jitter
                a = np.exp(logits - np.max(logits))
                a = a / a.sum()
                last_improve = time.time()

        # stop if we have been idle for a while
        if time.time() - last_improve > 12.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Very cheap local “mass‑transfer” refinement
# ----------------------------------------------------------------------


def _local_transfer_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Stochastic refinement that moves a tiny amount of probability
    from one coordinate to another while staying on the simplex.
    """
    a = a0.copy()
    conv = _full_convolution(a, fft_len=conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    eps = 1e-4                      # amount of mass to transfer
    max_trials = 300                # keep this low – each trial needs an FFT

    last_improve = time.time()
    for _ in range(max_trials):
        if time.time() > deadline:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or a[i] < eps:
            continue

        a_new = a.copy()
        delta = min(eps, a_new[i])
        a_new[i] -= delta
        a_new[j] += delta

        conv_new = _full_convolution(a_new, fft_len=conv_fft_len)
        val_new = 2.0 * n * conv_new.max()
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
            last_improve = time.time()

        if time.time() - last_improve > 8.0:
            break

    return best_a, best_val

    """

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Determine problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        best_seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n = best_seq.shape[0]
        print(f"Using previous best of length n={n}")
    else:
        n = 32768
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        print(f"No previous solution – starting from uniform vector, n={n}")

    if best_seq.sum() == 0:
        best_seq = np.random.dirichlet(np.ones(n))
    else:

    # FFT lengths – a power of two for speed

    corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

    # Helper to compute the exact objective (avoids calling evaluate_sequence)
    def _objective(a_vec: np.ndarray) -> float:
        conv = _full_convolution(a_vec, fft_len=conv_fft_len)
        return 2.0 * n * conv.max()

    best_val = _objective(best_seq)
    print(f"Initial objective: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Build a modest collection of seeds
    # ------------------------------------------------------------------

            )

                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[seed {sid}] Adam improvement: {best_val:.6f}")

            # 2) Projected‑gradient descent on the true max‑norm
            a_pgd, val_pgd = _pgd_refine(
                best_seq, n, conv_fft_len, corr_fft_len, deadline
            )
            if val_pgd < best_val:
                best_val = val_pgd
                best_seq = a_pgd.copy()
                print(f"[seed {sid}] PGD improvement: {best_val:.6f}")

            # 3) Tiny stochastic mass‑transfer refinement
            a_loc, val_loc = _local_transfer_refine(
                best_seq, n, conv_fft_len, deadline
            )
            if val_loc < best_val:
                best_val = val_loc
                best_seq = a_loc.copy()
                print(f"[seed {sid}] Local transfer improvement: {best_val:.6f}")

        except Exception as exc:
            print(f"[seed {sid}] exception: {exc}")

    # ------------------------------------------------------------------
    # Final jitter‑driven PGD sweeps (if time permits)
    # ------------------------------------------------------------------
    jitter_round = 0
    while time.time() < deadline - 1.0:
        jitter_round += 1
        jitter = np.random.normal(scale=0.01, size=n)
        logits = np.log(np.maximum(best_seq, 1e-12)) + jitter
        a0 = np.exp(logits - np.max(logits))
        a0 = a0 / a0.sum()

        a_pgd, val_pgd = _pgd_refine(
            a0, n, conv_fft_len, corr_fft_len, deadline
        )
        if val_pgd < best_val:
            best_val = val_pgd
            best_seq = a_pgd.copy()
            print(f"[jitter {jitter_round}] PGD new best: {best_val:.6f}")

        # tiny local refinement after each jittered PGD
        a_loc, val_loc = _local_transfer_refine(
            best_seq, n, conv_fft_len, deadline
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[jitter {jitter_round}] post‑PGD local improvement: {best_val:.6f}")

    print(f"Finished – best objective found: {best_val:.6f}")
    # Return as plain Python list as required by the external evaluator
    return best_seq.tolist()
