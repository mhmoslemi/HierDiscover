"""
Propose a step‑function (non‑negative vector) that minimises

    F(a) = 2 * n * max(a * a) / (sum(a)**2)

The algorithm uses three progressive stages:

1. *Coarse parametric sweep* on a small length (2048).  A large
   heterogeneous library of low‑dimensional families (power‑law edges,
   two‑block edge windows, Beta‑mixtures, sine‑bumps, Dirichlet draws,
   etc.) is sampled, each shape is normalised to sum‑one and evaluated
   by an inexpensive FFT based objective.  The ten best shapes are kept.

2. *Mid‑size refinement* – each selected shape is up‑sampled to a
   medium length (8192) and refined with two gradient‑based phases:
   an Adam optimiser on a smooth‑max surrogate of the convolution maximum
   (log‑sum‑exp with an annealed temperature) followed by an exact
   ℓ∞ sub‑gradient step (exponentiated‑gradient).  The best mid‑size
   vector is retained.

3. *Large‑size polishing* – the current best vector is up‑sampled to the
   target length (32768) and optimised again with the same two‑phase
   scheme, but with more iterations and a slower temperature schedule.
   A few random jitters around the final solution are also tried.

All optimisation respects the provided wall‑clock budget (default 1000 s)
and returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (zero‑padded) via FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation needed for the gradient of the smooth‑max."""
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def compute_objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) / (sum(seq)**2)."""
    # security checks are already performed by the external evaluator,
    # but we keep a tiny defensive wrapper.
    if seq.ndim != 1 or seq.size == 0:
        return float("inf")
    n = len(seq)
    conv = _full_convolution(seq)
    max_c = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


def _normalise(v: np.ndarray) -> np.ndarray:
    """Scale to have sum 1, guard against zero‑vector."""
    s = v.sum()
    if s <= 0:
        # fallback to a uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n, then normalise."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, seq)
    return _normalise(up)


# ----------------------------------------------------------------------
# Shape factories (low‑dimensional parametric families)
# ----------------------------------------------------------------------


def shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def shape_powerlaw_edges(n: int, alpha: float) -> np.ndarray:
    """Symmetric edge‑concentrated shape: (i·(n‑i+1))**alpha."""
    idx = np.arange(1, n + 1, dtype=float)
    prod = idx * (n - idx + 1.0)
    shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
    return _normalise(shape)


def shape_two_edge_blocks(n: int, p: float) -> np.ndarray:
    """Uniform mass on two edge blocks of relative width p (0<p<0.5)."""
    if not (0.0 < p < 0.5):
        return shape_uniform(n)
    block_len = max(1, int(p * n))
    shape = np.zeros(n, dtype=float)
    shape[:block_len] = 0.5 / block_len
    shape[-block_len:] = 0.5 / block_len
    return shape


def shape_beta_mixture(n: int, w: float, a: float, b: float) -> np.ndarray:
    """Mixture w·Beta(a,b) + (1‑w)·uniform, symmetric via a==b."""
    x = (np.arange(n, dtype=float) + 0.5) / n  # centre points
    # avoid division‑by‑zero when a or b < 1
    eps = 1e-12
    beta_pdf = np.power(np.maximum(x, eps), a - 1.0) * np.power(
        np.maximum(1.0 - x, eps), b - 1.0
    )
    shape = w * beta_pdf + (1.0 - w) * (1.0 / n)
    return _normalise(shape)


def shape_sine_bump(n: int, power: float) -> np.ndarray:
    """sin(π·(i+0.5)/n) raised to *power* (smooth bump)."""
    s = np.sin(np.pi * (np.arange(n, dtype=float) + 0.5) / n)
    shape = np.power(s, power)
    return _normalise(shape)


# ----------------------------------------------------------------------
# Optimisation primitives (Adam on smooth‑max + exact sub‑gradient)
# ----------------------------------------------------------------------


def adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = None,
    lam_init: float = 0.5,
    lam_factor: float = 1.01,
    lr: float = 0.04,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best vector found and its exact objective value.
    """
    eps = 1e-16
    a = np.maximum(a0, eps)
    a /= a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    if max_iters is None:
        max_iters = int(50000 * 1024 / n) + 2000

    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()
    lam = lam_init

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max to stay on simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.size)

        # gradient of smooth‑max w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature annealing
        lam = min(lam * lam_factor, 1e8)

        # exact objective (sum(a)=1)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # occasional jitter if stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


def subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 3500,
    eta0: float = 0.45,
) -> tuple[np.ndarray, float]:
    """
    Exact ℓ∞ sub‑gradient refinement (exponentiated‑gradient).
    Returns the best vector and its exact objective.
    """
    a = _normalise(a0)
    best_val = float("inf")
    best_a = a.copy()
    eta = eta0
    last_improve = time.time()

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        # indices where the max is attained (tolerance)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        eta *= 0.9995  # slowly decay step size

        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the wall‑clock budget and returns the
    best sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    total_deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Stage 1 – coarse parametric sweep on a small length
    # ------------------------------------------------------------------
    n_small = 2048
    conv_fft_small = _next_pow2(2 * n_small - 1)
    corr_fft_small = _next_pow2(3 * n_small - 2)

    small_candidates = []

    # uniform
    small_candidates.append(shape_uniform(n_small))

    # symmetric power‑law edges
    for alpha in np.linspace(-3.0, -0.2, 15):
        small_candidates.append(shape_powerlaw_edges(n_small, alpha))

    # two‑edge blocks
    for p in [0.01, 0.02, 0.05, 0.10, 0.15, 0.20, 0.25, 0.30]:
        small_candidates.append(shape_two_edge_blocks(n_small, p))

    # Beta‑mixture shapes (symmetric, a==b)
    betas = [0.2, 0.5, 0.8, 1.2, 2.0, 5.0]
    for w in [0.2, 0.4, 0.6, 0.8]:
        for a in betas:
            small_candidates.append(shape_beta_mixture(n_small, w, a, a))

    # sine‑bump family
    for power in [1.0, 2.0, 4.0, 8.0]:
        small_candidates.append(shape_sine_bump(n_small, power))

    # random Dirichlet draws
    for _ in range(10):
        small_candidates.append(np.random.dirichlet(np.ones(n_small)))

    # historic global variable, if any
    hseq = globals().get("height_sequence_1")
    if hseq is not None:
        h = np.asarray(hseq, dtype=float)
        if h.shape[0] != n_small:
            h = upsample(h, n_small)
        small_candidates.append(_normalise(h))

    # evaluate all small candidates
    evaluated = []
    for idx, cand in enumerate(small_candidates):
        val = compute_objective(cand)
        evaluated.append((val, cand))
        # keep output tidy
        if time.time() > total_deadline:
            break

    # keep the ten best
    evaluated.sort(key=lambda x: x[0])
    top_small = [cand for (_, cand) in evaluated[:10]]
    best_val = evaluated[0][0] if evaluated else float("inf")
    best_seq = evaluated[0][1] if evaluated else None
    print(f"[Stage 1] best small‑scale value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – medium length refinement
    # ------------------------------------------------------------------
    n_mid = 8192
    conv_fft_mid = _next_pow2(2 * n_mid - 1)
    corr_fft_mid = _next_pow2(3 * n_mid - 2)

    mid_deadline = start_time + 0.6 * budget_s
    best_mid_seq = None
    best_mid_val = float("inf")

    for idx, cand_small in enumerate(top_small):
        if time.time() > mid_deadline:
            break
        cand_mid = upsample(cand_small, n_mid)

        # Adam phase
        a_adam, val_adam = adam_optimize(
            cand_mid,
            n_mid,
            conv_fft_mid,
            corr_fft_mid,
            deadline=mid_deadline,
            max_iters=int(60000 * 1024 / n_mid) + 2000,
            lam_init=0.4,
            lam_factor=1.01,
            lr=0.04,
        )

        # Sub‑gradient refinement
        a_ref, val_ref = subgradient_refine(
            a_adam,
            n_mid,
            conv_fft_mid,
            corr_fft_mid,
            deadline=mid_deadline,
            max_iters=3500,
            eta0=0.45,
        )

        # track best across medium runs
        if val_ref < best_mid_val:
            best_mid_val = val_ref
            best_mid_seq = a_ref.copy()
            print(f"[Stage 2 cand={idx}] new mid best {best_mid_val:.6f}")
        elif val_adam < best_mid_val:
            best_mid_val = val_adam
            best_mid_seq = a_adam.copy()
            print(f"[Stage 2 cand={idx}] new mid best (Adam) {best_mid_val:.6f}")

    if best_mid_seq is None:
        # fallback to the best small candidate up‑sampled
        best_mid_seq = upsample(best_seq, n_mid)
        best_mid_val = compute_objective(best_mid_seq)

    # ------------------------------------------------------------------
    # Stage 3 – large length polishing (target size)
    # ------------------------------------------------------------------
    n_large = 32768
    conv_fft_large = _next_pow2(2 * n_large - 1)
    corr_fft_large = _next_pow2(3 * n_large - 2)

    large_deadline = total_deadline

    # start from up‑sampled medium optimum
    cand_large = upsample(best_mid_seq, n_large)

    # a few jittered restarts around the current best
    jitter_trials = 8
    for j in range(jitter_trials):
        if time.time() > large_deadline:
            break
        # create a jittered copy (log‑space perturbation)
        logits = np.log(np.maximum(cand_large, 1e-12))
        logits += np.random.normal(scale=0.015, size=n_large)
        jittered = np.exp(logits - np.max(logits))
        jittered = jittered / jittered.sum()

        a_adam, val_adam = adam_optimize(
            jittered,
            n_large,
            conv_fft_large,
            corr_fft_large,
            deadline=large_deadline,
            max_iters=int(80000 * 1024 / n_large) + 2000,
            lam_init=0.3,
            lam_factor=1.008,
            lr=0.035,
        )
        a_ref, val_ref = subgradient_refine(
            a_adam,
            n_large,
            conv_fft_large,
            corr_fft_large,
            deadline=large_deadline,
            max_iters=4000,
            eta0=0.43,
        )

        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Stage 3 jitter={j}] new global best {best_val:.6f}")
        elif val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Stage 3 jitter={j}] new global best (Adam) {best_val:.6f}")

    # If somehow we never improved, fall back to the large‑scale upsample
    if best_seq is None:
        best_seq = cand_large.copy()
        best_val = compute_objective(best_seq)

    # Final sanity check (optional – uses the supplied evaluator)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[Final] evaluate_sequence reports {final_score:.6f}")
    except Exception:
        # ignore, we already have best_val from our own computation
        final_score = best_val
        print(f"[Final] internal objective {final_score:.6f}")

    # Return as plain Python list
    return best_seq.tolist()
