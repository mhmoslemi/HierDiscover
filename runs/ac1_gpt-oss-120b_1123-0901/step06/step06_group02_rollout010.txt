def generate_candidate(n, shape):
    if shape == "exp":
        tau = np.random.uniform(0.05, 2.0) * n
        a = np.exp(-np.arange(n) / tau)
    elif shape == "pow":
        p = np.random.uniform(0.3, 2.0)
        a = 1.0 / (np.arange(1, n+1) ** p)
    elif shape == "hann":
        i = np.arange(n)
        a = 0.5 - 0.5 * np.cos(2*np.pi*i/(n-1))
    elif shape == "hamming":
        i = np.arange(n)
        a = 0.54 - 0.46 * np.cos(2*np.pi*i/(n-1))
    elif shape == "gauss":
        sigma = np.random.uniform(0.1, 0.3) * n
        i = np.arange(n) - (n-1)/2
        a = np.exp(-0.5 * (i/sigma)**2)
    elif shape == "bimodal":
        centers = np.random.choice(n, size=2, replace=False)
        tau1 = np.random.uniform(0.05, 0.5) * n
        tau2 = np.random.uniform(0.05, 0.5) * n
        a = np.exp(-np.abs(np.arange(n) - centers[0]) / tau1) + np.exp(-np.abs(np.arange(n) - centers[1]) / tau2)
    elif shape == "dirichlet":
        a = np.random.dirichlet(np.ones(n))
    else:
        a = np.random.rand(n)
    return a
```

    start = time.time()
    deadline = start + budget_s - 0.1

    best_seq = None
    best_score = np.inf

    def update_best(seq):
        nonlocal best_seq, best_score
        score = _score(seq)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] GLOBAL new best={best_score:.6f}  n={len(seq)}")
        return score

    # warm start
    hs = globals().get("height_sequence_1")
    if hs is not None:
        hs_arr = np.asarray(hs, dtype=float)

            n0 = hs_arr.shape[0]
            target0 = np.sqrt(2.0 * n0)
            hs_norm = hs_arr * (target0 / hs_arr.sum())
            update_best(hs_norm)

    # Phase 1: random parametric init
    while time.time() < deadline - 300:  # reserve for subsequent phases
        remaining = deadline - time.time()
        if remaining <= 0:

        # choose length

"""
Hybrid multi‑phase optimizer for the non‑negative step‑function bound.

The objective 2·n·max(conv(a,a))/sum(a)² is invariant under scaling,
so we normalise the L1‑norm to √(2·n) and minimise the peak of the
auto‑convolution.  The search proceeds through the following stages,
each respecting the overall time budget:

1. Warm‑start from any existing global construction (height_sequence_1).
2. Broad random initialisation using many parametric families
   (exponential, power‑law, Hann/Hamming, Gaussian, bimodal, Dirichlet).
3. Adam optimisation on a smooth log‑sum‑exp surrogate of the max‑convolution.
4. Evolutionary noise injection in log‑soft‑max (a simple CMA‑ES‑like loop).
5. Targeted projected sub‑gradient descent that follows the current worst index.
6. Simulated‑annealing hill‑climb with variable mass transfers.
7. Final tiny‑step random walk until time runs out.

All intermediate candidates are projected onto the simplex
{a ≥ 0, Σa = √(2·n)}.  The best sequence found is returned as a plain
Python list.  The implementation uses only NumPy and the standard library
and fits comfortably within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # Clip tiny negatives caused by rounding errors
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]          # sorted descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)     # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1        # zero‑based index of last positive entry
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Scoring helper (exact objective)
# ----------------------------------------------------------------------
def _score(seq: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv)/sum² (no clipping needed)."""
    conv = _full_convolution_fft(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 1e-12:
        return np.inf
    n = seq.shape[0]
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Random parametric candidate generator
# ----------------------------------------------------------------------
def _generate_candidate(n: int, shape: str) -> np.ndarray:
    """Generate a non‑negative vector of length n with the given shape."""
    idx = np.arange(n, dtype=float)
    if shape == "exp":
        tau = np.random.uniform(0.05, 2.0) * n
        a = np.exp(-idx / tau)
    elif shape == "pow":
        p = np.random.uniform(0.3, 2.0)
        a = 1.0 / ((idx + 1) ** p)
    elif shape == "hann":
        a = 0.5 - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1))
    elif shape == "hamming":
        a = 0.54 - 0.46 * np.cos(2.0 * np.pi * idx / (n - 1))
    elif shape == "gauss":
        sigma = np.random.uniform(0.1, 0.3) * n
        i = idx - (n - 1) / 2.0
        a = np.exp(-0.5 * (i / sigma) ** 2)
    elif shape == "bimodal":
        centers = np.random.choice(n, size=2, replace=False)
        tau1 = np.random.uniform(0.05, 0.5) * n
        tau2 = np.random.uniform(0.05, 0.5) * n
        a = np.exp(-np.abs(idx - centers[0]) / tau1) + np.exp(-np.abs(idx - centers[1]) / tau2)
    elif shape == "dirichlet":
        a = np.random.dirichlet(np.ones(n))
    else:   # uniform noise fallback
        a = np.random.rand(n)
    return a


# ----------------------------------------------------------------------
# Adam optimisation on the soft‑max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.10,
    min_tau_factor: float = 1e-4,
    alpha0: float = 0.02,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for a smooth surrogate of max‑convolution.
    Returns the best coefficient vector (on the simplex) and the exact
    max‑convolution value of that vector.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # map logits to simplex via softmax
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        p = exp_shift / np.sum(exp_shift)          # Σ p = 1
        a = target_sum * p                         # feasible a

        # exact max of convolution (used for logging)
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam n={n} new best max={cur_max:.6f}")

        # smoothed max via log‑sum‑exp
        tau = max(init_tau_factor * cur_max * (0.99 ** it), min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a: 2·a ⊛ w_rev
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)               # length 3·n‑2
        idx = (2 * n - 2) - np.arange(n)                      # positions for grad
        grad_a = 2.0 * conv2[idx]

        # back‑prop to logits (chain rule through softmax)
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slow decay of learning rate
        alpha *= 0.9995

    final_val = _full_convolution_fft(best_a, best_a).max()
    return best_a, float(final_val)


# ----------------------------------------------------------------------
# Targeted projected sub‑gradient descent (exact max)
# ----------------------------------------------------------------------
def _targeted_subgradient(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Follow the current worst index using projected sub‑gradient steps."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step = 0.2 * target_sum / n          # initial step size

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())          # index of current worst peak
        # gradient of conv_k w.r.t. a
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # projected step
        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)

        cand_val = _full_convolution_fft(cand, cand).max()
        if cand_val < best_val - 1e-12:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= 1.02                       # increase step after success
            if it % 200 == 0:
                print(f"[{time.time():.2f}] subgrad it={it} val={best_val:.6f}")
        else:
            step *= 0.70                       # reduce step on failure
            # occasional jitter
            if np.random.rand() < 0.07:
                jitter = np.random.rand(n) * 1e-4
                a = _project_onto_simplex(a + jitter, target_sum)

        if best_val < 1.503:
            break
    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (tiny mass transfers)
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Metropolis hill‑climb using random transfers of a small mass fraction."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # temperature schedule
    T0 = max(1e-3, best_val * 0.4)
    T_end = 1e-5
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue
        # transfer at most 8 % of mass at i
        delta = a[i] * np.random.rand() * 0.08
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()
        accept = False
        if new_val < best_val:
            accept = True
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")

        # re‑project occasionally to fight drift
        if it % 2500 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Evolutionary (CMA‑ES‑like) refinement on the log‑softmax parametrisation
# ----------------------------------------------------------------------
def _evolutionary_refinement(
    a_start: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simple evolutionary loop: add Gaussian noise to the logits,
    map back with a softmax, project onto the simplex.
    The best candidate is retained as the new centre for the next batch.
    """
    n = a_start.shape[0]
    logits = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_val = _score(best_a)

    sigma = 0.2                # initial mutation strength
    batch = 6                  # candidates per iteration

    while time.time() < deadline:
        candidates = []
        for _ in range(batch):
            noise = np.random.randn(n) * sigma
            l_new = logits + noise
            # softmax -> simplex
            max_l = np.max(l_new)
            exp_shift = np.exp(l_new - max_l)
            p = exp_shift / np.sum(exp_shift)
            a_cand = target_sum * p
            candidates.append(a_cand)

        for a in candidates:
            val = _score(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                logits = np.log(best_a + 1e-12)
                print(f"[{time.time():.2f}] evolutionary new best={best_val:.6f}")

        # decay mutation strength
        sigma *= 0.9995
        if sigma < 5e-7:
            sigma = 0.01   # occasional restart of exploration

    return best_a, best_val

                continue
            # transfer a very small fraction of mass from i to j
            delta = min(best_seq[i], target_sum * 0.001) * np.random.rand()
            a_new = best_seq.copy()
            a_new[i] -= delta
            a_new[j] += delta
            val = _score(a_new)
            if val < best_score:
                best_seq = a_new
                best_score = val
                print(f"[{time.time():.2f}] polishing new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be used)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = _score(best_seq)
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # Return a plain Python list as required by the evaluator
    return best_seq.tolist()
