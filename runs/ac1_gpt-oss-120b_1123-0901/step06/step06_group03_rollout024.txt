def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    n = 4096
    conv_fft_len = _next_pow2(2*n-1)
    corr_fft_len = _next_pow2(3*n-2)
    # build candidates
    candidates = _generate_candidate_pool(n, warm_start=None)
    # evaluate each
    best_seq, best_val = None, float('inf')
    top_k = 12
    scored = []
    for idx, a in enumerate(candidates):
         # ensure positivity
         a = np.maximum(a, 0.0)
         total = a.sum()
         if total < 1e-6:
             a = np.random.dirichlet(np.ones(n))
         else:
             a = a / total
         val = _objective(a, n, conv_fft_len)
         scored.append((val, a))
         if val < best_val:
             best_val, best_seq = val, a.copy()
         # maybe break if time nearing?
         if time.time() > deadline - 30:
             break
    # keep top candidates
    scored.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in scored[:top_k]]
    # deep optimization passes
    for i, a0 in enumerate(top_candidates):
        if time.time() >= deadline - 20:
            break
        print(f"[optim start {i}]")
        best_seq, best_val = _deep_optimize(a0, n, conv_fft_len, corr_fft_len, deadline,
                                            best_seq, best_val)
    # optional upsampling if time left
    if time.time() + 20 < deadline:
         n_big = 8192
         conv_fft_len_big = _next_pow2(2*n_big-1)
         corr_fft_len_big = _next_pow2(3*n_big-2)
         # interpolate best_seq onto new grid
         if best_seq is not None:
              # use np.interp
              x_old = np.linspace(0, 1, n, endpoint=False)
              x_new = np.linspace(0, 1, n_big, endpoint=False)
              a_big = np.interp(x_new, x_old, best_seq)
         else:
              a_big = np.random.dirichlet(np.ones(n_big))
         a_big = np.maximum(a_big, 0.0); a_big /= a_big.sum()
         best_seq, best_val = _deep_optimize(a_big, n_big, conv_fft_len_big, corr_fft_len_big,
                                             deadline, best_seq, best_val)
    # final return
    if best_seq is None:
        best_seq = np.full(n, 1.0/n)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return best_seq.tolist()

def _generate_candidate_pool(n, warm):
    pool = []
    # warm start
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n//4, n//2, 3*n//4):
            pool.append(np.roll(warm, shift))
    # uniform
    pool.append(np.full(n, 1.0/n))
    i = np.arange(n, dtype=float)
    # precompute prod
    prod = (i+1)*(n-i) # because i indexes 0..n-1 -> i+1 and n-i
    # power law edges
    alphas = np.linspace(-1.2, -0.2, 61) # fine grid
    for alpha in alphas:
        shape = np.where(prod>0, prod**alpha, 0.0)
        if shape.sum()>0:
            shape = shape/shape.sum()
            pool.append(shape)
    # double exponential
    lam_vals = np.linspace(0.1, 4.0, 81)
    for lam in lam_vals:
        shape = np.exp(-lam*i) + np.exp(-lam*(n-1-i))
        shape = shape/shape.sum()
        pool.append(shape)
    # gaussian bump
    sigma_vals = np.linspace(n*0.02, n*0.4, 20)
    center = (n-1)/2.0
    for sigma in sigma_vals:
        shape = np.exp(-0.5*((i-center)/sigma)**2)
        shape = shape/shape.sum()
        pool.append(shape)
    # beta-like shape: p,q negative
    p_vals = np.linspace(-1.0, -0.1, 10)
    q_vals = np.linspace(-1.0, -0.1, 10)
    x = (i+0.5)/n  # shift to (0,1)
    for p in p_vals:
        for q in q_vals:
            shape = (x**p) * ((1-x)**q)
            shape = np.where(np.isfinite(shape), shape, 0.0)
            shape = shape.clip(min=0)
            if shape.sum()>0:
                shape = shape/shape.sum()
                pool.append(shape)
    # mixture of uniform and powerlaw
    uniform = np.full(n, 1.0/n)
    for w in np.linspace(0,1,11):

    while time.time() < deadline:

        mx = logits.max()
        exp_log = np.exp(logits - mx)

        else:
            w /= w_sum

        # gradient w.r.t a (through smooth max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # gradient w.r.t logits via softmax Jacobian
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase temperature
        lam = min(lam * lam_factor, 1e6)

        # shake when stagnant
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

        it += 1
        # safety limit large but time based
        if it >= 1_200_000:
            break

    # Subgradient (exact max) phase
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0/n)
    else:
        a = a / total

    eta0 = 0.6
    sub_last_improve = time.time()
    for sub_it in range(500_000):

            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[Subgrad n={n} it={sub_it}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12)

#!/usr/bin/env python3
"""
Propose a non‑negative step‑function that (hopefully) improves the current best
upper bound on

        F(a) = 2·n·max(a∗a) / (∑a)² .

The routine follows three stages:

1. **Diverse parametric sweep** – builds a pool of >1 000 candidate vectors
   using several analytic families (power‑law edges, double‑exponential,
   Gaussian bumps, beta‑type shapes, mixtures and random Dirichlet draws).  Every
   candidate is normalised to unit sum and scored with a fast FFT‑based exact
   objective.  The best few (default 12) are retained.

2. **Two‑phase refinement** – each retained candidate is refined by
   (a) an Adam optimiser on a smooth‑max surrogate of the convolution
   (log‑sum‑exp), followed by (b) an exact sub‑gradient mirror‑descent phase
   operating on the true max‑convolution.  The optimiser respects the global
   deadline and constantly updates the globally best sequence/value.

3. **Upsampling (if time permits)** – the best vector is interpolated to a
   larger support (8192) and the whole refinement is repeated, which often
   yields a modest further reduction of the bound.

All numbers are forced to be non‑negative and clipped to the evaluator’s
range [0, 1000].  The function runs within the supplied time budget (default
1000 s) and returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) indexed by lag.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of 2·n·max(conv) / (sum(a)²).  Assumes sum(a) > 0."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Two‑phase refinement of a candidate sequence.

    Phase 1 – Adam on a smooth‑max surrogate.
    Phase 2 – Exact sub‑gradient mirror‑descent on the true max.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # work in log‑space (exponential map)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    # Adam hyper‑parameters (tuned for stability)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03                               # a bit smaller than the default

    # temperature for smooth‑max (log‑sum‑exp)
    lam = 0.3
    lam_factor = 1.005                      # increase slowly → smoother transition

    it = 0
    last_improve = time.time()

    # --------------------------------------------------- Adam phase
    while time.time() < deadline:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (useful for monitoring)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp formulation)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through the soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase temperature, but cap it
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter if we have been stagnant too long
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

        it += 1
        # hard safety bound – we rely mainly on the time‑check
        if it >= 1_200_000:
            break

    # ------------------------------------------------ Sub‑gradient phase (exact max)
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    total = a.sum()
    a = a / total if total > 0 else np.full(n, 1.0 / n)

    eta0 = 0.6
    sub_last_improve = time.time()
    for sub_it in range(500_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[Subgrad n={n} it={sub_it:06d}] new best {val:.6f}")

        # mask of active (maximal) indices
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (mirror‑descent on the simplex)
        eta = eta0 / np.sqrt(sub_it + 1)

        a = a * np.exp(-eta * grad_a)          # multiplicative update
        s = a.sum()
        if s < 1e-12:                          # prevent collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 8.0:
            # no improvement for a while → stop this phase
            break

    return global_best_seq, global_best_val

def _generate_candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of start vectors using several analytic families."""
    pool: list[np.ndarray] = []

    # ---- warm‑start and simple rotations ---------------------------------
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # ---- trivial baseline -------------------------------------------------
    pool.append(np.full(n, 1.0 / n))

    i = np.arange(n, dtype=np.float64)
    prod = (i + 1) * (n - i)                     # i·(n‑i+1) with 1‑based indexing

    # ---- power‑law edge‑concentrated shapes (fine sweep) -----------------
    alphas_fine = np.linspace(-1.2, -0.2, 61)   # dense grid around the likely optimum
    for alpha in alphas_fine:
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # ---- double‑exponential (two‑sided Laplace) families ----------------
    lam_vals = np.linspace(0.1, 4.0, 81)        # finer sweep than before
    for lam in lam_vals:
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        shape = shape.clip(min=0.0)
        shape /= shape.sum()
        pool.append(shape)

    # ---- Gaussian‑type bumps (centre) -----------------------------------
    sigma_vals = np.linspace(n * 0.02, n * 0.4, 20)
    centre = (n - 1) / 2.0
    for sigma in sigma_vals:
        shape = np.exp(-0.5 * ((i - centre) ** 2) / (sigma ** 2))
        shape /= shape.sum()
        pool.append(shape)

    # ---- beta‑like shapes (p,q negative → edge mass) --------------------
    p_vals = np.linspace(-1.0, -0.1, 10)
    q_vals = np.linspace(-1.0, -0.1, 10)
    x = (i + 0.5) / n                         # shift to (0,1)
    for p in p_vals:
        for q in q_vals:
            shape = (x ** p) * ((1.0 - x) ** q)
            # discard NaNs / infinities that appear for extreme exponents
            shape = np.where(np.isfinite(shape), shape, 0.0)
            if shape.sum() > 0:
                shape /= shape.sum()
                pool.append(shape)

    # ---- simple mixtures of uniform and a power‑law shape ---------------
    uniform = np.full(n, 1.0 / n)
    for w in np.linspace(0.0, 1.0, 11):
        for alpha in alphas_fine[::5]:          # coarse subset for speed
            power = np.where(prod > 0, prod ** alpha, 0.0)
            if power.sum() == 0:
                continue
            power /= power.sum()
            shape = w * uniform + (1.0 - w) * power
            shape /= shape.sum()
            pool.append(shape)

    # ---- random Dirichlet draws with a range of concentration parameters
    for alpha in np.logspace(-2, 0, 15):
        shape = np.random.dirichlet(np.full(n, alpha))
        pool.append(shape)

    return pool


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation metric.
    The routine respects the supplied time‑budget (default 1000 s) and returns
    the best sequence encountered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ---- primary support size ------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ---- warm‑start from a previously found construction, if any -------------
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        w = np.asarray(warm_start, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
        else:
            w = None
    else:
        w = None

    # ---- build the candidate pool -------------------------------------------
    candidates = _generate_candidate_pool(n, w)

    best_seq: np.ndarray | None = None
    best_val = float("inf")
    scored: list[tuple[float, np.ndarray]] = []

    print(f"[init] evaluating {len(candidates)} raw candidates")
    for idx, cand in enumerate(candidates):
        # safety checks – ensure non‑negative, normalise to unit sum
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 1e-6:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        val = _objective(a, n, conv_fft_len)
        scored.append((val, a))
        if val < best_val:
            best_val, best_seq = val, a.copy()
        if idx % 200 == 0:
            print(f"[screen {idx:04d}/{len(candidates)}] best={best_val:.6f}")
        # keep an eye on the deadline – stop early if we are too close
        if time.time() > deadline - 30.0:
            print("[screen] time low – terminating candidate sweep")
            break

    # ---- keep the most promising starts (up to K) ---------------------------
    K = 12
    scored.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in scored[:K]]
    print(f"[select] retained top {len(top_candidates)} candidates")

    # ---- deep optimisation passes (more time for the very best) ------------
    for cid, a0 in enumerate(top_candidates):
        if time.time() >= deadline - 20.0:
            print("[optim] deadline imminent – stopping further starts")
            break
        # allocate a little more time to the first candidate
        extra = 5.0 if cid == 0 else 0.0
        sub_deadline = min(deadline, time.time() + (budget_s * 0.4) + extra)
        print(f"[optim start {cid}] run deep optimisation (n={n})")
        best_seq, best_val = _deep_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_deadline, best_seq, best_val
        )
        print(f"[optim end {cid}] current global best {best_val:.6f}")

    # ---- optional up‑sampling to a larger grid if time permits ------------
    if time.time() + 20.0 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation onto the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("[big] start optimisation on enlarged support")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        print(f"[big] finished – best value {best_val:.6f}")

    # ---- finalisation -------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[final] returning sequence of length {best_seq.shape[0]}, value {best_val:.6f}")
    return best_seq.tolist()
