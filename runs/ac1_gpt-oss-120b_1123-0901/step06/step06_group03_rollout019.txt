"""
Propose a low‑autocorrelation step function.

The algorithm works in four stages:

1. **Rich pool generation** – a wide variety of analytic families
   (uniform, symmetric Beta, sinusoid, double‑exponential) together
   with many random Dirichlet draws and an optional warm‑start.
   Every vector is normalised to unit sum and the best few (≤8) are kept.

2. **Simple evolutionary search** – the selected vectors form a tiny
   population.  Off‑springs are created by convex combination of two
   parents plus a small stochastic “mutation”.  New candidates are
   evaluated with a fast FFT‑based objective and replace the worst
   individuals if they improve the global best.

3. **Refinement by Adam + exact sub‑gradient** – the current global best
   is refined with the smooth‑max Adam optimiser from the reference
   code, followed by an exponentiated‑gradient phase that works on the
   exact max‑convolution.  The schedule is a bit more aggressive
   (more iterations, slower temperature increase) and runs until the
   time budget is almost exhausted.

4. **Optional up‑sampling** – if time permits the refined distribution
   is interpolated to a larger support (8192 points) and the whole
   optimisation loop (steps 2‑3) is repeated, which often yields a
   strictly lower bound.

All numbers are forced to be non‑negative and ≤ 1000, satisfying the
security checks of ``evaluate_sequence``.  The routine respects the
provided time budget (default 1000 s) and returns the best sequence it
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two that is ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation w ★ a (same size as a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective value (sum(a) is assumed > 0)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate (log‑sum‑exp) followed by an exact
    max‑convolution sub‑gradient phase (exponentiated‑gradient mirror
    descent).  Returns the best sequence/value found (global across all
    calls).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                 # start in the simplex via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                                 # a touch larger than the reference

    lam = 0.3                                 # start temperature (more smoothing)
    lam_factor = 1.010                         # slower heating → more stable descent

    last_improve = time.time()
    it = 0

    # ------------------------------ Adam phase ------------------------------
    while time.time() < deadline:
        # project logits → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (useful for logging)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase temperature → approach true max
        lam = min(lam * lam_factor, 1e6)

        # add a tiny random shake if we have been idle for a while
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 5e-3
            last_improve = time.time()

        it += 1
        if it > 1_200_000:                # generous upper bound
            break

    # --------------------------- Exact sub‑gradient ---------------------------
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total

    eta0 = 0.6
    sub_last_improve = time.time()
    for sub_it in range(70_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[Subgrad n={n} it={sub_it:05d}] new best {val:.6f}")

        # active‑set mask (all positions attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent step on the simplex (exponentiated gradient)
        eta = eta0 / np.sqrt(sub_it + 1.0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid accidental collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 8.0:   # stagnation → stop early
            break

    return global_best_seq, global_best_val

def _candidate_pool(
    n: int,
    warm: np.ndarray | None,
) -> list[np.ndarray]:
    """Build a large and diverse set of start vectors."""
    pool: list[np.ndarray] = []

    # warm‑start (if supplied) and a few cyclic shifts
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # 1) uniform baseline
    pool.append(np.full(n, 1.0 / n))

    # 2) symmetric Beta families  (a = b)
    x = (np.arange(n, dtype=np.float64) + 0.5) / n   # avoid exact 0/1
    for a in np.logspace(-0.7, 0.7, 16):            # a≈0.2 … 5
        shape = np.power(x, a - 1.0) * np.power(1.0 - x, a - 1.0)
        s = shape.sum()
        if s > 0:
            pool.append(shape / s)

    # 3) double‑exponential (two‑sided Laplace) families
    i = np.arange(n, dtype=np.float64)
    for lam in np.linspace(0.1, 4.0, 12):
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        shape = np.maximum(shape, 0.0)
        s = shape.sum()
        if s > 0:
            pool.append(shape / s)

    # 4) single half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        pool.append(shape / s)

    # 5) a large set of random Dirichlet draws
    #    (different concentrations → from sparse to almost uniform)
    for alpha in np.logspace(-2, 0, 14):
        for _ in range(5):               # 5 draws per concentration
            shape = np.random.dirichlet(np.full(n, alpha))
            pool.append(shape)

    return pool

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the
    evaluation metric.  Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # primary support size (start modest, raise later if time permits)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm‑start handling – a previously found construction may be stored
    warm_start = globals().get("height_sequence_1")

        w = np.asarray(warm_start, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
        else:
            w = None
    else:
        w = None

    # ------------------------------------------------------------------
    # 1) generate a very diverse pool
    candidates = _candidate_pool(n, w)
    print(f"[setup] generated {len(candidates)} raw candidates")

    # ------------------------------------------------------------------
    # 2) quick screening: keep the best eight after a cheap objective eval
    scored = []
    best_val = float("inf")
    best_seq = None
    for idx, cand in enumerate(candidates):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:                     # degenerate – replace with fresh Dirichlet
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        val = _objective(a, n, conv_fft_len)
        scored.append((val, a))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        if idx % 50 == 0:
            print(f"[screen {idx:04d}/{len(candidates)}] val={val:.6f}")
    scored.sort(key=lambda x: x[0])
    population = [vec for _, vec in scored[:8]]
    print(f"[screen] best initial value {best_val:.6f}")

    # ------------------------------------------------------------------

        p1, p2 = np.random.choice(population, 2, replace=False)

        w_mix = np.random.rand()
        child = w_mix * p1 + (1.0 - w_mix) * p2
        # small mutation in log‑space (preserves positivity)
        noise = (np.random.randn(n) * 5e-3)       # 0.5 % multiplicative noise
        child = np.exp(np.log(child + 1e-12) + noise)
        child = np.maximum(child, 0.0)
        child_sum = child.sum()
        if child_sum < 1e-12:

        child /= child_sum

        worst_val, worst_idx = max((v, i) for i, (v, _) in enumerate(scored[-len(population):]))
        if val < worst_val:
            # update global best as well
            if val < best_val:
                best_val = val

        it += 1

        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("[big] start refinement on enlarged support")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        print(f"[big] final best {best_val:.6f}")

    # ------------------------------------------------------------------
    # final sanity – ensure compliance with the evaluator limits
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    total_time = time.time() - start_time
    print(f"[done] total runtime {total_time:.1f}s, best objective {best_val:.6f}")
