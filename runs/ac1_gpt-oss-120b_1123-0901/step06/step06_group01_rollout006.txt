"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.
=======================================================================

The search consists of three stages:

1. **Coarse evolutionary optimisation (DE)** – a simple Differential
   Evolution runs on a low‑dimensional parametrisation of the step
   function (K “knots” interpolated to length *n*).  The fitness is the
   exact max‑convolution, because with the normalisation Σa = √(2 n) the
   evaluation function reduces to that maximum.

2. **Gradient polishing** – from the best DE candidate we run both
   Projected Gradient Descent and an Adam optimiser on the exact
   max‑convolution.  These locally improve the result while keeping the
   vector on the simplex.

3. **Stochastic refinement** – a cheap hill‑climbing / simulated
   annealing phase moves small amounts of mass between indices,
   accepting occasional worsening moves to escape plateaux.

The whole routine respects the 1000 s budget, allocating roughly
60 % of the time to DE, the remainder to polishing and final refinement.
If a previous construction ``height_sequence_1`` is available, it is used
as a warm start (interpolated to the current length).  All helper
functions are defined at module level (no closures) and only NumPy and
standard library modules are required.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities ------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection onto {x | x_i ≥ 0, Σx_i = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(seq: np.ndarray) -> float:
    """Exact maximum of the linear convolution seq * seq."""
    n = seq.shape[0]
    Nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(seq, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
    return float(conv.max())

# Differential Evolution (very light‑weight custom implementation) --


def _eval_de_fitness(v: np.ndarray,
                     n: int,
                     target_sum: float,
                     knot_x: np.ndarray,
                     idx_arr: np.ndarray) -> float:
    """
    Decode a coarse vector v (length K) into a full sequence,
    normalise it and return the max‑convolution (the true objective).
    """
    seq0 = np.interp(idx_arr, knot_x, v)          # linear interpolation
    s = seq0.sum()
    if s <= 0.0:
        return float('inf')
    seq = seq0 * (target_sum / s)                 # enforce Σa = √(2 n)
    return _max_convolution(seq)


def _de_optimize(n: int,
                 target_sum: float,
                 rng: np.random.Generator,
                 deadline: float,
                 init_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Simple DE on a K‑dimensional “knot” representation.
    Returns the best full‑length sequence and its max‑convolution value.
    """
    # dimensionality of the low‑dim representation
    K = min(80, max(20, n // 20))

    # pre‑computed arrays for fast interpolation
    idx_arr = np.arange(n)
    knot_x = np.linspace(0, n - 1, K)

    # bounds for knot values
    low, high = 0.0, 5.0

    # initialise population
    popsize = max(150, 5 * K)        # a few hundred individuals
    pop = rng.uniform(low, high, size=(popsize, K))
    if init_seq is not None:
        v0 = np.interp(knot_x, idx_arr, init_seq)
        pop[0] = np.clip(v0, low, high)

    # evaluate initial fitness
    fitness = np.empty(popsize, dtype=float)
    for i in range(popsize):
        fitness[i] = _eval_de_fitness(pop[i], n, target_sum, knot_x, idx_arr)

    best_idx = int(np.argmin(fitness))
    best_v = pop[best_idx].copy()
    best_f = fitness[best_idx]

    # DE hyper‑parameters
    F = 0.8
    CR = 0.9
    max_generations = 200

    gen = 0
    while time.time() < deadline and gen < max_generations:
        for i in range(popsize):
            # choose three distinct indices different from i
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.clip(mutant, low, high)

            # binomial crossover
            cross_mask = rng.random(K) < CR
            if not np.any(cross_mask):
                cross_mask[rng.integers(K)] = True
            trial = np.where(cross_mask, mutant, pop[i])

            f_trial = _eval_de_fitness(trial, n, target_sum, knot_x, idx_arr)

            if f_trial < fitness[i]:
                pop[i] = trial
                fitness[i] = f_trial
                if f_trial < best_f:
                    best_f, best_v = f_trial, trial.copy()
        gen += 1
        if gen % 10 == 0:
            print(f"[DE n={n} gen={gen}] best={best_f:.6f}")

    # decode the best coarse vector to a full sequence
    seq0 = np.interp(idx_arr, knot_x, best_v)
    seq = seq0 * (target_sum / seq0.sum())
    return seq, best_f

# Optimisers (PGD and Adam) – copied from the previous state of the art

def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected Gradient Descent on the exact max‑convolution.
    Returns the best vector observed and its exact objective value.
    """
    target_sum = math.sqrt(2.0 * n)

    a = a0.copy()
    a = a * target_sum / a.sum()

    Nfft = 1 << ((3 * n - 2).bit_length())
    step0 = 0.001 * target_sum
    eps_active = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

        mask = conv >= M - eps_active
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = mask.astype(float) / float(cnt)

        # weighted reverse convolution (sub‑gradient)
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        rev_full = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        rev_full = rev_full[: len(w) + n - 1]
        idx_start = 2 * n - 2
        rev = rev_full[idx_start - np.arange(n)]

        g = 2.0 * rev

        lr = step0 / math.sqrt(it)
        a_tmp = a - lr * g
        a = _project_onto_simplex(a_tmp, target_sum)

        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)

    return best_seq, best_score


def _exp_gradient_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑its (exponentiated‑gradient) using a soft‑max
    approximation of the max‑convolution.
    Returns the best vector seen in this run together with its exact
    objective value.
    """
    target_sum = math.sqrt(2.0 * n)

    eps = 1e-16
    y = np.log(a0 + eps)

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.08
    jitter_interval = 3000

    tau_factor = 0.30
    decay_rate = max_iters / 8.0

    Nfft = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        logits = y - np.max(y)
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()
        a = target_sum * p

        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[Adam len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
        else:
            maxc = conv.max()
            ex = np.exp((conv - maxc) / tau)
            w = ex / ex.sum()

        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        rev_full = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        rev_full = rev_full[: len(w) + n - 1]
        idx_start = 2 * n - 2
        rev = rev_full[idx_start - np.arange(n)]

        g_a = 2.0 * rev
        avg = np.dot(p, g_a)
        g_y = target_sum * p * (g_a - avg)

        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        if it % jitter_interval == 0:
            jitter = rng.normal(scale=0.01, size=n)
            y += jitter

    return best_seq, best_score


# ----------------------------------------------------------------------
# Local stochastic refinement --------------------------------------------


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing with simulated‑annealing acceptance.
    Moves a small fraction of mass from a random index i to j.
    """
    target_sum = a.sum()
    cur_seq = a.copy()
    cur_score = best_score

    Nfft = 1 << ((2 * n - 1).bit_length())
    max_steps = 200_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (plain Python list) found within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0    # safety margin

    # ------------------------------------------------------------------
    # Load a previous construction if it exists
    base = None
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        pass

    # Choose a baseline length (the historically good size)
    n0 = base.shape[0] if base is not None else 1900
    # Try a second slightly larger length if time permits
    lengths = [n0]
    if time.time() + 30 < deadline:
        lengths.append(min(8000, int(n0 * 1.10)))

    best_seq = None
    best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        print(f"[run] start length = {n}")

        # ------------------------------------------------------------------
        # Prepare initial full‑length sequence (may be None)
        base_n = None
        if base is not None:
            if base.shape[0] != n:
                old_x = np.linspace(0.0, 1.0, base.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                interp = np.interp(new_x, old_x, base)
                interp = np.maximum(interp, 0.0)
                interp = interp / interp.sum() * target_sum
                base_n = interp
            else:
                base_n = base.copy()

        # ------------------------------------------------------------------
        # Stage 1 – differential evolution on a low‑dimensional sketch
        de_seq, de_M = _de_optimize(
            n=n,
            target_sum=target_sum,
            rng=rng,
            deadline=deadline,
            init_seq=base_n,
        )
        de_score = evaluate_sequence(de_seq.tolist())
        print(f"[DE] length {n} max_conv={de_M:.6f}, evaluate={de_score:.6f}")

        if de_score < best_score:
            best_score = de_score
            best_seq = de_seq.copy()

        # ------------------------------------------------------------------
        # Stage 2 – gradient polishing (PGD + Adam)
        remaining = deadline - time.time()
        if remaining < 1.0:
            continue
        # a modest iteration budget that fits the remaining time
        max_iters = int(200_000 * (budget_s / 1000.0))
        max_iters = max(100_000, min(max_iters, int(2_500_000 * 2100 / n)))

        # Projected Gradient Descent
        best_seq, best_score = _pgd_optimize(
            a0=de_seq.copy(),
            n=n,
            deadline=deadline,
            max_iters=max_iters,
            best_seq=best_seq if best_seq is not None else de_seq,
            best_score=best_score,
            rng=rng,
        )
        # Adam (exponentiated‑gradient) optimiser
        best_seq, best_score = _exp_gradient_optimize(
            a0=de_seq.copy(),
            n=n,
            deadline=deadline,
            max_iters=max_iters,
            best_seq=best_seq,
            best_score=best_score,
            rng=rng,
        )

        # ------------------------------------------------------------------
        # Stage 3 – cheap stochastic refinement
        if time.time() < deadline:
            refine_deadline = time.time() + min((deadline - time.time()) * 0.5, 30.0)
            best_seq, best_score = _local_refine(
                a=best_seq,
                n=best_seq.shape[0],
                deadline=refine_deadline,
                best_score=best_score,
                rng=rng,
            )

    # ------------------------------------------------------------------
    # Fallback (should never trigger)
    if best_seq is None:
        n = n0
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq.tolist()
