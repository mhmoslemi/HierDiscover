#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
Propose a step‑function (list of non‑negative heights) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)^2)

using a three‑phase strategy

1️⃣  A diverse pool of initial vectors (warm‑start, random Dirichlet,
    power‑law, exponentials, Hann, sine, Gaussian …).

2️⃣  Continuous optimisation on the simplex:
    – Adam on a smooth‑max surrogate of the max‑convolution,
    – Exponentiated‑gradient sub‑gradient on the true max‑convolution,
    – (for the smallest length 2048) a tiny linear‑program refinement
      that maximises the total mass while respecting the current max bound.

3️⃣  A cheap stochastic hill‑climb to squeeze out any remaining improvement.

The routine respects the supplied time budget (≈1000 s) and returns the best
sequence found as a plain Python list.  It deliberately avoids any
filesystem or network access and works with the externally provided
`evaluate_sequence` function.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative vector to ``new_n`` points by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:                     # protect against a zero vector
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s


def _generate_initial_candidates(n: int, prev_best):
    """
    Produce a heterogeneous pool of starting points for a given length ``n``.
    ``prev_best`` (if not ``None``) is up‑sampled and used as a warm‑start.
    """
    candidates = []

    # 1. warm‑start from the best sequence found so far (upsampled)
    if prev_best is not None:
        candidates.append(_upsample(prev_best, n))

    # 2. historic solution if the global variable exists
    hseq = globals().get('height_sequence_1')
    if isinstance(hseq, (list, np.ndarray)):
        h = np.asarray(hseq, dtype=np.float64)
        candidates.append(_upsample(h, n))

    # 3. random Dirichlet families (different concentrations)
    candidates.append(np.random.dirichlet(np.ones(n)))
    for conc in (0.2, 0.5, 0.8):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # 4. edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.1, 7):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # 5. monotone decreasing exponentials (left‑to‑right)
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i)
        shape /= shape.sum()
        candidates.append(shape)

    # 6. the reverse of the exponentials (right‑to‑left)
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        i = np.arange(n, dtype=np.float64)[::-1]
        shape = np.exp(-gamma * i)
        shape /= shape.sum()
        candidates.append(shape)

    # 7. symmetric two‑sided exponentials
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # 8. Hann window (raised‑cosine)
    i = np.arange(n, dtype=np.float64)
    hann = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    hann /= hann.sum()
    candidates.append(hann)

    # 9. half‑sine shape
    sine = np.sin(np.pi * i / (n - 1))
    sine = np.maximum(sine, 0)
    sine /= sine.sum()
    candidates.append(sine)

    # 10. centred Gaussian‑like bumps with several widths
    mu = (n - 1) / 2.0
    for sigma in (n / 10, n / 6, n / 4):
        shape = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        shape /= shape.sum()
        candidates.append(shape)

    # Normalise and discard pathological entries
    clean = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        clean.append(c / s)
    return clean

def _run_adam(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Adam optimiser on the smooth‑max surrogate of the objective.
    Returns the best sequence seen during the run and its exact objective value.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05                # a bit smaller than the previous code – more stable
    lam = 0.30               # temperature for the smooth‑max (soft‑max)
    lam_factor = 1.02        # slowly increase temperature

    best_a = a0.copy()
    best_val = float('inf')
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # map logits → point on the simplex (soft‑max)
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # exact objective (sum(a)=1 ⇒ denominator drops out)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t}] new best {val:.6f}")

        # smooth‑max weights (soft‑max on the convolution vector)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum != 0:
            w = w / w_sum
        else:
            w = np.full_like(w, 1.0 / w.size)

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # back‑prop through the soft‑max (logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e4)

        # occasional random shake if we are stuck
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val

def _run_subgrad(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Exponentiated‑gradient sub‑gradient phase on the true max‑convolution.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    last_improve = time.time()
    eta0 = 0.55                      # initial step size

    for it in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it}] new best {val:.6f}")

        # active set where the convolution actually attains the maximum
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (classical EG schedule)
        eta = eta0 / np.sqrt(it)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                        # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val

def _local_hill_climb(a0: np.ndarray, n: int, deadline: float,
                     max_trials: int = 200_000):
    """
    Very cheap stochastic local search.
    Repeatedly moves a small mass ε from one index to another and keeps the
    move if it reduces the exact objective.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    best_a = a0.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    no_improve = 0
    trials = 0
    while time.time() < deadline and trials < max_trials and no_improve < 5_000:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0.0, min(0.05, best_a[i]))
        if eps == 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = cand
            no_improve = 0
            print(f"[n={n} HillClimb] improved to {val:.6f}")
        else:
            no_improve += 1
        trials += 1

    return best_a, best_val

def _conv_constraint_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build sparse matrix A (shape (2n‑1, n)) such that

        (A @ g)[k] = (a * g)[k]      (linear convolution of a with variable g)

    This matrix is used in the LP that maximises Σ g_i under the bound
    (a * g)[k] ≤ M for all k.
    """
    n = a.shape[0]
    # each column i contributes a[i] on rows i … i+n‑1
    col_idx = np.repeat(np.arange(n), n)               # length n*n
    row_idx = col_idx + np.tile(np.arange(n), n)        # shifted rows
    data = np.repeat(a, n)
    A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A

def _run_lp_refine(a0: np.ndarray, n: int, deadline: float,
                   max_outer_iters: int = 5):
    """
    One‑step linear‑program refinement.
    Starting from ``a0`` we repeatedly:
        * compute the current bound M = max(a*a)
        * solve   max Σ g_i   s.t.   a * g ≤ M ,  g ≥ 0
        * scale g to the same ℓ₁‑norm as a
        * replace a ← (1‑t)·a + t·g  (t=0.5)
    The best vector encountered is returned together with its exact objective.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    conv_fft_len = _next_pow2(2 * n - 1)

    while time.time() < deadline and max_outer_iters > 0:
        max_outer_iters -= 1

        # current bound
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()
        val = 2.0 * n * M / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()

        # linear programme
        A = _conv_constraint_matrix(a)          # (2n‑1) × n
        b = np.full(2 * n - 1, M, dtype=np.float64)
        # maximise Σ g_i  ⇔ minimise -Σ g_i
        c = -np.ones(n, dtype=np.float64)
        remaining = max(0.0, deadline - time.time())
        opts = {'presolve': True}
        if remaining > 0:
            opts['time_limit'] = remaining
        res = opt.linprog(c, A_ub=A, b_ub=b,
                          bounds=[(0, None)] * n,
                          method='highs', options=opts)
        if not res.success:
            # LP failed – abort further refinements
            break
        g = res.x

        # scale g to the same ℓ₁‑norm as a (so that the denominator does not change)
        sum_a = a.sum()
        sum_g = g.sum()
        if sum_g == 0:
            break
        g = g * (sum_a / sum_g)

        # convex combination
        t = 0.5
        a = (1.0 - t) * a + t * g

    # final evaluation of the last a
    conv = _full_convolution(a, conv_fft_len)
    final_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000):
    """
    Search for a non‑negative step‑function (list of heights) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

    The algorithm:
      1. try several lengths (2048 … 16384);
      2. build a diverse pool of initial vectors (warm‑start, random Dirichlet,
         power‑law, exponentials, Hann, sine, Gaussian, …);
      3. run Adam on a smooth‑max surrogate;
      4. refine with an exponentiated‑gradient sub‑gradient phase;
      5. (only for the smallest length) perform a tiny LP refinement;
      6. finish with a cheap stochastic hill‑climb on the overall best.
    The whole search respects the supplied time budget and returns the best
    sequence encountered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_val = float('inf')

    # lengths – larger ones give more expressive power but cost more time
    lengths = [2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        candidates = _generate_initial_candidates(n, best_seq)

        # iteration budgets (scale roughly with 1/n)
        max_adam = int(25000 * 2048 / n) + 8_000   # a little larger than before
        max_subgrad = 7_000

        for cid, init in enumerate(candidates):
            if time.time() > deadline:
                break

            # ---- Adam (smooth‑max) ----
            a_adam, val_adam = _run_adam(init, n, deadline, max_adam)
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n} cand={cid}] Adam set new global best {val_adam:.6f}")

            # ---- Sub‑gradient (exponentiated‑gradient) ----
            a_sub, val_sub = _run_subgrad(a_adam, n, deadline, max_subgrad)
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[n={n} cand={cid}] Subgrad set new global best {val_sub:.6f}")

            # ---- LP refinement (only for the smallest size) ----
            if n == 2048 and best_seq is not None and len(best_seq) == n:
                a_lp, val_lp = _run_lp_refine(best_seq, n, deadline, max_outer_iters=5)
                if val_lp < best_val:
                    best_val = val_lp
                    best_seq = a_lp.copy()
                    print(f"[n={n} cand={cid}] LP refinement improved global best to {val_lp:.6f}")

            # ---- Short fine‑tune on the current global best for this length ----
            if best_seq is not None and len(best_seq) == n:
                a_fine, val_fine = _run_adam(best_seq, n, deadline,
                                             int(0.5 * max_adam))
                if val_fine < best_val:
                    best_val = val_fine
                    best_seq = a_fine.copy()
                    print(f"[n={n}] Fine‑tune improved global best to {val_fine:.6f}")

    # ------------------------------------------------------------------
    # Final cheap local search to squeeze out any remaining improvement
    if best_seq is not None and time.time() < deadline:
        print(f"\n=== Final local hill‑climb (n={len(best_seq)}) ===")
        a_final, val_final = _local_hill_climb(best_seq, len(best_seq), deadline)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final
            print(f"[final] hill‑climb improved global best to {val_final:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform if nothing was produced (should never happen)
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # expose the best sequence for possible warm‑starts in later runs
    globals()['height_sequence_1'] = best_seq.tolist()

    # Return a plain Python list, as required by the external evaluator
    return best_seq.tolist()
