"""
Optimise a non-negative step function to minimise
F(a) = 2 * n * max(conv(a,a)) / (sum(a)**2).

The method proceeds in several stages:
1. Generate a diverse pool of seeds (uniform, up‑sampled previous best,
   power‑law edge shapes, exponential ramps and a few random Dirichlets).
2. For each seed run a long Adam optimisation on a log‑sum‑exp
   smooth‑max surrogate of the ℓ∞‑norm.  The Adam loop is allowed
   many more iterations for the larger lengths than the reference code.
3. Refine the Adam output with an exact ℓ∞ sub‑gradient step using an
   exponentiated‑gradient update.
4. Starting from the best sequence of the current length, jitter the
   logits several times and repeat steps 2‑3 to escape local minima.
The whole search respects the supplied wall‑clock budget (default 1000 s)
and finally returns the best list that was found.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Core linear‑algebra utilities (FFT‑based convolution / correlation)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using a real FFT of length `fft_len`."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) required for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0 … n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5000,
    lr: float = 0.04,
    lam0: float = 0.5,
    lam_factor: float = 1.015,
) -> tuple[np.ndarray, float]:
    """Adam optimisation on the smooth‑max surrogate of the convolution maximum.
    Returns the best iterate (as a probability vector) and the exact objective value."""
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    lam = lam0
    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max projection onto the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # gradient of smooth‑max with respect to a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # back‑prop through soft‑max (the usual trick for the simplex)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature annealing – slow increase
        lam = min(lam * lam_factor, 1e7)

        # exact objective (sum is 1 by construction)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # occasional random jitter if stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 4000,
    eta: float = 0.45,
) -> tuple[np.ndarray, float]:
    """Exact ℓ∞ sub‑gradient refinement using an exponentiated‑gradient step."""
    a = a0 / a0.sum()
    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        # build a uniform distribution over all maximising indices
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient update on the simplex
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # fallback to a random point on the simplex
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        eta *= 0.9995                     # gradual decay

        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Helper to evaluate the exact objective (no safety checks)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact evaluation of 2·n·max(conv)/sum(a)^2."""
    n = a.shape[0]
    conv = _full_convolution(a, fft_len=1 << int(np.ceil(np.log2(2 * n - 1))))
    max_c = conv.max()
    s = a.sum()
    return 2.0 * n * max_c / (s * s)

"""
Optimise a non‑negative step function to minimise

    F(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

where conv denotes linear convolution.  The algorithm works in four stages:

1.  Build a diverse pool of seeds (uniform, up‑sampled previous best,
    symmetric power‑law “U‑shapes”, exponential ramps, sinusoidal bumps,
    a few random Dirichlet draws and the historic global variable if present).

2.  For each seed run a *long* Adam optimisation on a log‑sum‑exp smooth‑max
    surrogate of the ℓ∞‑norm.  The number of iterations is deliberately larger
    for the larger lengths than the reference implementation.

3.  Refine the Adam output with an exact ℓ∞ sub‑gradient step using an
    exponentiated‑gradient update (the classic EG‑method on the simplex).

4.  Starting from the current best of the given length, jitter its logits several
    times and repeat steps 2‑3 to escape possible local minima.

The routine respects the supplied wall‑clock budget (default 1000 s) and
returns the best sequence found as a plain Python list.  The best sequence is
stored in the global variable ``height_sequence_1`` for possible reuse by a
subsequent call.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution / correlation utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using a real FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) required for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]  for i = 0 … n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Adam optimisation on a smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5000,
    lr: float = 0.04,
    lam0: float = 0.5,
    lam_factor: float = 1.015,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max of the convolution.
    Returns the best iterate (as a probability vector) and its *exact*
    objective value.
    """
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    lam = lam0
    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ---------- projection onto the simplex (soft‑max) ----------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---------- convolution & smooth‑max weights ----------
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # ---------- gradient of the smooth‑max ----------
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # ---------- back‑prop through soft‑max ----------
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---------- temperature annealing ----------
        lam = min(lam * lam_factor, 1e7)

        # ---------- exact objective (sum is 1 by construction) ----------
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # ---------- simple stagnation handling ----------
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact ℓ∞ sub‑gradient refinement (exponentiated gradient)
# ----------------------------------------------------------------------


def _subgradient_refine(
    a0: np.ndarray,

    corr_fft_len: int,
    deadline: float,
    max_iters: int = 4000,

) -> tuple[np.ndarray, float]:

    Exact ℓ∞ sub‑gradient refinement using an exponentiated‑gradient step.
    Returns the refined vector and its exact objective value.
    """
    a = a0 / a0.sum()
    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(max_iters):

                    deadline,
                    max_iters=int(subgrad_iters * 0.3),
                )
                if val_ref < best_val:
                    best_val = val_ref
                    best_seq = a_ref.copy()
                    print(f"[n={n} jitter={jitter_id}] NEW GLOBAL BEST {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback safety net
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 4096
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = _objective(best_seq)
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    # expose for the next round (if any)
    globals()["height_sequence_1"] = best_seq.copy()
    print(f"Final best value = {best_val:.6f}, length = {len(best_seq)}")
    return best_seq.tolist()
