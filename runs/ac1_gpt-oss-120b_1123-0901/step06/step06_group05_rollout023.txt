def _generate_initial_candidates(n, prev_best):
    # produce list up to e.g., 12 shapes
    candidates = []
    # warm start if prev_best
    if prev_best is not None:
        candidates.append(_upsample(prev_best, n))
    # uniform Dirichlet
    candidates.append(np.random.dirichlet(np.ones(n)))
    # arcsine shape
    i = np.arange(n)
    x = (i + 0.5) / n
    shape = 1.0 / np.sqrt(x * (1 - x) + 1e-12)
    shape /= shape.sum()
    candidates.append(shape)
    # beta shapes: (α=0.3, β=0.3) and (α=0.7, β=0.7) etc.
    for a,b in [(0.3,0.3),(0.5,0.5),(0.8,0.8),(1.2,1.2)]:
        shape = ( (i+1) ** (a-1) ) * ( (n-i) ** (b-1) )
        # Note: (i+1) range 1..n, but maybe easier: x**(a-1)*(1-x)**(b-1)
        shape = np.power(i+1, a-1) * np.power(n-i, b-1)
        # Actually using normalized x for numeric stability
        shape = np.power(x, a-1) * np.power(1-x, b-1)
        shape = np.maximum(shape, 0)
        s = shape.sum()
        if s > 0:
            shape /= s
            candidates.append(shape)
    # Kaiser windows with betas = [2,4,6,8]
    from scipy.special import i0
    for beta in (2, 4, 6, 8):
        # Compute Kaiser window:
        # w[n] = i0(beta * sqrt(1 - ((n - (N-1)/2) / ((N-1)/2))**2 )) / i0(beta)
        n_arr = np.arange(n, dtype=np.float64)
        alpha = (n - 1) / 2.0
        ratio = (n_arr - alpha) / alpha
        ratio = np.clip(ratio, -1, 1)
        w = i0(beta * np.sqrt(1 - ratio**2)) / i0(beta)
        w = np.maximum(w, 0)
        w_sum = w.sum()
        if w_sum>0:
            w = w / w_sum
            candidates.append(w)

"""
Propose a step‑function (non‑negative vector whose sum is 1) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)^2)

using a hybrid stochastic‑gradient / Adam optimiser with a richer pool of
initial shapes (Kaiser, Blackman, Nuttall, arcsine, Beta‑shapes, etc.) and a
two‑phase schedule: quick exploration of many candidates, then a long
refinement of the current best.  A cheap hill‑climb is applied at the end.
The routine respects the supplied time budget (default ≈1000 s) and
returns a plain Python list suitable for ``evaluate_sequence``.
"""

import time
import random
import numpy as np
from scipy.special import i0  # Kaiser window

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x (fast integer version)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative vector to ``new_n`` points by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:                     # protect against a zero vector
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s


def _generate_initial_candidates(n: int, prev_best):
    """
    Produce a compact but diverse pool of starting points for a given length ``n``.
    ``prev_best`` (if not ``None``) is up‑sampled and used as a warm‑start.
    """
    candidates = []

    # 1. warm‑start from the best sequence found so far (upsampled)
    if prev_best is not None:
        candidates.append(_upsample(prev_best, n))

    # 2. uniform Dirichlet (conc=1) and a few other concentrations
    candidates.append(np.random.dirichlet(np.ones(n)))                # ≈ uniform
    for conc in (0.05, 0.2, 5.0):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # 3. arcsine‑like density (beta(0.5,0.5))
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    arcsine = 1.0 / np.sqrt(x * (1 - x) + 1e-12)
    arcsine = np.maximum(arcsine, 0.0)
    arcsine /= arcsine.sum()
    candidates.append(arcsine)

    # 4. Beta‑shapes (a=b to keep symmetry)
    for a in (0.3, 0.5, 0.7, 1.2):
        beta_shape = np.power(x, a - 1.0) * np.power(1 - x, a - 1.0)
        beta_shape = np.maximum(beta_shape, 0.0)
        s = beta_shape.sum()
        if s > 0:
            candidates.append(beta_shape / s)

    # 5. Kaiser windows with a handful of betas
    for beta in (2, 4, 6, 8):
        alpha = (n - 1) / 2.0
        ratio = (i - alpha) / alpha
        ratio = np.clip(ratio, -1.0, 1.0)
        wk = i0(beta * np.sqrt(1.0 - ratio ** 2)) / i0(beta)
        wk = np.maximum(wk, 0.0)
        wk /= wk.sum()
        candidates.append(wk)

    # 6. Blackman window
    blackman = 0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1)) + \
               0.08 * np.cos(4 * np.pi * i / (n - 1))
    blackman = np.maximum(blackman, 0.0)
    blackman /= blackman.sum()
    candidates.append(blackman)

    # 7. Nuttall window (very low sidelobes)
    nuttall = (0.355768 - 0.487396 * np.cos(2 * np.pi * i / (n - 1)) +
               0.144232 * np.cos(4 * np.pi * i / (n - 1)) -
               0.012604 * np.cos(6 * np.pi * i / (n - 1)))
    nuttall = np.maximum(nuttall, 0.0)
    nuttall /= nuttall.sum()
    candidates.append(nuttall)

    # 8. Gaussian bumps of several widths (centered)
    mu = (n - 1) / 2.0
    for sigma in (n / 10, n / 6, n / 4):
        gauss = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        gauss = np.maximum(gauss, 0.0)
        gauss /= gauss.sum()
        candidates.append(gauss)

    # Normalise and discard pathological entries
    clean = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        clean.append(c / s)

    # Shuffle and keep a modest number (max 10) to limit total work
    random.shuffle(clean)
    return clean[:10]


# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best sequence seen during the run and its exact value.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    logits = np.log(a0 + 1e-15)                # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05                                 # a safe learning rate
    lam = 0.5                                  # start with a bit more smoothing
    lam_factor = 1.05                         # increase temperature faster

    best_a = a0.copy()
    best_val = float('inf')
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # map logits → simplex (soft‑max)
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # exact objective (sum(a)=1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # back‑prop through soft‑max (logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e4)

        # occasional jitter if stuck
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _run_subgrad(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Exponentiated‑gradient sub‑gradient phase on the true max‑convolution.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    last_improve = time.time()
    eta0 = 0.80                      # a slightly larger initial step size

    for it in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it}] new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (classical EG schedule)
        eta = eta0 / np.sqrt(it)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 8.0:
            # give up on this run if it stalls
            break
        if time.time() > deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _local_hill_climb(a0: np.ndarray, n: int, deadline: float,
                     max_trials: int = 300_000):
    """
    Very cheap stochastic local search.
    Moves a small amount of mass ε from one index to another and
    keeps the move if it reduces the exact objective.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    best_a = a0.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    no_improve = 0
    trials = 0
    while time.time() < deadline and trials < max_trials and no_improve < 10_000:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0.0, min(0.07, best_a[i]))
        if eps == 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = cand
            no_improve = 0
            print(f"[n={n} HillClimb] improved to {val:.6f}")
        else:
            no_improve += 1
        trials += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the evaluation function.  The strategy is:

      1. Try three lengths 4096, 8192, 16384 (larger lengths give more expressive
         power).  For each length we generate a modest and diverse set of seed
         vectors (warm‑start, Dirichlet draws, arcsine, Kaiser, Blackman,
         Nuttall, Gaussian, Beta‑shapes, …).

      2. **Quick phase** – run a short Adam optimisation followed by a few
         sub‑gradient steps on every seed.  This cheaply discovers promising
         regions of the search space.

      3. **Refinement phase** – take the best candidate for the current length
         and run a long Adam run and a long sub‑gradient run to polish it.

      4. After all lengths are processed, perform a final stochastic hill‑climb
         on the overall best vector.

    The whole search respects the supplied time budget.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)               # keep both RNGs in sync
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float('inf')

    # Lengths – we focus on the larger sizes where the bound can be lower.
    lengths = [4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        # ------------------------------------------------------------------
        # 1. generate a modest pool of diverse starting points
        candidates = _generate_initial_candidates(n, best_seq)

        # 2. quick exploration for every candidate
        quick_adam = max(4_000, int(0.15 * (25000 * 2048 / n) + 2_000))
        quick_sub  = max(2_000, int(0.15 * 7_000))

        for cid, init in enumerate(candidates):
            if time.time() > deadline:
                break

            # ---- quick Adam (smooth‑max) ----
            a_adam, val_adam = _run_adam(init, n, deadline, quick_adam)
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n} cand={cid}] Adam quick new global best {val_adam:.6f}")

            # ---- quick sub‑gradient ----
            a_sub, val_sub = _run_subgrad(a_adam, n, deadline, quick_sub)
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[n={n} cand={cid}] Subgrad quick new global best {val_sub:.6f}")

        # ------------------------------------------------------------------
        # 3. fine‑tune the *current* best for this length (if lengths match)
        if best_seq is not None and len(best_seq) == n:
            fine_adam = max(20_000, int(25000 * 2048 / n) + 5_000)
            fine_sub  = max(8_000, int(7_000 * 1.2))

            a_fine, val_fine = _run_adam(best_seq, n, deadline, fine_adam)
            if val_fine < best_val:
                best_val = val_fine
                best_seq = a_fine.copy()
                print(f"[n={n}] Fine‑Adam improved global best to {val_fine:.6f}")

            a_fine2, val_fine2 = _run_subgrad(best_seq, n, deadline, fine_sub)
            if val_fine2 < best_val:
                best_val = val_fine2
                best_seq = a_fine2.copy()
                print(f"[n={n}] Fine‑Subgrad improved global best to {val_fine2:.6f}")

    # ----------------------------------------------------------------------
    # Final cheap local search to squeeze out any remaining improvement
    if best_seq is not None and time.time() < deadline:
        print(f"\n=== Final local hill‑climb (n={len(best_seq)}) ===")
        a_final, val_final = _local_hill_climb(best_seq, len(best_seq), deadline)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final
            print(f"[final] hill‑climb improved global best to {val_final:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – uniform if nothing was produced (should never happen)
    if best_seq is None:
        n_fallback = 4096
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
