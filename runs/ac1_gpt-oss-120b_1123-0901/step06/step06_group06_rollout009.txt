def resize_sequence(seq, new_n):

   old_n = len(seq)

   x_old = np.linspace(0, 1, old_n)

   x_new = np.linspace(0, 1, new_n)

   seq_new = np.interp(x_new, x_old, seq)

   seq_new = np.maximum(seq_new, 0)

   seq_new = seq_new / seq_new.sum()

   return seq_new

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1
    best_seq = None
    best_val = float('inf')
    # lengths list
    lengths = [1024, 2048, 4096, 8192]
    for n in lengths:
        if time.time() > deadline: break
        print(f"--- Optimising length {n} ---")
        conv_len = 1 << int(np.ceil(np.log2(2*n-1)))
        corr_len = 1 << int(np.ceil(np.log2(conv_len + n - 1)))
        # build candidates
        cand = []
        # if we have previous best, resize it
        if best_seq is not None:
            cand.append(resize_sequence(best_seq, n))
        # random Dirichlet variants
        for conc in (0.5, 1.0, 2.0):
            cand.append(np.random.dirichlet(np.full(n, conc)))
        # power-law edges
        idx = np.arange(1, n+1, dtype=np.float64)
        prod = idx * (n - idx + 1)
        shape = np.power(prod, -0.5, where=prod>0, out=np.zeros_like(prod))
        if shape.sum()>0:
            cand.append(shape/shape.sum())
        # Gaussian
        i = np.arange(n, dtype=np.float64)
        sigma = n/6.0
        gauss = np.exp(-0.5*((i-(n-1)/2)/sigma)**2)
        if gauss.sum()>0:
            cand.append(gauss/gauss.sum())
        # maybe add uniform as baseline
        cand.append(np.full(n, 1.0/n))
        # Iterate over candidates
        for cid, a0 in enumerate(cand):
            if time.time() > deadline: break
            a = np.asarray(a0, dtype=np.float64)
            if a.sum() == 0:
                a = np.random.dirichlet(np.ones(n))
            a = a / a.sum()
            # param for simplex via logits
            logits = np.log(a + 1e-15)
            # Adam optimization on smooth max
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1=0.9; beta2=0.999; eps=1e-8
            lr = 0.12 * np.sqrt(2048.0 / n)
            lam = 0.1
            lam_factor = 1.02
            max_iters_adam = int(60000 * 1024 / n) + 5000
            last_improve = time.time()
            for it in range(1, max_iters_adam+1):
                # softmax to get a
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()
                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum
                grad_a = 2.0 * _corr_fft(w, a, corr_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)
                m = beta1 * m + (1 - beta1) * grad_log
                v = beta2 * v + (1 - beta2) * (grad_log ** 2)
                m_hat = m / (1 - beta1**it)
                v_hat = v / (1 - beta2**it)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
                lam = min(lam * lam_factor, 1e6)
                # compute exact objective (since sum=1)
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cid} Adam iter={it}] new best {val:.6f}")
                # random shake if stalled
                if time.time() - last_improve > 5.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()
                if time.time() > deadline:
                    break
            # Exponentiated gradient refinement
            a = a / a.sum()
            eta = 0.6
            sub_last = time.time()
            max_iters_eg = 8000
            for sub_it in range(max_iters_eg):

                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)
                grad_a = 2.0 * _corr_fft(w, a, corr_len)
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:

                    s = a.sum()
                a = a / s
                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last = time.time()
                    print(f"[n={n} cand={cid} EG iter={sub_it}] new best {val:.6f}")
                if time.time() - sub_last > 6.0:
                    break
                if time.time() > deadline:
                    break
            # Local random move search

                    a_loc = np.maximum(a_loc, 0)
                    a_loc = a_loc / a_loc.sum()
                    loc_last = time.time()
                if time.time() > deadline:
                    break
        # End of candidates for length n
        # maybe perform a final fine-tune on global best of this length
        if best_seq is not None and len(best_seq) == n:
            # brief fine-tune using same Adam but fewer iterations
            print(f"[n={n}] final fine‑tune on global best")
            a = best_seq.copy()
            logits = np.log(a + 1e-15)
            m=np.zeros_like(logits); v=np.zeros_like(logits)
            beta1=0.9; beta2=0.999; eps=1e-8
            lr = 0.06 * np.sqrt(2048.0/n)
            lam=0.2; lam_factor=1.015
            max_fine = int(20000 * 1024 / n) + 2000
            last_improve = time.time()
            for it in range(1, max_fine+1):
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()
                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam*(conv-max_c))
                w_sum = w.sum()
                w = w / w_sum if w_sum>0 else np.full_like(w,1.0/w.size)
                grad_a = 2.0 * _corr_fft(w, a, corr_len)
                avg = np.dot(a, grad_a)
                grad_log = a*(grad_a-avg)
                m = beta1*m + (1-beta1)*grad_log
                v = beta2*v + (1-beta2)*(grad_log**2)
                m_hat = m/(1-beta1**it)
                v_hat = v/(1-beta2**it)
                logits -= lr * m_hat / (np.sqrt(v_hat)+eps)
                lam = min(lam * lam_factor, 1e6)
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} fine iter={it}] new best {val:.6f}")
                if time.time() - last_improve > 8.0:
                    logits += (np.random.rand(n)-0.5)*5e-3
                    last_improve = time.time()
                if time.time() > deadline:
                    break
    # fallback uniform if nothing found
    if best_seq is None:
        best_seq = np.full(1024, 1.0/1024)
    # expose global variable for next call
    global height_sequence_1
    height_sequence_1 = best_seq.tolist()
    return best_seq.tolist()

"""
Propose a non‑negative step‑function that (hopefully) beats the previous
best bound 1.508653.  The search works on several lengths (up to 8192),
starts from a rich pool of deterministic shapes plus interpolations of the
current best solution, and refines each candidate with three stages:

1. Adam optimisation on a smooth‑max surrogate of the true objective.
2. Exponentiated‑gradient (sub‑gradient) refinement that directly minimises
   the max‑convolution.
3. A cheap local random‑move search that swaps a tiny amount of mass
   between two positions.

All work is performed on the simplex by optimizing the log‑softmax
(logits) of the sequence, which guarantees non‑negativity and unit sum.
Fast FFT‑based convolution and cross‑correlation give O(n log n)
per iteration.  The routine respects a time‑budget (default 1000 s) and
stores the best sequence in the global variable ``height_sequence_1``
so that later calls can warm‑start from it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Resize ``seq`` to length ``new_n`` by linear interpolation,
    then renormalise to sum‑to‑one.
    """
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    resized = np.interp(x_new, x_old, seq)
    resized = np.maximum(resized, 0.0)
    total = resized.sum()
    if total < 1e-12:
        resized = np.full(new_n, 1.0 / new_n)
    else:
        resized /= total
    return resized


def _initial_candidates(n: int) -> list[np.ndarray]:
    """Create a modest yet diverse set of starting points for length ``n``."""
    cand = []

    # warm‑start from a globally stored solution (if any)
    global height_sequence_1
    if "height_sequence_1" in globals():
        h = np.asarray(height_sequence_1, dtype=np.float64)
        cand.append(_resize_sequence(h, n))

    # a few Dirichlet draws with different concentrations
    for conc in (0.5, 1.0, 2.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # edge‑concentrated power‑law (inverse‑triangular) shape
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    shape = np.power(prod, -0.5, where=prod > 0, out=np.zeros_like(prod))
    if shape.sum() > 0:
        cand.append(shape / shape.sum())

    # discrete Gaussian
    i = np.arange(n, dtype=np.float64)
    sigma = n / 6.0
    gauss = np.exp(-0.5 * ((i - (n - 1) / 2) / sigma) ** 2)
    if gauss.sum() > 0:
        cand.append(gauss / gauss.sum())

    # uniform (the trivial benchmark)
    cand.append(np.full(n, 1.0 / n))

    return cand


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low value of
    F(a) = 2·n·max(a*a) / (sum(a)²).

    The routine respects ``budget_s`` seconds, returns the best sequence
    discovered as a plain Python list, and stores it in the global variable
    ``height_sequence_1`` for possible warm‑starts on later calls.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # lengths to explore – larger lengths give a larger design space
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_len = _next_pow2(2 * n - 1)
        corr_len = _next_pow2(conv_len + n - 1)

        candidates = _initial_candidates(n)

        for cid, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            a = np.asarray(a0, dtype=np.float64)
            if a.sum() < 1e-12:
                a = np.random.dirichlet(np.ones(n))
            a = a / a.sum()
            logits = np.log(a + 1e-15)

            # -----------------------------------------------------------------
            # 1) Adam on a smooth‑max surrogate
            # -----------------------------------------------------------------
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.12 * np.sqrt(2048.0 / n)          # larger for smaller n
            lam = 0.1                                 # start “soft”
            lam_factor = 1.02

            max_adam_iters = int(60000 * 1024 / n) + 5000
            last_improve = time.time()

            for it in range(1, max_adam_iters + 1):
                # simplex via soft‑max
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                conv = _conv_fft(a, conv_len)
                max_c = conv.max()

                # smooth‑max weights
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # gradient of the surrogate (linear in w)
                grad_a = 2.0 * _corr_fft(w, a, corr_len)

                # back‑prop through the soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam step
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # temperature schedule

                # exact objective (sum(a)==1)

            # -----------------------------------------------------------------

            # -----------------------------------------------------------------

            eta = 0.6

            max_sub_iters = 8000

                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                grad_a = 2.0 * _corr_fft(w, a, corr_len)

                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:      # guard against collapse
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last = time.time()
                    print(f"[n={n} cand={cid} EG iter={sub_it:04d}] new best {val:.6f}")

                if time.time() - sub_last > 6.0:

                if time.time() > deadline:

            a_loc = a.copy()

                    best_seq = a_new.copy()
                    a_loc = a_new

                    print(f"[n={n} cand={cid} local step={ls:04d}] new best {val:.6f}")

                    a_loc = np.maximum(a_loc, 0.0)

                    local_last = time.time()

        # Final fine‑tune of the global best for the current length

        if best_seq is not None and len(best_seq) == n:

                avg = np.dot(a, grad_a)
