"""
Propose a low‑evaluation step function.

Algorithm
---------
1. **Rich parametric sampling** – generate a large and diverse set of
   non‑negative vectors (uniform, Dirichlet, edge‑exponential, edge‑power‑law,
   Gaussian edges, sinusoidal, triangular and a few random mixes).  Each
   vector is normalised to have unit sum.  The best few (default 6) are kept
   for deep optimisation.

2. **Smooth‑max Adam optimisation** – for every retained start we run Adam on
   a log‑sum‑exp surrogate of the max‑convolution.  The variables are the
   logits of a soft‑max representation (guaranteeing non‑negativity and
   simplex constraint).  A slowly growing temperature (``lam``) makes the
   surrogate approach the true max.  The Adam schedule is a bit more aggressive
   than before (larger learning rate, slower temperature increase).

3. **Exact sub‑gradient refinement** – after Adam we perform exponentiated‑gradient
   steps using the true active set of the max‑convolution.  This sharpens the
   solution.

4. **Resolution doubling** – the best 4096‑vector is interpolated to a larger
   grid (normally 8192).  The whole deep optimisation (Adam + sub‑gradient) is
   repeated on the larger grid with the remaining time budget.  More resolution
   generally yields a smaller bound.

5. **Time‑budget aware** – the function respects the 1000 s limit.  Time is
   split between candidate generation, deep optimisation of the top starts,
   and the final high‑resolution refinement.

The routine returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Fast objective (sum is assumed close to 1, but we compute the exact denominator)
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Adam optimiser on a smooth‑max surrogate
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
    max_iter: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp surrogate of the max‑convolution.
    Returns the best vector/value seen during the run.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # initialise logits (soft‑max)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.08                               # a touch larger than before

    lam = 0.1                               # start temperature (more smoothing)
    lam_factor = 1.008                      # slower growth → smoother optimisation
    lam_cap = 1e7

    best_seq = global_best_seq
    best_val = global_best_val
    last_improve = time.time()

    it = 0
    while time.time() < deadline and it < max_iter:
        # ---- simplex via soft‑max ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- convolution & exact objective (for logging) ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:07d}] new best {val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        w = np.exp(lam * (conv - max_c))          # all entries ≤ 1
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of surrogate (chain‑rule through soft‑max) ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)            # gradient w.r.t. logits

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, lam_cap)

        # ---- occasional random shake when stuck ----
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (exponentiated‑gradient)
def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_seq: np.ndarray | None,
    init_val: float,
    max_iter: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient steps on the true max‑convolution.
    """
    a = init_seq if init_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()                     # stay on the simplex

    best_seq = a.copy()
    best_val = init_val
    last_improve = time.time()

    eta = 0.6
    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it:05d}] new best {val:.6f}")

        # active set where convolution attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # stop early if stagnant
        if time.time() - last_improve > 8.0:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam (smooth‑max) followed by exact sub‑gradient refinement.
    Updates the globally best vector/value.
    """
    # Adam phase (the internal function respects the deadline)
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
        max_iter=300_000,
    )
    # Sub‑gradient phase
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
        max_iter=30_000,
    )
    return best_seq, best_val

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """
    Produce a large, diverse set of candidate vectors.
    All candidates are normalised to unit sum.
    """
    candidates = []

    # --- Warm start & simple rotations ---
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # --- Uniform baseline ---
    candidates.append(np.full(n, 1.0 / n))

    # --- Random Dirichlet draws with many concentrations ---
    rng = np.random.default_rng()
    for conc in (0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        for _ in range(3):
            cand = rng.dirichlet(np.full(n, conc))
            candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # --- Edge‑exponential families (mix with uniform) ---
    alphas = np.logspace(-2, 1.0, num=36)          # 0.01 … 10 (finer grid)
    for alpha in alphas:
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in np.linspace(0, 1, 11):           # mixing weight
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # --- Edge‑power‑law families ---
    ps = np.linspace(0.2, 5.0, num=30)
    for p in ps:
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in np.linspace(0, 1, 7):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # --- Gaussian‑edge families (mix with uniform) ---
    gammas = np.logspace(-2, 0.7, num=20)         # 0.01 … 5
    for gamma in gammas:
        edge = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        edge /= edge.sum()
        for w in np.linspace(0, 1, 9):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # --- Sinusoidal half‑wave (already used) ---
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # --- Triangular bump centred in the middle (for completeness) ---
    tri = np.minimum(i + 1, n - i)
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    candidates.append(tri)

    # --- Random convex combos of two Dirichlet vectors ---
    for _ in range(30):
        a = rng.dirichlet(np.ones(n))
        b = rng.dirichlet(np.ones(n))
        t = rng.random()
        cand = t * a + (1 - t) * b
        candidates.append(cand)

    # Ensure all candidates are proper probability vectors
    clean = []
    for cand in candidates:
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:                     # fallback to a fresh Dirichlet
            cand = rng.dirichlet(np.ones(n))
            s = cand.sum()
        clean.append(cand / s)

    return clean

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a low‑evaluation‑score step function.
    Returns the best sequence found (plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Base resolution (n = 4096) – a size that still runs fast enough
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start from the previous global solution if available
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a large pool of candidates and keep the top few
    all_candidates = _generate_parametric_candidates(n, warm_start)

    # quick evaluation of each candidate (using the same FFT length as the optimiser)
    cand_vals = []
    for idx, cand in enumerate(all_candidates):
        if time.time() >= deadline:
            break
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))
        # optional: print progress every few hundred candidates
        if (idx + 1) % 200 == 0:
            print(f"Evaluated {idx + 1}/{len(all_candidates)} candidates, current best {min(v for v, _ in cand_vals):.6f}")

    # keep only the best K candidates for the expensive optimisation
    cand_vals.sort(key=lambda x: x[0])
    top_k = 6
    top_candidates = [c for _, c in cand_vals[:top_k]]
    print(f"Selected top {len(top_candidates)} candidates for deep optimisation (best quick val {cand_vals[0][0]:.6f})")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation on the selected starts
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation on start #{start_id + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )
        print(f"After start #{start_id + 1}: best objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – if time permits, upscale to a larger grid (n = 8192) and refine
    time_left = deadline - time.time()
    if time_left > 40.0 and best_seq is not None:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation to the finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print(f"--- Optimising on larger grid (n={n_big}) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big  # remember final size

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded, return a uniform vector.
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    elapsed = time.time() - start_time
    print(f"*** Finished search (elapsed {elapsed:.1f}s): best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
