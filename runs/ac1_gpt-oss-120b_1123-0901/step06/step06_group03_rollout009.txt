"""
propose_candidate – multi‑scale, LBFGS‑+‑Adam search for a step‑function.
The algorithm works in three phases:

1. **Coarse exploration (n≈1024).**
   A large, diverse pool of candidate shapes (uniform, power‑law,
   double‑exponential, Gaussian, sinusoidal, cosine‑taper, sinc, random
   Dirichlet) is built.  All candidates are normalised to sum‑one and their
   exact objective (via an FFT‑based convolution) is evaluated.  The best
   few (default 12) are kept.

2. **Local refinement with L‑BFGS.**
   Each kept candidate is optimised on a smooth‑max surrogate of the
   convolution peak using `scipy.optimize.minimize` (L‑BFGS‑B) on the
   soft‑max (log‑softmax) parametrisation of the simplex.  The smoothness
   temperature is fixed (λ≈30) which gives a good approximation of the true
   max while keeping gradients stable.  After every optimisation the exact
   objective is recomputed and the global best is updated.

3. **Upsampling + deep polishing.**
   The current best shape is interpolated to a larger support (4096 or
   8192 points) and further refined by the original Adam → sub‑gradient
   routine (`_deep_optimize`).  This phase runs until the global deadline
   (≈1000 s) is nearly reached.

All numbers are forced to be non‑negative and ≤1000 as required by the
evaluator.  The function returns the best sequence encountered as a plain
Python list.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for gradient of the max‑convolution."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation: 2·n·max(a*a) / (∑a)² (∑a is 1 after normalisation)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _softmax(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    m = logits.max()
    e = np.exp(logits - m)
    return e / e.sum()


def _generate_candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse set of start vectors (including many parametric families)."""
    pool: list[np.ndarray] = []

    # warm‑start and its rotations
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # uniform baseline
    pool.append(np.full(n, 1.0 / n))

    i = np.arange(n, dtype=np.float64)

    # power‑law edge‑concentrated shapes (beta‑like)
    idx = i + 1.0
    prod = idx * (n - idx + 1.0)          # i·(n‑i+1)
    for alpha in np.linspace(-2.0, -0.2, 15):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # double‑exponential (two‑sided Laplace) families
    for lam in np.linspace(0.1, 3.0, 15):
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        pool.append(shape)

    # Gaussian‑type bumps centred in the middle (width sweep)
    for sigma in np.linspace(n * 0.03, n * 0.4, 10):
        shape = np.exp(-0.5 * ((i - n / 2) ** 2) / (sigma ** 2))
        shape /= shape.sum()
        pool.append(shape)

    # single half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # cosine‑taper (raised cosine)
    shape = (1 + np.cos(2 * np.pi * (i - n / 2) / n)) / 2.0
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # sinc bump (non‑negative part)
    shape = np.sinc((i - n / 2) / (n / 10))
    shape = np.maximum(shape, 0.0)
    if shape.sum() > 0:
        shape /= shape.sum()
        pool.append(shape)

    # random Dirichlet draws with varied concentration
    for alpha in np.logspace(-2, 0, 12):
        shape = np.random.dirichlet(np.full(n, alpha))
        pool.append(shape)

    return pool


# ----------------------------------------------------------------------
def _lbfgs_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
    maxiter: int = 1500,
) -> tuple[np.ndarray, float]:
    """
    Optimise a0 on a smooth‑max surrogate using L‑BFGS‑B.
    Returns possibly improved (global_best_seq, global_best_val).
    """
    eps = 1e-12
    logits = np.log(a0 + eps)  # start in log‑softmax space

    lam = 30.0  # temperature for smooth‑max (higher → closer to true max)

    # objective+gradient function required by scipy.optimize
    def fun_and_grad(logits_vec: np.ndarray):
        # soft‑max -> simplex
        a = _softmax(logits_vec)

        # convolution
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()

        # smooth‑max weights (log‑sum‑exp trick)
        w_exp = np.exp(lam * (conv - max_c))
        w_sum = w_exp.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w_exp / w_sum

        # gradient of smooth‑max w.r.t a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule to logits (soft‑max Jacobian)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # smooth‑max surrogate value (1/lam * log‑sum‑exp)
        # using log‑sum‑exp stability
        log_sum = max_c + np.log(np.exp(lam * (conv - max_c)).sum())
        smooth_val = (1.0 / lam) * log_sum
        # Full objective (sum(a)=1) is 2·n·smooth_val
        obj = 2.0 * n * smooth_val
        return obj, grad_log.astype(np.float64)

    # minimise
    res = _opt.minimize(
        fun_and_grad,
        logits,
        method="L-BFGS-B",
        jac=True,
        options=dict(maxiter=maxiter, maxcor=5, disp=False),
    )
    a_opt = _softmax(res.x)

    # compute exact objective (the evaluator’s metric)
    val = _objective(a_opt, n, conv_fft_len)

    if val < global_best_val:
        print(f"[LBFGS] new best {val:.6f} (iters={res.nit})")
        return a_opt, val
    return global_best_seq, global_best_val


# ----------------------------------------------------------------------
def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate followed by an exponentiated‑gradient
    sub‑gradient phase (as in the reference implementation).  The routine
    updates the global best sequence/value while respecting the overall
    deadline.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5          # start temperature for the smooth‑max
    lam_factor = 1.018

    last_improve = time.time()
    it = 0

    # --------------------------- Adam phase ---------------------------
    while time.time() < deadline:
        # soft‑max → simplex
        a = _softmax(logits)

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase the temperature (approach the true max)
        lam = min(lam * lam_factor, 1e6)

        # occasional random “shake” when we have been stagnant
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        if it > 500_000:          # safety guard
            break

    # -------------------- Sub‑gradient phase (exact max) --------------------
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total

    eta0 = 0.6
    sub_last_improve = time.time()
    for sub_it in range(30_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[n={n} Subgrad iter={sub_it:05d}] new best {val:.6f}")

        # active‑set mask (all indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (mirror descent)
        eta = eta0 / np.sqrt(sub_it + 1)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                 # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 6.0:
            break

    return global_best_seq, global_best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the metric
    defined in `evaluate_sequence`.  The search proceeds through a
    coarse sweep, L‑BFGS refinement and a final deep‑optimisation phase.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1) Coarse stage (moderate support)
    n0 = 1024
    conv_fft_len0 = _next_pow2(2 * n0 - 1)
    corr_fft_len0 = _next_pow2(3 * n0 - 2)

    # Warm start (downsample if we have a previous construction)
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        w_big = np.asarray(warm_start, dtype=np.float64)
        # down‑sample to n0 by simple interpolation
        x_big = np.linspace(0.0, 1.0, w_big.shape[0], endpoint=False)
        x_small = np.linspace(0.0, 1.0, n0, endpoint=False)
        w = np.interp(x_small, x_big, w_big)
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
        else:
            w = None
    else:
        w = None

    # generate diverse candidates
    candidates = _generate_candidate_pool(n0, w)

    # quick screening (exact objective)
    screened = []
    best_val = float("inf")
    best_seq = None
    for idx, cand in enumerate(candidates):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n0))
            s = a.sum()
        a = a / s
        val = _objective(a, n0, conv_fft_len0)
        screened.append((val, a))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        print(f"[screen {idx:03d}] obj={val:.6f}")

    # keep the most promising starts (up to 12)
    screened.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in screened[:12]]

    # ------------------------------------------------------------------
    # 2) L‑BFGS local refinement (still at coarse resolution)
    for cid, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        # allocate a modest budget per candidate (maxiter 1200)
        print(f"[L‑BFGS start {cid}]")
        best_seq, best_val = _lbfgs_optimize(
            a0,
            n0,
            conv_fft_len0,
            corr_fft_len0,
            best_seq,
            best_val,
            maxiter=1200,
        )

    # ------------------------------------------------------------------
    # 3) Upsample to a larger grid and deep‑optimize
    if time.time() + 12 < deadline:
        # decide target size (4096 if still enough time, else 8192)
        n_big = 4096 if time.time() + 30 < deadline else 2048
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # interpolate linearly to new grid
            x_old = np.linspace(0.0, 1.0, best_seq.shape[0], endpoint=False)
            x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # a quick L‑BFGS pass on the larger grid (more iterations)
        print("[Upsampled L‑BFGS] start")
        best_seq, best_val = _lbfgs_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            best_seq,
            best_val,
            maxiter=2000,
        )

        # final polishing with the Adam + sub‑gradient routine
        if time.time() + 2 < deadline:
            print("[final deep optimisation]")
            best_seq, best_val = _deep_optimize(
                best_seq,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                deadline,
                best_seq,
                best_val,
            )

    # ------------------------------------------------------------------
    # 4) Safety fall‑back
    if best_seq is None:
        best_seq = np.full(1024, 1.0 / 1024, dtype=np.float64)

    # Clip to evaluator limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return best_seq.tolist()
