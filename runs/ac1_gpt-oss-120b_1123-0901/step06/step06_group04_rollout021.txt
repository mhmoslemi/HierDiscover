import time
import math
import numpy as np

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _full_convolution(a, fft_len=None):
    """Linear convolution of a with itself via FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * n - 1]

def _cross_correlation(w, a, fft_len):
    """Cross‑correlation needed for gradient of Σ w_i (a*a)_i."""
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

def _compute_F(seq):
    """Fast evaluation of 2·n·max(conv(seq,seq))/sum(seq)^2."""
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    if n == 0:
        return float('inf')
    total = a.sum()
    if total < 1e-12:
        return float('inf')
    a = a / total
    conv = _full_convolution(a)
    max_c = conv.max()
    return 2.0 * n * max_c

def _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline):
    """
    Adam optimisation on a smooth‑max surrogate of max(conv).
    Returns best vector (sum=1) and its exact objective value.
    """
    eps = 1e-12
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    lr = 0.05
    lam = 1.0
    lam_factor = 1.02

    # iteration budget generous – will stop early if deadline approaches
    max_iters = int(200000 * 1024 / n) + 5000

    best_val = float('inf')
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max to keep a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # gradient w.r.t. a of Σ w_i conv_i
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # optional gradient clipping for stability
        grad_log = np.clip(grad_log, -10.0, 10.0)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # temperature (smooth‑max) annealing
        lam = min(lam * lam_factor, 1e8)

        # exact objective (since a sums to 1)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # stagnation handling: small random perturbation of logits
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val

def _subgradient_refine(a0, n, conv_fft_len, corr_fft_len, deadline):
    """
    Exact ℓ∞ sub‑gradient refinement via exponentiated‑gradient updates.
    Returns best vector (sum=1) and its exact objective value.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() == 0.0:
        a = np.random.dirichlet(np.ones(n))
    else:
        a = a / a.sum()

    best_a = a.copy()
    conv = _full_convolution(a, fft_len=conv_fft_len)
    best_val = 2.0 * n * conv.max()
    eta = 0.45
    last_improve = time.time()
    max_iters = 8000

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        # active set tolerance (absolute)
        active = np.isclose(conv, max_c, atol=1e-12)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        conv = _full_convolution(a, fft_len=conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # decay step size slowly
        eta *= 0.9995

        # early exit if no progress for a while
        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val

def _generate_candidate_shapes(prev_seq, n):
    """
    Produce a diverse list of seeds for length n.
    prev_seq may be None.
    Returns a list of numpy arrays (each normalized to sum 1).
    """
    candidates = []

    # 1. uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2. up/down‑sample previous best if available
    if prev_seq is not None:
        old = np.asarray(prev_seq, dtype=np.float64)
        if old.shape[0] == n:
            candidates.append(old.copy())
        else:
            # linear interpolation to target size
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, old)
            # fix possible tiny negatives caused by interpolation noise
            up = np.maximum(up, 0.0)
            if up.sum() > 0:
                up = up / up.sum()
            else:
                up = np.random.dirichlet(np.ones(n))
            candidates.append(up)

    # 3. symmetric Beta‑shape (parameter α) – prod^{α}
    idx = np.arange(1, n + 1, dtype=np.float64) - 0.5
    prod = idx * (n - idx + 0.0)
    alphas = [-2.0, -1.5, -1.2, -0.8, -0.5, -0.2, 0.0, 0.2, 0.5, 0.8, 1.0, 1.2, 1.5, 2.0]
    for alpha in alphas:
        shape = np.power(prod, alpha)
        # protect against overflow/underflow
        shape = np.where(np.isfinite(shape), shape, 0.0)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            candidates.append(shape)

    # 4. sine‑squared bump (already works well)
    sine = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    sine = sine / sine.sum()
    candidates.append(sine)

    # 5. Hann window (0.5 - 0.5*cos)
    i = np.arange(n, dtype=np.float64)
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    hann = hann / hann.sum()
    candidates.append(hann)

    # 6. Parabolic “flat‑top” shape: (1 - ((2*i/(n-1))-1)^2)^{gamma}
    gamma_vals = [0.5, 1.0, 2.0]
    for gamma in gamma_vals:
        x = (2.0 * i / (n - 1)) - 1.0
        base = 1.0 - x * x
        base = np.clip(base, 0.0, None)
        shape = np.power(base, gamma)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            candidates.append(shape)

    # 7. a few random Dirichlet draws
    for _ in range(3):
        candidates.append(np.random.dirichlet(np.ones(n)))

    # Limit total number for very large n (to keep runtime reasonable)
    max_cand = 12 if n <= 32768 else 8
    if len(candidates) > max_cand:
        candidates = candidates[:max_cand]

    return candidates

"""
Optimises a non‑negative step function to minimise

    F(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

by a two‑phase scheme:

1. Adam on a smooth‑max surrogate (log‑sum‑exp) with a soft‑max
   parametrisation guaranteeing non‑negativity and unit sum.
2. Exact ℓ∞ sub‑gradient refinement using an exponentiated‑gradient update.

A diverse pool of seeds (uniform, up‑sampled previous best,
symmetric Beta‑shapes, sinusoidal, Hann, flat‑top, random Dirichlet)
is processed for several lengths, with a generous iteration budget
(compared to the reference implementation) and additional jittering
of the current best.  The routine respects the supplied wall‑clock
budget and returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a * a using an FFT (returns length 2·len(a)-1)."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) required for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]  for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Objective evaluation (fast, for internal bookkeeping)
# ----------------------------------------------------------------------


def _compute_F(seq) -> float:
    """Fast evaluation of 2·n·max(conv)/sum(seq)^2 (sum is normalised to 1)."""
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    if n == 0:
        return float("inf")
    total = a.sum()
    if total < 1e-12:
        return float("inf")
    a = a / total
    conv = _full_convolution(a)
    max_c = conv.max()
    return 2.0 * n * max_c


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max surrogate.
    Returns (best_a, best_val) where best_val = 2·n·max(conv) for the best
    iterate discovered before the deadline.
    """
    eps = 1e-12
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    lr = 0.05
    lam = 1.0
    lam_factor = 1.02

    # generous iteration budget – will stop early if the wall‑clock deadline is hit
    max_iters = int(200_000 * 1024 / n) + 5_000

    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # --- simplex via soft‑max ---
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # --- convolution & smooth‑max weights ---
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # --- gradient of smooth‑max wrt a ---
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # --- back‑propagation through soft‑max ---
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Optional clipping for stability
        grad_log = np.clip(grad_log, -10.0, 10.0)

        # --- Adam update on logits ---
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # --- temperature annealing ---
        lam = min(lam * lam_factor, 1e8)

        # --- exact objective (since a sums to 1) ---
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # --- stagnation handling (small random jitter) ---
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Exact ℓ∞ sub‑gradient refinement via an exponentiated‑gradient step.
    Returns (best_a, best_val) after at most a few thousand iterations or
    when the deadline is reached.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() == 0.0:
        a = np.random.dirichlet(np.ones(n))
    else:
        a = a / a.sum()

    best_a = a.copy()
    conv = _full_convolution(a, fft_len=conv_fft_len)
    best_val = 2.0 * n * conv.max()
    eta = 0.45
    last_improve = time.time()
    max_iters = 8_000

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        conv = _full_convolution(a, fft_len=conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        eta *= 0.9995  # slow decay

        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Candidate generation
# ----------------------------------------------------------------------


def _generate_candidate_shapes(prev_seq, n):
    """
    Produce a heterogeneous pool of seeds for a given length n.
    The returned list contains Numpy arrays each already normalised to sum 1.
    """
    candidates = []

    # 1) uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) warm‑start / up‑sample from a previous solution
    if prev_seq is not None:
        old = np.asarray(prev_seq, dtype=np.float64)
        if old.shape[0] == n:
            candidates.append(old.copy())
        else:
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, old)
            up = np.maximum(up, 0.0)
            if up.sum() > 0:
                up = up / up.sum()
            else:
                up = np.random.dirichlet(np.ones(n))
            candidates.append(up)

    # 3) symmetric Beta‑type shapes: (i·(n‑i))^{α}
    idx = np.arange(1, n + 1, dtype=np.float64) - 0.5
    prod = idx * (n - idx + 0.0)
    alphas = [-2.0, -1.5, -1.2, -0.8, -0.5, -0.2,
              0.0, 0.2, 0.5, 0.8, 1.0, 1.2, 1.5, 2.0]
    for alpha in alphas:
        shape = np.power(prod, alpha)
        shape = np.where(np.isfinite(shape), shape, 0.0)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            candidates.append(shape)

    # 4) sine‑squared bump
    sine = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    sine = sine / sine.sum()
    candidates.append(sine)

    # 5) Hann window (0.5‑0.5·cos)
    i = np.arange(n, dtype=np.float64)
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    hann = hann / hann.sum()
    candidates.append(hann)

    # 6) flat‑top parabolic shapes: (1‑((2i/(n‑1))‑1)²)^{γ}
    for gamma in (0.5, 1.0, 2.0):
        x = (2.0 * i / (n - 1)) - 1.0
        base = 1.0 - x * x
        base = np.clip(base, 0.0, None)
        shape = np.power(base, gamma)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            candidates.append(shape)

    # 7) a few random Dirichlet draws
    for _ in range(3):
        candidates.append(np.random.dirichlet(np.ones(n)))

    # Limit the total number (keeps the run‑time reasonable)
    max_cand = 12 if n <= 32768 else 8
    if len(candidates) > max_cand:
        candidates = candidates[:max_cand]

    return candidates
