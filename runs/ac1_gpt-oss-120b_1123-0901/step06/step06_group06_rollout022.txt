def _local_search(a_start, best_val, deadline, max_iters=200000):
    a = a_start.copy()
    n = a.shape[0]
    conv_len = 2*n-1
    fft_len = 1 << int(np.ceil(np.log2(conv_len)))
    sigma = 0.1
    T = best_val * 0.01  # some scale
    decay = 0.9999
    iter = 0
    while time.time() < deadline and iter < max_iters:
        # propose neighbor
        delta = sigma * (np.random.rand(n) - 0.5)
        a_new = a + delta
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-12:
            continue
        a_new /= s
        val_new = _objective(a_new, n, fft_len)
        if val_new < best_val:
            best_val = val_new
            a = a_new
            sigma = max(sigma * 0.9, 1e-5)
            print(f"Local search iter {iter} new best {best_val:.6f}")
        else:
            # Simulated annealing acceptance
            if np.random.rand() < np.exp(-(val_new - best_val) / T):
                a = a_new
        T *= decay
        sigma *= 0.9995
        iter += 1
    return a

def _adam_optimize(a_init, n, conv_len_fft, corr_len_fft, max_iters, deadline, lr, lam0, lam_factor):
    a = a_init.copy()
    # ensure sum=1
    a = a / a.sum()
    logits = np.log(a + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lam = lam0
    best_val = float('inf')
    best_a = a.copy()
    for t in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # softmax
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()
        # conv
        conv = _conv_fft(a, conv_len_fft)
        max_c = conv.max()
        # smooth max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        # grad
        grad_a = 2.0 * _corr_fft(w, a, corr_len_fft)
        # backprop softmax
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # temperature schedule
        lam = min(lam * lam_factor, 1e6)
        # objective
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # optional random shake if stagnation?
        # Not needed.
    return best_a, best_val
```

"""
Propose a non‑negative step‑function (list of heights) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

using a multi‑stage strategy:

1.  Build a diversified pool of deterministic initial shapes (Dirichlet,
    power‑law, exponential, tent, cosine, Gaussian, arcsine, sparse spikes,
    plus an optional warm‑start from the global variable `height_sequence_1`).

2.  For each candidate run a two‑phase continuous optimisation
    – Adam on a smooth‑max surrogate of the convolution max,
    – a sub‑gradient (exponentiated‑gradient) phase that drives the true max
      to a (local) minimum.

3.  Keep the best ~5 candidates in a small “elite” pool and repeatedly
    generate mixed offspring (convex combinations of two elite members with
    a little noise).  Each offspring is refined with a shortened Adam+EG run.

4.  Upscale the current smallest‑length best sequence to a longer length
    (4096 → 8192) by linear interpolation and repeat steps 2‑3 on the larger
    problem.

5.  Finally perform a simple simulated‑annealing hill‑climb on the best
    sequence found so far.

The routine respects the supplied time budget (default 1000 s) and returns
the best sequence as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution / cross‑correlation utilities
def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]
    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact value of the objective (no clamping)."""
    conv = _conv_fft(a, fft_len)
    return 2.0 * n * conv.max()

# Candidate‑generation helpers
def _build_initial_candidates(n: int) -> list[np.ndarray]:
    """Diversified start‑up vectors, all normalised to sum‑to‑1."""
    cand = []

    # warm‑start from a previously found solution if available
    if "height_sequence_1" in globals():
        h = np.asarray(height_sequence_1, dtype=np.float64)
        if h.shape[0] > n:
            h = h[:n]
        else:
            h = np.pad(h, (0, n - h.shape[0]))
        h = np.maximum(h, 0.0)
        if h.sum() < 1e-8:
            h = np.random.dirichlet(np.ones(n))
        else:
            h = h / h.sum()
        cand.append(h)

    # Dirichlet families
    cand.append(np.random.dirichlet(np.ones(n)))
    for conc in (0.2, 0.4, 0.6, 0.8):
        cand.append(np.random.dirichlet(np.full(n, conc)))
    for conc in (1.5, 2.0, 3.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # Edge‑concentrated power‑law shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 9):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape = shape / shape.sum()
            cand.append(shape)

    # Two‑sided exponential decay
    for gamma in (0.0008, 0.0015, 0.003, 0.006):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape = shape / shape.sum()
        cand.append(shape)

    # Smooth tent
    left = np.linspace(0, 1, n // 2, endpoint=False)
    right = left[::-1]
    tent = np.concatenate([left, right])
    if tent.shape[0] < n:
        tent = np.concatenate([tent, [0.0]])
    tent = np.maximum(tent, 0)
    tent = tent / tent.sum()
    cand.append(tent)

    # Cosine window
    i = np.arange(n, dtype=np.float64)
    coswin = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    coswin = coswin / coswin.sum()
    cand.append(coswin)

    # Half‑sine window
    sinwin = np.sin(np.pi * i / (n - 1))
    sinwin = sinwin / sinwin.sum()
    cand.append(sinwin)

    # Discrete Gaussian
    sigma = n / 8.0
    gauss = np.exp(-0.5 * ((i - (n - 1) / 2) / sigma) ** 2)
    gauss = gauss / gauss.sum()
    cand.append(gauss)

    # Truncated arcsine
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcs = arcs / arcs.sum()
    cand.append(arcs)

    # Sparse combinatorial pattern (≈ sqrt(n) spikes)
    m_spikes = int(np.sqrt(2 * n))
    sparse = np.zeros(n, dtype=np.float64)
    indices = np.random.choice(n, size=m_spikes, replace=False)
    sparse[indices] = 1.0
    if sparse.sum() > 0:
        sparse = sparse / sparse.sum()
        cand.append(sparse)

    return cand


# ----------------------------------------------------------------------
# Optimisers (Adam on smooth‑max surrogate + exponentiated‑gradient)
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   fft_len_conv: int,
                   fft_len_corr: int,
                   max_iters: int,
                   deadline: float,
                   lr: float,
                   lam0: float,
                   lam_factor: float) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max surrogate. Returns best (a,val) seen."""
    a = a0.copy()
    a = a / a.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lam = lam0

    best_val = float('inf')
    best_a = a.copy()

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # soft‑max (simple projection onto the simplex)
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # convolution and smooth‑max weights
        conv = _conv_fft(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _corr_fft(w, a, fft_len_corr)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule – approach the true max
        lam = min(lam * lam_factor, 1e6)

        # exact objective (sum(a) == 1)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()

    return best_a, best_val


def _eg_optimize(a0: np.ndarray,
                 n: int,
                 fft_len_conv: int,
                 fft_len_corr: int,
                 max_iters: int,
                 deadline: float,
                 eta: float) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient phase (exact max)."""
    a = a0.copy()
    a = a / a.sum()

    best_val = float('inf')
    best_a = a.copy()

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_fft(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad_a = 2.0 * _corr_fft(w, a, fft_len_corr)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # protect against collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()

    return best_a, best_val


def _run_optimizations(a0: np.ndarray,
                       n: int,
                       deadline: float,
                       max_iters_adam: int,
                       max_iters_eg: int,
                       lr: float,
                       lam0: float,
                       lam_factor: float,
                       eta: float) -> tuple[np.ndarray, float]:
    """Adam + EG (with early‑stop on the global deadline)."""
    conv_len = 2 * n - 1
    fft_len_conv = 1 << int(np.ceil(np.log2(conv_len)))
    fft_len_corr = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

    a_adam, val_adam = _adam_optimize(a0, n,
                                      fft_len_conv,
                                      fft_len_corr,
                                      max_iters_adam,
                                      deadline,
                                      lr,
                                      lam0,
                                      lam_factor)

    # If there is still time, run the EG phase starting from the best Adam point
    if time.time() < deadline:
        a_eg, val_eg = _eg_optimize(a_adam, n,
                                    fft_len_conv,
                                    fft_len_corr,
                                    max_iters_eg,
                                    deadline,
                                    eta)
    else:
        a_eg, val_eg = a_adam, val_adam

    if val_eg < val_adam:
        return a_eg, val_eg
    else:
        return a_adam, val_adam

# Helper for up‑scaling a sequence to a larger length
def _upscale(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(a) → new_n, re‑normalised to sum‑1."""
    old_n = a.shape[0]
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new

    # try a few mixed offspring while time permits
    while time.time() < deadline:
        for _ in range(3):
            if time.time() >= deadline:
                break
            i, j = np.random.choice(len(pool), size=2, replace=False)
            w = np.random.rand()
            a_mix = w * pool[i][1] + (1.0 - w) * pool[j][1]

            # tiny random multiplicative noise to escape plateaus
            a_mix = a_mix * (1.0 + 0.04 * (np.random.rand(n) - 0.5))
            a_mix = np.maximum(a_mix, 0.0)
            a_mix = a_mix / a_mix.sum()

            a_opt, val_opt = _run_optimizations(a_mix, n, deadline,
                                                max_iters_adam, max_iters_eg,
                                                lr, lam0, lam_factor, eta)

            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[mix] new best {val_opt:.6f}")

            # insert into elite pool
            pool.append((val_opt, a_opt.copy()))
            pool.sort(key=lambda x: x[0])
            if len(pool) > 5:
                pool = pool[:5]

        # one outer iteration is enough – the loop will terminate once the
        # deadline is reached
        break

    return best_seq, best_val, pool

# Small simulated‑annealing hill‑climb on the current best sequence
def _local_search(a_start: np.ndarray,
                  best_val_start: float,
                  deadline: float) -> np.ndarray:
    """Very simple random‑walk optimiser with an annealed acceptance rule."""
    a = a_start.copy()
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << int(np.ceil(np.log2(conv_len)))

    best_val = best_val_start
    best_seq = a.copy()

    sigma = 0.02      # scale of additive perturbations
    T = 0.005         # temperature for occasional uphill moves
    decay_T = 0.9995
    it = 0
    max_iters = 200000

    while time.time() < deadline and it < max_iters:
        delta = sigma * (np.random.rand(n) - 0.5)   # additive noise
        a_new = a + delta
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            it += 1
            continue
        a_new = a_new / s
        val_new = _objective(a_new, n, fft_len)

        if val_new < best_val:
            best_val = val_new
            best_seq = a_new.copy()
            a = a_new
            sigma = max(sigma * 0.94, 1e-5)
            T = max(T * 0.94, 1e-7)
            print(f"[local] iter {it:06d} new best {best_val:.6f}")
        else:
            # Metropolis‑type acceptance
            if np.random.rand() < np.exp(-(val_new - best_val) / T):
                a = a_new

        T *= decay_T
        sigma *= 0.9999
        it += 1

    return best_seq

# Main entry point
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the evaluation function.  The function respects the time budget and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a tiny safety margin

    # ------------------------------------------------------------------
    # Stage 1 – optimise a medium size problem (n = 4096)
    n_small = 4096
    best_seq, best_val, elite = _optimize_length(n_small, deadline)

    # Stage 2 – mixed‑offspring refinement (if time allows)
    if time.time() < deadline:
        mix_seq, mix_val, elite = _mix_and_refine(elite, n_small, deadline)
        if mix_seq is not None and mix_val < best_val:
            best_seq, best_val = mix_seq, mix_val

    # ------------------------------------------------------------------
    # Stage 3 – upscale to a larger length (n = 8192) and repeat the process
    n_big = 8192
    if time.time() < deadline:
        if best_seq is not None and len(best_seq) == n_small:
            up_seq = _upscale(best_seq, n_big)
        else:
            up_seq = np.random.dirichlet(np.ones(n_big))

        big_seq, big_val, elite_big = _optimize_length(n_big, deadline, warm_seq=up_seq)

        if big_seq is not None and big_val < best_val:
            best_seq, best_val = big_seq, big_val
            elite = elite_big

        # a short mixing phase on the large problem
        if time.time() < deadline:
            mix_seq_big, mix_val_big, elite = _mix_and_refine(elite, n_big, deadline)
            if mix_seq_big is not None and mix_val_big < best_val:
                best_seq, best_val = mix_seq_big, mix_val_big

    # ------------------------------------------------------------------
    # Stage 4 – final local stochastic hill‑climb on the current best
    if best_seq is not None and time.time() < deadline:
        refined = _local_search(best_seq, best_val, deadline)
        # evaluate the refined sequence
        conv_len = 2 * refined.shape[0] - 1
        fft_len = 1 << int(np.ceil(np.log2(conv_len)))
        final_val = _objective(refined, refined.shape[0], fft_len)
        if final_val < best_val:
            best_seq = refined
            best_val = final_val
            print(f"[final] improved to {final_val:.6f}")

    # ------------------------------------------------------------------
    # Return the (normalised) best sequence as a plain list
    if best_seq is None:
        # a safe fallback – uniform distribution over a modest size
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    best_seq = best_seq / best_seq.sum()          # protect against tiny drift
    return best_seq.tolist()
