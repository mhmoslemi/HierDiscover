"""
Propose a step‑function (non‑negative height sequence) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

where “a*a” is the linear convolution of a with itself.
The algorithm builds a diverse pool of starting points (including a Sidon‑set
candidate and random sparse supports), optimises each with an Adam phase on a
smooth‑max surrogate, refines with an exponentiated‑gradient sub‑gradient
phase, performs several short fine‑tuning Adam passes, mixes the best result
with a few simple shapes (uniform, symmetric) and finally applies a cheap
stochastic hill‑climb.  All work respects the supplied time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two that is ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative vector to ``new_n`` points by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:                     # protect against a zero vector
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s


def _sidon_candidate(n: int) -> np.ndarray | None:
    """
    Greedy construction of a Sidon‑set (all pairwise sums distinct) inside {0,…,n‑1}.
    Returns a probability vector that places equal mass on the selected indices.
    """
    pos = []
    sums = set()
    # aim for roughly sqrt(n) points – that already gives a good spread
    target = max(2, int(np.sqrt(n) * 1.3))
    for i in range(n):
        conflict = False
        for p in pos:
            if (i + p) in sums:
                conflict = True
                break
        if not conflict:
            for p in pos:
                sums.add(i + p)
            pos.append(i)
            if len(pos) >= target:
                break
    if not pos:
        return None
    a = np.zeros(n, dtype=np.float64)
    a[pos] = 1.0 / len(pos)
    return a


def _sparse_random_candidate(n: int) -> np.ndarray:
    """Randomly choose ~sqrt(n) positions and put a Dirichlet random mass on them."""
    m = max(2, int(np.sqrt(n)))
    idx = np.random.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = np.random.dirichlet(np.ones(m))
    return a


def _generate_initial_candidates(n: int, prev_best):
    """
    Produce a heterogeneous pool of starting points for a given length ``n``.
    ``prev_best`` (if not ``None``) is up‑sampled and used as a warm‑start.
    """
    candidates = []

    # 1. Warm‑start from the best sequence found so far (upsampled)
    if prev_best is not None:
        candidates.append(_upsample(prev_best, n))

    # 2. Historic solution if the global variable exists
    hseq = globals().get('height_sequence_1')
    if isinstance(hseq, (list, np.ndarray)):
        h = np.asarray(hseq, dtype=np.float64)
        candidates.append(_upsample(h, n))

    # 3. Random Dirichlet draws (different concentrations)
    candidates.append(np.random.dirichlet(np.ones(n)))
    for conc in (0.2, 0.5, 0.8):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # 4. Edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.1, 7):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # 5. Symmetric two‑sided exponentials
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # 6. Hann window (raised‑cosine)
    i = np.arange(n, dtype=np.float64)
    hann = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    hann /= hann.sum()
    candidates.append(hann)

    # 7. Half‑sine shape
    sine = np.sin(np.pi * i / (n - 1))
    sine = np.maximum(sine, 0)
    sine /= sine.sum()
    candidates.append(sine)

    # 8. Gaussian‑like bumps
    mu = (n - 1) / 2.0
    for sigma in (n / 10, n / 6, n / 4):
        shape = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        shape /= shape.sum()
        candidates.append(shape)

    # 9. Sidon‑set based uniform support
    sidon = _sidon_candidate(n)
    if sidon is not None:
        candidates.append(sidon)

    # 10. Random sparse support
    candidates.append(_sparse_random_candidate(n))

    # Normalise and discard pathological entries
    clean = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        clean.append(c / s)
    return clean

def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) / (sum²) (no clipping – we keep values ≤1000)."""
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_c / (s * s)


def _run_adam(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Adam optimiser on a smooth‑max surrogate.
    Returns the best vector seen (on the simplex) and its exact objective.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # use a soft‑max parametrisation (logits → simplex)
    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.07                # a touch larger than the previous code
    lam = 0.15               # start temperature (smooth‑max)
    lam_factor = 1.025       # slowly warm‑up

    best_a = a0.copy()
    best_val = _objective(a0)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # map logits → simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)   # a.sum() is 1 but keep formula
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))      # shift by max for stability
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # back‑prop through the soft‑max (logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 2e4)

        # occasional random shake if stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val


def _run_subgrad(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Exponentiated‑gradient sub‑gradient phase on the exact max‑convolution.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)
    last_improve = time.time()
    eta0 = 0.60            # a slightly larger step‑size

    for it in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:         # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val


def _local_hill_climb(a0: np.ndarray, n: int, deadline: float,
                     max_trials: int = 250_000):
    """
    Very cheap stochastic hill‑climb.
    Moves a small mass ε from one index to another and accepts the move
    if it reduces the exact objective.
    """
    conv_fft_len = _next_pow2(2 * n - 1)

    best_a = a0.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (best_a.sum() ** 2)

    no_improve = 0
    trials = 0

    while time.time() < deadline and trials < max_trials and no_improve < 6_000:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0.0, min(0.05, best_a[i]))
        if eps == 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps

        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max() / (cand.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = cand
            no_improve = 0
            print(f"[n={n} HillClimb] improved to {val:.6f}")
        else:
            no_improve += 1
        trials += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000):
    """
    Search for a non‑negative step‑function that minimises
        F(a) = 2·n·max(a*a) / (sum(a)²)

    The method:
      1. tries several lengths (2048 … 16384),
      2. builds a rich pool of initial shapes (including Sidon‑set and sparse supports),
      3. runs a longer Adam optimiser on a smooth‑max surrogate,
      4. refines with an exponentiated‑gradient sub‑gradient phase,
      5. performs two short fine‑tune Adam passes,
      6. mixes the best candidate with a few simple shapes (uniform / symmetric),
      7. finishes with a cheap stochastic hill‑climb.
    The whole run respects the supplied time budget and returns the best sequence.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = np.inf

    # try a few lengths – larger n gives more freedom but costs more time
    lengths = [4096, 8192, 16384, 32768]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        candidates = _generate_initial_candidates(n, best_seq)

        # iteration budgets, scaled approx. inversely with n
        max_adam = int(30000 * 2048 / n) + 12_000
        max_subgrad = int(9000 * 2048 / n) + 2000

        for cid, init in enumerate(candidates):
            if time.time() > deadline:
                break

            # ----- Adam (smooth‑max) -----
            a_adam, val_adam = _run_adam(init, n, deadline, max_adam)
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n} cand={cid}] Adam new global best {val_adam:.6f}")

            # ----- Sub‑gradient (exponentiated‑gradient) -----
            a_sub, val_sub = _run_subgrad(a_adam, n, deadline, max_subgrad)
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[n={n} cand={cid}] Subgrad new global best {val_sub:.6f}")

            # ----- First short fine‑tune -----
            a_fine1, val_fine1 = _run_adam(best_seq, n, deadline,
                                           max_iters=int(0.4 * max_adam))
            if val_fine1 < best_val:
                best_val = val_fine1
                best_seq = a_fine1.copy()
                print(f"[n={n}] Fine‑tune1 improved global best to {val_fine1:.6f}")

            # ----- Second short fine‑tune (different LR) -----
            a_fine2, val_fine2 = _run_adam(best_seq, n, deadline,
                                           max_iters=int(0.2 * max_adam))
            if val_fine2 < best_val:
                best_val = val_fine2
                best_seq = a_fine2.copy()
                print(f"[n={n}] Fine‑tune2 improved global best to {val_fine2:.6f}")

    # ------------------------------------------------------------------
    # Mixing with very simple shapes (uniform / symmetric) – often flattens
    # the convolution a little more.
    if best_seq is not None and time.time() < deadline:
        n = len(best_seq)
        uniform = np.full(n, 1.0 / n)
        for t in np.linspace(0.0, 1.0, 13):          # 0,0.083,…,1.0
            mixed = (1.0 - t) * best_seq + t * uniform
            mixed = mixed / mixed.sum()
            val = _objective(mixed)
            if val < best_val:
                best_val = val
                best_seq = mixed
                print(f"[mix] uniform mix t={t:.3f} improved to {val:.6f}")

        # symmetrisation (a + reverse(a))/2
        sym = (best_seq + best_seq[::-1]) / 2.0
        sym = sym / sym.sum()
        val = _objective(sym)
        if val < best_val:
            best_val = val
            best_seq = sym
            print(f"[mix] symmetry improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Final cheap local search to squeeze any remaining improvement
    if best_seq is not None and time.time() < deadline:
        print(f"\n=== Final local hill‑climb (n={len(best_seq)}) ===")
        a_final, val_final = _local_hill_climb(best_seq, len(best_seq), deadline)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final
            print(f"[final] hill‑climb improved global best to {val_final:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform if nothing was produced (should never happen)
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Return a plain Python list, as required by the external evaluator
    return best_seq.tolist()
