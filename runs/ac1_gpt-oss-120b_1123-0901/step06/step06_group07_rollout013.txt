"""
Propose a low‑evaluation‑score non‑negative step function.

Algorithm
---------
1. **Rich parametric seeding** – generate a large pool of candidate
   vectors from many analytic families (edge‑exponential, edge‑power‑law,
   two‑Gaussian edges, mixtures with a uniform component, random Dirichlet,
   warm‑start rotations, sinusoidal half‑wave, …).  All candidates are normalised
   to a simplex (sum = 1).  The pool is evaluated with a cheap FFT‑based
   objective and the best few are kept.

2. **Progressive multi‑scale optimisation** – work on a moderate size
   (n = 4096) first, then linearly up‑sample the current best to n = 8192.
   On each scale a deep optimisation consists of:
   * **Smooth‑max Adam** – mirror‑descent on the soft‑max surrogate of the
     maximum convolution.  The temperature (λ) is increased very slowly
     (λ←λ·1.006) so that the surrogate follows the true max closely.
   * **Exact sub‑gradient refinement** – exponentiated‑gradient steps on the
     true max‑convolution using the active set (indices attaining the
     maximum).  The step‑size η decays as 1/√t for fine‑grained tuning.

   Both phases keep the best vector/value seen during the run.
   The number of iterations is limited only by the global deadline, but
   extra patience (larger λ‑cap, longer Adam runs, up‑to‑100 k sub‑gradient
   steps) is allowed when time permits.

3. **Random perturbations** – after the main optimisation a few jittered
   copies of the best vector are created (Gaussian noise in log‑space) and a
   short refinement is performed, allowing the search to escape shallow
   local minima.

The whole pipeline respects the 1000 s budget, prints progress and finally
returns the best sequence (as a plain Python list) clipped to the evaluator’s
range.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Fast evaluation of the objective (assumes sum(a)≈1).
    Returns 2·n·max(conv) .
    """
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
    *,
    init_lr: float = 0.05,
    lam_start: float = 0.2,
    lam_factor: float = 1.006,
    lam_cap: float = 5e5,
    max_iter: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a log‑sum‑exp (smooth‑max) surrogate.
    The vector is kept on the simplex via a soft‑max mapping.
    Returns the best vector/value observed.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # initialise logits
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = init_lr

    lam = lam_start
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        # ----- simplex via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- true objective (for tracking) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                     # sum(a) = 1 → denominator = 1
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights -----
        # w_i ∝ exp(λ·(conv_i‑max_c)), sum w = 1
        # the argument is ≤0, so no overflow worries
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # ----- gradient of the surrogate (chain‑rule through soft‑max) -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # Dϕ(a) for the soft‑max mirror map
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update (with clipping for safety) -----
        grad_log = np.clip(grad_log, -10.0, 10.0)
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, lam_cap)

        # ----- random shake if stuck -----
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
    *,
    eta0: float = 0.6,
    max_iter: int = 100_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient steps on the true max‑convolution.
    The step‑size η decays as η0 / sqrt(t+1).
    Returns the best vector/value observed.
    """
    # initialise from best seen so far, otherwise from a0
    a = best_seq if best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()                         # stay on simplex

    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it:06d}] new best {val:.6f}")

        # active set where convolution attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)

        # renormalise to the simplex, avoid collapse
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # early stop if stagnant for a while
        if time.time() - last_improve > 5.0:
            break

        it += 1

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run a full optimisation pass (Adam → sub‑gradient) and update the global best.
    """
    # Adam phase (more iterations than the original code)
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
        init_lr=0.05,
        lam_start=0.2,
        lam_factor=1.006,
        lam_cap=5e5,
        max_iter=400_000,
    )
    # Sub‑gradient phase (longer runs)
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
        eta0=0.6,
        max_iter=100_000,
    )
    return best_seq, best_val


# ----------------------------------------------------------------------
def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """
    Produce a diverse set of candidate vectors.
    Families include:
      * warm‑start and its rotations
      * uniform
      * Dirichlet draws (wide range of concentrations)
      * edge‑exponential (α ∈ [1e‑2, 20])
      * edge‑power‑law (p ∈ [0.1, 15])
      * two‑Gaussian edge peaks (σ ∈ [0.05, 4], optional offset)
      * sinusoidal half‑wave
    All candidates are normalised to sum = 1.
    """
    candidates = []

    # ------------------------------------------------------
    # Warm‑start and rotations
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # ------------------------------------------------------
    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # ------------------------------------------------------
    # Random Dirichlet vectors with many concentration parameters
    concs = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for c in concs:
        cand = np.random.dirichlet(np.full(n, c))
        candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # ------------------------------------------------------
    # Edge‑exponential families, mixed with uniform
    alphas = np.logspace(-2, np.log10(20), num=25)   # 0.01 … 20
    w_mix = np.linspace(0.0, 1.0, num=7)            # mixing weight with uniform
    for alpha in alphas:
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in w_mix:
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # ------------------------------------------------------
    # Edge‑power‑law families, mixed with uniform
    p_vals = np.linspace(0.2, 15.0, num=30)
    for p in p_vals:
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # ------------------------------------------------------
    # Two‑Gaussian edge peaks (optionally offset from the true edge)
    sigmas = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    offsets = [0, int(0.05 * n), int(0.1 * n), int(0.15 * n)]
    for sigma in sigmas:
        for off in offsets:
            left_center = off
            right_center = n - 1 - off
            gauss = np.exp(-((i - left_center) ** 2) / (2 * (sigma * n) ** 2))
            gauss += np.exp(-((i - right_center) ** 2) / (2 * (sigma * n) ** 2))
            gauss /= gauss.sum()
            candidates.append(gauss)

    # ------------------------------------------------------
    # Positive sinusoidal half‑wave (already used before)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    return candidates

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a non‑negative step function that
    minimises 2·n·max(a∗a)/(sum a)².

    Returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper to up‑sample a vector to a new length by linear interpolation
    def _interp_up(v: np.ndarray, new_n: int) -> np.ndarray:
        old_n = v.shape[0]
        x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
        v_new = np.interp(x_new, x_old, v)
        v_new = np.maximum(v_new, 0.0)
        s = v_new.sum()
        if s > 0:
            v_new /= s
        else:
            v_new = np.full(new_n, 1.0 / new_n)
        return v_new

    # ------------------------------------------------------------------
    # Multi‑scale schedule: start with 4096, finish with 8192 (if time permits)
    scales = [4096, 8192]

    for scale_idx, n in enumerate(scales):
        if time.time() >= deadline:
            break

        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        # ------------------------------------------------------------------
        # Warm‑start handling
        warm_start = None
        global_warm = globals().get("height_sequence_1")
        if global_warm is not None:
            ws = np.asarray(global_warm, dtype=np.float64)
            if ws.shape[0] != n:
                ws = _interp_up(ws, n)
            ws = np.clip(ws, 0.0, None)
            if ws.sum() >= 0.01:
                ws /= ws.sum()
                warm_start = ws
            else:
                warm_start = None

        # If we already have a best from previous scale, up‑sample it
        if best_seq is not None:
            warm_start = _interp_up(best_seq, n)

        # ------------------------------------------------------------------
        # Stage 1 – generate a rich pool of candidates and cheap‑evaluate them
        all_candidates = _generate_parametric_candidates(n, warm_start)

        cand_vals = []
        for idx, cand in enumerate(all_candidates):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s < 0.01:
                cand = np.random.dirichlet(np.ones(n))
                s = cand.sum()
            cand = cand / s
            val = _objective(cand, n, conv_fft_len)
            cand_vals.append((val, cand))
            # optional progress print
            # print(f"[scale {n} candidate {idx}] cheap val={val:.6f}")

        # Keep the top‑k seeds for deep optimisation
        cand_vals.sort(key=lambda x: x[0])
        top_k = 12 if n == 4096 else 8     # a bit fewer starts on the larger grid
        top_candidates = [c for _, c in cand_vals[:top_k]]

        # ------------------------------------------------------------------
        # Stage 2 – deep optimisation of the selected starts
        for start_id, a0 in enumerate(top_candidates):
            if time.time() >= deadline:
                break
            print(f"--- Scale n={n} – deep optimisation start #{start_id + 1}/{len(top_candidates)} ---")
            best_seq, best_val = _deep_optimize(
                a0,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                best_seq,
                best_val,
            )

        # ------------------------------------------------------------------
        # Stage 3 – jittered perturbations around the current best (if time left)
        time_left = deadline - time.time()
        if best_seq is not None and time_left > 20.0:
            jitter_runs = 6
            for j in range(jitter_runs):
                if time.time() >= deadline:
                    break
                # add small Gaussian noise in log‑space, then project back to simplex
                eps = 1e-2
                log_a = np.log(best_seq + 1e-15)
                pert = log_a + np.random.randn(n) * eps
                a_pert = np.exp(pert)
                a_pert /= a_pert.sum()
                print(f"--- Scale n={n} – jitter run {j + 1}/{jitter_runs} ---")
                best_seq, best_val = _deep_optimize(
                    a_pert,
                    n,
                    conv_fft_len,
                    corr_fft_len,
                    deadline,
                    best_seq,
                    best_val,
                )

        # At the end of this scale we keep the best_seq found
        # (it may have a different length; keep the vector as‑is)

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded, return a uniform vector on 4096.
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator's allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search finished: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
