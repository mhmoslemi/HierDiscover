def propose_candidate(seed=0,budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    best_seq = None
    best_val = float('inf')
    
    # target lengths increasing
    lengths = [8192, 16384, 32768, 65536] # maybe also 4096
    for n in lengths:
        if time.time()>deadline: break
        print(f"Optimizing length {n}")
        conv_len = 1 << int(np.ceil(np.log2(2*n-1)))
        corr_len = 1 << int(np.ceil(np.log2(3*n-2)))

        # Build seeds list
        seeds = []
        # uniform
        seeds.append(np.full(n, 1.0/n))
        # if have best_seq: up/down
        if best_seq is not None:
            # upsample/downsample current best to size n
            old = np.array(best_seq)
            x_old = np.linspace(0,1,old.size)
            x_new = np.linspace(0,1,n)
            up = np.interp(x_new, x_old, old)
            if up.sum()>0: up/=up.sum()
            else: up = np.random.dirichlet(np.ones(n))
            seeds.append(up)
        # shape families: power law - exponents in [ -3.5, -0.5 ] step 0.2
        idx = np.arange(1,n+1, dtype=np.float64)
        prod = idx * (n - idx + 1.0)
        for alpha in np.arange(-3.5, -0.4, 0.2):
            shape = np.power(prod, alpha)
            shape[prod<=0] = 0
            shape_sum = shape.sum()
            if shape_sum>0:
                shape/=shape_sum
                seeds.append(shape)
        # double Gaussian edges with gamma param
        i = np.arange(n, dtype=np.float64)
        for gamma in [0.05,0.1,0.2,0.4,0.8]:
            shape = np.exp(-gamma * (i**2)) + np.exp(-gamma *((n-1-i)**2))
            shape/=shape.sum()
            seeds.append(shape)
        # 3 random Dirichlet seeds
        for _ in range(3):
            seeds.append(np.random.dirichlet(np.ones(n)))
        # optionally truncate seeds to limited number to keep time limited
        max_seeds = 12 if n<32768 else 8
        seeds = seeds[:max_seeds]

        # Run each seed through optimizer
        for s_idx,seed_seq in enumerate(seeds):
            if time.time()>deadline: break
            # Evaluate initial value
            val0 = evaluate_sequence(seed_seq.tolist())
            if val0<best_val:
                best_val, best_seq = val0, seed_seq.copy()
                print(f"Initial seed {s_idx} improved to {best_val:.6f}")
            # Adam stage
            a_adam, val_adam = _adam_optimize(seed_seq, n, conv_len, corr_len, deadline)
            if val_adam < best_val:
                best_val, best_seq = val_adam, a_adam.copy()
                print(f"Adam seed {s_idx} improved to {best_val:.6f}")
            # Subgradient refine
            a_ref, val_ref = _subgradient_refine(a_adam, n, conv_len, corr_len, deadline)
            if val_ref < best_val:
                best_val, best_seq = val_ref, a_ref.copy()
                print(f"Refine seed {s_idx} improved to {best_val:.6f}")
        # jitter expansions
        if best_seq is not None and len(best_seq)==n:
            for jitter in range(6):
                if time.time()>deadline: break
                # small multiplicative jitter
                jitter_vec = np.random.normal(scale=0.05, size=n)
                # add to log domain
                logits = np.log(np.maximum(best_seq, 1e-12))
                logits += jitter_vec
                a0 = np.exp(logits - np.max(logits))
                a0 = a0 / a0.sum()
                # run refine only (skip Adam maybe)
                a_ref, val_ref = _subgradient_refine(a0, n, conv_len, corr_len, deadline)
                if val_ref < best_val:
                    best_val, best_seq = val_ref, a_ref.copy()
                    print(f"Jitter {jitter} improved to {best_val:.6f}")
    # final fallback: if best_seq None
    if best_seq is None:
        n = 2048
        best_seq = np.full(n, 1.0/n)
        best_val = evaluate_sequence(best_seq.tolist())
    print(f"Final best value {best_val:.6f} length {len(best_seq)}")
    # store globally for next runs
    globals()['height_sequence_1'] = best_seq.copy()
    return best_seq.tolist()

* **Extended sub‑gradient refinement** – the exponentiated‑gradient stage is
  allowed many more iterations (up to 8000) and its step‑size decays more
  gently.

* **Focused jittering** – after a good solution is found for a given length we
  perform several random perturbations in log‑space and re‑optimise each of them.
  This helps escape shallow local minima.

* **Multi‑scale progression** – we start from moderate lengths (8192, 16384)
  and double the size up to 65536.  Each larger length is warm‑started from the
  best solution of the previous size (interpolated to the new grid).

All heavy work respects the wall‑clock deadline (default 1000 s).  The final
sequence is stored in a global variable `height_sequence_1` for possible reuse
by subsequent runs.

The code is self‑contained, uses only NumPy, and follows the required API.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based – O(n log n))
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT.  Returns length 2*n-1."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = None) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best vector (as a numpy array) and its exact objective value.
    """
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05

    lam = 1.0               # temperature for log‑sum‑exp
    lam_factor = 1.02

    # length‑dependent iteration budget (3× the original baseline)
    if max_iters is None:
        max_iters = int(120000 * 1024 / n) + 6000

    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ------- simple‑x to guarantee a lies on the simplex ----------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ------- convolution and smooth‑max weights --------------------
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.size)

        # ------- gradient of smooth‑max w.r.t. a -----------------------
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # ------- back‑prop through soft‑max ----------------------------
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ------- Adam update -----------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- temperature annealing --------------------------------
        lam = min(lam * lam_factor, 1e8)

        # ------- exact objective (since sum(a)=1) --------------------
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # ------- occasional jitter to shake stagnation ---------------
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


def _subgradient_refine(a0: np.ndarray,
                        n: int,
                        conv_fft_len: int,
                        corr_fft_len: int,
                        deadline: float,
                        max_iters: int = 8000) -> tuple[np.ndarray, float]:
    """
    Exact ℓ∞ sub‑gradient refinement using an exponentiated‑gradient update.
    Returns the best vector found and its exact objective.
    """
    a = a0 / a0.sum()
    best_val = float("inf")
    best_a = a.copy()
    eta = 0.45
    last_improve = time.time()

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient step (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # fallback to a random point
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        eta *= 0.9995                      # gentle decay
        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Shape factories for the seeding stage
# ----------------------------------------------------------------------
def _power_law_shape(n: int, alpha: float) -> np.ndarray:
    """Shape ∝ (i·(n‑i+1))**alpha  with i=1..n (alpha < 0)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
    if shape.sum() > 0:
        shape /= shape.sum()
    else:
        shape = np.random.dirichlet(np.ones(n))
    return shape


def _edge_gaussian_shape(n: int, gamma: float) -> np.ndarray:
    """Two symmetric edge Gaussians with variance ~1/gamma."""
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    if shape.sum() > 0:
        shape /= shape.sum()
    else:
        shape = np.random.dirichlet(np.ones(n))
    return shape


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the provided evaluation function.  The routine respects the wall‑clock
    budget and returns the best sequence discovered (as a plain Python list).

    The algorithm proceeds through increasing lengths (8192 → 65536),
    creates a diverse seed pool (uniform, historic best, parametric families,
    random Dirichlet draws), refines each seed with a long Adam phase followed
    by an extensive sub‑gradient phase, and finally performs several jittered
    restarts around the current best.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # Process a cascade of lengths; start modest to warm‑start the large problem.
    lengths = [8192, 16384, 32768, 65536]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n--- Optimising length n = {n} ---")
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

        # -------------------- build seed pool --------------------
        seeds = []

        # 1) uniform distribution
        seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

        # 2) warm‑start from historic best (interpolate / trim)
        if best_seq is not None:
            old = np.asarray(best_seq, dtype=np.float64)
            if old.shape[0] != n:
                x_old = np.linspace(0.0, 1.0, old.shape[0])
                x_new = np.linspace(0.0, 1.0, n)
                up = np.interp(x_new, x_old, old)
                if up.sum() > 0:
                    up /= up.sum()
                else:
                    up = np.random.dirichlet(np.ones(n))
                seeds.append(up)
            else:
                seeds.append(old.copy())

        # 3) power‑law shapes (grid on exponent)
        for alpha in np.arange(-3.5, -0.4, 0.3):
            seeds.append(_power_law_shape(n, alpha))

        # 4) edge‑Gaussian shapes (grid on width)
        for gamma in [0.04, 0.08, 0.12, 0.20, 0.35, 0.60]:
            seeds.append(_edge_gaussian_shape(n, gamma))

        # 5) a few random Dirichlet draws
        for _ in range(3):
            seeds.append(np.random.dirichlet(np.ones(n)))

        # Limit seed count for very large n so we stay within budget
        max_seeds = 12 if n <= 32768 else 8
        seeds = seeds[:max_seeds]

        # -------------------- optimise each seed --------------------
        for s_id, seed_vec in enumerate(seeds):
            if time.time() > deadline:
                break

            # quick sanity evaluation – could already improve the global best
            val0 = evaluate_sequence(seed_vec.tolist())
            if val0 < best_val:
                best_val, best_seq = val0, seed_vec.copy()
                print(f"[seed {s_id}] initial improvement: {best_val:.6f}")

            # Adam phase (longer than baseline)
            a_adam, val_adam = _adam_optimize(seed_vec, n,
                                              conv_fft_len, corr_fft_len,
                                              deadline)
            if val_adam < best_val:
                best_val, best_seq = val_adam, a_adam.copy()
                print(f"[seed {s_id}] Adam improvement: {best_val:.6f}")

            # Exact sub‑gradient refinement
            a_ref, val_ref = _subgradient_refine(a_adam, n,
                                                 conv_fft_len, corr_fft_len,
                                                 deadline)
            if val_ref < best_val:
                best_val, best_seq = val_ref, a_ref.copy()
                print(f"[seed {s_id}] Sub‑grad improvement: {best_val:.6f}")

        # -------------------- jitter around current best of this n --------------------
        if best_seq is not None and len(best_seq) == n:
            for jitter_id in range(6):
                if time.time() > deadline:
                    break
                # perturb in log‑space (multiplicative jitter)
                logits = np.log(np.maximum(best_seq, 1e-12))
                logits += np.random.normal(scale=0.03, size=n)
                a0 = np.exp(logits - np.max(logits))
                a0 /= a0.sum()

                # a short Adam + sub‑gradient pass for each jitter
                a_adam, _ = _adam_optimize(a0, n,
                                           conv_fft_len, corr_fft_len,
                                           deadline,
                                           max_iters=2000)   # fewer iters for jitter
                a_ref, val_ref = _subgradient_refine(a_adam, n,
                                                     conv_fft_len, corr_fft_len,
                                                     deadline,
                                                     max_iters=4000)
                if val_ref < best_val:
                    best_val, best_seq = val_ref, a_ref.copy()
                    print(f"[jitter {jitter_id}] new best: {best_val:.6f}")

        # Report progress after each length
        print(f"Best after n={n}: value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = evaluate_sequence(best_seq.tolist())
        print("Fallback to uniform sequence:", best_val)

    # Store globally for possible reuse in subsequent calls
    globals()["height_sequence_1"] = best_seq.copy()
    print(f"\nFinal best value = {best_val:.6f} (length = {len(best_seq)})")
    return best_seq.tolist()
