def _adam_optimize_sym(a0, n, conv_fft_len, corr_fft_len, deadline):

    a = np.maximum(a0, EPS)
    a = a / a.sum()
    logits = np.log(a)   # shape (n,)

    # Adam parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.05
    eps_adam = 1e-8

    lam = 1.0
    lam_factor = 1.03

    # iteration budget based on n
    max_iters = int(50000 * 1024 / n) + 2000  # adjust perhaps
    best_val = float('inf')
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break

        # compute a from logits via softmax
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        a = (a + a[::-1]) * 0.5
        # re-normalize (due to floating round)
        total = a.sum()
        if total > 0:
            a /= total
        else:
            a = np.full_like(a, 1.0 / n)

        max_c = conv.max()

        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

""".
Search for non‑negative step‑function minimizing evaluation.
Uses symmetric projection, Adam on smooth‑max, exact sub‑gradient,
and a final random local search. The routine works within a wall‑clock budget.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    n = a.shape[0]
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

def _symmetrize(x: np.ndarray) -> np.ndarray:
    """Return (x + x[::-1]) / 2, keeping the sum unchanged."""
    y = (x + x[::-1]) * 0.5
    # numerical safety: make sure sum is still close to original
    s = y.sum()
    if s > 0:
        y /= s
    else:
        y = np.full_like(y, 1.0 / y.shape[0])
    return y

def _adam_optimize_sym(a0: np.ndarray, n: int, conv_fft_len: int,
                       corr_fft_len: int, deadline: float) -> tuple[np.ndarray, float]:
    """Adam on a smooth‑max surrogate. Enforces symmetry after each soft‑max."""
    eps = 1e-15
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    lr = 0.05
    eps_adam = 1e-8

    lam = 1.0
    lam_factor = 1.03

    # the number of iterations grows when n is small, shrinks for large n
    max_iters = int(50000 * 1024 / n) + 2000
    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max -> probability simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        # enforce symmetry
        a = _symmetrize(a)

        # convolution and smooth‑max weights
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.size)

        # gradient w.r.t. a (smooth‑max)
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)
        # keep gradient in the symmetric subspace
        grad_a = (grad_a + grad_a[::-1]) * 0.5

        # back‑propagate through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature annealing
        lam = min(lam * lam_factor, 1e8)

        # objective (exact because sum(a) = 1)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # occasional jitter if we stall for a few seconds
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val

def _subgradient_refine_sym(a0: np.ndarray, n: int, conv_fft_len: int,
                            corr_fft_len: int, deadline: float) -> tuple[np.ndarray, float]:
    """Exact ℓ∞ sub‑gradient refinement with an exponentiated‑gradient step.
    The iterate is forced to stay symmetric."""
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    a = _symmetrize(a)

    best_a = a.copy()
    best_val = float("inf")

Symmetric Adam + exact sub‑gradient search for the step‑function problem.

The routine works inside a wall‑clock budget (default 1000 s).  For each
candidate length it builds a heterogeneous pool of seeds, runs an Adam
optimiser on a smooth‑max surrogate of the convolution peak while always
projecting the iterate onto the symmetric simplex, follows up with an exact
ℓ∞ sub‑gradient refinement (exponentiated‑gradient) and finally performs a few
seconds of cheap stochastic local search.  The best sequence found over the
whole run is returned as a plain Python list.
"""

import time

def _next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _symmetrize(x: np.ndarray) -> np.ndarray:
    """Project x onto the symmetric simplex: (x + x[::-1])/2, re‑normalised."""
    y = (x + x[::-1]) * 0.5
    s = y.sum()
    if s > 0:
        y /= s
    else:
        y = np.full_like(y, 1.0 / y.shape[0])
    return y

                    print(f"[n={n} local search] new global best {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found
    # ----------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return as a plain Python list as required by the external evaluator
    return best_seq.tolist()
