"""
Hybrid search for low max-convolution step function.
Stage 1: generate diverse candidates (uniform, power‑law, double‑exponential,
Gaussian, sinusoid, Dirichlet, warm‑start).
Stage 2: for the K best candidates run an Adam optimizer on a smooth‑max
surrogate and a long exponentiated‑gradient sub‑gradient phase.
Stage 3: finish with a simple simulated‑annealing loop that perturbs the
log‑probabilities (softmax) while preserving the simplex constraint.
Stage 4: if time permits, up‑sample to a larger support (8192) and repeat
the Adam+sub‑gradient+annealing refinement.
The function respects the time budget, returns the best sequence found.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2*len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective (sum(a) assumed > 0)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate followed by a long exponentiated‑gradient
    sub‑gradient phase.  The routine updates the global best sequence/value.
    """
    eps = 1e-15
    # map to soft‑max logits to stay on the simplex
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5                # start temperature for smooth‑max
    lam_factor = 1.018

    last_update = time.time()
    it = 0

    # ------------------- Adam (smooth‑max) -------------------
    while time.time() < deadline:
        # Convert logits → probabilities on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # Exact objective for logging / possible early stop
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)    # a.sum()==1 ⇒ 2*n*max_c

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_update = time.time()
            print(f"[n={n} Adam iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through the soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase temperature (approach the true max)
        lam = min(lam * lam_factor, 1e6)

        # occasional random shake if we have been stagnant
        if time.time() - last_update > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_update = time.time()

        it += 1
        # safety bound – the loop will also stop because of the deadline
        if it > 800_000:
            break

    # ---------------------- Sub‑gradient phase ----------------------
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total

    eta0 = 0.6
    sub_last_improve = time.time()
    # we allow many more iterations than the original code
    for sub_it in range(200_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[n={n} Subgrad iter={sub_it:06d}] new best {val:.6f}")

        # active‑set mask (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (mirror‑descent on the simplex)
        eta = eta0 / np.sqrt(sub_it + 1)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # stop if stagnant for a while (more tolerant than original)
        if time.time() - sub_last_improve > 15.0:
            break

    return global_best_seq, global_best_val


def _generate_candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse set of initial probability vectors."""
    pool = []

    # warm‑start and simple rotations
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # uniform baseline
    pool.append(np.full(n, 1.0 / n))

    # power‑law edges
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)  # i·(n‑i+1)
    for alpha in np.linspace(-2.0, -0.2, 20):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # double‑exponential families
    i = np.arange(n, dtype=np.float64)
    for lam in np.linspace(0.05, 4.0, 20):
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        pool.append(shape)

    # Gaussian bumps
    for sigma in np.linspace(n * 0.02, n * 0.5, 12):
        shape = np.exp(-0.5 * ((i - n / 2) ** 2) / (sigma ** 2))
        shape /= shape.sum()
        pool.append(shape)

    # single half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # random Dirichlet draws
    for alpha in np.logspace(-3, 0, 16):
        shape = np.random.dirichlet(np.full(n, alpha))
        pool.append(shape)

    return pool


def _simulated_annealing(
    start: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    init_temp: float = 0.5,
    decay: float = 0.9995,
    min_temp: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing on the simplex using a log‑softmax
    parametrisation.  Returns the best vector seen and its objective.
    """
    eps = 1e-15
    logits = np.log(start + eps)
    best_logits = logits.copy()
    best_a = start.copy()
    best_val = _objective(best_a, n, conv_fft_len)

    temp = init_temp
    it = 0
    while time.time() < deadline:
        # propose a move
        step = np.random.randn(n) * temp
        new_logits = logits + step
        # soft‑max back to a probability vector
        mx = new_logits.max()
        expv = np.exp(new_logits - mx)
        a = expv / expv.sum()

        val = _objective(a, n, conv_fft_len)

        if val < best_val:
            best_val = val
            best_a = a
            best_logits = new_logits
            logits = new_logits
            # speed up cooling when we improve
            temp = max(temp * 0.95, min_temp)
            print(f"[SA iter={it}] new best {val:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(val - best_val) / max(temp, 1e-12))
            if np.random.rand() < prob:
                logits = new_logits

        # geometric cooling
        temp = max(temp * decay, min_temp)
        it += 1
        # safety stop (in case we happen to wander forever)
        if it % 50000 == 0 and time.time() > deadline:
            break

    return best_a, best_val


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Hybrid search for a non‑negative step‑function (length up to 8192)
    that minimises 2·n·max(a*a)/(sum a)^2.
    The routine respects the supplied time‑budget and returns the best
    sequence encountered as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # Primary support size (fast to evaluate)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Load possible warm start
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        w = np.asarray(warm_start, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
        else:
            w = None
    else:
        w = None

    # ------------------------------------------------------------------
    # 1. Candidate pool & quick screening
    candidates = _generate_candidate_pool(n, w)
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    screened = []

    for idx, cand in enumerate(candidates):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        val = _objective(a, n, conv_fft_len)
        screened.append((val, a))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        print(f"[screen {idx:03d}] obj={val:.6f}")

    # Keep a moderate number of promising starts
    screened.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in screened[:10]]

    # ------------------------------------------------------------------
    # 2. Deep optimisation (Adam + long sub‑gradient)
    for cid, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[optim start {cid}] launching deep optimisation")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # 3. Simulated‑annealing refinement on the current best vector
    if best_seq is not None and time.time() < deadline - 5:
        print("[SA] starting simulated‑annealing refinement")
        best_seq, best_val = _simulated_annealing(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            init_temp=0.3,
            decay=0.9997,
            min_temp=1e-7,
        )

    # ------------------------------------------------------------------
    # 4. Optional up‑sampling to a larger grid if time permits
    if time.time() + 8 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # Interpolate the best sequence onto the larger grid
        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        # Run the deep optimiser on the larger grid
        print("[big] deep optimisation on enlarged support")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )

        # Final SA on the enlarged grid (if time left)
        if time.time() < deadline - 2:
            print("[big SA] final simulated‑annealing")
            best_seq, best_val = _simulated_annealing(
                best_seq,
                n_big,
                conv_fft_len_big,
                deadline,
                init_temp=0.2,
                decay=0.9995,
                min_temp=1e-7,
            )

        # Down‑sample back to the original length, if desired.
        # The evaluator does not require a fixed length – we simply keep the
        # expanded sequence (it is usually better because the objective scales
        # with n).  However, to stay safe we clip the length to at most 8192.
        final_seq = best_seq
    else:
        final_seq = best_seq

    if final_seq is None:
        final_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    # Convert to plain Python list for the caller
    return final_seq.tolist()

"""
Hybrid search for a non‑negative step‑function that minimises

    F(a) = 2·n·max(a∗a) / (∑a)² .

The algorithm is split into four stages:

1. **Candidate pool** – a large, diverse set of probability vectors is
   created (uniform, power‑law, double‑exponential, Gaussian, sinusoid,
   Dirichlet draws and a warm‑start if available).  Every vector is
   normalised to unit sum and the best few are kept.

2. **Deep optimisation** – each kept start is refined by an Adam optimiser
   on a smooth‑max (log‑sum‑exp) surrogate of the true maximum, followed by a
   long exponentiated‑gradient sub‑gradient phase that works on the exact
   max.  The routine updates a global best sequence/value that survives across
   all starts.

3. **Simulated‑annealing refinement** – the current best vector is perturbed
   in the log‑probability (soft‑max) space.  A geometric cooling schedule and
   a Metropolis acceptance criterion are used to escape shallow local minima.

4. **Optional up‑sampling** – if time permits the best distribution is
   interpolated onto a larger grid (8192 points) and the whole optimisation
   (Adam + sub‑gradient + SA) is repeated; the enlarged sequence is returned.

All steps respect the supplied time budget (default 1000 s).  The final
sequence is clipped to the evaluator’s allowed range (non‑negative,
≤ 1000) and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than ``x``."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective (the sum of ``a`` is assumed > 0)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate followed by a long exponentiated‑gradient
    sub‑gradient phase.  The routine updates and returns the best sequence/value
    found (global across all calls).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)           # stay on the simplex via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5          # temperature for the smooth‑max
    lam_factor = 1.018

    last_improve = time.time()
    it = 0

    # ------------------------------------------------------- Adam phase
    while time.time() < deadline:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp formulation)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through the soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # occasional random “shake” when we have been stagnant
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        if it > 800_000:          # hard safety bound
            break

    # --------------------------------------------------- Sub‑gradient phase
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total

    eta0 = 0.6
    sub_last_improve = time.time()
    for sub_it in range(200_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[n={n} Subgrad iter={sub_it:06d}] new best {val:.6f}")

        # active‑set mask (all indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (mirror‑descent on the simplex)
        eta = eta0 / np.sqrt(sub_it + 1)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 15.0:
            break

    return global_best_seq, global_best_val


# ----------------------------------------------------------------------
def _generate_candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Build a diverse pool of start vectors (including a parametric sweep)."""
    pool = []

    # warm‑start and simple rotations
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # uniform baseline
    pool.append(np.full(n, 1.0 / n))

    # power‑law edge‑concentrated shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)            # i·(n‑i+1)
    for alpha in np.linspace(-2.0, -0.2, 20):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # double‑exponential (two‑sided Laplace) families
    i = np.arange(n, dtype=np.float64)
    for lam in np.linspace(0.05, 4.0, 20):
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        pool.append(shape)

    # Gaussian‑type bumps centred in the middle (width sweep)
    for sigma in np.linspace(n * 0.02, n * 0.5, 12):
        shape = np.exp(-0.5 * ((i - n / 2) ** 2) / (sigma ** 2))
        shape /= shape.sum()
        pool.append(shape)

    # single half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # random Dirichlet draws with a range of concentration parameters
    for alpha in np.logspace(-3, 0, 16):
        shape = np.random.dirichlet(np.full(n, alpha))
        pool.append(shape)

    return pool


# ----------------------------------------------------------------------
def _simulated_annealing(
    start: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    init_temp: float = 0.5,
    decay: float = 0.9995,
    min_temp: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing on the simplex (log‑softmax parametrisation).
    Returns the best vector seen and its objective.
    """
    eps = 1e-15
    logits = np.log(start + eps)
    best_logits = logits.copy()
    best_a = start.copy()
    best_val = _objective(best_a, n, conv_fft_len)

    temp = init_temp
    it = 0
    while time.time() < deadline:
        step = np.random.randn(n) * temp
        new_logits = logits + step
        mx = new_logits.max()
        expv = np.exp(new_logits - mx)
        a = expv / expv.sum()

        val = _objective(a, n, conv_fft_len)

        if val < best_val:
            best_val = val
            best_a = a
            best_logits = new_logits
            logits = new_logits
            # accelerate cooling after an improvement
            temp = max(temp * 0.95, min_temp)
            print(f"[SA iter={it}] new best {val:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(val - best_val) / max(temp, 1e-12))
            if np.random.rand() < prob:
                logits = new_logits

        temp = max(temp * decay, min_temp)
        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function minimising the evaluation metric.
    The routine respects the supplied time‑budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Primary support size (fast enough for many iterations)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Load a warm start if the user has supplied one
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        w = np.asarray(warm_start, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
        else:
            w = None
    else:
        w = None

    # ------------------------------------------------------------------
    # 1. Build a diverse pool and screen it quickly
    candidates = _generate_candidate_pool(n, w)

    best_seq: np.ndarray | None = None
    best_val = float("inf")
    screened = []

    for idx, cand in enumerate(candidates):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:                     # avoid degenerate zero‑sum vectors
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        val = _objective(a, n, conv_fft_len)
        screened.append((val, a))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        print(f"[screen {idx:03d}] obj={val:.6f}")

    # Keep a modest number of promising starts for the deep optimisation
    screened.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in screened[:5]]

    # ------------------------------------------------------------------
    # 2. Deep optimisation (Adam + long sub‑gradient) on the selected starts
    for cid, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[optim start {cid}] launching deep optimisation")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # 3. Simulated‑annealing refinement on the current best vector
    if best_seq is not None and time.time() < deadline - 5:
        print("[SA] starting simulated‑annealing refinement")
        best_seq, best_val = _simulated_annealing(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            init_temp=0.3,
            decay=0.9997,
            min_temp=1e-7,
        )

    # ------------------------------------------------------------------
    # 4. Optional up‑sampling to a larger grid if time permits
    if time.time() + 8 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("[big] deep optimisation on enlarged support")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )

        if time.time() < deadline - 2:
            print("[big SA] final simulated‑annealing")
            best_seq, best_val = _simulated_annealing(
                best_seq,
                n_big,
                conv_fft_len_big,
                deadline,
                init_temp=0.2,
                decay=0.9995,
                min_temp=1e-7,
            )
        final_seq = best_seq
    else:
        final_seq = best_seq

    if final_seq is None:
        final_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    return final_seq.tolist()
