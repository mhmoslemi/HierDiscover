def _generate_candidates_extended(n, warm_start):
   candidates = []
   # warm start
   if warm_start not None: add it and its roll.
   # uniform
   candidates.append(np.full(n, 1.0/n))
   # random Dirichlet
   for conc in (0.05,0.1,0.2,0.5,1.0,2.0):
       cand = np.random.dirichlet(np.full(n, conc))
       candidates.append(cand)
   # sorted Dirichlet (descending)
   for conc in (0.1,0.3,0.5,1.0):
       cand = np.random.dirichlet(np.full(n, conc))
       cand = np.sort(cand)[::-1]
       candidates.append(cand)
   # decreasing exponential
   i = np.arange(n)
   for alpha in np.logspace(-4, -0.5, num=15):  # 1e-4 to ~0.3
       dec = np.exp(-alpha * i)
       dec = dec/dec.sum()
       candidates.append(dec)
       # mixture with uniform
       for w in (0.2,0.4,0.6,0.8):
           shape = w*dec + (1-w)/n
           shape /= shape.sum()
           candidates.append(shape)
   # decreasing power law
   for p in np.linspace(0.5, 3.0, 12):
       dec = (i+1) ** (-p)
       dec /= dec.sum()
       candidates.append(dec)
   # linear decreasing
   linear = (n - i).astype(np.float64)
   linear /= linear.sum()
   candidates.append(linear)
   # quadratic decreasing
   quad = (n - i)**2
   quad /= quad.sum()

"""
Hybrid search for a low‑evaluation step function.

The algorithm consists of three stages:

1. **Rich, asymmetric candidate generation** – a large and diverse
   set of vectors is built (exponential‑decay, power‑law, linear,
   quadratic, sorted Dirichlet, mixtures with a uniform component,
   random permutations, the previous best, etc.).  All candidates are
   normalised to sum = 1 and quickly evaluated with a cheap FFT‑based
   objective.

2. **Smooth‑max Adam optimisation** – every retained start is optimised
   with Adam on a log‑sum‑exp surrogate of the max‑convolution while
   staying on the simplex via a soft‑max parametrisation.  The
   temperature (smooth‑max “λ”) is increased steadily so the surrogate
   converges to the true max.

3. **Exact sub‑gradient refinement + random jitter** – after Adam a few
   exponentiated‑gradient steps are taken on the true max‑convolution.
   A light random‑jitter phase is then performed to escape shallow
   minima.

If time permits the best vector is interpolated to a finer grid
(e.g. n = 8192) and the whole pipeline is repeated.  The routine respects
the 1000 s budget and finally returns the best sequence as a plain
Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)-1 using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Very fast evaluation (assumes a is already normalised)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp surrogate of the max‑convolution.
    Returns best sequence/value seen during the run.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5               # start temperature
    lam_factor = 1.02       # slower increase – gives smoother gradients
    lam_cap = 1e6

    last_improve = time.time()
    it = 0
    max_iter = 400_000

    while time.time() < deadline and it < max_iter:
        # ----- simplex via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- convolution & objective -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n} Adam it={it:06d}]  new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate (chain‑rule through soft‑max) -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, lam_cap)

        # occasional random shake if stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 2e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient steps on the true max‑convolution.
    """
    a = best_seq if best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()

    eta = 0.6
    last_improve = time.time()
    max_iter = 30_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n} Subgrad it={it:05d}]  new best {val:.6f}")

        # active set where convolution attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 7.0:
            break

    return best_seq, best_val


def _jitter_phase(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Small random perturbations followed by a short Adam run.
    Helps escape shallow local minima.
    """
    a = best_seq if best_seq is not None else a0
    # jitter in logits space
    jitter = (np.random.rand(n) - 0.5) * 2e-2
    logits = np.log(a + 1e-15) + jitter
    # a brief Adam run (few thousand iterations)
    best_seq, best_val = _adam_smooth_max(
        np.exp(logits), n, conv_fft_len, corr_fft_len, deadline,
        best_seq, best_val,
    )
    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """Runs Adam → sub‑gradient → jitter, updating the global best."""
    best_seq, best_val = _adam_smooth_max(
        a0, n, conv_fft_len, corr_fft_len, deadline, global_best_seq, global_best_val
    )
    best_seq, best_val = _subgradient_refinement(
        a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
    )
    # a couple of jitter‑Adam passes
    for _ in range(2):
        if time.time() >= deadline:
            break
        best_seq, best_val = _jitter_phase(
            a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
        )
    return best_seq, best_val


# ----------------------------------------------------------------------
def _generate_candidates_extended(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a diverse collection of plausible vectors.
    Includes monotone decreasing families, shuffled versions,
    random Dirichlet (sorted) samples, mixtures and a few symmetric baselines.
    """
    candidates: list[np.ndarray] = []
    i = np.arange(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Warm‑start and simple rotations
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # ------------------------------------------------------------------
    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # ------------------------------------------------------------------
    # Random Dirichlet vectors (various concentrations)
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    # ------------------------------------------------------------------
    # Sorted Dirichlet (descending) – forces a monotone shape
    for conc in (0.1, 0.3, 0.5, 1.0):
        cand = np.random.dirichlet(np.full(n, conc))
        cand = np.sort(cand)[::-1]
        candidates.append(cand)

    # ------------------------------------------------------------------
    # Pure decreasing exponential families
    for alpha in np.logspace(-4, -0.5, num=20):      # 1e‑4 … ~0.3
        dec = np.exp(-alpha * i)
        dec /= dec.sum()
        candidates.append(dec)
        # mixtures with a uniform component
        for w in (0.2, 0.4, 0.6, 0.8):
            shape = w * dec + (1.0 - w) / n
            shape /= shape.sum()
            candidates.append(shape)

    # ------------------------------------------------------------------
    # Decreasing power‑law families
    for p in np.linspace(0.5, 3.5, num=18):
        dec = (i + 1.0) ** (-p)
        dec /= dec.sum()
        candidates.append(dec)

    # ------------------------------------------------------------------
    # Simple linear / quadratic decreasing shapes
    linear = (n - i)
    linear /= linear.sum()
    candidates.append(linear)

    quad = (n - i) ** 2
    quad /= quad.sum()
    candidates.append(quad)

    # ------------------------------------------------------------------
    # Random permutations of a few already‑generated vectors (break symmetry)
    for base in candidates[:12]:
        perm = np.random.permutation(base)
        candidates.append(perm)

    # ------------------------------------------------------------------
    # Positive half‑wave sinusoid (used before)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    return candidates

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a low‑evaluation‑score step function.
    Returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Primary grid size – 4096 gives a good trade‑off between quality and speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Try to use the previous best as a warm start (global variable)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a large pool of candidates and keep the most promising
    all_cands = _generate_candidates_extended(n, warm)
    cand_scores = []

    for idx, cand in enumerate(all_cands):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:                         # avoid near‑zero vectors
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_scores.append((val, cand))

    # Keep a generous top‑k (more than the previous implementation)
    cand_scores.sort(key=lambda x: x[0])
    top_k = min(40, len(cand_scores))
    top_candidates = [c for _, c in cand_scores[:top_k]]
    print(f"Stage 1 – selected top {top_k} candidates (best quick ≈ {cand_scores[0][0]:.6f})")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation of each start
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"\n--- Deep optimisation on start #{start_id + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
        )

    # ------------------------------------------------------------------
    # Stage 3 – enlarge grid if time permits
    time_left = deadline - time.time()
    if time_left > 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("\n--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big,
            deadline, best_seq, best_val,
        )
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big

    # ------------------------------------------------------------------
    # Fallback – extremely unlikely but safe
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator's allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"\n*** Search finished: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    # expose the result for possible reuse in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
