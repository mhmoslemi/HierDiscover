"""
Propose a better non‑negative step‑function.

The algorithm consists of four loosely coupled phases that together explore
different parts of the search space while respecting the 1000 s time budget.

1️⃣ **Rich candidate pool** – builds a diverse set of starting vectors:
   uniform, power‑law edge‑concentrated, double‑exponential, Gaussian,
   half‑wave sinusoid, cosine‑squared edges, arcsine‑shaped edge,
   beta‑shaped (iⁿ·(N‑i)ᵐ) families and many random Dirichlet draws.
   Each vector is normalised to unit sum and evaluated with a fast FFT‑based
   convolution.  The best few (≤ 8) are kept.

2️⃣ **Deep optimisation** – each kept candidate is refined by an Adam optimiser
   on a smooth‑max surrogate of the maximal convolution (`log‑sum‑exp`),
   followed by a small‑step exponentiated‑gradient (mirror‑descent) phase
   that works on the exact max.  The routine is time‑aware, updates a global
   best, and runs longer when time permits.

3️⃣ **Stochastic local search** – starting from the current global best a
   series of cheap random moves are attempted:
      * transfer a small amount of mass between two positions,
      * draw a Dirichlet sample centred on the current best (concentration
        parameter slowly increased so the search becomes more local),
      * occasional full random restart if no improvement happened for a while.
   Moves are accepted only when they improve the exact objective; the
   temperature‑like schedule and early‑stop on stagnation keep the search
   efficient.

4️⃣ **Upsampling + final refinement** – if enough time remains the best
   distribution is interpolated to a finer grid (8192 points) and a short
   Adam+sub‑gradient pass is performed.  The larger support gives the optimiser
   more degrees of freedom, usually lowering the bound further.

All arithmetic is performed with NumPy, convolutions are computed via FFT,
and the sequence is always kept on the probability simplex (non‑negative,
unit‑sum).  The final list is clipped to the evaluator's allowed range
([0, 1000]) and returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of the objective (sum(a) is assumed > 0)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Shape proportional to (i+1)^α · (n-i)^β, i = 0,…,n‑1.
    Normalised to unit sum.
    """
    i = np.arange(n, dtype=np.float64) + 1.0
    shape = np.power(i, alpha) * np.power(n - i + 1.0, beta)
    shape[~np.isfinite(shape)] = 0.0
    if shape.sum() == 0:
        shape = np.ones_like(shape)
    return shape / shape.sum()


def _arcsine_shape(n: int) -> np.ndarray:
    """
    Discretised arcsine density 1/√(x(1‑x)), x∈(0,1).
    Normalised to unit sum.
    """
    eps = 1e-12
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = 1.0 / np.sqrt(np.clip(x * (1.0 - x), eps, None))
    shape /= shape.sum()
    return shape


def _cos_squared_edge_shape(n: int) -> np.ndarray:
    """
    Edge‑concentrated shape cos²(π·i/(n‑1)), i=0,…,n‑1.
    Peaks at both ends, zero in the centre.
    """
    i = np.arange(n, dtype=np.float64)
    shape = np.cos(np.pi * i / (n - 1.0)) ** 2
    shape /= shape.sum()
    return shape


def _generate_candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Build a diverse pool of start vectors (including many parametric families)."""
    pool: list[np.ndarray] = []

    # warm‑start and cyclic shifts
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # 1️⃣ uniform baseline
    pool.append(np.full(n, 1.0 / n))

    # 2️⃣ power‑law edge shapes (i·(n‑i+1))^α
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-4.0, -0.1, 20):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # 3️⃣ double‑exponential (two‑sided Laplace)
    i = np.arange(n, dtype=np.float64)
    for lam in np.linspace(0.05, 5.0, 20):
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        pool.append(shape)

    # 4️⃣ Gaussian bumps
    for sigma in np.linspace(n * 0.02, n * 0.35, 12):
        shape = np.exp(-0.5 * ((i - n / 2) ** 2) / (sigma ** 2))
        shape /= shape.sum()
        pool.append(shape)

    # 5️⃣ single half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # 6️⃣ cosine‑squared edge shape
    pool.append(_cos_squared_edge_shape(n))

    # 7️⃣ arcsine‑edge shape
    pool.append(_arcsine_shape(n))

    # 8️⃣ beta‑family (i+1)^α·(n‑i)^β
    for a in np.linspace(-2.0, 2.0, 7):
        for b in np.linspace(-2.0, 2.0, 7):
            shape = _beta_shape(n, a, b)
            pool.append(shape)

    # 9️⃣ random Dirichlet draws (various concentrations)
    for alpha in np.logspace(-2, 0, 14):
        shape = np.random.dirichlet(np.full(n, alpha))
        pool.append(shape)

    return pool

def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate (log‑sum‑exp) followed by a short
    exponentiated‑gradient (mirror‑descent) phase.
    Time‑aware: stops when the deadline is reached.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # start in the simplex via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5          # temperature for smooth‑max
    lam_factor = 1.02  # slower annealing – more accurate gradients

    last_improve = time.time()
    it = 0

    # ------------------------------------------------------------------
    # Adam on smooth‑max
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----- simplex projection via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- exact objective (for logging) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        # ----- keep global best -----
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)          # chain‑rule through soft‑max

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, 1e6)

        # ----- occasional random shake to escape plateaus -----
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        if it > 600_000:          # safety stop
            break

    # ------------------------------------------------------------------
    # Sub‑gradient phase on the exact max
    # ------------------------------------------------------------------
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total

    eta0 = 0.6
    sub_last_improve = time.time()
    for sub_it in range(40_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[Subgrad n={n} iter={sub_it:05d}] new best {val:.6f}")

        # active‑set mask (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (mirror‑descent on the simplex)
        eta = eta0 / np.sqrt(sub_it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 6.0:
            break

    return global_best_seq, global_best_val

def _local_search(
    a_start: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing around `a_start`.  Two move types are used:
      * transfer a small amount of mass between two random coordinates,
      * draw a Dirichlet sample centred on the current best (concentration
        slowly increased so the search becomes more local).
    The routine keeps the best ever seen and aborts after stagnation.
    """
    a = a_start.copy()
    best_seq = global_best_seq if global_best_seq is not None else a.copy()
    best_val = global_best_val
    last_improve = time.time()
    start_time = time.time()

    # parameters for the two move types
    max_transfer_fraction = 0.02   # ≤ 2 % of the total mass per transfer
    dirichlet_start_conc = 2.0
    dirichlet_end_conc = 200.0
    stagnation_limit = 30.0        # seconds without improvement → random restart

    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 0.01:
            break

        # choose move type
        if np.random.rand() < 0.6:
            # ---- small transfer between two indices ----
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            max_transfer = min(a[i], max_transfer_fraction)
            if max_transfer <= 0.0:
                continue
            delta = np.random.rand() * max_transfer
            a[i] -= delta
            a[j] += delta
        else:
            # ---- Dirichlet perturbation centred on best_seq ----
            frac = (deadline - time.time()) / (deadline - start_time)
            conc = dirichlet_start_conc + (dirichlet_end_conc - dirichlet_start_conc) * (1 - frac)
            # shape parameters proportional to current best (avoid zeros)
            shape = np.maximum(best_seq, 1e-12) * conc
            a = np.random.dirichlet(shape)

        # re‑normalise (numerical drift)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        val = _objective(a, n, conv_fft_len)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[LocalSearch n={n}] new best {val:.6f}")
        else:
            # reject non‑improving move (simple hill‑climb)
            a = best_seq.copy()

        # random restart after long stagnation
        if time.time() - last_improve > stagnation_limit:
            a = np.random.dirichlet(np.ones(n))
            a /= a.sum()
            last_improve = time.time()

    return best_seq, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the metric.
    The routine respects the supplied time‑budget and returns the best
    sequence encountered as a plain Python list.
    """
    np.random.seed(seed)
    safety = 0.5                     # keep a little buffer for the caller
    deadline = time.time() + max(0.0, budget_s - safety)

    # ------------------------------------------------------------------
    # 0. Basic parameters
    # ------------------------------------------------------------------
    n = 4096                         # primary resolution
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1. Warm‑start handling
    # ------------------------------------------------------------------
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        w = np.asarray(warm_start, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
        else:
            w = None
    else:
        w = None

    # ------------------------------------------------------------------
    # 2. Build a rich pool of candidates and evaluate them quickly
    # ------------------------------------------------------------------
    candidates = _generate_candidate_pool(n, w)
    scored = []   # (value, normalised vector)

    for idx, cand in enumerate(candidates):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        val = _objective(a, n, conv_fft_len)
        scored.append((val, a))
        print(f"[Screen {idx:03d}/{len(candidates)}] obj={val:.6f}")

    # Keep the most promising starts (up to eight)
    scored.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in scored[:8]]

    # ------------------------------------------------------------------
    # 3. Deep optimisation (Adam + sub‑gradient) on the top candidates
    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    for cid, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[DeepOpt start {cid}]")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )
        print(f"[DeepOpt end {cid}] best so far {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Local stochastic search around the global best
    # ------------------------------------------------------------------
    if time.time() + 2.0 < deadline and best_seq is not None:
        print("[Local search] starting")
        best_seq, best_val = _local_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            best_seq,
            best_val,
        )
        print(f"[Local search] finished, best {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Optional up‑sampling to a finer grid (8192) and a short refinement
    # ------------------------------------------------------------------
    if time.time() + 12.0 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # Interpolate the current best onto the finer grid
        if best_seq is not None:
            x_small = np.linspace(0.0, 1.0, n, endpoint=False)
            x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
            a_big = np.interp(x_big, x_small, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("[Upsample] start Adam+subgrad on larger support")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        print(f"[Upsample] finished, best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Finalisation
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator's allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return best_seq.tolist()
