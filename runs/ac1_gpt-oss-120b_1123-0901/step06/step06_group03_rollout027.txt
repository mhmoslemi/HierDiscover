def _simulated_annealing(a0, n, fft_len, deadline, max_iters=500000):
    a = a0.copy()
    best = a.copy()
    best_val = objective(a, n, fft_len)
    it = 0
    # schedule parameters
    sigma = 0.02  # initial max fraction to move
    temp = 0.02
    last_improve = time.time()
    while time.time() < deadline and it < max_iters:
        # choose indices based on gradient significance
        # compute gradient direction from current max conv
        conv = _full_convolution(a, fft_len)[:2*n-1]
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        # create weight w (same as subgrad)
        w = active.astype(float)
        if w.sum() > 0:
            w /= w.sum()
        else:
            w = np.zeros_like(conv)
        grad = 2 * _cross_correlation(w, a, _next_pow2(3*n-2))
        # pick i with high grad (positive) and j with low grad (negative)
        # Since we want to decrease max, move mass from large grad to small:
        # find i = argmax(grad), j = argmin(grad)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        # propose move delta
        delta = sigma * np.random.rand() * a[i]
        if delta > a[i]:
            delta = a[i]
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        val = objective(a_new, n, fft_len)
        # accept
        if val < best_val:
            best_val = val
            best = a_new.copy()
            a = a_new
            last_improve = time.time()
            # maybe reduce sigma slowly after improvement
        else:
            if np.random.rand() < np.exp(-(val - best_val) / max(temp, 1e-12)):
                a = a_new
        # schedule decay
        sigma *= 0.99995
        temp *= 0.99995
        it += 1
        if it % 5000 == 0:
            print(f"[SA n={n} iter={it}] best {best_val:.6f}")
        if time.time() - last_improve > 10.0:
            # random restart with slight perturbation
            a = a * (np.random.rand(n) * 0.2 + 0.9)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum < 1e-12:
                a = np.random.dirichlet(np.ones(n))
                a_sum = a.sum()
            a = a / a_sum
            sigma = 0.02
            temp = 0.02
            last_improve = time.time()
    return best, best_val

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    
    # 1. Determine base size
    n = 4096
    conv_fft_len = _next_pow2(2*n - 1)
    corr_fft_len = _next_pow2(3*n - 2)
    
    # warm start
    warm = globals().get('height_sequence_1')
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n-w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
        else:
            w = None
    else:
        w = None

    # generate candidates
    cand_pool = _generate_candidate_pool(n, w)
    best_seq = None
    best_val = float('inf')
    for cand in cand_pool:
        val = objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
    
    # 2. Deep optimization (Adam+Subgrad) on best_seq
    if time.time() < deadline:
        # run deep opt with limited time: we compute sub_deadline = deadline - 30 sec
        sub_deadline = deadline - 30.0  # keep some margin for later stages
        best_seq, best_val = _deep_optimize(best_seq, n, conv_fft_len, corr_fft_len,
                                            sub_deadline, best_seq, best_val)
    
    # 3. Simulated annealing refinement at current resolution
    if time.time() < deadline:
        best_seq, best_val = _simulated_annealing(best_seq, n, conv_fft_len,
                                                  deadline, max_iters=400000)
    
    # 4. Upsample to larger grid if time permits
    if time.time() + 5 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2*n_big - 1)
        # upsample by linear interpolation
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)

        # refine with SA on larger grid
        best_seq, best_val = _simulated_annealing(a_big, n_big, conv_fft_len_big,

    # final clip

def _simulated_annealing(a0, n, fft_len, deadline, max_iters=200000):
    a = a0.copy()
    best = a.copy()
    best_val = objective(a, n, fft_len)
    it = 0
    sigma = 0.02  # relative move size (max fraction of source mass)
    temp = 0.02
    last_improve = time.time()
    while time.time() < deadline and it < max_iters:
        # pick source index with mass > 0
        i = np.random.randint(0, n)
        if a[i] <= 0.0:
            continue
        # pick target distinct
        j = np.random.randint(0, n-1)
        if j >= i:
            j += 1
        # propose transfer
        delta = sigma * np.random.rand() * a[i]
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # evaluate
        val = objective(a_new, n, fft_len)
        # acceptance
        if val < best_val:
            best_val = val
            best = a_new.copy()
            a = a_new
            last_improve = time.time()
        else:
            # accept with probability exp(-(val - best_val) / temp)
            if temp > 0 and np.random.rand() < np.exp(-(val - best_val) / temp):
                a = a_new
        # schedule
        sigma *= 0.99995
        temp *= 0.99995
        it += 1
        # periodic logging
        if it % 5000 == 0:
            print(f"[SA n={n} iter={it}] current best {best_val:.6f}")

"""
search for a non‑negative step function that minimises

        F(a) = 2·n·max(a*a) / (∑a)² .

The method consists of three phases

1. **Diverse initial pool** – uniform, edge‑concentrated power‑law,
   double‑exponential, Gaussian and random Dirichlet vectors (plus a warm‑start
   if one is available).  Every vector is normalised to unit sum and the
   best few are retained.

2. **Adam + sub‑gradient refinement** – the retained vectors are optimised with
   the same Adam / mirror‑descent routine that the reference solution used.
   The optimiser runs until the majority of the time budget is spent.

3. **Stochastic‑annealing fine tuning** – starting from the best Adam result a
   simple simulated‑annealing on the simplex is performed (mass is moved
   between two randomly chosen indices).  If time permits the best shape is
   interpolated to a larger grid (8192 points) and the annealing is repeated.

All numbers are forced to be non‑negative and clipped to the evaluator’s
range (≤1000).  Progress is printed, and the routine respects the supplied
time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective value for a normalised sequence."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _generate_candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modestly diverse set of starting vectors."""
    pool = []

    # warm‑start and its rotations
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # uniform
    pool.append(np.full(n, 1.0 / n))

    # power‑law edges (U‑shaped)
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 12):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # double‑exponential (symmetric Laplace)
    i = np.arange(n, dtype=np.float64)
    for lam in np.linspace(0.1, 4.0, 12):
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        shape /= shape.sum()
        pool.append(shape)

    # Gaussian bumps (different widths)
    for sigma in np.linspace(n * 0.03, n * 0.45, 9):
        shape = np.exp(-0.5 * ((i - n / 2) ** 2) / (sigma ** 2))
        shape /= shape.sum()
        pool.append(shape)

    # single half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # random Dirichlet draws
    for alpha in np.logspace(-2, 0, 10):
        shape = np.random.dirichlet(np.full(n, alpha))
        pool.append(shape)

    return pool


# ----------------------------------------------------------------------
def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate (log‑sum‑exp) followed by an
    exponentiated‑gradient sub‑gradient phase.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5          # starting temperature
    lam_factor = 1.018

    last_improve = time.time()
    it = 0

    # ----------------------- Adam phase -----------------------
    while time.time() < deadline:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # current exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature increase (approach true max)
        lam = min(lam * lam_factor, 1e6)

        # shake if stagnant
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        if it > 500_000:
            break

    # -------------------- Sub‑gradient phase --------------------
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total

    eta0 = 0.6
    sub_last_improve = time.time()
    for sub_it in range(200_000):          # more iterations than original
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[n={n} Subgrad iter={sub_it:06d}] new best {val:.6f}")

        # active set mask
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (mirror‑descent on the simplex)
        eta = eta0 / np.sqrt(sub_it + 1)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                 # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 8.0:
            break

    return global_best_seq, global_best_val


# ----------------------------------------------------------------------
def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Simple SA on the simplex: repeatedly move a small amount of mass from
    a source index to a target index.  Accept improvements always and
    worse moves with a temperature‑dependent probability.
    """
    a = a0.copy()
    best = a.copy()
    best_val = _objective(a, n, fft_len)

    sigma = 0.03       # typical relative step size
    temp = 0.02        # initial temperature
    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < max_iters:
        # pick a source with non‑zero mass
        i = np.random.randint(0, n)
        if a[i] <= 0.0:
            continue
        # pick a different target
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction of source mass
        delta = sigma * np.random.rand() * a[i]
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        val = _objective(a_new, n, fft_len)

        if val < best_val:
            best_val = val
            best = a_new.copy()
            a = a_new
            last_improve = time.time()
        else:
            # Metropolis acceptance
            if temp > 0 and np.random.rand() < np.exp(-(val - best_val) / temp):
                a = a_new

        # schedule decay
        sigma *= 0.9999
        temp *= 0.9999
        it += 1

        if it % 5000 == 0:
            print(f"[SA n={n} iter={it}] best {best_val:.6f}")

        # if stuck for a while, shake the current point
        if time.time() - last_improve > 12.0:
            a = a * (0.9 + 0.2 * np.random.rand(n))
            a = np.maximum(a, 0.0)
            total = a.sum()
            if total < 1e-12:
                a = np.random.dirichlet(np.ones(n))
                total = a.sum()
            a /= total
            sigma = 0.03
            temp = 0.02
            last_improve = time.time()

    return best, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation metric.
    The routine respects the supplied time‑budget (default 1000 s) and returns
    the best sequence encountered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1️⃣  Primary support size and warm‑start handling
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        w = np.asarray(warm_start, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
        else:
            w = None
    else:
        w = None

    # --------------------------------------------------------------
    # 2️⃣  Build a pool of diverse starts and keep the most promising ones
    candidates = _generate_candidate_pool(n, w)

    best_seq = None
    best_val = float("inf")
    scored = []   # (value, vector)

    for idx, cand in enumerate(candidates):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        val = _objective(a, n, conv_fft_len)
        scored.append((val, a))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        print(f"[screen {idx:03d}] obj={val:.6f}")

    # keep a handful of top vectors
    scored.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in scored[:6]]

    # --------------------------------------------------------------
    # 3️⃣  Deep optimisation (Adam + long sub‑gradient phase)
    # allocate roughly 70 % of the remaining budget to this stage
    deep_deadline = time.time() + (deadline - time.time()) * 0.70
    for cid, a0 in enumerate(top_candidates):
        if time.time() >= deep_deadline:
            break
        print(f"[deep start {cid}]")
        best_seq, best_val = _deep_optimize(
            a0, n, conv_fft_len, corr_fft_len, deep_deadline, best_seq, best_val
        )

    # --------------------------------------------------------------
    # 4️⃣  Simulated‑annealing fine‑tuning on the current resolution
    if time.time() < deadline:
        print("[SA start on 4096]")
        best_seq, best_val = _simulated_annealing(
            best_seq, n, conv_fft_len, deadline, max_iters=500_000
        )

    # --------------------------------------------------------------
    # 5️⃣  Optional up‑sampling and additional refinement
    if time.time() + 5 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)

        # linear interpolation to the larger grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        print("[SA start on 8192]")
        best_seq, best_val = _simulated_annealing(
            a_big, n_big, conv_fft_len_big, deadline, max_iters=600_000
        )

    # --------------------------------------------------------------
    # 6️⃣  Final safety – ensure a valid return value
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"[final] best objective ≈ {best_val:.6f}")
    return best_seq.tolist()
