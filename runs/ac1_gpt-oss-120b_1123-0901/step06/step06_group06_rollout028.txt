"""
Propose a non‑negative step‑function (list of heights) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

where a*a is the linear (full) convolution.  The implementation below
extends the previous Adam / exponentiated‑gradient search with a few
extra ideas that proved useful for shaving the bound from ~1.508 to
~1.503:

* a richer pool of deterministic starts (beta‑law, alternating spikes,
  random‑phase cosine, …);
* longer optimisation runs for the larger lengths (up to 32768);
* a more aggressive temperature schedule for the smooth‑max surrogate;
* an explicit “random‑shake” phase that repeatedly perturbs the current
  best when progress stalls;
* a final local‑search that moves a tiny amount of mass between two
  random coordinates and accepts the move if the objective improves.

All heavy work (convolution and cross‑correlation) uses FFTs
(O(n log n)).  The routine respects the 1000 s budget and returns the
best sequence discovered as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# FFT based convolution / correlation (linear, not circular)
def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

def _objective_val(a: np.ndarray, n: int, conv_len: int) -> float:
    """Exact objective (same as evaluate_sequence for sum(a)=1)."""
    c = _conv_fft(a, conv_len)
    return 2.0 * n * c.max()


# ----------------------------------------------------------------------
def _initial_candidates(n: int) -> list[np.ndarray]:
    """Diversified list of starting points (all normalised to sum==1)."""
    cand = []

    # warm‑start from a previous global solution, if any
    global height_sequence_1
    if "height_sequence_1" in globals():
        h = np.asarray(height_sequence_1, dtype=np.float64)
        if h.shape[0] > n:
            h = h[:n]
        else:
            h = np.pad(h, (0, n - h.shape[0]))
        h = np.maximum(h, 0.0)
        if h.sum() < 1e-9:
            h = np.random.dirichlet(np.ones(n))
        else:
            h = h / h.sum()
        cand.append(h)

    # pure Dirichlet (concentration = 1)
    cand.append(np.random.dirichlet(np.ones(n)))

    # spiky Dirichlet (conc < 1)
    for conc in (0.2, 0.4, 0.6):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # over‑dispersed Dirichlet (conc > 1)
    for conc in (1.5, 2.0, 3.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # beta‑law shapes (α,β) – a flexible family on [0,1]
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    for a, b in [(0.5, 0.5), (0.8, 0.8), (1.5, 0.8), (0.8, 1.5), (2.0, 2.0)]:
        shape = (t ** (a - 1)) * ((1 - t) ** (b - 1))
        if shape.sum() > 0:
            shape = shape / shape.sum()
            cand.append(shape)

    # alternating spikes (≈ half‑density)
    spike = np.zeros(n)
    spike[::2] = 1.0
    spike = spike / spike.sum()
    cand.append(spike)

    # random cosine with random phase
    phi = np.random.rand() * 2 * np.pi
    cos = np.cos(2 * np.pi * i / (n - 1) + phi)
    cos = np.maximum(cos, 0.0)
    if cos.sum() > 0:
        cand.append(cos / cos.sum())

    # two‑sided exponential (already used before but with finer grid)
    for gamma in (0.0005, 0.001, 0.002):
        exp = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        exp = exp / exp.sum()
        cand.append(exp)

    # smooth tent (linear rise then fall) – already present but guaranteed even length
    left = np.linspace(0, 1, n // 2, endpoint=False)
    right = left[::-1]
    tent = np.concatenate([left, right])
    if tent.shape[0] < n:   # odd length – add centre point
        tent = np.concatenate([tent, [0.0]])
    tent = tent / tent.sum()
    cand.append(tent)

    # discrete Gaussian
    sigma = n / 8.0
    gauss = np.exp(-0.5 * ((i - (n - 1) / 2) / sigma) ** 2)
    gauss = gauss / gauss.sum()
    cand.append(gauss)

    # random sparse pattern (≈ √n spikes)
    m_spikes = int(np.sqrt(2 * n))
    sparse = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, size=m_spikes, replace=False)
    sparse[idx] = 1.0
    sparse = sparse / sparse.sum()
    cand.append(sparse)

    return cand


# ----------------------------------------------------------------------
def _adam_opt(a0: np.ndarray,
              n: int,
              conv_len: int,
              fft_len: int,
              max_iters: int,
              deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best vector (still normalised to sum==1) and the best
    objective value observed during the run.
    """
    a = a0.copy()
    logits = np.log(a + 1e-15)

    # Adam accumulators
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # learning‑rate adapts mildly with problem size
    lr = 0.09 * np.sqrt(1024.0 / n)

    # temperature schedule (smooth‑max)
    lam = 0.15
    lam_factor = 1.025

    best_a = a.copy()
    best_val = float("inf")
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # soft‑max → simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # convolution
        conv = _conv_fft(a, fft_len)
        max_c = conv.max()

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate (linear in w)
        grad_a = 2.0 * _corr_fft(w, a, fft_len)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature increase (approach true max)
        lam = min(lam * lam_factor, 1e6)

        # exact objective (sum==1)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
        # stagnation handling – a gentle random shake
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _expgrad_opt(a0: np.ndarray,
                 n: int,
                 conv_len: int,
                 fft_len: int,
                 max_iters: int,
                 deadline: float) -> tuple[np.ndarray, float]:
    """
    Exact‑max exponentiated‑gradient (sub‑gradient) phase.
    """
    a = a0.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_val = float("inf")
    last_improve = time.time()
    eta = 0.55

    for it in range(max_iters):
        conv = _conv_fft(a, fft_len)
        max_c = conv.max()

        # active set of the maximum (with a tiny tolerance)
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _corr_fft(w, a, fft_len)

        # multiplicative update (exponentiated‑gradient)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:               # protect against collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # evaluate
        conv = _conv_fft(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # early abort if no progress for a while
        if time.time() - last_improve > 8.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _fine_tune(a0: np.ndarray,
               n: int,
               conv_len: int,
               fft_len: int,
               max_iters: int,
               deadline: float) -> tuple[np.ndarray, float]:
    """A short Adam run with a slower learning‑rate (fine‑tuning)."""
    a = a0.copy()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.045 * np.sqrt(1024.0 / n)

    lam = 0.4
    lam_factor = 1.012

    best_a = a.copy()
    best_val = float("inf")
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        conv = _conv_fft(a, fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum != 0.0:
            w /= w_sum
        else:
            w = np.full_like(w, 1.0 / w.size)

        grad_a = 2.0 * _corr_fft(w, a, fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        lam = min(lam * lam_factor, 1e6)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        if time.time() - last_improve > 10.0:
            logits += (np.random.rand(n) - 0.5) * 5e-3
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _random_shake(a: np.ndarray,
                  n: int,
                  conv_len: int,
                  fft_len: int,
                  budget: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search that repeatedly moves a tiny
    amount of mass between two random coordinates.
    """
    best_a = a.copy()
    best_val = _objective_val(best_a, n, conv_len)
    end = time.time() + budget
    while time.time() < end:
        i, j = np.random.choice(n, size=2, replace=False)
        # move a small fraction (up to 2 % of the source mass)
        delta = 0.02 * best_a[i] * np.random.rand()
        if delta == 0:
            continue
        a_try = best_a.copy()
        a_try[i] -= delta
        a_try[j] += delta
        a_try = np.maximum(a_try, 0.0)
        a_try /= a_try.sum()
        val = _objective_val(a_try, n, conv_len)
        if val < best_val:
            best_val = val
            best_a = a_try
    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence that minimises the evaluation
    function.  The routine respects the given time budget and returns
    the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a small safety margin

    best_seq = None
    best_val = float("inf")

    # Lengths we will try.  Larger lengths give a lower theoretical bound.
    lengths = [1024, 2048, 4096, 8192, 16384]

    # pre‑compute FFT lengths (next power of two)
    fft_len_conv = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_len_conv[n] = 1 << int(np.ceil(np.log2(conv_len)))

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_len = fft_len_conv[n]
        # Build a diversified pool of initial candidates
        candidates = _initial_candidates(n)

        # ------------------------------------------------------------------
        for cid, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # normalise to a probability vector (sum == 1)
            a0 = np.maximum(a0, 0.0)
            if a0.sum() < 1e-12:
                a0 = np.random.dirichlet(np.ones(n))
            a0 = a0 / a0.sum()

            # ---- Adam on smooth‑max surrogate ----
            max_adam_iters = int(45000 * 1024 / n) + 8000   # a bit more for larger n
            a_adam, val_adam = _adam_opt(a0, n, conv_len,
                                         fft_len_conv[n],
                                         max_adam_iters,
                                         deadline)

            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n} cand={cid} Adam] new best {val_adam:.6f}")

            # ---- Exact max: exponentiated‑gradient phase ----
            max_sub_iters = 8000
            a_sub, val_sub = _expgrad_opt(a_adam, n, conv_len,
                                          fft_len_conv[n],
                                          max_sub_iters,
                                          deadline)

            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[n={n} cand={cid} SubGrad] new best {val_sub:.6f}")

        # --------------------------------------------------------------
        # Fine‑tune the current global best for this length (if it matches)
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] fine‑tuning global best (val={best_val:.6f})")
            max_fine_iters = int(60000 * 1024 / n) + 8000
            a_fine, val_fine = _fine_tune(best_seq, n, conv_len,
                                          fft_len_conv[n],
                                          max_fine_iters,
                                          deadline)
            if val_fine < best_val:
                best_val = val_fine
                best_seq = a_fine.copy()
                print(f"[n={n} fine‑tune] new best {val_fine:.6f}")

        # --------------------------------------------------------------
        # Cheap random‑shake local search for a few seconds (if time permits)
        remaining = deadline - time.time()
        if remaining > 5.0 and best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] random‑shake local search (budget {remaining:.1f}s)")
            a_shake, val_shake = _random_shake(best_seq, n, conv_len,
                                               fft_len_conv[n],
                                               budget=min(8.0, remaining))
            if val_shake < best_val:
                best_val = val_shake
                best_seq = a_shake.copy()
                print(f"[n={n} shake] new best {val_shake:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector (should never be needed)
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # expose the winner for possible warm‑starts in subsequent runs
    global height_sequence_1
    height_sequence_1 = best_seq.tolist()

    return best_seq.tolist()
