"""Hybrid optimisation for low max‑convolution step function.

The algorithm combines:

1. A diverse parametric sampling stage and quick ranking.
2. Adam optimisation on a smooth‑max surrogate of the max‑convolution.
3. Exact sub‑gradient optimisation on the true max‑convolution.
4. A final simulated‑annealing / stochastic‐gradient refinement on the
   log‑softmax (simplex) parametrisation.
5. Optional up‑sampling to a larger grid when time permits.

All steps respect the 1000 s budget and keep the sequence non‑negative.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT of length fft_len."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Fast evaluation of the objective.
    The vector a is assumed to be normalised (sum≈1), therefore
    the denominator is ≈1 and the objective is 2·n·max(conv).
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp surrogate of the max‑convolution.
    Returns the best vector/value seen during the run.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5               # start temperature for smooth‑max
    lam_factor = 1.02       # moderately fast schedule
    lam_cap = 1e6

    last_improve = time.time()
    it = 0
    max_iter = 200_000

    while time.time() < deadline and it < max_iter:
        # simplex via soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution & true objective (cheap)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        ws = w.sum()
        if ws == 0 or np.isnan(ws):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws

        # gradient of the surrogate (through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional random shake if stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient steps on the true max‑convolution.
    """
    a = best_seq if best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()

    eta = 0.6
    last_improve = time.time()
    max_iter = 15_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it:05d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam → sub‑gradient refinement, updating the global best.
    """
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """
    Produce a diverse set of candidate vectors using several parametric families.
    All candidates are normalised to sum ≈ 1.
    """
    candidates = []

    # Warm‑start and rotations
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # Uniform
    candidates.append(np.full(n, 1.0 / n))

    # Random Dirichlet with various concentrations
    for conc in (0.02, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # Edge‑exponential families, optionally mixed with uniform
    for alpha in np.logspace(-2, 1.0, num=16):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge = edge / edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # Edge‑power‑law families
    for p in np.linspace(0.2, 5.0, num=16):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge = edge / edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # Double‑exponential mixtures
    for a1 in (0.01, 0.1, 1.0):
        for a2 in (0.01, 0.1, 1.0):
            mix = 0.5 * (np.exp(-a1 * i) + np.exp(-a2 * (n - 1 - i)))
            mix = mix / mix.sum()
            for w in (0.0, 0.5, 1.0):
                shape = w * mix + (1.0 - w) / n
                candidates.append(shape)

    # Two‑Gaussian edges
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # Sinusoidal half‑wave
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # Plateau‑exponential mixture (central flat region)
    for k_frac in (0.05, 0.1, 0.2):
        k = max(1, int(k_frac * n))
        for alpha in (0.1, 0.5, 2.0):
            left = np.exp(-alpha * np.arange(k))
            right = np.exp(-alpha * np.arange(k)[::-1])
            middle = np.ones(n - 2 * k)
            shape = np.concatenate([left, middle, right])
            shape = shape / shape.sum()
            candidates.append(shape)

    return candidates

def _simulated_annealing_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Stochastic refinement on the simplex using a log‑softmax (exponential) map.
    Accepts uphill moves with a linearly cooling temperature.
    """
    eps = 1e-15
    u = np.log(a0 + eps)

    # start from the currently best sequence if it is better
    if best_seq is not None:
        cur_seq = best_seq.copy()
        cur_val = best_val
        u = np.log(cur_seq + eps)
    else:
        cur_seq = a0.copy()
        conv = np.fft.irfft(np.fft.rfft(cur_seq, conv_fft_len) *
                            np.fft.rfft(cur_seq, conv_fft_len), conv_fft_len)
        cur_val = 2.0 * n * conv[: 2 * n - 1].max()

    step = 0.08          # std of Gaussian perturbation in log‑space
    temp0 = 4e-4         # initial temperature
    temp = temp0
    start_time = time.time()
    last_improve = start_time
    it = 0

    while time.time() < deadline:
        # propose a new point
        delta = np.random.randn(n) * step
        u_new = u + delta
        mx = u_new.max()
        exp_u = np.exp(u_new - mx)
        a_new = exp_u / exp_u.sum()

        # evaluate
        Fa = np.fft.rfft(a_new, conv_fft_len)
        conv = np.fft.irfft(Fa * Fa, conv_fft_len)
        max_c = conv[: 2 * n - 1].max()
        new_val = 2.0 * n * max_c

        improve = new_val < cur_val
        if improve or np.random.rand() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            # accept
            u = u_new
            cur_seq = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = a_new.copy()
                last_improve = time.time()
                print(f"[SA iter={it:06d}] new best {new_val:.6f}")

        it += 1

        # cooling & step‑size reduction
        if it % 5000 == 0:
            elapsed = time.time() - start_time
            total = deadline - start_time
            # linear cooling
            temp = temp0 * max(0.0, 1.0 - elapsed / total)
            step *= 0.97

            # random shake if stuck
            if time.time() - last_improve > 20.0:
                # re‑centre around the current best with a modest perturbation
                u = np.log(best_seq + eps) + np.random.randn(n) * step * 2
                last_improve = time.time()

        # early stop if step becomes negligible
        if step < 1e-7:
            break

    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a non‑negative step function that
    minimises the evaluation metric.
    Uses a three‑stage pipeline (parametric sampling → deep optimisation →
    simulated‑annealing refinement) and optionally upsamples to a larger grid.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Basic grid size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start from a previously found construction if present
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – cheap generation & pre‑selection
    all_candidates = _generate_parametric_candidates(n, warm_start)

    cand_vals = []
    for idx, cand in enumerate(all_candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    cand_vals.sort(key=lambda x: x[0])
    top_k = 20
    top_candidates = [c for _, c in cand_vals[:top_k]]

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation of the selected starts
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation on start #{start_id + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – stochastic refinement (simulated annealing) on the current best
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 20.0:
        print("--- Simulated‑annealing refinement (n=4096) ---")
        best_seq, best_val = _simulated_annealing_refine(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 4 – if time permits, move to a finer grid (n = 8192)
    time_left = deadline - time.time()
    if time_left > 60.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        # upscale the current best (linear interpolation)
        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big = a_big / a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Optimising on larger grid (n=8192) ---")
        best_seq_big, best_val_big = _deep_optimize(
            a_big,

"""Hybrid optimisation for low max‑convolution step function.

The algorithm combines:

1. A diverse parametric sampling stage and quick ranking.
2. Adam optimisation on a smooth‑max surrogate of the max‑convolution.
3. Exact sub‑gradient optimisation on the true max‑convolution.
4. A final simulated‑annealing / stochastic‑gradient refinement on the
   log‑softmax (simplex) parametrisation.
5. Optional up‑sampling to a larger grid when time permits.

All steps respect the 1000 s budget and keep the sequence non‑negative.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT of length fft_len."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Fast evaluation of the objective.
    The vector a is assumed to be normalised (sum≈1), therefore
    the denominator is ≈1 and the objective is 2·n·max(conv).
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Adam on a log‑sum‑exp surrogate of the max‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5               # start temperature for smooth‑max
    lam_factor = 1.02       # schedule factor
    lam_cap = 1e6

    last_improve = time.time()
    it = 0
    max_iter = 200_000

    while time.time() < deadline and it < max_iter:
        # simplex via soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution & true objective (cheap)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        ws = w.sum()
        if ws == 0 or np.isnan(ws):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws

        # gradient of the surrogate (through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional random shake if stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient steps on the true max‑convolution."""
    a = best_seq if best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()

    eta = 0.6
    last_improve = time.time()
    max_iter = 15_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it:05d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """Run Adam → sub‑gradient refinement, updating the global best."""
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """Produce a diverse set of candidate vectors using several parametric families."""
    candidates = []

    # Warm‑start and rotations
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Random Dirichlet with various concentrations
    for conc in (0.02, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # Edge‑exponential families, optionally mixed with uniform
    for alpha in np.logspace(-2, 1.0, num=16):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge = edge / edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # Edge‑power‑law families
    for p in np.linspace(0.2, 5.0, num=16):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge = edge / edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # Double‑exponential mixtures
    for a1 in (0.01, 0.1, 1.0):
        for a2 in (0.01, 0.1, 1.0):
            mix = 0.5 * (np.exp(-a1 * i) + np.exp(-a2 * (n - 1 - i)))
            mix = mix / mix.sum()
            for w in (0.0, 0.5, 1.0):
                shape = w * mix + (1.0 - w) / n
                candidates.append(shape)

    # Two‑Gaussian edges
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # Sinusoidal half‑wave
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # Plateau‑exponential mixture (central flat region)
    for k_frac in (0.05, 0.1, 0.2):
        k = max(1, int(k_frac * n))
        for alpha in (0.1, 0.5, 2.0):
            left = np.exp(-alpha * np.arange(k))
            right = np.exp(-alpha * np.arange(k)[::-1])
            middle = np.ones(n - 2 * k)
            shape = np.concatenate([left, middle, right])
            shape = shape / shape.sum()
            candidates.append(shape)

    return candidates

def _simulated_annealing_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Stochastic refinement on the simplex using a log‑softmax (exponential) map.
    Accepts uphill moves with a linearly cooling temperature.
    """
    eps = 1e-15
    # initialise log‑space representation
    if best_seq is not None:
        cur_seq = best_seq.copy()
        cur_val = best_val
        u = np.log(cur_seq + eps)
    else:
        cur_seq = a0.copy()
        Fa = np.fft.rfft(cur_seq, conv_fft_len)
        conv = np.fft.irfft(Fa * Fa, conv_fft_len)
        cur_val = 2.0 * n * conv[: 2 * n - 1].max()
        u = np.log(cur_seq + eps)

    step = 0.08          # std. of Gaussian perturbation in log‑space
    temp0 = 4e-4         # initial temperature
    temp = temp0
    start_time = time.time()
    last_improve = start_time
    it = 0

    while time.time() < deadline:
        # propose a new point in log‑space
        delta = np.random.randn(n) * step
        u_new = u + delta
        mx = u_new.max()
        exp_u = np.exp(u_new - mx)
        a_new = exp_u / exp_u.sum()

        # evaluate objective (fast via FFT)
        Fa = np.fft.rfft(a_new, conv_fft_len)
        conv = np.fft.irfft(Fa * Fa, conv_fft_len)
        max_c = conv[: 2 * n - 1].max()
        new_val = 2.0 * n * max_c

        improve = new_val < cur_val
        if improve or np.random.rand() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            u = u_new
            cur_seq = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = a_new.copy()
                last_improve = time.time()
                print(f"[SA iter={it:06d}] new best {new_val:.6f}")

        it += 1

        # cooling and step‑size decay
        if it % 5000 == 0:
            elapsed = time.time() - start_time
            total = deadline - start_time
            temp = temp0 * max(0.0, 1.0 - elapsed / total)   # linear cooling
            step *= 0.97

            if time.time() - last_improve > 20.0:
                # random shake around the current best
                u = np.log(best_seq + eps) + np.random.randn(n) * step * 2
                last_improve = time.time()

        if step < 1e-7:
            break

    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a non‑negative step function that
    minimises the evaluation metric.  Uses a three‑stage pipeline
    (parametric sampling → deep optimisation → simulated‑annealing refinement)
    and optionally upsamples to a larger grid.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Base grid size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start from a previously found construction if present
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – cheap generation & pre‑selection
    all_candidates = _generate_parametric_candidates(n, warm_start)

    cand_vals = []
    for idx, cand in enumerate(all_candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    cand_vals.sort(key=lambda x: x[0])
    top_k = 20
    top_candidates = [c for _, c in cand_vals[:top_k]]

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation of the selected starts
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation on start #{start_id + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – stochastic refinement (simulated annealing) on the current best
    if best_seq is not None and (deadline - time.time()) > 20.0:
        print("--- Simulated‑annealing refinement (n=4096) ---")
        best_seq, best_val = _simulated_annealing_refine(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 4 – finer grid (n = 8192) if time permits
    if (deadline - time.time()) > 60.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)

        if best_seq is not None:
            # linear interpolation to the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big = a_big / a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Optimising on larger grid (n=8192) ---")
        best_seq_big, best_val_big = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            _next_pow2(3 * n_big - 2),
            deadline,
            best_seq,
            best_val,
        )

        # Final SA on the big grid if time remains
        if best_seq_big is not None and (deadline - time.time()) > 30.0:
            print("--- Simulated‑annealing refinement (n=8192) ---")
            best_seq_big, best_val_big = _simulated_annealing_refine(
                best_seq_big,
                n_big,
                conv_fft_len_big,
                deadline,
                best_seq_big,
                best_val_big,
            )

        if best_seq_big is not None and best_val_big < best_val:
            best_seq = best_seq_big
            best_val = best_val_big
            n = n_big

    # ------------------------------------------------------------------
    # Fallback – uniform if nothing succeeded
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator limits (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
