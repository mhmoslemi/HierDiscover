"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Initialisation** – Build a diverse set of feasible candidates for a
   given length n (uniform, Dirichlet, exponential, Hann/Blackman windows,
   sparse spikes, and the previously found construction if available).
   All candidates are projected onto the simplex
   {a_i≥0, Σa_i = √(2 n)} and symmetrised (a ← (a+rev a)/2).

2. **Genetic optimisation** – A small population (≈15) is evolved.
   *Selection*: tournament of two random individuals.
   *Crossover*: convex combination α·p1+(1‑α)·p2.
   *Mutation*: additive Gaussian noise (≈2 % of the mean mass per coordinate).
   *Symmetrisation*: applied with probability 0.4.
   *Replacement*: child replaces the worst individual if it improves the
   max‑convolution value (the exact objective when the sum is normalised).
   The loop runs until the time budget is exhausted.

3. **Projected‑gradient polishing** – The best individual from the GA is
   further refined with the exact sub‑gradient method (hard‑max PGD) from
   the previous state‑of‑the‑art code.  This can exploit the active set of
   binding convolution indices for fast convergence.

4. **Stochastic local refinement** – A cheap simulated‑annealing hill‑climb
   (mass transfers between random positions) is applied to the polished
   sequence for a few seconds.

The whole search respects the 1000 s budget, dynamically allocating time
to different lengths (≈±10 % of the size of the previously best sequence)

def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x | x_i ≥ 0, Σx_i = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Exact max of a∗a (linear convolution) using a single FFT."""
    A = rfft(a, n=Nfft)
    conv = irfft(A * A, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Generate a small diverse pool of feasible start vectors."""
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0
             - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    init.append(black / black.sum() * target_sum)

    # 7) small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    # 8) triangular shape (peak in the centre)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    init.append(tri / tri.sum() * target_sum)

    # 9) sparse random spikes (helps explore highly non‑uniform shapes)
    k = max(2, min(15, n // 100))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k)
    vals = vals / vals.sum() * target_sum
    sparse = np.zeros(n)
    sparse[idx] = vals
    init.append(sparse)

    return init

def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected Gradient Descent (hard‑max) on the exact max‑convolution.
    Keeps the best vector observed.
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise on the simplex and enforce sum
    a = a0.copy()
    a = a * target_sum / a.sum()

    Nfft = 1 << ((3 * n - 2).bit_length())
    step0 = 0.001 * target_sum
    eps_active = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current convolution and max
        A = rfft(a, n=Nfft)
        conv = irfft(A * A, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD n={n:5d} it={it:8d}] new best = {best_score:.6f}")

        # active set (all entries within eps of the max)
        mask = conv >= M - eps_active
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = mask.astype(float) / float(cnt)

        # weighted reverse convolution = gradient of max (sub‑gradient)
        W_rev = rfft(w[::-1], n=Nfft)
        rev = irfft(W_rev * A, n=Nfft)[: len(w) + n - 1]
        idx_start = 2 * n - 2
        rev = rev[idx_start - np.arange(n)]

        g = 2.0 * rev  # ∇_a max(a∗a)

        lr = step0 / math.sqrt(it)
        a_tmp = a - lr * g
        a = _project_onto_simplex(a_tmp, target_sum)

        # occasional jitter to avoid stagnation
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)

    return best_seq, best_score


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on the simplex:
    move a small fraction of mass from a random index i to j.
    Accept improvements always; accept worsening moves with a decaying
    simulated‑annealing probability.
    """
    target_sum = a.sum()           # already √(2 n)
    cur_seq = a.copy()
    cur_score = best_score
    Nfft = 1 << ((2 * n - 1).bit_length())
    max_steps = 200_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        # transfer a modest fraction of the source mass
        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        A = rfft(new_seq, n=Nfft)
        conv = irfft(A * A, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score
