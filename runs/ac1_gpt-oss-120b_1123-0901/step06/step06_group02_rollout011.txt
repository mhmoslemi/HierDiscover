"""
Hybrid step‑function optimizer with a targeted mass‑transfer hill‑climb.

The objective 2·n·max(conv(a,a))/sum(a)² is scale‑invariant,
so we fix Σaᵢ = √(2 n) and minimise the peak of the
autoconvolution.  The search proceeds in four stages:

1.  Diverse random initialisations (exponential, power‑law,
    Hann/Hamming windows, Dirichlet draws) over a wide range of
    lengths; the best candidate is retained.
2.  Adam optimisation on a smooth log‑sum‑exp surrogate of the
    max‑convolution, using a soft‑max parametrisation that respects
    the simplex constraint.
3.  Exact projected sub‑gradient descent that follows the gradient at
    the current worst convolution index.
4.  A *targeted* hill‑climb: at each step the index k where the
    convolution attains its maximum is identified, the gradient
    2·a_{k‑i} is formed, and mass is moved from a donor (large
    gradient entry) to a recipient (small gradient entry).  A line
    search on the step size is performed; moves are accepted only
    when they strictly lower the peak.  This focuses the search on
    the most harmful contributions and typically yields a finer
    reduction than a purely random annealing step.

All phases respect the supplied time budget (default 1000 s) and
share a global best solution.  The routine returns a plain Python
list of non‑negative heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of real vectors x and y."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # Clip tiny negatives caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]          # sorted descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)     # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1        # zero‑based index of last positive entry
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the log‑it (soft‑max) parametrisation
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.10,
    min_tau_factor: float = 1e-4,
    alpha0: float = 0.02,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed‑max surrogate.
    Returns the best coefficient vector (on the simplex) and its exact
    max‑convolution value.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # ----- map logits to the simplex (a = target_sum * softmax(x)) -----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)       # stable soft‑max
        p = exp_shift / np.sum(exp_shift)   # Σ p = 1
        a = target_sum * p

        # ----- exact max of the convolution (used for logging) -----
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam n={n} new best max={cur_max:.6f}")

        # ----- smoothed max (log‑sum‑exp) weights -----
        tau = max(init_tau_factor * cur_max * (0.99 ** it), min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t. a: 2·a⊛w_rev  -----
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # ----- back‑prop to logits (chain rule) -----
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slow decay of learning rate
        alpha *= 0.9995

    # final exact evaluation (max‑convolution) of the best vector
    final_val = _full_convolution_fft(best_a, best_a).max()
    return best_a, float(final_val)


# ----------------------------------------------------------------------
# Targeted projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _targeted_subgradient(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent that always follows the current worst index."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step = 0.2 * target_sum / n          # initial step size

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())            # index of current worst peak
        # gradient of conv_k w.r.t. a
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # take a projected step
        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)

        cand_val = _full_convolution_fft(cand, cand).max()
        if cand_val < best_val - 1e-12:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= 1.02                       # modestly increase step after success
            if it % 200 == 0:
                print(f"[{time.time():.2f}] subgrad it={it} val={best_val:.6f}")
        else:
            step *= 0.70                       # reduce step on failure
            # occasional jitter to avoid stagnation
            if np.random.rand() < 0.07:
                jitter = np.random.rand(n) * 1e-4
                a = _project_onto_simplex(a + jitter, target_sum)

        if best_val < 1.503:
            break
    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted hill‑climb that moves mass from donors with large
# gradient contribution to recipients with small contribution.
# ----------------------------------------------------------------------
def _targeted_hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer hill‑climb guided by the worst‑index gradient."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    start = time.time()
    it = 0

    # hyper‑parameters for the donor‑recipient move
    alpha = 0.5          # start by moving up to 50 % of the donor mass
    min_alpha = 1e-5
    eps = 1e-12

    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        k = int(conv.argmax())            # index of the current peak

        # Build the gradient vector v_i = a_{k-i} (0 if out of range)
        v = np.zeros(n)
        low_i = max(0, k - (n - 1))
        high_i = min(k, n - 1)            # inclusive
        if low_i <= high_i:
            idx = np.arange(low_i, high_i + 1, dtype=int)
            v[idx] = a[k - idx]

        # Choose donor (largest v_i among positions that actually hold mass)
        donor_mask = a > eps
        if not np.any(donor_mask):
            break
        v_donor = np.where(donor_mask, v, -np.inf)
        i_d = int(np.argmax(v_donor))
        if v_donor[i_d] == -np.inf:
            break          # no donor with positive gradient found

        # Choose recipient (smallest v_j)
        j_r = int(np.argmin(v))

        # If donor and recipient are the same or the move would not help,
        # fall back to a small random redistribution.
        if i_d == j_r or v[i_d] <= v[j_r] + eps:
            # random small move (similar to the original hill‑climb)
            i_rand, j_rand = np.random.randint(0, n, size=2)
            if a[i_rand] <= 0:
                continue
            delta = a[i_rand] * np.random.rand() * 0.08
            a_new = a.copy()
            a_new[i_rand] -= delta
            a_new[j_rand] += delta
        else:
            # Try to move a fraction of the donor mass towards the recipient.
            improved = False
            cur_alpha = alpha
            while cur_alpha > min_alpha:
                delta = a[i_d] * cur_alpha
                if delta <= 0:
                    break
                a_new = a.copy()
                a_new[i_d] -= delta
                a_new[j_r] += delta
                new_val = _full_convolution_fft(a_new, a_new).max()
                if new_val < cur_max - 1e-12:
                    improved = True
                    break
                cur_alpha *= 0.5
            if not improved:
                # fallback to a random tiny move
                i_rand, j_rand = np.random.randint(0, n, size=2)
                if a[i_rand] <= 0:
                    continue
                delta = a[i_rand] * np.random.rand() * 0.08
                a_new = a.copy()
                a_new[i_rand] -= delta
                a_new[j_rand] += delta

        # Accept the move
        a = a_new
        new_val = _full_convolution_fft(a, a).max()
        if new_val < best_val - 1e-12:
            best_val = new_val
            best_a = a.copy()
            print(f"[{time.time():.2f}] targeted hillclimb it={it} new best={best_val:.6f}")
            # When we succeed we allow a slightly larger step next time
            alpha = min(alpha * 1.1, 0.5)
        else:
            # If the move did not improve the global best, we shrink the step size
            alpha = max(alpha * 0.7, min_alpha)

        # Stop early if we have already reached the target bound
        if best_val < 1.503:
            break

        # Periodic re‑projection to avoid drift due to floating‑point errors
        if it % 2500 == 0:
            a = _project_onto_simplex(a, target_sum)

    return best_a, best_val

        elif shape == "hann":
            i = np.arange(n, dtype=float)
            a0 = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        elif shape == "hamming":
            i = np.arange(n, dtype=float)
            a0 = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
        else:  # dirichlet (uniform on simplex)
            a0 = np.random.dirichlet(np.ones(n))

        # optional tiny noise
        if np.random.rand() < 0.25:
            noise = np.random.randn(n) * np.max(a0) * 0.02
            a0 = np.maximum(a0 + noise, 0.0)

        # normalise to the required L1 sum
        a0 = a0 * (target_sum / a0.sum())
        _record(a0)

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on a smoothed max‑convolution
    # ------------------------------------------------------------------
    while best_seq is not None and time.time() < deadline - 150.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        # initialise logits from the current best (soft‑max parametrisation)
        x0 = np.log(best_seq + 1e-12)

        run_deadline = time.time() + min(20.0, deadline - time.time() - 30.0)
        seq_local, val_local = _optimise_adam_softmax(
            n, x0, target_sum, run_deadline,
            init_tau_factor=0.12,
            min_tau_factor=5e-5,
            alpha0=0.025,
        )
        if val_local < best_score:
            best_score = val_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] after Adam refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Targeted projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_deadline = deadline - 40.0   # reserve some time for the final hill‑climb
        seq_sub, val_sub = _targeted_subgradient(best_seq.copy(),
                                                 target_sum,
                                                 sub_deadline)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑gradient best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Final targeted hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        hill_deadline = deadline
        seq_hill, val_hill = _targeted_hillclimb(best_seq.copy(),
                                                target_sum,
                                                hill_deadline)
        if val_hill < best_score:
            best_score = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] after targeted hill‑climb best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be used)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = _full_convolution_fft(best_seq, best_seq).max()
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # Return a plain Python list as required by the evaluator
    return best_seq.tolist()
