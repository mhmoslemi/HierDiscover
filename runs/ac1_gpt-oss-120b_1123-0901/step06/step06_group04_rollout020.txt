        delta = np.random.uniform(-0.01,0.01) * a[i]  # transfer proportion
        # ensure positivity
        if a[i] - delta < 0 or a[j] + delta < 0: continue
        a[i] -= delta
        a[j] += delta
        val = true_objective(a)
        if val < best_val:
            best_val = val
            best = a.copy()
        else:
            # revert with probability? we can revert always
            a[i] += delta
            a[j] -= delta
    return best, best_val

"""
Propose a non‑negative step‑function (list of floats) that minimises

    F(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

where conv is the linear convolution.  

The algorithm builds a diverse pool of shaped seeds (uniform, edge‑concentrated,
beta‑distributed, Gaussian‑edge, etc.), evaluates them cheaply, keeps the
fewest‑valued seeds and refines them with two gradient phases:

1. Adam on a smooth‑max (log‑sum‑exp) surrogate – the vector is kept in the
   simplex via a soft‑max of logits.
2. Exact ℓ∞ sub‑gradient refinement using an exponentiated‑gradient update.

For the largest length the iteration budgets are increased and a few
jittered restarts around the current best are performed.  The routine respects
the 1000 s wall‑clock budget and returns the best sequence found as a plain
Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a * a (real FFT, returns length 2·len(a)-1)."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns an array of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)].
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Objective helpers (exact, no safety checks – inputs are already safe)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact evaluation of 2·n·max(conv)/sum(a)^2 for a (assumed non‑negative)."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


def _normalize(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (non‑negative, sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        # fallback to a uniform Dirichlet draw
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max surrogate.
    Returns (best_a, best_val) where best_val = 2·n·max(conv) for the best
    iterate discovered before the deadline.
    """
    eps = 1e-16
    # initialise logits of a soft‑max (guarantees a in simplex)
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.07

    lam = 1.0                     # temperature for smooth‑max
    lam_factor = 1.03

    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ----- soft‑max to stay on the simplex -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- convolution and smooth‑max weights -----
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # ----- gradient of smooth‑max w.r.t. a -----
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # ----- back‑prop through soft‑max -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, 1e8)

        # ----- exact objective (sum(a)=1) -----
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # ----- occasional jitter if stagnating -----
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Exact ℓ∞ sub‑gradient refinement (exponentiated‑gradient).
    Returns (best_a, best_val) after at most max_iters iterations or
    when the deadline is reached.
    """
    a = _normalize(a0)
    best_val = float("inf")
    best_a = a.copy()
    eta = 0.45
    last_improve = time.time()

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        # active constraints (allow a tiny tolerance)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient step (mirrored simplex)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                      # safeguard against collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # exact objective
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # slowly decay the step size
        eta *= 0.9995

        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Candidate generation
# ----------------------------------------------------------------------


def _candidate_pool(n: int, best_seq: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a heterogeneous set of seed vectors for a given length n.
    The pool mixes analytic shapes, up‑sampled previous bests and random
    Dirichlet draws.
    """
    candidates = []

    # 1) pure uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) warm start from previous solution of the same size
    if best_seq is not None and best_seq.shape[0] == n:
        candidates.append(best_seq.astype(np.float64))

    # 3) up‑sample / down‑sample previous global best (if any)
    if best_seq is not None and best_seq.shape[0] != n:
        old = best_seq.astype(np.float64)
        x_old = np.linspace(0.0, 1.0, old.shape[0])
        x_new = np.linspace(0.0, 1.0, n)
        up = np.interp(x_new, x_old, old)
        candidates.append(_normalize(up))

    # 4) historic global variable if present
    hseq = globals().get("height_sequence_1")
    if hseq is not None:
        h = np.asarray(hseq, dtype=np.float64)
        if h.shape[0] > n:
            h = h[:n]
        elif h.shape[0] < n:
            h = np.pad(h, (0, n - h.shape[0]))
        candidates.append(_normalize(h))

    # 5) power‑law edge‑concentrated shapes  (distance to edge)^{alpha}
    dist = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1)).astype(np.float64)
    for alpha in [-0.3, -0.5, -0.8, -1.0, -1.5, -2.0]:
        shape = np.power(dist, alpha)
        candidates.append(_normalize(shape))

    # 6) sums of two power‑laws (left + right) for more peaked edges
    idx = np.arange(n, dtype=np.float64)
    for alpha in [1.5, 2.0, 2.5, 3.0]:
        shape = (idx + 1) ** (-alpha) + (n - idx) ** (-alpha)
        candidates.append(_normalize(shape))

    # 7) beta‑distribution shapes (U‑shaped for a,b<1)
    x = (np.arange(n) + 0.5) / n
    for a_par, b_par in [(0.2, 0.2), (0.3, 0.3), (0.4, 0.4), (0.5, 0.5)]:
        shape = np.power(x, a_par - 1) * np.power(1 - x, b_par - 1)
        candidates.append(_normalize(shape))

    # 8) double‑Gaussian edges
    for gamma in [0.1, 0.3, 0.6, 1.0]:
        shape = np.exp(-gamma * (idx ** 2)) + np.exp(-gamma * ((n - 1 - idx) ** 2))
        candidates.append(_normalize(shape))

    # 9) sine‑squared bump (smooth interior)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    candidates.append(_normalize(shape))

    # 10) mixture of uniform and edge‑Gaussian
    edge = np.exp(-0.5 * (idx ** 2)) + np.exp(-0.5 * ((n - 1 - idx) ** 2))
    edge = edge / edge.sum()
    mix = 0.5 * np.full(n, 1.0 / n) + 0.5 * edge
    candidates.append(_normalize(mix))

    # 11) a few random Dirichlet draws with various concentration
    for alpha in [0.3, 0.6, 1.0, 2.0]:
        candidates.append(np.random.dirichlet(np.full(n, alpha)))
        candidates.append(np.random.dirichlet(np.full(n, alpha * 0.5)))

    # Limit the pool size for very large n (GPU / time)
    max_pool = 30 if n <= 16384 else 18
    return candidates[:max_pool]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the wall‑clock budget (default 1000 s)
    and returns the best sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    best_seq = None
    best_val = float("inf")

    # Progressively larger lengths – larger n gives a smaller theoretical bound
    lengths = [4096, 8192, 16384, 32768, 65536]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

        # --------------------------------------------------------------
        # 1) Build a heterogeneous pool of seeds
        # --------------------------------------------------------------
        pool = _candidate_pool(n, best_seq)

        # --------------------------------------------------------------
        # 2) Quick evaluation of the raw seeds, keep the top few
        # --------------------------------------------------------------
        scored = []
        for v in pool:
            val = _objective(v)
            scored.append((val, v))
        scored.sort(key=lambda t: t[0])
        top_k = 5  # number of seeds that will receive the heavy optimisation
        top_seeds = [vec for _, vec in scored[:top_k]]

        # --------------------------------------------------------------
        # 3) Refine each top seed
        # --------------------------------------------------------------
        for idx, seed_vec in enumerate(top_seeds):
            if time.time() > deadline:
                break
            try:
                # Adam phase – iteration budget grows with n
                max_adam_iters = 2000 if n <= 16384 else 4000
                a_adam, val_adam = _adam_optimize(
                    seed_vec,
                    n,
                    conv_fft_len,
                    corr_fft_len,
                    deadline,
                    max_iters=max_adam_iters,
                )
                # Sub‑gradient refinement – a few thousand steps
                a_ref, val_ref = _subgradient_refine(
                    a_adam,
                    n,
                    conv_fft_len,
                    corr_fft_len,
                    deadline,
                    max_iters=3500,
                )
                # Record best
                if val_ref < best_val:
                    best_val = val_ref
                    best_seq = a_ref.copy()
                    print(f"[n={n} seed={idx}] NEW GLOBAL BEST {best_val:.6f}")
                elif val_adam < best_val:
                    best_val = val_adam
                    best_seq = a_adam.copy()
                    print(f"[n={n} seed={idx}] new best (Adam) {best_val:.6f}")
            except Exception as exc:                     # defensive – should never happen
                print(f"[n={n} seed={idx}] exception: {exc}")

        # --------------------------------------------------------------
        # 4) Random jitter around the current best of this length
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.shape[0] == n:
            jitter_rounds = 6
            for j in range(jitter_rounds):
                if time.time() > deadline:
                    break
                logits = np.log(np.maximum(best_seq, 1e-12))
                logits += np.random.normal(scale=0.025, size=n)
                a0 = np.exp(logits - logits.max())
                a0 /= a0.sum()

                a_adam, val_adam = _adam_optimize(
                    a0,
                    n,
                    conv_fft_len,
                    corr_fft_len,
                    deadline,
                    max_iters=2500,
                )
                a_ref, val_ref = _subgradient_refine(
                    a_adam,
                    n,
                    conv_fft_len,
                    corr_fft_len,
                    deadline,
                    max_iters=3000,
                )
                if val_ref < best_val:
                    best_val = val_ref
                    best_seq = a_ref.copy()
                    print(f"[n={n} jitter={j}] NEW GLOBAL BEST {best_val:.6f}")

    # ------------------------------------------------------------------
    # Safety net – if nothing was found (should not happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 4096
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return as a plain Python list for the external evaluator
    return best_seq.tolist()
