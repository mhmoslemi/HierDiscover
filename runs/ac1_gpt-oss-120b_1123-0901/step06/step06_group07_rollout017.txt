"""
Improved search for a non‑negative step‑function that minimises

        F(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of three stages:

1. **Rich parametric seeding** – a large and diverse set of analytic
   families (exponential, power‑law, Gaussian, sinusoidal, random Dirichlet
   and mixtures thereof) is generated.  All members are normalised to
   unit ‑sum and quickly evaluated with an FFT‑based convolution.  The
   best K are kept.

2. **Smooth‑max Adam optimisation** – for each retained seed an Adam
   optimiser minimises a *log‑sum‑exp* surrogate of the max‑convolution.
   The surrogate temperature (``lam``) is increased very slowly so the
   optimiser follows a smooth path towards the true maximum.

3. **Exact sub‑gradient polishing** – the vector from (2) is refined by
   exponentiated‑gradient steps on the genuine max‑convolution (active‑set
   weighting).  A few tiny random “shakes’’ are added if the optimiser
   stalls.

If time permits the best 4096‑vector is linearly interpolated to a finer
grid (n = 8192) and the whole pipeline is repeated, usually yielding a
strictly smaller objective.  The routine respects the 1000 s budget,
periodically prints progress and finally returns the best vector found
as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns only the first 2·len(a)‑1 entries.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Very fast evaluation of the objective.
    ``a`` is assumed to be already normalised (sum≈1).
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

                     best_seq: np.ndarray | None,
                     best_val: float,
                     target: float = 1.503) -> tuple[np.ndarray | None, float]:
    """
    Adam on a log‑sum‑exp surrogate of max(conv(a,a)).
    Returns the best vector/value seen so far.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # start in log‑space (soft‑max)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04                                      # a tad smaller than before
    lam = 0.5                                      # initial temperature
    lam_factor = 1.008
    lam_cap = 2e6

    last_improve = time.time()
    it = 0
    max_iter = 500_000

    while time.time() < deadline and it < max_iter:
        # ---- simplex via soft‑max ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- surrogate objective ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")
            if best_val <= target:
                # stop early – we already beat the target
                break

        # ---- smooth‑max weights (log‑sum‑exp) ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # ---- gradient of the surrogate ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # chain rule through soft‑max (mirrored to logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, lam_cap)

        # ---- occasional random shake ----
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 2e-3   # tiny perturbation
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(a0: np.ndarray,
                            n: int,
                            conv_fft_len: int,
                            corr_fft_len: int,
                            deadline: float,
                            best_seq: np.ndarray | None,
                            best_val: float,
                            target: float = 1.503) -> tuple[np.ndarray | None, float]:
    """
    Exponentiated‑gradient steps on the *true* max‑convolution.
    """
    # start from the best point found by Adam (or a0 if none)
    a = best_seq if best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()                # stay on the simplex

    eta = 0.55
    it = 0
    last_improve = time.time()
    max_iter = 120_000

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it:06d}] new best {val:.6f}")
            if best_val <= target:
                break

        # active set where the max is attained
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                      # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # early‑stop if we are stuck for a while
        if time.time() - last_improve > 7.0:
            break

        it += 1

    return best_seq, best_val


def _deep_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   global_best_seq: np.ndarray | None,
                   global_best_val: float,
                   target: float = 1.503) -> tuple[np.ndarray | None, float]:
    """
    Run Adam (smooth‑max) followed by exact sub‑gradient polishing.
    """
    best_seq, best_val = _adam_smooth_max(
        a0, n, conv_fft_len, corr_fft_len,
        deadline, global_best_seq, global_best_val,
        target=target,
    )
    best_seq, best_val = _subgradient_refinement(
        a0, n, conv_fft_len, corr_fft_len,
        deadline, best_seq, best_val,
        target=target,
    )
    return best_seq, best_val

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """
    Produce a diverse portfolio of analytic shapes.
    All vectors are normalised to unit sum.
    """
    candidates = []

    # --------------------------------------------------------------
    # Warm‑start + simple rotations
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 6, n // 3, n // 2, 2 * n // 3):
            candidates.append(np.roll(warm_start, shift))

    # --------------------------------------------------------------
    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # --------------------------------------------------------------
    # Random Dirichlet draws with many concentration values
    for conc in (0.01, 0.05, 0.1, 0.3, 0.6, 1.0, 2.0, 5.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    # --------------------------------------------------------------
    # Edge‑exponential families (left, right and symmetric)
    i = np.arange(n, dtype=np.float64)
    for alpha in np.logspace(-2, 1.2, num=30):          # 0.01 … ~16
        left = np.exp(-alpha * i)
        right = np.exp(-alpha * (n - 1 - i))
        # 1) left only
        shape = left
        shape = shape / shape.sum()
        candidates.append(shape)
        # 2) right only
        shape = right
        shape = shape / shape.sum()
        candidates.append(shape)
        # 3) symmetric sum (previous edge‑exponential)
        shape = left + right
        shape = shape / shape.sum()
        # mixture with uniform
        for uni_w in (0.0, 0.25, 0.5, 0.75, 1.0):
            mixed = uni_w * (np.full(n, 1.0 / n)) + (1.0 - uni_w) * shape
            mixed = mixed / mixed.sum()
            candidates.append(mixed)

    # --------------------------------------------------------------
    # Edge‑power‑law families (symmetric) + uniform mixture
    for p in np.linspace(0.1, 6.0, num=30):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge = edge / edge.sum()
        for uni_w in (0.0, 0.4, 0.8):
            mixed = uni_w * (np.full(n, 1.0 / n)) + (1.0 - uni_w) * edge
            mixed = mixed / mixed.sum()
            candidates.append(mixed)

    # --------------------------------------------------------------
    # Two‑Gaussian blobs (different widths)
    for sigma in (0.02, 0.05, 0.1, 0.2, 0.4, 0.8):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((n - 1 - i) / (sigma * n)) ** 2)
        shape = shape / shape.sum()
        candidates.append(shape)

    # --------------------------------------------------------------
    # Sinusoidal half‑wave + uniform mixture
    sin = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin = np.maximum(sin, 0.0)
    sin = sin / sin.sum()
    for uni_w in (0.0, 0.3, 0.6):
        mixed = uni_w * (np.full(n, 1.0 / n)) + (1.0 - uni_w) * sin
        mixed = mixed / mixed.sum()
        candidates.append(mixed)

    # --------------------------------------------------------------
    # Random convex combination of three basis shapes (uniform, edge‑exp,
    # power‑law).  This generates a huge variety with only a few random
    # parameters.
    for _ in range(80):
        alpha = 10 ** np.random.uniform(-2, 1)          # exp‑decay
        p = np.random.uniform(0.2, 5.0)                 # power‑law
        # basis vectors
        left = np.exp(-alpha * i)
        right = np.exp(-alpha * (n - 1 - i))
        edge_exp = left + right
        edge_exp = edge_exp / edge_exp.sum()
        power = (i + 1.0) ** (-p) + (n - i) ** (-p)
        power = power / power.sum()
        uni = np.full(n, 1.0 / n)

        w = np.random.dirichlet(np.ones(3))
        shape = w[0] * uni + w[1] * edge_exp + w[2] * power
        shape = shape / shape.sum()
        candidates.append(shape)

    return candidates

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a low‑evaluation‑score step function.
    Returns the best sequence found (plain Python list).
    """
    np.random.seed(seed)
    # give a small safety margin for the host environment
    deadline = time.time() + budget_s - 0.7

    best_seq: np.ndarray | None = None
    best_val = float("inf")
    TARGET = 1.503                     # stop as soon as we beat this

    # --------------------------------------------------------------
    # 1. work initially with n = 4096 (rapid exploration)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # possible warm‑start from a previous run
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None

    # --------------------------------------------------------------
    # generate and pre‑evaluate a large pool of candidates
    all_candidates = _generate_parametric_candidates(n, warm_start)

    cand_vals = []
    for idx, cand in enumerate(all_candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))
        # occasional progress report (not too noisy)
        if idx % 100 == 0:
            print(f"[candidate {idx:04d}/{len(all_candidates)}] fast val = {val:.6f}")

    # keep the most promising starts
    cand_vals.sort(key=lambda x: x[0])
    top_k = 12
    top_starts = [c for _, c in cand_vals[:top_k]]

    # --------------------------------------------------------------
    # 2. Deep optimisation on the selected starts
    for start_id, a0 in enumerate(top_starts):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # Give each start at most (remaining / remaining_starts) seconds
        # but never less than a few seconds.
        print(f"--- Optimisation start #{start_id + 1}/{len(top_starts)} (n={n}) ---")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
            target=TARGET,
        )
        if best_val <= TARGET:
            print("Target reached – exiting early.")
            break

    # --------------------------------------------------------------
    # 3. If time permits, upscale to n = 8192 and repeat the pipeline
    time_left = deadline - time.time()
    if time_left > 45.0:                     # need enough budget for a full pass
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation onto the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big = a_big / a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        # a quick fresh set of candidates on the larger grid (re‑using the same
        # generator but with the larger n)
        print("--- Upscaled optimisation (n=8192) ---")
        top_candidates_big = _generate_parametric_candidates(n_big, None)
        # quick filter – keep only the 8 best of the huge pool
        quick_vals = []
        for cand in top_candidates_big[:2000]:        # limit to first 2000 for speed
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand = cand / cand.sum()
            val = _objective(cand, n_big, conv_fft_len_big)
            quick_vals.append((val, cand))
        quick_vals.sort(key=lambda x: x[0])
        top_big = [c for _, c in quick_vals[:8]]

        # run full optimisation on the up‑scaled starts
        for start_id, a0 in enumerate(top_big):
            if time.time() >= deadline:
                break
            print(f"--- Upscaled start #{start_id + 1}/{len(top_big)} (n={n_big}) ---")
            best_seq, best_val = _deep_optimize(
                a0,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                deadline,
                best_seq,
                best_val,
                target=TARGET,
            )
            if best_val <= TARGET:
                print("Target reached on up‑scaled grid – exiting.")
                break
        # keep the larger vector if it truly improved the bound
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big

    # --------------------------------------------------------------
    # fallback – if nothing succeeded return a uniform vector
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # final clipping to the evaluator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search finished – best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
