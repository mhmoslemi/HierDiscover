"""
Hybrid search for low‑autocorrelation step functions.

The objective after normalising ∑aᵢ = √(2 n) is simply max(conv(a,a)),
so we minimise the largest entry of the self‑convolution.
The optimiser works in five stages:

1.  Wide random exploration of many lengths (up to 20 000) and
    several parametric families (exponential, power‑law, Hann, Hamming,
    Gaussian, Beta, Dirichlet).  The best candidate is kept.

2.  Refine the best candidate with an Adam optimiser on a smooth
    surrogate of max (log‑sum‑exp) using a soft‑max parametrisation.
    This respects the simplex constraint automatically.

3.  Exact projected sub‑gradient descent on the true max‑convolution.
    Uses the gradient at the current worst index and adapts the step
    size.

4.  Simulated‑annealing hill‑climbing.  Tiny pairwise mass transfers
    are combined with occasional global Gaussian perturbations;
    moves are accepted with a Metropolis criterion.

5.  A final short Adam run to polish the result.

All stages respect the global time budget (default 1000 s) and the
global best sequence is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of two real vectors."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)            # clip numerical negatives
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(full_convolution_fft(a, a).max())


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]          # sort descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1        # zero‑based index of last positive entry
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Generation of parametric initial candidates
# ----------------------------------------------------------------------
def generate_candidate(n: int, shape: str) -> np.ndarray:
    """Return a non‑negative vector of length n for the given shape."""
    idx = np.arange(n, dtype=float)

    if shape == "exp":
        tau = np.random.uniform(0.1, 2.5) * n
        a = np.exp(-idx / tau)
    elif shape == "pow":
        p = np.random.uniform(0.2, 1.7)
        a = (idx + 1) ** (-p)
    elif shape == "hann":
        a = 0.5 - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1))
    elif shape == "hamming":
        a = 0.54 - 0.46 * np.cos(2.0 * np.pi * idx / (n - 1))
    elif shape == "gaussian":
        mu = np.random.uniform(0, n - 1)
        sigma = np.random.uniform(0.05, 0.2) * n
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
    elif shape == "beta":
        alpha = np.random.uniform(0.3, 3.0)
        beta = np.random.uniform(0.3, 3.0)
        t = np.linspace(0, 1, n)
        a = (t ** (alpha - 1)) * ((1 - t) ** (beta - 1))
    else:  # dirichlet / uniform on simplex
        a = np.random.dirichlet(np.ones(n))

    # occasionally add a small Gaussian perturbation
    if np.random.rand() < 0.25:
        noise = np.random.randn(n) * np.max(a) * 0.02
        a = np.maximum(a + noise, 0.0)

    return a


# ----------------------------------------------------------------------
# Adam optimiser on the log‑sum‑exp (soft‑max) surrogate of max(conv)
# ----------------------------------------------------------------------
def adam_softmax_optimize(a0: np.ndarray,
                          target_sum: float,
                          deadline: float,
                          init_tau_factor: float = 0.12,
                          min_tau_factor: float = 5e-5,
                          alpha0: float = 0.025,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Adam on the smooth max‑convolution surrogate."""
    n = a0.shape[0]
    # initialise logits for soft‑max parametrisation
    x = np.log(a0 + 1e-12)
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = a0.copy()
    best_val = max_convolution(a0)
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # map logits to the simplex
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)           # stable soft‑max
        p = exp_shift / exp_shift.sum()
        a = target_sum * p

        # exact max for book‑keeping
        conv = full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam new best max={cur_max:.6f}")

        # smoothed max (log‑sum‑exp) weights
        tau = max(init_tau_factor * cur_max * (0.99 ** it),
                  min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient wrt a: 2·a ⊛ w_rev
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # back‑prop through soft‑max
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to shake out of plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slow decay of the learning rate
        alpha *= 0.9995

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def subgradient_descent(a0: np.ndarray,
                        target_sum: float,
                        deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent following the current worst index."""
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = max_convolution(a)

    step = 0.2 * target_sum / n          # start step size
    it = 0

    while time.time() < deadline:
        it += 1
        conv = full_convolution_fft(a, a)
        k = int(conv.argmax())            # index of current worst peak

        # gradient of conv_k w.r.t. a
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # projected step
        cand = a - step * grad
        cand = project_onto_simplex(cand, target_sum)

        cand_val = max_convolution(cand)
        if cand_val < best_val - 1e-12:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= 1.02
            if it % 200 == 0:
                print(f"[{time.time():.2f}] subgrad it={it}  val={best_val:.6f}")
        else:
            step *= 0.70
            if np.random.rand() < 0.07:   # tiny jitter on failure
                jitter = np.random.rand(n) * 1e-4
                a = project_onto_simplex(a + jitter, target_sum)

        if best_val < 1.503:            # early stop if target is reached
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb on the exact objective
# ----------------------------------------------------------------------
def hill_climb(a0: np.ndarray,
               target_sum: float,
               deadline: float) -> tuple[np.ndarray, float]:
    """Move mass between indices with Metropolis acceptance."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = max_convolution(a)

    # temperature schedule (linear cooling)
    T0 = max(1e-3, best_val * 0.4)
    T_end = 1e-5
    start = time.time()
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        # decide move type
        if np.random.rand() < 0.7:
            # pairwise transfer
            i, j = np.random.randint(0, n, size=2)
            if i == j or a[i] <= 0:
                continue
            delta = a[i] * np.random.rand() * 0.09   # ≤9 % of source mass
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
        else:
            # global Gaussian perturbation
            perturb = np.random.randn(n)
            perturb -= perturb.mean()
            scale = target_sum * 0.025
            a_new = a + perturb * scale
            a_new = np.maximum(a_new, 0.0)
            a_new = project_onto_simplex(a_new, target_sum)

        new_val = max_convolution(a_new)

        if new_val < best_val:
            a = a_new
            best_val = new_val
            best_a = a_new.copy()
            print(f"[{time.time():.2f}] hillclimb it={it}  new best={best_val:.6f}")
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                a = a_new

        # periodic re‑normalisation (helps with accumulated drift)
        if it % 2500 == 0:
            a = project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val

# Main entry point required by the evaluator

    """

    max‑self‑convolution.  Returns the best sequence found as a list
    of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.10   # safety margin

    best_seq = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # Helper to record a candidate (updates global best)
    # ------------------------------------------------------------------
    def record(seq: np.ndarray):
        nonlocal best_seq, best_val
        val = max_convolution(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] GLOBAL new best={best_val:.6f}  n={len(seq)}")
        return val

    # ------------------------------------------------------------------
    # Warm‑start from a previously stored construction, if any
    # ------------------------------------------------------------------
    try:
        hs = globals().get("height_sequence_1")
        if hs is not None:
            hs_arr = np.asarray(hs, dtype=float)
            n0 = hs_arr.shape[0]
            target0 = np.sqrt(2.0 * n0)
            hs_norm = hs_arr * (target0 / hs_arr.sum())
            record(hs_norm)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – diverse random initialisations (up to ~20 k length)
    # ------------------------------------------------------------------
    while time.time() < deadline - 250.0:   # reserve ~250 s for later phases
        remaining = deadline - time.time()
        # adapt length range with remaining time

            n = np.random.randint(5000, 18000 + 1)
        elif remaining > 300:
            n = np.random.randint(4000, 15000 + 1)
        else:
            n = np.random.randint(3000, 12000 + 1)

        target_sum = np.sqrt(2.0 * n)

        shape = np.random.choice(
            ["exp", "pow", "hann", "hamming", "gaussian", "beta", "dirichlet"]
        )
        a0 = generate_candidate(n, shape)
        a0 = a0 * (target_sum / a0.sum())
        record(a0)

    # If nothing was discovered (unlikely), fall back to uniform
    if best_seq is None:

    # ------------------------------------------------------------------
