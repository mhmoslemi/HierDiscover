def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    # Determine best length
    target_n = 32768
    if not evaluation possible else try larger if time.

    best_seq = None
    best_val = float('inf')

    # function to evaluate and track best
    def try_seq(arr):
        nonlocal best_seq, best_val
        # arr is numpy ndarray, but evaluate_sequence expects list
        val = evaluate_sequence(arr.tolist())
        if val < best_val:
            best_val = val
            best_seq = arr.copy()
            print(f"New best {val:.6f}")

    # define helper functions for conv and gradient:
    use same _full_convolution & _cross_correlation.

    # helper to run optimization from initial a0
    def optimize_from(a0, n):
        # Determine FFT lengths
        conv_len = 1 << (2*n - 1).bit_length()
        corr_len = 1 << ((3*n - 2).bit_length())
        # Phase 1: Adam
        a_adam, val_adam = adam_optimize(a0, n, conv_len, corr_len, deadline)
        try_seq(a_adam)
        # Phase 2: subgradient refine
        a_ref, val_ref = subgradient_refine(a_adam, n, conv_len, corr_len, deadline)
        try_seq(a_ref)
        return a_ref

    # multi-scale strategy
    # Stage 1: small lengths list
    small_lengths = [2048, 4096, 8192, 16384]

    # We'll keep best shape at each step
    cur_seq = None
    for n in small_lengths:
        # generate seeds based on cur_seq (if any) scaled
        seeds = []
        # uniform
        seeds.append(np.full(n, 1.0/n))
        # if cur_seq exists, upsample or downsample
        if cur_seq is not None:
            # rescale to n via interpolation
            old = cur_seq
            x_old = np.linspace(0, 1, old.shape[0])
            x_new = np.linspace(0, 1, n)
            tmp = np.interp(x_new, x_old, old)
            tmp = np.clip(tmp, 0, None)
            if tmp.sum() > 0:
                tmp = tmp / tmp.sum()
            else:
                tmp = np.random.dirichlet(np.ones(n))
            seeds.append(tmp)

        # power law exponents
        idx = np.arange(1, n+1)
        prod = idx*(n-idx+1)
        for alpha in np.arange(-0.5, -5.5, -0.5):
            shape = np.power(prod, alpha, where=prod>0)
            if shape.sum() > 0:
                shape = shape / shape.sum()
                seeds.append(shape)

        # random Dirichlet few
        for _ in range(2):
            seeds.append(np.random.dirichlet(np.ones(n)))
        # limit number
        seeds = seeds[:20]

        # run optimization for each seed
        for s_id, seed_arr in enumerate(seeds):
            if time.time() > deadline:
                break
            cur_seq = optimize_from(seed_arr, n)
            # early exit if best value already below target 1.503 and n large enough
            if best_val < 1.503:
                break

        # possibly jitter best for this size
        if cur_seq is not None and time.time() < deadline:
            for jitter in range(3):
                if time.time() > deadline:
                    break
                logits = np.log(np.maximum(cur_seq, 1e-12))
                logits += np.random.normal(scale=0.03, size=n)
                a0 = np.exp(logits - np.max(logits))
                a0 = a0 / a0.sum()
                cur_seq = optimize_from(a0, n)

    # Stage 2: large length (target_n)
    n = target_n
    # generate seeds: upsample best from previous (maybe from 16384), plus power law etc.
    seeds_big = []
    if cur_seq is not None:

        x_old = np.linspace(0, 1, old.shape[0])
        x_new = np.linspace(0, 1, n)
        up = np.interp(x_new, x_old, old)

            up = up / up.sum()
        else:
            up = np.random.dirichlet(np.ones(n))
        seeds_big.append(up)
    else:
        seeds_big.append(np.full(n, 1.0/n))

    # include height_sequence_1 if exists
    if 'height_sequence_1' in globals():
        hs = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        if hs.shape[0] != n:
            # interpolate
            x_old = np.linspace(0,1, hs.shape[0])
            x_new = np.linspace(0,1, n)
            hs = np.interp(x_new, x_old, hs)
        hs = np.clip(hs, 0, None)
        if hs.sum()>0:
            hs = hs/hs.sum()
        else:
            hs = np.random.dirichlet(np.ones(n))
        seeds_big.append(hs)

"""
Propose a step‑function (non‑negative vector) that minimises

        F(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

The objective is homogeneous, therefore we keep the vector on the simplex
(sum = 1).  The search consists of three stages:

1.  Small‑scale warm‑start (n = 2 048 … 16 384).  A few diverse seeds are
    optimised with a modest number of iterations – this quickly discovers a
    good shape.

2.  Large‑scale refinement (n = 32 768).  The best shape from stage‑1 is
    interpolated to the target length and optimised aggressively:
        • Adam on a smooth‑max surrogate of the ℓ∞‑norm.
        • Exact sub‑gradient (exponentiated‑gradient) refinement.
    The number of iterations is adapted to the remaining wall‑clock time.

3.  A handful of jittered restarts around the current best solution to
    escape shallow local minima.

All optimisation respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real‑valued)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a, length 2*len(a)-1, using a power‑of‑two FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) where

        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)] , i=0…n‑1 .
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Objective helper (exact value, no extra overhead)
# ----------------------------------------------------------------------


def _objective_val(a: np.ndarray, n: int) -> float:
    """Exact value 2·n·max(conv(a,a)) for a already normalised (sum=1)."""
    conv = _full_convolution(a, fft_len=1 << int(np.ceil(np.log2(2 * n - 1))))
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Adam optimiser on a smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate:

        smooth_max_λ(x) = (1/λ)·log Σ_i exp(λ·x_i)

    The temperature λ starts small (heavy smoothing) and grows
    geometrically.  The vector is represented in the log‑softmax
    (logits) space, guaranteeing non‑negativity and unit sum.
    """
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.07               # a slightly larger learning rate than the baseline

    lam = 0.2               # initial smooth‑max temperature
    lam_factor = 1.04

    best_a = a.copy()
    best_val = _objective_val(a, n)
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ---- simplex projection via soft‑max ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- convolution and smooth‑max weights ----
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        # soft‑max over the convolution values
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum if w_sum > 0 else np.full_like(w, 1.0 / w.size)

        # ---- gradient of smooth‑max w.r.t. a ----
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # ---- back‑propagation through soft‑max (logits) ----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature annealing ----
        lam = min(lam * lam_factor, 1e8)

        # ---- track best exact value ----
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # ---- occasional stochastic shake‑up if stagnating ----
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact ℓ∞ sub‑gradient refinement (exponentiated‑gradient)
# ----------------------------------------------------------------------


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent on the simplex) using the
    true ℓ∞‑norm sub‑gradient.  The update is

        a ← a ⊙ exp(−η·g)

    where g is the average of the correlation with the active convolution
    indices.
    """
    a = a0 / a0.sum()
    best_a = a.copy()
    best_val = _objective_val(a, n)

    eta = 0.45
    last_improve = time.time()

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient step on the simplex
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # fallback to a random point if we collapsed
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        eta *= 0.9995                     # slowly decay step size

        if time.time() - last_improve > 6.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# High‑level optimiser for a single seed
# ----------------------------------------------------------------------


def _optimise_seed(seed_vec: np.ndarray, n: int, deadline: float) -> np.ndarray:
    """
    Run the two‑phase optimiser (Adam → sub‑gradient) on a single seed.
    Returns the best vector discovered for this seed.
    """
    conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

    # allocate iteration budgets proportionally to remaining time
    time_left = deadline - time.time()
    # a very rough estimate: one Adam iteration ≈ 0.0008 s for n≈3·10⁴
    # we conservatively reserve half the remaining time for Adam,
    # the rest for sub‑gradient.
    adam_iters = max(1000, int(min(20000, (time_left * 0.5) / 0.001)))
    subgrad_iters = max(500, int(min(8000, (time_left * 0.5) / 0.001)))

    a_adam, val_adam = _adam_optimize(
        seed_vec, n, conv_fft_len, corr_fft_len, deadline, max_iters=adam_iters
    )
    a_ref, val_ref = _subgradient_refine(
        a_adam, n, conv_fft_len, corr_fft_len, deadline, max_iters=subgrad_iters
    )
    return a_ref if val_ref <= val_adam else a_adam

        val = _objective_val(candidate, candidate.shape[0])
        if val < best_val:
            best_val = val
            best_seq = candidate.copy()
            print(f">>> New global best (n={candidate.shape[0]}) : {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – warm‑up on modest lengths
    # ------------------------------------------------------------------
    warm_lengths = [2048, 4096, 8192, 16384]

        if time.time() > deadline:
            break

        print(f"\n--- Warm‑up stage, length n={n} ---")
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

        # generate a small, diverse pool of seeds
        seeds = []

        # (1) uniform

            interp = np.clip(interp, 0.0, None)

            else:

        # (3) power‑law (edge‑heavy) family

                seeds.append(shape)

        # (4) a few random Dirichlet draws
        for _ in range(2):

            _maybe_update(refined)

        # keep the best shape of this length for the next stage
