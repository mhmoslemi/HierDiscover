```python
def project_onto_simplex(v):
    # v can be any real vector. Return projection onto {x>=0, sum(x)=1}
    # Based on algorithm from Duchi et al. 2008
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho: max i s.t. u_i - cssv_i / i > 0
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Improved step‑function search.

The algorithm works in three stages:

1. **Coarse search** – generate a small pool of random Dirichlet / Beta‑shaped
   candidates for a modest length (1024).  Each candidate is refined with a
   projected‑gradient descent on a smooth‑max surrogate of the objective.
   The best candidate is kept.

2. **Progressive up‑sampling** – the current best sequence is repeatedly
   up‑sampled (linear interpolation) to larger lengths 2048 → 4096 → 8192 →
   16384.  After each up‑sampling a few thousand PGD iterations and an
   exponentiated‑gradient sub‑gradient phase are run.  The best result over
   all lengths is stored.

3. **Final polishing** – on the globally best sequence we perform a long
   PGD run followed by a cheap stochastic hill‑climb that moves a tiny mass
   between two positions if it improves the exact objective.

All updates keep the vector on the probability simplex (non‑negative,
summing to one) – this eliminates the denominator in the original
evaluation function.  The code respects the given time budget (default
1000 s) and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Returns an array of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]
    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative vector to ``new_n`` points by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _run_pgd_smooth_max(a0: np.ndarray, n: int, deadline: float,
                       max_iters: int):
    """
    Projected‑gradient descent on the smooth‑max surrogate of the objective.
    Returns the best sequence seen and its exact objective value (sum(a)=1).
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    last_improve = time.time()

    lam = 0.30               # start temperature for smooth‑max
    lam_factor = 1.02
    init_lr = 0.12           # initial learning rate

    for t in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[PGD n={n} iter={t}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size schedule
        lr = init_lr / np.sqrt(t)

        # gradient step and projection onto simplex
        a = a - lr * grad
        a = _project_onto_simplex(a)

        # temperature schedule (smooth‑max becomes sharper)
        lam = min(lam * lam_factor, 1e4)

        # occasional jitter if we stagnate
        if time.time() - last_improve > 8.0:
            a += (np.random.rand(n) - 0.5) * 1e-3
            a = _project_onto_simplex(a)
            last_improve = time.time()

        if time.time() > deadline:
            break
    return best_a, best_val


def _run_subgrad(a0: np.ndarray, n: int, deadline: float,
                 max_iters: int):
    """
    Exponentiated‑gradient sub‑gradient optimisation on the *exact* max‑conv.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    last_improve = time.time()
    eta0 = 0.55

    for it in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} iter={it}] new best {val:.6f}")

        # active set of indices attaining the maximum
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # early stop if stagnated
        if time.time() - last_improve > 6.0:
            break
        if time.time() > deadline:
            break
    return best_a, best_val


def _local_hill_climb(a0: np.ndarray, n: int, deadline: float,
                     max_trials: int = 500_000):
    """
    Very cheap stochastic local search: move a tiny mass ε from one index
    to another, keep the move if it lowers the exact objective.
    """
    conv_fft_len = _next_pow2(2 * n - 1)

    best_a = a0.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    no_improve = 0
    trials = 0
    while time.time() < deadline and trials < max_trials and no_improve < 8_000:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0.0, min(0.07, best_a[i]))
        if eps == 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = cand
            no_improve = 0
            print(f"[HillClimb n={n} trial={trials}] improved to {val:.6f}")
        else:
            no_improve += 1
        trials += 1
    return best_a, best_val


def _initial_candidates(n: int, count: int = 12):
    """
    Produce a set of random starting points:
      * Dirichlet draws (concentration 1)
      * Dirichlet draws with small concentrations (spiky)
      * Beta‑distribution shaped vectors.
    """
    candidates = []
    # Dirichlet 1
    for _ in range(count // 3):
        candidates.append(np.random.dirichlet(np.ones(n)))
    # Dirichlet with low concentration → spikier
    for _ in range(count // 3):
        conc = np.random.choice([0.2, 0.5, 0.8])
        candidates.append(np.random.dirichlet(np.full(n, conc)))
    # Beta shapes
    for _ in range(count - 2 * (count // 3)):
        p = np.random.uniform(0.2, 5.0)
        q = np.random.uniform(0.2, 5.0)
        x = np.linspace(0.0, 1.0, n)
        shape = np.power(x, p - 1) * np.power(1 - x, q - 1)
        shape = np.maximum(shape, 0.0)
        s = shape.sum()
        if s > 1e-12:
            candidates.append(shape / s)
    # Normalise & discard pathological entries
    clean = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s > 1e-12:
            clean.append(c / s)
    return clean

def propose_candidate(seed: int = 0, budget_s: float = 1000):
    """
    Search for a non‑negative step‑function that minimises
        F(a) = 2·n·max(a*a) / (sum(a)²)
    within the given time budget (seconds).  Returns the best sequence
    found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float('inf')

    # ------------------------------------------------------------------
    # 1️⃣  Coarse optimisation at a modest size
    n0 = 1024
    print(f"\n=== Coarse optimisation (n={n0}) ===")
    candidates = _initial_candidates(n0, count=15)

    for cid, init in enumerate(candidates):
        if time.time() > deadline:
            break

        a_pg, val_pg = _run_pgd_smooth_max(init, n0, deadline,
                                           max_iters=45_000)
        if val_pg < best_val:
            best_val, best_seq = val_pg, a_pg.copy()
            print(f"[Coarse cand={cid}] PGD new global best {val_pg:.6f}")

        a_sub, val_sub = _run_subgrad(a_pg, n0, deadline, max_iters=10_000)
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Coarse cand={cid}] Subgrad new global best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Progressive up‑sampling to larger lengths
    # Target lengths (increase geometrically)
    lengths = [2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Upsample & refine (n={n}) ===")
        # up‑sample current best to new size
        a_up = _upsample(best_seq, n)

        # moderate PGD run
        max_iters = max(20_000 * 2048 // n, 4_000)
        a_pg, val_pg = _run_pgd_smooth_max(a_up, n, deadline, max_iters=max_iters)
        if val_pg < best_val:
            best_val, best_seq = val_pg, a_pg.copy()
            print(f"[Upsample n={n}] PGD improved global best to {val_pg:.6f}")

        # sub‑gradient refinement
        a_sub, val_sub = _run_subgrad(a_pg, n, deadline, max_iters=12_000)
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Upsample n={n}] Subgrad improved global best to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final polishing on the best length found
    if best_seq is not None and time.time() < deadline:
        n = len(best_seq)
        print(f"\n=== Final polishing (n={n}) ===")
        # longer PGD
        max_iters = max(60_000 * 2048 // n, 12_000)
        a_pg, val_pg = _run_pgd_smooth_max(best_seq, n, deadline, max_iters)
        if val_pg < best_val:
            best_val, best_seq = val_pg, a_pg.copy()
            print(f"[Final PGD] improved to {val_pg:.6f}")

        # final hill‑climb
        if time.time() < deadline:
            a_hc, val_hc = _local_hill_climb(best_seq, n, deadline,
                                             max_trials=800_000)
            if val_hc < best_val:
                best_val, best_seq = val_hc, a_hc.copy()
                print(f"[Final HillClimb] improved to {val_hc:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback (should never happen)
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback)

    # Normalise one last time (protect against rounding drift)
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s > 0:
        best_seq = best_seq / s
    else:
        best_seq = np.random.dirichlet(np.ones(len(best_seq)))

    # Return as plain Python list
    print(f"\n=== Finished ===\nBest value observed: {best_val:.6f}\nLength: {len(best_seq)}")
    return best_seq.tolist()
