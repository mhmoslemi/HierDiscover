"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

The search consists of three stages:

1. **Diverse parametric families** – for several lengths we sample
   functions such as Gaussian, exponential, cosine windows, power‑law,
   and random sparse spikes.  Each candidate is normalised to the
   required ℓ₁‑norm √(2 n) and its exact max‑convolution is obtained by
   a single FFT, giving a cheap evaluation of the objective
   (the objective equals the max‑convolution when the sum is normalised).

2. **Best‑candidate refinement** – the globally best sequence seen so far
   is refined with a simulated‑annealing hill‑climber that repeatedly
   transfers a small random fraction of mass between two indices.
   Improvements are always accepted; worsening moves are accepted with a
   temperature‑dependent probability, allowing the search to escape
   shallow plateaux.

3. **Return** – the best sequence found within the time budget is
   returned as a plain Python list.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities ------------------------------------------------------


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """
    Exact maximum of the linear convolution a * a using one real FFT.
    """
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _generate_shape(kind: str, n: int, rng: np.random.Generator,
                    target_sum: float) -> np.ndarray:
    """
    Produce a non‑negative vector of length n belonging to the specified
    parametric family.  The result is scaled so that its ℓ₁‑norm equals
    ``target_sum`` (except for the 'sparse' family, which is already
    scaled).
    """
    if kind == "uniform":
        a = np.full(n, 1.0)

    elif kind == "triangular":
        # classic triangular “tent” (zero at the ends)
        a = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)

    elif kind == "hann":
        a = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))

    elif kind == "blackman":
        a0, a1, a2 = 0.42, 0.5, 0.08
        i = np.arange(n)
        a = (a0
             - a1 * np.cos(2.0 * np.pi * i / (n - 1))
             + a2 * np.cos(4.0 * np.pi * i / (n - 1)))

    elif kind == "sine":
        a = np.sin(np.pi * np.arange(n) / (n - 1))

    elif kind == "gaussian":
        sigma = rng.uniform(n / 20.0, n / 2.0)
        centre = (n - 1) / 2.0
        a = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)

    elif kind == "exp_decay":
        rate = rng.uniform(1e-4, 0.02)
        a = np.exp(-rate * np.arange(n))

    elif kind == "laplace":
        tau = rng.uniform(n / 20.0, n / 2.0)
        centre = (n - 1) / 2.0
        a = np.exp(-np.abs(np.arange(n) - centre) / tau)

    elif kind == "power":
        p = rng.uniform(-0.5, 2.0)
        idx = np.minimum(np.arange(n), n - 1 - np.arange(n)) + 1
        a = idx.astype(float) ** p

    elif kind == "sparse":
        # a random set of spikes; the total mass is already target_sum
        k = rng.integers(2, max(3, min(30, n // 30)))
        idx = rng.choice(n, size=k, replace=False)
        vals = rng.random(k)
        vals = vals / vals.sum() * target_sum
        a = np.zeros(n, dtype=float)
        a[idx] = vals
        return a

    else:
        # fallback – uniform
        a = np.full(n, 1.0)

    # Project onto the simplex {a_i ≥ 0, Σ a_i = target_sum}
    total = a.sum()
    if total <= 0:
        # very unlikely, but guard against division by zero
        a = np.full(n, target_sum / n)
    else:
        a = a / total * target_sum
    return a


# ----------------------------------------------------------------------
# Simulated‑annealing refinement ----------------------------------------


def _refine_sequence(a0: np.ndarray, deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing on the simplex: repeatedly move a small
    fraction of mass from a random index i to j.
    """
    n = a0.shape[0]
    target_sum = a0.sum()
    cur = a0.copy()
    # exact max‑convolution of the current candidate
    Nfft = 1 << ((2 * n - 1).bit_length())
    cur_max = _max_convolution(cur, Nfft)

    # annealing schedule (temperature decays linearly)
    # we estimate an initial temperature as a modest fraction of the
    # current objective; this works well in practice.
    T0 = cur_max * 0.05
    steps = 0
    # we cap the number of inner steps to keep within the global deadline
    max_steps = int(min(500_000, (deadline - time.time()) * 1500))

    while steps < max_steps and time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur[i] <= 0.0:
            steps += 1
            continue

        # transfer a random fraction (0.5 % – 5 %) of the source mass
        frac = rng.uniform(0.005, 0.05)
        delta = cur[i] * frac
        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        new_max = _max_convolution(new, Nfft)

        if new_max < cur_max:
            cur, cur_max = new, new_max
            # record a global improvement
            # (the caller will keep the best seen across all lengths)
        else:
            # Metropolis acceptance with linearly decreasing temperature
            T = max(T0 * (1.0 - steps / max_steps), 1e-12)
            prob = math.exp(-(new_max - cur_max) / T)
            if rng.random() < prob:
                cur, cur_max = new, new_max
        steps += 1

    return cur, cur_max

    hill‑climber.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0   # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Seed the search with a previously discovered construction, if any.
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        n0 = base.shape[0]
        target0 = math.sqrt(2.0 * n0)
        base = np.maximum(base, 0.0)
        if base.sum() > 0:
            base = base / base.sum() * target0
            best_seq = base.copy()
            best_score = _max_convolution(base, 1 << ((2 * n0 - 1).bit_length()))
            print(f"[init] loaded previous best – len {n0}, score {best_score:.6f}")
    except Exception:
        base = None

    # ------------------------------------------------------------------
    # Choose a set of lengths to explore.
    length_set = set()
    if base is not None:
        length_set.add(base.shape[0])

    # A handful of lengths around the historically good size (≈1900)
    for factor in [0.8, 0.9, 1.0, 1.1, 1.2, 1.4]:
        if base is not None:
            n = int(base.shape[0] * factor)
        else:
            n = int(1900 * factor)
        length_set.add(max(300, min(n, 8000)))

    # Also explore a few absolute sizes.
    for n in [800, 1200, 1600, 2000, 2400, 2800, 3200, 3600, 4000]:
        length_set.add(n)

    lengths = sorted(length_set)

    families = [
        "uniform",
        "triangular",
        "hann",
        "blackman",
        "sine",
        "gaussian",
        "exp_decay",
        "laplace",
        "power",
        "sparse",
    ]

    # ------------------------------------------------------------------
    # Exhaustive family search for each length.
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        Nfft = 1 << ((2 * n - 1).bit_length())
        print(f"[search] length = {n}")

        # a single interpolated candidate from the current global best,
        # when the lengths differ.
        if best_seq is not None and best_seq.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, best_seq.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, best_seq)
            interp = np.maximum(interp, 0.0)
            interp = interp / interp.sum() * target_sum
            cand_score = _max_convolution(interp, Nfft)
            if cand_score < best_score:
                best_score = cand_score
                best_seq = interp.copy()
                print(f"   [interp] new global best = {best_score:.6f}")

        # how many random draws per length?  we adapt to remaining time
        # but cap at a generous upper bound.
        max_attempts = 5000
        attempts = 0
        while attempts < max_attempts and time.time() < deadline:
            kind = rng.choice(families)
            a = _generate_shape(kind, n, rng, target_sum)
            # quick evaluation
            cur_max = _max_convolution(a, Nfft)

            if cur_max < best_score:
                best_score = cur_max
                best_seq = a.copy()
                print(f"   [new best] len {n} it {attempts} kind {kind} → {best_score:.6f}")

            attempts += 1

    # ------------------------------------------------------------------
    # Local stochastic refinement on the best candidate seen so far.
    if best_seq is not None and time.time() < deadline:
        print("[refine] starting simulated‑annealing refinement")
        refined_seq, refined_score = _refine_sequence(best_seq, deadline, rng)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq
            print(f"[refine] improved global best → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety fallback.
    if best_seq is None:
        n = 1000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = _max_convolution(best_seq,
                                      1 << ((2 * n - 1).bit_length()))
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
