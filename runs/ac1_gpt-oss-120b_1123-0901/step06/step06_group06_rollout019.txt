def _conv_fft(a, fft_len):
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)

Corr: cross-correlation of weight w with a (like convolution with reversed a). Implementation as before:

```
def _corr_fft(w, a, fft_len):
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    # take slice
    start = n - 1

"""Proposes a step function sequence by a hybrid parametric search +
entropic mirror descent (exponentiated‑gradient) refinement.

- Generates many candidate shapes (Dirichlet, Beta, exponential, Gaussian,
  power‑sine, power‑cosine, mixed exponentials, arcsine‑type) for a given
  length.
- Quickly evaluates the objective using an FFT‑based convolution.
- Picks the best candidate and runs a long sub‑gradient mirror‑descent
  (exponentiated‑gradient) loop that directly minimises the max of the
  self‑convolution while keeping the vector non‑negative and normalised.
- Random perturbations are added when progress stalls.
- Respects the time budget (default 1000 s) and returns the best
  sequence found as a Python list.
"""

import time
import numpy as np

def _next_pow2(x):
    return 1 << int(np.ceil(np.log2(x)))

def _conv_fft(a, fft_len):
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _corr_fft(w, a, fft_len):
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

def _objective_from_conv(n, max_c, sum_a):
    # sum_a positive
    return 2.0 * n * max_c / (sum_a * sum_a)

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The function respects a time budget and
    returns the best sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq = None
    best_val = float('inf')

    # Lengths to explore – we start from the largest feasible.
    lengths = [8192, 4096, 2048]

    # Helper to evaluate a sequence using our own fast routine.
    def fast_eval(seq):
        # seq is a NumPy 1‑D array, all non‑negative.
        s = seq.sum()
        if s < 1e-12:
            return float('inf')
        n = seq.shape[0]
        conv_len = _next_pow2(2 * n - 1)
        conv = _conv_fft(seq, conv_len)
        max_c = conv.max()
        return 2.0 * n * max_c / (s * s)

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"--- Optimising length n={n} ---")
        # Pre‑compute FFT sizes
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)   # (2n‑1)+n‑1

        # ----------------------------------------------------------------
        # 1.   Generate a large, diverse pool of parametric candidates.
        # ----------------------------------------------------------------
        candidates = []

        # a) Dirichlet with random concentration
        for _ in range(30):
            conc = np.exp(np.random.uniform(np.log(0.1), np.log(3.0)))
            a = np.random.dirichlet(np.full(n, conc))
            candidates.append(a)

        # b) Symmetric Beta‑type shapes (x^(α‑1)*(1‑x)^(β‑1))
        for _ in range(30):
            α = np.exp(np.random.uniform(np.log(0.2), np.log(4.0)))
            β = np.exp(np.random.uniform(np.log(0.2), np.log(4.0)))
            x = np.linspace(0.0, 1.0, n)
            shape = (x ** (α - 1.0)) * ((1.0 - x) ** (β - 1.0))
            if shape.sum() == 0:
                continue
            shape /= shape.sum()
            candidates.append(shape)

        # c) Double‑exponential decay from both ends
        for _ in range(30):
            γ = np.exp(np.random.uniform(np.log(5e-4), np.log(2e-2)))
            i = np.arange(n)
            shape = np.exp(-γ * np.minimum(i, n - 1 - i))
            shape /= shape.sum()
            candidates.append(shape)

        # d) Centered Gaussian
        for _ in range(30):
            σ = np.exp(np.random.uniform(np.log(n / 30.0), np.log(n / 4.0)))
            i = np.arange(n)
            mid = (n - 1) / 2.0
            shape = np.exp(-0.5 * ((i - mid) / σ) ** 2)
            shape /= shape.sum()
            candidates.append(shape)

        # e) Power‑sine window (zero at both ends)
        for _ in range(30):
            δ = np.exp(np.random.uniform(np.log(0.2), np.log(4.0)))
            i = np.arange(n)
            shape = np.sin(np.pi * i / (n - 1)) ** δ
            shape = np.maximum(shape, 0.0)
            shape_sum = shape.sum()
            if shape_sum == 0:
                continue
            shape /= shape_sum
            candidates.append(shape)

        # f) Power‑cosine window (peaks at the sides)
        for _ in range(30):
            δ = np.exp(np.random.uniform(np.log(0.2), np.log(4.0)))
            i = np.arange(n)
            shape = (0.5 * (1 + np.cos(np.pi * i / (n - 1)))) ** δ
            shape_sum = shape.sum()
            if shape_sum == 0:
                continue
            shape /= shape_sum
            candidates.append(shape)

        # g) Mixed exponential mixture of two ends
        for _ in range(30):
            γ = np.exp(np.random.uniform(np.log(5e-4), np.log(2e-2)))
            w = np.random.rand()
            i = np.arange(n)
            left = np.exp(-γ * i)
            right = np.exp(-γ * (n - 1 - i))
            shape = w * left + (1 - w) * right
            shape /= shape.sum()
            candidates.append(shape)

        # h) Sparse random spikes (≈ √n non‑zero entries)
        for _ in range(20):
            m_spikes = max(2, int(np.sqrt(2 * n)))
            shape = np.zeros(n, dtype=np.float64)
            idx = np.random.choice(n, size=m_spikes, replace=False)
            shape[idx] = np.random.rand(m_spikes) + 0.1   # avoid exact zeros
            shape /= shape.sum()
            candidates.append(shape)

        # i) Arcsine‑type (inverse‑square‑root edges) with tunable exponent
        for _ in range(30):
            α = np.exp(np.random.uniform(np.log(0.1), np.log(2.0)))
            i = np.arange(n, dtype=np.float64)
            # avoid division by zero at the exact ends

            arcs = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
            shape = arcs ** α
            shape /= shape.sum()
            candidates.append(shape)

        if "height_sequence_1" in globals():

            if h.shape[0] > n:

            else:
                h = np.pad(h, (0, n - h.shape[0]))
            h = np.maximum(h, 0.0)
            if h.sum() > 1e-12:
                h = h / h.sum()
                candidates.append(h)

        # ----------------------------------------------------------------
        # 2.   Evaluate all candidates and keep the best one.
        # ----------------------------------------------------------------
        for idx, seq in enumerate(candidates):
            if time.time() > deadline:
                break
            # ensure non‑negative and normalised
            seq = np.maximum(seq, 0.0)
            s = seq.sum()
            if s < 1e-12:
                continue
            seq = seq / s
            # fast evaluation (no clipping)
            conv = _conv_fft(seq, conv_fft_len)
            max_c = conv.max()
            val = 2.0 * n * max_c / (s * s)   # s == 1
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[n={n} candidate {idx}] new best {val:.6f}")

        # ----------------------------------------------------------------
        # 3.   Mirror‑descent (exponentiated‑gradient) refinement of the
        #      current best sequence for this length.
        # ----------------------------------------------------------------
        if best_seq is None or best_seq.shape[0] != n:
            # nothing to refine for this length
            continue

        a = best_seq.copy()
        eta = 0.55      # initial step size
        last_improve = time.time()
        max_iters = 60000   # may be cut by time budget
        for it in range(max_iters):
            if time.time() > deadline:
                break

            conv = _conv_fft(a, conv_fft_len)
            max_c = conv.max()
            # active indices (allow a tiny tolerance)
            active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
            if active.any():
                w = np.zeros_like(conv)
                w[active] = 1.0 / active.sum()
            else:
                w = np.zeros_like(conv)

            grad = 2.0 * _corr_fft(w, a, corr_fft_len)

            # exponential update (mirror descent on the simplex)
            a = a * np.exp(-eta * grad)
            a_sum = a.sum()
            if a_sum < 1e-12:
                a = np.random.dirichlet(np.ones(n))
                a_sum = a.sum()
            a = a / a_sum

            # evaluate current objective
            conv = _conv_fft(a, conv_fft_len)
            max_c = conv.max()
            val = 2.0 * n * max_c   # sum a == 1

            if val < best_val:
                best_val = val
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[n={n} subgrad it={it}] new best {val:.6f}")

            # anneal step size slowly
            if (it + 1) % 5000 == 0:
                eta *= 0.9

            # random shake when stuck
            if time.time() - last_improve > 7.0:
                a = a + (np.random.rand(n) - 0.5) * 1e-3
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                last_improve = time.time()

    # --------------------------------------------------------------------
    # 4.   Fallback – uniform vector (should never be needed)
    # --------------------------------------------------------------------
    if best_seq is None:
        n = 1024
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Return plain Python list for the external evaluator.
    return best_seq.tolist()

"""
Hybrid parametric + exponential‑gradient search for a low‑value step function.

1.  Generate a large, diversified pool of analytic shapes (Dirichlet,
    Beta‑type, double‑exponential, centred Gaussian, power‑sine/cosine,
    mixed exponentials, sparse spikes, arcsine‑type, plus the previous
    global solution if available).  All candidates are normalised to sum
    to 1 (the objective is scale‑invariant).

2.  Evaluate each candidate with a fast FFT‑based convolution and keep the
    best few (default 5) for deeper optimisation.

3.  Refine each of those elite candidates by a long mirror‑descent
    (exponentiated‑gradient) loop that directly minimises the
    max‑convolution while staying on the probability simplex.
    Random “shakes’’ are injected when progress stalls.

4.  Finish with a short Adam optimisation on a smooth‑max surrogate of
    the true max to polish the current global best.

The routine respects a time budget (default 1000 s) and returns the best
sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper: next power of two ≥ x."""
    return 1 << int(np.ceil(np.log2(x)))


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) required for the gradient.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Lengths to explore.  Larger n gives a lower theoretical bound,
    # but also a larger search space.  We start from the largest
    # feasible size and fall back to smaller ones if time runs out.
    # ------------------------------------------------------------------
    lengths = [8192, 4096, 2048]

    # We keep the top‑k raw candidates for a longer mirror‑descent pass.
    TOP_K = 5
    top_candidates = []   # list of (value, np.ndarray)

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n--- Optimising length n={n} ---")
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)   # (2n‑1) + (n‑1)

        # --------------------------------------------------------------
        # 1.  Build a rich collection of parametric candidates.
        # --------------------------------------------------------------
        candidates = []

        # a) Dirichlet draws with random concentration
        for _ in range(80):
            conc = np.exp(np.random.uniform(np.log(0.05), np.log(4.0)))
            candidates.append(np.random.dirichlet(np.full(n, conc)))

        # b) Symmetric Beta‑type (x^{α‑1}(1‑x)^{β‑1})
        for _ in range(80):
            α = np.exp(np.random.uniform(np.log(0.2), np.log(4.0)))
            β = np.exp(np.random.uniform(np.log(0.2), np.log(4.0)))
            x = np.linspace(0.0, 1.0, n)
            shape = (x ** (α - 1.0)) * ((1.0 - x) ** (β - 1.0))
            if shape.sum() == 0:
                continue
            candidates.append(shape / shape.sum())

        # c) Double‑exponential decay from both ends
        for _ in range(80):
            γ = np.exp(np.random.uniform(np.log(5e-4), np.log(2e-2)))
            i = np.arange(n)
            shape = np.exp(-γ * np.minimum(i, n - 1 - i))
            candidates.append(shape / shape.sum())

        # d) Centered Gaussian
        for _ in range(80):
            σ = np.exp(np.random.uniform(np.log(n / 30.0), np.log(n / 4.0)))
            i = np.arange(n)
            mid = (n - 1) / 2.0
            shape = np.exp(-0.5 * ((i - mid) / σ) ** 2)
            candidates.append(shape / shape.sum())

        # e) Power‑sine window (zero at both ends)
        for _ in range(80):
            δ = np.exp(np.random.uniform(np.log(0.2), np.log(4.0)))
            i = np.arange(n)
            shape = np.sin(np.pi * i / (n - 1)) ** δ
            s = shape.sum()
            if s == 0:
                continue
            candidates.append(shape / s)

        # f) Power‑cosine window (peaks at the sides)
        for _ in range(80):
            δ = np.exp(np.random.uniform(np.log(0.2), np.log(4.0)))
            i = np.arange(n)
            shape = (0.5 * (1 + np.cos(np.pi * i / (n - 1)))) ** δ
            s = shape.sum()
            if s == 0:
                continue
            candidates.append(shape / s)

        # g) Mixed exponential mixture of the two ends
        for _ in range(80):
            γ = np.exp(np.random.uniform(np.log(5e-4), np.log(2e-2)))
            w = np.random.rand()
            i = np.arange(n)
            left = np.exp(-γ * i)
            right = np.exp(-γ * (n - 1 - i))
            shape = w * left + (1.0 - w) * right
            candidates.append(shape / shape.sum())

        # h) Sparse random spikes (≈ √n non‑zero entries)
        for _ in range(30):
            m_spikes = max(2, int(np.sqrt(2 * n)))
            shape = np.zeros(n, dtype=np.float64)
            idx = np.random.choice(n, size=m_spikes, replace=False)
            shape[idx] = np.random.rand(m_spikes) + 0.1
            candidates.append(shape / shape.sum())

        # i) Arcsine‑type (inverse‑square‑root edges) with tunable exponent
        for _ in range(80):
            α = np.exp(np.random.uniform(np.log(0.1), np.log(2.0)))
            i = np.arange(n, dtype=np.float64)
            eps = 1e-12
            arcs = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
            shape = arcs ** α
            candidates.append(shape / shape.sum())

        # j) Warm‑start from the previous global solution if present
        global height_sequence_1
        if "height_sequence_1" in globals():
            h = np.asarray(height_sequence_1, dtype=np.float64)
            if h.shape[0] > n:
                h = h[:n]
            else:
                h = np.pad(h, (0, n - h.shape[0]))
            h = np.maximum(h, 0.0)
            if h.sum() > 1e-12:
                candidates.append(h / h.sum())

        # --------------------------------------------------------------
        # 2.  Quick evaluation of all candidates; keep the overall best
        #     and the top‑K elite candidates for refinement.
        # --------------------------------------------------------------
        for seq in candidates:
            if time.time() > deadline:
                break
            seq = np.maximum(seq, 0.0)
            s = seq.sum()
            if s < 1e-12:
                continue
            seq = seq / s          # normalise to sum = 1  (scale‑invariant)

            conv = _conv_fft(seq, conv_fft_len)
            max_c = conv.max()
            val = 2.0 * n * max_c   # denominator s² = 1

            # update global best
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[n={n} raw] new overall best {val:.6f}")

            # maintain top‑K list
            if len(top_candidates) < TOP_K:
                top_candidates.append((val, seq.copy()))
                top_candidates.sort(key=lambda x: x[0])
            else:
                if val < top_candidates[-1][0]:
                    top_candidates[-1] = (val, seq.copy())
                    top_candidates.sort(key=lambda x: x[0])

        # --------------------------------------------------------------
        # 3.  Mirror‑descent (exponentiated‑gradient) refinement of the
        #     elite candidates.
        # --------------------------------------------------------------
        for rank, (init_val, a0) in enumerate(top_candidates):
            if time.time() > deadline:
                break
            print(f"[n={n} EG start #{rank+1}] init val {init_val:.6f}")

            a = a0.copy()
            eta = 0.55                      # initial step size
            last_improve = time.time()
            # iteration budget grows for smaller n
            max_iters = int(4e4 * 1024 / n) + 5000

            for it in range(max_iters):
                if time.time() > deadline:
                    break

                # convolution and identification of active (max) indices
                conv = _conv_fft(a, conv_fft_len)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)

                # weight vector w: uniform over all active positions
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                # gradient of the (non‑smooth) max‑objective
                grad = 2.0 * _corr_fft(w, a, corr_fft_len)

                # exponentiated‑gradient update (mirror descent on the simplex)
                a = a * np.exp(-eta * grad)
                a_sum = a.sum()
                if a_sum < 1e-12:            # avoid collapse
                    a = np.random.dirichlet(np.ones(n))
                    a_sum = a.sum()
                a = a / a_sum

                # current objective value
                conv = _conv_fft(a, conv_fft_len)
                max_c = conv.max()
                val = 2.0 * n * max_c

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} EG #{rank+1} it={it}] new best {val:.6f}")

                # simple annealing of the step‑size
                if (it + 1) % 5000 == 0:
                    eta *= 0.9

                # random shake if nothing improves for a while
                if time.time() - last_improve > 7.0:
                    a += (np.random.rand(n) - 0.5) * 1e-3
                    a = np.maximum(a, 0.0)
                    a /= a.sum()
                    last_improve = time.time()

        # --------------------------------------------------------------
        # 4.  Final polishing of the current global best with a smooth‑max
        #     Adam optimiser (few thousand iterations).
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.shape[0] == n:
            print(f"[n={n}] final Adam polish (current best {best_val:.6f})")
            a = best_seq.copy()
            logits = np.log(a + 1e-15)

            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.045 * np.sqrt(2048.0 / n)
            lam = 0.3
            lam_factor = 1.012
            last_improve = time.time()
            max_fine = int(2e4 * 1024 / n) + 2000

            for t in range(1, max_fine + 1):
                if time.time() > deadline:
                    break

                # soft‑max → simplex
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                conv = _conv_fft(a, conv_fft_len)
                max_c = conv.max()
                # smooth‑max weights
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                grad_a = 2.0 * _corr_fft(w, a, corr_fft_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update on logits
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                lam = min(lam * lam_factor, 1e6)

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} Adam fine it={t}] new best {val:.6f}")

                if time.time() - last_improve > 5.0:
                    # jitter to escape shallow plateaus
                    logits += (np.random.rand(n) - 0.5) * 5e-3
