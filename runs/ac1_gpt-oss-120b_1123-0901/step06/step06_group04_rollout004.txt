def full_convolution(a, fft_len=None):
    n = len(a)
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2*n-1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
    return conv

def cross_correlation(w, a, fft_len):
    n = len(a)
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

def local_improvement(a, n, conv_fft_len, corr_fft_len, deadline):
    best_a = a.copy()
    conv = full_convolution(best_a, conv_fft_len)
    max_c = conv.max()
    best_val = 2 * n * max_c / (best_a.sum()**2)  # sum is 1
    # For speed, we keep best_val.
    while time.time() < deadline:
        # choose step size epsilon relative to small magnitude
        # generate random move: i->j shift delta
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i: j += 1
        if best_a[i] <= 1e-12:
            continue
        delta = np.random.uniform(0, min(best_a[i], 0.01))
        # propose new a
        a_new = best_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # compute val
        conv_new = full_convolution(a_new, conv_fft_len)
        max_c_new = conv_new.max()
        val_new = 2 * n * max_c_new / (a_new.sum()**2)
        if val_new < best_val:
            best_val = val_new
            best_a = a_new
            print(f"Local improvement: new val {best_val:.6f}")
        # optionally limit number of iterations
        if time.time() > deadline - 0.2:
            break
    return best_a, best_val

"""
propose_candidate
-----------------

Search for a non‑negative step function (a probability vector) that
minimises

    F(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm is a multi‑stage heuristic:

1. **Seeding** – a diverse pool of candidates:
   * uniform distribution,
   * the global variable ``height_sequence_1`` if present,
   * symmetric Beta‑like shapes (edge‑concentrated power‑law),
   * two‑Gaussian edge shapes,
   * a few random Dirichlet draws.

2. **Smooth‑max Adam** – optimises a differentiable surrogate of the
   ℓ∞‑norm (log‑sum‑exp with an annealing temperature) using Adam on the
   logits of a soft‑max parametrisation (guarantees simplex constraints).

3. **Exact sub‑gradient refinement** – exponentiated‑gradient updates using
   the true ℓ∞‑sub‑gradient (active set of maximal convolution entries).

4. **Local random transfer** – cheap coordinate‑wise mass transfers that
   occasionally improve the exact objective; this step finishes the
   remaining budget.

All FFT‑based convolutions are performed with a power‑of‑two length for
speed.  The routine respects the supplied wall‑clock budget (default 1000 s)
and returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) required for the gradient.
    Both w and a are real vectors; ``w`` has length 2*len(a)-1.
    Returns a vector of length n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Objective evaluation (exact, no clipping)
# ----------------------------------------------------------------------


def _objective_val(a: np.ndarray, n: int) -> float:
    """Exact value of the evaluation function (no clipping, no guards)."""
    conv = _full_convolution(a)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Phase 1 – Adam on a smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp surrogate of max(conv).
    Returns (best_a, best_val) where best_val is the exact objective.
    """
    eps = 1e-16
    a = np.maximum(a0, eps)
    a /= a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.08                     # a bit larger than the previous code

    lam = 0.5                     # start with a modest temperature
    lam_factor = 1.035

    # iteration budget scaled with length; give a little more than before
    max_iters = int(200_000 * 1024 / n) + 3000
    best_val = np.inf
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # Soft‑max re‑parameterisation (projects onto the simplex)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # Convolution of current a
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()

        # Smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # Gradient of smooth‑max w.r.t. a (via cross‑correlation)
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # Back‑prop through the soft‑max (logits gradient)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # Temperature annealing (approaches the true max)
        lam = min(lam * lam_factor, 1e9)

        # Exact objective for bookkeeping
        val = 2.0 * n * max_c  # sum(a) == 1 after soft‑max
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # Simple stagnation escape
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    # Return the best exact value (computed with the exact objective)
    exact_val = _objective_val(best_a, n)
    return best_a, exact_val


# ----------------------------------------------------------------------
# Phase 2 – Exact ℓ∞ sub‑gradient refinement (exponentiated gradient)
# ----------------------------------------------------------------------


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Refine a vector using the exact ℓ∞ sub‑gradient and an
    exponentiated‑gradient (multiplicative) update.
    """
    a = a0.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective_val(a, n)
    eta = 0.48
    last_improve = time.time()

    for it in range(4000):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13)

        # Build a weight vector that averages over the active set
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        # Sub‑gradient of max(conv) w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # Exponentiated‑gradient step (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s <= 0:
            a = np.random.dirichlet(np.ones(n))
        else:
            a /= s

        # bookkeeping
        val = _objective_val(a, n)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # decay step size gently
        eta *= 0.9996

        # stop if nothing improved for a while
        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – Simple random local transfers (mass‑shifting)
# ----------------------------------------------------------------------


def _local_improve(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Perform cheap random transfers of a tiny amount of mass between two
    coordinates.  Accept only improving moves (no simulated annealing).
    """
    a = a0.copy()
    conv = _full_convolution(a, fft_len=conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        if best_a[i] <= 1e-12:
            it += 1
            continue

        # Transfer a small random amount (up to 1 % of a[i])
        delta = np.random.uniform(0, min(best_a[i], 0.01))
        a_new = best_a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # Exact evaluation
        conv_new = _full_convolution(a_new, fft_len=conv_fft_len)
        val_new = 2.0 * n * conv_new.max() / (a_new.sum() ** 2)

        if val_new < best_val:
            best_val = val_new
            best_a = a_new
            # occasional progress report
            print(f"  Local move improved objective to {best_val:.6f}")
        it += 1

    return best_a, best_val

# Candidate generators (seed shapes)

def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Symmetric Beta‑like shape on {0,…,n‑1}, proportional to
    (i+1)^{α-1} * (n-i)^{β-1}.  Normalised to sum 1.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    shape = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    shape = np.where(np.isfinite(shape), shape, 0.0)
    s = shape.sum()
    if s == 0.0:
        return np.random.dirichlet(np.ones(n))
    return shape / s


def _edge_power_law(n: int, exp: float) -> np.ndarray:
    """symmetrical power‑law a_i ∝ ((i+1)*(n-i))^{exp}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    prod = i * (n - i + 1.0)
    shape = np.where(prod > 0, np.power(prod, exp), 0.0)
    s = shape.sum()
    if s == 0.0:
        return np.random.dirichlet(np.ones(n))
    return shape / s


def _two_gaussian_edges(n: int, gamma: float) -> np.ndarray:
    """Mixture of two Gaussians centred at the ends."""
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    s = shape.sum()
    if s == 0.0:
        return np.random.dirichlet(np.ones(n))
    return shape / s

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function that minimises the provided evaluation
    function.  The routine respects the wall‑clock budget and returns the
    best sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a length.  32768 gave the best bound so far, but we also try
    # a larger size if time permits.
    # ------------------------------------------------------------------
    lengths = [32768, 65536]               # 65536 is attempted only if time allows
    best_seq = None
    best_val = np.inf

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        # --------------------------------------------------------------
        # Build a pool of diverse seeds
        # --------------------------------------------------------------
        seeds = []

        # 0) uniform
        seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

        # 1) use the global best if it matches the size
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            h = np.asarray(hseq, dtype=np.float64)
            if h.shape[0] == n:
                seeds.append(h / h.sum())
            else:
                # up‑sample / down‑sample to the current length
                x_old = np.linspace(0.0, 1.0, h.shape[0])
                x_new = np.linspace(0.0, 1.0, n)
                up = np.interp(x_new, x_old, h)
                if up.sum() > 0:
                    seeds.append(up / up.sum())
                else:
                    seeds.append(np.random.dirichlet(np.ones(n)))

        # 2) beta‑like shapes (both symmetric and mildly skewed)
        for α, β in [(0.3, 0.3), (0.5, 0.5), (0.8, 0.8), (1.2, 1.2), (2.0, 2.0)]:
            seeds.append(_beta_shape(n, α, β))

        # 3) symmetric power‑law (edge‑concentrated)
        for exp in [-3.0, -2.5, -2.0, -1.5, -1.0]:
            seeds.append(_edge_power_law(n, exp))

        # 4) two‑Gaussian edge shapes
        for γ in [0.1, 0.3, 0.6, 1.0, 2.0]:
            seeds.append(_two_gaussian_edges(n, γ))

        # 5) few random Dirichlet draws
        for _ in range(3):
            seeds.append(np.random.dirichlet(np.ones(n)))

        # Limit seeds to a manageable number
        seeds = seeds[:15]

        # --------------------------------------------------------------
        # Optimise each seed
        # --------------------------------------------------------------
        for sid, seed_vec in enumerate(seeds):
            if time.time() > deadline:
                break

            try:
                # Phase 1: smooth‑max Adam
                a_adam, val_adam = _adam_optimize(
                    seed_vec, n, conv_fft_len, corr_fft_len, deadline
                )

                # Phase 2: exact sub‑gradient refinement
                a_ref, val_ref = _subgradient_refine(
                    a_adam, n, conv_fft_len, corr_fft_len, deadline
                )

                # Phase 3: cheap local random transfers
                a_loc, val_loc = _local_improve(
                    a_ref, n, conv_fft_len, deadline, max_iters=1500
                )

                # Determine the best of the three
                cand_val = min(val_adam, val_ref, val_loc)
                cand_seq = None
                if cand_val == val_loc:
                    cand_seq = a_loc
                elif cand_val == val_ref:
                    cand_seq = a_ref
                else:
                    cand_seq = a_adam

                if cand_val < best_val:
                    best_val = cand_val
                    best_seq = cand_seq.copy()
                    print(f"[n={n} seed={sid}] new global best {best_val:.6f}")

            except Exception as exc:
                print(f"[n={n} seed={sid}] exception: {exc}")

        # --------------------------------------------------------------
        # After processing this length, attempt a few targeted jitters
        # --------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            for jitter_id in range(4):
                if time.time() > deadline:
                    break
                logits = np.log(np.maximum(best_seq, 1e-12))
                logits += np.random.normal(scale=0.025, size=n)
                a0 = np.exp(logits - np.max(logits))
                a0 /= a0.sum()

                a_adam, val_adam = _adam_optimize(
                    a0, n, conv_fft_len, corr_fft_len, deadline
                )
                a_ref, val_ref = _subgradient_refine(
                    a_adam, n, conv_fft_len, corr_fft_len, deadline
                )
                a_loc, val_loc = _local_improve(
                    a_ref, n, conv_fft_len, deadline, max_iters=1200
                )
                cand_val = min(val_adam, val_ref, val_loc)
                cand_seq = a_loc if cand_val == val_loc else (a_ref if cand_val == val_ref else a_adam)

                if cand_val < best_val:
                    best_val = cand_val
                    best_seq = cand_seq.copy()
                    print(f"[n={n} jitter={jitter_id}] new global best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely case that nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 32768
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list as required by the evaluator
    return best_seq.tolist()
