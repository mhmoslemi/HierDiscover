"""
search for a low‑value step‑function.

The algorithm proceeds in three phases:

1. **Diverse warm‑starts** – a modest pool of candidates that
   includes uniform, random Dirichlet, geometric decays, double‑Gaussian
   edges, and many “spike” patterns with a few non‑zero entries.
   The best of these is kept as the initial solution.

2. **Simulated‑annealing refinement** – repeatedly move a random
   fraction of mass from one index to another.  Moves are accepted
   according to a Metropolis rule with a slowly cooling temperature.
   Every 20 000 steps the current state is refreshed with a tiny
   Gaussian perturbation of the best‑so‑far solution to escape plateaus.

3. **Optional up‑scaling** – if time permits the final vector is
   linearly interpolated to a longer length (8192) and a short
   annealing run is performed there.  The result is projected back to
   the original length.

All convolutions are performed with a single FFT (O(n log n)) so that
hundreds of thousands of objective evaluations easily fit into the
1000 s budget.  The returned list satisfies the evaluator’s
non‑negativity and range constraints.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Exact objective F(a) = 2·n·max(a∗a) / (sum(a)²).
    The sum is kept close to 1, so the denominator is ≈1.
    """
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    # a.sum() is deliberately kept ≥ 0.01, but we keep the exact formula.
    return 2.0 * n * max_c / (a.sum() ** 2)


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Generate a modest set of diverse starting vectors."""
    cand = []

    # ----- warm start (if supplied) ------------------------------------
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:

        w = np.clip(w, 0.0, None)

        a = np.clip(a, 0.0, None)
        a /= a.sum()
        cand.append(a)

            -((n - 1 - xs) / (sigma * n)) ** 2
        )
        shape = np.clip(shape, 0.0, None)

        cand.append(shape)

    # ----- sinusoidal half‑wave -----------------------------------------
        # (included once, outside the loop)

def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑type annealing on the exact objective.
    Moves consist of transferring a random fraction of mass from i to j.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len)

    # exponential temperature schedule
    T0 = 1.0
    decay = 0.99996

    best_a = a.copy()
    best_val = cur_val

    it = 0
    while time.time() < deadline and it < max_iters:
        T = max(T0 * (decay ** it), 1e-12)

        # propose a move
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        # move a random fraction (up to 40 %) of the source mass
        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        val2 = _objective(a2, n, fft_len)

        # Metropolis acceptance
        if val2 < cur_val or np.random.rand() < np.exp(-(val2 - cur_val) / T):
            a = a2
            cur_val = val2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
                print(f"[SA iter={it:06d}] new best {best_val:.6f}")

        # occasional “shake” to break stagnation
        if (it + 1) % 25_000 == 0:
            noise = np.random.randn(n) * 1e-3
            a = np.clip(best_a + noise, 0.0, None)
            a_sum = a.sum()
            if a_sum < 0.01:
                a = np.random.dirichlet(np.ones(n))
            else:
                a /= a_sum
            cur_val = _objective(a, n, fft_len)
            print(f"[SA iter={it:06d}] restart – cur_val {cur_val:.6f}")

        it += 1

    return best_a, best_val
