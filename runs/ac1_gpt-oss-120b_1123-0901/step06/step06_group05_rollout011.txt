class LBFGSObjective:
    def __init__(self, n, lam):
        self.n = n
        self.lam = lam
        self.conv_len = _next_pow2(2*n-1)
        self.corr_len = _next_pow2(3*n-2)

    def __call__(self, logits):
        # compute a
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()
        conv = _full_convolution(a, self.conv_len)
        # smooth max via logsumexp
        # compute f = 2 * n * (logsumexp(lam * conv) / lam)
        # use scipy.special.logsumexp
        # lam conv can be large but conv small.
        # But it's fine.
        z = self.lam * conv
        f = 2.0 * self.n * (logsumexp(z) / self.lam)
        # compute weight vector w = exp(lam * conv) / sum exp(lam * conv)
        # we have z = lam * conv
        w = np.exp(z - logsumexp(z))
        # gradient w.r.t a
        grad_a = 2.0 * _cross_correlation(w, a, self.corr_len)
        # gradient w.r.t logits
        dot = np.dot(a, grad_a)
        grad_log = a * (grad_a - dot)
        return f, grad_log

"""
Proposal: Multi‑scale L‑BFGS optimisation of a smoothed max‑convolution
objective.

The search works as follows (all functions are defined at top level):

1.  Build a diverse pool of seed sequences (warm‑start, random Dirichlet,
    power‑law, exponentials, Hann, sine, Gaussian bumps …) for several
    lengths `n` (4096 … 32768).  The seeds are normalised to sum‑to‑1.

2.  For every seed run a **L‑BFGS** optimiser on a *smooth‑max* surrogate:
    ```
    f(a) = 2·n· ( log‑sum‑exp( λ·(a*a) ) / λ )
    ```
    where `λ` (temperature) is increased in stages
    `[1.0, 5.0, 20.0]`.  The optimisation variables are the
    logits of the soft‑max parametrisation `a = softmax(logits)`, which
    automatically enforces non‑negativity and the simplex constraint.
    The gradient is computed analytically using FFT‑based cross‑correlation.

3.  The best L‑BFGS result for a seed is refined with a short **exponentiated‑gradient
    sub‑gradient** phase on the *exact* max‑convolution (as in the reference
    code) to remove possible bias introduced by the smoothing.

4.  The globally best sequence seen so far is further polished by a fast
    stochastic **hill‑climb** that moves tiny mass pieces between indices.

All steps respect the global time budget (`budget_s`) and stop gracefully
when the deadline approaches.  The final sequence is returned as a plain
Python list; the external `evaluate_sequence` routine will then compute the
exact score.

The combination of a second‑order optimiser on a smooth surrogate,
temperature annealing and a final exact sub‑gradient/hill‑climb
substantially improves the upper bound compared with the pure Adam/SGD
pipeline.
"""

def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x (fast integer version)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative vector to ``new_n`` points by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:                     # protect against a zero vector
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s


def _generate_initial_candidates(n: int, prev_best):
    """
    Produce a heterogeneous pool of starting points for a given length ``n``.
    ``prev_best`` (if not ``None``) is up‑sampled and used as a warm‑start.
    """
    candidates = []

    # 1. Warm‑start from the best sequence found so far (upsampled)
    if prev_best is not None:
        candidates.append(_upsample(prev_best, n))

    # 2. Historic solution if the global variable exists
    hseq = globals().get('height_sequence_1')
    if isinstance(hseq, (list, np.ndarray)):
        h = np.asarray(hseq, dtype=np.float64)
        candidates.append(_upsample(h, n))

    # 3. Random Dirichlet families (different concentrations)
    candidates.append(np.random.dirichlet(np.ones(n)))
    for conc in (0.2, 0.5, 0.8):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # 4. Edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.1, 7):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # 5. Symmetric two‑sided exponentials
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # 6. Hann window (raised‑cosine)
    i = np.arange(n, dtype=np.float64)
    hann = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    hann /= hann.sum()
    candidates.append(hann)

    # 7. Half‑sine shape
    sine = np.sin(np.pi * i / (n - 1))
    sine = np.maximum(sine, 0)
    sine /= sine.sum()
    candidates.append(sine)

    # 8. Centered Gaussian‑like bumps with several widths
    mu = (n - 1) / 2.0
    for sigma in (n / 10, n / 6, n / 4):
        shape = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        shape /= shape.sum()
        candidates.append(shape)

    # Normalise and discard pathological entries
    clean = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        clean.append(c / s)
    return clean

def _run_adam(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Adam optimiser on the smooth‑max surrogate of the objective.
    (Kept as fallback if SciPy is unavailable.)
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.30
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float('inf')
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t}] new best {val:.6f}")

        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        lam = min(lam * lam_factor, 1e4)

        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val

def _run_subgrad(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Exponentiated‑gradient sub‑gradient phase on the true max‑convolution.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    last_improve = time.time()
    eta0 = 0.55

    for it in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val

def _local_hill_climb(a0: np.ndarray, n: int, deadline: float,
                     max_trials: int = 200_000):
    """
    Very cheap stochastic local search.  Moves ε mass from one index to
    another and keeps the move if it reduces the exact objective.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    best_a = a0.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    no_improve = 0
    trials = 0
    while time.time() < deadline and trials < max_trials and no_improve < 5_000:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0.0, min(0.05, best_a[i]))
        if eps == 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = cand
            no_improve = 0
            print(f"[n={n} HillClimb] improved to {val:.6f}")
        else:
            no_improve += 1
        trials += 1

    return best_a, best_val

# L‑BFGS on the smooth‑max surrogate
_GLOBAL_DEADLINE = None  # will be set inside `propose_candidate`


def _lbfgs_callback(xk):                 # pragma: no cover
    """Callback for SciPy L‑BFGS that aborts when the global deadline is hit."""
    if _GLOBAL_DEADLINE is not None and time.time() > _GLOBAL_DEADLINE:
        raise StopIteration


class _LBFGSObjective:
    """Callable object used by SciPy's L‑BFGS‑B optimizer."""

    def __init__(self, n: int, lam: float):
        self.n = n
        self.lam = lam
        self.conv_len = _next_pow2(2 * n - 1)
        self.corr_len = _next_pow2(3 * n - 2)

    def __call__(self, logits: np.ndarray):
        # soft‑max mapping (logits → probability vector)
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # convolution a*a
        conv = _full_convolution(a, self.conv_len)

        # smooth max via log‑sum‑exp
        z = self.lam * conv
        f = 2.0 * self.n * (logsumexp(z) / self.lam)

        # weights for the gradient (soft‑max over convolution entries)
        w = np.exp(z - logsumexp(z))

        # gradient wrt a (derived from convolution structure)
        grad_a = 2.0 * _cross_correlation(w, a, self.corr_len)

        # back‑prop through the soft‑max (logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        return f, grad_log


def _run_lbfgs_multi(a0: np.ndarray, n: int, deadline: float):
    """
    Run a short L‑BFGS optimisation with an increasing temperature schedule.
    Returns the best sequence found and its exact objective value.
    """
    best_a = a0.copy()
    best_val = float('inf')
    # Temperatures – start smooth, finish very sharp.
    lam_schedule = [1.0, 5.0, 20.0]

    for lam in lam_schedule:
        if time.time() > deadline - 2.0:
            break
        obj = _LBFGSObjective(n, lam)
        logits0 = np.log(best_a + 1e-15)

        try:
            res = _opt.minimize(
                obj,
                logits0,
                method='L-BFGS-B',
                jac=True,
                callback=_lbfgs_callback,
                options={'maxiter': 200, 'maxfun': 500, 'disp': False},
            )
        except Exception as e:  # includes StopIteration from callback
            print(f"[n={n} L‑BFGS lam={lam}] aborted: {e}")
            break

        logits_opt = res.x
        a_opt = np.exp(logits_opt - np.max(logits_opt))
        a_opt /= a_opt.sum()

        conv = _full_convolution(a_opt, _next_pow2(2 * n - 1))
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = a_opt
            print(f"[n={n} L‑BFGS lam={lam}] new best {val:.6f}")

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000):
    """
    Search for a non‑negative step‑function (list of heights) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

    The routine combines:
      • diverse initial seeds,
      • multi‑stage L‑BFGS on a smoothed max‑convolution,
      • short exponentiated‑gradient sub‑gradient refinement,
      • a final cheap stochastic hill‑climb.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Overall time budget in seconds.

    Returns
    -------
    list[float]
        The best height sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin
    global _GLOBAL_DEADLINE
    _GLOBAL_DEADLINE = deadline

    best_seq = None
    best_val = float('inf')

    # Lengths – larger sequences give more expressive power.
    lengths = [4096, 8192, 16384, 32768]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        candidates = _generate_initial_candidates(n, best_seq)

        # try a moderate number of candidates (the pool is already diverse)
        for cid, init in enumerate(candidates):
            if time.time() > deadline:
                break

            # ---- L‑BFGS (smooth‑max) ----
            a_lbfgs, val_lbfgs = _run_lbfgs_multi(init, n, deadline)
            if val_lbfgs < best_val:
                best_val = val_lbfgs
                best_seq = a_lbfgs.copy()
                print(f"[n={n} cand={cid}] L‑BFGS set new global best {val_lbfgs:.6f}")

            # ---- Sub‑gradient refinement (exact max) ----
            # Use a relatively small number of iterations – we only need a
            # quick polishing step after the L‑BFGS run.
            a_sub, val_sub = _run_subgrad(best_seq, n, deadline, max_iters=1500)
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[n={n} cand={cid}] Subgrad improved global best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Final cheap local search to squeeze out any remaining improvement
    if best_seq is not None and time.time() < deadline:
        print(f"\n=== Final local hill‑climb (n={len(best_seq)}) ===")
        a_final, val_final = _local_hill_climb(best_seq, len(best_seq), deadline)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final
            print(f"[final] Hill‑climb improved global best to {val_final:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform if nothing was produced (should never happen)
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Return a plain Python list, as required by the external evaluator
    return best_seq.tolist()
