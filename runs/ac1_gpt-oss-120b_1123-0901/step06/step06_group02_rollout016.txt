"""
Hybrid optimizer for the step‑function bound (target < 1.503).

The algorithm consists of four stages, all respecting a global time budget
(≈ 1000 s by default).

1️⃣  Random parametric warm‑start – diverse families (exp, power‑law,
    Hann, Hamming, Dirichlet) over a wide range of lengths.  The best
    candidates (up to 10) are kept for later refinement.

2️⃣  Adam optimisation on a smooth log‑sum‑exp surrogate of the
    max‑convolution.  The current best sequence is re‑parameterised by
    logits (soft‑max) and optimised for a few seconds per run; the
    best result is retained.

3️⃣  Exact projected sub‑gradient descent with a simple back‑tracking line
    search on the true max‑convolution.  This follows the worst peak
    (index of the current maximum) and adapts the step size automatically.

4️⃣  Targeted simulated‑annealing hill‑climb.  Moves transfer a small
    fraction of mass from a source index that contributes heavily to the
    worst peak to a random target, accepting worse moves with a
    temperature schedule.  Contributions are used to bias the source
    selection, leading to much more focused exploration.

All stages share a global “best” record; intermediate results are printed.
If nothing is found a uniform construction is returned (never happens
in practice).

The returned list is a non‑negative step‑function whose evaluation score
equals the maximum of its full self‑convolution (because the sum is
normalised to √(2 n)).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full convolution via real FFT
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear (full) convolution of two real vectors using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative artefacts
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex of radius s (non‑negative, sum = s)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]          # descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)     # 1‑based
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1        # zero‑based index of last positive entry
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the log‑sum‑exp (soft‑max) surrogate of max‑conv.
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 5e-5,
    alpha0: float = 0.025,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed‑max surrogate.
    Returns the best coefficient vector (on the simplex) and its exact
    max‑convolution value.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # --- soft‑max mapping (logits -> simplex) -----------------------
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)                # stable
        p = exp_shift / np.sum(exp_shift)            # Σ p = 1
        a = target_sum * p                            # simplex point

        # exact objective (max of convolution) -------------------------
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam n={n}  new best max={cur_max:.6f}")

        # --- smooth‑max (log‑sum‑exp) weights -------------------------
        tau = max(init_tau_factor * cur_max * (0.99 ** it),
                  min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a (2·a ⊛ w_rev) -----------------------------
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # back‑prop to logits (chain rule) -----------------------------
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # --- Adam update -----------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slowly decay learning rate
        alpha *= 0.9995

    final_val = _full_convolution_fft(best_a, best_a).max()
    return best_a, float(final_val)


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent with line search
# ----------------------------------------------------------------------
def _subgradient_line_search(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Follow the current worst peak with a back‑tracking line search.
    Returns the best vector found and its exact max‑convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step = target_sum / (10.0 * n)       # modest initial step

    while time.time() < deadline:
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())               # index of the worst peak

        # gradient of the peak (2·a mirrored)
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        improved = False
        # simple back‑tracking line search
        for _ in range(10):
            cand = a - step * grad
            cand = _project_onto_simplex(cand, target_sum)
            cand_val = _full_convolution_fft(cand, cand).max()
            if cand_val < best_val - 1e-12:
                a = cand
                best_a = cand.copy()
                best_val = cand_val
                improved = True
                step *= 1.05          # modestly increase after success
                break
            else:
                step *= 0.5           # shrink step on failure

        if not improved:
            # occasional jitter to avoid stagnation
            if np.random.rand() < 0.05:
                jitter = (np.random.rand(n) - 0.5) * np.max(a) * 1e-4
                a = _project_onto_simplex(a + jitter, target_sum)

        if best_val < 1.503:
            break
    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted simulated‑annealing hill‑climb on the exact objective
# ----------------------------------------------------------------------
def _hillclimb_targeted(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Move a small fraction of mass from a source that contributes heavily
    to the worst peak towards a random target; accept with a Metropolis
    rule using a linear cooling schedule.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    start = time.time()
    T0 = max(0.01, best_val * 0.5)   # initial temperature
    T_end = 1e-6
    rng = np.random.default_rng()

    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())
        start_i = max(0, k - (n - 1))
        end_i = min(k, n - 1)
        js = np.arange(start_i, end_i + 1)
        contrib = a[js] * a[k - js]
        total_contrib = contrib.sum()

        if total_contrib > 0:
            probs = contrib / total_contrib
            i = rng.choice(js, p=probs)   # source biased by contribution
        else:
            i = rng.integers(0, n)

        # choose a distinct target uniformly
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # transfer up to ~12 % of the source mass
        delta = a[i] * rng.random() * 0.12
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new[i] = max(a_new[i], 0.0)               # guard against round‑off

        new_val = _full_convolution_fft(a_new, a_new).max()
        accept = False
        if new_val < best_val:
            accept = True
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb new best {best_val:.6f}")

        # periodic re‑projection to cure drift
        if rng.random() < 0.001:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break
    return best_a, best_val


# ----------------------------------------------------------------------
# Random parametric initialisation utilities
# ----------------------------------------------------------------------
def _random_sequence(rng: np.random.Generator, n: int) -> np.ndarray:
    """Draw a random non‑negative sequence of length n from a diverse family."""
    family = rng.choice(["exp", "pow", "hann", "hamming", "dirichlet"])
    if family == "exp":
        tau = rng.uniform(0.05 * n, 5.0 * n)
        a = np.exp(-np.arange(n) / tau)
    elif family == "pow":
        p = rng.uniform(0.2, 2.5)
        a = np.arange(1, n + 1) ** (-p)
    elif family == "hann":
        i = np.arange(n, dtype=float)
        a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    elif family == "hamming":
        i = np.arange(n, dtype=float)
        a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    else:  # dirichlet / sparse gamma draws
        conc = rng.uniform(0.1, 5.0)
        a = rng.gamma(conc, 1.0, size=n)

    # tiny optional Gaussian noise
    if rng.random() < 0.20:
        noise = rng.normal(scale=np.max(a) * 0.02, size=n)
        a = np.maximum(a + noise, 0.0)

    # normalise to the required L1 sum √(2 n)
    target = np.sqrt(2.0 * n)
    a = a * (target / a.sum())
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid search for a non‑negative step‑function that minimises the
    evaluation score (equivalently the maximal self‑convolution).
    Returns the best sequence as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.10   # safety margin

    best_seq = None
    best_score = np.inf
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, seq)

    # --------------------------------------------------------------
    # Helper to keep the global best and a small elite pool
    # --------------------------------------------------------------
    def _record(seq: np.ndarray):
        nonlocal best_seq, best_score, top_candidates
        val = _full_convolution_fft(seq, seq).max()
        if val < best_score:
            best_score = float(val)
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f}  n={len(seq)}")
        # maintain a pool of the ten best candidates
        if len(top_candidates) < 10:
            top_candidates.append((val, seq.copy()))
            top_candidates.sort(key=lambda x: x[0])
        else:
            if val < top_candidates[-1][0]:
                top_candidates[-1] = (val, seq.copy())
                top_candidates.sort(key=lambda x: x[0])
        return val

    # --------------------------------------------------------------
    # Warm‑start from a possible existing construction
    # --------------------------------------------------------------
    try:
        hs = globals().get("height_sequence_1")
        if hs is not None:
            hs_arr = np.asarray(hs, dtype=float)
            n0 = hs_arr.shape[0]
            target0 = np.sqrt(2.0 * n0)
            hs_norm = hs_arr * (target0 / hs_arr.sum())
            _record(hs_norm)
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – Random parametric initialisations (≈ 30 % of budget)
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.30 * budget_s
    while time.time() < phase1_deadline:
        # length grows early, shrinks later – simple uniform draw works well
        n = rng.integers(1500, 8001)
        seq = _random_sequence(rng, n)
        _record(seq)

    # --------------------------------------------------------------
    # Phase 2 – Adam on the smoothed max‑convolution (≈ 20 % of budget)
    # --------------------------------------------------------------
    phase2_deadline = phase1_deadline + 0.20 * budget_s
    if best_seq is not None:
        while time.time() < phase2_deadline:
            n = best_seq.shape[0]
            target_sum = np.sqrt(2.0 * n)
            # initialise logits from the current best
            x0 = np.log(best_seq + 1e-12)
            run_deadline = min(time.time() + 15.0, phase2_deadline)
            seq_adam, val_adam = _optimise_adam_softmax(
                n, x0, target_sum, run_deadline,
                init_tau_factor=0.12,
                min_tau_factor=5e-5,
                alpha0=0.025,
            )
            _record(seq_adam)
            # break if we run out of time
            if time.time() >= phase2_deadline:
                break

    # --------------------------------------------------------------
    # Phase 3 – Exact projected sub‑gradient descent (≈ 20 % of budget)
    # --------------------------------------------------------------
    phase3_deadline = phase2_deadline + 0.20 * budget_s
    if best_seq is not None and time.time() < phase3_deadline:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        seq_sub, val_sub = _subgradient_line_search(
            best_seq.copy(),
            target_sum,
            phase3_deadline,
        )
        _record(seq_sub)

    # --------------------------------------------------------------
    # Phase 4 – Focused simulated‑annealing hill‑climb (remainder)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        seq_hill, val_hill = _hillclimb_targeted(
            best_seq.copy(),
            target_sum,
            deadline,
        )
        _record(seq_hill)

    # --------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 2500
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = _full_convolution_fft(best_seq, best_seq).max()
        print(f"[{time.time():.2f}] fallback uniform score = {best_score:.6f}")

    return best_seq.tolist()
