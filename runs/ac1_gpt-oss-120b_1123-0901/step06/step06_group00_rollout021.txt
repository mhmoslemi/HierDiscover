import time, numpy as np, math, sys

def _next_pow2(x):
    return 1 << (x-1).bit_length()

def _full_convolution(a, fft_len):
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[:2*a.shape[0] - 1]

def _cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]
```

def _generate_candidate(n, seed):
    rng = np.random.default_rng(seed)
    typ = rng.choice(['dirichlet', 'beta', 'exp', 'power', 'gauss', 'sin', 'uniform'])
    if typ == 'dirichlet':
        alpha = rng.uniform(0.01, 5.0)
        a = rng.dirichlet(np.full(n, alpha))
    elif typ == 'beta':
        a = rng.beta(0.5, 0.5, size=n)   # shape for edges
        a = a / a.sum()
    elif typ == 'exp':
        lam = rng.uniform(0.0005, 0.01)
        a = np.exp(-lam * np.arange(n))
        a = a / a.sum()
    elif typ == 'power':
        p = rng.uniform(-2.5, -0.3)
        idx = np.arange(1, n+1)
        a = (idx * (n - idx + 1)) ** p
        a = np.where(a > 0, a, 0)
        a = a / a.sum()
    elif typ == 'gauss':
        sigma = rng.uniform(n/20, n/2)
        x = np.arange(n)
        a = np.exp(-0.5 * ((x - n/2) / sigma) ** 2)
        a = a / a.sum()
    elif typ == 'sin':
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:  # uniform
        a = np.full(n, 1.0 / n)
    return a
```

def _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline,
                   max_iters=500000, init_lr=0.1):
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    lam = 0.5
    lam_factor = 1.01

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # softmax to get a on simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        # objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2) # sum=1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam iter={it}] new best {val:.6f}")

        # smooth max
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # gradient
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # compute log gradient as in original
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        lr = init_lr / (np.sqrt(it + 1) + 1)  # decaying lr
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, 1e6)

        # occasional shake
        if it % 5000 == 0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    # final a from logits at exit
    mx = logits.max()
    exp_log = np.exp(logits - mx)
    a_final = exp_log / exp_log.sum()
    final_val = _objective(a_final, n, conv_fft_len)
    if final_val < best_val:
        best_val = final_val
        best_a = a_final.copy()
    return best_a, best_val

def _subgradient_optimize(a0, n, conv_fft_len, corr_fft_len, deadline,
                          max_iters=400000, eta0=0.6):
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad iter={it}] new best {val:.6f}")

        # active set
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size schedule: eta = eta0 / sqrt(it+1)
        eta = eta0 / (np.sqrt(it + 1) + 1)
        a = a * np.exp(-eta * grad_a)

        # re-normalize
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        it += 1

    final_val = _objective(a, n, conv_fft_len)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()
    return best_a, best_val

            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it}] improved to {val2:.6f}")

"""
Fast multi‑restart optimizer for the step‑function problem.

The algorithm works in three layers:

1. **Diverse random restarts** (Dirichlet, exponential, power‑law, Gaussian,
   sinusoidal, uniform).  Each restart receives a time slice and runs a full
   optimisation pipeline.

2. **Full optimisation pipeline** for one candidate:
   – Adam on a smooth‑max (log‑sum‑exp) surrogate.
   – Exact‑max sub‑gradient descent with a decaying step‑size.
   – Very cheap random local moves (mass transfer between two positions).

3. **Upscaling**:  the best candidate is linearly interpolated to a larger
   grid (8192 points) and refined again while time permits.

All operations use NumPy FFTs (O(n log n)) and stay on the simplex
(non‑negative, sum = 1).  The routine respects the given time budget and
always returns a valid sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective (exact, not the surrogate)
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """2·n·max(a*a) / (sum(a)^2).  a is assumed non‑negative."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Random‑candidate generator
def _generate_candidate(n: int, seed: int) -> np.ndarray:
    """Return a random non‑negative vector of length n (sum = 1)."""
    rng = np.random.default_rng(seed)
    typ = rng.choice(
        ["dirichlet", "beta", "exp", "power", "gauss", "sin", "uniform"]
    )
    if typ == "dirichlet":
        alpha = rng.uniform(0.01, 5.0)
        a = rng.dirichlet(np.full(n, alpha))
    elif typ == "beta":
        a = rng.beta(0.5, 0.5, size=n)
        a = a / a.sum()
    elif typ == "exp":
        lam = rng.uniform(0.0005, 0.01)
        a = np.exp(-lam * np.arange(n))
        a = a / a.sum()
    elif typ == "power":
        p = rng.uniform(-2.5, -0.3)
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = (idx * (n - idx + 1)) ** p
        a = np.where(a > 0, a, 0.0)
        a = a / a.sum()
    elif typ == "gauss":
        sigma = rng.uniform(n / 20.0, n / 2.0)
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-0.5 * ((x - n / 2.0) / sigma) ** 2)
        a = a / a.sum()
    elif typ == "sin":
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
    else:  # uniform
        a = np.full(n, 1.0 / n, dtype=np.float64)
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    init_lr: float = 0.10,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    lam = 0.5          # temperature for the smooth‑max
    lam_factor = 1.01

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # simplex from logits
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (use the invariant sum==1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits (learning‑rate decays with sqrt(it))
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        lr = init_lr / (math.sqrt(it + 1) + 1.0)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to escape plateaus
        if it % 5_000 == 0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    # final point
    mx = logits.max()
    exp_log = np.exp(logits - mx)
    a_final = exp_log / exp_log.sum()
    final_val = _objective(a_final, n, conv_fft_len)
    if final_val < best_val:
        best_val = final_val
        best_a = a_final.copy()
    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    eta0: float = 0.60,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase (exponentiated‑gradient)."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2_000 == 0:
                print(f"[Subgrad iter={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step‑size
        eta = eta0 / (math.sqrt(it + 1) + 1.0)
        a = a * np.exp(-eta * grad_a)

        # re‑normalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    final_val = _objective(a, n, conv_fft_len)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()
    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 50_000,
) -> tuple[np.ndarray, float]:
    """Very cheap hill‑climbing: move a random fraction of mass."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        delta = a[i] * (np.random.rand() * 0.4)          # ≤40 % transfer
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val - 1e-12:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Full optimisation pipeline for a single starting vector
def _optimize_full(
    a0: np.ndarray, n: int, deadline: float
) -> tuple[np.ndarray, float]:
    """
    Run Adam → sub‑gradient → cheap local search on a single candidate.
    The time budget is split roughly 1/2‑1/2 between Adam and the remaining
    phases (sub‑gradient + local search).
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a0 = np.maximum(a0, 0.0)
    if a0.sum() < 1e-12:
        a0 = np.random.dirichlet(np.ones(n))
    a0 = a0 / a0.sum()

    now = time.time()
    if now >= deadline - 5.0:
        return a0, _objective(a0, n, conv_fft_len)

    # --- Adam phase -------------------------------------------------
    half_deadline = now + (deadline - now) * 0.5
    a_adam, val_adam = _adam_optimize(
        a0, n, conv_fft_len, corr_fft_len, half_deadline
    )

    # --- Sub‑gradient phase -----------------------------------------
    now = time.time()
    if now >= deadline - 5.0:
        # not enough time for sub‑gradient; return the best so far
        return a_adam, min(val_adam, _objective(a0, n, conv_fft_len))

    a_sub, val_sub = _subgradient_optimize(
        a_adam, n, conv_fft_len, corr_fft_len, deadline
    )

    # --- Local random search ----------------------------------------
    now = time.time()
    if now < deadline - 5.0:
        a_loc, val_loc = _local_random_search(
            a_sub, n, conv_fft_len, deadline
        )
        # pick the best among the three results
        candidates = [(a_adam, val_adam), (a_sub, val_sub), (a_loc, val_loc)]
        best_a, best_val = min(candidates, key=lambda x: x[1])
        return best_a, best_val
    else:
        # deadline imminent – return the best of Adam / sub‑gradient
        if val_sub < val_adam:
            return a_sub, val_sub
        else:
            return a_adam, val_adam


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

    The routine respects the given time budget (default 1000 s) and
    returns the best sequence found.  It uses several random restarts,
    each followed by an Adam → sub‑gradient → local‑search pipeline,
    and finally up‑scales the best solution to a larger grid (8192 points)
    if time permits.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Base length for the quick restarts
    n_small = 4096

    # Warm start from a previous solution, if provided
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n_small:
            warm = warm[:n_small]
        elif warm.shape[0] < n_small:
            warm = np.pad(warm, (0, n_small - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() > 1e-8:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Multiple random restarts (each gets a slice of the remaining budget)
    max_restarts = 5
    restart = 0
    while time.time() < deadline - 120 and restart < max_restarts:
        remaining = deadline - time.time()
        # Give each restart up to 180 s or half of the remaining time,
        # whichever is smaller.
        allocate = min(180.0, remaining * 0.5)
        if allocate < 30.0:
            break

        if restart == 0 and warm is not None:
            candidate = warm.copy()
        else:
            candidate = _generate_candidate(n_small, seed + restart * 9973)

        a_opt, val_opt = _optimize_full(candidate, n_small, time.time() + allocate)

        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[restart {restart}] new global best {val_opt:.6f}")

        restart += 1

    # ------------------------------------------------------------------
    # Upscale to a larger grid if we still have time.
    if best_seq is not None and time.time() < deadline - 30:
        n_big = 8192
        # Linear interpolation to the finer grid
        x_old = np.linspace(0.0, 1.0, best_seq.shape[0], endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        if a_big.sum() < 1e-12:
            a_big = np.random.dirichlet(np.ones(n_big))
        else:
            a_big = a_big / a_big.sum()

        a_opt_big, val_opt_big = _optimize_full(a_big, n_big, deadline)

        if val_opt_big < best_val:
            best_val = val_opt_big
            best_seq = a_opt_big.copy()
            print(f"[big upscale] new best {val_opt_big:.6f}")

    # ------------------------------------------------------------------
    # Fallback – ensure a valid return value.
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator's accepted range.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
