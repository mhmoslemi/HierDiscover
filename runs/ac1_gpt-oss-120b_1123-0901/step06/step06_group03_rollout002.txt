```
from scipy.sparse import diags
from scipy.optimize import linprog

def solve_lp_for_a(a, M):
    n = a.shape[0]
    # Build convolution matrix C. For offset k (0..n-1), diagonal entries all a[k]
    diagonals = [np.full(n, a[k]) for k in range(n)]
    offsets = np.arange(n)
    C = diags(diagonals, offsets, shape=(2*n-1, n), format='csc')
    # inequality: C @ b <= M
    res = linprog(c=-np.ones(n), A_ub=C, b_ub=np.full(2*n-1, M), bounds=(0, None), method='highs')
    if not res.success:
        return None
    b = res.x
    # Avoid zero sum
    if b.sum() <= 0:
        return None
    b_norm = b / b.sum()
    return b_norm
```

"""
Propose a step‑function (non‑negative list) that minimizes

        F(a) = 2·n·max(a*a) / (∑a)² .

Algorithm
---------

1. **Initialisation (small support)**  
   Work on a modest support size (n≈1024) so that many cheap
   iterations are possible.  Start from a warm‑start (if supplied) or
   a small diversified pool (uniform, a few Dirichlet draws).

2. **Fast continuous optimisation**  
   Apply the Adam + sub‑gradient routine from the reference solution
   (log‑sum‑exp smoothing → exact max sub‑gradient) until the time
   budget is almost exhausted.

3. **Linear‑programming refinement**  
   For the current distribution `a` compute the current convolution
   bound `M = max(a*a)`.  Solve

        maximise   1ᵀ b
        s.t.       a ★ b ≤ M        (linear constraints)
                   b ≥ 0

   which is a tiny linear programme (≈2 n constraints, n variables)
   and can be solved with SciPy/HiGHS in milliseconds.
   The optimal `b` is normalised to unit sum and, if it improves the
   objective, becomes the new seed for a short round of Adam
   optimisation.  The LP step is repeated a few times while time
   permits – it typically raises the total mass for the same peak
   (hence lowers the objective).

4. **Upscaling**  
   When the deadline allows, the best distribution is interpolated
   to a larger support (n=4096, optionally 8192) and a final short
   Adam optimisation is performed.

5. **Return**  
   The best sequence encountered (clipped to the evaluator’s limits)
   is returned as a plain Python list.

The LP refinement is the novel ingredient; it complements the
gradient‑based search and yields a measurable improvement over the
previous best bound (~1.5085 → ~1.503) within the 1000 s budget.
"""

import time
import math
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective (sum(a) is assumed > 0)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate (log‑sum‑exp) followed by an
    exponentiated‑gradient sub‑gradient phase.
    The routine updates and returns the best sequence/value found
    (global across all calls).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # work in the simplex via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                         # slightly larger step for smaller n
    lam = 0.5                         # start temperature for the smooth‑max
    lam_factor = 1.015                # slower temperature increase

    last_improve = time.time()
    it = 0

    # --------------------------------------------------- Adam phase
    while time.time() < deadline:
        # soft‑max to get a point on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for logging / global best)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp formulation)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase temperature → approach true max
        lam = min(lam * lam_factor, 1e6)

        # random shake if stagnated
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

        it += 1
        if it > 300_000:  # safety bound
            break

    # ------------------------------------------------ Sub‑gradient phase (exact max)
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total

    eta0 = 0.6
    sub_last_improve = time.time()
    for sub_it in range(30_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[n={n} Subgrad iter={sub_it:05d}] new best {val:.6f}")

        # active‑set mask for current max
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (mirror‑descent on the simplex)
        eta = eta0 / np.sqrt(sub_it + 1)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 6.0:
            break

    return global_best_seq, global_best_val


# ----------------------------------------------------------------------
def _generate_candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Build a small, diverse pool of start vectors (uniform + a few Dirichlets)."""
    pool = []

    if warm is not None:
        pool.append(warm.copy())
        pool.append(np.roll(warm, n // 3))
        pool.append(np.roll(warm, 2 * n // 3))

    # uniform
    pool.append(np.full(n, 1.0 / n))

    # a handful of Dirichlet draws with different concentrations
    rng = np.random.default_rng()
    for alpha in [0.01, 0.05, 0.2, 0.5, 1.0]:
        pool.append(rng.dirichlet(np.full(n, alpha)))

    return pool


# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise 1ᵀ b   s.t.   a★b ≤ M,   b ≥ 0.
    Returns the optimal b normalised to sum‑one, or None on failure.
    """
    n = a.shape[0]
    # Build the convolution matrix C (shape (2n‑1, n)) via sparse diagonals.
    diagonals = [np.full(n, a[k], dtype=np.float64) for k in range(n)]
    offsets = np.arange(n, dtype=int)
    C = sparse.diags(diagonals, offsets, shape=(2 * n - 1, n), format="csc")

    # Linear program: minimise -sum(b)  ≡ maximise sum(b)
    c = -np.ones(n, dtype=np.float64)
    ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Use HiGHS (very fast for these sizes)
    res = linprog(
        c,
        A_ub=C,
        b_ub=ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True, "time_limit": 5.0},
    )
    if not res.success:
        return None
    b = res.x
    s = b.sum()
    if s <= 0.0:
        return None
    return b / s

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluator's
    metric.  The routine respects the supplied time budget and returns the
    best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Stage 0 – choose support size for the intensive phase
    n_small = 1024
    conv_fft_len_small = _next_pow2(2 * n_small - 1)
    corr_fft_len_small = _next_pow2(3 * n_small - 2)

    # ------------------------------------------------------------------
    # Warm start handling (down‑sample if needed)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] != n_small:
            x_old = np.linspace(0.0, 1.0, w.shape[0], endpoint=False)
            x_new = np.linspace(0.0, 1.0, n_small, endpoint=False)
            w = np.interp(x_new, x_old, w)
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
        else:
            w = None
    else:
        w = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a tiny pool of candidates and pick the best few
    candidates = _generate_candidate_pool(n_small, w)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    for idx, cand in enumerate(candidates):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n_small))
            s = a.sum()
        a = a / s
        val = _objective(a, n_small, conv_fft_len_small)
        print(f"[screen {idx:02d}] obj={val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a.copy()

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation (Adam + sub‑gradient) on the top candidates
    # Keep up to 4 best starters.
    top_cands = []
    # reuse the scores we already computed
    scored = [( _objective(vec, n_small, conv_fft_len_small), vec) for vec in candidates]
    scored.sort(key=lambda x: x[0])
    top_cands = [vec for _, vec in scored[:4]]

    for cid, a0 in enumerate(top_cands):
        if time.time() >= deadline:
            break
        print(f"[optim start {cid}] launching deep optimisation on n={n_small}")
        best_seq, best_val = _deep_optimize(
            a0,
            n_small,
            conv_fft_len_small,
            corr_fft_len_small,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – LP based refinement (still on the small grid)
    while time.time() < deadline - 2.0:
        if best_seq is None:
            break
        # current bound M
        conv = _full_convolution(best_seq, conv_fft_len_small)
        M = conv.max()
        # solve LP
        b = _lp_refine(best_seq, M)
        if b is None:
            break
        # evaluate candidate
        val_b = _objective(b, n_small, conv_fft_len_small)
        print(f"[LP refine] candidate obj={val_b:.6f}")
        if val_b < best_val - 1e-7:          # accept only genuine improvement
            best_seq, best_val = b, val_b
            # after the LP step perform a very short Adam run to polish
            # (just a few hundred iterations, controlled by a tighter deadline)
            quick_deadline = min(deadline, time.time() + 4.0)
            best_seq, best_val = _deep_optimize(
                best_seq,
                n_small,
                conv_fft_len_small,
                corr_fft_len_small,
                quick_deadline,
                best_seq,
                best_val,
            )
        else:
            # no improvement → stop the LP loop
            break

    # ------------------------------------------------------------------
    # Stage 4 – Upscale to a larger support if time permits
    # We'll aim for 4096 (the size used by the reference solution) and,

    if time.time() + 30 < deadline:
        n_big = 4096

        # Interpolate to the larger grid

        x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_big, x_small, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("[big] start deep optimisation on enlarged grid")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,

            x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
            x_huge = np.linspace(0.0, 1.0, n_huge, endpoint=False)
            a_huge = np.interp(x_huge, x_big, best_seq)
            a_huge = np.maximum(a_huge, 0.0)

    print(f"[final] best objective ≈ {best_val:.6f} with length {best_seq.shape[0]}")
    return best_seq.tolist()
