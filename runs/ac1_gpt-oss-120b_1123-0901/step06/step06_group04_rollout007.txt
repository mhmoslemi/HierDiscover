def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    best_seq = None
    best_val = float('inf')
    # Choose length based on time budget. We'll start at 32768; if we have enough time, we will go to 65536.
    lengths = [32768, 65536]

    for n in lengths:
        if time.time() > deadline:
            break

        # Determine FFT lengths.
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

        # Build initial candidates
        candidates = []

        # 1) Interpolate previous best if exists
        if best_seq is not None:
            old = np.asarray(best_seq)
            # Interpolate to new size
            x_old = np.linspace(0, 1, len(old))
            x_new = np.linspace(0, 1, n)
            cand = np.interp(x_new, x_old, old)
            cand = np.maximum(cand, 0)
            if cand.sum() == 0:
                cand = np.random.dirichlet(np.ones(n))
            else:
                cand = cand / cand.sum()
            candidates.append(cand)

        # 2) Uniform
        candidates.append(np.full(n, 1.0 / n))

        # 3) Power law exponent sweep
        idx = np.arange(1, n+1, dtype=float)
        prod = idx * (n - idx + 1)
        for alpha in np.linspace(-4.0, -0.1, 15):
            shape = np.power(prod, alpha, where=prod>0)
            if shape.sum() == 0:
                continue
            shape = shape / shape.sum()
            candidates.append(shape)

        # 4) Exponential edge decay
        i = np.arange(n, dtype=float)
        for beta in np.linspace(0.0005, 0.02, 8):
            shape = np.exp(-beta * np.minimum(i, n-1-i))
            shape = shape / shape.sum()
            candidates.append(shape)

        # 5) Gaussian mixture
        for sigma in [0.05,0.1,0.2,0.3]:
            gauss = np.exp(-0.5 * ((i - 0) / sigma / n)**2)
            gauss2 = np.exp(-0.5 * ((i - (n-1)) / sigma / n)**2)
            shape = gauss + gauss2
            shape = shape / shape.sum()
            candidates.append(shape)

        # 6) Random Dirichlet (few)
        for _ in range(5):
            candidates.append(np.random.dirichlet(np.ones(n)))

        # Maybe random mixtures of uniform and edge shape

        # Limit number of candidates to keep runtime manageable: for large n maybe 30-40.
        candidates = candidates[:30]

        for ci, cand in enumerate(candidates):
            if time.time() > deadline:
                break

"""
Propose a step‑function (non‑negative vector) that minimises the evaluation
function

    F(a) = 2 * n * max(a * a) / (sum(a)**2)

The algorithm is a heavy‑weight two‑phase local optimiser with a
large‑scale final refinement:

* A diverse pool of *structured* seeds (uniform, up‑sampled previous best,
  power‑law edges, exponential edges, Gaussian edge mixtures and a few
  Dirichlet draws) is built for a large length (≈ 65536).  The pool is tiny
  (≤ 15 candidates) so that each seed can be refined for many iterations.

* Phase 1 – Adam on a smooth‑max surrogate.
  The surrogate replaces the ℓ∞‑norm by a soft‑max
  `log‑sum‑exp(λ·conv)` with a temperature λ that is increased very slowly.
  The vector is parametrised by logits → soft‑max guaranteeing non‑negativity
  and unit sum.  Adam’s learning‑rate is annealed and a patience counter
  injects tiny random jitter when progress stalls.

* Phase 2 – Exact sub‑gradient (exponentiated‑gradient) on the true ℓ∞‑norm.
  A few thousand exact sub‑gradient steps with a slowly decaying step size
  often push the solution a little further.

* After exhausting the initial seeds the current best solution is jittered
  repeatedly (log‑its perturbed with Gaussian noise) and each jitter is
  processed again by the two‑phase optimiser.  This “zoom‑in” loop continues
  until the time budget (default 1000 s) is exhausted, thereby using the
  full wall‑clock for refinement.

The code respects the 1000 s budget, prints progress, and finally returns
the best sequence as a plain ``list`` ready for the external ``evaluate_sequence``.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Convolution / correlation helpers (FFT based, re‑used many times)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT. Length 2*len(a)-1."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns length‑n vector.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Phase‑1: Adam optimiser on a smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best iterate (as a probability vector) and its exact objective.
    """
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    # Adam state
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.07                     # a little larger than the original code

    # smooth‑max temperature schedule
    lam = 0.5                     # start fairly smooth
    lam_factor = 1.01             # increase slowly
    lam_cap = 1e8

    # iteration budget – we want many more steps for large n
    if max_iters is None:
        max_iters = int(20000 * 1024 / n) + 4000
    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        # time check
        if time.time() > deadline:
            break

        # ---- simplex via soft‑max ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- convolution & smooth‑max weights ----
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        # soft‑max weighting over the 2n‑1 positions
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # ---- gradient of the smooth‑max (w.r.t. a) ----
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # ---- back‑propagation through soft‑max (logits gradient) ----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature annealing ----
        lam = min(lam * lam_factor, lam_cap)

        # ---- exact objective for bookkeeping ----
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # ---- stagnation handling (tiny random jitter) ----
        if time.time() - last_improve > 6.0:
            # add a very small random offset to escape flat regions
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase‑2: Exact ℓ∞ sub‑gradient refinement (exponentiated gradient)
# ----------------------------------------------------------------------


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Exact ℓ∞ sub‑gradient refinement using an exponentiated‑gradient update.
    Returns the best iterate and its exact objective.
    """
    a = a0 / a0.sum()
    best_val = float("inf")
    best_a = a.copy()
    eta = 0.45                     # initial step size
    last_improve = time.time()

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        # active set (indices attaining the maximum, tolerance generous)
        active = np.isclose(conv, max_c, atol=1e-12)

        # uniform distribution over the active indices
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:               # should not happen
            w = np.zeros_like(conv)

        # sub‑gradient of the ℓ∞‑norm
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:                     # numerical safeguard
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # evaluate exact objective
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # step‑size decay
        eta *= 0.9995

        # early stop if stagnating
        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Candidate generators (structured seeds)
# ----------------------------------------------------------------------


def _make_powerlaw_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric edge‑concentrated shape: (i·(n‑i+1))^α, normalised."""
    idx = np.arange(1, n + 1, dtype=float)
    prod = idx * (n - idx + 1.0)
    shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
    if shape.sum() == 0:
        return np.full(n, 1.0 / n)
    return shape / shape.sum()


def _make_exponential_edge(n: int, beta: float) -> np.ndarray:
    """Edge‑decay shape exp(-β·min(i, n‑1‑i))."""
    i = np.arange(n, dtype=float)
    shape = np.exp(-beta * np.minimum(i, n - 1 - i))
    if shape.sum() == 0:
        return np.full(n, 1.0 / n)
    return shape / shape.sum()


def _make_two_gaussian_edge(n: int, sigma: float) -> np.ndarray:
    """Two identical Gaussians placed at the ends."""
    i = np.arange(n, dtype=float)
    gauss = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    gauss2 = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = gauss + gauss2
    if shape.sum() == 0:
        return np.full(n, 1.0 / n)
    return shape / shape.sum()

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the provided
    ``evaluate_sequence`` objective.  The routine respects the wall‑clock
    limit ``budget_s`` (default 1000 s) and returns the best sequence found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Initialise best solution (maybe from a previous run)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    if "height_sequence_1" in globals():
        h = np.asarray(globals()["height_sequence_1"], dtype=float)
        if h.size > 0:
            best_seq = h / h.sum()
            best_val = evaluate_sequence(best_seq.tolist())
            print(f"Starting from previous global best: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Target length – choose the largest affordable size
    # ------------------------------------------------------------------
    # One very large run (≈ 65k) allows many optimisation steps.
    n_target = 65536
    conv_fft_len = 1 << int(np.ceil(np.log2(2 * n_target - 1)))
    corr_fft_len = 1 << int(np.ceil(np.log2(3 * n_target - 2)))

    # ------------------------------------------------------------------
    # Build an initial pool of structured seeds (≤ 15)
    # ------------------------------------------------------------------
    candidates = []

    # 1) Uniform
    candidates.append(np.full(n_target, 1.0 / n_target, dtype=float))

    # 2) Interpolated previous best (if any)
    if best_seq is not None:
        old = best_seq
        x_old = np.linspace(0.0, 1.0, len(old))
        x_new = np.linspace(0.0, 1.0, n_target)
        up = np.interp(x_new, x_old, old)
        up = np.maximum(up, 0.0)
        if up.sum() > 0:
            up = up / up.sum()
        else:
            up = np.random.dirichlet(np.ones(n_target))
        candidates.append(up)

    # 3) Power‑law edge shapes (grid over α)
    for alpha in np.linspace(-4.0, -0.2, 9):
        candidates.append(_make_powerlaw_shape(n_target, alpha))

    # 4) Exponential edge decays
    for beta in np.linspace(0.0003, 0.015, 6):
        candidates.append(_make_exponential_edge(n_target, beta))

    # 5) Two‑Gaussian edge mixtures
    for sigma in [0.04, 0.08, 0.12, 0.20]:
        candidates.append(_make_two_gaussian_edge(n_target, sigma))

    # 6) Few random Dirichlet draws (to provoke diversity)
    for _ in range(4):
        candidates.append(np.random.dirichlet(np.ones(n_target)))

    # Remove possible duplicates (very unlikely) and keep a modest size
    candidates = candidates[:15]

    # ------------------------------------------------------------------
    # Optimise each seed (Adam → sub‑gradient)
    # ------------------------------------------------------------------
    for cid, cand in enumerate(candidates):
        if time.time() > deadline:
            break

        cand = np.asarray(cand, dtype=float)
        cand = cand / cand.sum()                     # ensure unit simplex

        try:
            a_adam, val_adam = _adam_optimize(
                cand, n_target, conv_fft_len, corr_fft_len, deadline
            )
            a_ref, val_ref = _subgradient_refine(
                a_adam, n_target, conv_fft_len, corr_fft_len, deadline
            )
            # evaluate with the official evaluator (guards against clamping)
            val_seq = evaluate_sequence(a_ref.tolist())
            if val_seq < best_val:
                best_val = val_seq
                best_seq = a_ref.copy()
                print(
                    f"[seed {cid}] new global best {best_val:.6f}"
                )
            elif val_ref < best_val:   # fall‑back on the sub‑gradient value
                best_val = val_ref
                best_seq = a_ref.copy()
                print(
                    f"[seed {cid}] improved (sub‑grad) {best_val:.6f}"
                )
            elif val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(
                    f"[seed {cid}] improved (Adam) {best_val:.6f}"
                )
        except Exception as exc:
            print(f"[seed {cid}] optimisation exception: {exc}")

    # ------------------------------------------------------------------
    # Final “zoom‑in” loop: jitter the current best and re‑optimise.
    # This runs until the time budget is spent.
    # ------------------------------------------------------------------
    jitter_round = 0
    while time.time() < deadline:
        if best_seq is None:
            break
        jitter_round += 1
        # Perturb logits of the current best
        logits = np.log(np.maximum(best_seq, 1e-12))
        logits += np.random.normal(scale=0.02, size=n_target)
        a0 = np.exp(logits - np.max(logits))
        a0 = a0 / a0.sum()

        try:
            a_adam, _ = _adam_optimize(
                a0, n_target, conv_fft_len, corr_fft_len, deadline
            )
            a_ref, _ = _subgradient_refine(
                a_adam, n_target, conv_fft_len, corr_fft_len, deadline
            )
            val_seq = evaluate_sequence(a_ref.tolist())
            if val_seq < best_val:
                best_val = val_seq
                best_seq = a_ref.copy()
                print(
                    f"[jitter {jitter_round}] new global best {best_val:.6f}"
                )
        except Exception as exc:
            print(f"[jitter {jitter_round}] exception: {exc}")

        # safety break if we are within a second of the deadline
        if deadline - time.time() < 1.0:
            break

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        print("Fallback to uniform distribution.")
        best_seq = np.full(2048, 1.0 / 2048, dtype=float)

    # Return plain Python list as required by the external driver
    print(f"Final best value: {best_val:.6f} on length {len(best_seq)}")
    return best_seq.tolist()
