def propose_candidate(seed=0,budget_s=1000):
    import numpy as np, time, math

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    best_seq = None
    best_val = float('inf')

    # Choose a list of lengths to try, e.g., [2048,4096,8192,16384,32768,65536]
    lengths = [2048,4096,8192,16384,32768,65536]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"=== length {n} ===")
        conv_fft_len = 1 << int(np.ceil(np.log2(2*n-1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3*n-2)))

        # Build seeds
        seeds = []

        # uniform
        seeds.append(np.full(n,1.0/n))

        # previous global best interpolation
        if best_seq is not None:
            old = np.asarray(best_seq)
            x_old = np.linspace(0,1,old.shape[0])
            x_new = np.linspace(0,1,n)
            up = np.interp(x_new, x_old, old)
            up = np.maximum(up,0)
            if up.sum()>0:
                up = up/up.sum()
            else:
                up = np.random.dirichlet(np.ones(n))
            seeds.append(up)

        # random Dirichlet with alpha = 0.5 for spikier
        for _ in range(3):
            seeds.append(np.random.dirichlet(np.full(n, 0.5)))

        # Beta-shaped using sin^p etc.
        i = np.arange(n)
        for p in [0.5,1.5,3.0]:
            shape = (np.sin(np.pi * (i+0.5)/n)**p)
            shape = shape/shape.sum()
            seeds.append(shape)

        # power law edge shapes with alphas [-0.5,-1,-2,-3,-4]
        prod = (i+1)*(n-i)
        for alpha in [-0.5,-1,-1.5,-2,-3]:
            shape = np.where(prod>0, prod**alpha, 0)
            if shape.sum()>0:
                shape = shape/shape.sum()
                seeds.append(shape)

        # Limit seeds to say 15 for larger n
        max_seeds = 12 if n <= 32768 else 8
        seeds = seeds[:max_seeds]

        # Perform optimization for each seed
        for idx,seed_a in enumerate(seeds):
            if time.time() > deadline: break
            # run Adam + subgradient
            a_opt, val_opt = _optimize_full(seed_a, n, conv_fft_len, corr_fft_len, deadline)
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[n {n} seed {idx}] new best {best_val:.6f}")

        # After seeds, if best length matches n, do a local simulated annealing for remaining time
        if best_seq is not None and len(best_seq)==n:
            # allocate maybe 10% of remaining time to SA
            remaining = deadline - time.time()
            if remaining > 5.0:
                a_curr = best_seq.copy()
                val_curr = best_val
                # schedule
                T0 = 0.01
                T_end = 1e-6
                total_iters = int(remaining * 1000)  # 1000 steps per second approx
                for it in range(total_iters):

                    # propose perturbation: random Gaussian noise on logits
                    log_a = np.log(np.maximum(a_curr,1e-12))
                    noise = np.random.normal(scale=0.05, size=n)
                    log_new = log_a + noise * np.sqrt(T)  # scaling with T

"""
propose_candidate
=================

Search for a non‑negative step‑function (a probability vector) that minimizes

    F(a) = 2 * n * max(a * a) / (sum(a)**2)

where “*’’ is the linear convolution.  The objective is homogeneous; we keep
the simplex constraint sum(a)=1 and minimise the ℓ∞‑norm of the
self‑convolution.

Algorithm
---------

* **Multi‑scale search** – we try a cascade of lengths (2048 … 65536).  For
  each length we build a diverse pool of seed vectors (uniform, interpolated
  previous best, Dirichlet draws, power‑law edge shapes, beta‑shaped bumps,
  sparse spikes, …).

* **Phase 1 – Adam on a smooth‑max surrogate** – the max is approximated by a
  log‑sum‑exp with a temperature λ that is increased exponentially during the
  run.  The vector is represented in the log‑domain (logits) and the
  soft‑max enforces the simplex constraints.

* **Phase 2 – Exact ℓ∞ sub‑gradient refinement** – after the Adam phase we
  refine the solution with an exponentiated‑gradient update that uses the
  true ℓ∞‑sub‑gradient (uniform weighting over the active convolution
  entries).

* **Phase 3 – Simulated‑annealing jitter** – the current best of the current
  length is further perturbed in log‑space; Metropolis acceptance (temperature
  cooling) allows occasional uphill moves, helping to escape local minima.

All heavy work uses FFT‑based convolution (O(n log n)) and runs respect the
wall‑clock budget (default 1000 s).  The final best vector is returned as a
plain Python list – ready for `evaluate_sequence`.

The code is deliberately self‑contained, uses only `numpy`, `time` and
`math`, and avoids any file or network I/O.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length `fft_len`.
    Returns a vector of length 2*len(a)-1.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑n vector where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0 … n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = None,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smooth‑max surrogate of max(conv).
    Returns (best_a, best_val) where best_val = 2*n*max(conv) for the best
    iterate discovered before the deadline.
    """
    # Ensure positivity and simplex normalisation
    a = np.maximum(a0, 1e-12)
    a /= a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05

    lam = 1.0                # smooth‑max temperature
    lam_factor = 1.02

    # iteration budget: a few thousand iterations are enough for any length.
    if max_iters is None:
        max_iters = int(50000 * 1024 / n) + 3000
    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # -------- simplex via soft‑max ----------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # -------- smooth‑max weights ----------
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # -------- gradient of smooth‑max ----------
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # -------- back‑prop through soft‑max ----------
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # -------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # -------- temperature annealing ----------
        lam = min(lam * lam_factor, 1e9)

        # -------- exact objective evaluation ----------
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # -------- occasional jitter to avoid dead‑ends ----------
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Exact ℓ∞ sub‑gradient refinement using exponentiated‑gradient updates.
    Returns (best_a, best_val) after at most `max_iters` iterations or when
    the deadline is reached.
    """
    a = a0 / a0.sum()
    best_val = float("inf")
    best_a = a.copy()
    eta = 0.45
    last_improve = time.time()

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        # active set (all entries within atol of the max)
        active = np.isclose(conv, max_c, atol=1e-12)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient step (maintains positivity)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # fallback to fresh Dirichlet
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # evaluate
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        eta *= 0.9995                      # slow decay

        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val


def _optimize_full(
    seed_a: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam → sub‑gradient refinement and return the best of the two.
    """
    # Adam with a generous iteration budget
    a_adam, val_adam = _adam_optimize(
        seed_a, n, conv_fft_len, corr_fft_len, deadline
    )
    # Exact refinement
    a_ref, val_ref = _subgradient_refine(
        a_adam, n, conv_fft_len, corr_fft_len, deadline
    )
    if val_ref < val_adam:
        return a_ref, val_ref
    else:
        return a_adam, val_adam


# ----------------------------------------------------------------------
# Simulated‑annealing jitter (log‑space perturbations)
# ----------------------------------------------------------------------


def _anneal_jitter(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_val: float,
    best_seq: np.ndarray,
) -> tuple[np.ndarray, float]:
    """
    Perform a simple simulated‑annealing run in log‑space starting from `a0`.
    The function returns the best sequence seen during the run and its value.
    """
    a_curr = a0.copy()
    val_curr = best_val
    log_a = np.log(np.maximum(a_curr, 1e-12))

    # remaining budget gives us an approximate number of iterations
    remaining = max(0.0, deadline - time.time())
    if remaining <= 0.0:
        return best_seq, best_val

    # Roughly 1000 proposals per second (FFT dominates)
    total_iters = int(remaining * 800) + 1
    # temperature schedule (linear cooling)
    T_start = 0.04
    T_end = 1e-5

    for it in range(total_iters):
        # linear cooling
        t = T_start * (1 - it / total_iters) + T_end * (it / total_iters)

        # propose a jitter in log‑space (scale with sqrt(t) for smoother cooling)
        noise = np.random.normal(scale=0.08 * math.sqrt(t + 1e-12), size=n)
        log_new = log_a + noise
        # soft‑max to stay on the simplex
        mx = log_new.max()
        exp_new = np.exp(log_new - mx)
        a_new = exp_new / exp_new.sum()

        conv = _full_convolution(a_new, fft_len=conv_fft_len)
        val_new = 2.0 * n * conv.max()
        delta = val_new - val_curr

        # Metropolis acceptance
        if delta < 0 or np.random.rand() < math.exp(-delta / (t + 1e-12)):
            a_curr = a_new
            log_a = log_new
            val_curr = val_new
            if val_curr < best_val:
                best_val = val_curr
                best_seq = a_curr.copy()
                print(f"[SA n={n} it={it}] new global best {best_val:.6f}")

        # early exit if we are out of time
        if time.time() > deadline:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function within the given wall‑clock budget.  Returns the best sequence
    as a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Length schedule – start small (cheap) and grow.
    # --------------------------------------------------------------
    lengths = [2048, 4096, 8192, 16384, 32768, 65536]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

        # ----------------------------------------------------------
        # Build a heterogeneous seed pool
        # ----------------------------------------------------------
        seeds = []

        # 1) Uniform
        seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

        # 2) Interpolated previous global best (if any)
        if best_seq is not None:
            old = np.asarray(best_seq, dtype=np.float64)
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, old)
            up = np.maximum(up, 0.0)
            if up.sum() > 0:
                up /= up.sum()
            else:
                up = np.random.dirichlet(np.ones(n))
            seeds.append(up)

        # 3) Sparse random spikes (≈5 % non‑zeros)
        k = max(2, n // 2000)
        for _ in range(2):
            mask = np.zeros(n, dtype=bool)
            mask[np.random.choice(n, k, replace=False)] = True
            vals = np.random.rand(k)
            spike = np.zeros(n)
            spike[mask] = vals
            if spike.sum() > 0:
                spike /= spike.sum()
            else:
                spike = np.random.dirichlet(np.ones(n))
            seeds.append(spike)

        # 4) Dirichlet draws with low concentration (spikier than uniform)
        for conc in [0.3, 0.5, 0.8]:
            seeds.append(np.random.dirichlet(np.full(n, conc)))

        # 5) Power‑law edge shapes (U‑shaped)
        idx = np.arange(1, n + 1, dtype=np.float64)
        prod = idx * (n - idx + 1.0)
        for alpha in [-0.5, -1.0, -1.5, -2.0, -3.0]:
            shape = np.where(prod > 0, prod ** alpha, 0.0)
            if shape.sum() > 0:
                shape /= shape.sum()
                seeds.append(shape)

        # 6) Sine‑power bumps (central concentration)
        i = np.arange(n, dtype=np.float64)
        for p in [0.5, 1.0, 2.0, 3.0]:
            shape = np.sin(np.pi * (i + 0.5) / n) ** p
            shape /= shape.sum()
            seeds.append(shape)

        # 7) Two‑Gaussian edge mixtures
        for gamma in [0.1, 0.3, 0.6, 1.0]:
            shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            shape /= shape.sum()
            seeds.append(shape)

        # 8) Hybrid uniform–edge mixture
        edge = np.exp(-0.5 * (i ** 2)) + np.exp(-0.5 * ((n - 1 - i) ** 2))
        edge /= edge.sum()
        mix = 0.5 * np.full(n, 1.0 / n) + 0.5 * edge
        mix /= mix.sum()
        seeds.append(mix)

        # Limit the number of seeds for very large n to stay within budget
        max_seeds = 12 if n <= 32768 else 8
        seeds = seeds[:max_seeds]

        # ----------------------------------------------------------
        # Optimise each seed (Adam → sub‑gradient)
        # ----------------------------------------------------------
        for sid, seed_vec in enumerate(seeds):
            if time.time() > deadline:
                break
            try:
                a_opt, val_opt = _optimize_full(
                    seed_vec, n, conv_fft_len, corr_fft_len, deadline
                )
                if val_opt < best_val:
                    best_val = val_opt
                    best_seq = a_opt.copy()
                    print(f"[n={n} seed={sid}] new global best {best_val:.6f}")
            except Exception as exc:
                print(f"[n={n} seed={sid}] exception: {exc}")

        # ----------------------------------------------------------
        # Simulated‑annealing jitter around the current best of this n
        # ----------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            # allocate a modest portion of the remaining budget to SA
            sa_deadline = min(deadline, time.time() + 0.25 * (deadline - time.time()))
            best_seq, best_val = _anneal_jitter(
                best_seq,
                n,
                conv_fft_len,
                sa_deadline,
                best_val,
                best_seq,
            )

    # ------------------------------------------------------------------
    # Fallback: if nothing was found (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        print("Fallback to uniform sequence (unexpected).")

    # Optional: expose the best sequence globally for future runs
    globals()["height_sequence_1"] = best_seq.tolist()

    # Return as a plain Python list (required by external evaluator)
    return best_seq.tolist()
