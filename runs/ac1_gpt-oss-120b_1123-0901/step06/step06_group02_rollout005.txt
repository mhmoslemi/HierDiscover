def full_convolution_fft(x,y):
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx*fy, n=size)[:n]
    conv[conv<0] = 0.0
    return conv

def generate_parametric_sequence(n):
    kind = np.random.choice(['exp','exp_mix','pow','pow_exp','gamma','gaussian','uniform'])
    i = np.arange(n, dtype=float)
    if kind == 'exp':
        tau = np.random.uniform(0.1, 3.0) * n
        a = np.exp(-i / tau)
    elif kind == 'exp_mix':
        tau1 = np.random.uniform(0.1, 2.0) * n
        tau2 = np.random.uniform(0.1, 2.0) * n
        w = np.random.rand()
        a = w * np.exp(-i / tau1) + (1-w) * np.exp(-i / tau2)
    elif kind == 'pow':
        p = np.random.uniform(0.1, 2.5)
        a = (i+1) ** (-p)
    elif kind == 'pow_exp':
        p = np.random.uniform(0.1, 2.5)
        tau = np.random.uniform(0.1, 3.0) * n
        a = (i+1) ** (-p) * np.exp(-i / tau)
    elif kind == 'gamma':
        k = np.random.uniform(0.5, 5.0)
        theta = np.random.uniform(0.1, 2.0) * n
        a = (i+1) ** (k-1) * np.exp(- (i+1) / theta)
    elif kind == 'gaussian':
        mu = np.random.uniform(0, n-1)
        sigma = np.random.uniform(0.05, 0.3) * n
        a = np.exp(-0.5 * ((i - mu)/sigma) ** 2)
    else: # uniform Dirichlet
        a = np.random.dirichlet(np.ones(n))
    # Clip negative due to rounding (shouldn't have) and add small noise optionally
    a = np.maximum(a, 0.0)
    return a

def adam_softmax_optimize(n, x0, target_sum, deadline,
                         init_tau_factor=0.1,
                         min_tau_factor=1e-5,
                         alpha0=0.02,
                         beta1=0.9,
                         beta2=0.999,
                         eps=1e-8):
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    best_seq = None
    best_val = np.inf
    it = 0
    alpha = alpha0
    while time.time() < deadline:
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        p = exp_shift / np.sum(exp_shift)
        a = target_sum * p
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Adam best max={best_val:.6f}")

        # compute smooth max weights
        tau = max(init_tau_factor * cur_max * np.exp(-0.001*it), min_tau_factor * cur_max)
        # might also incorporate linear decay.
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum <= 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # compute gradient of weighted sum of conv entries:
        # d/da sum_k w_k conv_k = 2 * (a * (w_rev)). where w_rev is reversed w.
        w_rev = w[::-1]
        size2 = 1 << ((2*n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3*n - 2]
        idx = (2*n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]
        # chain rule to logits
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        it += 1
        m = beta1*m + (1-beta1)*grad_x
        v = beta2*v + (1-beta2)*(grad_x**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step
        # occasional jitter
        if it % 200 == 0 and np.random.rand() < 0.1:
            x += np.random.randn(n) * 0.001
        # decay learning rate
        alpha *= 0.9995
    return best_seq, float(best_val)

def subgradient_multi_peak(a0, target_sum, deadline, tol=1e-12):

    n = a.shape[0]
    best_a = a.copy()

        it += 1
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        # find active indices +- tolerance
        eps = max(1e-9, cur_max * 1e-7)
        active = np.where(conv >= cur_max - eps)[0]
        # compute gradient sum for active set
        # if too many active, limit to e.g., first 30
        if active.size > 30:
            active = np.random.choice(active, size=30, replace=False)
        # compute gradient via FFT as described (sum of w for each active)
        # We'll create indicator vector c of length 2n-1 with ones at active positions.
        c = np.zeros(2*n-1)

        a_rev = a[::-1]
        size2 = 1 << ((2*n - 2 + n - 1).bit_length())

"""
Hybrid global optimiser for the step‑function L∞ convolution bound.

Goal: minimise   max_k (a * a)[k]   subject to a_i ≥ 0 and Σ a_i = √(2 n).
The routine works in five stages:

1.  Load any known good solution from the global variable
    `height_sequence_1` and record it.
2.  Generate many diverse parametric candidates (exponential,
    mixed‑exponential, power‑law, gamma, Gaussian, Dirichlet).  Keep the
    best.
3.  Refine the current best with several short Adam‑softmax runs on a
    log‑sum‑exp surrogate of the max‑convolution.
4.  Perform a multi‑peak projected sub‑gradient descent that follows the
    convex hull of all active constraints (all near‑max convolution
    indices).  This often removes the dominant peak faster than a
    single‑index sub‑gradient.
5.  Final simulated‑annealing hill‑climb that moves tiny mass packets
    between 1‑3 random positions with a Metropolis acceptance rule.

All heavy work uses FFT‑based convolution (O(n log n)).  The optimiser
adapts its step‑sizes and temperature automatically and respects the
overall time budget (default 1000 s).  The returned object is a plain
Python list of non‑negative floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of real vectors x and y."""
    n = x.shape[0] + y.shape[0] - 1
    # next power of two for efficient FFT
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # Clip tiny negatives caused by rounding errors
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (non‑negative, sum = s)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)  # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Evaluation of the exact max‑convolution (the objective after normalisation)
# ----------------------------------------------------------------------
def _max_convolution_val(a: np.ndarray) -> float:
    """Return max_k (a * a)[k] for a real non‑negative vector a."""
    return float(_full_convolution_fft(a, a).max())


# ----------------------------------------------------------------------
# Parametric random initialisation (various families)
# ----------------------------------------------------------------------
def _generate_parametric_sequence(n: int) -> np.ndarray:
    """Draw a random non‑negative sequence of length n from a diverse family."""
    kind = np.random.choice(
        ["exp", "exp_mix", "pow", "pow_exp", "gamma", "gaussian", "dirichlet"]
    )
    i = np.arange(n, dtype=float)

    if kind == "exp":
        tau = np.random.uniform(0.1, 3.0) * n
        a = np.exp(-i / tau)

    elif kind == "exp_mix":
        tau1 = np.random.uniform(0.1, 2.0) * n
        tau2 = np.random.uniform(0.1, 2.0) * n
        w = np.random.rand()
        a = w * np.exp(-i / tau1) + (1.0 - w) * np.exp(-i / tau2)

    elif kind == "pow":
        p = np.random.uniform(0.1, 2.5)
        a = (i + 1) ** (-p)

    elif kind == "pow_exp":
        p = np.random.uniform(0.1, 2.5)
        tau = np.random.uniform(0.1, 3.0) * n
        a = (i + 1) ** (-p) * np.exp(-i / tau)

    elif kind == "gamma":
        k = np.random.uniform(0.5, 5.0)
        theta = np.random.uniform(0.1, 2.0) * n
        a = (i + 1) ** (k - 1) * np.exp(- (i + 1) / theta)

    elif kind == "gaussian":
        mu = np.random.uniform(0, n - 1)
        sigma = np.random.uniform(0.05, 0.3) * n
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)

    else:  # dirichlet (uniform on the simplex)
        a = np.random.dirichlet(np.ones(n))

    # keep non‑negative, add a tiny jitter with small probability
    a = np.maximum(a, 0.0)
    if np.random.rand() < 0.15:
        noise = np.random.randn(n) * np.max(a) * 0.02
        a = np.maximum(a + noise, 0.0)
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max (log‑sum‑exp) surrogate of max‑convolution
# ----------------------------------------------------------------------
def _adam_softmax_optimize(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 5e-5,
    alpha0: float = 0.025,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser for a soft‑max approximation of the max‑convolution."""
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # map logits → soft‑max probabilities → coefficients on simplex
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)  # stable
        p = exp_shift / np.sum(exp_shift)   # Σ p = 1
        a = target_sum * p

        # exact max‑convolution (for bookkeeping)
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam n={n} new best max={cur_max:.6f}")

        # smoothed‑max weights (log‑sum‑exp)
        tau = max(
            init_tau_factor * cur_max * (0.99 ** it),
            min_tau_factor * cur_max,
        )
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient wrt a of Σ w_k conv_k = 2·a ⊛ w_rev
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # back‑prop to logits
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slow decay of learning rate
        alpha *= 0.9995

    final_val = _max_convolution_val(best_a) if best_a is not None else np.inf
    return best_a, float(final_val)


# ----------------------------------------------------------------------
# Multi‑peak projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_multi_peak(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent that follows the convex hull of all active peaks."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _max_convolution_val(a)
    step = 0.2 * target_sum / n  # initial step size (≈ average coefficient)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()

        # indices where the convolution is within a tiny tolerance of the maximum
        eps = max(1e-9, cur_max * 1e-7)
        active = np.where(conv >= cur_max - eps)[0]

        # limit the active set for speed (random subset if too large)
        if active.size > 40:
            active = np.random.choice(active, size=40, replace=False)

        # indicator vector for the active constraints
        c = np.zeros(2 * n - 1, dtype=float)
        c[active] = 1.0

        # gradient = 2·a_rev ⊛ c   (sum of gradients of all active constraints)
        a_rev = a[::-1]
        size2 = 1 << ((3 * n - 2 - 1).bit_length())
        Fa_rev = np.fft.rfft(a_rev, n=size2)
        Fc = np.fft.rfft(c, n=size2)
        conv2 = np.fft.irfft(Fa_rev * Fc, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * conv2[idx]

        # projected step
        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)
        cand_val = _max_convolution_val(cand)

        if cand_val < best_val - 1e-12:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= 1.02  # increase step after a successful move
        else:
            step *= 0.70  # shrink step on failure
            if np.random.rand() < 0.07:
                jitter = np.random.rand(n) * 1e-4
                a = _project_onto_simplex(a + jitter, target_sum)

        if it % 200 == 0:
            print(f"[{time.time():.2f}] subgrad it={it} best={best_val:.6f}")

        if best_val < 1.503:
            break

    return best_a, float(best_val)


# ----------------------------------------------------------------------
# Enhanced stochastic hill‑climb with multi‑move Metropolis acceptance
# ----------------------------------------------------------------------
def _enhanced_hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing style hill‑climb moving tiny mass packets between 1‑3 positions."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _max_convolution_val(a)

    # temperature schedule (proportional to current peak)
    T0 = max(1e-3, best_val * 0.4)
    T_end = 1e-5
    start = time.time()
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        # decide how many elementary moves to perform this iteration
        n_moves = np.random.choice([1, 2, 3], p=[0.65, 0.28, 0.07])
        a_new = a.copy()

        for _ in range(n_moves):
            i = np.random.randint(0, n)
            if a_new[i] <= 0:
                continue
            j = np.random.randint(0, n)
            if i == j:
                continue
            # transfer at most 8 % of mass at i
            delta = a_new[i] * np.random.rand() * 0.08
            a_new[i] -= delta
            a_new[j] += delta

        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        new_val = _max_convolution_val(a_new)

        accept = False
        if new_val < best_val:
            accept = True
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")

        # occasional re‑normalisation to counteract numerical drift
        if it % 2500 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, float(best_val)

# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid search for a non‑negative step‑function with minimal evaluation score.
    Returns a plain Python list of coefficients.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.10   # safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 0. Warm‑start from any globally supplied construction
    # ------------------------------------------------------------------
    try:
        hs = globals().get("height_sequence_1")
        if hs is not None:
            hs_arr = np.asarray(hs, dtype=float)
            n0 = hs_arr.shape[0]
            target0 = np.sqrt(2.0 * n0)
            hs_norm = hs_arr * (target0 / hs_arr.sum())
            val = _max_convolution_val(hs_norm)
            best_seq = hs_norm
            best_score = val
            print(f"[{time.time():.2f}] global warm‑start n={n0} val={val:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Random parametric initialisations
    # ------------------------------------------------------------------
    while time.time() < deadline - 250.0:   # reserve ~250 s for later phases
        # Choose a length (larger lengths later in the budget)
        remaining = deadline - time.time()
        if remaining > 600:
            n = np.random.randint(2500, 8000 + 1)
        elif remaining > 300:
            n = np.random.randint(2000, 6000 + 1)
        else:
            n = np.random.randint(1500, 4000 + 1)

        target_sum = np.sqrt(2.0 * n)
        a0 = _generate_parametric_sequence(n)
        # Normalise to required L1 sum
        a0 = a0 * (target_sum / a0.sum())
        val = _max_convolution_val(a0)
        if val < best_score:
            best_score = val
            best_seq = a0.copy()
            print(f"[{time.time():.2f}] new best after init n={n} val={val:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam optimisation on a smoothed max‑convolution (multiple short runs)
    # ------------------------------------------------------------------
    while best_seq is not None and time.time() < deadline - 150.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        # initialise logits from current best (soft‑max parametrisation)
        x0 = np.log(best_seq + 1e-12)

        # allocate a modest slice of time for this Adam run (≈20 s)
        run_deadline = time.time() + min(20.0, deadline - time.time() - 30.0)
        seq_local, val_local = _adam_softmax_optimize(
            n,
            x0,
            target_sum,
            run_deadline,
            init_tau_factor=0.12,
            min_tau_factor=5e-5,
            alpha0=0.025,
        )
        if seq_local is not None and val_local < best_score:
            best_score = val_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] after Adam refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Multi‑peak projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_deadline = deadline - 40.0   # keep some time for the final hill‑climb
        seq_sub, val_sub = _subgradient_multi_peak(
            best_seq.copy(),
            target_sum,
            sub_deadline,
        )
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after multi‑peak sub‑gradient best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        hill_deadline = deadline
        seq_hill, val_hill = _enhanced_hillclimb(
            best_seq.copy(),
            target_sum,
            hill_deadline,
        )
        if val_hill < best_score:
            best_score = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] after hill‑climb best={best_score:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if best_seq is None:

        best_score = _max_convolution_val(best_seq)
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    print(f"[{time.time():.2f}] FINAL best value {best_score:.6f} (length {len(best_seq)})")
    # Return plain Python list as required by the evaluator
    return best_seq.tolist()
