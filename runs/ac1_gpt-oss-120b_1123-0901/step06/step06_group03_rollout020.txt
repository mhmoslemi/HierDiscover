def _project_simplex(v):
    # projection onto simplex sum=1, v>=0
    # algorithm from Duchi et al.
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non-positive, return uniform?
        theta = cssv[-1] / n
        w = np.maximum(v - theta, 0)
        return w
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    # small n
    n = 4096
    conv_fft_len = _next_pow2(2* n -1)
    corr_fft_len = _next_pow2(3 * n -2)
    # warm start
    warm_start = globals().get("height_sequence_1")
    warm = None
    if warm_start is not None:
        w = np.asarray(warm_start, dtype=np.float64)
        # adjust to length n
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]), mode='constant')
        w = np.maximum(w, 0.0)
        s = w.sum()
        if s >= 0.01:
            w = w / s
            warm = w
    # generate candidate pool
    candidates = _generate_candidate_pool(n, warm)
    best_seq = None
    best_val = float("inf")
    scored = []

    for idx, cand in enumerate(candidates):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        val = _objective(a, n, conv_fft_len)
        scored.append((val, a))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        print(f"[screen {idx:03d}] obj={val:.6f}")

    scored.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in scored[:8]]

    # deep optimization on small n
    for cid, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[optim start {cid}] launching deep optimisation")
        best_seq, best_val = _deep_optimize(
            a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val

    # optional upsample if time left
    if time.time() + 20 < deadline:

        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)

"""
Propose a step‑function (non‑negative sequence) that minimizes

        F(a) = 2·n·max(a∗a) / (∑a)² .

The algorithm consists of three stages:
1.  A diverse pool of initial vectors (uniform, power‑law, double‑exponential,
    Gaussian, sinusoid, Dirichlet draws and a warm‑start) is built and the best
    few are kept.
2.  Each kept start is refined by a two‑phase optimiser:
    – Adam on a smooth‑max (log‑sum‑exp) surrogate,
    – a mirror‑descent sub‑gradient phase (exact max).
    The optimiser works on a moderate support (n=4096) and keeps the global
    best sequence.
3.  If time permits the best distribution is interpolated to a larger grid
    (n=8192) and the optimiser is run again.  Afterwards a cheap additive
    projected‑gradient descent and a stochastic local‑search are performed to
    fine‑tune the solution.

All numbers are forced to be non‑negative; the final list is clipped to the
range required by `evaluate_sequence`.  The routine respects the supplied
time‑budget (default 1000 s) and returns the best sequence it found.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, ∑x=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective (sum(a) assumed = 1)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Two‑phase optimiser:
      – Adam on a smooth‑max surrogate (log‑sum‑exp),
      – mirror‑descent sub‑gradient on the exact max.
    The routine updates and returns the best sequence/value encountered.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # initialise soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5          # temperature for smooth‑max
    lam_factor = 1.018

    last_improve = time.time()
    it = 0

    # ----------------------- Adam phase -----------------------
    while time.time() < deadline:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (useful for logging)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp formulation)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (via chain rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase temperature (approach true max)
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter when stagnating
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        if it > 500_000:
            break

    # -------------------- Sub‑gradient phase --------------------
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total

    eta0 = 0.6
    sub_last_improve = time.time()
    for sub_it in range(30_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[n={n} Subgrad iter={sub_it:05d}] new best {val:.6f}")

        # active‑set mask (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (mirror‑descent on the simplex)
        eta = eta0 / np.sqrt(sub_it + 1)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                 # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 6.0:
            break

    return global_best_seq, global_best_val


def _generate_candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Construct a diverse set of start vectors (including a warm‑start)."""
    pool = []

    # warm‑start and cyclic shifts
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # uniform baseline
    pool.append(np.full(n, 1.0 / n))

    # power‑law edge‑concentrated shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)          # i·(n‑i+1)
    for alpha in np.linspace(-2.0, -0.2, 15):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # double‑exponential (two‑sided Laplace) families
    i = np.arange(n, dtype=np.float64)
    for lam in np.linspace(0.1, 3.0, 15):
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        pool.append(shape)

    # Gaussian bumps centred in the middle (width sweep)
    for sigma in np.linspace(n * 0.03, n * 0.4, 10):
        shape = np.exp(-0.5 * ((i - n / 2) ** 2) / (sigma ** 2))
        shape /= shape.sum()
        pool.append(shape)

    # single half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # random Dirichlet draws (wide range of concentrations)
    for alpha in np.logspace(-2, 0, 12):
        shape = np.random.dirichlet(np.full(n, alpha))
        pool.append(shape)

    return pool


def compute_obj(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast objective assuming sum(a)=1."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


def refine_by_proj_grad(
    a: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Additive projected‑gradient descent on the exact max‑convolution.
    Uses Euclidean projection onto the simplex after each step.
    """
    current = a.copy()
    best_seq = a.copy()
    best = best_val
    it = 0
    alpha0 = 0.5

    while time.time() < deadline:
        conv = _full_convolution(current, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if not active.any():
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, current, corr_fft_len)

        alpha = alpha0 / np.sqrt(it + 1.0)
        new = current - alpha * grad
        new = _project_simplex(new)

        # evaluate new point
        conv_new = _full_convolution(new, conv_fft_len)
        max_c_new = conv_new.max()
        val_new = 2.0 * n * max_c_new

        if val_new < best:
            best = val_new
            best_seq = new.copy()
            current = new.copy()
            print(f"[projgrad iter {it}] new best {val_new:.6f}")
        else:
            current = new

        it += 1
        # safety break if we are about to run out of time
        if time.time() + 0.1 > deadline:
            break

    return best_seq, best


def local_search(
    a: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing: move a random amount of mass from a
    donor index to a recipient index while keeping the vector on the simplex.
    Accept only improving moves.
    """
    current = a.copy()
    best_seq = a.copy()
    best = best_val
    it = 0
    delta = 0.2  # relative size of the transferred chunk

    while time.time() < deadline:
        # pick donor proportionally to its mass (guarantees a[i]>0)
        donor = np.random.choice(n, p=current)
        if current[donor] <= 1e-14:
            continue
        # pick a different recipient uniformly
        rec = np.random.randint(n - 1)
        if rec >= donor:
            rec += 1

        # amount to transfer (≤ delta·current[donor])
        eps = min(current[donor], delta * current[donor]) * np.random.rand()
        if eps <= 0.0:
            continue

        cand = current.copy()
        cand[donor] -= eps
        cand[rec] += eps

        val = compute_obj(cand, n, conv_fft_len)
        if val < best:
            best = val
            best_seq = cand.copy()
            current = cand.copy()
            print(f"[local iter {it}] new best {val:.6f}")

        # slowly shrink the step size for fine‑tuning
        delta = max(delta * 0.9999, 1e-6)
        it += 1

        if time.time() + 0.1 > deadline:
            break

    return best_seq, best


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation metric.
    The routine respects the supplied time‑budget and returns the best sequence
    encountered as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # -------------------------------------------------------------
    # 1.   Small‑scale optimisation (n = 4096)
    # -------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # warm‑start from a previous construction, if available
    warm_start = globals().get("height_sequence_1")
    warm = None
    if warm_start is not None:
        w = np.asarray(warm_start, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]), mode="constant")
        w = np.maximum(w, 0.0)
        s = w.sum()
        if s >= 0.01:
            warm = w / s

    # diverse candidate pool
    candidates = _generate_candidate_pool(n, warm)

    best_seq = None
    best_val = float("inf")
    scored = []  # (value, normalised vector)

    for idx, cand in enumerate(candidates):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        val = _objective(a, n, conv_fft_len)
        scored.append((val, a))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        print(f"[screen {idx:03d}] obj={val:.6f}")

    # keep the most promising starts (up to eight)
    scored.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in scored[:8]]

    # deep optimisation passes on the small grid
    for cid, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[optim start {cid}] launching deep optimisation")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,

        )

    # -------------------------------------------------------------
    # 2.   Upsample to a larger grid (n = 8192) if time permits
    # -------------------------------------------------------------
    if time.time() + 20.0 < deadline:

        conv_fft_len_big = _next_pow2(2 * n_big - 1)

            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("[big] starting optimisation on enlarged grid")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )

        # ---------------------------------------------------------
        # 3.   Fine‑tune the large‑scale solution
        # ---------------------------------------------------------

        if time.time() + 10.0 < deadline:
            print("[refine] projected gradient descent")

                n_big,

    # Clip to evaluator's allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return best_seq.tolist()
