def _simplex_projection(v):
    # ensures v >=0 and sum(v)=1
    # using algorithm from https...
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a non‑negative step‑function (list of heights) that lowers
F(a)=2·n·max(a∗a)/(sum(a)²).  The routine combines

* a large diverse pool of cheap random starts (Dirichlet, Beta‑shapes,
  random spikes, sinusoid, uniform and a warm‑start if available);
* quick selection of the best few candidates;
* a three‑phase local optimisation on the full resolution:
    - Adam on a smooth‑max (log‑sum‑exp) surrogate;
    - exponentiated‑gradient sub‑gradient descent;
    - additive gradient descent with Euclidean projection onto the simplex;
* a final cheap hill‑climbing that moves mass from the most harmful
  coordinates to the least harmful ones;
* optional up‑sampling to a larger grid if time permits.

All heavy work is done with FFT‑based convolutions (O(n log n)).
The code respects the supplied time budget (default 1000 s) and always
returns a valid sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for sub‑gradient / gradient.
    Returns a vector of length len(a):
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find rho
    rho = ind[u - cssv / ind > 0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho - 1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective -------------------------------------------------------------

def _objective(seq: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Compute F(a) = 2·n·max(a∗a) / (sum(a)²) for a given sequence.
    Returns np.inf for invalid (too small) sequences.
    """
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    conv = _full_convolution(seq, conv_fft_len)
    max_c = conv.max()
    return 2.0 * n * max_c / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Optimisers ------------------------------------------------------------

def _adam_optimize(
        a0: np.ndarray,
        n: int,
        conv_fft_len: int,
        corr_fft_len: int,
        deadline: float,
        max_iters: int = 120_000) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns best vector encountered (sum≈1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5          # start temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective(best_a, n, conv_fft_len)
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:05d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter when time is running out
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1
    return best_a, best_val


def _subgradient_optimize(
        a0: np.ndarray,
        n: int,
        conv_fft_len: int,
        corr_fft_len: int,
        deadline: float,
        max_iters: int = 120_000,
        eta: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated‑gradient) sub‑gradient descent on the
    exact max‑convolution objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it:05d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps a on simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1
    return best_a, best_val


def _additive_gradient_descent(
        a0: np.ndarray,
        n: int,
        conv_fft_len: int,
        corr_fft_len: int,
        deadline: float,
        max_iters: int = 150_000,
        init_lr: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Euclidean gradient descent with projection onto the simplex.
    Uses the exact max‑convolution sub‑gradient (the same w as in the
    exponentiated‑gradient phase).
    """
    a = np.maximum(a0, 0.0)
    a = a / a.sum()               # ensure simplex
    lr = init_lr
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[AddGrad it={it:05d}] new best {val:.6f}")

        # form sub‑gradient of max by averaging over active set
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # gradient step
        a = a - lr * grad_a
        a = _project_simplex(a)

        # simple decay of learning rate
        lr *= 0.99997
        it += 1
    return best_a, best_val


def _targeted_local_search(
        a0: np.ndarray,
        n: int,
        conv_fft_len: int,
        deadline: float,
        max_moves: int = 30_000,
        step_frac: float = 0.1) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing that moves a small fraction of mass from the
    coordinate with the largest gradient (most harmful) to the one with the
    smallest gradient (most beneficial).  Keeps moves that improve the
    exact objective.
    """
    a = a0.copy()
    best_val = _objective(a, n, conv_fft_len)
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # compute the sub‑gradient weights
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, conv_fft_len)   # reuse length

        i = np.argmax(grad)   # most harmful
        j = np.argmin(grad)   # most helpful
        if i == j or a[i] < 1e-12:
            # fallback to random move if gradient is flat
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j or a[i] < 1e-12:
                it += 1
                continue

        delta = min(a[i] * step_frac, a[i] * 0.5)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        val2 = _objective(a2, n, conv_fft_len)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            # occasional log
            if it % 5000 == 0:
                print(f"[LocalMove {it}] improved to {val2:.6f}")
        it += 1
    return best_a, best_val

def _make_warm_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    cand = []
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        s = w.sum()
        if s >= 0.01:
            w = w / s
            cand.append(w.copy())
            # rotated copies
            for shift in (n // 4, n // 2, 3 * n // 4):
                cand.append(np.roll(w, shift))
    return cand


def _random_dirichlet_candidates(n: int, count: int) -> list[np.ndarray]:
    return [np.random.dirichlet(np.ones(n)) for _ in range(count)]


def _beta_shape(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Generate a shape a_i ∝ (i+1)^alpha * (n-i)^beta, then normalise.
    The exponents may be negative.
    """
    i = np.arange(n, dtype=np.float64) + 1.0
    rev = (n - i + 1.0)
    shape = (i ** alpha) * (rev ** beta)
    shape = np.where(shape > 0, shape, 0.0)
    if shape.sum() == 0:
        shape = np.ones_like(shape)
    shape /= shape.sum()
    return shape


def _beta_candidates(n: int, count: int) -> list[np.ndarray]:
    cand = []
    for _ in range(count):
        alpha = np.random.uniform(-3.0, 3.0)
        beta = np.random.uniform(-3.0, 3.0)
        cand.append(_beta_shape(alpha, beta, n))
    return cand


def _spike_candidates(n: int, count: int) -> list[np.ndarray]:
    """
    Generate random sparse ‘spike’ candidates.  For each candidate,
    choose a random number k∈[16,256] of distinct positions and put
    equal mass on them.
    """
    out = []
    for _ in range(count):
        k = np.random.randint(16, 257)
        idx = np.random.choice(n, size=k, replace=False)
        a = np.zeros(n, dtype=np.float64)
        a[idx] = 1.0 / k
        out.append(a)
    return out


def _deterministic_shapes(n: int) -> list[np.ndarray]:
    """A few hand‑crafted shapes (sinusoid, triangular, half‑cosine, etc.)."""
    cand = []

    # half‑sinusoid (already used but keep)
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    cand.append(s)

    # triangular (peak in middle)
    t = np.arange(1, n + 1, dtype=np.float64)
    t = np.minimum(t, n - t + 1)
    t = t / t.sum()
    cand.append(t)

    # cosine‑window (positive half‑period)
    c = np.cos(np.pi * (np.arange(n) - n / 2) / n)
    c = np.maximum(c, 0.0)
    c /= c.sum()
    cand.append(c)

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    return cand

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Choose the working resolution
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start, if any
    warm_raw = globals().get("height_sequence_1")
    warm_candidates = _make_warm_candidates(n, warm_raw)

    # ------------------------------------------------------------------
    # Build a large pool of cheap initial candidates
    pool = []
    pool.extend(warm_candidates)

    # deterministic hand‑crafted shapes
    pool.extend(_deterministic_shapes(n))

    # random Dirichlet (20)
    pool.extend(_random_dirichlet_candidates(n, 20))

    # random beta shapes (20)
    pool.extend(_beta_candidates(n, 20))

    # random sparse spikes (20)
    pool.extend(_spike_candidates(n, 20))

    # ------------------------------------------------------------------
    # Quick evaluation of all candidates, keep the best few
    evals = []
    for idx, cand in enumerate(pool):
        # Normalise to sum=1 (objective is scale‑invariant)
        c = np.maximum(cand, 0.0)
        s = c.sum()
        if s < 0.01:
            c = np.random.dirichlet(np.ones(n))
            s = c.sum()
        c = c / s
        val = _objective(c, n, conv_fft_len)
        evals.append((val, c))
        if idx % 25 == 0:
            print(f"[Init] evaluated {idx+1}/{len(pool)} candidates, current best {val:.6f}")

    # sort by value, keep top K
    evals.sort(key=lambda x: x[0])
    K = 5
    top_candidates = [c for (_, c) in evals[:K]]
    best_seq = top_candidates[0].copy()
    best_val = evals[0][0]
    print(f"[Init] best of pool: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Optimise each of the top candidates (short runs)
    for cid, cand in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        a0 = cand.copy()
        local_deadline = min(deadline - 5.0, time.time() + 25.0)

        # Adam phase (smooth‑max)
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, local_deadline, max_iters=80_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[cand {cid}] Adam improved to {val_adam:.6f}")

        # Exponentiated‑gradient sub‑gradient phase
        if time.time() < deadline:
            a_sub, val_sub = _subgradient_optimize(
                best_seq, n, conv_fft_len, corr_fft_len, deadline, max_iters=60_000
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[cand {cid}] Subgrad improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Long additive‑gradient refinement on the current best
    if best_seq is not None and time.time() < deadline - 5:
        a_add, val_add = _additive_gradient_descent(
            best_seq, n, conv_fft_len, corr_fft_len, deadline, max_iters=200_000, init_lr=0.05
        )
        if val_add < best_val:
            best_val = val_add
            best_seq = a_add.copy()
            print(f"[AddGrad] refined best to {val_add:.6f}")

    # ------------------------------------------------------------------
    # Targeted cheap local search (very cheap moves)
    if best_seq is not None and time.time() < deadline - 5:
        a_loc, val_loc = _targeted_local_search(
            best_seq, n, conv_fft_len, deadline, max_moves=40_000, step_frac=0.1

        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        a_big_adam, val_big_adam = _adam_optimize(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big,
            deadline, max_iters=80_000
        )
        # sub‑gradient on large grid
        a_big_sub, val_big_sub = _subgradient_optimize(
            a_big_adam, n_big, conv_fft_len_big, corr_fft_len_big,
            deadline, max_iters=60_000
        )
        if val_big_sub < best_val:
            best_val = val_big_sub
            best_seq = a_big_sub.copy()
            print(f"[Big] up‑sampled best improved to {val_big_sub:.6f}")

    # ------------------------------------------------------------------
    # Fallback – ensure we always return something valid
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator's accepted range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
