```
def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    best_seq = None
    best_val = float('inf')
    # length schedule
    lengths = [1024, 2048, 4096, 8192, 16384, 32768, 65536]
    for n in lengths:
        if time.time()>deadline:
            break
        # compute fft lengths
        conv_len = 1 << int(np.ceil(np.log2(2*n-1)))
        corr_len = 1 << int(np.ceil(np.log2(3*n-2)))
        # generate seeds
        seeds = []
        # uniform
        seeds.append(np.full(n, 1.0/n))
        # previous best upsample/downsample
        if best_seq is not None:
            old = np.asarray(best_seq)
            if len(old)!=n:
                # interpolate
                x_old = np.linspace(0,1,old.shape[0])
                x_new = np.linspace(0,1,n)
                up = np.interp(x_new, x_old, old)
                up = up/ up.sum()
                seeds.append(up)
            else:
                seeds.append(old.copy())
        # random Dirichlet with various concentration
        for alpha in [0.3, 0.6, 1.0, 2.0]:
            seeds.append(np.random.dirichlet(np.full(n, alpha)))
        # power law edges
        idx = np.arange(1, n+1, dtype=float)
        prod = idx * (n - idx + 1.0)
        for alpha in [-3.0, -2.5, -2.0, -1.5]:
            shape = np.where(prod>0, np.power(prod, alpha), 0.0)
            if shape.sum()>0:
                shape = shape/shape.sum()
                seeds.append(shape)
        # log shape
        shape = 1.0/(np.arange(1,n+1)+np.arange(n,0,-1))
        shape = shape/shape.sum()
        seeds.append(shape)
        # two Gaussians at edges
        i = np.arange(n, dtype=float)
        for gamma in [0.05, 0.2, 0.5]:
            shape = np.exp(-gamma*i*i) + np.exp(-gamma*(n-1-i)**2)
            shape = shape/shape.sum()
            seeds.append(shape)

        # limit candidates
        max_candidates = 10 if n <= 8192 else 6
        seeds = seeds[:max_candidates]

        # For each seed run optimization
        for cand_idx, cand in enumerate(seeds):
            if time.time()>deadline:
                break
            # Phase 1: Adam with smooth-max
            a_adam, val_adam = _adam_optimize(cand, n, conv_len, corr_len, deadline)
            # Phase 2: exponentiated gradient refinement
            a_ref, val_ref = _subgradient_refine(a_adam, n, conv_len, corr_len, deadline)

            # Keep best
            if val_ref < best_val:
                best_val = val_ref
                best_seq = a_ref.copy()
                print(f"[n={n} cand={cand_idx}] new best {best_val:.6f}")
            elif val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n} cand={cand_idx} (adam)] new best {best_val:.6f}")

        # After candidates, perform local jitter refinement if best seq matches length
        if best_seq is not None and len(best_seq)==n:
            # jitter loops
            for jitter in range(10):
                if time.time()>deadline:
                    break
                # small logit perturbation
                logits = np.log(np.maximum(best_seq, 1e-12))
                logits += np.random.normal(scale=0.02, size=n)
                a0 = np.exp(logits - np.max(logits))
                a0 = a0/a0.sum()
                a_jit, val_jit = _adam_optimize(a0, n, conv_len, corr_len, deadline)
                a_jit, val_jit = _subgradient_refine(a_jit, n, conv_len, corr_len, deadline)
                if val_jit < best_val:

                    best_seq = a_jit.copy()
                    print(f"[n={n} jitter={jitter}] new best {best_val:.6f}")

        # final random mass-transfer local search for large n
        if best_seq is not None and len(best_seq)==n:

            for step in range(2000):
                if time.time()>deadline:

                max_c = conv.max()

                    if np.random.rand() < 0.001:

            print(f"[n={n}] after local swaps best {best_val:.6f}")

    if best_seq is None:
        best_seq = np.full(1024, 1.0/1024)

    # store globally for next runs
    globals()["height_sequence_1"] = best_seq.tolist()

    return best_seq.tolist()
```

assistantfinal**Solution Overview**

The optimizer searches for a non‑negative step‑function `a` that minimizes  

\[
F(a)=\frac{2\,n\,\max (a*a)}{(\sum a)^2},
\]

where `*` is the linear convolution.  
Because the objective is homogeneous we fix `∑a = 1`; the problem then
reduces to minimizing `max(a*a)` on the probability simplex.  
The function is convex, so a combination of smooth‑max gradient descent,
exact ℓ∞‑sub‑gradient updates (exponentiated‑gradient), and a rich set of
initialisation schemes works well in practice.

The algorithm proceeds in several stages:

1. **Length schedule** – start from modest lengths (`1024…65536`) and
   iteratively up‑sample the current best solution, allowing the shape
   to refine on finer grids.
2. **Diverse seeds** – for each length we generate many candidates:
   uniform, up‑sampled previous best, Dirichlet draws with several
   concentration parameters, edge‑concentrated power‑law shapes,
   harmonic, and two‑Gaussian edge shapes.
3. **Phase 1 – Adam on a smooth‑max surrogate** – we minimise the
   log‑sum‑exp of the convolution (`smooth_max`) using Adam while keeping
   the vector on the simplex via a soft‑max parametrisation.
4. **Phase 2 – exact ℓ∞ sub‑gradient refinement** – an exponentiated‑gradient
   (mirror‑descent) step using the true ℓ∞‑subgradient.
5. **Local jitter & mass‑transfer search** – after the best vector for a
   given length is found we perform several stochastic perturbations
   (logit jitter + re‑optimisation) and a cheap random‑swap local search
   to escape possible plateaus.
6. **Best‑so‑far bookkeeping** – the global best sequence and its value
   are kept across all lengths; the final sequence is also stored in the
   global variable `height_sequence_1` for future runs.

All heavy work (convolution, cross‑correlation) uses FFT‑based
implementations (`np.fft.rfft`), which are fast even for `n = 65536`.
Time‑budget checks (`deadline`) guarantee that the routine returns before
the 1000 s limit.

```python
"""
Propose a non‑negative step‑function (probability vector) that minimises
    F(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

Algorithm:
  • Process a schedule of lengths, always up‑sampling the current best.
  • Build a heterogeneous pool of seeds (uniform, Dirichlet, edge‑law,
    harmonic, two‑Gaussian etc.).
  • Phase‑1: Adam on a smooth‑max (log‑sum‑exp) surrogate of the convolution.
  • Phase‑2: Exact ℓ∞ sub‑gradient refinement via exponentiated‑gradient.
  • After each length perform jittering and cheap random mass‑transfer
    local search.
  • Keep and return the best sequence (also stored in global variable
    height_sequence_1 for reuse).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length `fft_len`.
    Returns an array of length 2*len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) required for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]  for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns (best_a, best_val) where best_val = 2*n*max(conv) for the best
    iterate seen before `deadline`.
    """
    eps = 1e-12
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05

    # start with a gentle temperature, then anneal quickly
    lam = 0.5
    lam_factor = 1.03

    # iteration budget (more for small n)
    max_iters = int(80000 * 1024 / n) + 2000

    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max re‑parameterisation (simplex projection)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum if w_sum > 0 else np.full_like(w, 1.0 / w.size)

        # gradient of smooth‑max w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal temperature
        lam = min(lam * lam_factor, 1e9)

        # exact objective (sum(a)=1)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # occasional random jitter if stagnation occurs
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


def _subgradient_refine(a0: np.ndarray,
                        n: int,
                        conv_fft_len: int,
                        corr_fft_len: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Exact ℓ∞ sub‑gradient refinement using exponentiated‑gradient updates.
    Returns (best_a, best_val).  A few thousand iterations are usually
    sufficient for convergence.
    """
    a = a0 / a0.sum()
    best_val = float("inf")
    best_a = a.copy()
    eta = 0.45
    last_improve = time.time()

    for it in range(3000):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient step (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:          # numerical safety – fallback to a random simplex point
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        eta *= 0.9995                       # gentle decay
        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function that minimises `evaluate_sequence`.
    The routine respects a wall‑clock budget (`budget_s` seconds) and
    returns the best sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # Length schedule – start small, finish with a large grid
    lengths = [1024, 2048, 4096, 8192, 16384, 32768, 65536]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

        # --------------------------------------------------------------
        # Build a diverse pool of seeds
        # --------------------------------------------------------------
        seeds = []

        # 1) uniform
        seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

        # 2) up‑sample / down‑sample previous best (if any)
        if best_seq is not None:
            old = np.asarray(best_seq, dtype=np.float64)
            if old.shape[0] != n:
                x_old = np.linspace(0.0, 1.0, old.shape[0])
                x_new = np.linspace(0.0, 1.0, n)
                up = np.interp(x_new, x_old, old)
                if up.sum() > 0:
                    up = up / up.sum()
                else:
                    up = np.random.dirichlet(np.ones(n))
                seeds.append(up)
            else:
                seeds.append(old.copy())

        # 3) Dirichlet draws with several concentration parameters
        for conc in [0.3, 0.6, 1.0, 2.0]:
            seeds.append(np.random.dirichlet(np.full(n, conc)))

        # 4) Edge‑concentrated power‑law shapes
        idx = np.arange(1, n + 1, dtype=np.float64)
        prod = idx * (n - idx + 1.0)
        for alpha in [-3.5, -3.0, -2.5, -2.0, -1.5]:
            shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
            if shape.sum() > 0:
                shape = shape / shape.sum()
                seeds.append(shape)

        # 5) Harmonic-like shape
        shape = 1.0 / (np.arange(1, n + 1) + np.arange(n, 0, -1))
        shape = shape / shape.sum()
        seeds.append(shape)

        # 6) Two Gaussian bumps at the edges
        i = np.arange(n, dtype=np.float64)
        for gamma in [0.05, 0.15, 0.3, 0.8]:
            shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            shape = shape / shape.sum()
            seeds.append(shape)

        # 7) Sine‑squared bump (smooth interior)
        shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
        shape = shape / shape.sum()
        seeds.append(shape)

        # Limit the pool size (keeps the total time under control)
        max_cand = 12 if n <= 8192 else 6
        seeds = seeds[:max_cand]

        # --------------------------------------------------------------
        # Optimise every seed
        # --------------------------------------------------------------
        for cand_id, cand in enumerate(seeds):
            if time.time() > deadline:
                break

            try:
                # Phase 1 – Adam on smooth‑max
                a_adam, val_adam = _adam_optimize(
                    cand, n, conv_fft_len, corr_fft_len, deadline
                )

                # Phase 2 – exact ℓ∞ sub‑gradient refinement
                a_ref, val_ref = _subgradient_refine(
                    a_adam, n, conv_fft_len, corr_fft_len, deadline
                )

                # Keep the best among both phases
                if val_ref < best_val:
                    best_val = val_ref
                    best_seq = a_ref.copy()
                    print(f"[n={n} cand={cand_id}] new global best {best_val:.6f}")
                elif val_adam < best_val:
                    best_val = val_adam
                    best_seq = a_adam.copy()
                    print(f"[n={n} cand={cand_id} (Adam)] new global best {best_val:.6f}")
            except Exception as exc:
                print(f"[n={n} cand={cand_id}] exception: {exc}")

        # --------------------------------------------------------------
        # Local jitter + refined search around the current best of this length
        # --------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            # Small logit jitter followed by a full two‑phase optimisation
            for jitter_id in range(8):
                if time.time() > deadline:
                    break
                logits = np.log(np.maximum(best_seq, 1e-12))
                logits += np.random.normal(scale=0.02, size=n)
                a0 = np.exp(logits - np.max(logits))
                a0 = a0 / a0.sum()

                a_jit, val_jit = _adam_optimize(
                    a0, n, conv_fft_len, corr_fft_len, deadline
                )
                a_jit, val_jit = _subgradient_refine(
                    a_jit, n, conv_fft_len, corr_fft_len, deadline
                )
                if val_jit < best_val:
                    best_val = val_jit
                    best_seq = a_jit.copy()
                    print(f"[n={n} jitter={jitter_id}] new global best {best_val:.6f}")

            # Cheap random mass‑transfer local search (exploit convexity)
            a_local = best_seq.copy()
            for step in range(2000):
                if time.time() > deadline:
                    break
                i1, i2 = np.random.randint(0, n, size=2)
                if i1 == i2:
                    continue
                delta = 0.01 * a_local[i1] * np.random.rand()
                a_new = a_local.copy()
                a_new[i1] -= delta
                a_new[i2] += delta

                conv = _full_convolution(a_new, fft_len=conv_fft_len)
                val = 2.0 * n * conv.max()
                if val < best_val:
                    best_val = val
                    best_seq = a_new.copy()
                    a_local = a_new
                # occasional probabilistic acceptance to avoid stagnation
                elif np.random.rand() < 0.001:
                    a_local = a_new

            print(f"[n={n}] post‑local‑search best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback safety net
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Store the result globally for future runs
    globals()["height_sequence_1"] = best_seq.tolist()

    # Return a plain Python list (required by the external evaluator)
    return best_seq.tolist()
```
