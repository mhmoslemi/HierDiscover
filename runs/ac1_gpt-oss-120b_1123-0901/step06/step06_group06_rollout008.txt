"""
Proposal:  Fast multi‑family initialization + two‑phase Adam / EG optimiser.

The function searches for a non‑negative vector a (step‑function heights) that
minimises

        F(a) = 2·n·max(a*a) / (sum(a)²) ,

where a*a denotes the linear convolution (numpy.convolve).  The scaling of a
does not affect the objective, therefore the optimiser works on the simplex
(sum(a)=1).  The algorithm proceeds as follows:

1.  A diversified pool of seed vectors is built for each length n.
    •  Simple parametric families (Beta, power‑law, exponential, Gaussian,
       cosine / sine windows, truncated arcsine) with several parameter
       settings.
    •  Random Dirichlet draws (different concentrations).
    •  The previously best sequence (global variable ``height_sequence_1``)
       resized to the current length.

2.  For every seed a run two optimisation phases:
    – Adam on a smooth‑max surrogate of the true max, i.e.
          Sλ(a) = (1/λ)·log Σ exp(λ·conv(a)).
      The temperature λ is increased exponentially so that the surrogate
      approaches the true max.
    – A short exponentiated‑gradient (sub‑gradient) phase that works on the
      exact max using the active set of indices attaining the maximum.

   Both phases keep the solution on the simplex via the log‑softmax
   parametrisation (logits).  The best vector seen so far (global best)
   is updated after each iteration.

3.  The search respects a user‑provided time budget (default 1000 s) and
    returns the best sequence discovered as a plain Python list.

The implementation uses FFT‑based convolutions (O(n log n)) and checks the
deadline frequently so that it never exceeds the allowed runtime.

# FFT based linear convolution a*a and the cross‑correlation needed for
# the gradient of the smooth‑max surrogate.
def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns an array of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

# Helpers for building families of initial candidates.
def _build_beta_shapes(n: int) -> list[np.ndarray]:
    """Beta‑distribution shaped vectors (a_i ∝ i^{α-1}(n‑i)^{β-1})."""
    i = np.arange(1, n + 1, dtype=np.float64)
    cand = []
    # a few (α,β) pairs that give U‑, bell‑ and edge‑concentrated shapes
    for alpha, beta in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5),
                        (2.0, 2.0), (3.0, 3.0), (5.0, 1.0), (1.0, 5.0)]:
        shape = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
        if shape.sum() == 0:
            continue
        shape = shape / shape.sum()
        cand.append(shape)
    return cand


def _build_power_shapes(n: int) -> list[np.ndarray]:
    """Shapes a_i ∝ (i·(n‑i+1))^{α} for a range of α (including negative)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    cand = []
    for alpha in np.linspace(-2.0, -0.2, 9):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() == 0:
            continue
        shape = shape / shape.sum()
        cand.append(shape)
    return cand


def _build_exponential_shapes(n: int) -> list[np.ndarray]:
    """Two‑sided exponential decay a_i = exp(-γ·i)+exp(-γ·(n‑1‑i))."""
    cand = []
    for gamma in (0.0005, 0.001, 0.002, 0.004, 0.008):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape = shape / shape.sum()
        cand.append(shape)
    return cand


def _build_gaussian_shapes(n: int) -> list[np.ndarray]:
    """Discrete Gaussian centred in the middle."""
    cand = []
    i = np.arange(n, dtype=np.float64)
    for sigma in (n / 10.0, n / 8.0, n / 6.0, n / 4.0):
        gauss = np.exp(-0.5 * ((i - (n - 1) / 2) / sigma) ** 2)
        if gauss.sum() == 0:
            continue
        gauss = gauss / gauss.sum()
        cand.append(gauss)
    return cand


def _build_window_shapes(n: int) -> list[np.ndarray]:
    """Cosine, half‑sine and tent windows."""
    i = np.arange(n, dtype=np.float64)
    cand = []

    # cosine window (single bump)
    coswin = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    coswin = coswin / coswin.sum()
    cand.append(coswin)

    # half‑sine window (zero at both ends)
    sinwin = np.sin(np.pi * i / (n - 1))
    sinwin = sinwin / sinwin.sum()
    cand.append(sinwin)

    # symmetric tent (linear rise then fall)
    left = np.linspace(0, 1, n // 2, endpoint=False)
    right = left[::-1]
    tent = np.concatenate([left, right])
    if tent.shape[0] < n:   # odd length – add centre zero
        tent = np.concatenate([tent, [0.0]])
    tent = np.maximum(tent, 0.0)
    tent = tent / tent.sum()
    cand.append(tent)

    return cand


def _build_arcsine_shape(n: int) -> list[np.ndarray]:
    """Truncated arcsine (inverse‑sqrt at the edges)."""
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcs = arcs / arcs.sum()
    return [arcs]


def _build_random_dirichlet(n: int, count: int = 4) -> list[np.ndarray]:
    """Random Dirichlet vectors with a few concentration values."""
    cand = []
    concentrations = [0.2, 0.5, 1.0, 2.0, 5.0]
    for _ in range(count):
        conc = np.random.choice(concentrations)
        vec = np.random.dirichlet(np.full(n, conc))
        cand.append(vec)
    return cand


def _resize_sequence(seq: np.ndarray, n: int) -> np.ndarray:
    """Resize a sequence to length n by linear interpolation."""
    if seq.shape[0] == n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, seq.shape[0])
    x_new = np.linspace(0.0, 1.0, n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:
        new_seq = np.random.dirichlet(np.ones(n))
    else:
        new_seq = new_seq / s
    return new_seq

    F(a)=2·n·max(a*a)/(sum(a)²).  The function respects the time budget and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + max(0.0, budget_s - 0.5)  # safety margin

    best_seq = None
    best_val = float("inf")

    # Lengths to explore – larger n give a lower theoretical bound.
    lengths = [1024, 2048, 4096, 8192]

    # Pre‑compute FFT lengths (next power of two) for each n.
    fft_len_conv = {}
    fft_len_corr = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_len_conv[n] = 1 << int(np.ceil(np.log2(conv_len)))
        # Corr needs conv_len + n - 1 points (linear correlation)
        fft_len_corr[n] = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_len = fft_len_conv[n]
        corr_len = fft_len_corr[n]

        # ------------------------------------------------------------------
        # Build the pool of initial candidates for the current length.
        candidates = []

        # 1) Previous best (if any) resized to the current length.
        if best_seq is not None:
            a_prev = np.asarray(best_seq, dtype=np.float64)
            cand = _resize_sequence(a_prev, n)
            candidates.append(cand)

        # 2) Global variable from the previous run, if present.
        if "height_sequence_1" in globals():
            h = np.asarray(height_sequence_1, dtype=np.float64)
            cand = _resize_sequence(h, n)
            candidates.append(cand)

        # 3) Parametric families.
        candidates.extend(_build_beta_shapes(n))
        candidates.extend(_build_power_shapes(n))
        candidates.extend(_build_exponential_shapes(n))
        candidates.extend(_build_gaussian_shapes(n))
        candidates.extend(_build_window_shapes(n))
        candidates.extend(_build_arcsine_shape(n))

        # 4) Random Dirichlet draws.
        candidates.extend(_build_random_dirichlet(n, count=6))

        # ------------------------------------------------------------------
        # Optimise each seed independently.
        for cid, seed_vec in enumerate(candidates):
            if time.time() > deadline:
                break

            # Normalise to the simplex.
            a = np.maximum(seed_vec, 0.0)
            s = a.sum()
            if s < 1e-12:
                continue
            a = a / s

            # Log‑softmax parametrisation.
            eps = 1e-15
            logits = np.log(a + eps)

            # ---------- Phase 1 – Adam on smooth‑max ----------
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

            # Learning‑rate scales mildly with problem size.
            base_lr = 0.12 * np.sqrt(2048.0 / n)

            # Temperature for the smooth‑max (starts small and grows).
            lam = 0.25
            lam_factor = 1.02  # aggressive growth

            # Iteration budget – more iterations for smaller n.
            max_iters = int(40000 * 1024.0 / n) + 5000
            last_improve_time = time.time()

            for t in range(1, max_iters + 1):
                # Softmax → a on the simplex.
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                # Convolution and smooth‑max weights.
                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # Gradient of the surrogate (linear in w).
                grad_a = 2.0 * _corr_fft(w, a, corr_len)

                # Back‑prop through the soft‑max.
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update.
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= base_lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                # Increase temperature.
                lam = min(lam * lam_factor, 1e6)

                # Exact objective (sum(a)=1 ⇒ denominator = 1).
                val = 2.0 * n * max_c

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve_time = time.time()
                    print(f"[n={n} cand={cid} Adam it={t}] new best {val:.6f}")

                # Gentle random shake if we stall for a few seconds.
                if time.time() - last_improve_time > 7.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve_time = time.time()

                if time.time() > deadline:
                    break

            # ---------- Phase 2 – Exact max (sub‑gradient) ----------
            a = a / a.sum()
            eta = 0.55
            sub_last_improve = time.time()
            sub_iters = 6000

            for sub_it in range(sub_iters):
                if time.time() > deadline:
                    break

                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                # Indices where the max is attained (tiny tolerance).
                active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
                w = np.zeros_like(conv)
                if active.any():
                    w[active] = 1.0 / active.sum()

                grad_a = 2.0 * _corr_fft(w, a, corr_len)

                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:                     # protect against collapse
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                val = 2.0 * n * max_c

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last_improve = time.time()
                    print(f"[n={n} cand={cid} Subgrad it={sub_it}] new best {val:.6f}")

                if time.time() - sub_last_improve > 8.0:
                    break

        # ---------- Global fine‑tuning for the current best of this length ----------
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] fine‑tuning global best (val={best_val:.6f})")
            a = best_seq.copy()
            logits = np.log(a + 1e-15)

            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

            lr = 0.045 * np.sqrt(2048.0 / n)
            lam = 0.4
            lam_factor = 1.012
            last_improve = time.time()
            fine_iters = int(60000 * 1024.0 / n) + 5000

            for t in range(1, fine_iters + 1):
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

                grad_a = 2.0 * _corr_fft(w, a, corr_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                lam = min(lam * lam_factor, 1e6)

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} fine it={t}] new best {val:.6f}")

                if time.time() - last_improve > 10.0:
                    logits += (np.random.rand(n) - 0.5) * 5e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

    # ------------------------------------------------------------------
    # Fallback – uniform vector (should never be needed).
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list – the external evaluator expects a list.
    return best_seq.tolist()
