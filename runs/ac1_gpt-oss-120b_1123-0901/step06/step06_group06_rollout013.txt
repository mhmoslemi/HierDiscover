"""
Propose a non‑negative step‑function minimizing
        F(a) = 2·n·max(a*a) / (sum(a)²)

Algorithm
---------
* **Broad random search** – for each target length *n* we draw many
  Dirichlet‑distributed vectors with a wide range of concentration
  parameters (including very spiky and very flat shapes).  The fast
  FFT‑based convolution `_conv_fft` gives an inexpensive estimate of the
  objective (sum(a) is 1, so F = 2·n·max(conv)).
* **Quick local polishing** – the current global best for a given length
  is refined with a short Adam optimisation on a smooth‑max surrogate.
* **Exact sub‑gradient phase** – after Adam we run a few exponentiated‑
  gradient steps that use the true max‑convolution (active set) as a
  sub‑gradient.
* **Final L‑BFGS fine‑tuning** – the best sequence overall is handed to
  SciPy’s L‑BFGS‑B optimiser on the same smooth‑max (large temperature)
  to push the maximum even lower.
* **Time‑budget aware** – the routine checks the deadline frequently and
  aborts gracefully, always returning the best sequence found so far.

The code uses only NumPy (and SciPy if available) and respects the
1000‑second budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a sequence of heights with a low evaluation score."""
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for the caller
    deadline = start_time + budget_s - 0.5

    best_seq = None          # numpy array, normalised to sum==1
    best_val = float("inf")  # current best value of F(a)

    # ------------------------------------------------------------------
    # Lengths we will try. Larger sizes give a lower theoretical bound,
    # but also cost more time; we stop when the deadline is near.
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Exploring length n={n} ===")

        # pre‑compute FFT lengths (next powers of two)
        conv_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_len = 1 << int(np.ceil(np.log2(2 * n - 1 + n - 1)))

        # ------------------------------------------------------------------
        # 1) RANDOM SEARCH -------------------------------------------------
        # Number of random draws adapts to remaining time and problem size.
        remaining = deadline - time.time()
        # Empirical guess: about 0.001 s per FFT for n≈4096 – be generous.
        max_draws = int(min(20000, max(2000, remaining / (0.001 * n / 1024))))
        print(f"[n={n}] random draws: {max_draws}")

        for draw in range(max_draws):
            # concentration uniformly sampled in log‑space (0.01 … 10)
            conc = 10.0 ** np.random.uniform(-2.0, 1.0)
            a = np.random.dirichlet(np.full(n, conc))

            # fast exact objective (sum==1 → F = 2·n·max(conv))
            conv = _conv_fft(a, conv_len)
            val = 2.0 * n * conv.max()

            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[n={n} random {draw+1}/{max_draws}] new best {val:.6f}")

            if time.time() > deadline:
                break

        # ------------------------------------------------------------------
        # 2) OPTIONAL warm‑start from a previously stored solution ----------
        if best_seq is not None and best_seq.shape[0] == n:
            # already have a candidate of the correct length – keep it
            pass
        elif "height_sequence_1" in globals():
            # try to reuse the historic best (cut / pad as needed)
            h = np.asarray(height_sequence_1, dtype=np.float64)
            if h.shape[0] > n:
                h = h[:n]
            else:
                h = np.pad(h, (0, n - h.shape[0]))
            h = np.maximum(h, 0.0)
            if h.sum() > 1e-8:
                a = h / h.sum()
                conv = _conv_fft(a, conv_len)
                val = 2.0 * n * conv.max()
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[n={n}] warm‑start from height_sequence_1 gives {val:.6f}")

        # ------------------------------------------------------------------
        # 3) QUICK ADAM REFINEMENT ON THE CURRENT BEST OF THIS length -----
        if best_seq is not None and best_seq.shape[0] == n:
            a = best_seq.copy()
            logits = np.log(a + 1e-15)          # work in the soft‑max domain
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            # learning‑rate scaled mildly with n
            lr = 0.07 * np.sqrt(2048.0 / n)
            lam = 0.2
            lam_factor = 1.02
            last_improve = time.time()
            max_iters = int(15000 * 1024 / n) + 2000

            for t in range(1, max_iters + 1):
                # ---- simplex projection via soft‑max -----------------------
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                # ---- smooth‑max of the convolution -------------------------
                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # ---- gradient of the surrogate -----------------------------
                grad_a = 2.0 * _corr_fft(w, a, corr_len)

                # back‑prop through the soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # ---- Adam update -------------------------------------------
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # temperature schedule – approach true max
                lam = min(lam * lam_factor, 1e6)

                # current exact objective (sum==1)
                val = 2.0 * n * max_c
                if val < best_val - 1e-12:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} Adam iter={t}] new best {val:.6f}")

                # small random shake if stuck too long
                if time.time() - last_improve > 5.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()

                if time.time() > deadline:
                    break

        # ------------------------------------------------------------------
        # 4) EXACT SUB‑GRADIENT (exponentiated‑gradient) phase -------------
        if best_seq is not None and best_seq.shape[0] == n:
            a = best_seq.copy()
            eta = 0.55
            sub_last_improve = time.time()
            max_sub_iters = 4000

            for sub_it in range(max_sub_iters):
                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                # active set – all indices attaining the maximum (tiny tolerance)
                active = np.isclose(conv, max_c, atol=1e-12)
                w = np.zeros_like(conv)
                if active.any():
                    w[active] = 1.0 / active.sum()

                grad_a = 2.0 * _corr_fft(w, a, corr_len)
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:                     # protect against collapse
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                conv = _conv_fft(a, conv_len)
                val = 2.0 * n * conv.max()
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last_improve = time.time()
                    print(f"[n={n} subgrad {sub_it}] new best {val:.6f}")

                if time.time() - sub_last_improve > 5.0 or time.time() > deadline:
                    break

        # --------------------------------------------------------------
        # If the time budget is almost exhausted we stop early.
        if time.time() > deadline - 2.0:
            break

    # ----------------------------------------------------------------------
    # 5) GLOBAL FINETUNING WITH L‑BFGS (if SciPy is available) ----------
    if best_seq is not None:
        n = best_seq.shape[0]
        conv_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_len = 1 << int(np.ceil(np.log2(2 * n - 1 + n - 1)))

        try:
            from scipy.optimize import minimize

            def smooth_objective(logits_arr):
                """Smooth‑max surrogate (log‑sum‑exp) of max(conv)."""
                a = np.exp(logits_arr - np.max(logits_arr))
                a = a / a.sum()
                conv = _conv_fft(a, conv_len)
                lam_big = 5.0e4                # large temperature ≈ true max
                max_c = conv.max()
                w_sum = np.exp(lam_big * (conv - max_c)).sum()
                # smooth max = max_c + log(w_sum)/lam_big
                return max_c + np.log(w_sum) / lam_big

            def smooth_jacobian(logits_arr):
                a = np.exp(logits_arr - np.max(logits_arr))
                a = a / a.sum()
                conv = _conv_fft(a, conv_len)
                lam_big = 5.0e4
                max_c = conv.max()
                exp_term = np.exp(lam_big * (conv - max_c))
                w_sum = exp_term.sum()
                w_norm = exp_term / w_sum
                grad_a = 2.0 * _corr_fft(w_norm, a, corr_len)

                avg = np.dot(a, grad_a)
                # chain rule through the soft‑max
                return a * (grad_a - avg)

            init_log = np.log(best_seq + 1e-15)
            res = minimize(
                smooth_objective,
                init_log,
                method="L-BFGS-B",
                jac=smooth_jacobian,
                options={"maxiter": 200, "ftol": 1e-12, "disp": False},
            )
            a_opt = np.exp(res.x - np.max(res.x))
            a_opt = a_opt / a_opt.sum()
            # evaluate with the official evaluator (clamping etc.)
            final_val = evaluate_sequence(a_opt.tolist())
            if final_val < best_val:
                best_val = final_val
                best_seq = a_opt.copy()
                print(f"[final L‑BFGS] new best {final_val:.6f}")

        except Exception as exc:          # SciPy missing or optimiser failure
            print("L‑BFGS fine‑tuning skipped:", exc)

    # ----------------------------------------------------------------------
    # 6) FALLBACK – uniform vector if nothing was found (should never happen)
    if best_seq is None:
        best_seq = np.full(1024, 1.0 / 1024.0, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator.
    return best_seq.tolist()
