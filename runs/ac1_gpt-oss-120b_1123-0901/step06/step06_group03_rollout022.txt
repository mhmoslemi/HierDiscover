"""
Propose a candidate step‑function sequence minimizing
F(a)=2·n·max(a∗a)/(∑a)².

The search consists of:
1. A rich parametric pool (uniform, Gaussian, exponential, beta,
   power‑law, sinusoids) and a warm‑start from the previous best.
   All vectors are normalised to sum‑one and the best few are kept.
2. For each kept start a two‑phase optimiser is run:
   – Adam on a log‑sum‑exp smooth‑max surrogate of the convolution.
   – Exact max‑subgradient mirror‑descent refinement.
   Both phases respect the global deadline and keep the overall best.
3. A cheap stochastic local‑search (small mass transfers) further
   polishes the solution.
4. If time permits the best shape is interpolated to a finer grid
   (8192 points) and the whole optimisation is repeated.

All numbers are projected onto the feasible set (non‑negative,
≤1000) before returning.  The routine prints progress and returns a
plain Python list.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact evaluation of the objective (assumes sum(a)>0)."""
    conv = _full_convolution(a, fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

def _candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Generate a diverse set of normalized, non‑negative start vectors."""
    pool = []

    # Warm‑start and rotations
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Uniform
    pool.append(np.full(n, 1.0 / n))

    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    # Triangular / polynomial central shapes
    prod = (i + 1) * (n - i)
    for alpha in np.linspace(0.2, 2.0, 12):
        shape = prod ** alpha
        shape_sum = shape.sum()
        if shape_sum > 0:
            shape /= shape_sum
            pool.append(shape)

    # Symmetric exponential (Laplacian) shapes
    for lam in np.linspace(0.05, 4.0, 16):
        shape = np.exp(-lam * np.abs(i - mid))
        shape_sum = shape.sum()
        if shape_sum > 0:
            shape /= shape_sum
            pool.append(shape)

    # Gaussian bumps of varying width
    for sigma in np.linspace(0.02 * n, 0.35 * n, 12):
        shape = np.exp(-0.5 * ((i - mid) ** 2) / (sigma ** 2))
        shape_sum = shape.sum()
        if shape_sum > 0:
            shape /= shape_sum
            pool.append(shape)

    # Beta‑distribution‑like shapes (symmetric)
    for a_param in np.logspace(-1, 1, 10):
        b_param = a_param
        x = i / (n - 1)
        # avoid 0^negative by adding tiny epsilon
        eps = 1e-12
        shape = (np.maximum(x, eps) ** (a_param - 1)) * (np.maximum(1 - x, eps) ** (b_param - 1))
        shape_sum = shape.sum()
        if shape_sum > 0:
            shape /= shape_sum
            pool.append(shape)

    # Single half‑wave sinusoid (already used before)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # Random Dirichlet draws with a wide range of concentration
    for alpha in np.logspace(-2, 0, 16):
        shape = np.random.dirichlet(np.full(n, alpha))
        pool.append(shape)

    # Clip to non‑negative and normalise (just in case)
    cleaned = []
    for v in pool:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s < 1e-12:
            continue
        v = v / s
        cleaned.append(v)
    return cleaned

def _adam_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                   corr_fft_len: int, deadline: float,
                   global_best_seq: np.ndarray | None,
                   global_best_val: float) -> tuple[np.ndarray, float]:
    """Adam on a smooth‑max surrogate, returns the best vector/value seen."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # work in the log‑simplex space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04

    lam = 0.5                         # start temperature for smooth‑max
    lam_factor = 1.015                # geometric increase

    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        # ----- convert logits → probabilities (simplex) -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()   # a lives on the simplex, sum=1

        # ----- evaluate exact objective (for bookkeeping) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c  # denominator is 1 because sum(a)=1
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights -----
        w = np.exp(lam * (conv - max_c))          # stabilised by subtracting max_c
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of smooth‑max w.r.t a -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- chain rule through soft‑max (logits) -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, 1e7)

        # ----- occasional jitter when stuck -----
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        # safety break
        if it >= 400_000:
            break

    return global_best_seq, global_best_val

def _subgradient_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                          corr_fft_len: int, deadline: float,
                          global_best_seq: np.ndarray | None,
                          global_best_val: float) -> tuple[np.ndarray, float]:
    """Exact max‑subgradient mirror descent (exponentiated gradients)."""
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    eta0 = 0.6
    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < 80_000:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} iter={it:05d}] new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent step (exponentiated gradient)
        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)

        # re‑normalize onto simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # stop if no improvement for a while
        if time.time() - last_improve > 7.0:
            break

        it += 1

    return global_best_seq, global_best_val

def _local_refinement(a: np.ndarray, n: int, conv_fft_len: int,
                      deadline: float, current_best_val: float) -> tuple[np.ndarray, float]:
    """Tiny random mass transfers; accepts only improvements."""
    a_cur = a.copy()
    best_val = current_best_val
    it = 0
    while time.time() < deadline and it < 5000:
        # pick two distinct indices
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        # propose moving a fraction of mass from i to j
        if a_cur[i] <= 0:
            it += 1
            continue
        frac = np.random.rand() * 0.15  # up to 15 % of a[i]
        delta = a_cur[i] * frac
        a_new = a_cur.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # evaluate
        conv = _full_convolution(a_new, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val - 1e-12:
            best_val = val
            a_cur = a_new
            # print modest progress
            print(f"[Local] improvement to {best_val:.6f}")
            # reset stagnation timer indirectly via loop condition
        it += 1
    return a_cur, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function minimising the supplied
    evaluate_sequence metric.  The routine obeys the given time budget
    and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    # leave a small safety margin for Python bookkeeping
    deadline = time.time() + budget_s - 0.3

    # ----- primary grid size ------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ----- warm start -------------------------------------------------------
    warm_start = globals().get("height_sequence_1")
    warm = None
    if warm_start is not None:
        w = np.asarray(warm_start, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            warm = w / w.sum()
        else:
            warm = None

    # ----- candidate pool ----------------------------------------------------
    pool = _candidate_pool(n, warm)

    best_seq = None
    best_val = float("inf")
    scored = []

    # quick screening of the pool (no gradients, just objective)
    for idx, cand in enumerate(pool):
        # ensure proper normalisation
        a = cand
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        val = _objective(a, n, conv_fft_len)
        scored.append((val, a))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        if time.time() >= deadline:
            break
        print(f"[Screen {idx:03d}] val={val:.6f}")

    # keep the most promising start vectors (up to 8)
    scored.sort(key=lambda t: t[0])
    top_vectors = [vec for _, vec in scored[:8]]

    # ----- deep optimisation on the top starts ------------------------------
    for cid, start_vec in enumerate(top_vectors):
        if time.time() >= deadline:
            break
        print(f"[Main Optim {cid}] start optimisation")
        # Adam phase
        best_seq, best_val = _adam_optimize(
            start_vec, n, conv_fft_len, corr_fft_len,
            deadline, best_seq, best_val
        )
        # Sub‑gradient exact phase
        best_seq, best_val = _subgradient_optimize(
            best_seq if best_seq is not None else start_vec,
            n, conv_fft_len, corr_fft_len,
            deadline, best_seq, best_val
        )
        # Local refinement
        best_seq, best_val = _local_refinement(
            best_seq, n, conv_fft_len, deadline, best_val
        )
        print(f"[Optim {cid}] best so far {best_val:.6f}")

    # ----- random restarts while time permits --------------------------------
    while time.time() + 2.0 < deadline:
        # random Dirichlet with random concentration
        alpha = 10 ** np.random.uniform(-2.0, 0.0)
        a0 = np.random.dirichlet(np.full(n, alpha))
        # short optimisation (fewer iterations) to keep budget low
        # we reuse the Adam routine but with a tighter iteration cap;
        # we achieve this by reducing the deadline a bit.
        sub_deadline = min(deadline, time.time() + 6.0)
        seq_tmp, val_tmp = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len,
            sub_deadline, best_seq, best_val
        )
        seq_tmp, val_tmp = _subgradient_optimize(
            seq_tmp, n, conv_fft_len, corr_fft_len,
            sub_deadline, seq_tmp, val_tmp
        )
        # a quick local search can occasionally unlock extra gain
        seq_tmp, val_tmp = _local_refinement(
            seq_tmp, n, conv_fft_len, sub_deadline, val_tmp
        )
        if val_tmp < best_val:
            best_val = val_tmp
            best_seq = seq_tmp
            print(f"[Random restart] new global best {best_val:.6f}")

    # ----- optional up‑sampling to a finer grid -------------------------------
    if time.time() + 20.0 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # interpolate the current best onto the larger grid
        if best_seq is not None:
            x_old = np.linspace(0.0, 1.0, n, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("[Upsample] start optimisation on larger support")

        # run the same two‑phase optimiser on the enlarged grid (shorter budget)
        big_deadline = deadline
        best_seq_big, best_val_big = _adam_optimize(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big,
            big_deadline, None, float("inf")
        )
        best_seq_big, best_val_big = _subgradient_optimize(
            best_seq_big, n_big, conv_fft_len_big, corr_fft_len_big,
            big_deadline, best_seq_big, best_val_big
        )
        best_seq_big, best_val_big = _local_refinement(
            best_seq_big, n_big, conv_fft_len_big, big_deadline, best_val_big
        )
        print(f"[Upsample] best value on big grid {best_val_big:.6f}")

        # Keep the better of the two resolutions
        if best_val_big < best_val:
            best_val = best_val_big
            # down‑sample back to original size for the evaluator (optional)
            # simple averaging over blocks
            best_seq = np.mean(best_seq_big.reshape(-1, 2), axis=1)
            best_seq = np.maximum(best_seq, 0.0)
            if best_seq.sum() > 0:
                best_seq /= best_seq.sum()
            else:
                best_seq = np.full(n, 1.0 / n)

    # ----- final sanity checks ------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Return as plain Python list (float)
    return best_seq.tolist()

"""
Propose a candidate step‑function sequence minimizing

        F(a) = 2·n·max(a∗a) / (∑a)² .

The search proceeds in four stages:

1. **Rich candidate pool** – uniform, Gaussian, exponential,
   beta‑distribution, polynomial‑central, sinusoid and a warm‑start
   (if available) are generated, normalised and quickly screened.
   The best few are kept.

2. **Two‑phase optimisation** for each kept start:
   – Adam on a log‑sum‑exp smooth‑max surrogate of the convolution;
   – Exact max‑subgradient mirror‑descent (exponentiated‑gradient)
     to polish the solution.  Both respect the global deadline
     and constantly update the overall best.

3. **Stochastic local refinement** – tiny random mass transfers are
   accepted only if they improve the exact objective.

4. **Optional up‑sampling** – if time remains, the best shape is
   interpolated to a finer grid (8192 points) and the whole optimisation
   is repeated; the best of the original and the up‑sampled result is
   returned.

All numbers are forced onto the feasible set (non‑negative,
≤ 1000) before returning.  Progress is printed to stdout.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than ``x``."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)–1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact evaluation of the objective (assumes sum(a)>0)."""
    conv = _full_convolution(a, fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Build a diverse set of normalised, non‑negative start vectors."""
    pool = []

    # warm‑start and simple rotations
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # uniform baseline
    pool.append(np.full(n, 1.0 / n))

    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0

    # polynomial central shapes (prod = i·(n‑i))
    prod = (i + 1) * (n - i)
    for alpha in np.linspace(0.2, 2.0, 12):
        shape = prod ** alpha
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # symmetric exponential (double‑sided Laplace) families
    for lam in np.linspace(0.05, 4.0, 16):
        shape = np.exp(-lam * np.abs(i - mid))
        shape /= shape.sum()
        pool.append(shape)

    # Gaussian bumps of varying width
    for sigma in np.linspace(0.02 * n, 0.35 * n, 12):
        shape = np.exp(-0.5 * ((i - mid) ** 2) / (sigma ** 2))
        shape /= shape.sum()
        pool.append(shape)

    # Beta‑distribution‑like symmetric shapes
    for a_param in np.logspace(-1, 1, 10):
        b_param = a_param
        x = i / (n - 1)
        eps = 1e-12
        shape = (np.maximum(x, eps) ** (a_param - 1)) * \
                (np.maximum(1 - x, eps) ** (b_param - 1))
        shape /= shape.sum()
        pool.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # Random Dirichlet draws with a range of concentration parameters
    for alpha in np.logspace(-2, 0, 16):
        shape = np.random.dirichlet(np.full(n, alpha))
        pool.append(shape)

    # final clean‑up
    cleaned = []
    for v in pool:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s < 1e-12:
            continue
        cleaned.append(v / s)
    return cleaned


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                   corr_fft_len: int, deadline: float,
                   global_best_seq: np.ndarray | None,
                   global_best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smooth‑max surrogate.
    Returns the best sequence/value discovered (global across calls).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # work in log‑simplex space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04

    lam = 0.5                         # starting temperature
    lam_factor = 1.015                # geometric increase

    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for logging / global best)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t a of the smooth‑max
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max (logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # jitter if stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        if it >= 400_000:
            break

    return global_best_seq, global_best_val


# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                          corr_fft_len: int, deadline: float,
                          global_best_seq: np.ndarray | None,
                          global_best_val: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    eta0 = 0.6
    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < 80_000:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} iter={it:05d}] new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 7.0:
            break

        it += 1

    return global_best_seq, global_best_val


# ----------------------------------------------------------------------
def _local_refinement(a: np.ndarray, n: int, conv_fft_len: int,
                      deadline: float, current_best_val: float) -> tuple[np.ndarray, float]:
    """
    Tiny random mass transfers; accepts only strict improvements.
    """
    a_cur = a.copy()
    best_val = current_best_val
    it = 0
    while time.time() < deadline and it < 5_000:
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        if a_cur[i] <= 0:
            it += 1
            continue
        frac = np.random.rand() * 0.15          # up to 15 % of a[i]
        delta = a_cur[i] * frac
        a_new = a_cur.copy()
        a_new[i] -= delta
        a_new[j] += delta

        conv = _full_convolution(a_new, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val - 1e-12:
            best_val = val
            a_cur = a_new
            print(f"[Local] improvement to {best_val:.6f}")
        it += 1
    return a_cur, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the
    evaluation metric.  The routine respects the supplied time budget
    (default 1000 s) and returns the best sequence encountered as a plain
    Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    # primary support size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # warm‑start from a previously found construction, if any
    warm_start = globals().get("height_sequence_1")
    warm = None
    if warm_start is not None:
        w = np.asarray(warm_start, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            warm = w / w.sum()

    # ------------------------------------------------------------------
    # 1️⃣  build & quickly screen a diverse candidate pool
    pool = _candidate_pool(n, warm)

    best_seq = None
    best_val = float("inf")
    scored = []

    for idx, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        a = cand
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        val = _objective(a, n, conv_fft_len)
        scored.append((val, a))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        print(f"[Screen {idx:03d}] val={val:.6f}")

    # keep the most promising starts (up to eight)
    scored.sort(key=lambda t: t[0])
    top_vectors = [vec for _, vec in scored[:8]]

    # ------------------------------------------------------------------
    # 2️⃣  deep optimisation on the top candidates
    for cid, start_vec in enumerate(top_vectors):
        if time.time() >= deadline:
            break
        print(f"[Optim {cid}] launching deep optimisation")
        best_seq, best_val = _adam_optimize(
            start_vec, n, conv_fft_len, corr_fft_len,
            deadline, best_seq, best_val
        )
        best_seq, best_val = _subgradient_optimize(
            best_seq if best_seq is not None else start_vec,
            n, conv_fft_len, corr_fft_len,
            deadline, best_seq, best_val
        )
        best_seq, best_val = _local_refinement(
            best_seq, n, conv_fft_len, deadline, best_val
        )
        print(f"[Optim {cid}] best so far {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  random restarts while time permits
    while time.time() + 2.0 < deadline:
        alpha = 10 ** np.random.uniform(-2.0, 0.0)
        a0 = np.random.dirichlet(np.full(n, alpha))
        temp_dead = min(deadline, time.time() + 6.0)

        seq_tmp, val_tmp = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len,
            temp_dead, best_seq, best_val
        )
        seq_tmp, val_tmp = _subgradient_optimize(
            seq_tmp, n, conv_fft_len, corr_fft_len,
            temp_dead, seq_tmp, val_tmp
        )
        seq_tmp, val_tmp = _local_refinement(
            seq_tmp, n, conv_fft_len, temp_dead, val_tmp
        )
        if val_tmp < best_val:
            best_val = val_tmp
            best_seq = seq_tmp
            print(f"[Random] new global best {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  optional up‑sampling to a finer grid
    if time.time() + 20.0 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            x_old = np.linspace(0.0, 1.0, n, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("[Upsample] start optimisation on larger support")
        best_seq_big, best_val_big = _adam_optimize(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big,
            deadline, None, float("inf")
        )
        best_seq_big, best_val_big = _subgradient_optimize(
            best_seq_big, n_big, conv_fft_len_big, corr_fft_len_big,
            deadline, best_seq_big, best_val_big
        )
        best_seq_big, best_val_big = _local_refinement(
            best_seq_big, n_big, conv_fft_len_big, deadline, best_val_big
        )
        print(f"[Upsample] best on big grid {best_val_big:.6f}")

        if best_val_big < best_val:
            best_val = best_val_big
            # down‑sample back to original size (simple averaging of pairs)
            best_seq = np.mean(best_seq_big.reshape(-1, 2), axis=1)
            best_seq = np.maximum(best_seq, 0.0)
            if best_seq.sum() > 0:
                best_seq /= best_seq.sum()
            else:
                best_seq = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # final sanity checks and return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return best_seq.tolist()
