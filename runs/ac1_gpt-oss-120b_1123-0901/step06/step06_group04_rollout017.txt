```
def hillclimb_optimize(a0, n, conv_len, corr_len, deadline, steps=10000):
    a = a0.copy()
    best_a = a.copy()
    best_val = objective(a)
    sigma = 0.5 # initial logit noise scale
    for step in range(steps):
        if time.time() > deadline:
            break
        # propose new by adding gaussian noise to logits representation
        logits = np.log(np.maximum(a, 1e-12))
        logits += np.random.normal(scale=sigma, size=n)
        a_new = np.exp(logits - np.max(logits))
        a_new /= a_new.sum()
        val_new = objective(a_new)
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new # accept
            sigma = max(sigma * 0.99, 1e-4) # reduce step size
        else:
            # maybe accept with probability for SA?
            if np.random.rand() < np.exp(-(val_new - best_val) / (sigma*10)):
                a = a_new
        # occasional increase sigma
        if step % 500 == 0:
            sigma = min(sigma * 1.02, 0.5)
    return best_a, best_val
```

"""
Optimises a non‑negative step‑function (vector) to minimise
    F(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm proceeds in three stages for each candidate length:
1. Adam optimisation of a smooth‑max surrogate (log‑sum‑exp) using a
   soft‑max parameterisation that guarantees a≥0 and Σa=1.
2. Exact ℓ∞ sub‑gradient refinement with an exponentiated‑gradient update.
3. A simple stochastic hill‑climb (perturb logits and accept improvements)
   that works as a final local‑search / simulated‑annealing phase.

We start from several diverse seeds (uniform, edge‑Gaussian,
product‑power‑law, beta‑shapes and random Dirichlet draws) and from the
previous global best if available.  A longer length (65536) is used because
larger dimensions give extra degrees of freedom and, empirically,
allow a lower bound than the 32768‑point construction.

All work respects the wall‑clock budget supplied by the caller.
"""

import time
import numpy as np

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT. Returns length 2*n‑1."""
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) required for the sub‑gradient."""
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

def _softmax(logits: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    mx = logits.max()
    exp = np.exp(logits - mx)
    return exp / exp.sum()

def _adam_optimize(a0: np.ndarray, n: int,
                   conv_fft_len: int, corr_fft_len: int,
                   deadline: float, max_iters: int = None) -> tuple[np.ndarray, float]:
    """Adam on a log‑sum‑exp surrogate of max(conv). Returns best a and its value."""
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.04                       # a little smaller than before

    # smooth‑max temperature schedule
    lam = 0.5
    lam_factor = 1.01

    if max_iters is None:
        # give more iterations than the reference implementation
        max_iters = int(80000 * 1024 / n) + 3000
        max_iters = max(3000, max_iters)

    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max to stay on simplex
        a = _softmax(logits)

        # convolution and smooth‑max weights
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        # compute weight vector approximating the arg‑max
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # gradient of smooth‑max w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # back‑propagation through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update of logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature annealing
        lam = min(lam * lam_factor, 1e8)

        # exact objective (sum(a)=1)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # occasional random jitter to escape plateaus
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val

def _subgradient_refine(a0: np.ndarray, n: int,
                        conv_fft_len: int, corr_fft_len: int,
                        deadline: float, max_iters: int = 4000) -> tuple[np.ndarray, float]:
    """Exact ℓ∞ sub‑gradient refinement with an exponentiated‑gradient update."""
    a = a0 / a0.sum()
    best_val = float("inf")
    best_a = a.copy()
    eta = 0.45
    last_improve = time.time()

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        # active set (indices that achieve the maximum up to tolerance)
        active = np.isclose(conv, max_c, atol=1e-12)

        # construct sub‑gradient weight vector
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient step (mirror descent with entropy)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # evaluate
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        eta *= 0.9995   # slowly decay step size

        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val

def _hillclimb_optimize(a0: np.ndarray, n: int,
                        conv_fft_len: int, corr_fft_len: int,
                        deadline: float, max_steps: int = 5000) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing / simulated‑annealing phase.
    Operates in the log‑probability (logits) space, proposes Gaussian
    perturbations and keeps improvements.
    """
    # make sure we start from a properly normalised vector
    a = a0 / a0.sum()
    best_a = a.copy()
    best_val = 2.0 * n * _full_convolution(a, fft_len=conv_fft_len).max()
    # starting scale for the Gaussian perturbation in logit space
    sigma = 0.5
    it = 0
    while it < max_steps and time.time() < deadline:
        it += 1
        # propose new logits
        logits = np.log(np.maximum(a, 1e-12))
        logits += np.random.normal(scale=sigma, size=n)
        a_new = np.exp(logits - np.max(logits))
        a_new /= a_new.sum()

        val_new = 2.0 * n * _full_convolution(a_new, fft_len=conv_fft_len).max()
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
            # tighten the search a little
            sigma = max(sigma * 0.99, 1e-4)
        else:
            # occasional uphill moves (simulated annealing style)
            # temperature is tied to sigma so that larger steps allow occasional climbs
            if np.random.rand() < np.exp(-(val_new - best_val) / (sigma * 0.02 + 1e-12)):
                a = a_new

        # periodically enlarge sigma a bit to keep exploration alive
        if it % 400 == 0:
            sigma = min(sigma * 1.02, 0.5)

    return best_a, best_val

def _generate_seeds(n: int, best_seq) -> list[np.ndarray]:
    """Return a modest collection of diverse initial vectors of length n."""
    seeds = []

    # 1) uniform distribution
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) up‑sample / down‑sample previous global best if we have one
    if best_seq is not None:
        old = np.asarray(best_seq, dtype=np.float64)
        x_old = np.linspace(0.0, 1.0, old.shape[0])
        x_new = np.linspace(0.0, 1.0, n)
        up = np.interp(x_new, x_old, old)
        up = np.maximum(up, 0.0)
        s = up.sum()
        if s > 0:
            up = up / s
        else:
            up = np.random.dirichlet(np.ones(n))
        seeds.append(up)

    # 3) edge‑Gaussian shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in [0.02, 0.05, 0.1, 0.2, 0.5]:
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape_sum = shape.sum()
        if shape_sum > 0:
            shape = shape / shape_sum
        seeds.append(shape)

    # 4) product‑power‑law (edge‑concentrated) shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in [-2.5, -2.0, -1.5, -1.0, -0.5]:
        shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
        s = shape.sum()
        if s > 0:
            shape = shape / s
        seeds.append(shape)

    # 5) symmetric beta‑like shapes (x^a * (1-x)^b)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    eps = 1e-12
    for a, b in [(-2.0, -2.0), (-1.5, -1.5), (-1.0, -1.0),
                 (-0.5, -0.5), (-2.0, -0.5), (-0.5, -2.0)]:
        shape = (np.maximum(x, eps) ** a) * (np.maximum(1.0 - x, eps) ** b)
        s = shape.sum()
        if s > 0:
            shape = shape / s
        seeds.append(shape)

    # 6) a few random Dirichlet draws for stochastic diversity
    for _ in range(3):
        seeds.append(np.random.dirichlet(np.ones(n)))

    # keep a reasonable number of candidates
    max_cand = 12 if n <= 32768 else 10
    return seeds[:max_cand]

"""
Optimises a non‑negative step‑function (vector) to minimise

    F(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works in three stages for each candidate length:

1. **Adam** on a smooth‑max (log‑sum‑exp) surrogate of the convolution
   maximum, using a soft‑max parametrisation that guarantees a≥0 and Σa=1.
2. **Exact ℓ∞ sub‑gradient refinement** with an exponentiated‑gradient (mirror‑descent)
   update that directly minimises the true maximum.
3. **Stochastic hill‑climb** (simulated‑annealing style) that perturbs the logits
   and keeps improvements, acting as a fine‑tuning stage.

We start from a diverse set of seeds (uniform, edge‑Gaussian, power‑law,
beta‑like, and random Dirichlet draws) and also from the previous global
best if it exists.  A large dimension (65536) is used because the extra
degrees of freedom typically allow a lower bound than the 32768‑point
construction while still fitting comfortably into the 1000 s budget.

All work respects the supplied wall‑clock budget; the best sequence found
is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution / correlation helpers (FFT based)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT. Returns array of length 2*n‑1."""
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the sub‑gradient."""
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _softmax(logits: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector (sum = 1)."""
    mx = logits.max()
    e = np.exp(logits - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray, n: int,
                   conv_fft_len: int, corr_fft_len: int,
                   deadline: float, max_iters: int = None):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns (best_a, best_value) where value = 2·n·max(conv) for the best iterate.
    """
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.04

    # smooth‑max temperature schedule
    lam = 0.5
    lam_factor = 1.01

    if max_iters is None:
        max_iters = int(80000 * 1024 / n) + 3000
        max_iters = max(3000, max_iters)

    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(logits)

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e8)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


def _subgradient_refine(a0: np.ndarray, n: int,
                        conv_fft_len: int, corr_fft_len: int,
                        deadline: float, max_iters: int = 4000):
    """
    Exact ℓ∞ sub‑gradient refinement using an exponentiated‑gradient update.
    Returns (best_a, best_value).
    """
    a = a0 / a0.sum()
    best_val = float("inf")
    best_a = a.copy()
    eta = 0.45
    last_improve = time.time()

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        eta *= 0.9995

        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val


def _hillclimb_optimize(a0: np.ndarray, n: int,
                        conv_fft_len: int, corr_fft_len: int,
                        deadline: float, max_steps: int = 5000):
    """
    Simple stochastic hill‑climb / simulated‑annealing phase.
    Works in log‑probability space, proposes Gaussian perturbations,
    and keeps improvements.
    """
    a = a0 / a0.sum()
    best_a = a.copy()
    best_val = 2.0 * n * _full_convolution(a, fft_len=conv_fft_len).max()
    sigma = 0.5
    it = 0

    while it < max_steps and time.time() < deadline:
        it += 1
        logits = np.log(np.maximum(a, 1e-12))
        logits += np.random.normal(scale=sigma, size=n)
        a_new = np.exp(logits - np.max(logits))
        a_new /= a_new.sum()

        val_new = 2.0 * n * _full_convolution(a_new, fft_len=conv_fft_len).max()
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
            sigma = max(sigma * 0.99, 1e-4)
        else:
            # occasional uphill moves (SA style)
            if np.random.rand() < np.exp(-(val_new - best_val) / (sigma * 0.02 + 1e-12)):
                a = a_new

        if it % 400 == 0:
            sigma = min(sigma * 1.02, 0.5)

    return best_a, best_val


# ----------------------------------------------------------------------
# Seed generation helpers
# ----------------------------------------------------------------------


def _generate_seeds(n: int, reference) :
    """
    Produce a modest collection of diverse initial vectors of length n.
    If `reference` is not None it is up‑sampled / down‑sampled to length n and added.
    """
    seeds = []

    # 1) uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) up‑sample / down‑sample a previous solution if available
    if reference is not None:
        old = np.asarray(reference, dtype=np.float64)
        x_old = np.linspace(0.0, 1.0, old.shape[0])
        x_new = np.linspace(0.0, 1.0, n)
        up = np.interp(x_new, x_old, old)
        up = np.maximum(up, 0.0)
        s = up.sum()
        if s > 0:
            up = up / s
        else:
            up = np.random.dirichlet(np.ones(n))
        seeds.append(up)

    # 3) edge‑Gaussian shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in [0.02, 0.05, 0.1, 0.2, 0.5]:
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape = shape / shape.sum()
        seeds.append(shape)

    # 4) product‑power‑law (edge‑concentrated) shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in [-2.5, -2.0, -1.5, -1.0, -0.5]:
        shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
        shape = shape / shape.sum()
        seeds.append(shape)

    # 5) beta‑like symmetric shapes: (x^a)*(1-x)^b
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    eps = 1e-12
    for a, b in [(-2.0, -2.0), (-1.5, -1.5), (-1.0, -1.0),
                 (-0.5, -0.5), (-2.0, -0.5), (-0.5, -2.0)]:
        shape = (np.maximum(x, eps) ** a) * (np.maximum(1.0 - x, eps) ** b)
        shape = shape / shape.sum()
        seeds.append(shape)

    # 6) a few random Dirichlet draws
    for _ in range(3):
        seeds.append(np.random.dirichlet(np.ones(n)))

    max_cand = 12 if n <= 32768 else 10
    return seeds[:max_cand]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises

        F(a) = 2·n·max(conv(a,a)) / (sum(a)²).

    The routine respects the supplied wall‑clock budget and returns the best
    sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for final checks

    # Prefer a very large dimension; fall back to a smaller one if time runs out.
    target_lengths = [65536, 32768]

    best_seq = None
    best_val = float("inf")

    # Retrieve any previously stored sequence (e.g. from a former run)
    previous = globals().get("height_sequence_1")

    for n in target_lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

        seed_vectors = _generate_seeds(n, previous if best_seq is None else best_seq)

        for idx, seed_vec in enumerate(seed_vectors):
            if time.time() > deadline:
                break
            try:
                # Phase 1 – Adam on a smooth‑max surrogate
                a_adam, val_adam = _adam_optimize(
                    seed_vec, n, conv_fft_len, corr_fft_len, deadline
                )
                # Phase 2 – exact ℓ∞ sub‑gradient refinement
                a_ref, val_ref = _subgradient_refine(
                    a_adam, n, conv_fft_len, corr_fft_len, deadline
                )
                # Phase 3 – stochastic hill‑climb fine‑tuning
                a_hc, val_hc = _hillclimb_optimize(
                    a_ref, n, conv_fft_len, corr_fft_len, deadline
                )

                for cand_val, cand_seq in [(val_adam, a_adam),
                                           (val_ref, a_ref),
                                           (val_hc, a_hc)]:
                    if cand_val < best_val:
                        best_val = cand_val
                        best_seq = cand_seq.copy()
                        print(f"[n={n} cand={idx}] new global best {best_val:.6f}")

            except Exception as exc:
                print(f"[n={n} cand={idx}] exception: {exc}")

        # Small jitter runs around the current best of *exact* length n
        if best_seq is not None and len(best_seq) == n:
            for jitter_id in range(4):
                if time.time() > deadline:
                    break
                logits = np.log(np.maximum(best_seq, 1e-12))
                logits += np.random.normal(scale=0.02, size=n)
                a0 = np.exp(logits - np.max(logits))
                a0 = a0 / a0.sum()

                a_adam, val_adam = _adam_optimize(
                    a0, n, conv_fft_len, corr_fft_len, deadline
                )
                a_ref, val_ref = _subgradient_refine(
                    a_adam, n, conv_fft_len, corr_fft_len, deadline
                )
                a_hc, val_hc = _hillclimb_optimize(
                    a_ref, n, conv_fft_len, corr_fft_len, deadline
                )
                for cand_val, cand_seq in [(val_adam, a_adam),
                                           (val_ref, a_ref),
                                           (val_hc, a_hc)]:
                    if cand_val < best_val:
                        best_val = cand_val
                        best_seq = cand_seq.copy()
                        print(f"[n={n} jitter={jitter_id}] new global best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Safety net – if nothing succeeded, fall back to a uniform distribution
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 32768
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Optional final sanity check (use the supplied evaluator if available)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final sequence length {len(best_seq)} → evaluate_sequence = {final_val:.6f}")
    except Exception as e:
        print(f"Could not evaluate final sequence: {e}")

    return best_seq.tolist()
