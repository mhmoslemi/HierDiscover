"""
Propose a non‑negative step function that minimises

    F(a) = 2 * n * max(a * a) / (sum(a)**2)

using a two‑phase optimiser:

1. Adam on a smooth‑max (log‑sum‑exp) surrogate of the convolution maximum.
   The vector is represented in the simplex by a soft‑max of free logits.
2. Exact ℓ∞ sub‑gradient refinement with an exponentiated‑gradient update.

The routine works on a series of lengths (increasingly large) and spends
as much time as the 1000 s wall‑clock budget allows.  After the main loop
any remaining time is used for extra jitter‑refinements of the current
global best.  The best sequence (as a plain Python list) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a using a power‑of‑two FFT.
    Returns an array of length 2*len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) required for the gradient.
    Result length = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a log‑sum‑exp surrogate of the convolution peak.
    Returns the best iterate (as a probability vector that sums to 1) and
    its exact objective value 2·n·max(conv).
    """
    # initialise on the simplex (soft‑max of logits)
    a = np.maximum(a0, 1e-12)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.12                # a bit larger than the original code
    lam = 0.5                # start with a smooth max, increase quickly
    lam_factor = 1.07

    max_iters = 20000        # generous budget – will be cut off by deadline
    best_a = a.copy()
    best_val = float("inf")
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ---- simplex projection (soft‑max) ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- convolution and smooth‑max weights ----
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        # smooth‑max (log‑sum‑exp) weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # ---- gradient of smooth‑max w.r.t. a ----
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # back‑propagation through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, 1e8)

        # ---- exact objective (sum is 1, so denominator = 1) ----
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # ---- occasional random shake‑up to escape plateaus ----
        if time.time() - last_improve > 3.5:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Exact ℓ∞ sub‑gradient refinement using an exponentiated‑gradient step.
    Returns the best iterate and its exact objective value.
    """
    a = a0 / a0.sum()
    best_val = float("inf")
    best_a = a.copy()
    eta = 0.45
    last_improve = time.time()

    for it in range(3000):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        # active set (indices achieving the maximum, tolerance 1e-12)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient update (keeps a on the simplex)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                      # safeguard – re‑sample from Dirichlet
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        eta *= 0.9995                       # slow decay
        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the wall‑clock budget (seconds) and
    returns the best sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Length schedule – start small, finish with a very large vector
    # --------------------------------------------------------------
    lengths = [8192, 16384, 32768, 65536]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

        # --------------------------------------------------------------
        # Build a diverse set of initial candidates
        # --------------------------------------------------------------
        candidates = []

        # 0) uniform
        candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # 1) warm‑start from a previous solution of the same size
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.asarray(best_seq, dtype=np.float64))

        # 2) up‑/down‑sample an existing best of different size
        if best_seq is not None and len(best_seq) != n:
            old = np.asarray(best_seq, dtype=np.float64)
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, old)
            if up.sum() > 0:
                up = up / up.sum()
            else:
                up = np.random.dirichlet(np.ones(n))
            candidates.append(up)

        # 3) global variable if it exists
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            h = np.asarray(hseq, dtype=np.float64)
            if h.shape[0] > n:
                h = h[:n]
            elif h.shape[0] < n:
                h = np.pad(h, (0, n - h.shape[0]))
            h = np.maximum(h, 0.0)
            if h.sum() > 0:
                h = h / h.sum()
            else:
                h = np.random.dirichlet(np.ones(n))
            candidates.append(h)

        # 4) a few random Dirichlet draws
        for _ in range(4):
            candidates.append(np.random.dirichlet(np.ones(n)))

        # 5) power‑law / edge‑concentrated shapes
        idx = np.arange(1, n + 1, dtype=np.float64)
        prod = idx * (n - idx + 1.0)
        for alpha in [-2.5, -2.0, -1.5, -1.0, -0.5]:
            shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
            if shape.sum() > 0:
                shape = shape / shape.sum()
                candidates.append(shape)

        # 6) two‑Gaussian edge shapes
        i = np.arange(n, dtype=np.float64)
        for gamma in [0.05, 0.15, 0.3, 0.6]:
            shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            shape = shape / shape.sum()
            candidates.append(shape)

        # Limit the pool to keep the wall‑clock budget a‑bit tighter
        max_candidates = 15 if n <= 32768 else 9
        candidates = candidates[:max_candidates]

        # --------------------------------------------------------------
        # Optimise each candidate
        # --------------------------------------------------------------
        for cand_id, cand in enumerate(candidates):
            if time.time() > deadline:
                break
            try:
                a_adam, val_adam = _adam_optimize(
                    cand, n, conv_fft_len, corr_fft_len, deadline
                )
                a_ref, val_ref = _subgradient_refine(
                    a_adam, n, conv_fft_len, corr_fft_len, deadline
                )

                # Keep the best of the two phases
                if val_ref < best_val:
                    best_val = val_ref
                    best_seq = a_ref.copy()
                    print(f"[n={n} cand={cand_id}] new global best {best_val:.6f}")
                elif val_adam < best_val:
                    best_val = val_adam
                    best_seq = a_adam.copy()
                    print(f"[n={n} cand={cand_id}] new global best (Adam) {best_val:.6f}")

            except Exception as exc:
                print(f"[n={n} cand={cand_id}] exception: {exc}")

        # --------------------------------------------------------------
        # Local jitter‑search around the current best of this length
        # --------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            for jitter_id in range(8):
                if time.time() > deadline:
                    break

                logits = np.log(np.maximum(best_seq, 1e-12))
                # a slightly larger Gaussian jitter than the original code
                logits += np.random.normal(scale=0.025, size=n)

                a0 = np.exp(logits - np.max(logits))
                a0 = a0 / a0.sum()

                a_adam, val_adam = _adam_optimize(
                    a0, n, conv_fft_len, corr_fft_len, deadline
                )
                a_ref, val_ref = _subgradient_refine(
                    a_adam, n, conv_fft_len, corr_fft_len, deadline
                )
                if val_ref < best_val:
                    best_val = val_ref
                    best_seq = a_ref.copy()
                    print(f"[n={n} jitter={jitter_id}] new global best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Use any remaining time to polish the best sequence we have
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = len(best_seq)
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))
        while time.time() + 0.2 < deadline:
            # extra jitter‑refinement loop
            logits = np.log(np.maximum(best_seq, 1e-12))
            logits += np.random.normal(scale=0.03, size=n)
            a0 = np.exp(logits - np.max(logits))
            a0 = a0 / a0.sum()
            a_adam, val_adam = _adam_optimize(
                a0, n, conv_fft_len, corr_fft_len, deadline
            )
            a_ref, val_ref = _subgradient_refine(
                a_adam, n, conv_fft_len, corr_fft_len, deadline
            )
            if val_ref < best_val:
                best_val = val_ref
                best_seq = a_ref.copy()
                print(f"[polish] improved global best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely case that nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 8192
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    print(f"\nFinished – best objective {best_val:.6f} with length {len(best_seq)}")
    return best_seq.tolist()

**Explanation of the improvements**

* **Longer optimisation budget** – the Adam phase now runs up to 20 000 iterations (instead of ≈4 000) and the sub‑gradient phase is left unchanged but benefits from more frequent restarts.  
* **More aggressive temperature schedule** (`lam` doubles faster) and a slightly larger learning rate help the optimiser focus on the true peak earlier.  
* **Additional candidate generators** (extra Dirichlet draws, a broader range of edge‑Gaussian shapes) increase diversity of starting points.  
* **Extended jitter loop** – after each length we perform eight jitter trials (instead of five) with a larger perturbation to escape local minima.  
* **Very large length (65 536)** – doubling the previous maximum length gives a finer discretisation, allowing the peak of the convolution to be flattened further (the theoretical lower bound approaches 1).  
* **Final polishing phase** – any remaining time is spent repeatedly jittering and re‑optimising the current global best, squeezing out the last bits of improvement.

All helper functions are top‑level, there are no closures or lambdas, and the routine respects the 1000 s wall‑clock limit while returning the best‑found sequence as required.
