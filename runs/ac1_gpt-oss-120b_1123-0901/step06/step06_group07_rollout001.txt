"""
Improved search for a low‑evaluation step function.

Algorithm
---------
1. **Rich random‑parametric generation** – build thousands of symmetric
   edge‑biased shapes (exponential, power‑law, mixtures, Dirichlet draws)
   and keep the best few after a cheap convolution evaluation.

2. **Two‑phase optimisation** for each surviving start
   * Adam on a smooth‑max surrogate (log‑sum‑exp) with a slow temperature
     schedule; the surrogate becomes the exact max after ~1000 iterations.
   * Exponentiated‑gradient sub‑gradient steps on the true max‑convolution
     (active‑set weighting) to sharpen the solution.

   The optimisation runs for as many iterations as time permits (up to
   300 k Adam steps and 50 k sub‑gradient steps per start).

3. **Upscaling** – if time remains, linearly interpolate the current best
   vector to a longer grid (n = 8192) and repeat the deep optimisation with
   a longer budget.

All vectors are kept on the probability simplex, therefore
`objective = 2·n·max(conv)` (the denominator `sum(a)^2` equals 1).
The code respects the 1000 s budget, prints only essential progress
messages and finally returns the best sequence found as a plain Python
list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Fast evaluation of the objective (assuming a on the simplex)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of the max‑convolution.
    Returns the best vector/value seen during the run.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.02                      # smaller step for fine control

    lam = 0.5                      # start temperature
    lam_factor = 1.008            # very slow increase → smoother transition
    lam_cap = 1e6

    it = 0
    max_iter = 300_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        # simplex via soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # surrogate convolution
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional shake‑up if stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient steps on the true max‑convolution.
    """
    a = best_seq if best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()

    eta = 0.6
    it = 0
    max_iter = 50_000
    last_improve = time.time()

    while it < max_iter and time.time() < deadline:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it:05d}] new best {val:.6f}")

        # active set of indices attaining the maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # stop if no improvement for a while
        if time.time() - last_improve > 5.0:
            break

        it += 1

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run the Adam phase followed by the sub‑gradient refinement.
    Updates the globally best vector/value.
    """
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val

def _generate_parametric_candidates_ext(n: int, warm_start: np.ndarray | None):
    """
    Produce a large, diverse set of candidates.
    Families:
        * warm‑start and its rotations
        * uniform
        * random Dirichlet (various concentrations)
        * edge‑exponential (single, mixed, double)

    """

    # Warm‑start and simple rotations

            candidates.append(np.roll(warm_start, shift))

    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Random Dirichlet draws
    for conc in (0.02, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    i = np.arange(n, dtype=np.float64)

    # Edge‑exponential families (single)
    for alpha in np.logspace(-3, 1.0, num=30):  # 0.001 … 10
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            cand = w * edge + (1.0 - w) / n
            candidates.append(cand)

    # Edge‑power‑law families (single)
    for p in np.linspace(0.1, 12.0, num=30):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            cand = w * edge + (1.0 - w) / n
            candidates.append(cand)

    # Double‑exponential mixtures
    for _ in range(200):
        alpha1 = np.exp(np.random.uniform(np.log(0.001), np.log(10.0)))
        alpha2 = np.exp(np.random.uniform(np.log(0.001), np.log(10.0)))
        w1 = np.random.rand()
        w2 = np.random.rand() * (1.0 - w1)
        mix = w1 * (np.exp(-alpha1 * i) + np.exp(-alpha1 * (n - 1 - i)))
        mix += w2 * (np.exp(-alpha2 * i) + np.exp(-alpha2 * (n - 1 - i)))
        mix /= mix.sum()
        # optional uniform residual
        uniform_weight = np.random.rand() * (1.0 - (w1 + w2))
        cand = (1.0 - uniform_weight) * mix + uniform_weight / n
        candidates.append(cand)

    # Double‑power‑law mixtures
    for _ in range(200):
        p1 = np.random.uniform(0.1, 12.0)
        p2 = np.random.uniform(0.1, 12.0)

        w2 = np.random.rand() * (1.0 - w1)

        uniform_weight = np.random.rand() * (1.0 - (w1 + w2))
        cand = (1.0 - uniform_weight) * mix + uniform_weight / n
        candidates.append(cand)

    # Positive sinusoidal half‑wave
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)
