def solve_lp_min_max(f):
    n = len(f)
    f = np.asarray(f, dtype=np.float64)
    # Build sparse matrix A_ub for convolution constraints
    row_diag_len = n
    data = []
    offsets = []
    for d in range(n):
        data.append(np.full(row_diag_len, f[d]))
        offsets.append(d)
    A_f = sp.diags(data, offsets, shape=(2*n-1, n), format='csr')
    # Append column for variable t: -1 for each row
    # Use sp.hstack
    t_col = sp.csr_matrix((-np.ones((2*n-1, 1)), (np.arange(2*n-1), np.zeros(2*n-1))), shape=(2*n-1, 1))
    A_ub = sp.hstack([A_f, t_col], format='csr')
    # Right-hand side zeroes
    b_ub = np.zeros(2*n-1)
    # Equality: sum b =1
    A_eq = sp.csr_matrix(np.ones((1,n+1)), dtype=np.float64)
    b_eq = np.array([1.0])
    # variable bounds: b_i >=0, t >=0
    bounds = [(0, None)]*n + [(0, None)]
    # Objective c: minimize t, so c = [0,...,0, 1]
    c = np.zeros(n+1)
    c[-1] = 1.0
    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if not res.success:
        # fallback: near uniform
        return np.full(n, 1.0/n), None
    b_opt = res.x[:n]
    t_opt = res.x[-1]
    # Might normalise b_opt to sum=1
    return b_opt, t_opt

def generate_seeds(n, best_seq):
    seeds = []
    # Uniform
    seeds.append(np.full(n, 1.0/n))
    # Edge shape
    idx = np.arange(1, n+1, dtype=np.float64)
    prod = idx * (n - idx + 1)
    for alpha in [-2.0, -1.5, -1.0]:
        shape = np.where(prod>0, np.power(prod, alpha), 0)
        if shape.sum()>0:
            shape = shape / shape.sum()
            seeds.append(shape)
    # Gaussian edges
    i = np.arange(n, dtype=np.float64)
    for gamma in [0.2, 0.5, 1.0]:
        shape = np.exp(-gamma * (i**2)) + np.exp(-gamma*((n-1-i)**2))
        shape = shape / shape.sum()
        seeds.append(shape)
    # Sin^2 bump
    shape = np.sin(np.pi * (np.arange(n)+0.5)/n) ** 2
    shape = shape/shape.sum()
    seeds.append(shape)
    # Mix uniform and edge
    edge = np.exp(-0.5 * (i**2)) + np.exp(-0.5*((n-1-i)**2))
    edge = edge/edge.sum()
    mix = 0.5*np.full(n, 1.0/n) + 0.5*edge
    mix = mix/mix.sum()
    seeds.append(mix)
    # random Dirichlet
    for _ in range(3):
        seeds.append(np.random.dirichlet(np.ones(n)))
    # upsample downsample of global best
    if best_seq is not None:
        old = np.asarray(best_seq, dtype=np.float64)
        x_old = np.linspace(0,1,old.shape[0])
        x_new = np.linspace(0,1,n)
        up = np.interp(x_new, x_old, old)
        up = np.maximum(up, 0)
        if up.sum()>0:
            up = up / up.sum()
            seeds.append(up)
    # optional symmetric shape: enforce symmetry
    # keep seeds length limited to maybe 12
    return seeds[:12]

def solve_lp_min_max(f):
    import numpy as np, scipy.sparse as sp, scipy.optimize as opt, time
    n = len(f)
    f = np.asarray(f, dtype=np.float64)
    # Build sparse Toeplitz matrix for convolution constraints
    offsets = np.arange(n, dtype=int)
    data = [f[d] for d in range(n)]  # scalars
    A_f = sp.diags(data, offsets, shape=(2*n-1, n), format='csr')
    neg_one_col = sp.csr_matrix(-np.ones((2*n-1, 1)))
    A_ub = sp.hstack([A_f, neg_one_col], format='csr')
    b_ub = np.zeros(2*n-1, dtype=np.float64)
    # Equality sum b = 1
    A_eq = sp.csr_matrix(np.ones((1, n+1)), dtype=np.float64)
    b_eq = np.array([1.0])
    # bounds: b_i >=0, t >=0
    bounds = [(0, None)]*n + [(0, None)]
    c = np.zeros(n+1)
    c[-1] = 1.0
    # Use hiGHS solver
    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        return None, None
    b_opt = res.x[:n]
    t_opt = res.x[-1]
    # ensure positivity and normalization
    b_opt = np.maximum(b_opt, 0)
    if b_opt.sum() == 0:
        b_opt = np.random.dirichlet(np.ones(n))
    else:
        b_opt = b_opt / b_opt.sum()
    return b_opt, t_opt

"""
Propose a non‑negative step function that (approximately) minimises

    F(a) = 2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm is a hybrid of three ideas:

1. **Gradient phase** – Adam on a smooth‐max surrogate of the ℓ∞‑norm,
   followed by exact ℓ∞ sub‑gradient refinement (exponentiated‑gradient).

2. **Linear‑program phase** – For a given vector `f` we solve the LP

        minimise   t
        s.t.       (f * b)_k ≤ t   for all k
                   b_i ≥ 0
                   Σ b_i = 1

   The optimal `b` gives a direction that cannot increase the current
   maximal convolution value.  We mix `b` into the current iterate.

3. **Multi‑stage search** – A pool of heterogeneous seeds (uniform,
   edge‑concentrated power‑laws, Gaussian edges, sine‑squared bumps,
   Dirichlet draws, up‑sampled previous best) is generated for a series
   of lengths.  Each seed is refined by the gradient phase, optionally
   by the LP phase, and finally jittered.  The best sequence seen before
   the deadline is returned.

The implementation respects the 1000 s wall‑clock budget, uses FFT for
fast convolutions and SciPy’s HiGHS LP solver for the linear‑program
step (performed only for moderate lengths to keep memory usage bounded).
All vectors are kept on the probability simplex and are symmetrised after
each update – the optimal solution is expected to be symmetric.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Convolution helpers (FFT based)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]  for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Symmetrisation utilities
# ----------------------------------------------------------------------


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce a[i] == a[n‑1‑i] (element‑wise averaging)."""
    return (v + v[::-1]) * 0.5


# ----------------------------------------------------------------------
# Adam optimisation on a smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Adam optimisation of the log‑sum‑exp smooth‑max of the convolution."""
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05

    lam = 1.0               # smooth‑max temperature
    lam_factor = 1.02

    # iteration budget – more generous than the previous version
    max_iters = int(60000 * 1024 / n) + 3000
    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum if w_sum > 0 else np.full_like(w, 1.0 / w.size)

        # gradient of smooth‑max w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # back‑propagation through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature annealing (gently increase)
        lam = min(lam * lam_factor, 1e8)

        # exact evaluation (sum(a) == 1)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # occasional jitter if we stall
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact ℓ∞ sub‑gradient refinement (exponentiated‑gradient)
# ----------------------------------------------------------------------


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Exact ℓ∞ sub‑gradient refinement using exponentiated‑gradient updates."""
    a = a0 / a0.sum()
    best_val = float("inf")
    best_a = a.copy()
    eta = 0.45
    last_improve = time.time()

    for it in range(3000):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:  # fallback – random simplex point
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # exact evaluation
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        eta *= 0.9995  # slow decay

        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑program (LP) refinement – minimise maximal convolution with fixed f
# ----------------------------------------------------------------------


def _solve_lp_min_max(f: np.ndarray):
    """
    Solve
        min   t
        s.t.  (f * b)_k ≤ t   for k = 0…2n‑2
              b_i ≥ 0
              Σ_i b_i = 1
    Returns the optimal b (normalised) and the optimal t.
    """
    n = f.shape[0]
    # Build a sparse Toeplitz matrix representing (f * b)
    # Row k, column i has entry f[k‑i] if 0 ≤ k‑i < n else 0.
    offsets = np.arange(n, dtype=int)            # 0,1,…,n‑1
    data = [float(f[d]) for d in range(n)]      # scalar for each diagonal
    A_f = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    # Column for the variable t (coefficient –1 in every inequality)
    neg_one_col = sp.csr_matrix(-np.ones((2 * n - 1, 1)))
    A_ub = sp.hstack([A_f, neg_one_col], format="csr")
    b_ub = np.zeros(2 * n - 1, dtype=float)

    # Equality Σ b_i = 1
    A_eq = sp.csr_matrix(np.ones((1, n + 1)), dtype=float)
    b_eq = np.array([1.0])

    # Bounds – all variables non‑negative
    bounds = [(0, None)] * n + [(0, None)]  # last bound is for t

    # Objective: minimise t
    c = np.zeros(n + 1, dtype=float)
    c[-1] = 1.0

    res = opt.linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None, None

    b_opt = res.x[:n]
    t_opt = res.x[-1]

    # Normalise b (numerical noise may break the exact sum‑to‑1)
    b_opt = np.maximum(b_opt, 0.0)
    s = b_opt.sum()
    if s < 1e-12:
        b_opt = np.random.dirichlet(np.ones(n))
    else:
        b_opt = b_opt / s
    return b_opt, t_opt


# ----------------------------------------------------------------------
# Seed generation (heterogeneous pool)
# ----------------------------------------------------------------------


def _generate_seeds(n: int, best_seq):
    """Return a small, diverse list of candidate starting vectors of length n."""
    seeds = []

    # 1) Uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) Edge‑concentrated power‑law shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in (-2.5, -2.0, -1.5, -1.0, -0.5):
        shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            seeds.append(shape)

    # 3) Two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.1, 0.3, 0.6, 1.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape = shape / shape.sum()
        seeds.append(shape)

    # 4) Smooth sine‑squared bump
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    shape = shape / shape.sum()
    seeds.append(shape)

    # 5) Mixture of uniform and edge Gaussian
    edge = np.exp(-0.5 * (i ** 2)) + np.exp(-0.5 * ((n - 1 - i) ** 2))
    edge = edge / edge.sum()
    mix = 0.5 * np.full(n, 1.0 / n) + 0.5 * edge
    mix = mix / mix.sum()
    seeds.append(mix)

    # 6) A few random Dirichlet draws
    for _ in range(3):
        seeds.append(np.random.dirichlet(np.ones(n)))

    # 7) Up‑sample / down‑sample previous global best (if available)
    if best_seq is not None:
        old = np.asarray(best_seq, dtype=np.float64)
        x_old = np.linspace(0.0, 1.0, old.shape[0])
        x_new = np.linspace(0.0, 1.0, n)
        interp = np.interp(x_new, x_old, old)
        if interp.sum() > 0:
            interp = interp / interp.sum()
        else:
            interp = np.random.dirichlet(np.ones(n))
        seeds.append(interp)

    # Limit the pool size (keeps the total wall‑clock reasonable)
    return seeds[:12]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the provided
    evaluation function.  The routine respects a 1000 s wall‑clock budget,
    combines gradient descent, an LP refinement, and stochastic jitter,
    and returns the best sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    best_seq = None
    best_val = float("inf")

    # Lengths are processed from small to large – cheap warm‑starts first.
    lengths = [2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

        # ------------------------------------------------------------------
        # Build a heterogeneous pool of seeds for the current length
        # ------------------------------------------------------------------
        seeds = _generate_seeds(n, best_seq)

        # ------------------------------------------------------------------
        # Optimise each seed
        # ------------------------------------------------------------------
        for sid, seed_vec in enumerate(seeds):
            if time.time() > deadline:
                break

            a0 = np.maximum(seed_vec, 0.0)
            if a0.sum() == 0:
                a0 = np.random.dirichlet(np.ones(n))
            a0 = a0 / a0.sum()
            a0 = _symmetrize(a0)

            # Phase 1: Adam on smooth‑max surrogate
            try:
                a_adam, _ = _adam_optimize(
                    a0, n, conv_fft_len, corr_fft_len, deadline
                )
            except Exception as e:
                print(f"[n={n} seed={sid}] Adam failed: {e}")
                continue

            a_adam = _symmetrize(a_adam)

            # Phase 2: Exact ℓ∞ sub‑gradient refinement
            try:
                a_ref, _ = _subgradient_refine(
                    a_adam, n, conv_fft_len, corr_fft_len, deadline
                )
            except Exception as e:
                print(f"[n={n} seed={sid}] Sub‑gradient failed: {e}")
                continue

            a_ref = _symmetrize(a_ref)

            # Evaluate the refined candidate
            try:
                val = evaluate_sequence(a_ref.tolist())
            except Exception as e:
                print(f"[n={n} seed={sid}] evaluate_sequence exception: {e}")
                continue

            if val < best_val:
                best_val = val
                best_seq = a_ref.copy()
                print(f"[n={n} seed={sid}] new global best {best_val:.6f}")

            # ------------------------------------------------------------------
            # Optional LP refinement (only for moderate sizes to keep memory sane)
            # ------------------------------------------------------------------
            if n <= 4096 and time.time() < deadline:
                try:
                    b_opt, t_opt = _solve_lp_min_max(best_seq)
                    if b_opt is not None:
                        # Mix the LP direction into the current best.
                        mix = 0.20  # modest mixing to keep stability
                        cand = (1 - mix) * best_seq + mix * b_opt
                        cand = _symmetrize(cand)
                        cand = np.maximum(cand, 0.0)
                        cand /= cand.sum()
                        val_lp = evaluate_sequence(cand.tolist())
                        if val_lp < best_val:
                            best_val = val_lp
                            best_seq = cand.copy()
                            print(f"[n={n} LP] new global best {best_val:.6f}")
                except Exception as e:
                    print(f"[n={n} LP] exception: {e}")

        # ------------------------------------------------------------------
        # Small random jitters around the current best of this length
        # ------------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            for jitter_id in range(5):
                if time.time() > deadline:
                    break
                logits = np.log(np.maximum(best_seq, 1e-12))
                logits += np.random.normal(scale=0.02, size=n)
                a0 = np.exp(logits - np.max(logits))
                a0 = a0 / a0.sum()
                a0 = _symmetrize(a0)

                try:
                    a_adam, _ = _adam_optimize(
                        a0, n, conv_fft_len, corr_fft_len, deadline
                    )
                except Exception as e:
                    print(f"[n={n} jitter={jitter_id}] Adam failed: {e}")
                    continue
                a_adam = _symmetrize(a_adam)

                try:
                    a_ref, _ = _subgradient_refine(
                        a_adam, n, conv_fft_len, corr_fft_len, deadline
                    )
                except Exception as e:
                    print(f"[n={n} jitter={jitter_id}] Sub‑gradient failed: {e}")
                    continue

                a_ref = _symmetrize(a_ref)
                try:
                    val = evaluate_sequence(a_ref.tolist())
                except Exception as e:
                    print(f"[n={n} jitter={jitter_id}] evaluation exception: {e}")
                    continue

                if val < best_val:
                    best_val = val
                    best_seq = a_ref.copy()
                    print(f"[n={n} jitter={jitter_id}] new global best {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Final high‑precision refinement on the overall best (if time permits)
    # ----------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = len(best_seq)
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))
        try:
            a_adam, _ = _adam_optimize(
                best_seq, n, conv_fft_len, corr_fft_len, deadline
            )
            a_adam = _symmetrize(a_adam)
            a_ref, _ = _subgradient_refine(
                a_adam, n, conv_fft_len, corr_fft_len, deadline
            )
            a_ref = _symmetrize(a_ref)
            val = evaluate_sequence(a_ref.tolist())
            if val < best_val:
                best_val = val
                best_seq = a_ref.copy()
                print(f"[final refinement] improved best to {best_val:.6f}")
        except Exception as e:
            print(f"[final refinement] exception: {e}")

    # ----------------------------------------------------------------------
    # Safety fallback (should never be needed)
    # ----------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = evaluate_sequence(best_seq.tolist())

    print(f"\nFinished – best value {best_val:.6f} with length {len(best_seq)}")
    return best_seq.tolist()
```

This script defines all required helper functions and a `propose_candidate` implementation that respects the time budget, exploits gradient‑based optimisation, adds a convex‑programming refinement step, and leverages diverse seeding & jittering to escape local minima. The final returned list complies with the evaluator’s expectations.
