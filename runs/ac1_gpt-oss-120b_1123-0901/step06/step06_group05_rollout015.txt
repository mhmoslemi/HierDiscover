"""
Propose a high‑quality step‑function (non‑negative heights) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

where * denotes linear convolution.
The search combines:
  • a rich pool of parametric initial shapes (Beta, power‑law, generalized Gaussian,
    mixtures, classic windows, random Dirichlet);
  • a quick pre‑selection based on the exact objective;
  • a two‑phase continuous optimisation (Adam on a smooth‑max surrogate +
    exponentiated‑gradient sub‑gradient on the true max);
  • a final cheap stochastic hill‑climb.
The algorithm runs for the supplied time budget (≈1000 s) and returns the best
sequence found as a plain Python list.  All vectors are kept on the simplex
(sum = 1), thus the denominator of F is constant and the objective reduces to
`2·n·max(conv)`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]
    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative vector to ``new_n`` points by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:                     # protect against a zero vector
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s


# ----------------------------------------------------------------------
# Parametric families – all return a **probability vector** (sum = 1)
# ----------------------------------------------------------------------
def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric or asymmetric Beta(α,β) pdf sampled on {@0,…,n‑1}."""
    x = (np.arange(n) + 0.5) / n                # avoid exact 0/1
    shape = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s < 1e-12:
        shape = np.random.dirichlet(np.ones(n))
        s = shape.sum()
    return shape / s


def _powerlaw_shape(n: int, alpha: float) -> np.ndarray:
    """Shape ∝ (i·(n‑i+1))^α, i = 1…n."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s < 1e-12:
        shape = np.random.dirichlet(np.ones(n))
        s = shape.sum()
    return shape / s


def _gen_gaussian_shape(n: int, sigma: float, p: float = 2.0) -> np.ndarray:
    """Generalised Gaussian ∝ exp( -|x‑μ|^p / (2·σ^p) )."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    shape = np.exp(-0.5 * np.abs((i - mu) / sigma) ** p)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s < 1e-12:
        shape = np.random.dirichlet(np.ones(n))
        s = shape.sum()
    return shape / s


def _two_gaussian_shape(n: int, sigma: float, p: float, offset: float) -> np.ndarray:
    """Two symmetric bumps centred at μ±offset."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    left = np.exp(-0.5 * np.abs((i - (mu - offset)) / sigma) ** p)
    right = np.exp(-0.5 * np.abs((i - (mu + offset)) / sigma) ** p)
    shape = left + right
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s < 1e-12:
        shape = np.random.dirichlet(np.ones(n))
        s = shape.sum()
    return shape / s


# ----------------------------------------------------------------------
# Scoring (exact objective, sum assumed to be 1)
# ----------------------------------------------------------------------
def _exact_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) for a vector whose entries sum to 1."""
    n = a.shape[0]
    conv_fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Candidate pool generation (diverse families + warm‑starts)
# ----------------------------------------------------------------------
def _generate_initial_candidates(n: int, prev_best):
    """Return a list of normalized arrays to be used as starting points."""
    cand = []

    # 1. warm‑start from the best sequence found so far
    if prev_best is not None:
        cand.append(_upsample(prev_best, n))

    # 2. historic variable if present
    h = globals().get('height_sequence_1')
    if isinstance(h, (list, np.ndarray)):
        h = np.asarray(h, dtype=np.float64)
        cand.append(_upsample(h, n))

    # 3. classic deterministic windows
    i = np.arange(n, dtype=np.float64)
    hann = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    cand.append(hann / hann.sum())

    sine = np.sin(np.pi * i / (n - 1))
    cand.append(sine / sine.sum())

    # 4. Random Dirichlet families (different concentrations)
    for conc in (0.2, 0.5, 0.8, 1.0, 2.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 5. Power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.1, 7):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # 6. Symmetric exponential tails
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        cand.append(shape)

    # 7. Parametric families ------------------------------------------------
    #    – Beta distributions (both symmetric and skewed)
    for alpha in np.logspace(-0.5, 0.7, 5):          # roughly 0.3 … 5
        for beta in np.logspace(-0.5, 0.7, 5):
            cand.append(_beta_shape(n, alpha, beta))

    #    – Generalised Gaussians (different widths & exponents)
    for sigma in (n / 10.0, n / 6.0, n / 4.0):
        for p in (0.8, 1.0, 1.2, 1.5):
            cand.append(_gen_gaussian_shape(n, sigma, p))

    #    – Mixture of two bumps (offset = 0.2·n … 0.4·n)
    for sigma in (n / 12.0, n / 8.0):
        for p in (1.0, 1.5):
            for off in (n * 0.2, n * 0.3, n * 0.4):
                cand.append(_two_gaussian_shape(n, sigma, p, off))

    # Normalise & discard pathological entries
    cleaned = []
    for c in cand:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        cleaned.append(c / s)
    return cleaned


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _run_adam(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.30
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float('inf')
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # soft‑max → point on the simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # exact objective (denominator = 1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

        # gradient wrt a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # back‑prop through soft‑max (logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e4)

        # occasional random shake if stuck
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient sub‑gradient phase (exact max)
# ----------------------------------------------------------------------
def _run_subgrad(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    last_improve = time.time()
    eta0 = 0.55

    for it in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it}] new best {val:.6f}")

        # active set where max is attained
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (classical EG)
        eta = eta0 / np.sqrt(it)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Very cheap stochastic local search – moves a small mass ε
# ----------------------------------------------------------------------
def _local_hill_climb(a0: np.ndarray, n: int, deadline: float,
                     max_trials: int = 200_000):
    conv_fft_len = _next_pow2(2 * n - 1)
    best_a = a0.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    no_improve = 0
    trials = 0
    while time.time() < deadline and trials < max_trials and no_improve < 5_000:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0.0, min(0.05, best_a[i]))
        if eps == 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = cand
            no_improve = 0
            print(f"[n={n} HillClimb] improved to {val:.6f}")
        else:
            no_improve += 1
        trials += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point ----------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000.0):
    """
    Search for a non‑negative step‑function that minimises the evaluation metric.
    Returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    margin = 0.5                     # safety margin before the hard deadline
    deadline = time.time() + budget_s - margin

    best_seq = None
    best_val = float('inf')

    # Lengths to explore – larger n gives more expressive power but costs more.
    lengths = [2048, 4096, 8192, 16384, 32768]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # ------------------------------------------------------------------
        # 1) Build a large variety of raw candidates (all sum to 1)
        # ------------------------------------------------------------------
        raw_cands = _generate_initial_candidates(n, best_seq)

        # ------------------------------------------------------------------
        # 2) Quick pre‑selection: keep the K best according to the exact score
        # ------------------------------------------------------------------
        scores = [_exact_score(c) for c in raw_cands]
        order = np.argsort(scores)
        K = min(5, len(raw_cands))                     # at most 5 heavy runs
        top_idxs = order[:K]
        candidates = [raw_cands[i] for i in top_idxs]

        # ------------------------------------------------------------------
        # 3) Heavy continuous optimisation on the selected candidates
        # ------------------------------------------------------------------
        # iteration budgets – scale roughly with 1/n so total runtime stays bounded
        max_adam = int(26000 * 2048 / n) + 9_000      # a bit larger than before
        max_subgrad = 9_000

        for cid, init in enumerate(candidates):
            if time.time() > deadline:
                break

            # ---- Adam (smooth‑max) ----
            a_adam, val_adam = _run_adam(init, n, deadline, max_adam)
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n} cand={cid}] Adam new global best {val_adam:.6f}")

            # ---- Sub‑gradient (exact max) ----
            a_sub, val_sub = _run_subgrad(a_adam, n, deadline, max_subgrad)
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[n={n} cand={cid}] Subgrad new global best {val_sub:.6f}")

        # ------------------------------------------------------------------
        # 4) Short fine‑tune of the (currently) best sequence for this length
        # ------------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            a_fine, val_fine = _run_adam(best_seq, n, deadline,
                                         int(0.6 * max_adam))
            if val_fine < best_val:
                best_val = val_fine
                best_seq = a_fine.copy()
                print(f"[n={n}] Fine‑tune improved global best to {val_fine:.6f}")

    # ----------------------------------------------------------------------
    # 5) Final cheap hill‑climb (use a generous trial budget if time permits)
    # ----------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print(f"\n=== Final local hill‑climb (n={len(best_seq)}) ===")
        remaining = deadline - time.time()
        # allocate at most half the remaining time to hill‑climb
        max_trials = int(2e5 * (remaining / budget_s))
        a_final, val_final = _local_hill_climb(best_seq, len(best_seq), deadline,
                                               max_trials=max_trials)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final
            print(f"[final] hill‑climb improved global best to {val_final:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ----------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
