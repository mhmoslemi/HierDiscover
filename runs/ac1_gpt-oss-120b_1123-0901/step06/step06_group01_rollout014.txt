```
def compute_objective(seq: np.ndarray) -> float:
    n = seq.shape[0]
    # ensure positive
    a = np.maximum(seq, 0.0)
    # compute max convolution using FFT
    Nfft = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[:2*n-1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float('inf')
    return 2 * n * max_b / (sum_a * sum_a)
```

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

The search proceeds in three stages:

1. **Parametric exploration** – A handful of simple symmetric families
   (Gaussian, exponential, cosine‑power, polynomial, beta‑pdf) are sampled
   over a modest grid of shape parameters for a few candidate lengths.
   Each candidate is normalised to unit ℓ₁‑norm and its exact objective
   (2·n·max(a∗a)/‖a‖₁²) is computed via a single FFT.  The best shape
   across all lengths is retained.

2. **Refinement by first‑order optimisation** – Starting from the best
   parametric shape, two complementary optimisers are run:
   * Projected Gradient Descent on the exact max‑convolution (hard max).
   * Adam optimiser on a soft‑max smoothed max‑convolution (exponentiated‑gradient).
   Both work on the simplex {a_i≥0, Σa_i=√(2 n)} and keep the best vector
   ever seen.

3. **Cheap stochastic polishing** – A simulated‑annealing hill‑climber
   repeatedly transfers a small random fraction of mass between two
   coordinates, accepting improvements always and occasional worsening
   moves with a temperature schedule.

All phases respect the 1000 s budget (a 5 s safety margin is kept).
The final best sequence is returned as a plain Python list.
"""

def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x_i≥0, Σx_i = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)

    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Exact max of a∗a (linear convolution) using a single FFT."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def compute_objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum(seq)² (no safety‑checks)."""
    a = np.maximum(seq, 0.0)
    n = a.shape[0]
    if a.sum() < 1e-12:
        return float("inf")
    Nfft = 1 << ((2 * n - 1).bit_length())
    maxb = _max_convolution(a, Nfft)
    return 2.0 * n * maxb / (a.sum() ** 2)


# ----------------------------------------------------------------------
# Parametric families ----------------------------------------------------


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a


def shape_exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    return a


def shape_cosine_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.cos(np.pi * (i - centre) / (n - 1))
    a = np.maximum(a, 0.0) ** p
    return a


def shape_polynomial(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n)
    a = (i * (n - 1 - i)) ** alpha
    return a


def shape_beta_symmetric(n: int, alpha: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n)
    a = (x ** alpha) * ((1.0 - x) ** alpha)
    return a

# Optimisers -----------------------------------------------------------


def _pgd_optimize(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  max_iters: int,
                  best_seq: np.ndarray,
                  best_score: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Projected Gradient Descent on the exact max‑convolution."""
    target_sum = math.sqrt(2.0 * n)

    a = a0.copy()
    a = a * target_sum / a.sum()

    Nfft = 1 << ((3 * n - 2).bit_length())

    step0 = 0.001 * target_sum
    eps_active = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # forward convolution (max)
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # keep global best
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD n={n:4d} it={it:7d}] new best max={best_score:.6f}")

        # active set (hard max)
        mask = conv >= M - eps_active
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = mask.astype(float) / float(cnt)

        # weighted reverse convolution (sub‑gradient)
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[: len(w) + n - 1]
        idx_start = 2 * n - 2
        rev = conv_rev[idx_start - np.arange(n)]

        g = 2.0 * rev

        # gradient step + projection
        lr = step0 / math.sqrt(it)
        a_tmp = a - lr * g
        a = _project_onto_simplex(a_tmp, target_sum)

        # occasional jitter
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)

    return best_seq, best_score


def _exp_gradient_optimize(a0: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int,
                           best_seq: np.ndarray,
                           best_score: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient Adam on a soft‑max approximation of the max‑conv."""
    target_sum = math.sqrt(2.0 * n)

    # initialise logits
    eps = 1e-16
    y = np.log(a0 + eps)

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.08
    jitter_interval = 3000

    tau_factor = 0.30
    decay_rate = max_iters / 8.0

    Nfft = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current vector on simplex
        logits = y - np.max(y)
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()
        a = target_sum * p

        # forward conv
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[Adam n={n:4d} it={it:7d}] new best max={best_score:.6f}")

        # soft‑max weights
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
        else:
            maxc = conv.max()
            ex = np.exp((conv - maxc) / tau)
            w = ex / ex.sum()

        # reverse conv for gradient wrt a
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[: len(w) + n - 1]
        idx_start = 2 * n - 2
        rev = conv_rev[idx_start - np.arange(n)]

        g_a = 2.0 * rev

        # mirror‑descent to logits
        avg = np.dot(p, g_a)
        g_y = target_sum * p * (g_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        if it % jitter_interval == 0:
            jitter = rng.normal(scale=0.01, size=n)
            y += jitter

    return best_seq, best_score


def _local_refine(a: np.ndarray,
                  n: int,
                  deadline: float,
                  best_score: float,
                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simple stochastic hill‑climbing with simulated annealing."""
    cur_seq = a.copy()
    cur_score = best_score
    Nfft = 1 << ((2 * n - 1).bit_length())

    max_steps = 200_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # max‑convolution
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (plain Python list) found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Load any previously discovered construction if available
    best_seq = None
    best_val = float("inf")
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        base_val = compute_objective(base)
        best_seq, best_val = base, base_val
        print("[init] loaded previous best, obj =", base_val)
    except Exception:
        base = None

    # ------------------------------------------------------------------
    # Choose a modest set of lengths to explore (around the historically good region)
    lengths = [1800, 1890, 2000, 2100, 2400]
    # ------------------------------------------------------------------
    # Stage 1 – parametric sweep
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[stage‑1] exploring length {n}")

        # pre‑compute FFT size for objective evaluations
        Nfft_obj = 1 << ((2 * n - 1).bit_length())

        def eval_shape(arr):
            a = np.maximum(arr, 0.0)
            if a.sum() == 0:
                return float("inf")
            # normalise to ℓ₁=1 for comparison (objective invariant to scaling)
            a = a / a.sum()
            maxb = _max_convolution(a, Nfft_obj)
            return 2.0 * n * maxb  # denominator = 1²

        # grids for the five families
        centre = (n - 1) / 2.0
        # 1) Gaussian
        sigma_vals = np.geomspace(n / 200.0, n / 2.0, num=12)
        for sigma in sigma_vals:
            a = shape_gaussian(n, sigma)
            val = eval_shape(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[param] Gaussian sigma={sigma:.2f} → obj={val:.6f}")

        # 2) Exponential (symmetric)
        rate_vals = np.geomspace(5e-4, 2e-2, num=12)
        for rate in rate_vals:
            a = shape_exponential(n, rate)
            val = eval_shape(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[param] Exponential rate={rate:.5f} → obj={val:.6f}")

        # 3) Cosine‑power
        p_vals = [0.5, 1, 2, 4, 8, 16]
        for p in p_vals:
            a = shape_cosine_power(n, p)
            val = eval_shape(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[param] Cosine^p p={p} → obj={val:.6f}")

        # 4) Polynomial i·(n‑1‑i) raised to α
        alpha_vals = [0.3, 0.5, 0.8, 1, 1.5, 2, 3]
        for alpha in alpha_vals:
            a = shape_polynomial(n, alpha)
            val = eval_shape(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[param] Polynomial α={alpha} → obj={val:.6f}")

        # 5) Symmetric beta‑pdf (x·(1‑x))^α
        for alpha in alpha_vals:
            a = shape_beta_symmetric(n, alpha)
            val = eval_shape(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[param] Beta α={alpha} → obj={val:.6f}")

        # Occasionally supplement with a sparse random spike pattern
        k = max(2, min(12, n // 150))
        idx = rng.choice(n, size=k, replace=False)
        vals = rng.random(k)
        vals = vals / vals.sum()
        sparse = np.zeros(n)
        sparse[idx] = vals
        val = eval_shape(sparse)
        if val < best_val:
            best_val, best_seq = val, sparse.copy()
            print(f"[param] Sparse spikes → obj={val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – first‑order refinement (PGD + Adam) on the best shape
    if best_seq is not None and time.time() < deadline:
        # ensure we work on the chosen length of the best sequence
        n = best_seq.shape[0]
        target_sum = math.sqrt(2.0 * n)

        # normalise to the required ℓ₁‑norm for the optimisers
        a0 = best_seq.copy()
        a0 = a0 * target_sum / a0.sum()

        # allocate iteration budget roughly proportional to remaining time
        remaining = deadline - time.time()
        # estimate 0.0005 s per iteration (conservative)
        max_iters = max(100_000, int(remaining / 0.0005))

        print(f"[stage‑2] refinement on length {n}, iters≈{max_iters}")

        # Adam (exponentiated‑gradient)
        best_seq, best_score = _exp_gradient_optimize(
            a0=a0.copy(),
            n=n,
            deadline=deadline,
            max_iters=max_iters // 2,
            best_seq=best_seq,
            best_score=best_val,   # note: best_val stores 2·n·maxb, not the max itself
            rng=rng,
        )
        # Convert back to the original scaling for the next stage
        best_val = 2.0 * n * best_score / (target_sum ** 2)

        # PGD (hard max)
        best_seq, best_score = _pgd_optimize(
            a0=best_seq.copy(),
            n=n,
            deadline=deadline,
            max_iters=max_iters // 2,
            best_seq=best_seq,
            best_score=best_score,
            rng=rng,
        )
        best_val = 2.0 * n * best_score / (target_sum ** 2)

    # ------------------------------------------------------------------
    # Stage 3 – cheap stochastic polishing (simulated annealing)
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        # allocate at most half of the remaining budget (capped at 30 s)
        remaining = deadline - time.time()
        refine_deadline = time.time() + min(remaining * 0.5, 30.0)
        best_seq, best_score = _local_refine(
            a=best_seq,
            n=n,
            deadline=refine_deadline,
            best_score=best_score,
            rng=rng,
        )
        best_val = 2.0 * n * best_score / (np.sum(best_seq) ** 2)

    # ------------------------------------------------------------------
    # Fallback – uniform if nothing was ever produced
    if best_seq is None:
        n = 2000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_val = compute_objective(best_seq)
        print("[fallback] using uniform sequence")

    print(f"[done] best objective = {best_val:.6f}, length = {best_seq.shape[0]}")
    # Return as plain Python list (evaluate_sequence expects a list)
    return best_seq.tolist()
