"""
Propose a non‑negative step‑function (list of heights) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

where a*a is the linear convolution (numpy.convolve).

The implementation follows a very lightweight strategy:

* If a previously discovered sequence is available in the global variable
  ``height_sequence_1`` we start from it, otherwise we fall back to a uniform
  vector.
* A fast FFT‑based routine computes the objective (the same formula that
  ``evaluate_sequence`` uses, but without the heavy validation).
* A simple simulated‑annealing / hill‑climbing loop performs many tiny
  mass‑redistribution moves while respecting the non‑negativity constraint.
  The loop stops when the time budget (default 1000 s) is exhausted.
* The best sequence found is returned as a plain Python list.

The search is deliberately kept lightweight to guarantee a result well within
the 1000 s budget while still being able to improve on the initial guess.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
def _objective_fft(a: np.ndarray) -> float:
    """Exact evaluation of the objective 2·n·max(conv)/sum(a)² using FFT."""
    n = a.shape[0]
    # Length of linear convolution output
    conv_len = 2 * n - 1
    # Next power‑of‑two for the FFT (fast & exact for zero‑padded signals)
    fft_len = 1 << int(np.ceil(np.log2(conv_len)))
    # Linear convolution via FFT
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:conv_len]
    max_c = conv.max()
    sum_a = a.sum()
    # Guard against the (unlikely) zero‑sum case – the external
    # ``evaluate_sequence`` returns ``inf`` there.
    if sum_a < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (sum_a ** 2)


# ----------------------------------------------------------------------
def _initial_sequence(n: int) -> np.ndarray:
    """
    Return a reasonable starting point of length ``n``.
    Preference order:
    1. Previously found global best (``height_sequence_1``) if present.
    2. Uniform vector (the theoretical optimum under the non‑negative
       constraint).
    """
    # Try to reuse the globally stored best sequence
    if "height_sequence_1" in globals():
        h = np.asarray(height_sequence_1, dtype=np.float64)
        if h.size > n:
            h = h[:n]
        else:
            h = np.pad(h, (0, n - h.size))
        h = np.maximum(h, 0.0)               # enforce non‑negativity
        if h.sum() < 1e-8:                    # degenerate – fall back to uniform
            return np.full(n, 1.0 / n, dtype=np.float64)
        return h / h.sum()                    # normalise to sum‑one
    # Uniform fallback
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                deadline: float,
                max_iters: int = 200_000,
                init_temp: float = 1.0,
                min_temp: float = 1e-4,
                decay: float = 0.9995) -> tuple[np.ndarray, float]:
    """
    Stochastic local search that repeatedly moves a small amount of mass from
    one index to another while preserving the total sum.
    The move is accepted if it improves the objective, otherwise it may be
    accepted with a probability that decreases with a simulated‑annealing
    temperature schedule.
    Returns the best sequence and its objective value found during the run.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective_fft(a)
    cur_val = best_val
    n = a.shape[0]

    # Pre‑compute a small epsilon to avoid divisions by zero
    eps = 1e-12

    # Temperature schedule
    temp = init_temp

    # For efficient random choice of source indices with non‑zero mass
    # we keep a list of candidate indices that currently hold >eps.
    nonzero_idx = np.where(a > eps)[0]

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # Choose a source (with some mass) and a destination (different)
        src = np.random.choice(nonzero_idx)
        dst = np.random.randint(0, n - 1)
        if dst >= src:
            dst += 1

        # Transfer a random fraction of the source's mass (up to 10 % by default)
        max_transfer = min(0.1 * a[src], a[src])   # never more than what's available
        if max_transfer < eps:
            continue
        delta = max_transfer * np.random.rand()

        # Propose the new vector (mass conservation)
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        # Numerical safety – enforce non‑negativity (tiny negative values may appear)
        a_new = np.maximum(a_new, 0.0)

        # Re‑normalise to keep the sum exactly the same (helps with round‑off)
        s = a_new.sum()
        if s < eps:               # degenerate – skip this move
            continue
        a_new /= s

        # Evaluate
        new_val = _objective_fft(a_new)

        # Acceptance test
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Simulated‑annealing acceptance probability
            prob = math.exp(-(new_val - cur_val) / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_new
            cur_val = new_val
            # Update the non‑zero index list (cheap update)
            if a[src] <= eps:
                nonzero_idx = np.setdiff1d(nonzero_idx, [src], assume_unique=True)
            if a[dst] > eps and dst not in nonzero_idx:
                nonzero_idx = np.append(nonzero_idx, dst)

            # Track the best ever seen
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()

        # Temperature decay (geometric)
        temp = max(temp * decay, min_temp)

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence that minimises the evaluation score.
    The routine respects the supplied time budget and returns the best sequence
    found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin for the caller

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Explore a few different problem sizes.  Larger n gives a larger
    # search space, but each convolution gets more expensive (≈ n log n).
    # The set below is a compromise that fits comfortably into a 1000 s
    # budget on a typical 2‑core machine.
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n--- Optimising length n = {n} ---")
        # Initialise a sensible starting point (global best or uniform)
        a0 = _initial_sequence(n)

        # Evaluate the start
        val0 = _objective_fft(a0)
        print(f"  start value: {val0:.6f}")

        if val0 < best_val:
            best_val = val0
            best_seq = a0.copy()
            print(f"  ** new global best (start) {best_val:.6f}")

        # ----------------------------------------------------------------
        # Local stochastic optimisation (hill‑climbing / simulated annealing)
        # The loop respects the overall deadline.
        a_opt, val_opt = _hill_climb(a0, deadline)

        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"  ** new global best (local) {best_val:.6f}")

        # ----------------------------------------------------------------
        # Occasionally try a fresh random Dirichlet start – cheap exploration.
        # Only a few attempts per length are performed to keep the runtime low.
        for _ in range(3):
            if time.time() > deadline:
                break
            a_rand = np.random.dirichlet(np.ones(n))
            val_rand = _objective_fft(a_rand)
            if val_rand < best_val:
                best_val = val_rand
                best_seq = a_rand.copy()
                print(f"  ** new global best (rand) {best_val:.6f}")

            # One extra hill‑climb from this random point
            a_opt2, val_opt2 = _hill_climb(a_rand, deadline)
            if val_opt2 < best_val:
                best_val = val_opt2
                best_seq = a_opt2.copy()
                print(f"  ** new global best (rand‑local) {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed because the loop always creates a
    # candidate, but it guards against pathological early termination.
    if best_seq is None:
        best_seq = np.full(1024, 1.0 / 1024.0, dtype=np.float64)

    # Return a plain list as required by the external evaluator.
    return best_seq.tolist()
