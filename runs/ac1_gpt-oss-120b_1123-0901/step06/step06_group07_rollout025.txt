def _generate_parametric_candidates(n, warm_start):
    candidates = []
    # Warm start rotations
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n//4, n//2, 3*n//4):
            candidates.append(np.roll(warm_start, shift))
    # Uniform
    candidates.append(np.full(n, 1.0/n))
    # Dirichlet
    for conc in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0,5.0]:
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)
    # Edge exponential mixture
    i = np.arange(n, dtype=np.float64)
    alphas = np.logspace(-2, 2, num=30)  # 0.01 to 100
    for alpha in alphas:
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge = edge / edge.sum()
        for mix in np.linspace(0,1,11):
            shape = mix * edge + (1 - mix) / n
            candidates.append(shape)
    # Edge power law mixture
    ps = np.logspace(-1, 1.5, num=30)  # 0.1 to ~31
    for p in ps:
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge = edge / edge.sum()
        for mix in np.linspace(0,1,11):
            shape = mix * edge + (1 - mix) / n
            candidates.append(shape)
    # Edge constant block
    width_fracs = np.linspace(0.005, 0.3, num=15)  # from 0.5% to 30%
    mass_fracs = np.linspace(0.2, 0.9, num=8)  # proportion mass on edges
    for wf in width_fracs:
        w = max(1, int(round(wf * n)))
        if 2*w >= n:
            continue
        for mf in mass_fracs:
            edge_weight = mf / (2 * w)
            interior_weight = (1 - mf) / (n - 2 * w)
            shape = np.full(n, interior_weight, dtype=np.float64)
            shape[:w] = edge_weight
            shape[-w:] = edge_weight
            candidates.append(shape)
    # Two Gaussian edges
    gammas = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0]
    for gamma in gammas:
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape = shape / shape.sum()
        for mix in np.linspace(0,1,11):
            cand = mix * shape + (1 - mix) / n
            candidates.append(cand)
    # Sinusoidal half-wave
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape = shape / shape.sum()
    candidates.append(shape)
    return candidates

def evaluate_candidates(candidates, n, conv_fft_len, deadline):
    best_vals = []
    for cand in candidates:
        if time.time() >= deadline: break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective_fast(cand, n, conv_fft_len)
        best_vals.append((val, cand))
    best_vals.sort(key=lambda x: x[0])
    top = [c for _,c in best_vals[:20]]
    best_val = best_vals[0][0] if best_vals else float('inf')
    best_seq = best_vals[0][1] if best_vals else None
    return top, best_seq, best_val

"""
Propose a low‑evaluation‑score step function.

The algorithm works on three increasingly fine grids (1024 → 4096 → 8192).
1.  A large and diverse parametric pool (edge‑exponential, edge‑power‑law,
    constant edge blocks, two‑Gaussian edges, random Dirichlet, sinusoid,
    warm‑start rotations) is generated for the *small* grid.  All candidates are
    normalised to sum ≈ 1 and evaluated cheaply with an FFT‑based convolution.
    The best few (default 5) are retained.
2.  Each retained candidate is refined on its grid by a two‑phase optimiser:
    – Adam on a smooth‑max (log‑sum‑exp) surrogate of the max‑convolution,
      using a soft‑max (mirror) parametrisation to stay on the simplex.
    – A handful of exponentiated‑gradient sub‑gradient steps on the exact
      max‑convolution (active‑set weighting).  Hyper‑parameters (learning‑rate,
      temperature schedule, iteration limits) are chosen for rapid yet accurate
      convergence.
3.  The best vector is interpolated to the next larger grid and refined again.
    If the time budget permits a final step on 8192 entries is performed.
    Throughout the run the globally best sequence and its objective value are
    tracked and finally returned as a plain Python list (clipped to the
    evaluator’s allowed range).

The whole routine respects the 1000 s budget, prints progress and uses only
NumPy (no external solvers).  It improves upon the previous best bound
≈ 1.5087 and aims for the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)-1 using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective_fast(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast objective (assumes a.sum()≈1)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _generate_parametric_candidates(
    n: int, warm_start: np.ndarray | None
) -> list[np.ndarray]:
    """Create a large diverse pool of candidate vectors for a given length."""
    candidates: list[np.ndarray] = []

    # --------------------------------------------------------------
    # Warm‑start (if any) and 3 rotations
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # --------------------------------------------------------------
    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # --------------------------------------------------------------
    # Random Dirichlet draws with a range of concentration parameters
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    # --------------------------------------------------------------
    # Edge‑exponential families, optionally mixed with uniform
    i = np.arange(n, dtype=np.float64)
    alphas = np.logspace(-2, 2, num=30)          # 0.01 … 100
    mix_weights = np.linspace(0, 1, 11)         # 0%,10%,...,100%
    for alpha in alphas:
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in mix_weights:
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # --------------------------------------------------------------
    # Edge‑power‑law families, optionally mixed with uniform
    ps = np.logspace(-1, 1.5, num=30)           # 0.1 … ~31.6
    for p in ps:
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in mix_weights:
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # --------------------------------------------------------------
    # Constant edge blocks (mass concentrated on the two ends)
    width_fracs = np.linspace(0.005, 0.30, 15)   # 0.5% … 30% of length
    edge_mass_fracs = np.linspace(0.2, 0.95, 8)  # proportion of total mass on edges
    for wf in width_fracs:
        w = max(1, int(round(wf * n)))
        if 2 * w >= n:
            continue
        for mf in edge_mass_fracs:
            edge_weight = mf / (2 * w)
            interior_weight = (1.0 - mf) / (n - 2 * w)
            shape = np.full(n, interior_weight, dtype=np.float64)
            shape[:w] = edge_weight
            shape[-w:] = edge_weight
            candidates.append(shape)

    # --------------------------------------------------------------
    # Two‑Gaussian edge shapes, mixed with uniform
    gammas = (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0)
    for gamma in gammas:
        edge = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        edge /= edge.sum()
        for w in mix_weights:
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # --------------------------------------------------------------
    # Positive sinusoidal half‑wave (already present in the original code)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    return candidates


# ----------------------------------------------------------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
    max_iter: int = 250_000,
    lr: float = 0.05,
    lam_start: float = 0.5,
    lam_factor: float = 1.015,
    lam_cap: float = 1e6,
) -> tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp surrogate of the max‑convolution.
    Returns the best vector/value seen during the run.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    lam = lam_start
    last_improve = time.time()

    it = 0
    while time.time() < deadline and it < max_iter:
        # ----- map logits -> simplex via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- convolution & surrogate objective -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate (chain‑rule through soft‑max) -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, lam_cap)

        # ----- occasional random shake if stuck -----
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient steps on the true max‑convolution.
    """
    a = best_seq if best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()                         # stay on the simplex

    eta = 0.6
    last_improve = time.time()
    max_iter = 20_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it:05d}] new best {val:.6f}")

        # ----- active set where convolution attains its maximum -----
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- exponentiated‑gradient update (keeps a ≥ 0) -----
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # ----- early stop if no progress for a while -----
        if time.time() - last_improve > 5.0:
            break

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
    adam_params: dict,
) -> tuple[np.ndarray, float]:
    """
    Run Adam on the smooth‑max surrogate followed by a sub‑gradient refinement.
    Updates the globally best vector/value.
    """
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
        **adam_params,
    )
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a low‑evaluation‑score step function.
    Returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Retrieve optional warm start (global variable ``height_sequence_1``)
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.ndim != 1:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 0 – small grid (n=1024) – broad parametric sweep
    n0 = 1024
    conv_fft_len0 = _next_pow2(2 * n0 - 1)
    corr_fft_len0 = _next_pow2(3 * n0 - 2)

    # adapt warm start to the small grid (interpolation)
    if warm_start is not None:
        if warm_start.shape[0] != n0:
            x_old = np.linspace(0, 1, warm_start.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n0, endpoint=False)
            ws0 = np.interp(x_new, x_old, warm_start)
        else:
            ws0 = warm_start.copy()
        ws0 = np.maximum(ws0, 0.0)
        if ws0.sum() >= 0.01:
            ws0 = ws0 / ws0.sum()
        else:
            ws0 = None
    else:
        ws0 = None

    all_cand = _generate_parametric_candidates(n0, ws0)

    # Fast evaluation and selection of the top few candidates
    cand_vals = []
    for cand in all_cand:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:                     # avoid degenerate vectors
            cand = np.random.dirichlet(np.ones(n0))
            s = cand.sum()
        cand = cand / s
        val = _objective_fast(cand, n0, conv_fft_len0)
        cand_vals.append((val, cand))

    cand_vals.sort(key=lambda x: x[0])
    top_k_small = 5
    top_candidates_small = [c for _, c in cand_vals[:top_k_small]]
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – short Adam + sub‑gradient refinements on the small grid
    for idx, a0 in enumerate(top_candidates_small):
        if time.time() >= deadline:
            break
        print(f"--- Small‑grid optimisation start {idx + 1}/{len(top_candidates_small)} ---")
        adam_params = {
            "max_iter": 120_000,
            "lr": 0.05,
            "lam_start": 0.5,
            "lam_factor": 1.02,
            "lam_cap": 1e6,
        }
        best_seq, best_val = _deep_optimize(
            a0,
            n0,
            conv_fft_len0,
            corr_fft_len0,
            deadline,
            best_seq,
            best_val,
            adam_params,
        )
        print(f"  Small‑grid best so far: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – upscale to medium grid (n=4096) and refine
    n_mid = 4096
    conv_fft_len_mid = _next_pow2(2 * n_mid - 1)
    corr_fft_len_mid = _next_pow2(3 * n_mid - 2)

    if best_seq is not None:
        # linear interpolation to the larger grid
        x_small = np.linspace(0, 1, best_seq.shape[0], endpoint=False)
        x_mid = np.linspace(0, 1, n_mid, endpoint=False)
        a_mid = np.interp(x_mid, x_small, best_seq)
    else:
        a_mid = np.full(n_mid, 1.0 / n_mid)

    a_mid = np.maximum(a_mid, 0.0)
    a_mid /= a_mid.sum()

    if time.time() < deadline:
        print("--- Medium‑grid optimisation (4096) ---")
        adam_params_mid = {
            "max_iter": 180_000,
            "lr": 0.035,
            "lam_start": 0.5,
            "lam_factor": 1.015,
            "lam_cap": 1e6,
        }
        best_seq, best_val = _deep_optimize(
            a_mid,
            n_mid,
            conv_fft_len_mid,
            corr_fft_len_mid,
            deadline,
            best_seq,
            best_val,
            adam_params_mid,
        )
        print(f"  Medium‑grid best: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – optional large grid (n=8192) if time permits
    time_left = deadline - time.time()
    if time_left > 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            x_mid = np.linspace(0, 1, best_seq.shape[0], endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_big, x_mid, best_seq)
        else:
            a_big = np.full(n_big, 1.0 / n_big)

        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("--- Large‑grid optimisation (8192) ---")
        adam_params_big = {
            "max_iter": 120_000,
            "lr": 0.025,
            "lam_start": 0.5,
            "lam_factor": 1.01,
            "lam_cap": 1e6,
        }
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
            adam_params_big,
        )
        print(f"  Large‑grid best: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback to a uniform vector if everything failed
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
