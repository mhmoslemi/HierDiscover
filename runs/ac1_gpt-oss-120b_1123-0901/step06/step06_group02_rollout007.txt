"""
Hybrid search for a non‑negative step‑function that minimises
the evaluation score 2·n·max(conv(a,a))/sum(a)².

The algorithm consists of four stages:

1. **Diverse random initialisations** – a large set of parametric families
   (exponential, power‑law, Gaussian, logistic, parabolic, double‑exponential,
   Dirichlet, uniform) with random lengths (1500‑20000).  Each candidate is
   normalised to the required L1‑sum √(2 n) and the best is recorded.

2. **Adam optimisation on a smooth surrogate** – the max‑convolution is
   approximated by a log‑sum‑exp (temperature τ).  The parameters are the
   logits of a soft‑max, guaranteeing the simplex constraint.  τ is annealed
   aggressively to approach the true max.

3. **Exact projected sub‑gradient descent** – we follow the gradient of the
   currently worst convolution entry and project back onto the simplex.

4. **Fine‑grained simulated‑annealing hill‑climb** – tiny mass transfers
   between random positions are accepted with a Metropolis rule.  The
   temperature decays to pinpoint a local optimum.

All stages share a global best solution and respect the overall time budget.
If no candidate is found a uniform sequence is returned as a safe fallback.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of real vectors x and y."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # Clip tiny negatives caused by rounding errors
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]          # sorted descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)     # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1        # zero‑based index of last positive entry
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the log‑sum‑exp (soft‑max) surrogate of the max‑conv.
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    logits: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 5e-5,
    alpha0: float = 0.025,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed‑max surrogate.
    Returns the best coefficient vector (on the simplex) and its exact
    max‑convolution value.
    """
    x = logits.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # ----- map logits to the simplex (a = target_sum * softmax(x)) -----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)               # stable soft‑max
        p = exp_shift / np.sum(exp_shift)           # Σ p = 1
        a = target_sum * p

        # ----- exact max of the convolution (used for logging) -----
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam n={n} new best max={cur_max:.6f}")

        # ----- smoothed max (log‑sum‑exp) weights -----
        tau = max(init_tau_factor * cur_max * (0.99 ** it), min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t. a: 2·a⊛w_rev  -----
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # ----- back‑prop to logits (chain rule) -----
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slow decay of learning rate
        alpha *= 0.9995

    # final exact evaluation of the best vector
    final_val = _full_convolution_fft(best_a, best_a).max()
    return best_a, float(final_val)


# ----------------------------------------------------------------------
# Targeted projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _targeted_subgradient(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent that always follows the current worst index."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step = 0.25 * target_sum / n          # initial step size (a bit larger)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())            # index of current worst peak
        # gradient of conv_k w.r.t. a
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # projected step
        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)

        cand_val = _full_convolution_fft(cand, cand).max()
        if cand_val < best_val - 1e-12:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= 1.02                       # modestly increase step after success
            if it % 200 == 0:
                print(f"[{time.time():.2f}] subgrad it={it} val={best_val:.6f}")
        else:
            step *= 0.70                       # reduce step on failure
            if np.random.rand() < 0.07:        # occasional jitter
                jitter = np.random.rand(n) * 1e-4
                a = _project_onto_simplex(a + jitter, target_sum)

        if best_val < 1.503:   # early stop if we reach the target
            break
    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (simulated annealing) on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Move tiny amounts of mass between positions with a Metropolis rule."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # initial temperature proportional to current peak
    T0 = max(1e-3, best_val * 0.4)
    T_end = 1e-5
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue
        # transfer at most 8 % of mass at i
        delta = a[i] * np.random.rand() * 0.08
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()
        accept = False
        if new_val < best_val:
            accept = True
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")

        # full re‑normalisation every few thousand steps
        if it % 2500 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val

def _random_shape_sequence(n: int) -> np.ndarray:
    """Generate a random non‑negative sequence of length n from a variety of families."""
    shape = np.random.choice([
        "exp", "pow", "gauss", "logistic", "parabolic",
        "double_exp", "uniform", "dirichlet"
    ])
    idx = np.arange(n, dtype=float)

    if shape == "exp":
        tau = np.random.uniform(0.05, 2.0) * n
        a = np.exp(-idx / tau)
        if np.random.rand() < 0.5:
            a = a[::-1]

    elif shape == "pow":
        p = np.random.uniform(0.1, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** -p)
        if np.random.rand() < 0.5:
            a = a[::-1]

    elif shape == "gauss":
        mu = np.random.uniform(0, n - 1)
        sigma = np.random.uniform(0.05, 0.3) * n
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)

    elif shape == "logistic":
        mu = np.random.uniform(0, n - 1)
        scale = np.random.uniform(0.02, 0.2) * n
        a = 1.0 / (1.0 + np.exp((idx - mu) / scale))

    elif shape == "parabolic":
        mu = np.random.uniform(0, n - 1)
        width = np.random.uniform(0.2, 0.6) * n
        a = np.maximum(0.0, 1.0 - ((idx - mu) / width) ** 2)

    elif shape == "double_exp":
        # two exponentials meeting at a random pivot
        pivot = np.random.randint(1, n - 1)
        tau_l = np.random.uniform(0.05, 2.0) * pivot
        tau_r = np.random.uniform(0.05, 2.0) * (n - pivot - 1)
        left = np.exp(-np.arange(pivot) / tau_l)
        right = np.exp(-np.arange(n - pivot - 1)[::-1] / tau_r)
        a = np.concatenate([left, [1.0], right])

    elif shape == "uniform":
        a = np.ones(n, dtype=float)

    else:  # dirichlet
        a = np.random.dirichlet(np.ones(n))

    # add a sprinkle of noise (optional)
    if np.random.rand() < 0.3:
        noise = np.random.randn(n) * np.max(a) * 0.02
        a = np.maximum(a + noise, 0.0)

    # avoid the degenerate all‑zero case
    if a.sum() == 0.0:
        a = np.full(n, 1.0)

    return a
