"""
Propose a step‑function (non‑negative vector) that minimises

        F(a) = 2·n·max(a∗a) / (∑a)²

using a diverse pool of parametrised shapes, many random permutations,
followed by two optimisation phases:

1. Adam on a smooth‑max (log‑sum‑exp) surrogate of the convolution peak.
   The temperature λ grows slowly (factor 1.01) so the landscape stays
   smooth for a long time, allowing the optimiser to escape shallow minima.

2. Projected sub‑gradient descent on the exact max‑convolution.  The
   sub‑gradient is the average of the gradients of all active (tight)
   convolution entries, and each step is projected onto the probability
   simplex (guaranteeing non‑negativity and unit sum).

If time permits the best distribution is interpolated to a larger grid
(8192 points) and the whole refinement is repeated.  The routine respects
the supplied time budget (default 1000 s) and prints progress messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of the objective (sum(a) is assumed > 0)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / v.shape[0]
    else:
        rho = ind[cond].max()
        theta = cssv[cond][ -1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _beta_shape(alpha: float, beta: float, n: int) -> np.ndarray:
    """Beta‑distribution pdf evaluated on {0,…,n‑1} and normalised."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n  # centre of each bin
    # avoid log(0)
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    log_pdf = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x)
    shape = np.exp(log_pdf)
    shape = np.maximum(shape, 0.0)
    if shape.sum() == 0:
        shape = np.full(n, 1.0 / n)
    else:
        shape /= shape.sum()
    return shape


def _exp_mixture(p: float, lam1: float, lam2: float, n: int) -> np.ndarray:
    """Two‑sided exponential mixture, normalised."""
    i = np.arange(n, dtype=np.float64)
    shape = p * np.exp(-lam1 * i) + (1.0 - p) * np.exp(-lam2 * (n - 1 - i))
    shape = np.maximum(shape, 0.0)
    if shape.sum() == 0:
        shape = np.full(n, 1.0 / n)
    else:
        shape /= shape.sum()
    return shape


def _generate_candidate_pool(n: int,
                             warm: np.ndarray | None,
                             rng: np.random.Generator) -> list[np.ndarray]:
    """Construct a diverse set of start vectors (including many permutations)."""
    pool = []

    # ------------------------------------------------------------------
    # 1) Warm‑start and simple rotations
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # ------------------------------------------------------------------
    # 2) Classical families (mostly identical to the reference code)
    pool.append(np.full(n, 1.0 / n))                      # uniform

    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)                         # i·(n‑i+1)
    for alpha in np.linspace(-2.0, -0.2, 12):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    i = np.arange(n, dtype=np.float64)
    for lam in np.linspace(0.1, 3.0, 12):
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        pool.append(shape)

    for sigma in np.linspace(n * 0.03, n * 0.4, 8):
        shape = np.exp(-0.5 * ((i - n / 2) ** 2) / (sigma ** 2))
        shape /= shape.sum()
        pool.append(shape)

    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # ------------------------------------------------------------------
    # 3) New families
    #   a) Beta distributions (both symmetric & asymmetric)
    for a in np.logspace(-1, 1, 6):
        for b in np.logspace(-1, 1, 6):
            shape = _beta_shape(a, b, n)
            pool.append(shape)

    #   b) Two‑sided exponential mixtures
    for _ in range(12):
        p = rng.random()
        lam1 = rng.uniform(0.1, 4.0)
        lam2 = rng.uniform(0.1, 4.0)
        shape = _exp_mixture(p, lam1, lam2, n)
        pool.append(shape)

    #   c) Simple linear ramps (incl. decreasing)
    for slope in np.linspace(-3.0, 3.0, 7):
        shape = np.clip(slope * (np.arange(n) - n / 2) + 1.0, 0, None)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    #   d) Narrow Gaussian bumps (small sigma)
    for sigma in np.linspace(n * 0.005, n * 0.02, 5):
        shape = np.exp(-0.5 * ((i - n / 2) ** 2) / (sigma ** 2))
        shape /= shape.sum()
        pool.append(shape)

    # ------------------------------------------------------------------
    # 4) Random Dirichlet draws (various concentration)
    for conc in np.logspace(-2, 0, 14):
        shape = rng.dirichlet(np.full(n, conc))
        pool.append(shape)

    # ------------------------------------------------------------------
    # 5) Random permutations of the above shapes (adds asymmetry)
    permuted = []
    for base in pool:
        # only keep a limited number of permutations per base shape
        for _ in range(2):
            perm = rng.permutation(n)
            permuted.append(base[perm])
    pool.extend(permuted)

    # ------------------------------------------------------------------
    # 6) A few handcrafted alternating high/low patterns
    for high_frac in (0.1, 0.2, 0.3):
        high_val = high_frac
        low_val = (1.0 - high_val) / (n - int(high_frac * n))
        pattern = np.full(n, low_val)
        hi_indices = rng.choice(n, size=int(high_frac * n), replace=False)
        pattern[hi_indices] = high_val
        pattern = np.maximum(pattern, 0.0)
        pattern /= pattern.sum()
        pool.append(pattern)

    # ------------------------------------------------------------------
    # final sanitise (clip, normalise)
    clean = []
    for vec in pool:
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s < 1e-12:
            continue
        clean.append(vec / s)

    return clean

def _deep_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   global_best_seq: np.ndarray | None,
                   global_best_val: float) -> tuple[np.ndarray, float]:
    """
    Two‑phase refinement:
      1) Adam on a smooth‑max surrogate (log‑sum‑exp);
      2) Projected sub‑gradient on the exact max.
    Global best sequence/value are updated and returned.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5          # temperature for smooth‑max
    lam_factor = 1.01  # slower increase than before
    lam_max = 1e5

    last_improve = time.time()
    it = 0

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smooth‑max
    while time.time() < deadline:
        # soft‑max → simplex (probability vector)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (used for bookkeeping)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights (soft‑argmax)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate w.r.t. a (via chain rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_max)

        # occasional shake when stuck
        if time.time() - last_improve > 5.0:
            logits += (rng.random(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        if it > 400_000:          # safety cap
            break

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient on exact max
    a = (global_best_seq if global_best_seq is not None else a0).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    eta0 = 0.6
    sub_last_improve = time.time()
    max_sub_iters = 30_000

    for sub_it in range(max_sub_iters):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[Subgrad n={n} iter={sub_it:05d}] new best {val:.6f}")

        # active set (indices where conv equals its maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size (standard 1/√t decay)
        eta = eta0 / np.sqrt(sub_it + 1.0)
        a_new = a - eta * grad_a
        # project back onto the simplex
        a = _proj_simplex(a_new)

        # stagnation check
        if time.time() - sub_last_improve > 7.0:
            break

    return global_best_seq, global_best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation metric.
    Returns the best sequence found (plain Python list).  The routine respects a
    time budget (default 1000 s) and prints progress information.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # warm‑start (if present)
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        w = np.asarray(warm_start, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
        else:
            w = None
    else:
        w = None

    # --------------------------------------------------------------
    # generate a broad pool of candidates
    candidates = _generate_candidate_pool(n, w, rng)

    # quick scoring of the pool
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    scored = []  # list of (value, vector)

    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            a = rng.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        val = _objective(a, n, conv_fft_len)
        scored.append((val, a))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        if idx % 50 == 0:
            print(f"[pool {idx:04d}/{len(candidates)}] val={val:.6f}")

    # keep the most promising starters (up to 12)
    scored.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in scored[:12]]

    # --------------------------------------------------------------
    # deep optimisation passes for the top candidates
    for cid, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[optim start {cid}] launching deep optimisation (n={n})")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )
        print(f"[optim start {cid}] best so far {best_val:.6f}")

    # --------------------------------------------------------------
    # Optional up‑sampling to a larger grid if time permits
    if time.time() + 12.0 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_big, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = rng.dirichlet(np.ones(n_big))

        print("[big] starting optimisation on enlarged grid")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        print(f"[big] final best {best_val:.6f}")

    # --------------------------------------------------------------
    # final fallback (should never happen)
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator limits and return as plain list
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # store for potential warm‑starts in later runs
    globals()["height_sequence_1"] = best_seq.tolist()

    return best_seq.tolist()
