def smooth_max_conv(logits, n, conv_fft_len, beta):
    a = softmax(logits)
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    # use log-sum-exp: s = (1/beta) * log(np.sum(np.exp(beta*(conv - max_c)))) + max_c
    # This is numerically stable.
    diff = conv - max_c
    exp_beta = np.exp(beta * diff)
    sum_exp = exp_beta.sum()
    s = max_c + np.log(sum_exp) / beta  # because max_c + (1/beta)log sum exp(beta*(conv-max_c))
    # compute gradient:
    w = exp_beta / sum_exp  # weight per conv index
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)  # same as before but weighted
    # grad w.r.t. logits:
    dot_a_grad = np.dot(a, grad_a)
    grad_log = a * (grad_a - dot_a_grad)
    return s, grad_log

def _project_simplex(v):
    """Project vector v onto probability simplex {x>=0, sum=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho = max {j | u_j - cssv_j / j > 0}
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all <=0
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Multi‑phase optimizer for the step‑function bound.

The pipeline is:
1. Build a diverse pool of start vectors (including a warm start if present,
   Dirichlet samples with various concentration parameters, edge‑concentrated
   power‑law shapes, double‑Gaussian, sinusoid and sparse “spike” patterns).
2. For each start vector run a short Adam optimisation on a smooth‑max
   surrogate (log‑sum‑exp) with a temperature that is increased until the
   surrogate almost equals the true maximum.
3. Refine the current global best with a projected‑gradient sub‑gradient
   method (exact max).  The step size decays as η/√t and each step is
   projected onto the probability simplex.
4. Perform a cheap simulated‑annealing local search that swaps a random
   fraction of mass between two coordinates.
5. If time permits, interpolate the best vector to a finer grid
   (8192 points) and repeat steps 2‑4.

All heavy computations (convolutions, cross‑correlations) use FFTs
(`np.fft.rfft`) for O(n log n) speed.
The algorithm respects the supplied time budget (default 1000 s) and
always returns a valid non‑negative list.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _softmax(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = logits.max()
    e = np.exp(logits - mx)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components are non‑positive after projection
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector (∑=1) seen and its exact objective value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5          # temperature start
    lam_factor = 1.02  # geometric increase

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # ------- simplex via soft‑max -------
        a = _softmax(logits)

        # ------- exact objective (for monitoring) -------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ------- smooth‑max weights -------
        # w_j ∝ exp(lam * (conv_j - max_c))
        exp_term = np.exp(lam * (conv - max_c))
        w_sum = exp_term.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = exp_term / w_sum

        # ------- gradient wrt a -------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ------- chain rule to logits -------
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------- temperature schedule -------
        lam = min(lam * lam_factor, 1e6)

        # occasional tiny random perturbation to avoid stagnation
        if it % 5000 == 0 and (deadline - time.time()) < 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgrad_proj_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max projected sub‑gradient descent.
    Uses a diminishing step size η_t = η0 / sqrt(t+1) and projects onto the
    probability simplex after each step.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if not active.any():
            # degenerate case – treat whole vector uniformly
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step size
        step = eta0 / math.sqrt(it + 1)

        a = a - step * grad_a
        # project back onto the simplex
        a = _project_simplex(a)

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 40_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap stochastic hill‑climbing (with optional annealing).
    Moves a random fraction of mass from i to j and keeps the change if it
    improves the objective.  Occasionally accepts a worse move with a tiny
    probability to escape shallow traps.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # transfer a random fraction (up to 40 %) of the source mass
        delta = a[i] * (np.random.rand() * 0.4)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        accept = False
        if val2 < best_val:
            accept = True
        else:
            # tiny Metropolis acceptance to occasionally escape local minima
            T = 1e-4  # temperature for annealing
            if np.random.rand() < math.exp(-(val2 - best_val) / T):
                accept = True

        if accept:
            a = a2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
                if it % 3000 == 0:
                    print(f"[Local move {it}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of start vectors (probability distributions)."""
    cand = []

    # warm‑start and a few rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet with various concentration levels
    for alpha in (1.0, 0.3, 0.1):
        cand.append(np.random.dirichlet(np.full(n, alpha)))

    # power‑law edge‑concentrated shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # double‑Gaussian edges
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # spike families (equally spaced non‑zero entries)
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    return cand
