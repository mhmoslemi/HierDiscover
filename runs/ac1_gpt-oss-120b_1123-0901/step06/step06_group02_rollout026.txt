"""
Hybrid optimizer for the step‑function bound (improved version).

The algorithm proceeds in four stages:

1. **Diverse random initialisations** – generates a few thousand candidate
   sequences of lengths between 1500 and 12000 using many parametric families
   (exponential, exponential+offset, power‑law, Gaussian, Hann, Hamming,
   Dirichlet).  The best K=3 candidates (by exact max‑convolution) are kept.

2. **Continuation L‑BFGS on a smooth‑max surrogate** – each kept candidate is
   refined by a sequence of decreasing temperatures τ.  The objective is
   τ·log‑sum‑exp(conv/τ) (a smooth approximation of the true max).  Gradients
   are computed analytically via FFT‑based convolution, and L‑BFGS‑B (from
   SciPy) performs the optimisation on the log‑softmax parametrisation
   (`a = S·softmax(x)`) which automatically respects the simplex constraint
   ( Σa = √(2n) , a ≥ 0 ).  After each τ‑stage the exact objective is
   evaluated and the global best is updated.

3. **Targeted projected sub‑gradient descent** – follows the exact worst
   index of the convolution and takes projected steps on the simplex.

4. **Final simulated‑annealing hill‑climb** – moves tiny amounts of mass
   between random positions, accepting moves with a Metropolis rule.
   This gently polishes the solution.

The routine respects the supplied time budget (default 1000 s) and returns
the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of real vectors x and y."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # Clip tiny negatives caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (s ≥ 0)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]          # sorted descending
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)     # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1        # zero‑based index of last positive entry
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact evaluation (max of the full self‑convolution)
# ----------------------------------------------------------------------
def _exact_max_conv(seq: np.ndarray) -> float:
    """Return max(conv(seq,seq)) after normalising sum to √(2n)."""
    n = seq.shape[0]
    target = math.sqrt(2.0 * n)
    if seq.sum() == 0:
        return np.inf
    a = seq * (target / seq.sum())
    conv = _full_convolution_fft(a, a)
    return float(conv.max())


# ----------------------------------------------------------------------
# Surrogate (smooth‑max) evaluation and its gradient w.r.t. logits x
# ----------------------------------------------------------------------
def _surrogate_and_grad(x: np.ndarray,
                        target_sum: float,
                        tau: float) -> tuple[float, np.ndarray]:
    """
    For a given temperature τ compute
        φτ(x) = τ·log Σ exp(conv(a,a)/τ)    (smooth‑max)
    where a = target_sum·softmax(x).  Return φτ and ∇x φτ.
    """
    # soft‑max mapping
    max_x = np.max(x)
    exp_shift = np.exp(x - max_x)                # stable
    p = exp_shift / np.sum(exp_shift)            # Σp = 1
    a = target_sum * p

    # full convolution of a with itself
    conv = _full_convolution_fft(a, a)
    max_conv = np.max(conv)

    # soft‑max weights over the convolution entries (stable)
    shifted = conv - max_conv                     # ≤ 0
    w_exp = np.exp(shifted / tau)                # exp((conv-max)/τ)
    w_sum = np.sum(w_exp)
    if w_sum == 0.0:                             # should not happen
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w = w_exp / w_sum                         # Σ w = 1

    # gradient of φτ w.r.t. a : 2 * (a *⋆ w)   (cross‑correlation)
    # cross‑correlation a ⋆ w = convolution of a with w reversed
    w_rev = w[::-1]
    conv2 = _full_convolution_fft(a, w_rev)      # length 3n‑2
    n = a.shape[0]
    idx = (2 * n - 2) - np.arange(n)              # slice yielding length‑n vector
    grad_a = 2.0 * conv2[idx]

    # chain rule to get gradient w.r.t. logits x
    dotp = np.dot(p, grad_a)                     # scalar
    grad_x = target_sum * p * (grad_a - dotp)

    # smooth‑max value (identical to max_conv + τ·log Σ exp((conv-max)/τ))
    phi = max_conv + tau * math.log(w_sum)

    return phi, grad_x


# ----------------------------------------------------------------------
# L‑BFGS refinement with a continuation on τ
# ----------------------------------------------------------------------
def _l_bfgs_continuation(seq: np.ndarray,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS on a decreasing schedule of temperatures τ.
    Returns the best sequence found (exact max‑convolution) and its
    maximum convolution value.
    """
    n = seq.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # initialise logits from the current sequence (soft‑max parametrisation)
    x = np.log(seq + 1e-12)

    # temperature schedule – start relatively large and shrink
    cur_conv = _full_convolution_fft(seq, seq).max()
    tau_sched = [max(cur_conv * f, 1e-4) for f in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01)]

    best_a = seq.copy()
    best_val = float(cur_conv)

    for tau in tau_sched:
        if time.time() >= deadline:
            break

        # callback for SciPy: returns (value, grad)
        def fun_and_grad(x_inner):
            return _surrogate_and_grad(x_inner, target_sum, tau)

        # L‑BFGS‑B – we limit the number of iterations for safety
        res = minimize(fun_and_grad,
                       x,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 200, 'gtol': 1e-9, 'disp': False})
        x = res.x

        # evaluate exact max‑convolution of the current point
        max_x = np.max(x)
        p = np.exp(x - max_x)
        p = p / np.sum(p)
        a = target_sum * p
        cur_max = _full_convolution_fft(a, a).max()
        if cur_max < best_val:
            best_val = float(cur_max)
            best_a = a.copy()
            print(f"[{time.time():.2f}] L‑BFGS τ={tau:.5g} improved max={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------
def _targeted_subgradient(a0: np.ndarray,
                          deadline: float) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent that follows the current worst index."""
    a = a0.copy()
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step = 0.2 * target_sum / n          # initial step size

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())            # index of current worst peak
        # gradient of conv_k w.r.t. a
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)

        cand_val = _full_convolution_fft(cand, cand).max()
        if cand_val < best_val - 1e-12:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= 1.02                       # modest increase after success
            if it % 200 == 0:
                print(f"[{time.time():.2f}] subgrad it={it}  val={best_val:.6f}")
        else:
            step *= 0.70                       # reduce step on failure
            if np.random.rand() < 0.07:
                jitter = np.random.rand(n) * 1e-4
                a = _project_onto_simplex(a + jitter, target_sum)

        if best_val < 1.503:                  # early exit if we reached target
            break
    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (simulated annealing) on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray,
               deadline: float) -> tuple[np.ndarray, float]:
    """Move tiny amounts of mass between positions with a Metropolis rule."""
    a = a0.copy()
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)

    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    T0 = max(1e-3, best_val * 0.4)     # initial temperature
    T_end = 1e-5
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue
        delta = a[i] * np.random.rand() * 0.08   # move ≤8 % of mass at i
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()
        accept = False
        if new_val < best_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it}  new best={best_val:.6f}")

        if it % 2500 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break
    return best_a, best_val


# ----------------------------------------------------------------------
# Random parametric initialisations – keep the K best candidates
# ----------------------------------------------------------------------
def _random_initialisations(K: int,
                            deadline: float) -> list[tuple[np.ndarray, float]]:
    """
    Generate random sequences of various lengths and families.
    Return a list of up to K tuples (seq, max_conv) representing the
    best candidates encountered.
    """
    best = []  # list of (seq, max_conv) sorted by max_conv ascending

    while time.time() < deadline:
        # pick a length – favour larger lengths while we still have time
        remaining = deadline - time.time()
        if remaining > 600:
            n = np.random.randint(3000, 10000 + 1)
        elif remaining > 300:
            n = np.random.randint(2500, 8000 + 1)
        else:
            n = np.random.randint(2000, 6000 + 1)

        target_sum = math.sqrt(2.0 * n)

        shape = np.random.choice(
            ["exp", "exp_off", "pow", "gauss", "hann", "hamming", "dirichlet"]
        )
        idx = np.arange(n, dtype=float)

        if shape == "exp":
            tau = np.random.uniform(0.2, 6.0) * n
            a0 = np.exp(-idx / tau)
        elif shape == "exp_off":
            tau = np.random.uniform(0.2, 6.0) * n
            offset = np.random.uniform(0.0, 0.2) * np.exp(-0.0 / tau)
            a0 = offset + np.exp(-idx / tau)
        elif shape == "pow":
            p = np.random.uniform(0.1, 2.0)
            a0 = (idx + 1) ** (-p)
        elif shape == "gauss":
            sigma = np.random.uniform(0.1, 0.5) * n
            a0 = np.exp(-0.5 * (idx / sigma) ** 2)
        elif shape == "hann":
            a0 = 0.5 - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1))
        elif shape == "hamming":
            a0 = 0.54 - 0.46 * np.cos(2.0 * np.pi * idx / (n - 1))
        else:  # dirichlet (uniform on simplex)
            a0 = np.random.dirichlet(np.ones(n))

        # optional small Gaussian jitter
        if np.random.rand() < 0.2:
            noise = np.random.randn(n) * np.max(a0) * 0.02
            a0 = np.maximum(a0 + noise, 0.0)

        # normalise to the required L1 sum
        a0 = a0 * (target_sum / a0.sum())

        cur_max = _full_convolution_fft(a0, a0).max()
        # keep the top‑K
        if len(best) < K:
            best.append((a0, cur_max))
            best.sort(key=lambda x: x[1])
        else:
            if cur_max < best[-1][1]:
                best[-1] = (a0, cur_max)
                best.sort(key=lambda x: x[1])

    return best


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Hybrid search for a non‑negative step‑function with minimal evaluation
    score.  The routine respects the supplied time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.10   # small safety margin

    best_seq = None
    best_score = np.inf

    # convenience updater
    def _record(seq: np.ndarray):
        nonlocal best_seq, best_score
        score = _full_convolution_fft(seq, seq).max()
        if score < best_score:
            best_score = float(score)
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] GLOBAL new best={best_score:.6f}  n={len(seq)}")
        return score

    # ------------------------------------------------------------------
    # Warm‑start from any global construction that may already exist
    # ------------------------------------------------------------------
    try:
        hs = globals().get("height_sequence_1")
        if hs is not None:
            hs_arr = np.asarray(hs, dtype=float)
            n0 = hs_arr.shape[0]
            target0 = math.sqrt(2.0 * n0)
            hs_norm = hs_arr * (target0 / hs_arr.sum())
            _record(hs_norm)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – random diverse initialisations (keep top K)
    # ------------------------------------------------------------------
    K = 3
    init_deadline = deadline - 250.0
    top_candidates = _random_initialisations(K, init_deadline)

    # update global best with the candidates we just found
    for seq, _ in top_candidates:
        _record(seq)

    # ------------------------------------------------------------------
    # Phase 2 – L‑BFGS refinement with temperature continuation
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 150.0:
        # make a copy of the top‑K candidates and refine each one
        for idx, (seq, _) in enumerate(top_candidates):
            if time.time() >= deadline - 150.0:
                break
            print(f"[{time.time():.2f}] L‑BFGS refinement start on candidate {idx+1}")
            seq_refined, val_refined = _l_bfgs_continuation(seq, deadline)
            if val_refined < best_score:
                best_score = val_refined
                best_seq = seq_refined.copy()
                print(f"[{time.time():.2f}] after L‑BFGS candidate {idx+1} best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Targeted projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        n = best_seq.shape[0]
        sub_deadline = deadline - 40.0
        seq_sub, val_sub = _targeted_subgradient(best_seq.copy(),
                                                 sub_deadline)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑gradient best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Final simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        hill_deadline = deadline
        seq_hill, val_hill = _hillclimb(best_seq.copy(),
                                        hill_deadline)
        if val_hill < best_score:
            best_score = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] after hill‑climb best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be used)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 3000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = _full_convolution_fft(best_seq, best_seq).max()
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    return best_seq.tolist()
