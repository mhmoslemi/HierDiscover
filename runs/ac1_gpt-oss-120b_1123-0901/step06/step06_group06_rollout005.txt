    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    best_seq = None
    best_val = float('inf')
    best_n = None

    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline: break
        print(f'\n=== length {n} ===')
        conv_len = 1 << int(np.ceil(np.log2(2*n-1)))
        corr_len = 1 << int(np.ceil(np.log2(2*n-1 + n-1))) # as before

        # Build candidates
        candidates = _build_initial_candidates(n)
        # if we have previous best_seq, upscale and add
        if best_seq is not None:
            up_seq = upscale(best_seq, n)
            candidates.append(up_seq)

        # Deduplicate candidates maybe.

        # For each candidate, run gradient descent
        for cid, cand in enumerate(candidates):
            if time.time() > deadline: break

            # Ensure sum 1 and nonnegative
            a = np.maximum(cand, 0.0)
            s = a.sum()
            if s < 1e-12:
                a = np.random.dirichlet(np.ones(n))
                s = a.sum()
            a = a / s

            max_iters = int(5000 * 8192 / n) + 2000  # more for small n
            eta = 0.5
            # Optionally use schedule decreasing eta: eta_t = eta / sqrt(t)
            for it in range(max_iters):
                if time.time() > deadline: break
                # conv
                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                val = 2.0 * n * max_c
                # best update
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    best_n = n
                    print(f"[n={n} cand={cid} EG iter={it}] new best {val:.6f}")

                # Compute subgradient
                # active indices where conv near max

                mask = conv >= max_c - eps

                if mask.any():
                    w[mask] = 1.0 / mask.sum()
                # gradient = 2 * corr(w, a)
                grad = 2.0 * _corr_fft(w, a, corr_len)

                # multiplicative update
                a = a * np.exp(-eta * grad)
                # renorm
                s = a.sum()
                if s < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()

"""
Propose a non‑negative step‑function (list of heights) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

The search works in several stages:

1. **Multi‑scale progressive refinement** – we start with a short
   sequence (n≈1024) and repeatedly double the length, interpolating the
   best solution found so far.  Larger n gives a lower theoretical bound,
   but the optimisation becomes more expensive, so we allocate time
   adaptively.

2. **Two‑phase local optimisation** for every candidate:
   * *Adam on a smooth‑max surrogate* (log‑sum‑exp with a growing
     temperature).  The parameters are kept on the log‑softmax (logits)
     manifold, guaranteeing non‑negativity and a unit sum.
   * *Exponentiated‑gradient sub‑gradient descent* on the exact
     objective (max‑convolution).  This stage tightens the solution
     after the smoother Adam phase.

3. **Simulated annealing** – a cheap stochastic “shake” that moves a
   tiny amount of mass between two random positions.  It helps to escape
   shallow plateaus that deterministic updates may miss.

All heavy operations (convolution and the needed cross‑correlation)
use real FFTs, i.e. O(n log n) per iteration.  The routine respects the
overall time budget (default 1000 s) and returns the best sequence it
found as a plain Python list.

The implementation is deliberately lightweight, uses only NumPy and the
standard library, and does not rely on any external files or network
access.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based linear convolution a*a and cross‑correlation (w ★ a)
def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

def _build_initial_candidates(n: int) -> list[np.ndarray]:
    """Generate a modest, diverse set of starting points (all sum to 1)."""
    cand = []

    # 1) warm‑start from a previously found solution, if present
    global height_sequence_1
    if "height_sequence_1" in globals():
        h = np.asarray(height_sequence_1, dtype=np.float64)
        if h.shape[0] > n:
            h = h[:n]
        else:
            h = np.pad(h, (0, n - h.shape[0]))
        h = np.maximum(h, 0.0)
        if h.sum() < 1e-8:
            h = np.random.dirichlet(np.ones(n))
        else:
            h = h / h.sum()
        cand.append(h)

    # 2) pure Dirichlet (uniform‑like)
    cand.append(np.random.dirichlet(np.ones(n)))

    # 3) spiky Dirichlet (concentration < 1)
    for conc in (0.2, 0.4, 0.6):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 4) over‑dispersed Dirichlet (conc > 1)
    for conc in (1.5, 2.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 5) simple power‑law edge‑concentrated shape
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    shape = np.power(prod, -0.8, where=prod > 0, out=np.zeros_like(prod))
    if shape.sum() > 0:
        shape = shape / shape.sum()
        cand.append(shape)

    # 6) symmetric exponential decay
    gamma = 0.001
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
    shape = shape / shape.sum()
    cand.append(shape)

    # 7) cosine window (single bump)
    coswin = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    coswin = coswin / coswin.sum()
    cand.append(coswin)

    return cand


def _upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate ``seq`` to length ``new_n`` and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:
        new_seq = np.random.dirichlet(np.ones(new_n))
    else:
        new_seq = new_seq / s
    return new_seq


def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_len: int,
                   corr_len: int,
                   max_iters: int,
                   deadline: float,
                   best_val: float,
                   best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max surrogate.
    Returns the locally best sequence (on the unit‑sum simplex) and its value.
    """
    # initialise logits (soft‑max variables)
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    lr = 0.08 * np.sqrt(2048.0 / n)          # size‑adapted learning rate
    lam = 0.2                                # initial temperature
    lam_factor = 1.018                       # aggressive ramp‑up

    last_improve = time.time()
    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max (logits → probabilities)
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # convolution and smooth‑max weights
        conv = _conv_fft(a, conv_len)
        max_c = conv.max()
        # true objective (sum(a)=1) – used for bookkeeping only
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={t:05d}] new best {val:.6f}")

        # smooth‑max (log‑sum‑exp) weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (linear in w)
        grad_a = 2.0 * _corr_fft(w, a, corr_len)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule (approaches true max)
        lam = min(lam * lam_factor, 1e6)

        # occasional random shake when stuck
        if time.time() - last_improve > 7.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    # return the best sequence seen in this Adam run
    return best_seq, best_val


def _expgrad_optimize(a0: np.ndarray,
                      n: int,
                      conv_len: int,
                      corr_len: int,
                      max_iters: int,
                      deadline: float,
                      best_val: float,
                      best_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient / sub‑gradient descent on the exact max‑convolution.
    """
    a = a0 / a0.sum()
    eta = 0.55                     # base step size
    last_improve = time.time()

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[ExpGrad n={n} iter={it:05d}] new best {val:.6f}")

        # active set (indices attaining the maximum, up to tolerance)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _corr_fft(w, a, corr_len)

        # multiplicative update (protect against overflow)
        grad_clip = np.clip(grad_a, -20.0 / eta, 20.0 / eta)
        a = a * np.exp(-eta * grad_clip)

        # renormalise – also guard against collapse to all zeros
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # simple decay of the step size
        eta *= 0.9999

        # early abort if stagnating
        if time.time() - last_improve > 8.0:
            break

    return best_seq, best_val


def _simulated_anneal(a0: np.ndarray,
                      n: int,
                      conv_len: int,
                      budget_s: float,
                      current_best_val: float) -> tuple[np.ndarray, float]:
    """
    Very lightweight simulated‑annealing on the exact objective.
    Moves a small amount of mass between two random positions.
    """
    best_a = a0.copy()
    best_val = current_best_val
    start = time.time()
    # temperature linearly decays from T0 to Tf
    T0, Tf = 1e-2, 1e-5

    while time.time() - start < budget_s:
        elapsed = time.time() - start
        T = T0 * (1.0 - elapsed / budget_s) + Tf
        # pick a source index with non‑zero mass
        i = np.random.randint(n)
        if best_a[i] < 1e-12:
            continue
        # pick a distinct destination index
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        # amount to transfer (up to a few percent of the source mass)
        frac = np.random.rand() * 0.05
        delta = best_a[i] * frac
        new_a = best_a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        # evaluate
        conv = _conv_fft(new_a, conv_len)
        val = 2.0 * n * conv.max()

        # accept if better or by Metropolis criterion
        if val < best_val or np.random.rand() < np.exp((best_val - val) / T):
            best_a = new_a
            if val < best_val:
                best_val = val

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the supplied time budget (seconds) and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")
    best_n = None

    # Length schedule – we start small and double up to the largest
    # size that still leaves us enough time for the final fine‑tune.
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        conv_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_len = 1 << int(np.ceil(np.log2(2 * n - 1 + n - 1)))

        # ----- build a tiny but diverse set of start points -------------
        candidates = []

        # a) upscale the current global best (if any)
        if best_seq is not None:
            up = _upscale(best_seq, n)
            candidates.append(up)

        # b) a fresh Dirichlet draw (uniform‑like)
        candidates.append(np.random.dirichlet(np.ones(n)))

        # c) a power‑law edge‑concentrated shape (often a good starter)
        idx = np.arange(1, n + 1, dtype=np.float64)
        prod = idx * (n - idx + 1.0)
        shape = np.power(prod, -0.8, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

        # d) symmetric exponential decay (another useful prototype)
        gamma = 0.001
        i = np.arange(n, dtype=np.float64)
        exp_shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        exp_shape /= exp_shape.sum()
        candidates.append(exp_shape)

        # Keep only a few distinct candidates
        seen = set()
        uniq_cands = []
        for c in candidates:
            key = tuple(np.round(c, 6))  # crude deduplication
            if key not in seen:
                seen.add(key)
                uniq_cands.append(c)
        candidates = uniq_cands

        # ----- optimise each candidate ----------------------------------
        for cid, cand in enumerate(candidates):
            if time.time() > deadline:
                break

            # Normalise to a probability vector
            a0 = np.maximum(cand, 0.0)
            if a0.sum() < 1e-12:
                a0 = np.random.dirichlet(np.ones(n))
            a0 = a0 / a0.sum()

            # Heuristic iteration budgets (more for smaller n)
            adam_iters = int(12000 * 2048.0 / n) + 2000
            eg_iters   = int(6000  * 2048.0 / n) + 2000

            # 1) Adam on smooth‑max surrogate
            best_seq, best_val = _adam_optimize(
                a0, n, conv_len, corr_len, adam_iters,
                deadline, best_val, best_seq
            )

            # 2) Exponentiated‑gradient (exact max) refinement
            best_seq, best_val = _expgrad_optimize(
                best_seq, n, conv_len, corr_len, eg_iters,
                deadline, best_val, best_seq
            )

            # 3) Occasional random shake if we still have time
            # (a very light version of simulated annealing)
            if time.time() < deadline and np.random.rand() < 0.15:
                shake_iters = 1500
                best_seq, best_val = _expgrad_optimize(
                    best_seq, n, conv_len, corr_len, shake_iters,
                    deadline, best_val, best_seq
                )

        # ----- post‑optimisation simulated annealing on the current best
        if best_seq is not None and best_seq.shape[0] == n:
            remaining = deadline - time.time()
            if remaining > 5.0:
                sa_budget = min(remaining * 0.5, 30.0)
                print(f"[n={n}] simulated annealing for {sa_budget:.1f}s")
                best_seq, best_val = _simulated_anneal(
                    best_seq, n, conv_len, sa_budget, best_val
                )
                print(f"[n={n}] post‑SA best {best_val:.6f}")

        best_n = n   # remember the length that produced the current optimum

    # ------------------------------------------------------------------
    # Final fine‑tune on the longest length we managed to obtain.
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        conv_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_len = 1 << int(np.ceil(np.log2(2 * n - 1 + n - 1)))
        remaining = deadline - time.time()
        # run a long Adam run (no strict iteration cap – stop by deadline)
        max_iters = int(200000 * 2048.0 / n)   # generous upper bound
        print(f"[final] fine‑tuning n={n} for up to {remaining:.1f}s")
        best_seq, best_val = _adam_optimize(
            best_seq, n, conv_len, corr_len, max_iters,
            deadline, best_val, best_seq
        )
        # one more short EG pass
        best_seq, best_val = _expgrad_optimize(
            best_seq, n, conv_len, corr_len, 3000,
            deadline, best_val, best_seq
        )
        print(f"[final] best value {best_val:.6f}")

    # Fallback in the unlikely event that nothing was found
    if best_seq is None:
        best_seq = np.full(1024, 1.0 / 1024, dtype=np.float64)

    # Convert to plain Python list as required by the outer evaluator
    return best_seq.tolist()
