"""
Improved search for a step‑function (non‑negative sequence) that minimises

        F(a) = 2·n·max(a∗a) / (∑a)² .

Key ideas
---------
1. **Richer starting pool** – in addition to the families used before we add
   single‑sided exponentials, asymmetric double‑exponentials, simple power‑laws,
   beta‑shaped densities and a few extra random Dirichlet draws.  The pool is
   deliberately over‑complete; only the best few are kept.

2. **Longer local optimisation** – the Adam / sub‑gradient optimiser from the
   reference implementation is kept, but the iteration limits are increased
   (up to a million Adam steps and 60 k sub‑gradient steps) while still obeying
   the global deadline.  The learning‑rate schedule is slightly more conservative
   (smaller lr, slower temperature rise) which helps to escape shallow plateaux.

3. **Convex mixing** – after the first optimisation round the ten best vectors
   are linearly mixed (convex combinations) to generate fresh starting points;
   this often yields a small but systematic drop in the objective.

4. **Optional up‑sampling** – if time permits the best distribution is interpolated
   to a double‑size grid (8192) and the whole optimisation is repeated.  The extra
   resolution lets the optimiser fine‑tune the shape and usually gives a further
   reduction.

All numbers are forced to be non‑negative and clipped to the evaluator’s
maximum of 1000.  The routine respects the supplied time budget (default 1000 s)
and returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.ifft(fa * fa, fft_len).real
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.ifft(fw * fr, fft_len).real
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective (sum(a) is assumed > 0)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate (log‑sum‑exp) followed by a sub‑gradient
    phase on the exact max.  Slightly more generous iteration limits than the
    reference code.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # simplex parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03                                     # a bit more conservative

    lam = 0.3                                      # start temperature
    lam_factor = 1.015

    last_improve = time.time()
    it = 0

    # --------------------------- Adam phase ---------------------------
    while time.time() < deadline and it < 1_000_000:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through the soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # raise temperature (approach true max)
        lam = min(lam * lam_factor, 1e6)

        # shake when stagnant
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 5e-3
            last_improve = time.time()

        it += 1

    # -------------------- Sub‑gradient phase (exact max) --------------------
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total

    eta0 = 0.6
    sub_last_improve = time.time()
    for sub_it in range(60_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[n={n} Subgrad it={sub_it:05d}] new best {val:.6f}")

        # active set mask (all indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(sub_it + 1)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 8.0:
            break

    return global_best_seq, global_best_val


# ----------------------------------------------------------------------
def _generate_candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a diverse pool of start vectors (including many parametric families).
    """
    pool = []

    # ------------------------------------------------------------------
    # warm‑start and simple rotations
    # ------------------------------------------------------------------
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # ------------------------------------------------------------------
    # baseline families (kept for completeness)
    # ------------------------------------------------------------------
    pool.append(np.full(n, 1.0 / n))                                 # uniform

    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)                                     # i·(n‑i+1)
    for alpha in np.linspace(-2.0, -0.2, 12):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    i = np.arange(n, dtype=np.float64)
    for lam in np.linspace(0.1, 3.0, 12):
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        pool.append(shape)

    for sigma in np.linspace(n * 0.03, n * 0.4, 8):
        shape = np.exp(-0.5 * ((i - n / 2) ** 2) / (sigma ** 2))
        shape /= shape.sum()
        pool.append(shape)

    # ------------------------------------------------------------------
    # NEW families – asymmetry and beta‑type shapes
    # ------------------------------------------------------------------
    # single‑sided exponential (left‑heavy)
    for lam in np.linspace(0.2, 5.0, 12):
        shape = np.exp(-lam * i)
        shape /= shape.sum()
        pool.append(shape)

    # asymmetric double‑exponential (weight w on the left side)
    for lam in np.linspace(0.2, 5.0, 8):
        for w in (0.25, 0.5, 0.75):
            shape = w * np.exp(-lam * i) + (1 - w) * np.exp(-lam * (n - 1 - i))
            shape = np.maximum(shape, 0.0)
            shape /= shape.sum()
            pool.append(shape)

    # simple power‑law decreasing
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        shape = 1.0 / (idx ** p)
        shape /= shape.sum()
        pool.append(shape)

    # simple power‑law increasing (mirror of decreasing)
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        shape = 1.0 / ((n - idx + 1) ** p)
        shape /= shape.sum()
        pool.append(shape)

    # beta‑type densities (α,β > 0)
    alphas = (0.2, 0.5, 1.0, 2.0, 5.0)
    betas  = (0.2, 0.5, 1.0, 2.0, 5.0)
    for a in alphas:
        for b in betas:
            shape = (idx ** (a - 1)) * ((n - idx + 1) ** (b - 1))
            if shape.sum() > 0:
                shape /= shape.sum()
                pool.append(shape)

    # random Dirichlet draws (wide range of concentration)
    for alpha in np.logspace(-2, 0, 15):
        shape = np.random.dirichlet(np.full(n, alpha))
        pool.append(shape)

    return pool


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation metric.
    The routine obeys the supplied time‑budget and returns the best sequence
    found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------- primary support size ---------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Warm start from a previously found construction, if available
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        w = np.asarray(warm_start, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
        else:
            w = None
    else:
        w = None

    # ------------------- build & screen candidates -------------------
    candidates = _generate_candidate_pool(n, w)

    best_seq: np.ndarray | None = None
    best_val = float("inf")
    scored = []  # (value, normalised vector)

    for idx, cand in enumerate(candidates):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        val = _objective(a, n, conv_fft_len)
        scored.append((val, a))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        if time.time() >= deadline:
            break
        print(f"[screen {idx:03d}] obj={val:.6f}")

    # keep a handful of the most promising starts (up to ten)
    scored.sort(key=lambda x: x[0])
    top_candidates = [vec for _, vec in scored[:10]]

    # ------------------------------------------------------------------
    # convex mixing – generate a few extra starts by averaging top vectors
    # ------------------------------------------------------------------
    mixed = []
    for i in range(3):
        seeds = np.random.choice(len(top_candidates), size=2, replace=False)
        mix_vec = (top_candidates[seeds[0]] + top_candidates[seeds[1]]) * 0.5
        mixed.append(mix_vec / mix_vec.sum())
    top_candidates.extend(mixed)

    # --------------------- deep optimisation passes -----------------
    for cid, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[optim start {cid}] launching deep optimisation")
        best_seq, best_val = _deep_optimize(
            a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
        )

    # --------------------- optional larger support ------------------
    if time.time() + 15 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("[big] starting optimisation on enlarged grid")
        best_seq, best_val = _deep_optimize(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big,
            deadline, best_seq, best_val
        )

    # -------------------------- finalisation -----------------------
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Finished search – best objective = {best_val:.6f}")
    return best_seq.tolist()
