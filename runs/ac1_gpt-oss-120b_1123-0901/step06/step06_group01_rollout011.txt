"""
Propose a non‑negative step function that (hopefully) beats the previous
best bound (≈ 1.50833).  

The search consists of three stages:

1. **Random‑shape exploration** – a cheap, highly diverse stochastic
   generator produces vectors of many different analytic forms
   (exponential, power‑law, Gaussian, Beta, random spikes, etc.).
   Each candidate is normalised to Σa = √(2 n) and evaluated with
   ``evaluate_sequence``.  The best vector seen so far is retained.

2. **Projected Gradient Descent (PGD)** – the convex objective
   `max(a∗a)` is minimised directly.  The sub‑gradient is obtained from the
   active (max‑attaining) convolution indices via a single FFT per
   iteration.  After each step the vector is projected onto the simplex
   `{a_i ≥ 0, Σa_i = √(2 n)}`.  A generous step‑size schedule and a
   deadline‑driven termination guarantee that the optimiser runs as long
   as time permits.

3. **Stochastic local refinement** – a simple simulated‑annealing hill‑climber
   moves a small fraction of mass between random indices, always accepting
   improvements and occasionally accepting worsening moves.  This cleans up
   any remaining irregularities.

All stages respect a global time budget (default 1000 s).  The routine
returns the best candidate found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities ------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x_i ≥ 0, Σx_i = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Exact maximum of the linear convolution a * a (using one FFT)."""
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected Gradient Descent on the exact max‑convolution.
    Returns the best vector observed (as a NumPy array) and its objective value.
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise on the simplex
    a = a0.copy()
    a = a * target_sum / a.sum()

    # FFT length sufficient for the linear convolution of length 3 n‑2
    Nfft = 1 << ((3 * n - 2).bit_length())

    # base step (≈ 1 % of total mass) – a relatively aggressive start
    step0 = 0.01 * target_sum
    eps_active = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ------- current convolution & its maximum -------
        A = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # keep global best
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD n={n:5d} it={it:8d}] new best = {best_score:.6f}")

        # ------- active set (hard max) -------
        mask = conv >= M - eps_active
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = mask.astype(float) / float(cnt)

        # ------- weighted reverse convolution (sub‑gradient) -------
        W_rev = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev * A, n=Nfft)
        conv_rev = conv_rev[: len(w) + n - 1]                 # linear part
        idx_start = 2 * n - 2
        rev = conv_rev[idx_start - np.arange(n)]

        g = 2.0 * rev                                      # ∇_a max(a∗a)

        # ------- gradient step + projection -------
        lr = step0 / math.sqrt(it)                         # annealed step
        a_tmp = a - lr * g
        a = _project_onto_simplex(a_tmp, target_sum)

        # occasional jitter to escape plateaux
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)

    return best_seq, best_score


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing on the simplex.
    Moves a small fraction of mass from a random source index to a random
    destination index.  Always accepts improvements; otherwise accepts with
    a temperature‑controlled probability.
    """
    target_sum = a.sum()
    cur_seq = a.copy()
    cur_score = best_score

    # FFT size for fast max‑convolution (exact, not padded)
    Nfft = 1 << ((2 * n - 1).bit_length())

    max_steps = 250_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-14:
            step += 1
            continue

        # transfer a random fraction (1 %–5 %) of the source mass
        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # compute new max‑convolution
        A = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score


def _random_shape_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random non‑negative vector of length n using a diverse set of
    analytic shapes.  The vector is NOT normalised.
    """
    shape_type = rng.integers(0, 6)
    i = np.arange(n, dtype=float)

    if shape_type == 0:  # exponential decay (left‑to‑right)
        rate = rng.uniform(0.0002, 0.02)
        v = np.exp(-rate * i)
    elif shape_type == 1:  # exponential increase (right‑to‑left)
        rate = rng.uniform(0.0002, 0.02)
        v = np.exp(-rate * (n - 1 - i))
    elif shape_type == 2:  # power‑law (i+1)^{-p}
        p = rng.uniform(0.0, 2.5)
        v = (i + 1) ** (-p)
    elif shape_type == 3:  # Gaussian bump (maybe shifted)
        mu = rng.uniform(0, n - 1)
        sigma = rng.uniform(n / 20, n / 2)
        v = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    elif shape_type == 4:  # Beta‑like shape on (0,1)
        alpha = rng.uniform(0.2, 5.0)
        beta = rng.uniform(0.2, 5.0)
        x = (i + 0.5) / n
        v = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    else:  # sparse random spikes
        k = max(2, min(15, n // 200))
        idx = rng.choice(n, size=k, replace=False)
        v = np.zeros(n)
        v[idx] = rng.random(k)

    # ensure non‑negativity and return
    v = np.maximum(v, 0.0)
    return v


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (as a plain Python list) found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a few seconds as safety margin
    deadline = start_time + budget_s - 5.0

    best_seq = None          # NumPy array
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – random analytic shapes
    # ------------------------------------------------------------------
    # lengths to explore – a mixture of modest and fairly large sizes
    lengths = [500, 800, 1200, 1600, 1900, 2100, 2400, 2800, 3200, 3800, 4600, 5200]
    # ensure we stay within reasonable memory (≤ 8000)
    lengths = [n for n in lengths if n <= 8000]

    # allocate roughly half the time to this stage
    search_deadline = time.time() + (deadline - time.time()) * 0.5

    it = 0
    while time.time() < search_deadline:
        it += 1
        n = int(rng.choice(lengths))
        target = math.sqrt(2.0 * n)

        # generate a random shape and normalise it to the required sum
        a_raw = _random_shape_vector(n, rng)
        s = a_raw.sum()
        if s == 0.0:
            continue
        a = a_raw * (target / s)

        # evaluate (evaluate_sequence expects a plain list)
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[rand  iter {it:6d}] new best = {best_score:.6f} (len={n})")

    # ------------------------------------------------------------------
    # Stage 2 – deterministic projected‑gradient descent
    # ------------------------------------------------------------------
    if best_seq is not None:
        n = best_seq.shape[0]
        # give the optimiser the rest of the budget (deadline already set)
        # a very large iteration count – the loop will stop when the deadline hits
        max_iters = 10_000_000
        best_seq, best_score = _pgd_optimize(
            a0=best_seq.copy(),
            n=n,
            deadline=deadline,
            max_iters=max_iters,
            best_seq=best_seq,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Stage 3 – stochastic local refinement (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # allocate up to 30 s or half of the remaining time
        refine_deadline = time.time() + min(30.0, remaining * 0.5)
        best_seq, best_score = _local_refine(
            a=best_seq,
            n=best_seq.shape[0],
            deadline=refine_deadline,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Fallback handling (should never trigger)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1900
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
