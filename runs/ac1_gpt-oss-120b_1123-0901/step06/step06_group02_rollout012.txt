"""
Hybrid optimizer for the step‑function bound (improved).

Key ideas
---------
* Enforce symmetry a[i] = a[n‑1‑i] after each update – the optimal
  sequence is believed to be symmetric, this reduces the search space.
* Occasionally project onto a monotone shape (non‑increasing away from the
  centre) – a simple heuristic that frequently lowers the convolution peak.
* Search a wide range of lengths (up to 25 000) and progressively
  upscale the current best candidate by linear interpolation.
* Four optimisation phases:
  1. Diverse random initialisations (exponential, power‑law, windows,
     bimodal, Dirichlet) with symmetry/monotonicity.
  2. Adam optimiser on a smoothed max‑convolution (log‑sum‑exp surrogate)
     with the symmetry projection.
  3. Targeted projected sub‑gradient descent on the exact max‑convolution.
  4. Simulated‑annealing hill‑climb (Metropolis) respecting symmetry.
* All phases respect a global time budget (default 1000 s) and keep a
  global best solution.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of real vectors x and y."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)            # eliminate tiny negatives
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple symmetry & monotonicity utilities
# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Replace v by (v + v[::-1]) / 2."""
    return (v + v[::-1]) * 0.5


def _enforce_monotone_sym(v: np.ndarray) -> np.ndarray:
    """
    Force a symmetric, non‑increasing shape away from the centre.
    The operation is cheap and keeps total mass unchanged (renormalisation
    is performed later by the caller).
    """
    a = v.copy()
    n = a.shape[0]
    centre = (n - 1) // 2
    # ensure a[i] ≤ a[i+1] for i < centre (values increase towards the centre)
    for i in range(centre - 1, -1, -1):
        if a[i] > a[i + 1]:
            a[i] = a[i + 1]
    # enforce symmetry
    a = _symmetrize(a)
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed‑max surrogate (with symmetry)
# ----------------------------------------------------------------------
def _optimise_adam_sym(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 5e-5,
    alpha0: float = 0.025,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the log‑sum‑exp surrogate of max(conv(a,a)).
    The iterate a is always symmetrised and normalised to the required
    L¹‑sum.
    Returns the best coefficient vector (on the simplex) and its exact
    max‑convolution value.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # logits → softmax → a
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)           # stable soft‑max
        p = exp_shift / np.sum(exp_shift)       # Σ p = 1
        a = target_sum * p

        # enforce symmetry and renormalise
        a = _symmetrize(a)
        a *= target_sum / a.sum()

        # exact max of the convolution (used for logging & weighting)
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam‑sym n={n}  new best max={cur_max:.6f}")

        # smoothed max (log‑sum‑exp) weights
        tau = max(init_tau_factor * cur_max * (0.99 ** it),
                  min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a: 2·a ⊛ w_rev
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # back‑propagation to logits (chain rule of soft‑max)
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slow decay of learning rate
        alpha *= 0.9995

    final_val = _full_convolution_fft(best_a, best_a).max()
    return best_a, float(final_val)

# Targeted projected sub‑gradient descent on the exact max‑convolution
# (with symmetry & optional monotonicity)
# ----------------------------------------------------------------------
def _targeted_subgradient_sym(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent that follows the current worst convolution index."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step = 0.2 * target_sum / n          # initial step size

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())            # index of current worst peak

        # gradient of conv_k w.r.t. a
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # projected step
        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)

        # enforce symmetry & (occasionally) monotonicity
        cand = _symmetrize(cand)
        cand *= target_sum / cand.sum()
        if np.random.rand() < 0.15:
            cand = _enforce_monotone_sym(cand)
            cand *= target_sum / cand.sum()

        cand_val = _full_convolution_fft(cand, cand).max()
        if cand_val < best_val - 1e-12:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= 1.02                       # modestly increase step after success

        else:

                jitter = np.random.rand(n) * 1e-4
                a = _project_onto_simplex(a + jitter, target_sum)
                a = _symmetrize(a)
                a *= target_sum / a.sum()

        if best_val < 1.503:

    return best_a, best_val

def _hillclimb_sym(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Move tiny amounts of mass between positions with a Metropolis rule."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # initial temperature proportional to current peak
    T0 = max(1e-3, best_val * 0.4)
    T_end = 1e-5
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue
        # transfer at most 8 % of mass at i
        delta = a[i] * np.random.rand() * 0.08
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # enforce symmetry and normalise
        a_new = _symmetrize(a_new)
        a_new *= target_sum / a_new.sum()

        new_val = _full_convolution_fft(a_new, a_new).max()
        accept = False
        if new_val < best_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb‑sym it={it}  new best={best_val:.6f}")

        # occasional re‑normalisation / symmetry enforcement
        if it % 2500 == 0:
            a = _project_onto_simplex(a, target_sum)
            a = _symmetrize(a)
            a *= target_sum / a.sum()

        if best_val < 1.503:
            break

    return best_a, best_val

            a0 *= target_sum / a0.sum()

        # evaluate and possibly record as global best
        val = _full_convolution_fft(a0, a0).max()
        if val < best_score:
            best_score = float(val)
            best_seq = a0.copy()
            print(f"[{time.time():.2f}] GLOBAL new best={best_score:.6f}  n={n}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on a smoothed max‑convolution (symmetry)
    # ------------------------------------------------------------------
    while best_seq is not None and time.time() < deadline - 200.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        x0 = np.log(best_seq + 1e-12)               # initialise logits
        # allocate a modest slice of the remaining budget for this run
        run_deadline = time.time() + min(
            25.0, deadline - time.time() - 30.0
        )
        seq_opt, val_opt = _optimise_adam_sym(
            n,
            x0,
            target_sum,
            run_deadline,
            init_tau_factor=0.15,
            min_tau_factor=1e-5,
            alpha0=0.03,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = seq_opt.copy()
            print(f"[{time.time():.2f}] after Adam‑sym best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Progressive up‑scaling to longer lengths
    # ------------------------------------------------------------------
    while best_seq is not None and time.time() < deadline - 120.0:
        n = best_seq.shape[0]
        if n >= 25000:
            break
        # choose a random upscale factor
        factor = np.random.uniform(1.3, 1.7)
        N = int(min(round(n * factor), 25000))
        # linear interpolation to the larger grid
        xs = np.linspace(0, n - 1, N)
        a_up = np.interp(xs, np.arange(n), best_seq)
        target_sum_N = np.sqrt(2.0 * N)
        a_up = a_up * (target_sum_N / a_up.sum())
        a_up = _symmetrize(a_up)
        if np.random.rand() < 0.25:
            a_up = _enforce_monotone_sym(a_up)
            a_up *= target_sum_N / a_up.sum()

        # record the up‑scaled candidate
        val = _full_convolution_fft(a_up, a_up).max()
        if val < best_score:
            best_score = float(val)
            best_seq = a_up.copy()
            print(f"[{time.time():.2f}] GLOBAL up‑scale new best={best_score:.6f}  n={N}")

        # short Adam refinement on the larger problem
        x0 = np.log(a_up + 1e-12)
        run_deadline = time.time() + min(
            18.0, deadline - time.time() - 30.0
        )
        seq_opt, val_opt = _optimise_adam_sym(
            N,
            x0,
            target_sum_N,
            run_deadline,
            init_tau_factor=0.12,
            min_tau_factor=1e-5,
            alpha0=0.025,
        )
        if val_opt < best_score:
            best_score = val_opt
            best_seq = seq_opt.copy()
            print(f"[{time.time():.2f}] after up‑scale+Adam best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Targeted projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_deadline = deadline - 30.0
        seq_sub, val_sub = _targeted_subgradient_sym(
            best_seq.copy(), target_sum, sub_deadline
        )
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑grad best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Final simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        hill_deadline = deadline
        seq_hill, val_hill = _hillclimb_sym(
            best_seq.copy(), target_sum, hill_deadline
        )
        if val_hill < best_score:
            best_score = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] final hill‑climb best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be used)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 3000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = _full_convolution_fft(best_seq, best_seq).max()
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # Return a plain Python list as required by the evaluator
    return best_seq.tolist()
