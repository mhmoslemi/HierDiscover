Implement _compute_objective(seq): uses next_pow2 for convolution length and uses np.fft.rfft.

```
def _next_pow2(x):
    return 1 << (x-1).bit_length()
def _full_convolution(a):
    n=len(a)
    fft_len=_next_pow2(2*n-1)
    fa=np.fft.rfft(a,fft_len)
    conv=np.fft.irfft(fa*fa,fft_len)[:2*n-1]
    return conv
def objective(seq):
    conv=_full_convolution(seq)
    max_c=conv.max()
    n=len(seq)
    sum_a=seq.sum()
    return 2.0*n*max_c/(sum_a**2)

"""
Propose a candidate sequence of non‑negative heights that (approximately) minimizes

        F(a) = 2·n·max(a*a) / (sum(a)²)

where “*” is the linear convolution.  The method combines:

* a fast exact evaluation based on FFTs,
* a large random search over a few low‑dimensional parametric families
  (beta‑shapes, exponentials, mixtures of Gaussians, Dirichlet draws),
* an occasional up‑sampling of the best solution found for a smaller length,
* a short Adam‑style refinement on the smooth‑max surrogate,
* a cheap stochastic hill‑climb that moves a small amount of mass between
  two random positions while keeping the total mass constant.

The whole search is bounded by the supplied time budget (≈1000 s).  The
returned value is a plain Python list of floats, ready to be fed to the
external ``evaluate_sequence`` routine.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers (no closures – top‑level functions only)
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective (fast, no Python loops)
def _objective(seq: np.ndarray) -> float:
    """Exact evaluation of 2·n·max(conv) / (sum(seq)²).  Assumes seq ≥ 0."""
    n = seq.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(seq, fft_len)
    max_c = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) – used for warm‑starts
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative vector to ``new_n`` points by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:                     # protect against a zero vector
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s


# ----------------------------------------------------------------------
# Simple Adam optimiser on the smooth‑max surrogate (few hundred steps)
def _run_adam_simple(a0: np.ndarray, n: int, deadline: float, max_iters: int = 5000):
    """Adam optimisation on a soft‑max surrogate of the max‑convolution."""
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # store logits → simplex via softmax
    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.04
    lam = 0.3                # temperature
    lam_factor = 1.01

    best_a = a0.copy()
    best_val = _objective(best_a)

    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max → probability vector on the simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # exact objective (for bookkeeping)
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            # small progress report
            print(f"[Adam n={n} iter={t}] new best {val:.6f}")

        # smooth‑max (log‑sum‑exp) weights on the convolution vector
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (cross‑correlation)
        # ------------------------------------------------
        # cross‑correlation w ★ a
        rev_a = a[::-1]
        fw = np.fft.rfft(w, corr_fft_len)
        fr = np.fft.rfft(rev_a, corr_fft_len)
        corr = np.fft.irfft(fw * fr, corr_fft_len)
        start = n - 1
        grad_a = 2.0 * corr[start:start + n]

        # back‑prop through the soft‑max (logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule – slowly tighten the surrogate
        lam = min(lam * lam_factor, 1e4)

        # occasional tiny random kick if we are stuck
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val

# Very cheap stochastic hill‑climb
def _local_hill_climb(seq0: np.ndarray, n: int,
                      deadline: float, max_trials: int = 100_000):
    """
    Repeatedly move a small mass ε from one index to another,
    keeping the sum constant. Accept the move iff it reduces the exact objective.
    """
    conv_fft_len = _next_pow2(2 * n - 1)

    best_a = seq0.copy()
    best_val = _objective(best_a)

    no_improve = 0
    trials = 0

    while time.time() < deadline and trials < max_trials and no_improve < 5_000:
        i, j = np.random.choice(n, 2, replace=False)
        # move at most 5 % of the source mass (or the whole mass if tiny)
        eps = np.random.uniform(0.0, min(0.05, best_a[i]))
        if eps <= 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps

        # maintain feasibility (non‑negative)
        if cand[i] < 0.0:
            trials += 1
            continue

        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_a = cand
            no_improve = 0
            print(f"[HillClimb n={n}] improved to {val:.6f}")
        else:
            no_improve += 1
        trials += 1

    return best_a, best_val

# Families of parametric sequences (all return a normalised ndarray)
def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like density on a discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # enforce safe exponents – avoid infinities at the boundaries
    a = np.power(x, max(alpha - 1.0, -0.999))
    b = np.power(1.0 - x, max(beta - 1.0, -0.999))
    seq = a * b
    seq = np.maximum(seq, 0.0)
    s = seq.sum()
    if s < 1e-12:
        return None
    return seq / s


def _exponential_shape(n: int, gamma: float) -> np.ndarray:
    """One–sided exponential decay."""
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-gamma * i)
    seq = np.maximum(seq, 0.0)
    s = seq.sum()
    if s < 1e-12:
        return None
    return seq / s


def _two_gaussian_shape(n: int, mu1: float, mu2: float, sigma: float) -> np.ndarray:
    """Mixture of two Gaussian bumps."""
    i = np.arange(n, dtype=np.float64)
    g1 = np.exp(-0.5 * ((i - mu1) / sigma) ** 2)
    g2 = np.exp(-0.5 * ((i - mu2) / sigma) ** 2)
    seq = g1 + g2
    seq = np.maximum(seq, 0.0)
    s = seq.sum()
    if s < 1e-12:
        return None
    return seq / s


def _powerlaw_shape(n: int, p: float) -> np.ndarray:
    """Monotone power‑law tail a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = np.power(i, -p)
    seq = np.maximum(seq, 0.0)
    s = seq.sum()
    if s < 1e-12:
        return None
    return seq / s

                break
            p = np.random.uniform(0.3, 3.0)
            seq = _powerlaw_shape(n, p)
            if seq is None:
                continue
            val = _objective(seq)
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print(f"[pow] new best {val:.6f}")

        # --------------------------------------------------------------
        # 3.  Short Adam refinement on the current global best (if any)
        if best_seq is not None and best_seq.shape[0] == n:
            # 5000 iterations is usually enough for a noticeable improvement
            a_ref, v_ref = _run_adam_simple(best_seq, n, deadline, max_iters=5000)
            if v_ref < best_val:
                best_val, best_seq = v_ref, a_ref.copy()
                print(f"[Adam fine] improved global best to {v_ref:.6f}")

        # --------------------------------------------------------------
        # 4.  Local cheap hill‑climb (tuned for the current length)
        if best_seq is not None and best_seq.shape[0] == n:
            a_hc, v_hc = _local_hill_climb(best_seq, n, deadline,
                                           max_trials=30_000)
            if v_hc < best_val:
                best_val, best_seq = v_hc, a_hc.copy()
                print(f"[HillClimb] improved global best to {v_hc:.6f}")

        # preserve this length's winner for up‑sampling to the next size
        prev_best = best_seq.copy() if best_seq is not None else None

    # ------------------------------------------------------------------
    # Final global hill‑climb on whatever the best sequence is (any length)
    if best_seq is not None and time.time() < deadline:
        print(f"\n>>> Final global hill‑climb (n={len(best_seq)})")
        a_final, v_final = _local_hill_climb(best_seq, len(best_seq),
                                             deadline, max_trials=50_000)
        if v_final < best_val:
            best_val, best_seq = v_final, a_final.copy()
            print(f"[final hill] improved to {v_final:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform when nothing was produced (should never happen)
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Return plain Python list as required by the external evaluator
    return best_seq.tolist()
