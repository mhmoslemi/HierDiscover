def _project_simplex(v):
    # From https://arxiv.org/pdf/1101.6081.pdf
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _run_pgd(a0, n, deadline, max_iters):
    conv_fft_len = _next_pow2(2*n - 1)
    corr_fft_len = _next_pow2(3*n - 2)
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    last_improve = time.time()
    # Step size schedule
    step0 = 0.5
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # compute conv
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0*n*max_c
        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} PGD iter={it}] new best {val:.6f}")
        # gradient
        active = np.isclose(conv, max_c, atol=1e-12)
        # If many active indices, weight equally
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # step size
        step = step0 / np.sqrt(it)
        # gradient descent step with projection onto simplex
        a_new = a - step * grad
        a_new = _project_simplex(a_new)
        a = a_new
        # early break if no improvement for some time
        if time.time() - last_improve > 8.0:
            break
    return best_a, best_val
```

def propose_candidate(seed: int = 0, budget_s: float = 1000):
    """Search for a step function sequence minimizing the objective.
    Uses a combination of simplex‑projected gradient descent, random restarts,
    and a final hill‑climb.  The algorithm respects the given time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_val = float('inf')
    # list of lengths to try
    lengths = [2048, 4096, 8192]  # can add 16384 if time permits
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Length {n} ===")
        # Build candidate pool
        candidates = _generate_initial_candidates(n, best_seq)
        # number of iterations for PGD
        max_iters = int(25000 * 2048 / n) + 3000
        for cid, init in enumerate(candidates):
            if time.time() > deadline:
                break
            a_opt, val = _run_pgd(init, n, deadline, max_iters)
            if val < best_val:
                best_val = val
                best_seq = a_opt.copy()
                print(f"[n={n} cand={cid}] PGD new global best {val:.6f}")

            # Also try a cheap hill climb on this candidate directly
            a_hc, val_hc = _local_hill_climb(a_opt, n, deadline, max_trials=50000)
            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.copy()
                print(f"[n={n} cand={cid}] hill‑climb improved global best {val_hc:.6f}")

    # final global hill‑climb
    if best_seq is not None and time.time() < deadline:
        print(f"\n=== Final global hill‑climb (n={len(best_seq)}) ===")
        a_final, val_final = _local_hill_climb(best_seq, len(best_seq), deadline, max_trials=200000)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final
            print(f"[final] global hill‑climb improved best to {val_final:.6f}")

    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback)

    return best_seq.tolist()
```

"""
Propose a step‑function (non‑negative heights) that minimises

    F(a) = 2·n·max(conv(a,a)) / (sum(a)²),

where conv(a,a) is the linear convolution of a with itself.
The algorithm works as follows:

1.  Hierarchical refinement – start with a small size (256) and
    progressively double the length up to 16384, each time up‑sampling the
    best sequence found so far.
2.  For every length generate a diverse pool of initial vectors
    (warm‑start, random Dirichlet, monotone, exponential, power‑law,
    Gaussian bumps, sparse vectors, …) and normalise them to a simplex.
3.  Run a **projected‑gradient descent** (PGD) on the true max‑convolution
    objective.  The sub‑gradient of max(conv) is computed with an FFT‑based
    cross‑correlation; after each step the iterate is projected back onto
    the probability simplex.
4.  A cheap stochastic hill‑climb (mass transfer between two positions)
    is applied to each PGD result to squeeze out any remaining improvement.
5.  After all lengths are processed a final global hill‑climb refines the
    best candidate.

All steps respect the supplied time budget (default 1000 s).  The
returned list sums to 1, contains only non‑negative floats, and can be
directly fed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)] ,   n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative vector to ``new_n`` points by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:                     # protect against a zero vector
        new_seq = np.random.dirichlet(np.ones(new_n))
        s = new_seq.sum()
    return new_seq / s


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(v, 1.0 / len(v))
    else:
        w /= w_sum
    return w


def _generate_initial_candidates(n: int, prev_best):
    """
    Produce a heterogeneous pool of starting points for a given length ``n``.
    ``prev_best`` (if not ``None``) is up‑sampled and used as a warm‑start.
    """
    candidates = []

    # 1. warm‑start from the best sequence found so far (upsampled)
    if prev_best is not None:
        candidates.append(_upsample(prev_best, n))

    # 2. historic solution if the global variable exists
    hseq = globals().get('height_sequence_1')
    if isinstance(hseq, (list, np.ndarray)):
        h = np.asarray(hseq, dtype=np.float64)
        candidates.append(_upsample(h, n))

    # 3. random Dirichlet families (different concentrations)
    candidates.append(np.random.dirichlet(np.ones(n)))
    for conc in (0.2, 0.5, 0.8):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # 4. monotone decreasing random vectors
    dec = np.sort(np.random.rand(n))[::-1]
    candidates.append(dec / dec.sum())

    # 5. symmetric decreasing shape (mirror the left half)
    half = (n + 1) // 2
    left = np.sort(np.random.rand(half))[::-1]
    right = left[:-1] if n % 2 else left
    sym = np.concatenate([left, right[::-1]])
    candidates.append(sym / sym.sum())

    # 6. exponential (two‑sided)
    gamma = np.random.uniform(0.0005, 0.01)
    i = np.arange(n)
    exp_shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
    candidates.append(exp_shape / exp_shape.sum())

    # 7. power‑law
    alpha = np.random.uniform(0.5, 3.0)
    idx = np.arange(1, n + 1, dtype=np.float64)
    power = idx ** (-alpha)
    candidates.append(power / power.sum())

    # 8. double Gaussian bumps
    mu1 = np.random.uniform(0.2, 0.4) * n
    mu2 = np.random.uniform(0.6, 0.8) * n
    sigma = n / 10.0
    gauss = np.exp(-0.5 * ((i - mu1) / sigma) ** 2) + np.exp(-0.5 * ((i - mu2) / sigma) ** 2)
    candidates.append(gauss / gauss.sum())

    # 9. sparse vectors with a few non‑zero entries
    for k in (3, 5, 10):
        mask = np.zeros(n)
        idxs = np.random.choice(n, k, replace=False)
        mask[idxs] = np.random.rand(k)
        if mask.sum() > 0:
            candidates.append(mask / mask.sum())

    # Clean and normalise
    clean = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        clean.append(c / s)
    return clean


def _run_pgd(a0: np.ndarray, n: int, deadline: float, max_iters: int):
    """
    Projected‑gradient descent on the exact max‑convolution objective.
    Returns the best sequence seen and its exact value.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    last_improve = time.time()
    step0 = 0.5  # initial step size

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # evaluate current objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} PGD iter={it}] new best {val:.6f}")

        # sub‑gradient of max(conv)
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size schedule
        step = step0 / np.sqrt(it)

        # gradient step followed by projection onto the simplex
        a = _project_simplex(a - step * grad)

        # stop if we have been idle for a while
        if time.time() - last_improve > 8.0:
            break

    return best_a, best_val


def _local_hill_climb(a0: np.ndarray, n: int, deadline: float,
                     max_trials: int = 200_000):
    """
    Very cheap stochastic local search.
    Repeatedly moves a small mass ε from one index to another and keeps the
    move if it reduces the exact objective.
    """
    conv_fft_len = _next_pow2(2 * n - 1)

    best_a = a0.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    no_improve = 0
    trials = 0
    while time.time() < deadline and trials < max_trials and no_improve < 5_000:
        i, j = np.random.choice(n, 2, replace=False)
        eps = np.random.uniform(0.0, min(0.05, best_a[i]))
        if eps == 0.0:
            trials += 1
            continue
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = cand
            no_improve = 0
            print(f"[n={n} HillClimb] improved to {val:.6f}")
        else:
            no_improve += 1
        trials += 1

    return best_a, best_val


def propose_candidate(seed: int = 0, budget_s: float = 1000):
    """
    Search for a non‑negative step‑function (list of heights) that minimises

        F(a) = 2·n·max(conv(a,a)) / (sum(a)²)

    The algorithm performs a multi‑scale projected‑gradient descent,
    interleaved with stochastic hill‑climbing, while respecting the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float('inf')

    # hierarchical list of lengths (doubling each stage)
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # Build a diverse set of starting points
        candidates = _generate_initial_candidates(n, best_seq)

        # iteration budget – larger n gets fewer iterations
        max_iters = int(20000 * 256 / n) + 2000
        max_iters = max(max_iters, 1_000)   # at least a thousand

        for cid, init in enumerate(candidates):
            if time.time() > deadline:
                break

            # ----- Projected‑gradient descent -----
            a_pgd, val_pgd = _run_pgd(init, n, deadline, max_iters)
            if val_pgd < best_val:
                best_val = val_pgd
                best_seq = a_pgd.copy()
                print(f"[n={n} cand={cid}] PGD new global best {val_pgd:.6f}")

            # ----- Cheap local hill‑climb on the PGD result -----
            a_hc, val_hc = _local_hill_climb(a_pgd, n, deadline, max_trials=30_000)
            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.copy()
                print(f"[n={n} cand={cid}] Hill‑climb improved global best {val_hc:.6f}")

    # ------------------------------------------------------------------
    # Final global hill‑climb to squeeze out any remaining gain
    if best_seq is not None and time.time() < deadline:
        print(f"\n=== Final global hill‑climb (n={len(best_seq)}) ===")
        a_final, val_final = _local_hill_climb(best_seq, len(best_seq), deadline,
                                               max_trials=200_000)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final

    # Fallback – uniform distribution (should never be used)
    if best_seq is None:

        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    return best_seq.tolist()
