def _smooth_obj_and_grad(logits, lam, n, conv_fft_len, corr_fft_len):
    a = _softmax(logits)
    conv = _full_convolution(a, conv_fft_len)   # length 2*n-1
    # smooth max
    max_smooth = logsumexp(lam * conv) / lam
    # weights for gradient (softmax over conv)
    w = np.exp(lam * (conv - max_smooth))
    # gradient w.r.t a
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)   # d M_smooth / d a
    # chain rule through softmax
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)
    # full objective: 2 * n * max_smooth
    obj = 2.0 * n * max_smooth
    # gradient w.r.t logits: multiply by 2*n
    grad = 2.0 * n * grad_log
    return obj, grad

def _lbgfs_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val):
    # initial logits
    eps = 1e-15
    logits = np.log(a0 + eps)
    for lam in (5.0, 20.0, 100.0):
        if time.time() >= deadline:
            break
        # define objective function closure
        def fun(logit):
            val, grad = _smooth_obj_and_grad(logit, lam, n, conv_fft_len, corr_fft_len)
            return val, grad
        # run L-BFGS-B
        maxiter = 200
        # compute remaining time budget for this run
        available = deadline - time.time()
        # ensure we don't exceed
        options = {'maxiter': maxiter, 'disp': False}
        res = scipy.optimize.minimize(fun, logits, method='L-BFGS-B', jac=True, options=options)
        logits = res.x
        # evaluate true objective (non‑smooth)
        a = _softmax(logits)
        # clip tiny values to avoid numeric issues
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 0.01:
            a = np.random.dirichlet(np.ones(n))
            a_sum = a.sum()
        a = a / a_sum
        conv = _full_convolution(a, conv_fft_len)
        val_true = 2.0 * n * conv.max() / (a_sum ** 2)
        if val_true < best_val:
            best_val = val_true
            best_seq = a.copy()
            print(f"[n={n} L‑BFGS lam={lam}] new best {val_true:.6f}")
    return best_seq, best_val

def _generate_parametric_candidates_extended(n, warm_start):
    candidates = []
    # Add warm start and shifts
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))
    # Uniform
    candidates.append(np.full(n, 1.0 / n))
    # Dirichlet draws
    for conc in (0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        for _ in range(2):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)
    # Edge exponential + uniform mix (same as before)
    i = np.arange(n, dtype=np.float64)
    for alpha in np.logspace(-2, 1.0, num=15):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)
    # Edge power law
    for p in np.linspace(0.2, 5.0, num=15):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)
    # Gaussian edges
    for gamma in (0.05, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)
    # Sinusoidal half-wave
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)
    # Beta distributions
    x = (np.arange(n) + 0.5) / n   # avoid exact 0 and 1
    for a in (0.2, 0.5, 0.8, 1.2, 2.0):
        for b in (0.2, 0.5, 0.8, 1.2, 2.0):

            weights = np.random.rand(k)

    best_seq = None
    best_val = float('inf')

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # warm start if available
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        s = warm_start.sum()
        if s >= 0.01:
            warm_start = warm_start / s
        else:
            warm_start = None
    else:
        warm_start = None

    all_candidates = _generate_parametric_candidates_extended(n, warm_start)

    cand_vals = []

        s = cand.sum()

            s = cand.sum()

    # pick top k
    cand_vals.sort(key=lambda x: x[0])
    top_k = 12

    # deep optimization for each top candidate
    for start_id, a0 in enumerate(top_candidates):

            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,

"""
Hybrid search for a non‑negative step function that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)²)

The algorithm consists of three stages:

1. **Rich parametric seeding** – generate many diverse shapes
   (uniform, Dirichlet, edge‑exponential, edge‑power‑law, mixtures,
   Beta‑pdf, random sparse spikes, sinusoidal half‑wave, etc.) and keep
   the few with the smallest fast‑computed objective.

2. **L‑BFGS on a smooth‑max surrogate** – each selected start is
   optimised with SciPy’s L‑BFGS‑B on the log‑sum‑exp approximation of the
   max‑convolution.  The temperature (λ) is increased in stages
   (λ≈5, 20, 100) so the surrogate gradually becomes the true max.  The
   variables are logits of a soft‑max to stay on the simplex.

3. **Exact sub‑gradient refinement** – a few exponentiated‑gradient steps
   using the true max‑convolution (active‑set weighting) sharpen the
   solution.

If time permits a second pass on a doubled grid (n≈8192) is performed
by linear interpolation of the best 4096‑vector followed by the same
optimisation pipeline.

All numbers are forced to be non‑negative and ≤ 1000 before returning.
"""

import time
import numpy as np
import scipy.optimize
from scipy.special import logsumexp


# ----------------------------------------------------------------------
# FFT‑based convolution / correlation utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)-1 using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _softmax(logits: np.ndarray) -> np.ndarray:
    """Stable soft‑max (maps ℝⁿ → simplex)."""
    m = logits.max()
    e = np.exp(logits - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smooth‑max objective and its gradient (for L‑BFGS)
def _smooth_obj_and_grad(logits: np.ndarray,
                         lam: float,
                         n: int,
                         conv_fft_len: int,
                         corr_fft_len: int):
    """
    Returns (value, gradient) of  2·n·smooth_max(conv(a))  where
    a = softmax(logits) and smooth_max uses log‑sum‑exp with temperature λ.
    """
    a = _softmax(logits)

    # convolution a*a
    conv = _full_convolution(a, conv_fft_len)          # length 2n‑1

    # smooth approximation of max(conv)
    max_smooth = logsumexp(lam * conv) / lam

    # weights w_i = exp(λ·(conv_i‑max_smooth)), they sum to 1
    w = np.exp(lam * (conv - max_smooth))

    # gradient of smooth_max w.r.t a  (factor 2 comes from derivative of a*a)
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    # chain rule through soft‑max
    avg = np.dot(a, grad_a)
    grad_logits = a * (grad_a - avg)

    # full objective (2·n·max_smooth) and its gradient w.r.t logits
    value = 2.0 * n * max_smooth
    grad = 2.0 * n * grad_logits
    return value, grad


# ----------------------------------------------------------------------
# L‑BFGS optimisation on a given start vector
def _lbgfs_optimize(a0: np.ndarray,
                    n: int,
                    conv_fft_len: int,
                    corr_fft_len: int,
                    deadline: float,
                    best_seq: np.ndarray | None,
                    best_val: float):
    """
    Run L‑BFGS‑B on a sequence of temperature stages.
    Returns the best vector/value found during the run.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)           # initialise logits from the start

    # temperature schedule – larger λ → tighter approximation to max
    for lam in (5.0, 20.0, 100.0):
        if time.time() >= deadline:
            break

        def fun(logit_vec):
            return _smooth_obj_and_grad(logit_vec, lam, n,
                                        conv_fft_len, corr_fft_len)

        # L‑BFGS‑B – we keep the iteration budget modest (≈200) per stage
        res = scipy.optimize.minimize(fun,
                                      logits,
                                      method='L-BFGS-B',
                                      jac=True,
                                      options={'maxiter': 200, 'disp': False})
        logits = res.x

        # evaluate the *true* objective (no smooth approximation)
        a = _softmax(logits)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max() / (s ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[n={n} L‑BFGS λ={lam:.0f}] new best  {val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient sub‑gradient refinement (exact max)
def _subgradient_refinement(a0: np.ndarray,
                            n: int,
                            conv_fft_len: int,
                            corr_fft_len: int,
                            deadline: float,
                            best_seq: np.ndarray | None,
                            best_val: float):
    """
    A few exponentiated‑gradient steps on the true max‑convolution.
    """
    # initialise from the best vector seen so far (or from the start)
    a = best_seq if best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a = a / a.sum()

    eta = 0.6                     # step size
    last_improve = time.time()
    max_iter = 15000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it:05d}] new best {val:.6f}")

        # active set where the maximum is attained
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # stop early if no improvement for a while
        if time.time() - last_improve > 5.0:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Diverse parametric candidate generator
def _generate_parametric_candidates_extended(n: int,
                                             warm_start: np.ndarray | None):
    """
    Produce a broad set of plausible probability vectors.
    The set includes the warm‑start, uniform, Dirichlet draws,
    edge‑exponential / power‑law families, Gaussian edges,
    sinusoidal half‑wave, Beta‑pdf shapes and random sparse spikes.
    """
    candidates = []

    # 1) Warm‑start and rotations
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # 2) Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # 3) Random Dirichlet vectors (different concentrations)
    for conc in (0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        for _ in range(2):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # 4) Edge‑exponential families with optional uniform mixing
    for alpha in np.logspace(-2, 1.0, num=15):          # 0.01 … 10
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # 5) Edge‑power‑law families
    for p in np.linspace(0.2, 5.0, num=15):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # 6) Two‑Gaussian edges (different widths)
    for gamma in (0.05, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # 7) Positive sinusoidal half‑wave
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # 8) Beta‑distribution shapes (flexible edge / interior concentration)
    x = (np.arange(n) + 0.5) / n       # avoid exact 0/1
    for a in (0.2, 0.5, 0.8, 1.2, 2.0):
        for b in (0.2, 0.5, 0.8, 1.2, 2.0):
            shape = np.power(x, a - 1) * np.power(1 - x, b - 1)
            shape = np.nan_to_num(shape, nan=0.0, posinf=0.0, neginf=0.0)
            s = shape.sum()
            if s < 1e-12:
                continue
            shape /= s
            candidates.append(shape)

    # 9) Random sparse spikes (2‑5 non‑zero entries)
    for k in (2, 3, 4):
        for _ in range(12):
            idx = np.random.choice(n, size=k, replace=False)
            weights = np.random.rand(k)
            weights /= weights.sum()
            vec = np.zeros(n)
            vec[idx] = weights
            candidates.append(vec)

    return candidates


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a low‑objective step function.
    Returns the best sequence found (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # small safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Primary grid size (the size that gave the previous best)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Assemble a warm start from a previously discovered construction, if any
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        s = warm_start.sum()
        if s >= 0.01:
            warm_start = warm_start / s
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a large pool of plausible candidates
    all_candidates = _generate_parametric_candidates_extended(n, warm_start)

    cand_vals = []
    for idx, cand in enumerate(all_candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s

        # cheap evaluation (uses the same FFT length as the optimiser)
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max() / (s ** 2)
        cand_vals.append((val, cand))

    # keep the most promising starts
    cand_vals.sort(key=lambda x: x[0])
    top_k = 12
    top_candidates = [c for _, c in cand_vals[:top_k]]

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation (L‑BFGS + sub‑gradient) for each start
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"\n--- Optimising start #{start_id + 1}/{len(top_candidates)} (n={n}) ---")
        best_seq, best_val = _lbgfs_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,

            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,

    # ------------------------------------------------------------------
    # Stage 3 – if time permits, repeat on a doubled grid (n = 8192)
    time_left = deadline - time.time()
    if time_left > 30.0:                     # need a small buffer
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation from the coarse grid to the finer one
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big = a_big / a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("\n--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _lbgfs_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,

            best_seq,

            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded, return a uniform vector.
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"\n*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
