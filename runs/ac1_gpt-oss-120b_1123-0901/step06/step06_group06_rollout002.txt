    a = np.exp(logits - np.max(logits))
    a = a / np.sum(a)
    conv = _conv_fft(a, fft_len)

    w_sum = w.sum()

def _objective_and_gradient(logits, n, conv_fft_len, corr_fft_len, lam):
    # compute a via softmax
    max_log = np.max(logits)
    exp_logits = np.exp(logits - max_log)
    a = exp_logits / np.sum(exp_logits)
    # convolution
    conv = _conv_fft(a, conv_fft_len)
    # max conv
    max_c = np.max(conv)
    # compute w
    w = np.exp(lam * (conv - max_c))
    w_sum = np.sum(w)
    if w_sum == 0.0:
        w = np.full_like(conv, 1.0/conv.size)
    else:
        w = w / w_sum
    # gradient wrt a
    grad_a = 2.0 * _corr_fft_fixed(w, a, corr_fft_len)
    # gradient wrt logits
    avg = np.dot(a, grad_a)
    grad_logits = a * (grad_a - avg)
    # smooth objective to be minimized
    smooth_val = np.dot(w, conv)  # = weighted avg of conv, approx max_c
    val = 2.0 * n * smooth_val
    return val, grad_logits

"""
Hybrid continuation‑L‑BFGS optimiser for the step‑function bound.

The algorithm:
* builds a diverse pool of initial shapes (Dirichlet, power‑law, exponential,
  sinusoidal, Gaussian, arcsine, sparse spikes, and a resized copy of the best
  solution from a smaller length);
* for each candidate it runs a sequence of smooth‑max surrogates with increasing
  temperature (lam) – a continuation method – and minimises each surrogate with
  L‑BFGS‑B on the unconstrained logits (soft‑max guarantees a ≥0 and Σa=1).
* after the continuation a final fine‑tune with many L‑BFGS iterations is
  performed on the best found sequence.
* the exact objective  F(a)=2·n·max(a*a)/(∑a)² (with the sum fixed to 1) is
  evaluated after every optimisation stage, and the globally best sequence is
  returned.

No closures or lambda functions are used; all helpers are top‑level.
"""

import time
import math
import numpy as np
from scipy import optimize

# ---------- FFT helpers -------------------------------------------------

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (int(math.ceil(math.log2(x))) if x > 0 else 0)


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length fft_len."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft_fixed(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]
    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ---------- Objective & gradient (smooth‑max surrogate) ---------------

def _objective_and_gradient(logits: np.ndarray,
                            n: int,
                            conv_fft_len: int,
                            corr_fft_len: int,
                            lam: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max surrogate of the true objective and its gradient w.r.t. logits.

    logits – unconstrained parameters; a = softmax(logits) is a point on the
    probability simplex (Σa = 1, a_i ≥ 0).

    lam – temperature; as lam → ∞ the surrogate tends to the true max.
    """
    # soft‑max → simplex
    max_log = np.max(logits)
    exp_logits = np.exp(logits - max_log)
    a = exp_logits / exp_logits.sum()

    # convolution a*a
    conv = _conv_fft(a, conv_fft_len)
    max_c = np.max(conv)

    # soft‑max weights over the convolution entries
    w = np.exp(lam * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum

    # gradient of weighted sum Σ w·conv w.r.t. a
    grad_a = 2.0 * _corr_fft_fixed(w, a, corr_fft_len)

    # back‑prop through soft‑max
    avg = np.dot(a, grad_a)
    grad_logits = a * (grad_a - avg)

    # surrogate objective (smooth approximation of max)
    smooth_val = np.dot(w, conv)          # = E_w[conv] ≤ max_c
    val = 2.0 * n * smooth_val

    return val, grad_logits


def _obj_and_grad_wrapper(logits: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          lam: float) -> tuple[float, np.ndarray]:
    """Thin wrapper required by scipy.optimize.minimize (no closures)."""
    return _objective_and_gradient(logits, n, conv_fft_len, corr_fft_len, lam)


# ---------- Candidate generation ----------------------------------------

def _build_initial_candidates(n: int) -> list[np.ndarray]:
    """
    Produce a diversified list of starting points for the optimisation.
    All candidates are normalised to sum to 1 (scaling does not affect the
    objective, but a fixed sum makes the optimiser more stable).
    """
    cand = []

    # 1) warm‑start from a previously found solution (if any)
    if "height_sequence_1" in globals():
        h = np.asarray(height_sequence_1, dtype=np.float64)
        if h.shape[0] > n:
            h = h[:n]
        else:
            h = np.pad(h, (0, n - h.shape[0]))
        h = np.maximum(h, 0.0)
        if h.sum() < 1e-8:
            h = np.random.dirichlet(np.ones(n))
        else:
            h = h / h.sum()
        cand.append(h)

    # 2) Dirichlet (uniform concentration)
    cand.append(np.random.dirichlet(np.ones(n)))

    # 3) spiky Dirichlet (low concentration)
    for conc in (0.2, 0.4, 0.6, 0.8):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 4) over‑dispersed Dirichlet (high concentration → near‑uniform)
    for conc in (1.5, 2.0, 3.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 5) edge‑concentrated power‑law shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 9):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape = shape / shape.sum()
            cand.append(shape)

    # 6) two‑sided exponential decay
    for gamma in (0.0008, 0.0015, 0.003, 0.006):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape = shape / shape.sum()
        cand.append(shape)

    # 7) cosine window (single bump)
    i = np.arange(n, dtype=np.float64)
    coswin = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    coswin = coswin / coswin.sum()
    cand.append(coswin)

    # 8) half‑sine window
    sinwin = np.sin(np.pi * i / (n - 1))
    sinwin = sinwin / sinwin.sum()
    cand.append(sinwin)

    # 9) discrete Gaussian (σ ∝ n)
    sigma = n / 8.0
    gauss = np.exp(-0.5 * ((i - (n - 1) / 2) / sigma) ** 2)
    gauss = gauss / gauss.sum()
    cand.append(gauss)

    # 10) truncated arcsine (inverse‑sqrt edges)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcs = arcs / arcs.sum()
    cand.append(arcs)

    # 11) random sparse pattern (≈ √n spikes)
    m_spikes = int(np.sqrt(2 * n))
    sparse = np.zeros(n, dtype=np.float64)
    idxs = np.random.choice(n, size=m_spikes, replace=False)
    sparse[idxs] = 1.0
    if sparse.sum() > 0:
        sparse = sparse / sparse.sum()
        cand.append(sparse)

    return cand


def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence to a new length by linear interpolation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    resized = np.interp(new_x, old_x, seq)
    return resized


# ---------- Main search routine -----------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    F(a)=2·n·max(a*a)/(sum(a)²).  The routine respects the time budget and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # Sequence lengths to explore (larger lengths give more expressive power)
    lengths = [1024, 2048, 4096, 8192]

    # Store the best sequence per completed length for up‑scaling
    best_by_len = {}

    # Temperature schedule for continuation (smooth‑max → true max)
    lam_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0,
                    100.0, 200.0, 500.0, 1000.0, 2000.0]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_len = 2 * n - 1
        conv_fft_len = _next_pow2(conv_len)
        corr_fft_len = _next_pow2(conv_len + n - 1)

        # Build candidate pool (including up‑scaled best from previous size)
        candidates = _build_initial_candidates(n)
        if n in best_by_len:
            # Add a resized copy of the best from the previous length
            resized = _resize_sequence(best_by_len[n], n)
            resized = np.maximum(resized, 0.0)
            if resized.sum() < 1e-12:
                resized = np.random.dirichlet(np.ones(n))
            resized = resized / resized.sum()
            candidates.append(resized)

        # Optimise each candidate
        for cid, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # Ensure correct length, non‑negativity and normalisation
            if a0.shape[0] > n:
                a = a0[:n]
            else:
                a = np.pad(a0, (0, n - a0.shape[0]))
            a = np.maximum(a, 0.0)
            if a.sum() < 1e-12:
                a = np.random.dirichlet(np.ones(n))
            a = a / a.sum()

            # Logits for the soft‑max parametrisation
            logits = np.log(a + 1e-15)

            # Continuation over increasing λ
            for lam in lam_schedule:
                if time.time() > deadline:
                    break

                # L‑BFGS‑B optimisation of the smooth‑max surrogate
                opt_res = optimize.minimize(fun=_obj_and_grad_wrapper,
                                            x0=logits,
                                            args=(n, conv_fft_len, corr_fft_len, lam),
                                            method='L-BFGS-B',
                                            jac=True,
                                            options={'maxiter': 50,
                                                     'ftol': 1e-12,
                                                     'gtol': 1e-12,
                                                     'disp': False})

                logits = opt_res.x

                # Recover a from logits
                mx = np.max(logits)
                a_opt = np.exp(logits - mx)
                a_opt = a_opt / a_opt.sum()

                # Exact objective (using true max of the convolution)
                conv = _conv_fft(a_opt, conv_fft_len)
                max_c = np.max(conv)
                val_exact = 2.0 * n * max_c

                if val_exact < best_val:
                    best_val = val_exact
                    best_seq = a_opt.copy()
                    print(f"[n={n} cand={cid} lam={lam:.1f} it={opt_res.nit}] "
                          f"new best {val_exact:.6f}")

                # Early stop if optimiser made no progress
                if opt_res.nit == 0:
                    break

        # Store the best for this length (if any) for possible up‑scaling
        if best_seq is not None and len(best_seq) == n:
            best_by_len[n] = best_seq.copy()
            # -------- Fine‑tune the current global best for this length -------
            print(f"[n={n}] fine‑tuning global best (val={best_val:.6f})")
            logits = np.log(best_seq + 1e-15)
            for lam in lam_schedule:
                if time.time() > deadline:
                    break
                opt_res = optimize.minimize(fun=_obj_and_grad_wrapper,
                                            x0=logits,
                                            args=(n, conv_fft_len, corr_fft_len, lam),
                                            method='L-BFGS-B',
                                            jac=True,
                                            options={'maxiter': 200,
                                                     'ftol': 1e-13,
                                                     'gtol': 1e-13,
                                                     'disp': False})
                logits = opt_res.x
                mx = np.max(logits)
                a_fine = np.exp(logits - mx)
                a_fine = a_fine / a_fine.sum()
                conv = _conv_fft(a_fine, conv_fft_len)
                val_exact = 2.0 * n * np.max(conv)
                if val_exact < best_val:
                    best_val = val_exact
                    best_seq = a_fine.copy()
                    print(f"[n={n} fine lam={lam:.1f}] new best {val_exact:.6f}")

    # ----------------------------------------------------------------------
    # Fallback: uniform distribution (should never be needed)
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = 2.0 * fallback_n * np.max(_conv_fft(best_seq,
                                                      _next_pow2(2 * fallback_n - 1)))

    # Store for potential reuse by subsequent calls
    global height_sequence_1
    height_sequence_1 = best_seq.tolist()

    return best_seq.tolist()
