def project_onto_simplex(v):
    # v is a 1-D np array
    # ensure sum(v)=1, entries >=0
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
propose_candidate
=================

Search for a non‑negative step function that (approximately) minimises

    F(a) = 2 * n * max(conv(a, a)) / (sum(a)**2)

with the constraint sum(a)=1.  The optimisation proceeds in three stages:

1.  A diverse pool of seed vectors is built (uniform, edge‑heavy,
    arcsine‑like, Gaussian mixtures, random Dirichlet draws and the
    previous best if any).

2.  Each seed is first refined by Adam on a smooth‑max (log‑sum‑exp)
    surrogate of the convolution peak.  The Adam schedule is
    deliberately long for larger lengths.

3.  The result of Adam is then polished by a *projected‑gradient*
    descent on the exact ℓ∞‑norm.  The sub‑gradient of the
    max‑convolution is obtained from the active set and the iterate is
    projected onto the probability simplex after every step.

During the search many candidates are jittered around the current best
to escape local minima.  All work respects the wall‑clock budget
(`budget_s`, default 1000 s).  The function finally returns the best
sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a*a using an FFT of power‑of‑two length.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of
    max(conv(a, a)).
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)], i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Projection onto the simplex (sum = 1, non‑negative)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero after projection
        return np.zeros_like(v)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimisation on a smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp surrogate of max(conv).
    Returns the best vector seen (normalised to sum=1) and its exact
    objective value 2·n·max(conv).
    """
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05

    lam = 1.0
    lam_factor = 1.02

    # give larger budgets for the long vectors
    max_iters = int(80000 * 1024 / n) + 3000
    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max to stay in the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0:
            w = w / w_sum
        else:
            w = np.full_like(w, 1.0 / w.size)

        # gradient of smooth‑max w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # back‑propagation through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase temperature (sharpen smooth‑max)
        lam = min(lam * lam_factor, 1e8)

        # exact objective (sum(a)=1)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # occasional jitter if stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact projected‑gradient refinement on the ℓ∞‑norm
# ----------------------------------------------------------------------


def _projected_gradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 20000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true ℓ∞‑norm with projection onto the
    probability simplex.  Returns the best iterate and its objective.
    """
    a = a0 / a0.sum()
    best_a = a.copy()
    conv = _full_convolution(a, fft_len=conv_fft_len)
    max_c = conv.max()
    best_val = 2.0 * n * max_c

    eta = eta0
    last_improve = time.time()
    tol = 1e-12

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # convolution & active set
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=tol)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            # fallback – should rarely happen
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0

        # sub‑gradient of max(conv) w.r.t. a
        grad = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # gradient step
        a_next = a - eta * grad
        a_next = _project_onto_simplex(a_next)

        # evaluate
        conv_next = _full_convolution(a_next, fft_len=conv_fft_len)
        max_c_next = conv_next.max()
        val_next = 2.0 * n * max_c_next

        # keep the best ever seen
        if val_next < best_val:
            best_val = val_next
            best_a = a_next.copy()
            last_improve = time.time()

        # accept the step (even if not improving – this is standard sub‑grad.)
        a = a_next

        # decay step size slowly
        eta *= 0.9985
        if eta < 1e-7:
            eta = 1e-7

        # stop early if no progress for a while
        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the supplied
    evaluation function.  The routine respects the wall‑clock budget and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    best_seq = None
    best_val = float("inf")

    # Lengths to explore – start small for cheap warm‑starts, end with a
    # large vector (the larger the better, up to the time limit).
    lengths = [2048, 4096, 8192, 16384, 32768]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

        # ------------------------------------------------------------------
        # Build a heterogeneous set of seeds
        # ------------------------------------------------------------------
        candidates = []

        # (1) uniform
        candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # (2) previous best (if same size)
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.asarray(best_seq, dtype=np.float64))

        # (3) up‑sample / down‑sample the current global best
        if best_seq is not None:
            old = np.asarray(best_seq, dtype=np.float64)
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            interp = np.interp(x_new, x_old, old)
            if interp.sum() > 0:
                interp = interp / interp.sum()
            else:
                interp = np.random.dirichlet(np.ones(n))
            candidates.append(interp)

        # (4) arcsine‑like edge‑heavy shape
        i = np.arange(n, dtype=np.float64)
        eps = 1e-12
        arcsine = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
        arcsine = arcsine / arcsine.sum()
        candidates.append(arcsine)

        # (5) symmetric power‑law shapes (including mildly centre‑heavy)
        prod = (i + 1) * (n - i)
        for gamma in [-2.5, -2.0, -1.5, -1.0, -0.5, -0.25, 0.0, 0.25, 0.5]:
            shape = np.where(prod > 0, prod ** gamma, 0.0)
            if shape.sum() > 0:
                shape = shape / shape.sum()
                candidates.append(shape)

        # (6) two‑Gaussian edge shapes
        for sigma in [0.05, 0.1, 0.2, 0.4, 0.8]:
            shape = np.exp(-sigma * (i ** 2)) + np.exp(-sigma * ((n - 1 - i) ** 2))
            shape = shape / shape.sum()
            candidates.append(shape)

        # (7) random Dirichlet draws
        for _ in range(3):
            candidates.append(np.random.dirichlet(np.ones(n)))

        # Limit the pool size for the large vectors (budget‑aware)
        max_cand = 20 if n <= 8192 else 12
        candidates = candidates[:max_cand]

        # ------------------------------------------------------------------
        # Optimise every seed
        # ------------------------------------------------------------------
        for cand_id, cand in enumerate(candidates):
            if time.time() > deadline:
                break

            try:
                # Ensure normalisation before any optimisation
                cand = np.maximum(cand, 0.0)
                if cand.sum() == 0:
                    cand = np.random.dirichlet(np.ones(n))
                cand = cand / cand.sum()

                # Phase 1: Adam on smooth‑max
                a_adam, val_adam = _adam_optimize(
                    cand, n, conv_fft_len, corr_fft_len, deadline
                )

                # Phase 2: exact projected‑gradient refinement
                a_ref, val_ref = _projected_gradient_refine(
                    a_adam, n, conv_fft_len, corr_fft_len, deadline
                )

                # Keep the best candidate seen so far
                if val_ref < best_val:
                    best_val = val_ref
                    best_seq = a_ref.copy()
                    print(
                        f"[n={n} cand={cand_id}] new global best {best_val:.6f}"
                    )
                elif val_adam < best_val:
                    best_val = val_adam
                    best_seq = a_adam.copy()
                    print(
                        f"[n={n} cand={cand_id}] new global best (Adam) {best_val:.6f}"
                    )
            except Exception as exc:
                print(f"[n={n} cand={cand_id}] exception: {exc}")

        # ------------------------------------------------------------------
        # Small random jitters around the current best of this length
        # ------------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            for jitter_id in range(5):
                if time.time() > deadline:
                    break
                logits = np.log(np.maximum(best_seq, 1e-12))
                logits += np.random.normal(scale=0.02, size=n)
                a0 = np.exp(logits - np.max(logits))
                a0 = a0 / a0.sum()

                a_jit, val_jit = _projected_gradient_refine(
                    a0, n, conv_fft_len, corr_fft_len, deadline
                )
                if val_jit < best_val:
                    best_val = val_jit
                    best_seq = a_jit.copy()
                    print(
                        f"[n={n} jitter={jitter_id}] new global best {best_val:.6f}"
                    )

    # ----------------------------------------------------------------------
    # Fallback: uniform vector if nothing succeeded (should never happen)
    # ----------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
