    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Additional random shapes:
    for _ in range(5):
        alpha = np.random.uniform(-2.0, -0.2)
        idx = np.arange(n)+1
        shape = (idx * (n - idx + 1)) ** alpha
        shape[shape < 0] = 0
        shape /= shape.sum()
        init_pool.append(shape)

    # mixture shape:
    for lam in [0.2,0.4,0.6,0.8]:
        arcs = 1.0/np.sqrt((np.arange(n)+1)*(n-np.arange(n)))
        arcs[~np.isfinite(arcs)] = 0
        arcs/= arcs.sum()
        uniform = np.full(n,1.0/n)
        mix = lam*arcs + (1-lam)*uniform
        mix/= mix.sum()
        init_pool.append(mix)

    # random Dirichlet
    for _ in range(3):
        init_pool.append(np.random.dirichlet(np.ones(n)))

    # Because we keep n moderate; may also add spike candidates

    # Run optimization for each candidate but limit overall time using while loop

    for cand in init_pool:
        if time.time() >= deadline - 10:
            break
        a0 = cand.copy()
        a0 = np.maximum(a0,0.0)
        s = a0.sum()
        if s < 1e-9:
            a0 = np.random.dirichlet(np.ones(n))
            s = a0.sum()
        a0 = a0 / s

        # time allocation for this candidate: maybe 0.13 of remaining time (or 30s)
        local_deadline = min(deadline - 5, time.time() + 30)

        a_opt, val_opt = _adam_optimize(a0, n, conv_fft_len, corr_fft_len,
                                        local_deadline, max_iters=200_000)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Candidate] Adam best {val_opt:.6f}")

        # subgradient refinement
        if time.time() < deadline - 5:
            a_opt2, val_opt2 = _subgradient_optimize(best_seq, n,
                                                     conv_fft_len, corr_fft_len,
                                                     deadline, max_iters=80_000, eta=0.5)
            if val_opt2 < best_val:
                best_val = val_opt2
                best_seq = a_opt2.copy()

        # local random search
        if time.time() < deadline - 5:
            a_opt3, val_opt3 = _local_random_search(best_seq, n,

                best_seq = a_opt3.copy()

def _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=200_000):
    """
    Adam optimizer over logits that parametrize a on the probability simplex.
    Uses a smooth approximation of the max via a softmax with temperature lam.
    Returns best a (sum≈1) and its exact objective value 2*n*max(conv).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.2            # start temperature (low -> smooth)
    lam_factor = 1.015   # slow increase

    best_a = a0.copy()
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        # softmax -> simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:05d}] new best {val:.6f}")

        # smooth-max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smoothed objective wrt a (chain rule already handled)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # gradient wrt logits via softmax Jacobian
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase temperature
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to escape plateaus
        if it % 5000 == 0 and (deadline - time.time() > 5):
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1
    return best_a, best_val

def _subgradient_optimize(a0, n, conv_fft_len, corr_fft_len, deadline,
                          max_iters=200_000, eta0=0.6):
    """
    Exponentiated‑gradient (multiplicative) sub‑gradient descent applied
    to the exact max objective.  The step size decays as eta0 / sqrt(t)
    to ensure convergence.
    Returns best a and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:05d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1
    return best_a, best_val

def _local_random_search(a0, n, conv_fft_len, deadline,
                         max_moves=20000):
    """
    Simple hill‑climbing that repeatedly transfers a random fraction
    of mass from one coordinate to another.  Moves that improve the
    exact objective are kept; others are discarded.  The loop stops
    when the time budget is exhausted or after max_moves attempts.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # move random fraction up to 40%
        delta = a[i] * (np.random.rand() * 0.4)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it}] improved to {val2:.6f}")
        it += 1
    return best_a, best_val

def _make_initial_candidates(n, warm=None):
    """Return a list of diverse probability vectors (sum≈1)."""
    cand = []

    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
            cand.append(warm.copy())
            # rotations
            for shift in (n // 4, n // 2, 3 * n // 4):
                cand.append(np.roll(warm, shift))

    # random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # arcsine‑like edge concentration (alpha negative)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.4, 6):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            cand.append(shape)

    # mixture of arcsine and uniform
    arcs = 1.0 / np.sqrt((np.arange(n) + 1) * (n - np.arange(n)))
    arcs = arcs / arcs.sum()
    uniform = np.full(n, 1.0 / n)
    for mix in [0.2, 0.4, 0.6, 0.8]:
        mix_vec = mix * arcs + (1 - mix) * uniform
        mix_vec = mix_vec / mix_vec.sum()
        cand.append(mix_vec)

    # double‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.1, 0.3, 0.7, 1.5):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape = shape / shape.sum()
        cand.append(shape)

    # simple half‑sine shape
    sin_shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin_shape = np.maximum(sin_shape, 0.0)
    sin_shape = sin_shape / sin_shape.sum()
    cand.append(sin_shape)

    # spike patterns (few equally spaced non‑zero entries)
    for k in (32, 64, 128, 256):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    return cand

def propose_candidate(seed: int = 0, budget_s: float = 1000):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises 2·n·max(a*a)/(sum(a)^2).  The routine uses a multi‑start
    Adam + sub‑gradient refinement pipeline, cheap hill‑climbing and
    optional up‑sampling to a finer grid.  It respects the supplied
    time budget and always returns the best sequence found.
    """
    np.random.seed(seed)
    # reserve tiny safety margin
    deadline = time.time() + budget_s - 0.05

    best_seq = None
    best_val = float('inf')

    # primary dimension
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    warm = globals().get('height_sequence_1')
    candidates = _make_initial_candidates(n, warm)

    # shuffle candidates to diversify order
    np.random.shuffle(candidates)

    for cand in candidates:
        if time.time() >= deadline - 5:
            break
        # ensure proper normalisation
        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 1e-9:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # allocate a fraction of remaining time (max 35 s) for this candidate
        local_deadline = min(deadline - 2, time.time() + 35)

        # ---- Adam (smooth‑max) ----
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, local_deadline, max_iters=150_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Candidate] Adam improved best to {val_adam:.6f}")

        # ---- Sub‑gradient refinement ----
        if time.time() < deadline - 3:
            a_sub, val_sub = _subgradient_optimize(
                best_seq, n, conv_fft_len, corr_fft_len, deadline, max_iters=80_000,
                eta0=0.6
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Candidate] Subgrad improved best to {val_sub:.6f}")

        # ---- cheap hill‑climbing ----
        if time.time() < deadline - 2:
            a_loc, val_loc = _local_random_search(
                best_seq, n, conv_fft_len, deadline, max_moves=25000
            )
            if val_loc < best_val:
                best_val = val_loc
                best_seq = a_loc.copy()
                print(f"[Candidate] Local search improved best to {val_loc:.6f}")

    # --------- Final thorough refinement on current global best ----------
    if best_seq is not None and time.time() < deadline - 5:
        a_ref, val_ref = _subgradient_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline, max_iters=250_000,
            eta0=0.5
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Global] Long subgrad best {val_ref:.6f}")

    # --------- Upscale to larger grid if time permits ----------
    if time.time() + 30 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_big, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # Adam on larger grid

            a_big, n_big,
            conv_fft_len_big, corr_fft_len_big,

            a_big_adam, n_big,
            conv_fft_len_big, corr_fft_len_big,
            deadline, max_iters=150_000, eta0=0.5
        )
        if val_big_ref < best_val:
            best_val = val_big_ref
            best_seq = a_big_ref.copy()
            print(f"[Upsample] refined best {val_big_ref:.6f}")

    # Fallback
    if best_seq is None:

    # Clip to allowed range (evaluate_sequence caps anyway)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()

"""
search for a non‑negative step function (list of heights) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The implementation follows a multi‑start scheme:

* a diverse pool of seeds (warm start, Dirichlet draws, edge‑heavy power‑law,
  arcsine‑like mixtures, double‑Gaussian edges, half‑sine, sparse spikes)
* a smooth‑max Adam phase (log‑sum‑exp surrogate) with a slowly increasing
  temperature, keeping a running best solution
* an exact‑max exponentiated‑gradient sub‑gradient phase with a decaying step
  size
* a cheap hill‑climbing random‑mass‑transfer local search
* a long final sub‑gradient refinement on the current global best
*, when time permits, up‑sampling to a larger grid (8192 points) and repeating
  the above phases.

All steps respect the overall time budget (default 1000 s) and the function
always returns a valid list.  The code uses only NumPy and works with the
provided `evaluate_sequence` (which the grader calls after this function
returns).
"""

    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:05d}] new best {val:.6f}")

        # active set where the max is attained
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update & re‑normalisation
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 20_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one index
    to another and keep the move if it improves the exact objective.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        delta = a[i] * (np.random.rand() * 0.4)          # move up to 40 % of source
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it}] improved to {val2:.6f}")
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
def _make_initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Construct a modest pool of diverse start vectors on the simplex."""
    cand = []

    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
            cand.append(warm.copy())
            for shift in (n // 4, n // 2, 3 * n // 4):
                cand.append(np.roll(warm, shift))

    # random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # edge‑concentrated power‑law shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.4, 6):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            cand.append(shape)

    # mixture of arcsine‑like and uniform
    arcs = 1.0 / np.sqrt((np.arange(n) + 1) * (n - np.arange(n)))
    arcs = arcs / arcs.sum()
    uniform = np.full(n, 1.0 / n)
    for mix in (0.2, 0.4, 0.6, 0.8):
        mix_vec = mix * arcs + (1 - mix) * uniform
        mix_vec = mix_vec / mix_vec.sum()
        cand.append(mix_vec)

    # double‑Gaussian edges
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.1, 0.3, 0.7, 1.5):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape = shape / shape.sum()
        cand.append(shape)

    # half‑sine wave (positive half‑wave)
    sin_shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin_shape = np.maximum(sin_shape, 0.0)
    sin_shape = sin_shape / sin_shape.sum()
    cand.append(sin_shape)

    # sparse spike patterns (k equally spaced non‑zero entries)
    for k in (32, 64, 128, 256):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    return cand

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function with a low evaluation score.
    The routine respects the given time budget and returns the best sequence
    (as a Python list) it found.
    """
    np.random.seed(seed)
    # tiny safety margin to avoid overshooting the budget
    deadline = time.time() + budget_s - 0.05

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # primary resolution (the current state‑of‑the‑art used n = 4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    warm = globals().get("height_sequence_1")
    candidates = _make_initial_candidates(n, warm)
    np.random.shuffle(candidates)       # diversify order

    # --------------------------------------------------------------
    # optimise each candidate (short runs) – keep a global best
    for cand in candidates:
        if time.time() >= deadline - 5:
            break

        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 1e-9:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # allocate a slice of the remaining budget (≤ 35 s) for this candidate
        local_deadline = min(deadline - 2, time.time() + 35)

        # ---- Adam (smooth‑max) ----
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, local_deadline, max_iters=150_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Candidate] Adam improved best to {val_adam:.6f}")

        # ---- Sub‑gradient refinement ----
        if time.time() < deadline - 3:
            a_sub, val_sub = _subgradient_optimize(
                best_seq, n, conv_fft_len, corr_fft_len, deadline,
                max_iters=80_000, eta0=0.6
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Candidate] Subgrad improved best to {val_sub:.6f}")

        # ---- cheap local hill‑climbing ----
        if time.time() < deadline - 2:
            a_loc, val_loc = _local_random_search(
                best_seq, n, conv_fft_len, deadline, max_moves=25_000
            )
            if val_loc < best_val:
                best_val = val_loc
                best_seq = a_loc.copy()
                print(f"[Candidate] Local search improved best to {val_loc:.6f}")

    # --------------------------------------------------------------
    # long sub‑gradient refinement on the current global best
    if best_seq is not None and time.time() < deadline - 5:
        a_ref, val_ref = _subgradient_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=250_000, eta0=0.5
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Global] Long subgrad best {val_ref:.6f}")

    # --------------------------------------------------------------
    # optional up‑sampling to a larger grid (8192) if time permits
    if time.time() + 30 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_big, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # Adam on the large grid (short run)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big, n_big,
            conv_fft_len_big, corr_fft_len_big,
            deadline, max_iters=120_000
        )
        # Sub‑gradient refinement on the large grid
        a_big_ref, val_big_ref = _subgradient_optimize(
            a_big_adam, n_big,
            conv_fft_len_big, corr_fft_len_big,
            deadline, max_iters=150_000, eta0=0.5
        )
        if val_big_ref < best_val:
            best_val = val_big_ref
            best_seq = a_big_ref.copy()
            print(f"[Upsample] refined best {val_big_ref:.6f}")

    # --------------------------------------------------------------
    # fallback – guarantee a valid return value
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the range accepted by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
