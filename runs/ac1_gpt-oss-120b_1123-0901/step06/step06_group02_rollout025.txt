The evaluation function is scale‑invariant, therefore we can force  

```
Σ a_i = √(2 n)   (target_sum)
```

and minimise `max(conv(a,a))`.  
The optimiser works in four stages:

1. **Diverse parametric initialisations** – exponential, power‑law,
   Hann/Hamming windows, Gaussian, β‑distribution and Dirichlet draws
   for several lengths.  
2. **Soft‑max Adam optimisation** – a smooth surrogate of the max‐convolution
   (log‑sum‑exp) is minimised on the simplex using a soft‑max
   parametrisation (`a = target_sum·softmax(x)`).
3. **Targeted projected sub‑gradient descent** – exact gradient at the
   current worst index (`k = argmax conv`) is used and the step is
   projected onto the simplex.
4. **Guided hill‑climb** – the gradient of the *worst* convolution
   identifies a source index (large contribution) and a destination
   index (small contribution).  A small amount of mass is transferred
   from source to destination; the move is accepted if it improves the
   worst convolution (or with a tiny Metropolis probability during the
   early phase).

All phases respect the global time budget (`budget_s`).  
The best sequence seen so far is stored globally and returned as a plain
Python list.

```python
"""
Hybrid optimiser for the step‑function bound

The objective 2·n·max(conv(a,a))/sum(a)² is scale‑invariant, therefore we
fix Σ aᵢ = √(2 n) and minimise max(conv(a,a)).  The search consists of

1.   Random parametric initialisations over a wide range of lengths.
2.   Soft‑max Adam optimisation on a log‑sum‑exp surrogate of the max‑convolution.
3.   Exact sub‑gradient descent that follows the current worst convolution index.
4.   Guided hill‑climb that moves mass from indices contributing most to the
     worst peak towards indices that contribute least.

All phases share a global best solution and are limited by the supplied
time budget (default 1000 s).  If nothing is found a uniform construction
is returned as a safe fallback.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT of power‑of‑2 length."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())      # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # Numerical noise may produce tiny negative values – clip them.
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex of radius s (non‑negative, sum = s)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Normalisation to the required L1 sum √(2·n)
# ----------------------------------------------------------------------
def _normalise_to_target(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    target = np.sqrt(2.0 * n)
    if a.sum() == 0.0:
        a = np.full_like(a, target / n)
    else:
        a = a * (target / a.sum())
    return a


# ----------------------------------------------------------------------
# Random parametric families (used in Phase 1)
# ----------------------------------------------------------------------
def _random_initial_sequence(rng: np.random.Generator, n: int) -> np.ndarray:
    shape = rng.choice(["exp", "pow", "hann", "hamming", "gauss", "beta", "dirichlet"])
    i = np.arange(n, dtype=float)

    if shape == "exp":
        tau = rng.uniform(0.1, 2.5) * n
        a = np.exp(-i / tau)

    elif shape == "pow":
        p = rng.uniform(0.2, 1.7)
        a = (i + 1) ** (-p)

    elif shape == "hann":
        a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))

    elif shape == "hamming":
        a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))

    elif shape == "gauss":
        sigma = rng.uniform(0.05, 0.30) * n
        centre = (n - 1) / 2.0
        a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)

    elif shape == "beta":
        # symmetric beta (α = β) – α controls concentration.
        alpha = rng.uniform(1.0, 5.0)
        x = (i + 0.5) / n
        a = (x ** (alpha - 1)) * ((1 - x) ** (alpha - 1))

    else:                     # dirichlet (uniform on simplex)
        a = rng.dirichlet(np.ones(n))

    # optional tiny zero‑mean noise
    if rng.rand() < 0.25:
        noise = rng.randn(n) * a.max() * 0.02
        a = np.maximum(a + noise, 0.0)

    return _normalise_to_target(a)


# ----------------------------------------------------------------------
# Adam optimiser on the log‑sum‑exp surrogate of the max‑convolution
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 5e-5,
    alpha0: float = 0.025,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed‑max surrogate.
    Returns the best coefficient vector (on the simplex) and its exact
    max‑convolution value.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # ----- Soft‑max mapping -----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)               # stable softmax
        p = exp_shift / np.sum(exp_shift)           # Σ p = 1
        a = target_sum * p

        # ----- Exact max‑convolution (used for logging) -----
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam n={n}  new best max={cur_max:.6f}")

        # ----- Smoothed‑max weights (log‑sum‑exp) -----
        tau = max(init_tau_factor * cur_max * (0.99 ** it), min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- Gradient wrt a (2·a ⊛ w_rev) -----
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # ----- Back‑propagation to logits -----
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaux
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slow decay of learning rate
        alpha *= 0.9995

    final_val = _full_convolution_fft(best_a, best_a).max()
    return best_a, float(final_val)


# ----------------------------------------------------------------------
# Targeted projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------
def _targeted_subgradient(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent that always follows the current worst index."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    step = 0.2 * target_sum / n          # initial step size
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())            # index of current worst peak

        # gradient of conv_k w.r.t. a is 2·a_{k‑i}
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        cand = a - step * grad
        cand = _project_onto_simplex(cand, target_sum)

        cand_val = _full_convolution_fft(cand, cand).max()
        if cand_val < best_val - 1e-12:
            a = cand
            best_a = cand.copy()
            best_val = cand_val
            step *= 1.02
            if it % 200 == 0:
                print(f"[{time.time():.2f}] subgrad it={it}  val={best_val:.6f}")
        else:
            step *= 0.70
            if np.random.rand() < 0.07:
                jitter = np.random.rand(n) * 1e-4
                a = _project_onto_simplex(a + jitter, target_sum)

        # early exit if we already reached the target
        if best_val < 1.503:
            break
    return best_a, best_val


# ----------------------------------------------------------------------
# Guided hill‑climb (mass transfer based on worst‑peak gradient)
# ----------------------------------------------------------------------
def _guided_hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Move mass from indices that contribute most to the worst peak to
    indices that contribute least, accepting only improving moves."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    print(f"[{time.time():.2f}] guided hill‑climb start max={best_val:.6f}")

    # step size as a fraction of the source entry (adapted during the run)
    step_factor = 0.08
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        k = int(conv.argmax())                # current worst index

        # gradient contribution of each entry to this worst peak:
        # g_i = a_{k-i}   (zero if index out of range)
        idx = k - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[valid] = a[idx[valid]]

        # source: maximise (g_i * a_i) -> biggest impact on the peak
        src_score = g * a
        i_src = int(np.argmax(src_score))

        # destination: minimise g_j (ideally zero) – avoid picking the source again
        candidates = np.arange(n)
        candidates = candidates[candidates != i_src]
        if candidates.size == 0:
            break
        i_dst = int(candidates[np.argmin(g[candidates])])

        # amount to move – never exceed the source entry
        delta = min(a[i_src] * step_factor, a[i_src] * 0.5)
        if delta <= 0:
            break

        a_new = a.copy()
        a_new[i_src] -= delta
        a_new[i_dst] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()
        if new_val < best_val:
            a = a_new
            best_a = a_new.copy()
            best_val = float(new_val)
            print(f"[{time.time():.2f}] guided hill‑climb it={it}  new best={best_val:.6f}")

            # on success slowly enlarge the step, otherwise shrink
            step_factor = min(step_factor * 1.02, 0.20)
        else:
            step_factor = max(step_factor * 0.95, 0.001)

        # occasional re‑projection onto the simplex to fix accumulated round‑off
        if it % 1500 == 0:
            a = _project_onto_simplex(a, target_sum)
            best_a = _project_onto_simplex(best_a, target_sum)

        # early exit if the target is already met
        if best_val < 1.503:
            break

    return best_a, best_val

            run_deadline,
            init_tau_factor=0.12,
            min_tau_factor=5e-5,
            alpha0=0.025,
        )
        if val_local < best_score:
            best_score = val_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] after Adam refinement best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Targeted projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 80.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_deadline = deadline - 40.0       # reserve a few seconds for hill‑climb
        seq_sub, val_sub = _targeted_subgradient(best_seq.copy(),
                                                 target_sum,
                                                 sub_deadline)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑gradient best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Guided hill‑climb (final refinement)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        hill_deadline = deadline
        seq_hill, val_hill = _guided_hillclimb(best_seq.copy(),
                                               target_sum,
                                               hill_deadline)
        if val_hill < best_score:
            best_score = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] after guided hill‑climb best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be used)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = _full_convolution_fft(best_seq, best_seq).max()
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # Return a plain Python list as required by the evaluator
    return best_seq.tolist()
```
