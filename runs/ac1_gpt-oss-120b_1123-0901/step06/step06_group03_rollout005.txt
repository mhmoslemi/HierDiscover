"""
Propose a step‑function (non‑negative vector) that minimizes

        F(a) = 2·n·max(a∗a) / (∑a)²

by a multi‑scale gradient‑based search.

Main ideas
---------
1. **Two‑stage resolution** – First optimise on a moderate grid (n≈4096)
   to obtain a good shape quickly, then up‑sample to a finer grid
   (n≈8192) and continue the optimisation.  The objective is scale‑free,
   so the refinement can push the constant 2·max(p∗p) closer to its
   conjectured optimum ≈1.5.

2. **Improved Adam + sub‑gradient** – The same smooth‑max surrogate as
   the reference code is used, but the stagnation tolerances are
   increased (15 s for Adam, 30 s for the exact sub‑gradient phase) and
   the sub‑gradient loop is allowed many more iterations.  This gives
   the optimiser enough time to escape plateaus.

3. **Rich candidate pool** – Besides uniform, sine, Gaussian, double‑exponential
   shapes we also add a triangular and a cosine‑squared envelope and a
   larger set of Dirichlet draws.  The best few (≤8) are taken to the
   deep optimisation stage.

4. **Random re‑shakes** – If Adam stagnates, a small random jitter is
   added to the logits to help escape local minima.

The routine obeys the supplied time‑budget (default 1000 s) and returns
the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Result has the same length as a.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Exact objective 2·n·max(conv) / (sum(a)²).
    Assumes sum(a) > 0.
    """
    conv = _full_convolution(a, fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
    adam_stagnation: float = 15.0,
    subgrad_stagnation: float = 30.0,
    max_subgrad_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Refine a probability vector `a0` (sum = 1) by
    * Adam on a smoothed max (log‑sum‑exp) surrogate,
    * followed by an exponentiated‑gradient sub‑gradient phase on the true max.
    The function updates and returns the best sequence/value found across
    all calls.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # parameters of a softmax
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04                                      # slight reduction for stability
    lam = 0.5                                      # initial temperature
    lam_factor = 1.018

    last_improve = time.time()
    it = 0

    # ------------------- Adam on smooth‑max ------------------- #
    while time.time() < deadline:
        # Soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # Exact objective (good for logging)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={it:06d}] new best {val:.6f}")

        # Smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient of surrogate (chain‑rule through the soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # Increase temperature
        lam = min(lam * lam_factor, 1e6)

        # Random shake when stagnating
        if time.time() - last_improve > adam_stagnation:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        # safety bound – very large n could otherwise loop a long time
        if it > 2_000_000:
            break

    # -------------------- Exact sub‑gradient phase -------------------- #
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total

    sub_last_improve = time.time()
    for sub_it in range(max_subgrad_iters):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[n={n} Subgrad it={sub_it:06d}] new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Mirror‑descent step size (decays like 1/√t)
        eta = 0.6 / np.sqrt(sub_it + 1)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > subgrad_stagnation:
            # No improvement for a while – stop this phase
            print(f"[n={n} Subgrad] stagnated after {sub_it} iters")
            break

    return global_best_seq, global_best_val

def _generate_candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a diverse pool of start vectors (including a parametric sweep).
    All vectors are normalised to sum = 1.
    """
    pool = []

    # Warm‑start and simple rolls
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Baselines
    pool.append(np.full(n, 1.0 / n))                     # uniform

    # Power‑law edge‑concentrated shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)                        # i·(n‑i+1)
    for alpha in np.linspace(-2.0, -0.2, 12):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # Double‑exponential (two‑sided Laplace) families
    i = np.arange(n, dtype=np.float64)
    for lam in np.linspace(0.1, 3.0, 12):
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        pool.append(shape)

    # Gaussian bumps (various widths)
    for sigma in np.linspace(n * 0.03, n * 0.4, 9):
        shape = np.exp(-0.5 * ((i - n / 2) ** 2) / (sigma ** 2))
        shape /= shape.sum()
        pool.append(shape)

    # Single half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # Triangular envelope
    tri = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1))
    tri = tri.astype(np.float64)
    tri /= tri.sum()
    pool.append(tri)

    # Cosine‑squared envelope (smooth bell)
    cos2 = np.cos(np.pi * (np.arange(n) - (n - 1) / 2) / n) ** 2
    cos2 = np.maximum(cos2, 0.0)
    cos2 /= cos2.sum()
    pool.append(cos2)

    # Random Dirichlet draws with many concentration values
    for alpha in np.logspace(-2, 0, 20):
        shape = np.random.dirichlet(np.full(n, alpha))
        pool.append(shape)

    return pool

            if time.time() >= big_deadline:
                break
            print(f"[big optim start {cid}] launching deep optimisation (n={n_big})")
            best_seq_small, best_val = _deep_optimize(
                a0,
                n_big,
                conv_fft_big,
                corr_fft_big,
                big_deadline,
                best_seq_small,
                best_val,
                adam_stagnation=18.0,
                subgrad_stagnation=30.0,
                max_subgrad_iters=200_000,
            )

    # ------------------------------------------------------------------
    # 3. Final sanity – ensure we have a valid vector
    # ------------------------------------------------------------------
    if best_seq_small is None:
        # Fallback: uniform distribution of a reasonable size
        best_seq_small = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator's allowed range (non‑negative, ≤ 1000)
    best_seq_small = np.clip(best_seq_small, 0.0, 1000.0)

    # Convert to plain list for the outer harness
    return best_seq_small.tolist()
