def _pgd_optimize(a0, n, deadline, max_iters, best_seq, best_score, rng):
    target_sum = math.sqrt(2.0*n)
    a = a0.copy()
    a = a * target_sum / a.sum()
    Nfft = 1 << ((3*n-2).bit_length())
    step0 = 0.003 * target_sum  # maybe larger than previous 0.001
    eps_active = 1e-12
    momentum = np.zeros_like(a)
    gamma = 0.9
    for it in range(1,max_iters+1):
        if time.time() >= deadline:
            break
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[:2*n-1]
        M = float(conv.max())
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD len {n} iter {it}] new best = {best_score:.6f}")
        mask = conv >= M - eps_active
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = mask.astype(float) / float(cnt)
        # weighted reverse convolution
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[:len(w) + n - 1]
        idx_start = 2*n - 2
        rev = conv_rev[idx_start - np.arange(n)]
        g = 2.0 * rev
        # momentum
        momentum = gamma * momentum + (1 - gamma) * g
        lr = step0 / math.sqrt(it)
        a_tmp = a - lr * momentum
        a = _project_onto_simplex(a_tmp, target_sum)
        # jitter occasional
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)
    return best_seq, best_score

def _exp_gradient_optimize(...):
    target_sum = sqrt(2*n)
    eps=1e-16
    y = np.log(a0 + eps)
    m=np.zeros_like(y); v=np.zeros_like(y)
    beta1,beta2,epsadam = 0.9,0.999,1e-8
    lr0=0.12
    tau_factor=0.4
    decay_rate = max_iters / 6.0
    jitter_interval = 2000

    Nfft = 1 << ((3*n -2).bit_length())
    for it in range(1, max_iters+1):
        if time.time() >= deadline: break
        logits = y - np.max(y)
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()
        a = target_sum * p
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[:2*n-1]
        M = float(conv.max())
        if M < best_score: ...
        # compute weight w
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-8:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
        else:
            maxc = conv.max()
            ex = np.exp((conv - maxc) / tau)
            w = ex / ex.sum()
        # reverse conv
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[:len(w) + n - 1]
        idx_start = 2*n - 2
        rev = conv_rev[idx_start - np.arange(n)]
        g_a = 2.0 * rev
        avg = np.dot(p, g_a)
        g_y = target_sum * p * (g_a - avg)
        # param updates
        m = beta1 * m + (1-beta1) * g_y
        v = beta2 * v + (1-beta2) * (g_y * g_y)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)
        if it % jitter_interval == 0:
            jitter = rng.normal(scale=0.01, size=n)
            y += jitter
    return best_seq, best_score

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

The strategy is a multi‑scale, multi‑start optimisation:

1. **Initial shapes** – a diverse pool of feasible vectors is built from
   classical windows (uniform, Hann, Blackman, exponential, …) together with
   parametric families (Beta‑pdf, sine bump, random cosine windows, etc.).
   If a previous best construction ``height_sequence_1`` is available it is
   interpolated to the current length and added to the pool.

2. **Two complementary continuous optimisers** are applied to every start
   vector:
   * an *exponentiated‑gradient Adam* optimiser on a smooth‑max
     approximation of the convolution (soft‑max with an annealing temperature);
   * a *projected gradient descent* (PGD) on the exact max‑convolution,
     using the sub‑gradient derived from the active set.  PGD includes a
     momentum term and occasional Gaussian jitter to escape plateaus.

   Both methods keep the iterate on the simplex
   `{a_i≥0, Σa_i = √(2 n)}` by Euclidean projection.

3. **Multi‑scale refinement** – the optimisation starts at a modest length
   (≈ 500) and the best vector is linearly interpolated to larger sizes
   (≈ 2000, 4000 …).  Each scale receives its own optimisation budget,
   allowing the algorithm to discover increasingly subtle shapes.

4. **Stochastic local refinement** – after the global search a fast
   simulated‑annealing hill‑climber moves a small fraction of mass between
   random indices, accepting improvements always and worsening moves with a
   temperature that decays to zero.

The whole process respects the 1000 s time budget (a 5 s safety margin is
kept).  The best sequence seen is returned as a plain Python list.
"""

# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x_i≥0, Σx_i = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Create a small, diverse set of feasible start vectors."""
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0
             - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    init.append(black / black.sum() * target_sum)

    # 7) small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    # 8) triangular shape (peak in the centre)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    init.append(tri / tri.sum() * target_sum)

    # 9) sparse random spikes
    k = max(2, min(15, n // 100))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k)
    vals = vals / vals.sum() * target_sum
    sparse = np.zeros(n)
    sparse[idx] = vals
    init.append(sparse)

    return init


def _make_parametric_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Additional start vectors from simple parametric families."""
    vectors = []

    # Beta‑pdf family (3 random draws)
    for _ in range(3):
        alpha = 10 ** rng.uniform(-1, 1)   # ∈[0.1,10]
        beta = 10 ** rng.uniform(-1, 1)
        x = np.linspace(0, 1, n)
        # use a small epsilon to avoid 0**negative
        v = np.where((x > 0) & (x < 1),
                     (x ** (alpha - 1)) * ((1 - x) ** (beta - 1)),
                     0.0)
        v = np.maximum(v, 0.0)
        if v.sum() == 0:
            continue
        vectors.append(v / v.sum() * target_sum)

    # Single sine bump (half‑wave)
    x = np.arange(n)
    v = np.sin(np.pi * (x + 0.5) / n)   # positive half‑sine
    vectors.append(v / v.sum() * target_sum)

    # Random cosine windows (2 variants)
    for _ in range(2):
        width_frac = rng.uniform(0.2, 0.6)          # proportion of the length
        width = max(2, int(width_frac * n))
        start = rng.integers(0, max(1, n - width + 1))
        idx = np.arange(start, start + width)
        win = 0.5 - 0.5 * np.cos(2.0 * np.pi * (idx - start) / (width - 1))
        v = np.zeros(n)
        v[start:start + width] = win
        vectors.append(v / v.sum() * target_sum)

    # Random Gamma shape (2 draws, different concentration)
    for _ in range(2):
        shape = rng.uniform(0.3, 3.0)
        scale = 1.0
        v = rng.gamma(shape, scale, size=n)
        vectors.append(v / v.sum() * target_sum)

    return vectors

def _pgd_optimize(

    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """PGD with momentum on the exact max‑convolution."""
    target_sum = math.sqrt(2.0 * n)

    # start on the simplex
    a = a0.copy()
    a = a * target_sum / a.sum()

    # FFT size (≥ 3n‑2)
    Nfft = 1 << ((3 * n - 2).bit_length())

    step0 = 0.003 * target_sum          # base step size
    eps_active = 1e-12
    gamma = 0.9                         # momentum coefficient
    momentum = np.zeros_like(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # record best
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

        # active set → weights w
        mask = conv >= M - eps_active
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = mask.astype(float) / float(cnt)

        # weighted reverse convolution (sub‑gradient)
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[: len(w) + n - 1]
        idx_start = 2 * n - 2
        rev = conv_rev[idx_start - np.arange(n)]
        g = 2.0 * rev                     # ∇_a max(a∗a)

        # momentum update and projected step
        momentum = gamma * momentum + (1.0 - gamma) * g
        lr = step0 / math.sqrt(it)
        a_tmp = a - lr * momentum
        a = _project_onto_simplex(a_tmp, target_sum)

        # occasional jitter to avoid stagnation
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient Adam on a soft‑max approximation of max‑conv
# ----------------------------------------------------------------------
def _exp_gradient_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the log‑sum‑exp smooth max of the convolution."""
    target_sum = math.sqrt(2.0 * n)

    eps = 1e-16
    y = np.log(a0 + eps)                     # logits on the simplex

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.12                                 # base learning rate for logits
    jitter_interval = 2000

    # temperature schedule for soft‑max
    tau_factor = 0.40
    decay_rate = max_iters / 6.0

    # FFT buffer size (≥ 3n‑2)
    Nfft = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current probability vector
        logits = y - np.max(y)
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()
        a = target_sum * p                     # lies on the simplex

        # convolution and smooth max weights
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # keep global best (exact max is what we report)
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[Adam len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

        # soft‑max temperature
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
        else:
            maxc = conv.max()
            ex = np.exp((conv - maxc) / tau)
            w = ex / ex.sum()

        # weighted reverse convolution (gradient of smoothed max)
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[: len(w) + n - 1]
        idx_start = 2 * n - 2
        rev = conv_rev[idx_start - np.arange(n)]

        g_a = 2.0 * rev                         # ∂M/∂a (smooth)
        avg = np.dot(p, g_a)                    # ⟨p, g_a⟩
        g_y = target_sum * p * (g_a - avg)      # chain‑rule to logits

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # occasional jitter on logits
        if it % jitter_interval == 0:
            jitter = rng.normal(scale=0.01, size=n)
            y += jitter

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple stochastic refinement (simulated annealing)
# ----------------------------------------------------------------------
def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing on the simplex.
    Moves a random fraction of mass from one index to another.
    Accepts improvements always; worsening moves with a decaying temperature.
    """
    target_sum = a.sum()                     # already √(2 n)
    cur_seq = a.copy()
    cur_score = best_score

    # FFT buffer for linear convolution of length 2n‑1
    Nfft = 1 << ((2 * n - 1).bit_length())

    max_steps = 300_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-14:
            step += 1
            continue

        # transfer a modest random fraction of the source mass
        frac = rng.uniform(0.005, 0.07)          # 0.5 % – 7 %
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # evaluate new max‑convolution (exact)
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        # deterministic improvement
        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.

    Returns the best sequence (plain Python list) discovered within the time
    budget.  If a previous construction ``height_sequence_1`` is present it is
    used as a warm start and interpolated to larger lengths.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # reserve a tiny safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load previous construction, if any
    base = None
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Choose a cascade of lengths (coarse → fine)
    if base is not None:
        n0 = int(base.shape[0])
    else:
        n0 = 2000

    # a handful of length factors; cap at 8000 for safety
    length_factors = [0.5, 0.75, 1.0, 1.5, 2.0, 2.5, 3.0]
    lengths = sorted({max(300, min(int(n0 * f), 8000)) for f in length_factors})

    best_seq = None
    best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # ------------------------------------------------------------------
        # Build diverse starting vectors
        init_vectors = _make_initial_vectors(n, target_sum, rng)
        init_vectors.extend(_make_parametric_vectors(n, target_sum, rng))

        # Interpolate previous best to the new size (if available)
        if base is not None:
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
            interp = np.maximum(interp, 0.0)
            interp = interp / interp.sum() * target_sum
            init_vectors.append(interp)

        # ------------------------------------------------------------------
        # Iteration budget – more iterations for smaller n (they are cheap)
        max_iters = int(3_500_000 * 2000 / n)
        max_iters = max(150_000, max_iters)

        print(f"[run] length={n}, max_iters≈{max_iters:,}, target_sum={target_sum:.3f}")

        for idx, a0 in enumerate(init_vectors):
            if time.time() >= deadline:
                break
            print(f"  start {idx+1}/{len(init_vectors)} for length {n}")

            # Adam‑based exponentiated‑gradient optimisation
            best_seq, best_score = _exp_gradient_optimize(
                a0=a0.copy(),
                n=n,
                deadline=deadline,
                max_iters=max_iters,
                best_seq=best_seq if best_seq is not None else a0,
                best_score=best_score,
                rng=rng,
            )

            # Projected Gradient Descent optimisation (independent)
            best_seq, best_score = _pgd_optimize(
                a0=a0.copy(),
                n=n,
                deadline=deadline,
                max_iters=max_iters,
                best_seq=best_seq,
                best_score=best_score,
                rng=rng,
            )

    # ------------------------------------------------------------------
    # Final stochastic refinement on the global best (if time permits)
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # allocate at most half of remaining time (capped to 45 s) for refinement
        refine_deadline = time.time() + min(remaining * 0.5, 45.0)
        best_seq, best_score = _local_refine(
            a=best_seq,
            n=best_seq.shape[0],
            deadline=refine_deadline,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Fallback – uniform sequence (unlikely to be used)
    if best_seq is None:
        n = n0
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = float("inf")
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
