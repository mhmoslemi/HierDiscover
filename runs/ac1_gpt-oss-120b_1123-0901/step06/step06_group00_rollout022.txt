def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = 4096
    fft_len = _next_pow2(2*n - 1)

    # initial candidate generation
    init = generate_initial(n)

    best_a = None
    best_val = float('inf')

    # iterate over candidates within time
    for cand in init:
        if time.time() > deadline: break
        a = np.maximum(cand,0)
        a = a / max(a.sum(), 1e-12)  # normalize sum to 1 for easier.

        # gradient descent phases
        a, val = adam_max_smooth(a, n, fft_len, deadline)
        a, val = subgrad_exact(a, n, fft_len, deadline)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print("candidate improved", val)

    # main refinement: run simulated annealing from best_a
    if best_a is not None:
        best_a, best_val = simulated_annealing(best_a, n, fft_len, deadline, best_val)

    # optional upscale to 8192 if time permits
    if time.time() + 20 < deadline:
        best_a, best_val = upscale_and_refine(best_a, n, fft_len, deadline)

    # ensure format list
    best_a = np.clip(best_a, 0.0, 1000.0)
    print("final best", best_val)
    return best_a.tolist()
```

"""
Hybrid heuristic for minimizing 2·n·max(a*a)/(sum a)^2.
Combines diverse warm‑starts, Adam on a smooth‑max surrogate,
exact sub‑gradient descent, and a fast simulated‑annealing
local search that updates the convolution in O(n) per move.
If time permits the solution is interpolated to a longer grid.
"""
import time
import numpy as np

# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a heterogeneous pool of starting vectors."""
    cand = []

    # warm start and a few rotations (if any)
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet (uniform) and spiky Dirichlet (small alpha)
    cand.append(np.random.dirichlet(np.ones(n)))
    cand.append(np.random.dirichlet(0.2 * np.ones(n)))   # spiky

    # power‑law edge shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in (-2.0, -1.5, -1.0, -0.75):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # arcsine‑type shape (β(½,½) analogue)
    arcs = 1.0 / np.sqrt(np.arange(1, n + 1) * np.arange(n, 0, -1))
    arcs /= arcs.sum()
    cand.append(arcs)

    # double‑Gaussian edges
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.3, 0.7, 1.5):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # half‑sine (already used before)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # spike families (k equally‑spaced non‑zero entries)
    for k in (64, 128, 256, 512, 1024):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    return cand

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp (smooth max) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03            # modest learning rate – empirically stable
    lam = 0.3            # start temperature for smooth‑max
    lam_factor = 1.015   # slow anneal

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (sum(a)=1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:05d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal temperature
        lam = min(lam * lam_factor, 1e6)

        # occasional small random perturbation (helps escape plateaus)
        if it % 4000 == 0 and time.time() > deadline - 5:
            logits += (np.random.rand(n) - 0.5) * 1e-3

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 300_000,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it:05d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update (exponentiated gradient)
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1
    return best_a, best_val

def _fast_move(a: np.ndarray,
               conv: np.ndarray,
               i: int,
               j: int,
               delta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Perform a deterministic move: transfer `delta` from index i to j.
    Updates both the vector a and its convolution conv in O(n) time.
    """
    n = a.shape[0]
    # update convolution – first remove contribution of the removed mass
    conv[i : i + n] -= 2.0 * delta * a
    # then add contribution of the added mass
    conv[j : j + n] += 2.0 * delta * a
    # quadratic terms (the delta² correction)
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    conv[i + j] -= 2.0 * delta * delta

    # update the vector itself
    a[i] -= delta
    a[j] += delta
    return a, conv


def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         cur_best_val: float) -> tuple[np.ndarray, float]:
    """
    Fast SA that works directly on the convolution.
    The temperature drops linearly with the elapsed fraction of the budget.
    """
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)
        # linear cooling from 0.1 down to 1e‑6
        T = max(1e-6, 0.1 * (1.0 - fraction))
        # move magnitude shrinks as we cool
        move_scale = max(0.01, 0.3 * (1.0 - fraction))

        # pick source index proportionally to its mass (helps focusing on large peaks)
        if a.sum() == 0:
            break
        i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-12:
            continue
        # pick a different target uniformly
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * (np.random.rand() * move_scale)

        # apply move (working on copies)
        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # update convolution using the original a (not a_new)
        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_max = conv_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            cur_max = new_max
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:05d}] new best {best_val:.6f}")

        it += 1
        if it % 5000 == 0:
            print(f"[SA it={it:05d}] best {best_val:.6f}")

    return best_a, best_val


def _interpolate_and_refine(a_small: np.ndarray,
                            n_big: int,
                            deadline: float) -> tuple[np.ndarray, float]:
    """
    Linear interpolation to a larger grid followed by a short Adam+sub‑gradient pass.
    """
    n_small = a_small.shape[0]
    x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
    a_big = np.interp(x_big, x_small, a_small)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()

    conv_fft_len_big = _next_pow2(2 * n_big - 1)
    corr_fft_len_big = _next_pow2(3 * n_big - 2)

    # short Adam + sub‑gradient
    a_adam, _ = _adam_optimize(a_big, n_big, conv_fft_len_big,
                               corr_fft_len_big, deadline, max_iters=80_000)
    a_ref, val_ref = _subgradient_optimize(a_adam, n_big,
                                           conv_fft_len_big,
                                           corr_fft_len_big,
                                           deadline, max_iters=100_000)
    return a_ref, val_ref

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimizes
    `2·n·max(a∗a)/(sum a)^2`.  The procedure combines several

    descent and a fast simulated‑annealing local search).  If time
    permits the solution is interpolated to a larger grid and refined
    again.
    """

    deadline = time.time() + budget_s - 0.5   # a tiny safety margin

    n = 4096

    corr_fft_len = _next_pow2(3 * n - 2)

                                                   conv_fft_len,
                                                   corr_fft_len,
                                                   deadline,
                                                   max_iters=80_000)
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[cand {cid}] Sub‑grad improved best to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Heavier refinement on the global best using simulated annealing
    if best_seq is not None and time.time() < deadline - 5:

                                            conv_fft_len,
                                            deadline,
                                            best_val)
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()

"""
Hybrid heuristic for minimizing

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

The routine combines several ideas that proved useful in earlier
experiments:

* a rich pool of warm‑starts (uniform, Dirichlet, power‑law,
  arcsine‑type, double‑Gaussian edges, spike patterns);
* Adam optimisation on a smooth‑max surrogate (log‑sum‑exp) with a
  slowly increasing temperature;
* exact‑max sub‑gradient descent using exponentiated‑gradient updates;
* a fast simulated‑annealing local search that updates the convolution
  in O(n) per elementary mass‑transfer move;
* optional up‑scaling to a longer grid (8192) and a short refinement.

The function respects the supplied time budget (default 1000 s) and
always returns a non‑negative vector.  All heavy work is carried out
with FFT‑based convolutions for speed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of start vectors."""
    cand = []

    # warm start & a few rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet (uniform) and a spiky Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))
    cand.append(np.random.dirichlet(0.2 * np.ones(n)))

    # power‑law edge‑concentrated shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in (-2.0, -1.5, -1.0, -0.75):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # arcsine‑type (Beta(½,½) analogue)
    arcs = 1.0 / np.sqrt(np.arange(1, n + 1) * np.arange(n, 0, -1))
    arcs /= arcs.sum()
    cand.append(arcs)

    # two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.3, 0.7, 1.5):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # half‑sine (positive half‑wave)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # spike families (k equally‑spaced non‑zero entries)
    for k in (64, 128, 256, 512, 1024):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    return cand


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp (smooth‑max) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03                # modest learning rate
    lam = 0.3                # start temperature for the smooth‑max
    lam_factor = 1.015       # slow anneal

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:05d}] new best {val:.6f}")

        # smooth‑max (log‑sum‑exp) weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to escape plateaus
        if it % 4000 == 0 and time.time() > deadline - 5:
            logits += (np.random.rand(n) - 0.5) * 1e-3

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 300_000,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it:05d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step‑size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         cur_best_val: float) -> tuple[np.ndarray, float]:
    """Fast SA that works directly on the convolution (O(n) per move)."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)

        # linear cooling from 0.1 down to 1e‑6
        T = max(1e-6, 0.1 * (1.0 - fraction))
        # move magnitude shrinks as we cool
        move_scale = max(0.01, 0.3 * (1.0 - fraction))

        # pick source index proportionally to its mass
        if a.sum() == 0:
            break
        i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-12:
            continue
        # pick a different target uniformly
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * (np.random.rand() * move_scale)

        # apply the move on copies
        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # update convolution using the *old* a (the one before the move)
        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_max = conv_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            cur_max = new_max
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:05d}] new best {best_val:.6f}")

        it += 1
        if it % 5000 == 0:
            print(f"[SA it={it:05d}] best {best_val:.6f}")

    return best_a, best_val


def _interpolate_and_refine(a_small: np.ndarray,
                            n_big: int,
                            deadline: float) -> tuple[np.ndarray, float]:
    """Interpolate to a larger grid and run a short Adam+sub‑gradient pass."""
    n_small = a_small.shape[0]
    x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
    a_big = np.interp(x_big, x_small, a_small)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()

    conv_fft_len_big = _next_pow2(2 * n_big - 1)
    corr_fft_len_big = _next_pow2(3 * n_big - 2)

    a_adam, _ = _adam_optimize(a_big, n_big, conv_fft_len_big,
                               corr_fft_len_big, deadline,
                               max_iters=80_000)
    a_ref, val_ref = _subgradient_optimize(a_adam, n_big,
                                           conv_fft_len_big,
                                           corr_fft_len_big,
                                           deadline, max_iters=100_000)
    return a_ref, val_ref
