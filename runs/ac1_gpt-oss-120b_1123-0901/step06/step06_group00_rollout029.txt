from scipy.sparse import csr_matrix, coo_matrix

def solve_lp(f, M):
    n = len(f)
    L = 2*n - 1
    rows = []
    cols = []
    data = []
    for s in range(L):
        # indices j where 0 <= j < n and 0 <= s-j < n <=> max(0, s-(n-1)) <= j <= min(n-1, s)
        j_start = max(0, s - (n - 1))
        j_end = min(n - 1, s)
        # For each j, coefficient f[s - j]
        for j in range(j_start, j_end + 1):
            i = s - j
            rows.append(s)
            cols.append(j)
            data.append(f[i])
    A = csr_matrix((data, (rows, cols)), shape=(L, n))
    # constraints: A @ b <= M
    c = -np.ones(n)  # maximize sum b -> minimize -sum b
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(L, M), bounds=bounds, method='highs')
    if not res.success:
        return None
    b = res.x
    return b

"""
Propose a non‑negative step function (a list of floats) that minimizes

    F(a) = 2·n·max(a∗a) / (sum(a)²)

where *∗* is the ordinary linear convolution (np.convolve).  
The algorithm is a two‑stage stochastic optimiser:

1. **Broad exploration** – many short Adam‑optimisation runs on random
   Dirichlet initialisations.  The surrogate is the log‑sum‑exp of the
   convolution (a smooth approximation of the true max).  The temperature
   is increased slowly so that the surrogate becomes tighter over time.

2. **Focused refinement** – the best vector from stage 1 is polished
   with a long exact‑max sub‑gradient phase (exponentiated‑gradient /
   mirror descent) and finally a cheap random local‑search that swaps
   small amounts of mass between coordinates.

All operations use FFT‑based convolution (O(n log n)).  The routine
respects the supplied time budget (default 1000 s) and always returns a
valid non‑negative sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns an array of length len(a):
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective (sum(a) is assumed to be 1)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return 2.0 * n * max_c


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    lr: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector encountered (sum≈1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    # start with a modest temperature and increase it exponentially
    lam = 0.4
    lam_factor = 1.025

    best_a = a0.copy()
    best_val = _objective(best_a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for logging / best‑track
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report (quiet when many iterations)
            if it % 20_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max (log‑sum‑exp) weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # a tiny random shake every few thousand steps to escape plateaus
        if it % 6_000 == 0:
            logits += (np.random.rand(n) - 0.5) * 1e-3

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta: float = 0.55,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-13)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climbing: move a random fraction of mass from one index to
    another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # move up to 35 % of the source mass
        delta = a[i] * (np.random.rand() * 0.35)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 10_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point for the competition.  Performs a batch of short Adam
    runs on diverse random starts, keeps the best, then refines it with a
    long sub‑gradient phase and a final random local search.
    The function respects the supplied time budget and always returns a
    valid non‑negative list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # problem size – 4096 is a good compromise between expressiveness
    # and speed.  If enough time is left we later upscale to 8192.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 1. Broad exploration: many short Adam runs on random Dirichlet
    #    starts.  The time budget is split into equal slices.
    slice_len = 30.0                     # seconds per short run
    while time.time() + slice_len < deadline:
        a0 = np.random.dirichlet(np.ones(n))
        local_deadline = time.time() + slice_len

        a_adam, val_adam = _adam_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            local_deadline,
            max_iters=250_000,
            lr=0.07,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Explore] new global best {val_adam:.6f}")

    # --------------------------------------------------------------
    # 2. Focused refinement on the current best candidate (if we have
    #    found something).  The remaining time is spent here.
    if best_seq is None:
        # fallback – uniform distribution
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # long sub‑gradient phase
    sub_deadline = min(deadline, time.time() + 250.0)
    a_ref, val_ref = _subgradient_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        sub_deadline,
        max_iters=500_000,
        eta=0.55,
    )
    if val_ref < best_val:
        best_val = val_ref
        best_seq = a_ref.copy()
        print(f"[Refine] sub‑grad best {val_ref:.6f}")

    # cheap local random search
    if time.time() + 30.0 < deadline:
        local_deadline = deadline
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            local_deadline,
            max_moves=200_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # --------------------------------------------------------------
    # 3. Optional up‑scaling: if enough time is left we interpolate to a
    #    longer grid (8192) and repeat a short Adam + sub‑grad phase.
    if time.time() + 150.0 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best onto the finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        big_deadline = min(deadline, time.time() + 120.0)

        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_deadline,
            max_iters=200_000,
            lr=0.07,
        )
        sub_deadline_big = min(deadline, time.time() + 60.0)
        a_big_ref, val_big_ref = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            sub_deadline_big,
            max_iters=300_000,
            eta=0.55,
        )
        if val_big_ref < best_val:
            best_val = val_big_ref
            best_seq = a_big_ref.copy()
            print(f"[Big] refined best {val_big_ref:.6f}")

    # --------------------------------------------------------------
    # 4. Final safety – ensure the sequence is valid for evaluate_sequence
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
