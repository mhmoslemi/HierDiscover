"""
Propose a step‑function (non‑negative vector) that minimises

        F(a) = 2·n·max(conv(a,a)) / (sum a)²

where `conv` is the linear convolution.
The search combines many diverse initialisations, a long Adam stage on a
smoothed max, an exact sub‑gradient (exponentiated‑gradient) stage with a
decreasing step‑size, and a cheap hill‑climbing refinement.  More time is
spent on the longest length (16384) where the objective is most sensitive.
The routine respects the global time‑budget (default 1000 s) and returns a
plain Python list.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small utility returning the smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) = n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)²."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence by linear interpolation and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new


def _generate_initial_candidates(
    n: int,
    best_seq: np.ndarray | None,
    hist_seq: np.ndarray | None,
) -> list[np.ndarray]:
    """
    Produce a diverse set of start vectors for the optimisation.
    All vectors are non‑negative and sum to 1.
    """
    cand: list[np.ndarray] = []

    # warm start from global best (exact or interpolated)
    if best_seq is not None:
        if best_seq.shape[0] == n:
            cand.append(best_seq.copy())
        else:
            cand.append(_interpolate_seq(best_seq, n))

    # historic sequence (if supplied)
    if hist_seq is not None:
        if hist_seq.shape[0] > n:
            tmp = hist_seq[:n]
        else:
            tmp = np.pad(hist_seq, (0, n - hist_seq.shape[0]))
        tmp = np.maximum(tmp, 0.0)
        tot = tmp.sum()
        if tot < 1e-12:
            tmp = np.random.dirichlet(np.ones(n))
        else:
            tmp = tmp / tot
        cand.append(tmp)

    # a few pure random Dirichlet draws
    cand.append(np.random.dirichlet(np.ones(n)))
    for _ in range(4):
        cand.append(np.random.dirichlet(np.ones(n) * np.random.uniform(0.5, 2.0)))

    # edge‑concentrated power‑law shapes ((i+1)*(n‑i))**α , α<0
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-6.0, -0.2, 6):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # double‑Gaussian edges with a wider range of widths
    for gamma in (0.2, 0.5, 1.0, 2.0, 4.0):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        cand.append(shape / shape.sum())

    # inverse‑root edge shapes (i+1)**‑α + (n‑i)**‑α
    for alpha in (0.5, 0.8, 1.0, 1.5):
        i = np.arange(n, dtype=np.float64) + 1.0
        shape = np.power(i, -alpha) + np.power(n - i + 1.0, -alpha)
        cand.append(shape / shape.sum())

    # three‑peak Gaussians (0.25,0.5,0.75) with various widths
    for sigma in (0.05, 0.10, 0.20, 0.30):
        x = np.arange(n, dtype=np.float64) / (n - 1)
        peaks = sum(np.exp(-0.5 * ((x - p) / sigma) ** 2) for p in (0.25, 0.5, 0.75))
        cand.append(peaks / peaks.sum())

    # logarithmic edge bias
    eps = 1e-12
    log_shape = np.log(idx + eps) + np.log((n - idx + 1) + eps)
    cand.append(log_shape / log_shape.sum())

    # shuffle and keep a modest number (≤30)
    np.random.shuffle(cand)
    return cand[:30]

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    max_iters: int,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best (a, value) seen before the deadline.
    """
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    # Adam state
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05

    # smooth‑max temperature
    lam = 0.5
    lam_factor = 1.03

    best_a = a.copy()
    conv = _full_convolution(a, fft_len_conv)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # simplex projection via soft‑max
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # smooth‑max weights on the convolution vector
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (via cross‑correlation)
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through the soft‑max (gradient w.r.t. logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule (approaches true max)
        lam = min(lam * lam_factor, 1e7)

        cur_val = 2.0 * n * max_c / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={t:06d}] new best {cur_val:.6f}")

        # small random shake after a few seconds without improvement
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


def _subgrad_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient optimisation using exponentiated‑gradient.
    A decreasing step‑size (η) is employed for finer convergence.
    """
    a = a0 / a0.sum()
    best_a = a.copy()
    if init_val is None:
        conv = _full_convolution(a, fft_len_conv)
        best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    else:
        best_val = init_val

    last_improve = time.time()
    eta0 = 0.6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponential update with a decaying step‑size
        eta = eta0 * (0.95 ** it)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        conv = _full_convolution(a, fft_len_conv)
        cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it}] new best {cur_val:.6f}")

        # early stop if stalled
        if time.time() - last_improve > 3.0:
            break

    return best_a, best_val


def _hill_climb_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climber: move a tiny amount of mass between
    two random coordinates and keep the move if it improves the exact
    objective.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_conv) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2:
            continue

        # shift up to 5 % of the mass from i1 to i2
        eps = np.random.rand() * 0.05 * a[i1]
        if eps <= 0.0:
            continue

        new_a = a.copy()
        new_a[i1] -= eps
        new_a[i2] += eps
        if new_a[i1] < -1e-15:
            continue
        new_a /= new_a.sum()

        new_val = _objective(new_a, n, fft_len_conv)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            last_improve = time.time()
            if it % 100 == 0:
                print(f"[HillClimb n={n} it={it}] new best {new_val:.6f}")
        else:
            # occasional uphill move to escape flat plateaus
            if np.random.rand() < 0.001:
                a = new_a

        # stop early if no improvement for a while
        if time.time() - last_improve > 4.0:
            break

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the supplied
    evaluation function.  The routine works in stages:
      * generate many diverse warm‑starts,
      * long Adam optimisation on a smooth‑max surrogate,
      * exact sub‑gradient (exponentiated‑gradient) optimisation,
      * cheap hill‑climbing,
      * final refinement on the global best.
    The algorithm focuses most of the budget on the longest length
    (n=16384) where the objective is most expressive.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val: float = float("inf")

    # retrieve historic solution if present
    hist_seq = None
    if "height_sequence_1" in globals():
        tmp = globals()["height_sequence_1"]
        if isinstance(tmp, (list, tuple, np.ndarray)):
            hist_seq = np.asarray(tmp, dtype=np.float64)

    # length hierarchy – start with a cheap medium size, finish with the largest
    lengths = [2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        fft_len_c = _next_pow2(2 * n - 1)
        fft_len_x = _next_pow2(3 * n)        # generous upper bound for correlation

        # build a rich pool of starts
        candidates = _generate_initial_candidates(n, best_seq, hist_seq)

        # allocate iteration budget – longer runs for the largest size
        base_adam = int(8000 * 1024 / n) + 3000
        max_adam_iters = max(2000, base_adam)          # never too tiny
        max_subgrad_iters = 4000
        max_hc_iters = 4000

        # run the three‑phase pipeline on each start
        for cid, a_start in enumerate(candidates):
            if time.time() > deadline:
                break

            a_adam, val_adam = _adam_optimize(
                a_start, n, fft_len_c, fft_len_x, deadline, max_adam_iters
            )
            a_sub, val_sub = _subgrad_optimize(
                a_adam, n, fft_len_c, fft_len_x, deadline,
                init_val=val_adam, max_iters=max_subgrad_iters
            )
            a_hc, val_hc = _hill_climb_optimize(
                a_sub, n, fft_len_c, deadline, init_val=val_sub,
                max_iters=max_hc_iters,
            )

            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.copy()
                print(f"*** Global new best (n={n} cand={cid}) value={best_val:.6f}")

        # keep a copy for warm‑starting the next length
        hist_seq = best_seq.copy() if best_seq is not None else None

        # final long‑run refinement on the current global best if time permits
        if best_seq is not None and time.time() < deadline - 5.0:
            print("\n--- Long‑run refinement on current best ---")
            extra_adam = int(15000 * 1024 / n) + 6000
            refined_seq, refined_val = _adam_optimize(
                best_seq, n, fft_len_c, fft_len_x, deadline, max_iters=extra_adam
            )
            if refined_val < best_val:
                best_val = refined_val
                best_seq = refined_seq.copy()
                print(f"*** Refined best after length {n}: value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (unlikely)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    print(f"\n=== Finished search: best value = {best_val:.6f} (length {best_seq.shape[0]}) ===")
    return best_seq.tolist()
