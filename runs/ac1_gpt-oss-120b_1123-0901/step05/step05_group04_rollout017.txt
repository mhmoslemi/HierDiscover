def propose_candidate(seed=42, budget_s=1000.0, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    best_seq = None
    best_score = float('inf')
    target_sum_for_n = lambda n: np.sqrt(2 * n)

    # Parameter grid search: loop over n, exponent p, maybe tau.
    for n in [2000, 2500, 3000, 3500, 4000]:
        target_sum = target_sum_for_n(n)
        idx = np.arange(1, n+1, dtype=float)

        # power law exponents
        for p in np.linspace(0.45, 0.85, 41):
            shape = idx ** (-p)
            shape += rng.random(n) * 1e-7  # jitter
            shape = shape / shape.sum() * target_sum
            shape = np.clip(shape, 0, 1000)
            score = evaluate_sequence(shape.tolist())
            if score < best_score:
                best_score = score
                best_seq = shape
                print(f'[{time.time():.2f}] Initial power p={p:.3f}, n={n}, score={score:.6f}')
            # early exit if time low
            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

    # exponential family
    # For each n maybe also test tau values.
    # run similar.

    # after grid, run refinement if time left
    if time.time() < deadline:
        n = best_seq.size
        target_sum = target_sum_for_n(n)
        a = best_seq * (target_sum / best_seq.sum())
        x0 = np.log(a + 1e-12)
        seq_refined, score_refined = _optimise_adam_softmax(
            n, x0, target_sum, deadline, rng,
            init_tau_factor=0.12,
            min_tau_factor=1e-5,
            alpha=0.03,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            tau_decay=0.94,
            max_iter=None
        )
        if score_refined < best_score:
            best_score = score_refined
            best_seq = seq_refined
            print(f'Refined after gradient, score={best_score:.6f}')

    # if somehow no seq, return uniform
    if best_seq is None:
        n = 2000
        target_sum = target_sum_for_n(n)
        best_seq = np.full(n, target_sum / n)

"""
A fast multi‑family parameter sweep followed by a short Adam refinement.

The search tries several simple analytic families (power‑law, exponential,
Gaussian, triangular, polynomial and sine‑based shapes) for a few
different problem sizes.  For each candidate the sequence is scaled to
the canonical sum √(2 n) (the evaluation function is scale invariant) and
evaluated with the supplied ``evaluate_sequence``.  The best candidate
found is then fine‑tuned with a soft‑max Adam optimiser that uses a
log‑sum‑exp surrogate of the max‑convolution.  The routine respects the
global time budget (default 1000 s) and returns the best sequence as a
plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – full linear convolution using a power‑of‑2 FFT (fast for Adam)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)            # eliminate tiny negatives
    return conv


# ----------------------------------------------------------------------
# Analytic families of non‑negative shapes
# ----------------------------------------------------------------------
def _shape_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, decreasing from the left."""
    idx = np.arange(1, n + 1, dtype=float)
    return idx ** (-p)


def _shape_exponential(n: int, tau: float) -> np.ndarray:
    """a_i ∝ exp(-i/τ), decreasing exponential."""
    idx = np.arange(n, dtype=float)
    return np.exp(-idx / tau)


def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(-((i‑centre)/σ)^2), symmetric bell."""
    centre = (n - 1) / 2.0
    idx = np.arange(n, dtype=float)
    return np.exp(-((idx - centre) / sigma) ** 2)


def _shape_triangular(n: int) -> np.ndarray:
    """Linear ramp up then down, peaks in the middle."""
    centre = (n - 1) / 2.0
    idx = np.arange(n, dtype=float)
    return 1.0 - np.abs(idx - centre) / (centre + 1.0)


def _shape_polynomial(n: int, q: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n‑i))^q, a smooth bump controlled by q."""
    i = np.arange(1, n + 1, dtype=float)
    factor = i * (n - i + 1)
    return factor ** q


def _shape_sine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+1)/(n+1))^p, symmetric and zero at the ends."""
    i = np.arange(1, n + 1, dtype=float)
    vals = np.sin(np.pi * i / (n + 1))
    return vals ** p


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max (log‑it) variables
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    rng: np.random.Generator,
    *,
    init_tau_factor: float = 0.25,
    min_tau_factor: float = 5e-4,
    alpha: float = 0.07,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau_decay: float = 0.96,
    max_iter: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a smoothed‑max surrogate.
    Returns the best sequence (as a NumPy array) and its official score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_val = float("inf")
    iteration = 0
    last_improve = 0

    while time.time() < run_deadline:
        # ---- map logits to the simplex (a = S·softmax(x)) ----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)               # stabilise
        p = exp_shift / exp_shift.sum()
        a = target_sum * p

        # ---- convolution and (smooth) max ----
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()

        # keep the best encountered point
        if max_conv < best_val:
            best_val = max_conv
            best_seq = a.copy()
            last_improve = iteration
            print(f"[{time.time():.2f}]  n={n}  new local best = {max_conv:.8f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        tau = max(init_tau_factor * max_conv * (tau_decay ** iteration),
                  min_tau_factor * max_conv)
        scaled = (conv - max_conv) / tau               # ≤0
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient w.r.t. a (derivative of Σ w_k·conv_k) ----
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)       # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)               # slice needed
        grad_a = 2.0 * conv2[idx]

        # ---- gradient w.r.t. logits x (chain rule through soft‑max) ----
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # optional clipping to avoid exploding steps
        gnorm = np.linalg.norm(grad_x)
        if gnorm > 10.0:
            grad_x = grad_x * (10.0 / gnorm)

        # ---- Adam update ----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to help escape plateaux
        if iteration % 300 == 0 and rng.random() < 0.15:
            x += rng.normal(size=n) * 0.001

        # slowly decay the learning rate
        alpha *= 0.9997

        # early stop if no improvement for a long stretch
        if iteration - last_improve > 3000:
            break
        if max_iter is not None and iteration >= max_iter:
            break

    # final evaluation with the official evaluator (preserves rounding)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score
