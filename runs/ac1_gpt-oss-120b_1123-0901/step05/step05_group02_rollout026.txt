"""
Propose a non‑negative step‑function (list of floats) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

where a*a is the full linear convolution (numpy.convolve).

The search combines several ideas:

* many diverse initialisations (uniform, Dirichlet, power‑law,
  two‑sided exponentials, sparse “binary” patterns, mixtures)
* an Adam optimiser on a smooth‑max (log‑sum‑exp) surrogate of the
  convolution maximum;
* a short exponentiated‑gradient (sub‑gradient) phase on the exact
  max‑convolution;
* a final fine‑tuning pass on the current global best;
* aggressive time‑budget handling – the routine stops well before the
  supplied limit and always returns a valid sequence.

The implementation works for any length up to a few ten‑thousands,
uses only NumPy (and optional SciPy for a fast simplex projection)
and respects the 2‑CPU limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the
    smooth‑max.  Both arrays are real, ``w`` has length 2·len(a)-1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, sum x_i = 1}.  Implements the algorithm of
    Duchi et al. (2008) – O(n log n) due to a sort.
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    # ensure the vector is finite
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # Find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # numerical safety: force exact sum 1
    w /= w.sum()
    return w


def _objective(a: np.ndarray, n: int) -> float:
    """
    Exact objective 2·n·max(a*a) / (sum(a)²).  ``a`` is assumed non‑negative.
    """
    conv = _full_convolution(a, 1 << int(np.ceil(np.log2(2 * n - 1))))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    ``seed`` fixes the NumPy RNG, ``budget_s`` is the wall‑clock budget
    (seconds).  The function returns a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Lengths to explore – small to moderate so that many iterations fit
    # into the time budget.
    lengths = [1024, 2048, 4096, 8192]

    # Pre‑compute FFT sizes (next power of two) for speed.
    fft_len_conv = {}
    fft_len_corr = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_len_conv[n] = 1 << int(np.ceil(np.log2(conv_len)))
        # For correlation we need at most conv_len + n – 1 ≤ 3·n.
        fft_len_corr[n] = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

    # ------------------------------------------------------------------
    # Helper to compute the exact objective without the costly
    # convolution if we already have the max of the convolution.
    def _exact_val_from_maxc(max_c, n):
        return 2.0 * n * max_c / (1.0)      # sum == 1 after normalisation

    # ------------------------------------------------------------------
    # Main loop over lengths ------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_len = fft_len_conv[n]
        corr_len = fft_len_corr[n]

        # -------------------------- build initial candidates ---------------
        candidates = []

        # Warm‑start from a previously found solution of the same length
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.asarray(best_seq, dtype=np.float64))

        # Random Dirichlet (concentration 1) – fully dense
        candidates.append(np.random.dirichlet(np.ones(n)))

        # More spiky Dirichlet samples (concentration <1)
        for conc in (0.2, 0.5, 0.8):
            candidates.append(np.random.dirichlet(np.full(n, conc)))

        # Power‑law “edge‑concentrated” shapes
        idx = np.arange(n, dtype=np.float64) + 1.0
        prod = idx * (n - idx + 1.0)
        for alpha in np.linspace(-2.0, -0.1, 7):
            shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
            if shape.sum() > 0:
                shape = shape / shape.sum()
                candidates.append(shape)

        # Two‑sided exponential shapes
        for gamma in (0.0005, 0.001, 0.002, 0.004):
            i = np.arange(n, dtype=np.float64)
            shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
            shape /= shape.sum()
            candidates.append(shape)

        # Sparse binary patterns – roughly sqrt(2n) non‑zeros
        m_target = max(2, int(round(np.sqrt(2 * n))))
        for _ in range(6):
            idxs = np.random.choice(n, size=m_target, replace=False)
            a = np.zeros(n, dtype=np.float64)
            a[idxs] = 1.0
            a = a / a.sum()
            candidates.append(a)

        # Sparse weighted patterns (Dirichlet on the chosen support)
        for _ in range(4):
            idxs = np.random.choice(n, size=m_target, replace=False)
            weights = np.random.dirichlet(np.ones(m_target))
            a = np.zeros(n, dtype=np.float64)
            a[idxs] = weights
            a = a / a.sum()
            candidates.append(a)

        # Mixed candidates – average of two Dirichlet draws
        for _ in range(4):
            a1 = np.random.dirichlet(np.ones(n))
            a2 = np.random.dirichlet(np.ones(n))
            a = 0.5 * a1 + 0.5 * a2
            a /= a.sum()
            candidates.append(a)

        # ------------------------------------------------------------------
        # Optimise each candidate independently
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # Normalise to a probability vector (sum = 1)
            a = a0 / a0.sum()
            # Work in the log‑softmax (logits) domain – guarantees positivity
            logits = np.log(a + 1e-15)

            # ---- Adam on a smooth‑max surrogate --------------------------------
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.05                # a bit larger than the reference code
            lam = 0.25               # start temperature
            lam_factor = 1.02        # slower heating for more stable descent

            last_improve = time.time()
            # iteration budget grows for larger n, but never exceeds ~30 k
            max_iters = int(30000 * 1024 / n) + 8000

            for t in range(1, max_iters + 1):
                # ---- projection onto the simplex (soft‑max) --------------------
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                # ---- convolution and smooth‑max weights -----------------------
                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                # log‑sum‑exp style weights (smooth max)
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w = w / w_sum

                # ---- gradient of the smoothed objective -----------------------
                grad_a = 2.0 * _cross_correlation(w, a, corr_len)

                # back‑prop through the soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # ---- Adam update ---------------------------------------------
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # temperature schedule
                lam = min(lam * lam_factor, 1e6)

                # exact objective (sum==1)
                val = 2.0 * n * max_c

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Adam iter={t:05d}] new best {val:.6f}")

                # simple stagnation handling: occasional random shake
                if time.time() - last_improve > 7.0:
                    logits += (np.random.rand(n) - 0.5) * 5e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # ---- Sub‑gradient (exponentiated‑gradient) phase on exact max -----
            a = a / a.sum()
            eta = 0.55                     # step size for EG
            sub_last_improve = time.time()
            max_sub_iters = 4000

            for sub_it in range(max_sub_iters):
                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                # active set = indices at (almost) the maximum
                active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)

                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                grad_a = 2.0 * _cross_correlation(w, a, corr_len)

                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:                     # avoid collapse
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad iter={sub_it}] new best {val:.6f}")

                if time.time() - sub_last_improve > 6.0:
                    break
                if time.time() > deadline:
                    break

        # ------------------------------------------------------------------
        # Global fine‑tuning on the best sequence of this length, if any -----
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] final fine‑tuning best={best_val:.6f}")
            a = best_seq.copy()
            logits = np.log(a + 1e-15)
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.035
            lam = 0.4
            lam_factor = 1.008
            last_improve = time.time()
            fine_iters = int(50000 * 1024 / n) + 8000

            for t in range(1, fine_iters + 1):
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

                grad_a = 2.0 * _cross_correlation(w, a, corr_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                lam = min(lam * lam_factor, 1e6)

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} fine‑tune iter={t:05d}] new best {val:.6f}")

                # occasional tiny shake to escape very flat plateaus
                if time.time() - last_improve > 9.0:
                    logits += (np.random.rand(n) - 0.5) * 4e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

    # ------------------------------------------------------------------
    # Fallback – uniform if nothing found (should never happen)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return plain Python list as required by the external evaluator
    return best_seq.tolist()
