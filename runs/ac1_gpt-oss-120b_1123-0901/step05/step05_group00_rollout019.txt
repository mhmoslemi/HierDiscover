def upsample(seq, n):
    old = np.asarray(seq)
    x_old = np.linspace(0,1,len(old))
    x_new = np.linspace(0,1,n)
    up = np.interp(x_new, x_old, old)
    up = np.maximum(up, 0.0)
    s= up.sum()
    if s>0:
        up = up/s
    else:
        up = np.random.dirichlet(np.ones(n))
    return up
```

"""
Propose a step‑function (non‑negative vector) that minimises
    F(a)=2·n·max(a*a)/(sum(a)²)

Algorithm
---------
*   Work successively on increasing lengths (powers of two).  The best
    solution for a smaller length is linearly up‑sampled and used as a warm
    start for the next length.
*   For each length a diverse pool of seeds is generated:
    – uniform, – previous best, – up‑sampled best,
    – Beta‑distribution shapes, – edge‑concentrated power laws,
    – two‑Gaussian edge shapes, – sine‑squared bump,
    – a few random Dirichlet draws and small perturbations of the current
      best.
*   Each seed is optimised in two phases:
    1. Adam on a smooth‑max (log‑sum‑exp) surrogate of the convolution
       maximum.  The surrogate temperature λ is slowly increased.
    2. A pure sub‑gradient method on the true max (active‑set weight).
    Both phases run many more iterations than the reference implementation.
*   After the length loop a fast local hill‑climbing phase (random mass
    transfers) tries to squeeze a little more out of the current best.
*   The routine respects the supplied wall‑clock budget (default 1000 s) and
    returns the best vector (as a plain list) discovered before the deadline.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper functions – keep them at top level (no closures)

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length fft_len (power of two)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length n (where n = len(a)).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective_value(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Compute the exact objective 2·n·max(conv) given a (already normalised to sum=1).
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return 2.0 * n * max_c


def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density discretised on n points (mid‑point rule)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n  # points in (0,1)
    # Avoid overflow for very small alpha – clip exponentiated terms
    shape = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    shape[~np.isfinite(shape)] = 0.0
    if shape.sum() == 0.0:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= shape.sum()
    return shape


def _powerlaw_edge(n: int, alpha: float) -> np.ndarray:
    """Edge‑concentrated shape a_i ∝ (i·(n‑i‑1))^α  (α negative)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
    if shape.sum() == 0.0:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= shape.sum()
    return shape


def _two_gaussian_edge(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _sine_bump(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n) ** 2
    shape /= shape.sum()
    return shape


def _upsample(seq: List[float], n: int) -> np.ndarray:
    """Linear interpolation of a shorter sequence up to length n."""
    old = np.asarray(seq, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old.shape[0])
    x_new = np.linspace(0.0, 1.0, n)
    up = np.interp(x_new, x_old, old)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.random.dirichlet(np.ones(n))
    return up


def _perturb_logit(logits: np.ndarray, scale: float = 1e-3) -> np.ndarray:
    """Add a tiny random jitter to logits (used when stagnating)."""
    return logits + (np.random.rand(logits.size) - 0.5) * scale


def _random_dirichlet(n: int) -> np.ndarray:
    return np.random.dirichlet(np.ones(n))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a*a)/(sum(a)²).

    The routine respects a wall‑clock budget and returns the best sequence
    discovered (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin so we can always return something sensible
    deadline = start_time + budget_s - 0.5

    best_seq = None                # numpy array (normalised)
    best_val = float("inf")        # objective value

    # Lengths processed from small to large (powers of two)
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384, 32768, 65536]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")

        # FFT sizes – chosen as the next power of two for efficiency
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)   # (2n‑1)+(n‑1)

        # --------------------------------------------------------------
        # Build a diverse pool of initial candidates
        candidates = []

        # 1) uniform
        candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # 2) warm‑start from an exact‑size previous solution
        if best_seq is not None and best_seq.shape[0] == n:
            candidates.append(best_seq.copy())

        # 3) up‑sample a shorter best (linear interpolation)
        if best_seq is not None and best_seq.shape[0] < n:
            candidates.append(_upsample(best_seq.tolist(), n))

        # 4) historic global variable, if it exists
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            h = np.asarray(hseq, dtype=np.float64)
            if h.shape[0] > n:
                h = h[:n]
            elif h.shape[0] < n:
                h = np.pad(h, (0, n - h.shape[0]))
            h = np.maximum(h, 0.0)
            s = h.sum()
            if s > 0:
                h /= s
            else:
                h = np.random.dirichlet(np.ones(n))
            candidates.append(h)

        # 5) a few random Dirichlet draws
        for _ in range(2):
            candidates.append(_random_dirichlet(n))

        # 6) symmetric Beta families (α in a modest range)
        for alpha in [0.7, 1.0, 1.5, 2.0, 3.0, 5.0]:
            candidates.append(_beta_shape(n, alpha))

        # 7) power‑law edge‑concentrated families (α negative)
        for a in [-3.0, -2.0, -1.5, -1.0]:
            candidates.append(_powerlaw_edge(n, a))

        # 8) two‑Gaussian edge shapes
        for gamma in [0.2, 0.5, 1.0, 2.0]:
            candidates.append(_two_gaussian_edge(n, gamma))

        # 9) smooth sine‑squared bump
        candidates.append(_sine_bump(n))

        # 10) tiny perturbation of the current global best (if we have one)
        if best_seq is not None:
            eps = 1e-3
            logit = np.log(np.maximum(best_seq, 1e-16))
            perturbed = np.exp(_perturb_logit(logit, eps))
            perturbed /= perturbed.sum()
            candidates.append(perturbed)

        # --------------------------------------------------------------
        # Optimise every candidate
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # initialise from a0 (guaranteed non‑negative)
            a = np.maximum(a0, 1e-16).astype(np.float64)
            a /= a.sum()
            logits = np.log(a + 1e-32)   # logits for the soft‑max parametrisation

            # ---- Adam on the smooth‑max surrogate --------------------
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2 = 0.9, 0.999
            eps_adam = 1e-8
            lr = 0.08                     # a bit larger than the reference
            lam = 0.5                     # temperature of the log‑sum‑exp
            lam_factor = 1.015
            last_improve = time.time()

            # iteration budget – larger n receives fewer iterations, but we
            # give a comfortable margin for the hardest length (16384‑65536)
            max_iters = int(30000 * 1024 / n) + 4000   # ~8000 for 16384

            for it in range(1, max_iters + 1):
                # soft‑max → simplex
                mx = logits.max()
                exp_log = np.exp(logits - mx)
                a = exp_log / exp_log.sum()

                # convolution and smooth‑max weights
                conv = _full_convolution(a, conv_fft_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # gradient of smooth‑max w.r.t. a
                grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

                # back‑propagation through the soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update on logits
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                # anneal temperature
                lam = min(lam * lam_factor, 1e7)

                # exact objective (a is normalised → denominator = 1)
                val = 2.0 * n * max_c

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Adam it={it:05d}] new best {val:.6f}")

                # stagnation handling – small random shake of logits
                if time.time() - last_improve > 5.0:
                    logits = _perturb_logit(logits, 1e-3)
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # ---- Sub‑gradient refinement on the exact max ------------
            # Ensure a is properly normalised before the sub‑gradient phase
            a = a / a.sum()
            eta = 0.6                      # initial step size
            sub_last = time.time()
            for sub_it in range(8000):    # more steps than reference
                conv = _full_convolution(a, conv_fft_len)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
                a = a * np.exp(-eta * grad_a)
                # keep numbers well‑behaved
                a = np.maximum(a, 1e-30)
                s = a.sum()
                if s < 1e-12:                     # numerical safety net
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a /= s

                # evaluate exact objective
                conv = _full_convolution(a, conv_fft_len)
                max_c = conv.max()
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad it={sub_it:04d}] new best {val:.6f}")

                # simple annealing of eta
                eta *= 0.9999

                # stop early if we have been idle for a few seconds
                if time.time() - sub_last > 5.0:
                    break
                if time.time() > deadline:
                    break

    # ------------------------------------------------------------------
    # Final local hill‑climbing (tiny random transfers) – uses any remaining time
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        conv_fft_len = _next_pow2(2 * n - 1)
        local_start = time.time()
        while time.time() < deadline - 0.1:
            # pick two distinct indices
            i, j = np.random.choice(n, 2, replace=False)
            # propose moving a small fraction of mass from i to j
            delta = 0.02 * a[i] * np.random.rand()
            if delta <= 0.0 or a[i] - delta < 0.0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # normalisation is unnecessary – total mass unchanged
            # evaluate objective
            val = _objective_value(a_new, n, conv_fft_len)
            if val < best_val:
                best_val = val
                best_seq = a_new.copy()
                a = a_new
                # occasional progress print
                if int(time.time() - local_start) % 5 == 0:
                    print(f"[local search] new best {val:.6f}")
        print(f"Local search finished – best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded (unlikely) return a uniform vector
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Convert to plain Python list as required by the external evaluator
    return best_seq.tolist()
