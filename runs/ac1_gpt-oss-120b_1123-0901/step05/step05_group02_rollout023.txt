    A_ub = scipy.sparse.lil_matrix((rows, n+1))
    for i in range(n):
        # rows i..i+n-1 inclusive: total n rows.
        start = i
        end = i + n
        # slice assignment for column i
        A_ub[start:end, i] = a[i]  # assign constant a[i] to those rows.
    # column for M
    A_ub[:, n] = -1.0

def adam_optimize(a0, n, conv_len, deadline):
    a = a0.copy()
    # ensure sum 1
    a = a / a.sum()
    logits = np.log(a + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1=0.9; beta2=0.999; eps=1e-8
    lr=0.05
    lam=0.1
    lam_factor=1.02
    best_val = float('inf')
    best_a = a.copy()
    max_iters = 20000  # param
    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # softmax
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()
        conv = full_convolution(a, conv_len)  # returns array length 2n-1
        max_c = conv.max()
        # smooth max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient: 2 * cross-correlation of w with a
        grad = 2.0 * cross_correlation(w, a, conv_len)  # must compute cross correlation length maybe similar
        # backprop through softmax
        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)
        # adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # temp increase
        lam = min(lam * lam_factor, 1e6)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # optionally random perturb if stuck
    return best_a, best_val

    rows = 2*n-1
    col_ind = np.repeat(np.arange(n), n)
    row_ind = np.concatenate([np.arange(i, i+n) for i in range(n)])
    data = a[col_ind]
    A = scipy.sparse.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))
    # Add column for M
    M_col = scipy.sparse.csr_matrix((-np.ones((rows, 1)), (np.arange(rows), np.zeros(rows))), shape=(rows, 1))

    # Use hstack
    A_ub = scipy.sparse.hstack([A, -np.ones((rows, 1))], format='csr')
    c = np.concatenate([np.zeros(n), [1.0]])  # minimize M
    # equality sum(b)=1
    A_eq = scipy.sparse.csr_matrix((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

                                 bounds=bounds,
                                 method='highs',
                                 options={'presolve': True})
    if not res.success:
        return None, None, False
    b = res.x[:n]
    M = res.x[-1]
    return b, M, True

"""
Hybrid optimizer for the step‑function problem.

The evaluation objective is   F(a)=2·n·max(conv(a,a))/sum(a)² .
For a fixed ℓ₁‑norm the denominator is constant, so we minimise
the peak of the linear convolution a∗a.

The algorithm works in three stages:

1. **Diverse initialisation** – a handful of Dirichlet draws,
   power‑law, exponential and the current best solution.

2. **Smooth‑max Adam optimisation** – the variables are the logits of a
   soft‑max (guaranteeing non‑negativity and the simplex).  A smooth
   max (log‑sum‑exp) surrogate of the convolution peak is minimised.
   The temperature is increased slowly so the surrogate approaches the
   true max.  Adam provides adaptive step‑sizes.

3. **Linear‑program refinement** – given a candidate a, we solve
   the linear programme

        minimise   M
        subject to  (a∗b)[k] ≤ M   (k=0…2n‑2)
                    Σ b = 1
                    b ≥ 0

   which computes the best response b to a.  Because the problem is
   linear the solution is exact; we replace a by b (or a convex
   combination) and repeat a few times.  This step dramatically
   flattens the convolution and typically lowers the objective beyond
   pure gradient descent.

All work respects the user‑supplied time budget (default 1000 s) and
returns the best sequence discovered as a plain Python list.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a using an FFT of length ``fft_len``.
    The returned array has length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the
    smooth‑max surrogate.  The result has length len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_len: int,
                   corr_len: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate of max(conv(a,a)).

    Returns the best vector found (still in the probability simplex) and
    the corresponding exact objective value 2·n·max(conv).
    """
    # initialise logits for a soft‑max parametrisation
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.2                                 # start temperature
    lam_factor = 1.012

    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    # iteration budget grows for larger n
    max_iters = int(20000 * 1024 / n) + 3000

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ----- soft‑max (projects onto the simplex) -----
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ----- convolution and smooth‑max weights -----
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth‑max surrogate -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        # ----- back‑prop through the soft‑max -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, 1e6)

        # ----- exact objective (sum(a)=1) -----
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # ----- occasional random shake if stuck -----
        if time.time() - last_improve > 8.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val

def _build_convolution_matrix_pattern(n: int):
    """
    Pre‑computes the sparsity pattern (row/col indices) of the
    convolution matrix that maps b → a∗b for a fixed length n.
    The pattern does not depend on the actual values of ``a``.
    Returns (row_ind, col_ind) suitable for constructing a CSR matrix.
    """
    # each column i appears in rows i … i+n‑1
    col_ind = np.repeat(np.arange(n, dtype=np.int32), n)
    row_blocks = [np.arange(i, i + n, dtype=np.int32) for i in range(n)]
    row_ind = np.concatenate(row_blocks)
    return row_ind, col_ind


def _solve_lp_for_b(a: np.ndarray,
                    n: int,
                    row_ind: np.ndarray,
                    col_ind: np.ndarray) -> tuple[np.ndarray, float, bool]:
    """
    Linear programme:
        minimise   M
        subject to  (a∗b)[k] ≤ M   for all k
                    Σ b = 1
                    b ≥ 0

    Returns (b, M, success).  The matrix A_ub is built on‑the‑fly
    using the pre‑computed sparsity pattern.
    """
    rows = 2 * n - 1

    # data for the convolution part of the constraints
    data = a[col_ind]                     # each entry uses the coefficient a_i
    A_conv = scipy.sparse.csr_matrix((data, (row_ind, col_ind)),
                                      shape=(rows, n))

    # append the column for the variable M (coefficient -1)
    M_column = scipy.sparse.csr_matrix((-np.ones(rows), (np.arange(rows), np.zeros(rows))),
                                       shape=(rows, 1))
    A_ub = scipy.sparse.hstack([A_conv, M_column], format='csr')

    # objective: minimise M
    c = np.concatenate([np.zeros(n), [1.0]])

    # equality Σ b = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # RHS for inequality constraints
    b_ub = np.zeros(rows)

    bounds = [(0.0, None)] * n + [(0.0, None)]  # b_j ≥ 0, M ≥ 0

    # solve via HiGHS (the fastest open‑source LP solver in SciPy)
    res = scipy.optimize.linprog(c,
                                 A_ub=A_ub,
                                 b_ub=b_ub,
                                 A_eq=A_eq,
                                 b_eq=b_eq,
                                 bounds=bounds,
                                 method='highs',
                                 options={'presolve': True, 'disp': False})

    if not res.success:
        return None, None, False

    b = res.x[:n]
    M = res.x[-1]
    return b, M, True


def _lp_refine(a: np.ndarray,
               n: int,
               deadline: float,
               row_ind: np.ndarray,
               col_ind: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Repeatedly replaces ``a`` by the LP‑optimal response to flatten the
    convolution.  Typically only a few iterations are needed.
    Returns the refined vector and its exact objective value.
    """
    best_a = a.copy()
    best_val = 2.0 * n * np.convolve(best_a, best_a).max()
    it = 0
    while time.time() < deadline and it < 5:
        b, M, ok = _solve_lp_for_b(best_a, n, row_ind, col_ind)
        if not ok:
            break
        # evaluate the exact objective of the candidate
        val = 2.0 * n * np.convolve(b, b).max()
        if val < best_val:
            best_val = val
            best_a = b
        else:
            # a small convex combination often helps escape tiny plateaux
            best_a = 0.5 * best_a + 0.5 * b
        it += 1
    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The function respects the time budget and returns the best sequence
    found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Lengths to explore – larger n gives a finer granularity but is
    # more expensive.  We start modestly and move upward if time permits.
    lengths = [256, 512, 1024, 2048, 4096]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n--- Optimising length n={n} ---")
        # FFT lengths for the two convolutions we need
        conv_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

        # Pre‑compute the sparsity pattern for the LP matrix (used later)
        row_ind, col_ind = _build_convolution_matrix_pattern(n)

        # --------------------------------------------------------------
        # Build a modest set of diverse seeds
        candidates = []

        # 1) Uniform (the classic baseline)
        candidates.append(np.full(n, 1.0 / n))

        # 2) Dirichlet draws with several concentrations
        for conc in [0.1, 0.3, 0.7, 1.0, 2.0]:
            candidates.append(np.random.dirichlet(np.full(n, conc)))

        # 3) Power‑law (edge‑concentrated) shapes
        idx = np.arange(1, n + 1, dtype=np.float64)
        prod = idx * (n - idx + 1)
        for alpha in [-2.0, -1.5, -1.0, -0.5]:
            shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
            if shape.sum() > 0:
                shape /= shape.sum()
                candidates.append(shape)

        # 4) Two‑sided exponential decay
        for gamma in [0.0005, 0.001, 0.002, 0.005]:
            i = np.arange(n, dtype=np.float64)
            shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
            shape /= shape.sum()
            candidates.append(shape)

        # 5) A simple tent (linear rise then fall)
        left = np.linspace(0, 1, n // 2, endpoint=False)
        right = left[::-1]
        tent = np.concatenate([left, right])
        tent = np.maximum(tent, 0)
        tent /= tent.sum()
        candidates.append(tent)

        # 6) Warm‑start from the current global best (if length matches)
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.asarray(best_seq, dtype=np.float64))

        # --------------------------------------------------------------
        # Optimise each seed
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            a0 = np.maximum(a0, 0.0)
            if a0.sum() < 1e-12:
                a0 = np.random.dirichlet(np.ones(n))
            a0 = a0 / a0.sum()

            # ---- Adam on smooth‑max surrogate ----
            a_adam, val_adam = _adam_optimize(a0, n, conv_len, corr_len, deadline)

            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n} cand={cand_id} Adam] new best {val_adam:.6f}")

            # ---- Linear‑program refinement (few iterations) ----
            if time.time() < deadline:
                a_ref, val_ref = _lp_refine(a_adam, n, deadline, row_ind, col_ind)
                if val_ref < best_val:
                    best_val = val_ref
                    best_seq = a_ref.copy()
                    print(f"[n={n} cand={cand_id} LP   ] new best {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded (should not happen)
    if best_seq is None:
        fallback_n = 256
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
