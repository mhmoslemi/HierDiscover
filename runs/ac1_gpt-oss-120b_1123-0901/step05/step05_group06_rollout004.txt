"""step_function_optimizer.py
A multi‑stage optimizer for the non‑negative step‑function problem.

The optimizer works in three phases:

1. **Diverse initialisation** – builds several start vectors for each
   chosen length (uniform, Dirichlet, exponential, windows, and an
   interpolated copy of the previous best construction).

2. **Soft‑max Adam on the log‑it simplex** – a mirror‑descent Adam optimiser
   minimises a smooth approximation of the peak of the autocorrelation
   `a∗a`.  The gradient of the soft‑maxed max‑convolution is obtained by a
   weighted reverse convolution (FFT‑based O(n log n)).  The temperature
   is annealed, jitter is added regularly and the learning‑rate schedule
   is aggressive to escape plateaus.

3. **Active‑set linear‑programming refinement** – after the Adam phase,
   we solve a tiny linear programme that maximises the total mass of a
   candidate `g` under the *active* convolution constraints
   `a∗g ≤ M` (where `M` is the current peak).  The LP is solved with
   `scipy.optimize.linprog` using only the rows where the convolution is
   near‑tight, which makes the problem tiny (≤ 30 constraints).  The
   solution is rescaled to the required ℓ₁‑norm and mixed with the current
   vector by a small convex‑combination step.

Finally a cheap stochastic hill‑climb moves a tiny fraction of mass
between random indices, accepting deteriorations with a simulated‑annealing
schedule.  The whole search respects the 1000 s budget and returns the
best sequence found as a plain Python list.

The objective equals the maximal value of `a∗a` because the evaluation
function normalises the ℓ₁‑sum to √(2 n).  Minimising this peak improves
the reported bound.
"""

import time
import math
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Helper functions – all are top‑level (no closures)

def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values.
    Returns (weights, smoothed max).
    """
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k * a_{k-i} (out‑of‑range indices ignored)
    using FFTs.  This is the gradient of the (soft‑max) max‑convolution.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    # reverse w before FFT
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # keep the linear part of the convolution (length = 3 n‑2)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Return max_k (a∗a)[k] using an FFT of length Nfft."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _conv_self(a: np.ndarray, Nfft: int) -> np.ndarray:
    """Full linear self‑convolution a∗a (length 2 n‑1)."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return conv


def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Create a small diverse set of normalised start vectors."""
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0 - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    init.append(black / black.sum() * target_sum)

    # 7) small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    # 8) triangular shape (peak in the centre)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    init.append(tri / tri.sum() * target_sum)

    return init


def _exp_gradient_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑its (exponentiated‑gradient) using a
    soft‑max approximation of the max‑convolution.
    Returns the best vector seen in this run together with its exact
    objective value (i.e. max(a∗a) after normalisation).
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise logits from the start vector (add tiny epsilon to avoid log(0))
    eps = 1e-16
    y = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.08                      # base learning rate for logits
    jitter_interval = 2500         # more frequent jitter

    # temperature (soft‑max) schedule
    tau_factor = 0.50               # start temperature as a fraction of M
    decay_rate = max_iters / 6.0    # exponential decay factor

    # FFT size – must be large enough for linear conv of length 3 n‑2
    Nfft = 1 << ((3 * n - 2).bit_length())

    no_improve = 0
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- current vector (softmax of logits) -----
        logits = y - np.max(y)               # shift for numerical stability
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()    # simplex probabilities
        a = target_sum * p                   # Σa = target_sum, a_i≥0

        # ----- exact convolution -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- soft‑max weights for the max -----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ----- weighted reverse convolution (gradient of smoothed max) -----
        rev = _weighted_rev(a, w, Nfft)

        # ----- gradient wrt a (2·rev) -----
        g_a = 2.0 * rev

        # ----- chain rule to logits (mirror‑descent on the simplex) -----
        avg = np.dot(p, g_a)                 # ⟨p, g_a⟩
        g_y = target_sum * p * (g_a - avg)   # ∂f/∂y_i

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # occasional jitter to escape plateaux
        if it % jitter_interval == 0:
            jitter = rng.normal(scale=0.01, size=n)   # additive jitter on logits
            y += jitter

        # ----- keep best -----
        if M < best_score - 1e-12:
            best_score = M
            best_seq = a.copy()
            no_improve = 0
            print(f"[len {n:5d} iter {it:8d}] new best = {best_score:.6f}")
        else:
            no_improve += 1
            # occasional restart if stuck for a long time
            if no_improve > 50000 and it % 1000 == 0:
                # random re‑initialisation of logits around current a
                y = np.log(a + eps) + rng.normal(scale=0.02, size=n)

    return best_seq, best_score


def _build_active_matrix(f: np.ndarray, active_rows: np.ndarray) -> np.ndarray:
    """
    Build a dense matrix A of shape (len(active_rows), len(f)) where
    A[k, j] = f[k-j] if the index is valid, otherwise 0.
    This matrix encodes the active convolution constraints f * b ≤ M.
    """
    n = f.shape[0]
    k_cnt = active_rows.shape[0]
    A = np.zeros((k_cnt, n), dtype=float)
    for idx, k in enumerate(active_rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k - i_range
        A[idx, j_range] = f[i_range]
    return A


def _lp_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    target_sum: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Perform a few active‑set LP refinement steps.
    Returns an improved sequence (or the original if no improvement) and
    its max‑convolution value.
    """
    # FFT size for convolutions (exact not required for LP, but needed for M)
    Nfft = 1 << ((2 * n - 1).bit_length())

    cur_seq = a.copy()
    cur_score = _max_convolution(cur_seq, Nfft)

    # LP parameters
    max_lp_steps = 5
    alpha = 0.2                     # mixing weight for the convex combination

    for step in range(max_lp_steps):
        if time.time() >= deadline:
            break

        # compute full self‑convolution and identify active constraints
        conv = _conv_self(cur_seq, Nfft)
        M = float(conv.max())
        eps_active = max(1e-6, 0.001 * M)          # keep constraints within 0.1 %
        active = np.where(conv >= M - eps_active)[0]

        # limit the number of active rows to keep the LP tiny
        if active.size > 30:
            # take the 30 largest values
            largest_idx = np.argsort(-conv[active])[:30]
            active = active[largest_idx]

        # Build active‑set matrix
        A_active = _build_active_matrix(cur_seq, active)

        # Linear programme: maximise sum(b)  ⇒ minimise -sum(b)
        c = -np.ones(n, dtype=float)
        b_ub = np.full(A_active.shape[0], M, dtype=float)
        bounds = [(0.0, None) for _ in range(n)]

        # Solve with HiGHS (fast, bundled with SciPy)
        res = scipy.optimize.linprog(
            c,
            A_ub=A_active,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
        )
        if not res.success:
            print("[lp] linprog failed, aborting refinement")
            break

        g_raw = res.x
        sum_g = g_raw.sum()
        if sum_g <= 1e-12:
            break

        # Rescale to satisfy the ℓ₁‑norm constraint
        g = g_raw * (target_sum / sum_g)

        # Convex combination step
        new_seq = (1.0 - alpha) * cur_seq + alpha * g
        new_score = _max_convolution(new_seq, Nfft)

        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[lp step {step}] improvement → {cur_score:.6f}")
            # keep the step size for further attempts
        else:
            # if no progress, shrink the mixing weight
            alpha *= 0.5
            if alpha < 1e-3:
                break

    return cur_seq, cur_score


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on the simplex:
    repeatedly move a small fraction of mass from a random index i to j.
    Accept improvements always; accept worsening moves with a decaying
    simulated‑annealing probability.
    """
    target_sum = a.sum()                     # already √(2 n)
    cur_seq = a.copy()
    cur_score = best_score

    # FFT size for fast max‑convolution
    Nfft = 1 << ((2 * n - 1).bit_length())

    # annealing schedule
    max_steps = 200_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        # transfer a random fraction (1 % – 5 %) of the source mass
        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # compute new max‑convolution
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        # deterministic improvement
        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score

    # Load any pre‑discovered construction (if present)
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        base = None

    # ------------------------------------------------------------------
    # Choose lengths to explore (original size plus enlargements, up to 8000)
    if base is not None:
        n0 = int(base.shape[0])
    else:
        n0 = 2100               # historic fallback

    # Wider spread of lengths, favouring modest enlargements
    length_factors = [1.0, 1.33, 1.66, 2.0, 2.5]
    lengths = sorted({max(300, min(int(n0 * f), 8000)) for f in length_factors})

    # ------------------------------------------------------------------
    # Global best tracking
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Main optimisation over the selected lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # generate diverse starts; also interpolate the old best if sizes differ
        init_vectors = _make_initial_vectors(n, target_sum, rng)

        if base is not None and base.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
            interp = np.maximum(interp, 0.0)
            interp = interp / interp.sum() * target_sum
            init_vectors.append(interp)

        # iteration budget – more steps for smaller n, but never less than 200 k
        max_iters = int(2_500_000 * 2100 / n)
        max_iters = max(200_000, max_iters)

        print(f"[run] length={n}, max_iters≈{max_iters}, target_sum={target_sum:.3f}")

        for idx, a0 in enumerate(init_vectors):
            if time.time() >= deadline:
                break
            print(f"  start {idx+1}/{len(init_vectors)} for length {n}")

            a0 = np.maximum(a0, 0.0)
            a0 = a0 / a0.sum() * target_sum

            # Adam‑based continuous optimisation
            best_seq, best_score = _exp_gradient_optimize(
                a0=a0.copy(),
                n=n,
                deadline=deadline,
                max_iters=max_iters,
                best_seq=best_seq if best_seq is not None else a0,
                best_score=best_score,
                rng=rng,
            )

        # ------------------------------------------------------------------
        # Active‑set LP refinement on the current global best for this length
        if best_seq is not None and best_seq.shape[0] == n and time.time() < deadline:
            print(f"[lp] start active‑set refinement for length {n}")
            refined_seq, refined_score = _lp_refine(
                a=best_seq.copy(),
                n=n,
                deadline=deadline,
                target_sum=target_sum,
                rng=rng,
            )
            if refined_score < best_score:
                best_score = refined_score
                best_seq = refined_seq.copy()
                print(f"[global] improved by LP → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Local stochastic refinement on the global best (if time permits)
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        refine_deadline = time.time() + min(remaining * 0.5, 20.0)
        best_seq, best_score = _local_refine(
            a=best_seq,
            n=best_seq.shape[0],
            deadline=refine_deadline,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Fallback in the unlikely case nothing was found
    if best_seq is None:
        n = n0
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = float('inf')
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
