    # logistic param, symmetry enforcement
    logits = np.log(a0 + 1e-12)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1,beta2 = 0.9,0.999

    lam = 1.0 # start small? maybe 0.5

    it=0
    while time.time() < deadline:

        logits = (logits + logits[::-1]) / 2

        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        # optionally enforce minor symmetry again: a = (a + a[::-1]) /2
        a = (a + a[::-1]) / 2
        a = a / a.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)  # sum close to 1
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={it}] new best {val:.6f}")
        # smooth max
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        avg = np.dot(a, grad_a)

        it1 = it + 1

"""
propose_candidate
-----------------
Search for a non‑negative step function (probability vector) that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)²)

The implementation follows three ideas that improve on the previous
submission:

1. **Symmetry enforcement** – the objective is invariant under reversal,
   therefore we keep all iterates symmetric (logits and the vector itself)
   which reduces the search space and yields smoother gradients.

2. **Improved Adam schedule** – a slower temperature increase for the
   smooth‑max surrogate, a smaller learning‑rate and periodic “shake”
   when progress stalls.  The Adam state is kept for the whole run.

3. **Multi‑scale refinement** – we start with n = 4096 (the current best
   size), optimise it, then, if time permits, interpolate the best
   distribution to larger sizes (8192 and 16384) and continue the optimisation.
   Larger n gives the algorithm more freedom to lower the maximal
   auto‑convolution.

Finally a cheap random‑perturbation local search is applied to the best
vector found.  All constraints required by ``evaluate_sequence`` are
satisfied (non‑negative, sum>0, values≤1000).  The routine respects the
provided time budget and returns the best sequence as a plain Python list.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact value of the evaluation function (sum(a) assumed >0)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a smooth‑max surrogate followed by an
    exponentiated‑gradient sub‑gradient phase.  All iterates are kept
    symmetric.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # initialise logits for soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    lr = 0.03
    lam = 0.5            # start temperature for smooth‑max
    lam_factor = 1.015

    last_improve = time.time()
    it = 0

    # ------------------------------------------------------------------
    # Adam + smooth‑max phase
    while time.time() < deadline:
        # enforce symmetry of logits
        logits = (logits + logits[::-1]) * 0.5

        # soft‑max → simplex, then re‑symmetrise the vector itself
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        a = (a + a[::-1]) * 0.5
        a /= a.sum()

        # exact objective for bookkeeping
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # grad wrt logits (entropy mirror map)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        it1 = it + 1
        m_hat = m / (1.0 - beta1 ** it1)
        v_hat = v / (1.0 - beta2 ** it1)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # temperature schedule – slowly tighten towards the true max
        lam = min(lam * lam_factor, 1e7)

        # occasional random shake if we are stuck
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        if it >= 500_000:     # safety cap
            break

    # ------------------------------------------------------------------
    # Exponentiated‑gradient sub‑gradient phase on the true max
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()
    a = (a + a[::-1]) * 0.5
    a /= a.sum()

    eta = 0.6
    sub_last_improve = time.time()
    for sub_it in range(30_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[Subgrad n={n} iter={sub_it:05d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps non‑negativity)
        a = a * np.exp(-eta * grad_a)

        # projection onto the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # re‑symmetrise
        a = (a + a[::-1]) * 0.5
        a /= a.sum()

        if time.time() - sub_last_improve > 8.0:
            break

    return global_best_seq, global_best_val


def _local_random_search(
    a: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_seq: np.ndarray,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climbing around the current best vector.
    Random Gaussian perturbations are projected back onto the symmetric simplex.
    """
    rng = np.random.default_rng()
    sigma = 0.002  # perturbation magnitude (relative to total mass)

    while time.time() < deadline:
        delta = rng.normal(scale=sigma, size=n)
        a_new = a + delta
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s < 1e-12:
            continue
        a_new /= s
        a_new = (a_new + a_new[::-1]) * 0.5
        a_new /= a_new.sum()

        val = _objective(a_new, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a_new.copy()
            a = a_new
            print(f"[Local search] improved to {val:.6f}")

    return best_seq, best_val


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a small, diverse set of starting vectors.
    """
    cand = []

    # warm start and simple shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # pure Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # power‑law edge concentrate
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # double‑Gaussian edges
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # half‑sine wave
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises the evaluation score.  The routine respects the given
    time budget and returns the best sequence found as a plain list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Base problem size (current known good size)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Warm start from previously discovered best, if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] != n:
            # resize by cropping/padding
            if warm.shape[0] > n:
                warm = warm[:n]
            else:
                warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # Assemble diverse start points
    candidates = _initial_candidates(n, warm)

    # ------------------------------------------------------------------
    # Optimise each candidate while time permits
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break

        a0 = np.maximum(cand, 0.0)
        s = a0.sum()
        if s < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
            s = a0.sum()
        a0 /= s

        best_seq, best_val = _adam_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Local random refinement on the best vector found so far
    if best_seq is not None and time.time() < deadline - 2.0:
        best_seq, best_val = _local_random_search(
            best_seq, n, conv_fft_len, deadline, best_seq, best_val
        )

    # ------------------------------------------------------------------
    # If time permits, go to a larger grid (8192) – a finer discretisation
    if time.time() + 15.0 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation onto the finer grid
            x_old = np.linspace(0.0, 1.0, n, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        best_seq, best_val = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )

        # final local refinement on the large grid
        if best_seq is not None and time.time() < deadline - 2.0:
            best_seq, best_val = _local_random_search(
                best_seq,
                n_big,
                conv_fft_len_big,
                deadline,
                best_seq,
                best_val,
            )

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded, return a uniform distribution.
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Return as plain Python list
    return best_seq.tolist()
