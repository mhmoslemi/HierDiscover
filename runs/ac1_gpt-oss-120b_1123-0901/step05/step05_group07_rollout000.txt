"""

    F(a) = 2·n·max(a*a) / (sum(a)^2)

The algorithm combines several complementary search phases:

1. **Diverse initialisation** – draws a small pool of candidates from
   Dirichlet, Gaussian‑shaped and sine‑power families (the latter often
   lie close to the optimum).

2. **Smooth‑max Adam optimisation** – each candidate is optimised in the
   log‑softmax (logits) space with a temperature‑controlled smooth maximum.
   The temperature is annealed linearly to a very small value, after which
   the exact max (indicator weights) is used, yielding a true sub‑gradient.

3. **Local stochastic refinement** – from the best sequence found so far a
   simple simulated‑annealing style mass‑transfer move is performed many
   times (transfer a tiny amount of probability mass from one index to
   another).  Improvements are always accepted, occasional uphill moves are
   allowed to escape shallow basins.

All phases respect a global wall‑time budget (default 1000 s).  The best
sequence discovered (as a plain Python list) is returned.

The implementation uses only NumPy, `time` and the standard library, and
avoids any file or network I/O.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT‑based convolution / correlation (fast O(n log n))
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length n:
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Objective and gradient utilities
# ----------------------------------------------------------------------
def objective_from_a(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) (a is assumed to sum to 1)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return 2.0 * n * np.max(conv)


def smooth_weights(conv: np.ndarray, tau: float):
    """Return a smooth‑max value and the corresponding softmax weights."""
    c_max = conv.max()
    shifted = (conv - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    smooth = c_max + tau * np.log(sum_exp)
    weights = exp_shifted / sum_exp
    return smooth, weights


def exact_weights(conv: np.ndarray, tol: float = 1e-12):
    """Return the exact max‑indicator weights (sub‑gradient of max)."""
    c_max = conv.max()
    mask = conv >= c_max - tol
    count = np.count_nonzero(mask)
    if count == 0:  # should never happen
        return conv.max(), np.ones_like(conv) / conv.size
    weights = mask.astype(float) / float(count)
    return c_max, weights


# ----------------------------------------------------------------------
# Generation of diverse starting points
# ----------------------------------------------------------------------
def generate_initial_candidates(n: int, rng: np.random.Generator, max_candidates=12):
    """Yield a handful of promising initial probability vectors."""
    mids = (n - 1) / 2.0
    idx = np.arange(n, dtype=float)

    # 1) Uniform (always a good baseline)
    yield np.full(n, 1.0 / n, dtype=float)

    # 2) Dirichlet random
    for _ in range(2):
        a = rng.dirichlet(np.ones(n))
        yield a

    # 3) Gaussian‑shaped bumps with random width
    for _ in range(3):
        sigma = rng.uniform(n * 0.07, n * 0.30)  # width relative to n
        a = np.exp(-0.5 * ((idx - mids) / sigma) ** 2)
        a = np.maximum(a, 0.0)
        if a.sum() == 0:
            continue
        a /= a.sum()
        yield a

    # 4) Sine‑power families
    for p in rng.uniform(0.5, 4.0, size=3):
        a = np.sin(np.pi * (idx + 0.5) / n) ** p
        a = np.maximum(a, 0.0)
        if a.sum() == 0:
            continue
        a /= a.sum()
        yield a

    # 5) If a previous best sequence is present, use it (trim or pad)
    if 'height_sequence_1' in globals():
        prev = np.asarray(height_sequence_1, dtype=float)
        if prev.shape[0] >= n:
            a = prev[:n].copy()
        else:
            a = np.pad(prev, (0, n - prev.shape[0]))
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-6:
            a = rng.dirichlet(np.ones(n))
        else:
            a /= a.sum()
        yield a


# ----------------------------------------------------------------------
# Adam optimiser working on logits (soft‑max parametrisation)
# ----------------------------------------------------------------------
def adam_optimize_logits(
    init_a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 25000,
    tau_final: float = 5e-7,
    tau_initial: float = 0.12,
    lr_initial: float = 0.09,
    decay_every: int = 4000,
):
    """
    Perform Adam optimisation on the log‑softmax variables.
    Returns the best sequence (probability vector) and its objective value.
    """
    # Soft‑max initialisation
    x = np.log(init_a + 1e-15)

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_initial

    fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))   # ≥ (2n‑1)+n‑1

    best_a = None
    best_val = float("inf")
    it = 0
    last_improve = time.time()

    # Main loop
    while time.time() < deadline and it < max_iters:
        # ----- soft‑max: obtain a probability vector (sum = 1) -----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ----- convolution -----
        conv = full_convolution(a, fft_len=fft_len_conv)

        # ----- temperature schedule -----
        tau = max(tau_final, tau_initial * (1.0 - it / max_iters))

        if tau > 1e-9:
            _, weights = smooth_weights(conv, tau)
        else:
            _, weights = exact_weights(conv, tol=1e-12)

        # ----- objective (exact) -----
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            # optional progress print
            print(
                f"[n={n} iter={it:05d}] new best {cur_val:.6f} (tau={tau:.2e})"
            )

        # ----- gradient of smooth‑max w.r.t. a -----
        grad_a = 2.0 * cross_correlation(weights, a, fft_len=fft_len_corr)

        # ----- back‑prop through the soft‑max -----
        inner = np.dot(a, grad_a)
        grad_x = a * (grad_a - inner)

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # learning‑rate decay
        if it % decay_every == 0:
            lr *= 0.71

        # occasional random jitter if stagnating
        if time.time() - last_improve > 30.0:
            jitter = (np.random.rand(n) - 0.5) * 4e-3
            x += jitter
            last_improve = time.time()
            print(f"[n={n} iter={it}] shake (stagnation)")

    return best_a, best_val


# ----------------------------------------------------------------------
# Local stochastic refinement (mass‑transfer simulated annealing)
# ----------------------------------------------------------------------
def local_refinement(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_moves: int = 30000,
    init_temp: float = 1e-2,
    temp_decay: float = 0.9995,
    min_temp: float = 1e-7,
):
    """
    Starting from a0 (probability vector), perform many random mass transfers.
    Accept improvements always; accept uphill moves with probability exp(-Δ/T).
    Returns the best vector found and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = objective_from_a(a)

    fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    T = init_temp
    moves = 0

    while time.time() < deadline and moves < max_moves:
        # pick two distinct indices
        i, j = np.random.choice(n, 2, replace=False)

        # propose moving a small fraction of a[i] to a[j]
        if a[i] <= 1e-12:
            moves += 1
            continue
        # epsilon is a random fraction of the source mass (2‑20 %)
        eps = a[i] * (np.random.rand() * 0.18 + 0.02)
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        # compute objective (fast FFT)
        conv = full_convolution(a_new, fft_len=fft_len)
        new_val = 2.0 * n * np.max(conv)

        delta = new_val - best_val
        if delta < -1e-12:
            # improvement – always accept
            a = a_new
            best_a = a_new.copy()
            best_val = new_val
            # optional progress print (sparse)
            if moves % 2000 == 0:
                print(f"[local] improved to {new_val:.6f} (move {moves})")
        else:
            # uphill – accept with probability exp(-Δ/T)
            if T > min_temp and np.random.rand() < math.exp(-delta / T):
                a = a_new
        # temperature schedule
        T = max(min_temp, T * temp_decay)
        moves += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the overall wall‑time budget (default
    1000 seconds) and returns the best sequence (as a plain Python list)
    discovered within that time.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    best_seq = None
    best_val = float("inf")
    rng = np.random.default_rng(seed)

    # Lengths to try – start with modest size, then expand if time permits.
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # Pre‑compute FFT lengths used repeatedly
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))

        # Determine a reasonable number of Adam restarts (more for smaller n)
        restarts = 5 if n <= 1024 else 3 if n <= 2048 else 2

        for r in range(restarts):
            if time.time() > deadline:
                break

            # ---- initialise a candidate ---------------------------------
            # Prefer a warm start from the global best if it matches length.
            if best_seq is not None and best_seq.shape[0] == n:
                a0 = best_seq.copy()
                a0 = np.maximum(a0, 0.0)
                if a0.sum() < 1e-6:
                    a0 = rng.dirichlet(np.ones(n))
                else:
                    a0 = a0 / a0.sum()
                print(f"[n={n} restart={r}] warm start from global best")
            else:
                # pick one from a diverse pool
                candidates = list(generate_initial_candidates(n, rng, max_candidates=12))
                a0 = rng.choice(candidates)

            # ---- Adam optimisation ---------------------------------------
            max_iters = 25000 if n <= 1024 else 18000 if n <= 2048 else 12000
            a_opt, val_opt = adam_optimize_logits(
                init_a=a0,
                n=n,
                deadline=deadline,
                max_iters=max_iters,
                tau_final=5e-7,
                tau_initial=0.12,
                lr_initial=0.09,
                decay_every=4000,
            )
            if a_opt is not None and val_opt < best_val - 1e-12:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[n={n} restart={r}] Adam best {val_opt:.6f}")

            # ---- Local stochastic refinement -----------------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a_ref, val_ref = local_refinement(
                    a0=best_seq,
                    n=n,
                    deadline=deadline,
                    max_moves=30000,
                    init_temp=1e-2,
                    temp_decay=0.9995,
                )
                if val_ref < best_val - 1e-12:
                    best_val = val_ref
                    best_seq = a_ref.copy()
                    print(f"[n={n} restart={r}] local refine best {val_ref:.6f}")

    # Fallback to a uniform vector if nothing was found (should not happen)
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=float)
        best_val = objective_from_a(best_seq)

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
