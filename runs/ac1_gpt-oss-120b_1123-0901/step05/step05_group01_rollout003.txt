import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) = n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)² (sum a is assumed > 0)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)

def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence by linear interpolation, then re‑normalise to unit sum."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new

        tot = tmp.sum()
        if tot < 1e-12:
            tmp = np.random.dirichlet(np.ones(n))
        else:
            tmp = tmp / tot
        candidates.append(tmp)

    # 4.  Pure random Dirichlet
    candidates.append(np.random.dirichlet(np.ones(n)))

    # 5.  Edge‑concentrated power‑law shapes: ((i+1)*(n-i))**α,  α∈[-2,-0.2]
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 6):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            candidates.append(shape / shape.sum())

    # 6.  Two‑Gaussian edge shapes
    for gamma in (0.1, 0.3, 0.7, 1.5):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        candidates.append(shape / shape.sum())

    # 7.  Inverse‑root edge shapes (i+1)**‑α + (n‑i)**‑α
    for alpha in (0.4, 0.7, 1.2):
        i = np.arange(n, dtype=np.float64) + 1.0
        shape = np.power(i, -alpha) + np.power(n - i + 1.0, -alpha)
        candidates.append(shape / shape.sum())

    # 8.  Three‑peak Gaussian mixture
    for sigma in (0.04, 0.09, 0.15):
        x = np.arange(n, dtype=np.float64) / (n - 1)
        peaks = sum(np.exp(-0.5 * ((x - p) / sigma) ** 2) for p in (0.2, 0.5, 0.8))
        candidates.append(peaks / peaks.sum())

    # 9.  Random perturbations of the current best (if any)
    if best_seq is not None:
        for _ in range(3):
            eps = np.random.normal(scale=0.02, size=n)
            pert = best_seq + eps
            pert = np.maximum(pert, 0.0)
            if pert.sum() < 1e-12:
                pert = np.random.dirichlet(np.ones(n))
            else:
                pert = pert / pert.sum()
            candidates.append(pert)

    np.random.shuffle(candidates)
    # Limit the number to keep the total budget reasonable
    return candidates[:16]

# Optimisation phases ----------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_c: int,
    fft_len_x: int,
    deadline: float,
    max_iters: int,
    lam_start: float = 0.05,
    lam_factor: float = 1.03,
    lam_max: float = 1e7,
    lr: float = 0.06,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the log‑sum‑exp smooth‑max surrogate.
    Returns the best (a, value) seen before the deadline.
    """
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    lam = lam_start
    best_a = a.copy()
    best_val = _objective(a, n, fft_len_c)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # Soft‑max projection onto the simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # Smooth‑max weights (log‑sum‑exp trick)
        conv = _full_convolution(a, fft_len_c)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        ws = w.sum()
        if ws == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws

        # Gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_x)

        # Back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # λ‑schedule (approaches the true max)
        lam = min(lam * lam_factor, lam_max)

        cur_val = _objective(a, n, fft_len_c)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t:06d}] new best {cur_val:.6f}")

        # Random shake if stalled
        if time.time() - last_improve > 3.5:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


def _subgrad_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_c: int,
    fft_len_x: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 4000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent (exponentiated‑gradient) on the true max.
    The step size η_t decays as η0 / sqrt(t).
    """
    a = a0 / a0.sum()
    best_a = a.copy()
    if init_val is None:
        best_val = _objective(a, n, fft_len_c)
    else:
        best_val = init_val
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_c)
        max_c = conv.max()
        # active set (allow a tiny tolerance)
        active = conv >= max_c - 1e-14
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_x)

        eta = eta0 / np.sqrt(it)

        # multiplicative update (keeps positivity)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        cur_val = _objective(a, n, fft_len_c)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it}] new best {cur_val:.6f}")

        # early stop if no improvement for a while
        if time.time() - last_improve > 3.0:
            break

    return best_a, best_val


def _hill_climb_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_c: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Cheap stochastic hill‑climbing: shift a random amount of mass between two indices.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_c) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2:
            continue

        # move up to 6 % of mass from i1 to i2
        eps = np.random.rand() * 0.06 * a[i1]
        if eps <= 0.0:
            continue
        new_a = a.copy()
        new_a[i1] -= eps
        new_a[i2] += eps
        if new_a[i1] < -1e-15:
            continue
        new_a /= new_a.sum()

        new_val = _objective(new_a, n, fft_len_c)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            last_improve = time.time()
            if it % 100 == 0:
                print(f"[n={n} HillClimb it={it}] new best {new_val:.6f}")
        else:
            # occasional uphill move for escaping plateaus
            if np.random.rand() < 0.001:
                a = new_a

        if time.time() - last_improve > 2.5:
            # no improvement for a while → stop early
            break

    return best_a, best_val


def _annealing_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_c: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 2500,
    start_temp: float = 1e-3,
    end_temp: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style local search.
    Each move redistributes a small amount of probability mass among three positions.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_c) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # temperature schedule (exponential decay)
        tau = it / max_iters
        temp = start_temp * (end_temp / start_temp) ** tau

        # choose three distinct indices
        i, j, k = np.random.choice(n, size=3, replace=False)
        # amount to move from i to (j,k)
        eps = np.random.rand() * 0.04 * a[i]
        if eps <= 0.0:
            continue
        # split eps between j and k
        split = np.random.rand()
        new_a = a.copy()
        new_a[i] -= eps
        new_a[j] += eps * split
        new_a[k] += eps * (1.0 - split)
        if new_a[i] < -1e-15:
            continue
        new_a /= new_a.sum()

        new_val = _objective(new_a, n, fft_len_c)

        # accept if better, otherwise with Metropolis probability
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[n={n} Anneal it={it}] new best {new_val:.6f}")

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function sequence that minimises
    F(a) = 2·n·max(a∗a) / (sum a)².
    Returns the best sequence found within the given time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Grab a historic solution if the environment already provides one.
    hist_seq = None
    if "height_sequence_1" in globals():
        tmp = globals()["height_sequence_1"]
        if isinstance(tmp, (list, tuple, np.ndarray)):
            hist_seq = np.asarray(tmp, dtype=np.float64)

    # ------------------------------------------------------------------
    # Length schedule – start modest, finish with the largest affordable size.
    # Larger n gives a lower theoretical lower bound but costs more time.
    lengths = [4096, 8192, 16384, 32768]   # 32768 still fits comfortably in RAM

    # Pre‑compute FFT lengths (next power of two) for each n.
    fft_len_conv = {}
    fft_len_corr = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_len_conv[n] = _next_pow2(conv_len)
        # correlation length = conv_len + n - 1 ≤ 3·n, use next pow2 for safety
        fft_len_corr[n] = _next_pow2(conv_len + n - 1)

    # ------------------------------------------------------------------
    # MAIN optimisation loop over the chosen lengths.
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        fft_c = fft_len_conv[n]
        fft_x = fft_len_corr[n]

        # Generate a varied set of start points.
        candidates = _generate_initial_candidates(n, best_seq, hist_seq)

        # Adam iteration budget – more for smaller n, but still generous.
        max_adam_iters = int(18000 * 1024 / n) + 2500

        for cid, cand in enumerate(candidates):
            if time.time() > deadline:
                break

            # --------------------------------------------------------------
            # Phase 1 – smooth‑max Adam
            a_adam, val_adam = _adam_optimize(
                cand,
                n,
                fft_c,
                fft_x,
                deadline,
                max_iters=max_adam_iters,
                lam_start=0.04,
                lam_factor=1.02,
                lr=0.05,
            )

            # --------------------------------------------------------------
            # Phase 2 – exact sub‑gradient
            a_sub, val_sub = _subgrad_optimize(
                a_adam,
                n,
                fft_c,
                fft_x,
                deadline,
                init_val=val_adam,
                max_iters=3500,
                eta0=0.9,
            )

            # --------------------------------------------------------------
            # Phase 3 – cheap hill‑climb
            a_hc, val_hc = _hill_climb_optimize(
                a_sub,
                n,
                fft_c,
                deadline,
                init_val=val_sub,
                max_iters=3000,
            )

            # --------------------------------------------------------------
            # Phase 4 – simulated annealing polish (if time permits)
            if time.time() < deadline - 5.0:
                a_ann, val_ann = _annealing_optimize(
                    a_hc, n, fft_c, deadline, init_val=val_hc, max_iters=2500
                )
            else:
                a_ann, val_ann = a_hc, val_hc

            # --------------------------------------------------------------
            # Update global best
            if val_ann < best_val:
                best_val = val_ann
                best_seq = a_ann.copy()
                print(f"*** Global new best (n={n} cand={cid}) value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Long‑run refinement on the overall best (if any time remains)
    if best_seq is not None and time.time() < deadline - 2.0:
        n = best_seq.shape[0]
        fft_c = fft_len_conv.get(n, _next_pow2(2 * n - 1))
        fft_x = fft_len_corr.get(n, _next_pow2(3 * n))
        print("\n=== Final long‑run refinement of the global best ===")
        refined_seq, refined_val = _adam_optimize(
            best_seq,
            n,
            fft_c,
            fft_x,
            deadline,
            max_iters=int(2.5 * (18000 * 1024 / n) + 3500),
            lam_start=0.02,
            lam_factor=1.015,
            lr=0.04,
        )
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq.copy()
            print(f"*** Refined global best value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback guard – should never trigger because we always have a candidate.
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # The evaluator expects a plain Python list.
    return best_seq.tolist()
