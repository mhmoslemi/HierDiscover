def _subgradient_multi_active(a0, target_sum, deadline):
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_val = _full_convolution_fft(a,a).max()
    # parameters
    epsilon = 1e-3  # fraction for active set tolerance
    step = 0.1 * target_sum / n
    prev_grad = None
    prev_a = None
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a,a)
        max_val = conv.max()
        # active set indices where conv >= max_val * (1 - epsilon)
        active = np.where(conv >= max_val * (1 - epsilon))[0]
        # compute weight vector w: uniform over active set (or normalized)
        w = np.zeros_like(conv)
        if active.size > 0:
            w[active] = 1.0 / active.size
        else:
            # fallback: use only max index
            w[np.argmax(conv)] = 1.0
        # gradient
        grad = _weighted_gradient(a, w)
        # Barzilai-Borwein step size
        if prev_a is not None and prev_grad is not None:
            s = a - prev_a
            y = grad - prev_grad
            s_norm2 = np.dot(s, s)
            sy = np.dot(s, y)
            if sy > 0:
                step = s_norm2 / sy
            else:
                step = 0.1 * target_sum / n
        # update
        a_candidate = a - step * grad
        a_candidate = _project_onto_simplex(a_candidate, target_sum)
        val_candidate = _full_convolution_fft(a_candidate, a_candidate).max()
        if val_candidate < best_val - 1e-12:
            best_val = val_candidate
            best_a = a_candidate.copy()
            # optionally increase step
            step *= 1.05
        else:
            step *= 0.7
        prev_a = a
        prev_grad = grad
        a = a_candidate
        # early stop if best_val < 1.503
        if best_val < 1.503:

        # occasional jitter
        if it % 300 == 0 and np.random.rand() < 0.1:
            a = _project_onto_simplex(a + np.random.rand(n) * 1e-4, target_sum)
    return best_a, best_val

        i, j = np.random.randint(0, n, size=2)

"""
Hybrid optimizer for the step‑function problem.

Strategy
---------
1. Warm‑start from the globally available `height_sequence_1` if present.
2. Perform many short restarts that minimise a *smooth* approximation of the
   max‑convolution using L‑BFGS‑B on the logits of a soft‑max
   parametrisation (the loss is τ·log‑sum‑exp(conv/τ)).  The temperature τ is
   reduced in several stages, always warm‑starting the next stage from the
   previous solution.
3. Refine the best candidate with a projected sub‑gradient method that
   aggregates **all** almost‑active convolution constraints (indices whose
   value is within a relative tolerance of the current maximum).  The step‑size
   is chosen by a Barzilai‑Borwein rule.
4. Final stochastic hill‑climbing that transfers a small amount of mass
   between two indices (and occasionally swaps whole entries) using a
   simulated‑annealing acceptance rule.

All three phases respect the global time budget (default 1000 s) and the
sequence is always kept on the simplex
    Σ a_i = √(2 n),   a_i ≥ 0,
so the evaluation score equals the maximal convolution value.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())            # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)               # eliminate tiny negatives
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. 2008
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {w ≥ 0, Σw = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of a weighted sum of convolution entries
# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    `w` must have length 2·n‑1 where n = len(a) and Σ w = 1.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    L = 3 * n - 2               # length of the full a * w_rev convolution
    size = 1 << ((L - 1).bit_length())
    Fa = np.fft.rfft(a, n=size)
    Fw = np.fft.rfft(w_rev, n=size)
    conv = np.fft.irfft(Fa * Fw, n=size)[:L]
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * conv[idx]
    return grad


# ----------------------------------------------------------------------
# Smooth‑max loss & gradient on the logits (soft‑max parametrisation)
# ----------------------------------------------------------------------
def _smooth_max_and_grad(
    x: np.ndarray, target_sum: float, tau: float
) -> tuple[float, np.ndarray]:
    """
    For logits `x` returns (loss, grad_x) where loss = τ·log‑sum‑exp(conv/τ)
    and conv = a*a with a = target_sum·softmax(x).  The gradient is with
    respect to `x`.
    """
    # soft‑max to obtain a satisfying Σa = target_sum
    max_x = np.max(x)
    exp_shift = np.exp(x - max_x)
    p = exp_shift / np.sum(exp_shift)          # probabilities, Σp = 1
    a = target_sum * p

    # convolution and smooth‑max (log‑sum‑exp) computation
    conv = _full_convolution_fft(a, a)
    scaled = conv / tau
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    sum_exp = np.sum(exp_shifted)
    lse = max_scaled + np.log(sum_exp)         # log‑sum‑exp
    loss = tau * lse

    # weights w = softmax(conv/τ)  (they already sum to 1)
    w = exp_shifted / sum_exp

    # gradient w.r.t. a
    grad_a = _weighted_gradient(a, w)

    # back‑prop through the soft‑max parametrisation
    dotp = np.dot(p, grad_a)
    grad_x = target_sum * p * (grad_a - dotp)

    return loss, grad_x


# ----------------------------------------------------------------------
# L‑BFGS optimisation of the smooth‑max
# ----------------------------------------------------------------------
def _lbfgs_optimize(
    n: int,
    target_sum: float,
    init_a: np.ndarray,
    deadline: float,
) -> np.ndarray:
    """
    Minimises the smooth‑max approximation using a sequence of decreasing
    temperatures τ and L‑BFGS‑B.  Returns the final coefficient vector.
    """
    try:
        import scipy.optimize
    except Exception:               # SciPy not available – abort this phase
        return init_a.copy()

    # initialise logits
    x = np.log(init_a + 1e-12)

    # get an initial temperature from the initial max‑convolution
    init_conv = _full_convolution_fft(init_a, init_a)
    cur_tau = max(init_conv.max() * 0.20, 1e-4)   # start relatively warm

    # temperature schedule – geometric decay
    taus = [cur_tau, cur_tau * 0.5, cur_tau * 0.25, cur_tau * 0.125, cur_tau * 0.0625]

    # keep the best logits seen so far (in case optimiser diverges)
    best_x = x.copy()
    best_val = np.inf

    for tau in taus:
        # abort if we run out of time
        if time.time() >= deadline:
            break

        max_iter = 80                                   # modest number of iterations
        res = scipy.optimize.minimize(
            fun=_smooth_max_and_grad,
            x0=best_x,
            args=(target_sum, tau),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': max_iter, 'disp': False},
        )
        best_x = res.x
        # evaluate the exact objective for the obtained logits
        max_x = np.max(best_x)
        exp_shift = np.exp(best_x - max_x)
        p = exp_shift / np.sum(exp_shift)
        a_cur = target_sum * p
        # store the best candidate (exact max of convolution)
        max_conv = _full_convolution_fft(a_cur, a_cur).max()
        if max_conv < best_val:
            best_val = max_conv
            best_seq = a_cur.copy()
        # continue with the next, colder temperature, warm‑starting from the
        # current logits
        # (no explicit line‑search – the optimiser already adapts step‑sizes)

    return best_seq if best_val < np.inf else init_a.copy()


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with multi‑active constraints
# ----------------------------------------------------------------------
def _subgradient_multi_active(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    rel_tol_start: float = 0.02,
    rel_tol_end: float = 1e-5,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent where the gradient aggregates *all* constraints
    whose value is within a relative tolerance of the current maximum.
    Uses a Barzilai‑Borwein step‑size rule.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # schedule for the relative tolerance (tighten as optimisation proceeds)
    start_time = time.time()
    rel_tol = rel_tol_start

    prev_a = None
    prev_grad = None
    step = 0.1 * target_sum / n                      # initial step (≈ 1 % of mean)

    it = 0
    while time.time() < deadline:
        it += 1
        # recompute tolerance depending on elapsed fraction
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        rel_tol = rel_tol_start * (1 - frac) + rel_tol_end * frac

        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        # active set: indices whose value ≥ max·(1‑rel_tol)
        active = np.where(conv >= max_val * (1 - rel_tol))[0]

        # build weight vector (uniform over active set)
        w = np.zeros_like(conv)
        if active.size > 0:
            w[active] = 1.0 / active.size
        else:                     # fall back to the single maximum index
            w[np.argmax(conv)] = 1.0

        grad = _weighted_gradient(a, w)

        # Barzilai‑Borwein step size (if we have a previous step)
        if prev_a is not None and prev_grad is not None:
            s = a - prev_a
            y = grad - prev_grad
            s_norm2 = np.dot(s, s)
            sy = np.dot(s, y)
            if sy > 0.0:
                step = s_norm2 / sy
            else:
                step = 0.1 * target_sum / n

        a_candidate = a - step * grad
        a_candidate = _project_onto_simplex(a_candidate, target_sum)

        val_candidate = _full_convolution_fft(a_candidate, a_candidate).max()
        if val_candidate < best_val - 1e-12:
            best_val = val_candidate
            best_a = a_candidate.copy()
            step *= 1.03                       # slightly increase if success
        else:
            step *= 0.70                       # otherwise shrink

        prev_a = a
        prev_grad = grad
        a = a_candidate

        # occasional jitter to escape plateaus
        if it % 300 == 0 and np.random.rand() < 0.12:
            a = _project_onto_simplex(a + np.random.rand(n) * 1e-4, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Enhanced stochastic hill‑climbing
# ----------------------------------------------------------------------
def _hillclimb_enhanced(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Local search that moves a small amount of mass between two indices and
    also occasionally swaps whole entries.  Acceptance follows a simple
    simulated‑annealing schedule.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    T0 = best_val * 0.30            # initial temperature
    T_end = 1e-5
    start = time.time()
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # transfer a tiny fraction of mass from i to j
        delta = a[i] * np.random.rand() * 0.10
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # with a small probability, attempt a whole‑entry swap
        if np.random.rand() < 0.08:
            k = np.random.randint(0, n)
            a_swapped = a.copy()
            a_swapped[i], a_swapped[k] = a_swapped[k], a_swapped[i]
            a_new = a_swapped

        new_val = _full_convolution_fft(a_new, a_new).max()

        accept = False
        if new_val < best_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hill‑climb it={it} new best={best_val:.6f}")

        # periodic re‑normalisation to avoid numerical drift
        if it % 2500 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid search for a step‑function minimising
        2·n·max(conv(a,a)) / (sum(a)**2)

    The routine respects the time budget, runs a series of L‑BFGS
    refinements on a smooth surrogate, follows up with a multi‑active
    sub‑gradient descent and finishes with a stochastic hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15      # tiny safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Warm‑start from the globally supplied construction, if any.
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        n0 = hs_arr.shape[0]
        target0 = math.sqrt(2.0 * n0)
        hs_norm = hs_arr * (target0 / hs_arr.sum())
        score0 = evaluate_sequence(hs_norm.tolist())
        best_seq = hs_norm.copy()
        best_score = float(score0)
        print(f"[{time.time():.2f}] warm‑start seed score={best_score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – many short L‑BFGS runs on the smoothed max.
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        # allocate a bite of the remaining budget for a single L‑BFGS run
        slice_time = min(15.0, remaining * 0.40)

        # choose a problem size – bias towards cheaper runs when time runs low
        if remaining > 400:
            n = np.random.randint(1800, 3500 + 1)
        elif remaining > 180:
            n = np.random.randint(1300, 2800 + 1)
        else:
            n = np.random.randint( 800, 2000 + 1)

        target_sum = math.sqrt(2.0 * n)

        # ----- initialise a candidate -----
        if np.random.rand() < 0.65:
            a0 = np.random.rand(n)
        else:
            p_exp = np.random.uniform(0.20, 0.80)
            idx = np.arange(1, n + 1, dtype=float)
            a0 = idx ** (-p_exp)
            if np.random.rand() < 0.45:
                tau_exp = np.random.uniform(0.5, 2.0) * n
                a0 *= np.exp(-idx / tau_exp)

        a0 = a0 / a0.sum() * target_sum

        # occasionally inject the current best (trim / pad)
        if best_seq is not None and np.random.rand() < 0.30:
            if best_seq.shape[0] >= n:
                a0 = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.shape[0]) * 1e-6
                a0 = np.concatenate([best_seq, pad])
            a0 = a0 / a0.sum() * target_sum

        run_deadline = time.time() + slice_time
        seq_local = _lbfgs_optimize(n, target_sum, a0, run_deadline)

        # exact evaluation of the obtained sequence
        score_local = evaluate_sequence(seq_local.tolist())
        if score_local < best_score:
            best_score = float(score_local)
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] GLOBAL new best={best_score:.6f} (n={n})")
            if best_score < 1.503:
                break

    # --------------------------------------------------------------
    # Phase 2 – projected multi‑active sub‑gradient descent.
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 12.0:
        n = best_seq.shape[0]
        target_sum = math.sqrt(2.0 * n)
        sub_deadline = time.time() + min(40.0, time_left * 0.45)
        seq_sub, val_sub = _subgradient_multi_active(best_seq.copy(),
                                                     target_sum,
                                                     sub_deadline)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑grad best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final stochastic hill‑climb for fine‑tuning.
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 6.0:
        n = best_seq.shape[0]
        target_sum = math.sqrt(2.0 * n)
        hill_deadline = time.time() + time_left
        seq_hill, val_hill = _hillclimb_enhanced(best_seq.copy(),
                                                target_sum,
                                                hill_deadline)
        if val_hill < best_score:
            best_score = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] after hill‑climb best={best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback – uniform distribution (should never be used).
    # --------------------------------------------------------------
    if best_seq is None:
        n = 1500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # Return a plain Python list as required by the evaluator.
    return best_seq.tolist()
