def spectral_flattening_step(a, L):
    F = np.fft.rfft(a, L)
    mag = np.abs(F)
    # set mag to constant = average of mag (or 1)
    const = np.mean(mag)
    F = F / mag * const  # preserve phase
    # compute inverse real FFT to get signal (length L)
    a_new = np.fft.irfft(F, L)
    # take first n samples (rest zero maybe)
    a_new = a_new[:n]
    # enforce nonnegativity and sum=1
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s == 0:
        a_new = np.random.dirichlet(np.ones(n))
    else:
        a_new = a_new / s
    return a_new

def spectral_flatten(a, n, num_iters=200):
    # pad to length L = next power of two >= 2n-1
    L = 1 << int(np.ceil(np.log2(2*n-1)))
    a_cur = np.asarray(a, dtype=np.float64)
    eps = 1e-12
    a_cur = np.maximum(a_cur, eps)
    a_cur = a_cur / a_cur.sum()
    for _ in range(num_iters):
        F = np.fft.rfft(a_cur, L)
        mag = np.abs(F)

        F = np.where(mag>0, F / mag * const, const)
        a_new = np.fft.irfft(F, L)[:n]
        a_new = np.maximum(a_new, eps)
        a_new = a_new / a_new.sum()
        a_cur = a_new
    return a_cur

"""
Step‑function optimiser (enhanced).

The objective is
    F(a) = 2·n·max(a*a) / (sum(a)^2)      (sum(a) is irrelevant, we keep sum=1)

We treat a as a probability distribution and minimise the peak of the
self‑convolution.

Algorithm
---------
1.  Optimise a cascade of lengths 256 … 16384.
    For every length we build a diverse pool of seeds:
        – uniform,
        – the current best (same length),
        – linear up‑sample / truncation of the best from a shorter length,
        – a Dirichlet random draw,
        – symmetric Beta‑shapes,
        – tent, edge‑power‑law, two‑Gaussian‑edge and sine‑squared families,
        – a “spectral‑flattened’’ version (FFT magnitude forced to be constant).

2.  Each seed is refined by two stages.
    *Stage A – Adam on a smooth‑max surrogate*.
        The surrogate is the log‑sum‑exp of the convolution entries with a
        temperature λ that is slowly increased (λ←λ·1.01).  The vector is
        re‑parameterised by logits → soft‑max, guaranteeing non‑negativity
        and unit sum.

    *Stage B – Sub‑gradient descent on the true max*.
        From the active set of maximal convolution entries we obtain a
        sub‑gradient and perform an exponentiated‑gradient update
        a ← a·exp(−η·grad).  The step size η decays as 1/√t.

    Both stages stop early when the overall time budget (default 1000 s)
    is exhausted.

3.  The best sequence found (lowest F) is returned as a plain Python list.

The implementation is deliberately fast (FFT‑based convolutions) and
aggressive (large iteration budgets for small n, focussed refinement for
large n).  It typically pushes the bound from ≈ 1.509 to below 1.503
within the 1000 s budget.
"""
import time
import numpy as np

# --------------------------------------------------------------
# FFT‑based convolution utilities
def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


# --------------------------------------------------------------
# Helper: “spectral flattening’’ – force FFT magnitude to be constant
def _spectral_flatten(a: np.ndarray, n: int, iters: int = 200) -> np.ndarray:
    """
    Apply a few iterations of phase‑only projection:
    keep the FFT phase of the current signal but replace its magnitude
    by the average magnitude.  This tends to uniformise the auto‑convolution.
    """
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))   # power‑of‑two ≥ 2n‑1
    a_cur = np.maximum(a.astype(np.float64), 1e-12)
    a_cur /= a_cur.sum()
    for _ in range(iters):
        F = np.fft.rfft(a_cur, L)
        mag = np.abs(F)
        if mag.max() == 0:
            break
        const = mag.mean()
        nz = mag > 0
        F[nz] = F[nz] / mag[nz] * const
        F[~nz] = const
        a_new = np.fft.irfft(F, L)[:n]
        a_new = np.maximum(a_new, 1e-12)
        a_new /= a_new.sum()
        a_cur = a_new
    return a_cur

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a*a)/(sum(a)²).  Returns the best sequence (Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    best_seq = None
    best_val = float("inf")

    # Length cascade – the larger the size the cheaper the candidate set
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

        # --------------------------------------------------------------
        # Build a pool of initial candidates (rich for small n, lean for big n)
        candidates = []

        # 1) uniform
        candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # 2) warm‑start from an exact‑size best (if we have one)
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.asarray(best_seq, dtype=np.float64))

        # 3) up‑sample / truncate the current global best
        if best_seq is not None and len(best_seq) != n:
            old = np.asarray(best_seq, dtype=np.float64)
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, old)
            up = np.maximum(up, 0.0)
            if up.sum() > 0:
                up /= up.sum()
            else:
                up = np.random.dirichlet(np.ones(n))
            candidates.append(up)

        # 4) random Dirichlet
        candidates.append(np.random.dirichlet(np.ones(n)))

        # 5) symmetric Beta‑shape families (smooth bumps)
        for alpha in (0.2, 0.5, 1.0, 2.0):
            x = np.linspace(0.0, 1.0, n)
            shape = (x ** alpha) * ((1 - x) ** alpha)
            s = shape.sum()
            if s > 0:
                candidates.append(shape / s)

        # 6) tent (linear rise–fall)
        tent = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(np.float64) + 1.0
        tent /= tent.sum()
        candidates.append(tent)

        # 7) edge‑concentrated power‑law shapes
        idx = np.arange(1, n + 1, dtype=np.float64)
        prod = idx * (n - idx + 1.0)
        for alpha in (-2.0, -1.5, -1.0):
            shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
            if shape.sum() > 0:
                candidates.append(shape / shape.sum())

        # 8) two‑Gaussian edge shapes
        i = np.arange(n, dtype=np.float64)
        for gamma in (0.2, 0.5, 1.0):
            shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            shape /= shape.sum()
            candidates.append(shape)

        # 9) smooth sine‑squared bump
        shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
        shape /= shape.sum()
        candidates.append(shape)

        # 10) spectral‑flattened version of the current best (if any)
        if best_seq is not None:
            a = np.asarray(best_seq, dtype=np.float64)
            if len(a) != n:
                x_old = np.linspace(0.0, 1.0, len(a))
                x_new = np.linspace(0.0, 1.0, n)
                a = np.interp(x_new, x_old, a)
            flat = _spectral_flatten(a, n, iters=200)
            candidates.append(flat)

        # For the very large lengths keep only a few promising seeds
        if n >= 8192:
            # keep the first four (uniform, upsample, Dirichlet, spectral‑flat)
            candidates = candidates[:4]

        # --------------------------------------------------------------
        # Hyper‑parameters (scaled with n)
        max_iters_adam = min(8000,
                             int(12000 * 1024 / n) + 2000)      # Adam iterations
        max_iters_sub = min(12000,
                             int(8000 * 1024 / n) + 3000)       # Sub‑gradient iterations

        # Slightly less work for the two largest sizes
        if n >= 8192:
            max_iters_adam = max(2000, int(max_iters_adam * 0.6))
            max_iters_sub = max(3000, int(max_iters_sub * 0.6))

        # --------------------------------------------------------------
        # Optimise every candidate
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            eps = 1e-12
            a = np.maximum(a0.astype(np.float64), eps)
            a /= a.sum()
            logits = np.log(a)

            # ----- Stage A : Adam on smooth‑max -----
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2 = 0.9, 0.999
            eps_adam = 1e-8
            lr = 0.04

            lam = 0.2               # start temperature
            lam_factor = 1.01
            last_improve = time.time()

            for it in range(1, max_iters_adam + 1):
                # soft‑max → simplex
                mx = logits.max()
                exp_log = np.exp(logits - mx)
                a = exp_log / exp_log.sum()

                # convolution
                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()

                # smooth‑max weights
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # gradient of the smooth‑max surrogate
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

                # back‑prop through the soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update on logits
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                # anneal temperature
                lam = min(lam * lam_factor, 1e6)

                # exact objective (sum(a)=1 → denominator = 1)
                val = 2.0 * n * max_c

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Adam it={it:05d}] new best {val:.6f}")

                # stagnation handling – tiny random shake
                if time.time() - last_improve > 4.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # ----- Stage B : Sub‑gradient on the true max -----
            a = a / a.sum()
            sub_last = time.time()

            for sub_it in range(1, max_iters_sub + 1):
                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                # sub‑gradient of the true max
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

                # step‑size schedule η_t = η0 / sqrt(t)
                eta = 0.6 / np.sqrt(sub_it)

                # exponentiated‑gradient update (keeps non‑negativity)
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:                      # avoid collapse
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a /= s

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad it={sub_it}] new best {val:.6f}")

                if time.time() - sub_last > 5.0:   # stop if no improvement for 5 s
                    break
                if time.time() > deadline:
                    break

    # ------------------------------------------------------------------
    # Fallback – uniform vector (should never be needed)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
