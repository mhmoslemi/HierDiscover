"""
Propose a non‑negative step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The optimizer works in three stages:

1. **Many short Adam runs** on the simplex (soft‑max parametrisation) with a
   smoothed log‑sum‑exp surrogate of the max‑convolution.  The runs are
   performed with diverse random initial shapes and lengths, keeping the
   best few candidates as warm‑starts.

2. **Linear‑programming refinement** – for the current best sequence we
   linearise the maximum‑convolution around the active (tight) constraints
   and solve a tiny LP that attempts to lower the maximal entry while
   preserving the total sum and non‑negativity.

3. **Stochastic hill‑climbing** – tiny mass transfers between random
   positions are accepted if they improve the exact objective.  This can
   exploit tiny non‑convexities left after the previous stages.

All computations use FFT‑based convolutions (O(n log n)) and a fast
implementation of the evaluation function, identical to the reference
`evaluate_sequence`.  The whole search respects the 1000 s budget and
returns the best sequence found as a plain Python list.
"""
import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Fast convolution (full linear convolution) via FFT
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of two real vectors using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative values caused by floating‑point noise
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Exact (but cheap) replica of the official evaluator
# ----------------------------------------------------------------------
def _fast_evaluate(seq: np.ndarray) -> float:
    """Compute the same value as `evaluate_sequence` but without safety checks."""
    seq = np.clip(seq, 0.0, 1000.0)
    n = seq.shape[0]
    conv = _full_convolution_fft(seq, seq)
    max_conv = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2 * n * max_conv / (sum_a ** 2))


# ----------------------------------------------------------------------
# Random initial shapes (all are scaled later to the target sum)
# ----------------------------------------------------------------------
def _initial_shape_random(n: int, rng: np.random.Generator) -> np.ndarray:
    typ = rng.choice(['power', 'exponential', 'mixed', 'uniform'])
    idx = np.arange(1, n + 1, dtype=float)

    if typ == 'power':
        p = rng.uniform(0.1, 2.0)
        shape = idx ** (-p)
    elif typ == 'exponential':
        tau = rng.uniform(0.2, 0.9) * n
        shape = np.exp(-idx / tau)
    elif typ == 'mixed':
        split = rng.integers(1, n - 1)
        left = idx[:split] ** (-rng.uniform(0.2, 1.0))
        right = (n - idx[split:] + 1) ** (-rng.uniform(0.2, 1.0))
        shape = np.concatenate([left, right])
    else:  # uniform
        shape = np.ones_like(idx)

    shape += rng.random(n) * 1e-6           # tiny jitter
    shape = np.maximum(shape, 1e-12)
    return shape


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max (log‑it) variables with a smooth max surrogate
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    rng: np.random.Generator,
    *,
    init_tau_factor: float = 0.25,
    min_tau_factor: float = 5e-4,
    alpha: float = 0.07,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau_decay: float = 0.96,
    max_iter: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a smoothed‑max surrogate.
    Returns the best sequence encountered (as a NumPy array) and its
    official score (via ``evaluate_sequence``).
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_val = float("inf")
    iteration = 0
    last_improve = 0

    while time.time() < run_deadline:
        # 1) soft‑max mapping (a is on the simplex, sum = target_sum)
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)          # numerical stabilisation
        p = exp_shift / exp_shift.sum()
        a = target_sum * p

        # 2) current convolution and maximum entry
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()

        if max_conv < best_val:
            best_val = max_conv
            best_seq = a.copy()
            last_improve = iteration
            print(f"[{time.time():.2f}] n={n}  local best = {max_conv:.8f}")

        # 3) smooth‑max weights (log‑sum‑exp)
        tau = max(init_tau_factor * max_conv * (tau_decay ** iteration),
                  min_tau_factor * max_conv)
        scaled = (conv - max_conv) / tau               # ≤ 0
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # 4) gradient w.r.t. a  (derivative of Σ w_k·conv_k)
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)        # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)                # pick the needed slice
        grad_a = 2.0 * conv2[idx]

        # 5) gradient w.r.t. logits x (chain rule through the soft‑max)
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # optional clipping to avoid huge steps
        gnorm = np.linalg.norm(grad_x)
        if gnorm > 10.0:
            grad_x = grad_x * (10.0 / gnorm)

        # 6) Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaux
        if iteration % 300 == 0 and rng.random() < 0.15:
            x += rng.normal(size=n) * 0.001

        # slowly decay the learning rate
        alpha *= 0.9997

        # early stop if no progress for a while
        if iteration - last_improve > 2000:
            break

        if max_iter is not None and iteration >= max_iter:
            break

    # final official evaluation of the best candidate found in this run
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (first‑order improvement of the max‑convolution)
# ----------------------------------------------------------------------
def _lp_refine_one_step(seq: np.ndarray, rng: np.random.Generator) -> tuple[np.ndarray, bool]:
    """
    Perform a single LP‑based refinement step.
    Returns the (potentially) improved sequence and a flag indicating success.
    """
    n = seq.size
    conv = _full_convolution_fft(seq, seq)
    M = conv.max()

    # active set – indices whose value is within relative tolerance of the maximum
    rel_tol = 5e-9
    tol = max(rel_tol * M, 1e-12)
    active = np.where(conv >= M - tol)[0]
    if active.size == 0:
        return seq, False

    # Build the matrices for `linprog`
    # Variables: Δ_0 … Δ_{n‑1}, t
    rows = []
    for k in active:
        row = np.zeros(n)
        # row[j] = seq[k‑j] whenever the index is in range
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        idx_j = np.arange(j_start, j_end + 1)
        row[idx_j] = seq[k - idx_j]
        rows.append(2.0 * row)       # factor 2 from the linearisation

    A_ub = np.hstack([np.vstack(rows), -np.ones((len(rows), 1))])
    b_ub = -M * np.ones(len(rows))

    # Equality: sum_i Δ_i = 0 (preserve total mass)
    A_eq = np.concatenate([np.ones((1, n)), np.zeros((1, 1))], axis=1)
    b_eq = np.array([0.0])

    # Bounds: Δ_i >= -seq_i   (keep non‑negative), t >= 0
    bounds = [(-float(v), None) for v in seq] + [(0.0, None)]

    # Objective: minimise t
    c = np.concatenate([np.zeros(n), np.array([1.0])])

    res = linprog(c, A_ub, b_ub, A_eq, b_eq, bounds=bounds, method='highs')
    if not res.success:
        return seq, False

    delta = res.x[:n]
    new_seq = seq + delta
    new_seq = np.clip(new_seq, 0.0, 1000.0)

    # Renormalise to the canonical sum (tiny drift may appear)
    target_sum = np.sqrt(2.0 * n)
    sum_new = new_seq.sum()
    if abs(sum_new - target_sum) > 1e-8:
        new_seq *= target_sum / sum_new

    old_score = _fast_evaluate(seq)
    new_score = _fast_evaluate(new_seq)
    if new_score < old_score - 1e-9:
        return new_seq, True
    return seq, False

        n = cur_seq.size
        avg_val = cur_seq.mean()
        step = max(avg_val * 0.05, 1e-3)            # start with ~5 % of the average height

        while time.time() < deadline - 0.5:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            max_transfer = min(step, cur_seq[i])
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer
            new_seq = cur_seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta

            new_score = _fast_evaluate(new_seq)
            if new_score < cur_score - 1e-9:
                cur_seq = new_seq
                cur_score = new_score
                step = min(step * 1.05, avg_val * 0.2)   # slowly increase step on success
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = cur_seq.copy()
                    print(f"[{time.time():.2f}] Hill‑climb improved to {best_score:.8f}")
            else:
                step = max(step * 0.97, avg_val * 1e-4)   # shrink step on failure

    # ------------------------------------------------------------------
    # Fallback if nothing was ever generated
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Final official evaluation (ensures identical rounding behaviour)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time():.2f}] Final official score = {final_score:.8f}")

    # Return a plain Python list as required by the harness
    return best_seq.tolist()
