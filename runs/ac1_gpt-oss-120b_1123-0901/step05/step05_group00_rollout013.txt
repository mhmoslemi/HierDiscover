def solve_lp_max_sum(a, M):
    n = len(a)

    if not res.success:
        return None
    else:
        return res.x

"""
Step‑function optimiser (improved).

This optimiser searches for a non‑negative vector a (the heights of a step
function) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)²)

where “*” denotes the linear convolution.  The method combines several ideas:

* A rich pool of initial families (uniform, Dirichlet draws, power‑law,
  edge Gaussians, central bumps, sine‑squared, up‑sampled best, …).
* A symmetric Adam optimiser on a smooth‑max (log‑sum‑exp) surrogate.
  The vector is kept on the probability simplex by a soft‑max parametrisation
  and forced to be mirror‑symmetric after every step – this greatly helps
  flatten the self‑convolution.
* For moderate lengths (≤ 4096) an exact linear‑programming step:
  with the current vector f fixed, we maximise Σ gᵢ subject to
  f * g ≤ max(f * f).  The LP direction is then mixed into f.
* A final exact sub‑gradient descent on the true max‑convolution.
* After the cascade of lengths the best vector is up‑sampled to a larger
  length (if time permits) and refined again.

The routine respects a wall‑clock budget (default 1000 s) and returns the
best sequence it discovered as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper FFT‑based convolution utilities
def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


def _symmetrize(x: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry (x[i] = x[-1-i]) and renormalise."""
    y = (x + x[::-1]) * 0.5
    s = y.sum()
    if s == 0:
        return np.full_like(y, 1.0 / y.size)
    return y / s


# ----------------------------------------------------------------------
# Linear‑programming direction (used for modest lengths)
def _lp_direction(a: np.ndarray, current_max: float):
    """
    Solve   max   Σ g_i
            s.t.  a * g ≤ current_max   (componentwise)
                  g_i ≥ 0
    Returns the optimal g (may be None on failure).
    """
    n = a.size
    rows = 2 * n - 1
    # Build the convolution matrix A such that conv = A @ g
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    # Upper‑bound constraints
    ub = np.full(rows, current_max, dtype=np.float64)
    # Objective: maximise Σ g_i  → minimise -Σ g_i
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method="highs")
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Optimisation of a single candidate (Adam → optional LP → sub‑gradient)
def _optimise_candidate(
    a0: np.ndarray,
    deadline: float,
    max_iters_factor: float = 5.0,
    subgrad_iters: int = 2000,
    lp_steps: int = 5,
    lp_mix: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Return a (vector, value) pair: the refined distribution and its
    objective value 2·n·max(conv).  The routine respects the global deadline.
    """
    n = a0.size
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    a = _symmetrize(a)

    # Adam on smooth‑max surrogate -------------------------------------------------
    logits = np.log(a + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05

    lam = 0.5                 # temperature for smooth‑max
    lam_factor = 1.01

    conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

    max_iters = int(max_iters_factor * 1024 / n) + 5000
    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        a = _symmetrize(a)                # enforce symmetry each step

        # convolution and smooth‑max weights
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # back‑propagation through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature annealing
        lam = min(lam * lam_factor, 1e7)

        # exact objective (sum(a)=1 → denominator = 1)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={it:05d}] new best {val:.6f}")

        # stagnation handling
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    # Optional LP‑based “direction” step (only for modest n) ----------------------
    if n <= 4096 and time.time() < deadline:
        a = best_a.copy()
        for lp_it in range(lp_steps):
            if time.time() > deadline:
                break
            conv = _full_convolution(a, fft_len=conv_fft_len)
            cur_max = conv.max()
            g0 = _lp_direction(a, cur_max)
            if g0 is None or g0.sum() == 0:
                break
            g = _symmetrize(g0 / g0.sum())
            a = (1.0 - lp_mix) * a + lp_mix * g
            a = _symmetrize(a)
            # evaluate
            conv = _full_convolution(a, fft_len=conv_fft_len)
            val = 2.0 * n * conv.max()
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[n={n} LP it={lp_it}] new best {val:.6f}")

    # Exact sub‑gradient refinement ------------------------------------------------
    a = best_a.copy()
    eta = 0.6
    sub_last = time.time()
    for sub_it in range(subgrad_iters):
        if time.time() > deadline:
            break
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        a = a * np.exp(-eta * grad_a)
        a = _symmetrize(a)

        s = a.sum()
        if s < 1e-14:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            sub_last = time.time()
            print(f"[n={n} Subgrad it={sub_it}] new best {val:.6f}")

        if time.time() - sub_last > 5.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a*a)/(sum(a)²).

    The routine proceeds through a cascade of lengths,
    generates diverse starts, refines each with a symmetric
    Adam optimiser, occasionally augments it with an LP direction,
    finishes with an exact sub‑gradient step and finally
    refines the best vector at a larger scale if time permits.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper to assemble a diverse pool of initial vectors for a given n
    def _initial_candidates(n, prev_best):
        cand = []

        # 1) uniform
        cand.append(np.full(n, 1.0 / n, dtype=np.float64))

        # 2) previous best of exactly the same size
        if prev_best is not None and len(prev_best) == n:
            cand.append(np.array(prev_best, dtype=np.float64))

        # 3) up‑sample previous best (linear interpolation)
        if prev_best is not None and len(prev_best) < n:
            old = np.asarray(prev_best, dtype=np.float64)
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, old)
            up = np.maximum(up, 0.0)
            up = up / up.sum() if up.sum() > 0 else np.random.dirichlet(np.ones(n))
            cand.append(up)

        # 4) power‑law edge‑concentrated shapes
        idx = np.arange(1, n + 1, dtype=np.float64)
        prod = idx * (n - idx + 1.0)
        for alpha in np.linspace(-2.0, -0.5, 4):
            shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
            if shape.sum() > 0:
                shape = shape / shape.sum()
                cand.append(shape)

        # 5) two‑Gaussian edge shapes
        i = np.arange(n, dtype=np.float64)
        for gamma in [0.2, 0.5, 1.0, 2.0]:
            shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            shape = shape / shape.sum()
            cand.append(shape)

        # 6) central Gaussian bump
        sigma = n / 8.0
        cen = i - (n - 1) / 2.0
        shape = np.exp(-0.5 * (cen / sigma) ** 2)
        shape = shape / shape.sum()
        cand.append(shape)

        # 7) double‑edge narrow bumps
        sigma_e = n / 32.0
        shape = np.exp(-0.5 * (i / sigma_e) ** 2) + np.exp(-0.5 * ((n - 1 - i) / sigma_e) ** 2)
        shape = shape / shape.sum()
        cand.append(shape)

        # 8) smooth sine‑squared bump
        shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
        shape = shape / shape.sum()
        cand.append(shape)

        # 9) random Dirichlet draws (two of them)
        cand.append(np.random.dirichlet(np.ones(n)))
        cand.append(np.random.dirichlet(np.ones(n)))

        # 10) historic global variable if present
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            h = np.asarray(hseq, dtype=np.float64)
            if h.shape[0] > n:
                h = h[:n]
            elif h.shape[0] < n:
                h = np.pad(h, (0, n - h.shape[0]))
            h = np.maximum(h, 0.0)
            h = h / h.sum() if h.sum() > 0 else np.random.dirichlet(np.ones(n))
            cand.append(h)

        return cand

    # ------------------------------------------------------------------
    # Process a cascade of sizes – small to large
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        candidates = _initial_candidates(n, best_seq)

        for cid, a0 in enumerate(candidates):
            if time.time() > deadline:
                break
            # Run the optimiser on this start
            try:
                a_opt, val = _optimise_candidate(a0, deadline)
            except Exception as e:
                print(f"  Candidate {cid} raised an exception and is skipped: {e}")
                continue

            if val < best_val:
                best_val = val
                best_seq = a_opt.copy()
                print(f"  *** Global new best at n={n} cand={cid}: {val:.6f}")

    # ------------------------------------------------------------------
    # If time permits, upscale the best vector and give it one more long
    # refinement pass (larger n usually yields a smaller bound).
    if best_seq is not None and time.time() < deadline - 10.0:
        n_big = 65536
        print(f"\n=== Final upscale to n={n_big} for extra polishing ===")
        # Linear interpolation to the larger grid
        x_old = np.linspace(0.0, 1.0, best_seq.size)
        x_new = np.linspace(0.0, 1.0, n_big)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()
        a_big = _symmetrize(a_big)

        # A more generous optimisation budget for the final size
        a_big_opt, val_big = _optimise_candidate(
            a_big,
            deadline,
            max_iters_factor=8.0,          # more Adam iterations
            subgrad_iters=4000,            # more sub‑gradient steps
            lp_steps=0                      # LP not needed for this size
        )
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big_opt.copy()
            print(f"  *** Global new best after upscale: {val_big:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was discovered (highly unlikely)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Ensure the list is plain Python floats (the external evaluator expects this)
    return best_seq.tolist()
