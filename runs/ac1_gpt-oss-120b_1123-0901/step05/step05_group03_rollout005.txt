def simplex_projection(v):
    u = np.sort(v)[::-1]
    sv = np.cumsum(u)
    rho = np.where(u > (sv - 1) / np.arange(1, len(v)+1))[0][-1]
    theta = (sv[rho] - 1) / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def optimize(a, n, deadline):
    conv_fft_len = _next_pow2(2*n-1)
    corr_fft_len = _next_pow2(3*n-2)
    best_a = a.copy()
    best_val = objective(a)
    step = 0.1
    it = 0
    last_improve = time.time()
    while time.time() < deadline and it < max_iters:
        # compute conv and gradient
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # compute objective
        val = 2 * n * max_c / (a.sum()**2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} GD iter={it}] new best {val:.6f}")
        # compute grad
        active = np.isclose(conv, max_c, atol=1e-12)
        if not active.any():
            # fallback: set w uniform
            w = np.ones_like(conv) / len(conv)
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        grad = 2 * _cross_correlation(w, a, corr_fft_len)
        # Nesterov's momentum: maintain v
        # but implement simple step
        # compute tentative new a via projection
        a_new = a - step * grad
        a_new = simplex_projection(a_new)
        # compute new objective to accept if improves else shrink step
        conv_new = _full_convolution(a_new, conv_fft_len)
        max_c_new = conv_new.max()
        val_new = 2 * n * max_c_new / (a_new.sum()**2)
        if val_new < val:
            a = a_new
            # maybe increase step slightly
            step = step * 1.02
        else:
            # shrink step
            step = step * 0.5
            # optionally add random perturbation

            last_improve = time.time()
        it += 1

    return best_a, best_val

"""
search algorithm for minimum of 2·n·max_convolution / (sum)^2

Approach:
  1. Build several diverse initial probability vectors (warm‑start if
     available, Dirichlet, power‑law edges, edge‑Gaussians, sinusoid,
     uniform).
  2. For each start run a projected gradient descent on the exact objective.
     The gradient is the (scaled) cross‑correlation of the uniform weight on
     the active convolution entries (where the max is attained).  After each
     gradient step we clip to non‑negative and re‑normalise (simplex projection).
  3. The step size decays slowly; if no improvement is seen for a few seconds
     a small random perturbation (“shake”) is added.
  4. When time permits the best distribution is interpolated to a larger grid
     (doubling n) and the whole procedure is repeated.  Larger n gives a finer
     search space and often lowers the bound.
  5. The best sequence found before the deadline is returned as a plain list.

All heavy work is done with FFT‑based convolution/correlation,
so a single iteration on n≈8192 costs only a few milliseconds, allowing
hundreds of thousands of iterations within the 1000 s budget.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective value F = 2·n·max(a*a) / (sum(a)^2)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returns the closest vector w with w_i ≥ 0 and Σ w_i = 1.
    """
    if v.ndim != 1:
        raise ValueError("simplex_projection expects a 1‑d array")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u > (cssv - 1) / np.arange(1, len(v) + 1))[0]
    if rho.size == 0:
        # all entries are zero or negative -> uniform distribution
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # renormalise to avoid tiny numerical drift
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest pool of diverse start vectors for the optimiser."""
    cand = []

    # warm start and its rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # power‑law edge‑concentrated shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # simple sinusoidal shape (positive half‑wave)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform vector (baseline)
    cand.append(np.full(n, 1.0 / n))

    return cand

def _pgd_optimize(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the exact max‑convolution objective.
    Returns the best vector and its objective value found within the time limit.
    """
    a = a0.copy()
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # initial best values
    best_seq = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    # step‑size schedule
    step = 0.2            # initial additive step size
    decay = 0.9995        # decay per iteration

    it = 0
    last_improve = time.time()
    # tolerance for active set detection
    atol = 1e-12

    while time.time() < deadline and it < max_iters:
        # convolution & max
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        # record best
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} PGD iter={it:05d}] new best {val:.6f}")

        # compute sub‑gradient of the exact max
        active = np.isclose(conv, max_c, atol=atol)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.full_like(conv, 1.0 / conv.size)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # gradient step + projection onto the simplex
        a = a - step * grad
        a = np.maximum(a, 0.0)                # keep non‑negative
        s = a.sum()
        if s < 1e-12:
            # collapse – jump to a fresh Dirichlet point
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # step‑size decay
        step *= decay

        # random shake if stagnating
        if time.time() - last_improve > 5.0:
            a = a + (np.random.rand(n) - 0.5) * 1e-2
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            last_improve = time.time()

        it += 1

    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (probability vector) that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)²)

    using a multi‑scale projected‑gradient descent with diverse initialisations
    and occasional random perturbations.  The routine respects the given time
    budget and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Warm start (previous best) if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        warm = np.maximum(warm, 0.0)
        s = warm.sum()
        if s > 0.01:
            warm = warm / s
        else:
            warm = None
    else:
        warm = None

    best_seq = None
    best_val = float("inf")
    # start at a moderate size – 4096 gave good results previously
    n = 4096
    print(f"=== start optimisation, n={n} ===")

    # ------------------------------------------------------------------
    # First optimisation level (n = 4096)
    candidates = _initial_candidates(n, warm)
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        s = a0.sum()
        if s < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / s

        seq, val = _pgd_optimize(a0, n, deadline)
        if val < best_val:
            best_val = val
            best_seq = seq
            print(f"[global] new best after candidate {idx} → {val:.6f}")

    # ------------------------------------------------------------------
    # If time permits, move to a larger grid (doubling size each round)
    while time.time() + 10 < deadline:
        n_big = n * 2
        if n_big > 16384:        # arbitrary upper bound to keep memory reasonable
            break
        print(f"=== upscale to n={n_big} ===")
        if best_seq is not None:
            # linear interpolation to the larger grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        seq, val = _pgd_optimize(a_big, n_big, deadline)
        if val < best_val:
            best_val = val
            best_seq = seq
            print(f"[global] new best after upscale → n={n_big} val={val:.6f}")

        # prepare for next possible upscale
        n = n_big

    # ------------------------------------------------------------------
    # Safety fallback – if nothing succeeded produce a uniform vector.
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    elapsed = time.time() - start_time
    print(f"search finished in {elapsed:.1f}s, best value {best_val:.6f}")

    return best_seq.tolist()

        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # simple positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform baseline
    cand.append(np.full(n, 1.0 / n))

    return cand


def _pgd_optimize(a0: np.ndarray,
                  n: int,
                  deadline: float,
                  max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the exact max‑convolution objective.
    Returns the best vector and its objective value found within the time limit.
    """
    a = a0.copy()
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    step = 0.2         # additive step size
    decay = 0.9995     # exponential decay per iteration
    atol = 1e-12
    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} PGD iter={it:05d}] new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=atol)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.full_like(conv, 1.0 / conv.size)

        # sub‑gradient of the exact max
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # gradient step + projection onto the simplex
        a = a - step * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # decay step size
        step *= decay

        # random shake if we have been stagnant for a while
        if time.time() - last_improve > 5.0:
            a = a + (np.random.rand(n) - 0.5) * 1e-2
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (probability vector) that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)²)

    using a multi‑scale projected‑gradient descent with diverse initialisations
    and occasional random perturbations.  The routine respects the given time
    budget and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Warm start (previous best) if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        warm = np.maximum(warm, 0.0)
        s = warm.sum()
        if s > 0.01:
            warm = warm / s
        else:
            warm = None
    else:
        warm = None

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Primary optimisation at n = 4096 (the current state‑of‑the‑art size)
    n = 4096
    print(f"=== start optimisation, n={n} ===")
    candidates = _initial_candidates(n, warm)

    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        s = a0.sum()
        if s < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / s

        seq, val = _pgd_optimize(a0, n, deadline)
        if val < best_val:
            best_val = val
            best_seq = seq
            print(f"[global] new best after candidate {cid} → {val:.6f}")

    # ------------------------------------------------------------------
    # Upscale to larger grids while time permits (doubling each round)
    while time.time() + 10 < deadline:
        n_big = n * 2
        if n_big > 16384:   # keep memory reasonable
            break
        print(f"=== upscale to n={n_big} ===")
        if best_seq is not None:
            # linear interpolation to the larger grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        seq, val = _pgd_optimize(a_big, n_big, deadline)
        if val < best_val:
            best_val = val
            best_seq = seq
            print(f"[global] new best after upscale → n={n_big} val={val:.6f}")

        n = n_big

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded produce a uniform vector.
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    elapsed = time.time() - start_time
    print(f"search finished in {elapsed:.1f}s, best value {best_val:.6f}")

    return best_seq.tolist()
