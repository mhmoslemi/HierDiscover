"""
Improvements over the baseline
--------------------------------
* Multi‑scale optimisation: start with a modest length (≈500),
  optimise aggressively, then interpolate to larger sizes (≈2000‑4000)
  and continue optimisation.  This lets the optimiser discover a good
  “shape’’ quickly and then refine it on finer grids.
* Enforced symmetry: the convolution of a step function and its reverse is
  symmetric, therefore the optimal (or near‑optimal) solution is also
  symmetric.  After every update we replace a by (a + a[::-1]) / 2.
  The search space is halved and the optimisation becomes far more stable.
* Exact sub‑gradient of the max‑convolution (instead of a smooth soft‑max)
  is used inside an Adam update on the simplex‑parameterisation
  (log‑its → soft‑max).  This removes the bias introduced by the temperature
  schedule of the original implementation.
* A more aggressive learning‑rate schedule and tighter jittering improve the

# Helper utilities -----------------------------------------------------

def _project_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex {x≥0, sum x = target_sum}."""
    # Duchi et al. (2008) – O(n log n) due to the sort, fine for n ≤ 8000.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force the vector to be symmetric (a[i] == a[-i-1])."""
    return (a + a[::-1]) * 0.5


def _max_convolution(a: np.ndarray, Nfft: int) -> (np.ndarray, float):
    """Full linear convolution a*a (via FFT) and its maximum."""
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * a.shape[0] - 1]
    return conv, float(conv.max())


def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """A small, diverse pool of starts (all already normalized)."""
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(1.0, 1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) small random perturbation of uniform
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.02 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    return init


# ----------------------------------------------------------------------
# Core optimiser (Adam on log‑its, exact sub‑gradient of max‑conv) -----

def _subgradient_adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> (np.ndarray, float):
    """
    Adam optimisation of the log‑its (exponentiated‑gradient) using the exact
    sub‑gradient of the max‑convolution.  Symmetry is enforced after each
    update.  Returns the best vector (with sum = sqrt(2n)) seen in this run
    together with its exact max‑convolution value.
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise logits from the start vector (tiny epsilon to avoid log(0))
    eps = 1e-16
    y = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.12                # base learning rate on logits
    jitter_interval = 2000

    # FFT length – must cover a full linear convolution (length 2n‑1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- softmax → current vector (guaranteed Σa = target_sum) -----
        logits = y - np.max(y)                 # for numerical stability
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()
        a = target_sum * p

        # enforce symmetry (keeps the sum unchanged)
        a = _symmetrize(a)

        # ----- exact convolution and max -----
        conv, M = _max_convolution(a, Nfft)

        # ----- keep global best -----
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

        # ----- sub‑gradient of max (pick a random active shift) -----
        # active set: all k where conv[k] is within a tiny tolerance of the max
        tol = max(1e-12, 1e-9 * M)
        active = np.where(conv >= M - tol)[0]
        k = rng.choice(active)

        # gradient wrt a (2·a_{k‑i} where defined, else 0)
        idx = np.arange(n)
        k_minus_idx = k - idx
        mask = (k_minus_idx >= 0) & (k_minus_idx < n)
        g_a = np.zeros_like(a)
        g_a[mask] = 2.0 * a[k_minus_idx[mask]]

        # ----- chain rule to logits (mirror‑descent on the simplex) -----
        avg = np.dot(p, g_a)                     # ⟨p, g⟩
        g_y = target_sum * p * (g_a - avg)       # ∂f/∂y_i

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # occasional jitter to escape flat regions
        if it % jitter_interval == 0:
            y += rng.normal(scale=0.01, size=n)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stochastic refinement (pairwise mass transfer + simulated annealing) --

def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> (np.ndarray, float):
    """Cheap hill‑climbing on the simplex (pairwise transfers)."""
    target_sum = a.sum()                     # already √(2 n)
    cur_seq = a.copy()
    cur_score = best_score

    # FFT size for fast max‑convolution
    Nfft = 1 << ((2 * n - 1).bit_length())

    max_steps = 400_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        frac = rng.uniform(0.01, 0.06)           # amount of mass to move
        delta = cur_seq[i] * frac
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        conv, new_score = _max_convolution(new_seq, Nfft)

        # deterministic improvement
        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score


# ----------------------------------------------------------------------
# Multi‑scale driver ---------------------------------------------------

def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine works in several stages:
        * start with a modest length (≈500) and run a large‑scale Adam optimiser;
        * iteratively double the length, interpolating the current best
          and continuing optimisation (fewer iterations for larger sizes);
        * finish with a cheap stochastic refinement.
    Returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0               # safety margin

    # ------------------------------------------------------------------
    # Retrieve any previously discovered construction (if present)
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        base = None

    # ------------------------------------------------------------------
    # Length schedule (geometric growth, capped at 8000)
    lengths = [512, 1024, 2048, 4096, 6000]               # fine‑grained early, coarse later
    lengths = [min(8000, max(300, n)) for n in lengths]

    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    prev_full = None

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # ---- initial vectors -------------------------------------------------
        init_vectors = _make_initial_vectors(n, target_sum, rng)

        # if we have a previous solution of different size, interpolate it
        if prev_full is not None and prev_full.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, prev_full.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, prev_full)
            interp = np.maximum(interp, 0.0)
            interp = interp / interp.sum() * target_sum
            init_vectors.append(interp)

        # ------------------------------------------------------------------
        # iteration budget – more for smaller n, fewer for larger n
        base_iters = 1_800_000                     # tuned for the 2100‑case of the baseline
        max_iters = int(base_iters * 2100 / n)     # keep roughly constant *n*iterations
        max_iters = max(150_000, min(max_iters, 1_200_000))

        print(f"[run] length={n}, max_iters≈{max_iters}, target_sum={target_sum:.3f}")

        for idx, a0 in enumerate(init_vectors):
            if time.time() >= deadline:
                break
            print(f"  start {idx+1}/{len(init_vectors)} for length {n}")

            best_seq, best_score = _subgradient_adam_optimize(
                a0=a0.copy(),
                n=n,
                deadline=deadline,
                max_iters=max_iters,
                best_seq=best_seq if best_seq is not None else a0,
                best_score=best_score,
                rng=rng,
            )

        # store the best full vector for the next size‑up step
        prev_full = best_seq.copy()

    # ------------------------------------------------------------------
    # Final stochastic refinement on the global best (if time permits)
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # allocate up to half of the remaining budget for refinement
        refine_deadline = time.time() + min(remaining * 0.5, 30.0)
        best_seq, best_score = _local_refine(
            a=best_seq,
            n=best_seq.shape[0],
            deadline=refine_deadline,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Fallback – should never happen, but keep it for safety
    if best_seq is None:
        n = 1024
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = float("inf")
        print("[fallback] using uniform sequence")

    # ------------------------------------------------------------------
    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
