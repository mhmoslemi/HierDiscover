def spectral_flatten(a, steps, fft_len): # where a sum=1

    for i in range(steps):
        A = np.fft.rfft(a, fft_len)
        mag = np.abs(A)
        avg_mag = np.mean(mag)
        # Set new magnitude to avg_mag, keep phase
        A_new = avg_mag * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, fft_len)[:len(a)]  # keep length n
        # Ensure nonnegative
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-12: # if all zero, reinitialize
            a_new = np.random.dirichlet(np.ones(len(a)))
            s = a_new.sum()
        a = a_new / s
    return a

"""
Propose a non‑negative step‑function (list of floats) that minimizes

        F(a) = 2·n·max(a*a) / (sum(a)²)

where a*a is the linear convolution.  The algorithm combines several
ideas that were missing or only lightly used in the reference code:

* **Diverse initialisation** – Dirichlet draws, exponential tails,
  power‑law edges, low‑pass filtered random signals and a “constant‑magnitude
  spectral” projection.
* **Spectral projection** – repeatedly enforce a (nearly) constant magnitude
  of the zero‑padded DFT while keeping the phase, then project back onto the
  non‑negative simplex.  This creates very different basins for the later
  optimiser.
* **Adam optimisation on a smooth‑max surrogate** – identical in spirit to the
  reference but with a slower temperature schedule, a larger iteration budget
  for larger sizes and a tighter stagnation criterion.
* **Exact sub‑gradient phase** – exponentiated‑gradient updates using the true
  max of the convolution.
* **Final fine‑tuning & random shake** – a short extra Adam run followed by
  cheap random perturbations (simulated‑annealing‑style) to escape possible
  local minima.

The routine respects a user‑supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list.  All helper
functions are defined at the top level so they can be pickled or
inspected if needed.
"""

import time
import numpy as np
import math


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a via FFT of length ``fft_len``.
    Returns the full convolution (length 2·len(a)‑1).
    """
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)
    # slice to the linear convolution length (the rest are zero‑padded)
    return conv_full[: 2 * a.shape[0] - 1]


def _cross_corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) using FFT.
    Result length = len(w) + len(a) – 1.
    """
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(a[::-1], fft_len)
    corr_full = np.fft.irfft(fw * fr, fft_len)
    return corr_full


def _objective(a: np.ndarray) -> float:
    """
    Exact evaluation of F(a) (identical to evaluate_sequence for non‑negative a).
    """
    s = a.sum()
    if s < 0.01:
        return float("inf")
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    conv = _conv_fft(a, fft_len)[:conv_len]
    max_c = conv.max()
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
def _constant_magnitude_projection(a: np.ndarray, fft_len: int, steps: int = 20) -> np.ndarray:
    """
    Project a onto the set of vectors whose zero‑padded DFT has (approximately)
    constant magnitude.  After each projection the vector is forced back onto
    the non‑negative simplex.
    """
    for _ in range(steps):
        A = np.fft.rfft(a, fft_len)
        target = np.mean(np.abs(A))
        A_new = target * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, fft_len)[: a.shape[0]]
        a_new = np.maximum(a_new, 0.0)
        total = a_new.sum()
        if total < 1e-12:
            a_new = np.random.dirichlet(np.ones_like(a))
            total = a_new.sum()
        a = a_new / total
    return a


def _low_pass_candidate(n: int, cutoff_frac: float) -> np.ndarray:
    """
    Create a smooth candidate by keeping only the lowest ``cutoff_frac`` of the
    DFT coefficients of a random vector.
    """
    x = np.random.rand(n)
    A = np.fft.rfft(x)
    keep = int(math.ceil(cutoff_frac * (A.shape[0] - 1)))
    A[keep + 1 :] = 0.0
    a = np.fft.irfft(A, n)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        total = a.sum()
    return a / total


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # Lengths to explore – larger n gives more expressive power but costs more.
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_len = 2 * n - 1
        fft_len_conv = _next_pow2(conv_len)
        fft_len_corr = _next_pow2(conv_len + n - 1)

        # ------------------------------------------------------------------
        # Build a diverse pool of starting points
        candidates = []

        # 1) Warm‑start from the globally best sequence (pad / truncate)
        if best_seq is not None:
            if len(best_seq) == n:
                candidates.append(np.asarray(best_seq, copy=True))
            else:
                arr = np.zeros(n, dtype=np.float64)
                l = min(len(best_seq), n)
                arr[:l] = best_seq[:l]
                # If we trimmed, renormalise; if we padded, mass is already less than 1
                if arr.sum() < 0.01:
                    arr = np.random.dirichlet(np.ones(n))
                else:
                    arr = arr / arr.sum()
                candidates.append(arr)

        # 2) Uniform vector (baseline)
        candidates.append(np.full(n, 1.0 / n))

        # 3) Random Dirichlet (concentration 1)
        candidates.append(np.random.dirichlet(np.ones(n)))

        # 4) Spiky Dirichlet (concentration < 1)
        for conc in (0.3, 0.5, 0.8):
            candidates.append(np.random.dirichlet(np.full(n, conc)))

        # 5) Two‑sided exponential tails
        for gamma in (0.0005, 0.001, 0.002, 0.005):
            i = np.arange(n, dtype=np.float64)
            shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
            shape /= shape.sum()
            candidates.append(shape)

        # 6) Edge‑concentrated power‑law shape (α<0)
        idx = np.arange(1, n + 1, dtype=np.float64)
        prod = idx * (n - idx + 1)
        for alpha in np.linspace(-2.0, -0.1, 7):
            shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
            if shape.sum() > 0:
                shape /= shape.sum()
                candidates.append(shape)

        # 7) Low‑pass filtered random vectors (different cut‑offs)
        for frac in (0.05, 0.10, 0.20):
            candidates.append(_low_pass_candidate(n, frac))

        # 8) Constant‑magnitude spectral projection of a random Dirichlet
        for _ in range(3):
            a0 = np.random.dirichlet(np.ones(n))
            a0 = _constant_magnitude_projection(a0, fft_len_conv, steps=15)
            candidates.append(a0)

        # ------------------------------------------------------------------
        # Optimise each candidate
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # Normalise to a probability vector
            a = np.maximum(a0, 0.0)
            if a.sum() < 0.01:
                a = np.random.dirichlet(np.ones(n))
            a = a / a.sum()

            # ------ Phase A : Spectral constant‑magnitude projection (optional shake) ------
            a = _constant_magnitude_projection(a, fft_len_conv, steps=20)

            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[n={n} cand={cand_id} init] new best {val:.6f}")

            # ------ Phase B : Adam on a smooth‑max surrogate ------
            logits = np.log(a + 1e-15)                # soft‑max parameters
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.03                                 # a bit smaller for stability
            lam = 0.1                                 # start temperature (smooth‑max)
            lam_factor = 1.02                         # slower growth → gentler approximation
            max_iters = int(30000 * 1024 / n) + 2000  # more iterations for small n
            last_improve = time.time()

            for t in range(1, max_iters + 1):
                # soft‑max → a lives on the simplex
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                # convolution via FFT
                conv = _conv_fft(a, fft_len_conv)[:conv_len]
                max_c = conv.max()

                # smooth‑max weights
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w.fill(1.0 / w.size)
                else:
                    w /= w_sum

                # gradient of the surrogate (2·cross‑correlation)
                corr = _cross_corr_fft(w, a, fft_len_corr)
                start = n - 1
                grad_a = 2.0 * corr[start : start + n]

                # back‑prop through the soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # temperature schedule
                lam = min(lam * lam_factor, 1e6)

                # exact objective (sum(a)=1)
                val = 2.0 * n * max_c

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Adam iter={t}] new best {val:.6f}")

                # occasional jitter if stuck
                if time.time() - last_improve > 6.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # ------ Phase C : Exact sub‑gradient (exponentiated gradient) ------
            a = a / a.sum()
            eta = 0.55
            sub_last_improve = time.time()
            max_sub_iters = 2500

            for sub_it in range(max_sub_iters):
                conv = _conv_fft(a, fft_len_conv)[:conv_len]
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12)

                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                corr = _cross_corr_fft(w, a, fft_len_corr)
                grad_a = 2.0 * corr[start : start + n]

                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad iter={sub_it}] new best {val:.6f}")

                if time.time() - sub_last_improve > 5.0:
                    break
                if time.time() > deadline:
                    break

        # ------------------------------------------------------------------
        # Optional fine‑tuning on the currently best sequence of this length
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] fine‑tuning global best (val={best_val:.6f})")
            a = best_seq.copy()
            logits = np.log(a + 1e-15)
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.02
            lam = 0.3
            lam_factor = 1.01
            last_imp = time.time()
            fine_iters = int(20000 * 1024 / n) + 3000

            for t in range(1, fine_iters + 1):
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                conv = _conv_fft(a, fft_len_conv)[:conv_len]
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum != 0.0:
                    w /= w_sum
                else:
                    w.fill(1.0 / w.size)

                corr = _cross_corr_fft(w, a, fft_len_corr)
                grad_a = 2.0 * corr[start : start + n]
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                lam = min(lam * lam_factor, 1e6)

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_imp = time.time()
                    print(f"[n={n} fine‑tune iter={t}] new best {val:.6f}")

                # small random shake if stagnating
                if time.time() - last_imp > 8.0:
                    logits += (np.random.rand(n) - 0.5) * 5e-3
                    last_imp = time.time()

                if time.time() > deadline:
                    break

        # ------------------------------------------------------------------
        # Light random‑shake post‑processing (simulated‑annealing flavour)
        if best_seq is not None and len(best_seq) == n:
            shake_iters = 2000
            for it in range(shake_iters):
                if time.time() > deadline:
                    break
                # small Gaussian perturbation, re‑project onto simplex
                noise = np.random.normal(scale=0.004, size=n)
                a2 = best_seq + noise
                a2 = np.maximum(a2, 0.0)
                s = a2.sum()
                if s < 0.01:
                    continue
                a2 = a2 / s
                val = _objective(a2)
                if val < best_val:
                    best_val = val
                    best_seq = a2.copy()
                    print(f"[n={n} shake {it}] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (should never happen)
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return as a plain Python list (the external evaluator expects this)
    return best_seq.tolist()
