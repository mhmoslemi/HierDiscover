"""
Propose candidate: L‑BFGS‑based optimizer with log‑sum‑exp surrogate.

Uses several families of initial step‑functions (uniform, Dirichlet,
edge‑concentrated power‑law, two‑Gaussian, sine‑squared bump,
beta‑like product, random piecewise). For each candidate a multi‑stage
L‑BFGS optimisation is performed with increasing smooth‑max temperature,
followed by a short exact‑max sub‑gradient refinement. The best sequence
found within the time budget is returned.
"""
import time, math, numpy as np
from scipy.optimize import minimize

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a)."""
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

class SmoothMaxObjective:
    """Callable that returns smooth‑max of a*a and its gradient w.r.t. logits."""
    def __init__(self, n: int, fft_len_conv: int, fft_len_corr: int, lam: float):
        self.n = n
        self.fft_len_conv = fft_len_conv
        self.fft_len_corr = fft_len_corr
        self.lam = lam

    def __call__(self, logits: np.ndarray):
        # soft‑max to obtain a on the simplex
        mx = np.max(logits)
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution a*a
        conv = _full_convolution(a, self.fft_len_conv)

        # smooth‑max via log‑sum‑exp (stable)
        lam = self.lam
        scaled = lam * conv
        smax = np.max(scaled)
        exp_term = np.exp(scaled - smax)
        sum_exp = exp_term.sum()
        smooth_val = (smax + np.log(sum_exp)) / lam

        # weights for the gradient (soft‑max over convolution entries)
        w = exp_term / sum_exp          # shape (2n‑1)

        # gradient w.r.t. a  (2 * w ★ a)
        grad_a = 2.0 * _cross_correlation(w, a, self.fft_len_corr)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_logits = a * (grad_a - avg)

        return smooth_val, grad_logits

def refine_subgradient(a: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_corr: int,
                       deadline: float) -> np.ndarray:
    """A few exponentiated‑gradient steps on the true max‑convolution."""
    best_a = a.copy()
    best_val = float('inf')
    last_improve = time.time()
    eta = 0.6

    while time.time() < deadline:
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s <= 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
        if time.time() - last_improve > 5.0:
            break
    return best_a

def run_lbfgs(a0: np.ndarray,
              n: int,
              deadline: float) -> np.ndarray:
    """Multi‑stage L‑BFGS optimisation with an increasing temperature."""
    # pre‑compute FFT lengths
    fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fft_len_corr = 1 << int(np.ceil(np.log2(3 * n - 2)))

    # initialise logits (soft‑max parametrisation)
    logits = np.log(np.maximum(a0, 1e-16))

    # temperature schedule – larger λ makes the surrogate closer to a true max
    schedule = [0.5, 5.0, 20.0]

    for lam in schedule:
        if time.time() > deadline:
            break
        obj = SmoothMaxObjective(n, fft_len_conv, fft_len_corr, lam)
        # a fairly generous iteration budget – L‑BFGS uses line‑search internally
        # we cap the number of L‑BFGS iterations to keep the overall budget safe.
        res = minimize(obj,
                       logits,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 80, 'disp': False})
        logits = res.x

    # recover the probability vector
    mx = np.max(logits)
    exp_log = np.exp(logits - mx)
    a = exp_log / exp_log.sum()

    # final short exact‑max polishing
    a = refine_subgradient(a, n, fft_len_conv, fft_len_corr, deadline)
    return a

def generate_candidates(n: int, best_seq: np.ndarray) -> list:
    """Return a diverse collection of starting vectors for length n."""
    candidates = []

    # 1. uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2. warm‑start from previous solution of the same size
    if best_seq is not None and best_seq.shape[0] == n:
        candidates.append(best_seq.copy())

    # 3. up‑sample a shorter best (linear interpolation)
    if best_seq is not None and best_seq.shape[0] < n:
        old = best_seq
        x_old = np.linspace(0.0, 1.0, old.shape[0])
        x_new = np.linspace(0.0, 1.0, n)
        up = np.interp(x_new, x_old, old)
        up = np.maximum(up, 0.0)
        s = up.sum()
        if s > 0:
            up = up / s
        else:
            up = np.random.dirichlet(np.ones(n))
        candidates.append(up)

    # 4. historic global variable if present
    hseq = globals().get("height_sequence_1")
    if hseq is not None:
        h = np.asarray(hseq, dtype=np.float64)
        if h.shape[0] > n:
            h = h[:n]
        elif h.shape[0] < n:
            h = np.pad(h, (0, n - h.shape[0]))
        h = np.maximum(h, 0.0)
        s = h.sum()
        if s > 0:
            h = h / s
        else:
            h = np.random.dirichlet(np.ones(n))
        candidates.append(h)

    # 5. Dirichlet draws with several concentration parameters
    for alpha in [0.2, 0.5, 1.0, 2.0, 5.0]:
        cand = np.random.dirichlet(alpha * np.ones(n))
        candidates.append(cand)

    # 6. power‑law edge‑concentrated shapes  (prod^α)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-3.0, -0.2, 6):
        shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            candidates.append(shape)

    # 7. two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in [0.2, 0.5, 1.0, 2.0]:
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape = shape / shape.sum()
        candidates.append(shape)

    # 8. smooth sine‑squared bump
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    shape = shape / shape.sum()
    candidates.append(shape)

    # 9. random beta‑like product shape a_i ∝ (i+1)^p (n-i)^q
    for _ in range(4):
        p = np.random.uniform(-2.0, 2.0)
        q = np.random.uniform(-2.0, 2.0)
        i = np.arange(n, dtype=np.float64) + 1.0
        shape = np.power(i, p) * np.power((n - i + 1), q)
        shape = np.maximum(shape, 0.0)
        s = shape.sum()
        if s > 0:
            shape = shape / s
            candidates.append(shape)

    # 10. random piecewise‑constant with 5–10 segments
    for _ in range(3):
        seg = np.random.randint(5, 11)
        cut_points = np.sort(np.random.choice(n - 1, seg - 1, replace=False) + 1)
        values = np.random.rand(seg)
        shape = np.empty(n, dtype=np.float64)
        start = 0
        for cp, v in zip(np.append(cut_points, n), values):
            shape[start:cp] = v
            start = cp
        shape = shape / shape.sum()
        candidates.append(shape)

    return candidates

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a*a)/(sum(a)²).

    The algorithm combines a rich pool of initial families with a
    multi‑stage L‑BFGS optimisation on a smooth‑max surrogate and a short
    exact‑max sub‑gradient polishing. It respects the wall‑clock budget and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # keep a safety margin

    best_seq = None
    best_val = float("inf")

    # Lengths processed from small to large – cheap runs warm‑start larger problems
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        candidates = generate_candidates(n, best_seq if best_seq is not None else None)

        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # ensure a proper probability vector
            a0 = np.maximum(a0, 1e-16)
            a0 = a0 / a0.sum()

            # run the optimiser
            try:
                a_opt = run_lbfgs(a0, n, deadline)
            except Exception as exc:
                print(f"  candidate {cand_id} raised exception: {exc}")
                continue

            # evaluate the exact objective (uses the external checker)
            try:
                val = evaluate_sequence(a_opt.tolist())
            except Exception as exc:
                print(f"  evaluate_sequence failed: {exc}")
                continue

            if val < best_val:
                best_val = val
                best_seq = a_opt.copy()
                print(f"  [n={n} cand={cand_id}] new global best {val:.6f}")

            # early exit if we already beat the target threshold
            if best_val <= 1.503:
                print("Target reached – terminating early.")
                return best_seq.tolist()

    # fallback – uniform vector if nothing was found
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = evaluate_sequence(best_seq.tolist())
        print(f"Fallback value {best_val:.6f}")

    print(f"\nFinal best value: {best_val:.6f} (length {best_seq.shape[0]})")
    return best_seq.tolist()

"""
Propose candidate – L‑BFGS optimiser with a log‑sum‑exp surrogate.

The optimiser works on a probability vector a (soft‑max of free logits).
For a fixed smoothing parameter λ it minimises the smooth‑max of the
self‑convolution a*a, i.e.

    Sλ(a) = (1/λ)·log Σ_k exp( λ·(a*a)_k ).

The gradient of Sλ is obtained analytically via a cross‑correlation.
A multi‑stage L‑BFGS run with increasing λ (0.5 → 5 → 20) drives the
solution close to the true max‑convolution.  A short exponentiated‑gradient
refinement on the exact max follows.  A diverse pool of initial
vectors (uniform, Dirichlet, power‑law edges, Gaussian edges, sine‑bump,
beta‑like products, random piecewise‑constant shapes and warm‑starts from
previous lengths) is explored for several lengths (1024 … 16384).  The
best sequence found before the wall‑clock deadline is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT‑based convolution / cross‑correlation utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.

    Returns a vector of length len(a) where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


# ----------------------------------------------------------------------
# Smooth‑max surrogate objective (callable for SciPy)
# ----------------------------------------------------------------------
class SmoothMaxObjective:
    """Callable returning (smooth‑max, gradient) for a given logits vector."""
    def __init__(self, n: int, fft_len_conv: int, fft_len_corr: int, lam: float):
        self.n = n
        self.fft_len_conv = fft_len_conv
        self.fft_len_corr = fft_len_corr
        self.lam = lam

    def __call__(self, logits: np.ndarray):
        # soft‑max → probability vector on the simplex
        mx = np.max(logits)
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution a*a
        conv = _full_convolution(a, self.fft_len_conv)

        # smooth‑max via stable log‑sum‑exp
        lam = self.lam
        scaled = lam * conv
        smax = np.max(scaled)
        exp_term = np.exp(scaled - smax)
        sum_exp = exp_term.sum()
        smooth_val = (smax + np.log(sum_exp)) / lam

        # gradient w.r.t. a: 2 * (soft‑max weights) ★ a
        w = exp_term / sum_exp               # shape (2n‑1)
        grad_a = 2.0 * _cross_correlation(w, a, self.fft_len_corr)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_logits = a * (grad_a - avg)

        return smooth_val, grad_logits


# ----------------------------------------------------------------------
# Exact‑max sub‑gradient polishing (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _refine_subgradient(a: np.ndarray,
                        n: int,
                        fft_len_conv: int,
                        fft_len_corr: int,
                        deadline: float) -> np.ndarray:
    """Few exponentiated‑gradient steps on the true max‑convolution."""
    best_a = a.copy()
    best_val = float("inf")
    last_improve = time.time()
    eta = 0.6

    while time.time() < deadline:
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient update (keeps a non‑negative and normalised)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s <= 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
        if time.time() - last_improve > 5.0:
            break
    return best_a


# ----------------------------------------------------------------------
# Multi‑stage L‑BFGS optimiser (smooth‑max → exact max)
# ----------------------------------------------------------------------
def _run_lbfgs(a0: np.ndarray, n: int, deadline: float) -> np.ndarray:
    """Optimise a start vector a0 with a staged L‑BFGS scheme."""
    fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fft_len_corr = 1 << int(np.ceil(np.log2(3 * n - 2)))
    logits = np.log(np.maximum(a0, 1e-16))

    # temperature schedule – larger λ makes the surrogate closer to a true max
    for lam in (0.5, 5.0, 20.0):
        if time.time() > deadline:
            break
        obj = SmoothMaxObjective(n, fft_len_conv, fft_len_corr, lam)
        # L‑BFGS on the smooth surrogate
        res = minimize(obj,
                       logits,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 80, 'disp': False})
        logits = res.x

    # recover the probability vector
    mx = np.max(logits)
    exp_log = np.exp(logits - mx)
    a = exp_log / exp_log.sum()

    # final polishing on the true max‑convolution
    a = _refine_subgradient(a, n, fft_len_conv, fft_len_corr, deadline)
    return a


# ----------------------------------------------------------------------
# Candidate factory – diverse families of start vectors
# ----------------------------------------------------------------------
def _generate_candidates(n: int, best_seq: np.ndarray) -> list:
    """Create a list of plausible start vectors for length n."""
    candidates = []

    # 1. Uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2. Warm‑start from a previous solution of the same length
    if best_seq is not None and best_seq.shape[0] == n:
        candidates.append(best_seq.copy())

    # 3. Upsampled warm‑start from a shorter solution (linear interpolation)
    if best_seq is not None and best_seq.shape[0] < n:
        old = best_seq
        x_old = np.linspace(0.0, 1.0, old.shape[0])
        x_new = np.linspace(0.0, 1.0, n)
        up = np.interp(x_new, x_old, old)
        up = np.maximum(up, 0.0)
        s = up.sum()
        up = up / s if s > 0 else np.random.dirichlet(np.ones(n))
        candidates.append(up)

    # 4. Historic global variable, if present
    hseq = globals().get("height_sequence_1")
    if hseq is not None:
        h = np.asarray(hseq, dtype=np.float64)
        if h.shape[0] > n:
            h = h[:n]
        elif h.shape[0] < n:
            h = np.pad(h, (0, n - h.shape[0]))
        h = np.maximum(h, 0.0)
        s = h.sum()
        h = h / s if s > 0 else np.random.dirichlet(np.ones(n))
        candidates.append(h)

    # 5. Dirichlet draws with a range of concentration parameters
    for alpha in (0.2, 0.5, 1.0, 2.0, 5.0):
        candidates.append(np.random.dirichlet(alpha * np.ones(n)))

    # 6. Edge‑concentrated power‑law shapes  (prod(i·(n‑i+1))^α)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-3.0, -0.2, 6):
        shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
        if shape.sum() > 0:
            candidates.append(shape / shape.sum())

    # 7. Two‑Gaussian edges
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        candidates.append(shape / shape.sum())

    # 8. Smooth sine‑squared bump
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    candidates.append(shape / shape.sum())

    # 9. Random beta‑like product (i+1)^p · (n‑i)^q
    for _ in range(4):
        p = np.random.uniform(-2.0, 2.0)
        q = np.random.uniform(-2.0, 2.0)
        i = np.arange(n, dtype=np.float64) + 1.0
        shape = np.power(i, p) * np.power((n - i + 1), q)
        shape = np.maximum(shape, 0.0)
        s = shape.sum()
        if s > 0:
            candidates.append(shape / s)

    # 10. Random piece‑wise constant (5‑10 segments)
    for _ in range(3):
        seg = np.random.randint(5, 11)
        cuts = np.sort(np.random.choice(n - 1, seg - 1, replace=False) + 1)
        values = np.random.rand(seg)
        piece = np.empty(n, dtype=np.float64)
        start = 0
        for cp, v in zip(np.append(cuts, n), values):
            piece[start:cp] = v
            start = cp
        candidates.append(piece / piece.sum())

    return candidates
