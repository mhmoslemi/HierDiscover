def _conditional_gradient_optimize(a0, n, deadline, current_best_val):
    a = a0.copy()
    best_val = float('inf')
    best_seq = a.copy()
    iter = 0
    while time.time() < deadline:
        # compute conv and M
        conv = np.convolve(a, a)
        M = conv.max()
        val = 2 * n * M / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        # Stop condition: if val not improving enough after many iterations
        if iter > 5000:
            break
        # Solve LP for g
        g = _solve_lp_for_g(a, M)
        if g is None:
            break
        # mixing weight
        t = 0.5  # constant or decaying
        # mix
        h = (1 - t) * a + t * g
        # renormalize to sum=1
        s = h.sum()
        if s < 1e-12:
            h = np.random.dirichlet(np.ones(n))
        else:
            h = h / s
        a = h
        # optionally adapt t if not improving
        iter += 1
        # maybe occasional random shake
        if iter % 100 == 0:
            # small perturbation
            a += (np.random.rand(n) - 0.5) * 0.01
            a = np.maximum(a, 0)
            a /= a.sum()
    return best_seq, best_val

def _solve_lp_for_g(f, M):
    n = len(f)
    # Build matrix A of shape (2n-1, n)
    rows = 2*n-1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_min = max(0, k-(n-1))
        i_max = min(k, n-1)
        A[k, i_min:i_max+1] = f[i_min:i_max+1]
    b = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n)  # maximize sum => min -sum
    # Use hiGHS method
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if res.success:
        g = res.x
        # normalize sum to 1 for consistency (optional)
        if g.sum() > 0:
            g = g / g.sum()
        return g
    return None

    method).  For a short vector (n≈256) we repeatedly:
        * compute M = max(a∗a)
        * solve a linear programme
              maximise ∑ g_i   s.t.    (a∗g)[k] ≤ M ,   g_i ≥ 0
          (this finds the direction that can increase the L1‑norm of
          the step‑function without worsening the worst convolution value)
        * move part‑way towards g and renormalise to sum = 1.
    The LP is tiny (≈500×256) and solved by SciPy’s HiGHS solver.
    Several diverse initial vectors (warm start, uniform, Dirichlet,
    edge‑concentrated power‑law, absolute Gaussian) are tried; the best
    small‑scale shape is kept.

2.  Upsample the best small vector to a larger grid (4096 points) by
    linear interpolation, renormalise and clip to the evaluator’s limits.

3.  Refine the large vector with a gradient‑based optimiser that works
    directly on the original objective:
        * Adam optimiser on a smooth‑max surrogate (log‑sum‑exp of the
          convolution) – this quickly reduces the peak of a∗a.
        * A short exponentiated‑gradient sub‑gradient phase on the exact
          max‑convolution (the same “sub‑grad” step used in the reference
          solution).

The whole pipeline respects the 1000 s budget and returns the best
sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the linear programme in the CG step.
try:
    from scipy.optimize import linprog
except Exception:  # pragma: no cover
    linprog = None


def _max_convolution(a: np.ndarray) -> float:
    """Exact max of the linear convolution a*a (no FFT, fine for small n)."""
    conv = np.convolve(a, a)
    return float(conv.max())


# ----------------------------------------------------------------------
# Conditional‑gradient (Frank‑Wolfe) optimiser on a small grid.
def _solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise ∑ g_i   s.t. (f∗g)[k] ≤ M   (for all k),   g_i ≥ 0.
    Returns the raw solution (not normalised).  Returns None on failure.
    """
    if linprog is None:  # SciPy not available – abort CG step.
        return None

    n = f.shape[0]
    rows = 2 * n - 1
    # Build the constraint matrix A where A[k,i] = f[i] if 0 ≤ k-i < n else 0.
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        A[k, i_min:i_max + 1] = f[i_min:i_max + 1]

    # Upper bounds for each convolution entry.
    b = np.full(rows, M, dtype=np.float64)

    # maximise sum(g)   →   minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)

    # Solve with HiGHS (fast and robust for modest sizes).
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method="highs")
    if res.success:
        return res.x
    return None


def _conditional_gradient_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    global_best_val: float,
    global_best_seq: np.ndarray | None,
) -> tuple[np.ndarray | None, float]:
    """
    Frank‑Wolfe style improvement for the max‑convolution objective.
    Operates on vectors of length `n` (normally 256).  Returns the best
    sequence discovered (or None) and its objective value.
    """
    a = a0.copy()
    best_seq_local = a.copy()
    best_val_local = float("inf")
    last_improve = time.time()
    it = 0

    while time.time() < deadline:
        # ---- compute current objective (fast, no clipping needed) ----
        M = _max_convolution(a)
        val = 2.0 * n * M / (a.sum() ** 2)

        if val < best_val_local:
            best_val_local = val
            best_seq_local = a.copy()
            last_improve = time.time()
            print(f"[CG n={n} iter={it:05d}] new local best {val:.6f}")

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            print(f"[CG] new global best {val:.6f}")

        # ---- solve LP for the ascent direction ----------------------
        g = _solve_lp_for_g(a, M)
        if g is None:
            # LP failed – abort the CG loop.
            break

        # ---- mix towards g ------------------------------------------------
        # step size schedule: start at 0.5 and decay exponentially
        t = 0.5 * (0.97 ** it)
        h_raw = (1.0 - t) * a + t * g

        # renormalise to keep the scale (objective is scale‑invariant)
        s = h_raw.sum()
        if s < 1e-12:
            h_raw = np.random.dirichlet(np.ones(n))
            s = h_raw.sum()
        a = h_raw / s

        # occasional jitter to escape plateaus
        if time.time() - last_improve > 5.0:
            jitter = (np.random.rand(n) - 0.5) * 1e-3
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
            last_improve = time.time()

        it += 1
        if it >= 2000:      # safety cap – enough for most budgets
            break

    # return the best sequence found in this run (or None)
    return (best_seq_local if best_seq_local is not None else None, best_val_local)


# ----------------------------------------------------------------------
# FFT‑based convolution utilities for the large‑scale Adam optimiser.
def _next_pow2(x: int) -> int:
    """Small utility – next power of two (≥ x)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length = 2*n‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)].
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max (log‑sum‑exp) surrogate followed by a short
    exponentiated‑gradient refinement on the exact max.  Works on the
    simplex via a soft‑max parametrisation.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # initialise soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5          # start temperature for smooth‑max
    lam_factor = 1.02

    last_improve = time.time()
    it = 0

    while time.time() < deadline:
        # ----- soft‑max → simplex ------------------------------------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- evaluate exact objective (for logging) ----------------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={it:05d}] new global best {val:.6f}")

        # ----- smooth‑max weights ------------------------------------
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate (chain‑rule) ----------------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update on logits --------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule ---------------------------------
        lam = min(lam * lam_factor, 1e6)

        # ----- occasional shake (stagnation handling) ---------------
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        if it >= 120_000:   # safety limit
            break

    # -----------------------------------------------------------------
    # Sub‑gradient phase on the exact max (exponentiated‑gradient)
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()

    eta = 0.6
    sub_last_improve = time.time()
    for sub_it in range(10_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[Subgrad n={n} iter={sub_it:05d}] new global best {val:.6f}")

        # active set of maximal entries
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 6.0:
            break

    return global_best_seq, global_best_val


# ----------------------------------------------------------------------
def _interpolate(a: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation from len(a) → target_n (periodic, no wrap)."""
    x_old = np.linspace(0.0, 1.0, a.shape[0], endpoint=False)
    x_new = np.linspace(0.0, 1.0, target_n, endpoint=False)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function minimising the evaluation
    function.  Returns the best sequence (list of floats) found within the
    allotted time.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------- small‑scale CG optimisation -----------------
    n_small = 256
    # Assemble a modest set of diverse starters.
    candidates_small = []

    # Warm start (if a previous construction exists)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] >= n_small:
            cand = w[:n_small]
        else:
            cand = np.pad(w, (0, n_small - w.shape[0]))
        cand = np.maximum(cand, 0.0)
        if cand.sum() > 0.01:
            cand = cand / cand.sum()
            candidates_small.append(cand)

    # Uniform baseline
    candidates_small.append(np.full(n_small, 1.0 / n_small))

    # Random Dirichlet
    candidates_small.append(np.random.dirichlet(np.ones(n_small)))

    # Absolute Gaussian (|N(0,1)|) – a simple “random‑positive” shape
    gauss = np.abs(np.random.randn(n_small))
    gauss /= gauss.sum()
    candidates_small.append(gauss)

    # Edge‑concentrated power‑law shapes
    idx = np.arange(1, n_small + 1, dtype=np.float64)
    prod = idx * (n_small - idx + 1.0)
    for alpha in (-2.0, -1.5, -1.0, -0.5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates_small.append(shape)

    # Two‑Gaussian bumps near the edges
    i = np.arange(n_small, dtype=np.float64)
    for gamma in (0.2, 0.5, 1.0):
        bump = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n_small - 1 - i) ** 2))
        bump /= bump.sum()
        candidates_small.append(bump)

    # --------------------------------------------------------------
    # Run the conditional‑gradient optimiser on each candidate.
    for cid, cand in enumerate(candidates_small):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        s = a0.sum()
        if s < 0.01:
            a0 = np.random.dirichlet(np.ones(n_small))
        else:
            a0 = a0 / s

        seq, val = _conditional_gradient_optimize(
            a0,
            n_small,
            deadline,
            best_val,
            best_seq,
        )
        if seq is not None and val < best_val:
            best_val = val
            best_seq = seq
            print(f"[main] CG candidate {cid} improved global best to {val:.6f}")

    # ------------------- upscale & large‑scale refinement -------------
    if best_seq is None:
        best_seq = np.full(n_small, 1.0 / n_small)

    # First upscale to a medium size (4096) – provides a good balance.
    target_n = 4096
    a_big = _interpolate(best_seq, target_n)

    # Pre‑compute FFT lengths for the large optimiser.
    conv_fft_len = _next_pow2(2 * target_n - 1)
    corr_fft_len = _next_pow2(3 * target_n - 2)

    # Run the Adam + sub‑gradient refinement (respecting the deadline).
    best_seq, best_val = _deep_optimize(
        a_big,
        target_n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )

    # Optional: if a bit of time remains, try an even larger grid.
    if time.time() + 8 < deadline:
        n_big = 8192
        if best_seq is not None:
            a_big2 = _interpolate(best_seq, n_big)
        else:
            a_big2 = np.random.dirichlet(np.ones(n_big))
        conv_fft_big = _next_pow2(2 * n_big - 1)
        corr_fft_big = _next_pow2(3 * n_big - 2)
        best_seq, best_val = _deep_optimize(
            a_big2,
            n_big,
            conv_fft_big,
            corr_fft_big,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------- final sanity & return -----------------------
    if best_seq is None:
        # Fallback – uniform vector of a reasonable size.
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip according to the evaluator’s safety checks.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return best_seq.tolist()
