"""
Propose a non‑negative step‑function (probability vector) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)^2)

The implementation combines three ideas that were not all used together in the
reference code:

1. **Analytic initialisations** – the coefficients of the power series of
   (1‑z)^‑½ (i.e. a_i = C(2i,i)/4^i) and the simple power‑law
   a_i ∝ (i+1)^‑½ are known to give a convolution that is almost flat,
   i.e. an objective ≈ π/2≈1.57.  Starting from these points gives a much
   better basin of attraction than a purely random start.

2. **Smooth‑max Adam optimisation** – we keep the probability simplex by
   parametrising the vector with logits x (soft‑max).  The true max of the
   convolution is replaced by a temperature‑controlled log‑sum‑exp
   (smooth‑max).  The temperature is annealed very slowly (linear schedule)
   so the landscape stays smooth for a long time, helping to escape local
   minima.

3. **Greedy mass‑shuffling refinement** – after the Adam phase we perform a
   cheap local search that repeatedly moves a tiny amount of probability mass
   from one coordinate to another if it reduces the objective.  This often
   squeezes a few more percent out of the solution.

The search runs for the whole wall‑time (default 1000 s).  Several problem
sizes are tried (256, 512, 1024, 2048, 4096) – larger n gives a lower
theoretical bound, but each size costs more time, so we allocate restarts
according to the available budget.  The best sequence found (as a plain
Python list) is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are top‑level as required
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def smooth_max_and_weights(conv: np.ndarray, tau: float):
    """
    Compute a temperature‑smoothed max of `conv` and the corresponding
    softmax weights (∂smooth_max/∂conv).
    """
    c_max = conv.max()
    shifted = (conv - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    smooth = c_max + tau * np.log(sum_exp)
    weights = exp_shifted / sum_exp
    return smooth, weights


def series_sequence_sqrtinv(n: int) -> np.ndarray:
    """
    Coefficients of (1‑z)^‑½ :
        a_i = binom(2i,i) / 4**i  =  product_{j=0}^{i-1} (2j+1)/(2j+2)
    The sequence is decreasing like i⁻½.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1) / (2.0 * i)
    return a


def powerlaw_sequence(n: int, p: float = 0.5) -> np.ndarray:
    """
    a_i ∝ (i+1)^{-p}.  p=0.5 gives the classic x⁻½ shape.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return a


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Used only for sanity checks – the main optimisation works in log‑space.
    """
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Core optimisation routine for a single length
# ----------------------------------------------------------------------


def optimise_one_length(
    n: int,
    deadline: float,
    init_candidates: list[np.ndarray],
    max_total_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Run Adam optimisation (smooth max) for a given length ``n`` using a list
    of initial candidates.  Returns the best sequence (sum=1) and its value.
    """
    best_seq = None
    best_val = float("inf")

    fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))  # safe upper bound

    # hyper‑parameters (same for all candidates, but we keep them mutable)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr_initial = 0.15
    decay_every = 4000
    tau_initial = 0.3
    tau_final = 2e-7

    for cand_id, a0 in enumerate(init_candidates):
        if time.time() > deadline:
            break

        # Normalise to a probability vector
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 1e-12:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # Log‑space parameters (soft‑max)
        x = np.log(a0 + 1e-15)

        # Adam state
        m = np.zeros_like(x)
        v = np.zeros_like(x)

        lr = lr_initial
        t = 0
        last_improve = time.time()

        while time.time() < deadline and t < max_total_iters:
            # ---- probability vector ------------------------------------------------
            max_x = np.max(x)
            exp_x = np.exp(x - max_x)
            a = exp_x / exp_x.sum()               # Σa = 1, a ≥ 0

            # ---- convolution & smooth‑max -----------------------------------------
            conv = full_convolution(a, fft_len=fft_len_conv)

            # anneal temperature linearly
            tau = max(tau_final, tau_initial * (1.0 - t / max_total_iters))
            _, weights = smooth_max_and_weights(conv, tau)

            cur_val = 2.0 * n * conv.max()        # exact objective (sum=1)

            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_seq = a.copy()
                last_improve = time.time()
                print(
                    f"[n={n} cand={cand_id} iter={t:06d}] "
                    f"new best {cur_val:.6f}"
                )

            # ---- gradient of smooth‑max w.r.t. a -----------------------------------
            grad_a = 2.0 * cross_correlation(weights, a, fft_len=fft_len_corr)

            # back‑prop through soft‑max (the usual “log‑soft‑max” Jacobian)
            inner = np.dot(a, grad_a)
            grad_x = a * (grad_a - inner)

            # ---- Adam update -------------------------------------------------------
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

            # learning‑rate decay
            if t % decay_every == 0:
                lr *= 0.7

            # ---- occasional shake to avoid stagnation -------------------------------
            if time.time() - last_improve > 30.0:
                x += (np.random.rand(n) - 0.5) * 4e-3
                last_improve = time.time()
                print(f"[n={n} cand={cand_id} iter={t}] shake")

        # ----------------------------------------------------------------------
        # final fine‑tuning at the tiniest temperature (true max)
        # ----------------------------------------------------------------------
        if best_seq is not None and best_seq.shape[0] == n:
            a = best_seq.copy()
            x = np.log(a + 1e-15)
            m = np.zeros_like(x)
            v = np.zeros_like(x)
            lr = 0.02
            fine_iters = 8000
            for ft in range(fine_iters):
                if time.time() > deadline:
                    break
                a = np.exp(x - np.max(x))
                a = a / a.sum()
                conv = full_convolution(a, fft_len=fft_len_conv)
                cur_val = 2.0 * n * conv.max()
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[n={n} fine] improved to {cur_val:.6f}")

                # gradient with true max (tau ~ 0)
                _, weights = smooth_max_and_weights(conv, tau_final)
                grad_a = 2.0 * cross_correlation(weights, a, fft_len=fft_len_corr)
                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                # Adam step
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** (ft + 1))
                v_hat = v / (1.0 - beta2 ** (ft + 1))
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----------------------------------------------------------------------
        # cheap greedy refinement – mass shuffling
        # ----------------------------------------------------------------------
        if best_seq is not None and best_seq.shape[0] == n:
            a_local = best_seq.copy()
            local_iters = 4000
            for it in range(local_iters):
                if time.time() > deadline:
                    break
                i, j = np.random.choice(n, 2, replace=False)
                move = min(a_local[i], 0.006) * (np.random.rand() * 0.6 + 0.2)
                if move <= 0:
                    continue
                a_new = a_local.copy()
                a_new[i] -= move
                a_new[j] += move
                conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                val_new = 2.0 * n * conv_new.max()
                if val_new < best_val - 1e-12:
                    best_val = val_new
                    best_seq = a_new.copy()
                    a_local = a_new
                    print(f"[n={n} refine] improved to {val_new:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000.0) -> list:
    """
    Time‑bounded optimisation that returns the best non‑negative sequence
    found so far.  The search uses a mixture of analytical starts,
    Adam optimisation with a smooth‑max surrogate, and a final greedy
    mass‑shuffling refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # problem sizes – larger ``n`` gives a lower theoretical bound.
    # We start from small to medium lengths (fast cheap passes) and finish
    # with the biggest sizes that still fit into the remaining budget.
    lengths = [256, 512, 1024, 2048, 4096]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # ----- Build a list of initial candidates --------------------------------
        init = []

        # 1. Analytic series (coefficients of (1‑z)^‑½)
        init.append(series_sequence_sqrtinv(n))

        # 2. Power‑law i⁻½ (exact discrete version)
        init.append(powerlaw_sequence(n, p=0.5))

        # 3. A few random Dirichlet draws
        for _ in range(2):
            init.append(np.random.dirichlet(np.ones(n)))

        # 4. Warm‑start from the globally best sequence seen so far (if length matches)
        if best_seq is not None and len(best_seq) == n:
            init.append(np.array(best_seq, dtype=np.float64))

        # 5. Warm‑start from the historic variable ``height_sequence_1`` (if present)
        if 'height_sequence_1' in globals():
            cand = np.array(height_sequence_1, dtype=np.float64)
            if cand.shape[0] >= n:
                cand = cand[:n]
            else:
                cand = np.pad(cand, (0, n - cand.shape[0]))
            init.append(cand)

        # Run optimisation for this length
        seq_n, val_n = optimise_one_length(
            n,
            deadline,
            init_candidates=init,
            max_total_iters=140_000 if n <= 1024 else 100_000,
        )
        if seq_n is not None and val_n < best_val:
            best_val = val_n
            best_seq = seq_n.copy()
            print(f"\n>>> New global best after length {n}: {best_val:.6f}")

    # -------------------------------------------------------------------------
    # Fallback – uniform vector (objective = 2) – should never be used
    # -------------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = 2.0

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
