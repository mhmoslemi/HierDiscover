"""
Proposes a step‑function that minimizes 
    F(a) = 2·n·max(a⋆a)/(∑a)²

Approach
---------
1. Build a diverse set of initial non‑negative vectors (uniform,
   sinusoidal, power‑law, monotone, Beta‑shapes, warm‑start).
2. For each candidate run a continuation scheme:
   - Minimize a smooth‑max surrogate 𝑆_λ(a)= (1/λ)·log∑exp(λ·(a⋆a))
     via L‑BFGS‑B on the soft‑max parametrisation (logits).
   - λ is increased geometrically (0.5 → 1e5); the solution of one
     stage is warm‑started for the next.
3. After the continuation, perform a short exponentiated‑gradient
   “exact‑max” refinement to drive the true max‑convolution down.
4. Keep the best sequence seen across all candidates and both lengths
   (4096 and 8192). The optimisation respects the 1000 s time budget
   and returns the best sequence as a plain Python list.
"""

import time
import numpy as np
# optional SciPy for deterministic L‑BFGS‑B; fall back to Adam if unavailable
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _softmax(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = logits.max()
    exp_log = np.exp(logits - mx)
    return exp_log / exp_log.sum()


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a pool of diverse start vectors (including a warm start)."""
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        # rotate the warm start a few times – useful if the optimum is not
        # translation invariant (the problem *is* shift invariant, but we keep it)
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # power‑law edge‑concentrated shapes (already used in the reference code)
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # sinusoidal positive half‑wave
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # monotone decreasing power‑law
    for p in (0.5, 1.0, 2.0, 3.0):
        shape = (np.arange(n, 0, -1, dtype=np.float64) ** p)
        shape /= shape.sum()
        cand.append(shape)

    # triangular shape
    tri = np.arange(1, n // 2 + 1, dtype=np.float64)
    if n % 2 == 0:
        tri = np.concatenate([tri, tri[::-1]])
    else:
        tri = np.concatenate([tri, [n // 2 + 1], tri[::-1]])
    tri /= tri.sum()
    cand.append(tri)

    # Beta‑type shapes (a_i ∝ x^{α‑1}(1‑x)^{β‑1})
    x = np.linspace(0, 1, n, dtype=np.float64)
    for alpha, beta in [(0.5, 0.5), (1.0, 1.0), (2.0, 2.0), (2.0, 1.0), (1.0, 2.0)]:
        shape = (np.power(x, max(alpha - 1, 0)) *
                 np.power(1 - x, max(beta - 1, 0)) + 1e-12)
        shape /= shape.sum()
        cand.append(shape)

    # uniform baseline
    cand.append(np.full(n, 1.0 / n))

    return cand


# ----------------------------------------------------------------------
def _adam_fallback(
    logits: np.ndarray,
    lam: float,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    max_iter: int = 200,
) -> np.ndarray:
    """
    Very small Adam loop used only when SciPy is unavailable.
    It optimises the smooth‑max surrogate for a fixed λ.
    """
    eps = 1e-15
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    for it in range(max_iter):
        a = _softmax(logits)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
    return logits

def _optimize_candidate(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray | None, float]:
    """
    Optimises a single starting vector.
    Uses a continuation of the smooth‑max surrogate (L‑BFGS‑B when SciPy
    is present, otherwise a few Adam steps) followed by a short
    exponentiated‑gradient refinement.
    Returns the (possibly) updated global best sequence/value.
    """
    eps = 1e-15
    # initialise logits for the soft‑max parametrisation
    logits = np.log(np.maximum(a0, eps))

    # schedule of λ values (smooth‑max temperature)
    lam_schedule = [
        0.5,
        1.0,
        2.0,
        5.0,
        10.0,
        20.0,
        50.0,
        100.0,
        200.0,
        500.0,
        1000.0,
        2000.0,
        5000.0,
        1e4,
        2e4,
        5e4,
        1e5,
    ]

    for lam in lam_schedule:
        if time.time() >= deadline:
            break

        # ----- smooth‑max surrogate -----
        if _opt is not None:
            # use SciPy L‑BFGS‑B; we request both value and gradient
            def _fun_and_grad(logits_vec, lam_val=lam):
                a = _softmax(logits_vec)
                conv = _full_convolution(a, conv_fft_len)
                max_c = conv.max()
                w = np.exp(lam_val * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0 or np.isnan(w_sum):
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # smooth‑max value (log‑sum‑exp)
                smooth_val = max_c + np.log(w_sum) / lam_val

                # gradient wrt a
                grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)
                return smooth_val, grad_log

            res = _opt.minimize(
                fun=_fun_and_grad,
                x0=logits,
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": 30, "ftol": 1e-12, "disp": False},
            )
            logits = res.x
        else:
            logits = _adam_fallback(logits, lam, n, conv_fft_len, corr_fft_len, max_iter=200)

        # ----- evaluate the true objective -----
        a = _softmax(logits)
        # evaluate via the provided black‑box (fast enough)
        try:
            val = evaluate_sequence(a.tolist())
        except Exception:  # pragma: no cover
            # fallback manual computation (identical to evaluate_sequence)
            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()
            sum_a = a.sum()
            if sum_a < 0.01:
                val = float("inf")
            else:
                val = float(2.0 * n * max_c / (sum_a ** 2))
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            print(f"[n={n} λ={lam:.0e}] new best {val:.6f}")

    # ----- exact‑max sub‑gradient refinement -----
    if time.time() < deadline:
        a = global_best_seq if global_best_seq is not None else a0.copy()
        a = np.maximum(a, 0.0)
        a /= a.sum()
        eta = 0.6
        sub_last_improve = time.time()
        for sub_it in range(8000):
            if time.time() >= deadline:
                break
            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()
            val = float(2.0 * n * max_c / (a.sum() ** 2))
            if val < global_best_val:
                global_best_val = val
                global_best_seq = a.copy()
                sub_last_improve = time.time()
                print(f"[n={n} subgrad it={sub_it:04d}] new best {val:.6f}")

            active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
            if active.any():
                w = np.zeros_like(conv)
                w[active] = 1.0 / active.sum()
            else:
                w = np.zeros_like(conv)

            grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
            a = a * np.exp(-eta * grad_a)
            s = a.sum()
            if s < 1e-12:
                a = np.random.dirichlet(np.ones(n))
                s = a.sum()
            a /= s

            if time.time() - sub_last_improve > 10.0:
                break

    return global_best_seq, global_best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    score within the given time budget.  Returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Try the larger grid first – more degrees of freedom often give a
    # lower constant c in 2·n·max(conv) = 2·c.
    for n in (8192, 4096):
        if time.time() >= deadline:
            break

        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        # Warm‑start coming from the best sequence seen so far (if any)
        warm = None
        if best_seq is not None:
            if best_seq.shape[0] == n:
                warm = best_seq.copy()
            else:
                # linear interpolation / resampling
                x_old = np.linspace(0, 1, best_seq.shape[0], endpoint=False)
                x_new = np.linspace(0, 1, n, endpoint=False)
                warm = np.interp(x_new, x_old, best_seq)
                warm = np.maximum(warm, 0.0)
                if warm.sum() > 0:
                    warm = warm / warm.sum()
                else:
                    warm = None

        candidates = _initial_candidates(n, warm)

        # Limit the number of candidates loosely by the remaining time.
        # The most promising ones (warm start and a few randoms) are tried
        # first; we stop when the deadline is reached.
        for idx, cand in enumerate(candidates):
            if time.time() >= deadline:
                break

            a0 = np.maximum(cand, 0.0)
            s = a0.sum()
            if s < 0.01:
                a0 = np.random.dirichlet(np.ones(n))
            else:
                a0 = a0 / s

            best_seq, best_val = _optimize_candidate(
                a0,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                best_seq,
                best_val,
            )
            print(
                f"[n={n} candidate={idx}/{len(candidates)}] current best {best_val:.6f}"
            )

    # ------------------------------------------------------------------
    # Fallback – if nothing was produced create a uniform vector.
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return best_seq.tolist()

        cand.append(shape)

    # positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # monotone decreasing power‑law shapes
    for p in (0.5, 1.0, 2.0, 3.0):
        shape = (np.arange(n, 0, -1, dtype=np.float64) ** p)
        shape /= shape.sum()
        cand.append(shape)

    # triangular shape
    tri = np.arange(1, n // 2 + 1, dtype=np.float64)
    if n % 2 == 0:
        tri = np.concatenate([tri, tri[::-1]])
    else:
        tri = np.concatenate([tri, [n // 2 + 1], tri[::-1]])
    tri /= tri.sum()
    cand.append(tri)

    # Beta‑type shapes (a_i ∝ x^{α‑1}(1‑x)^{β‑1})
    x = np.linspace(0, 1, n, dtype=np.float64)
    for alpha, beta in [(0.5, 0.5), (1.0, 1.0), (2.0, 2.0), (2.0, 1.0), (1.0, 2.0)]:
        shape = (np.power(x, max(alpha - 1, 0)) *
                 np.power(1 - x, max(beta - 1, 0)) + 1e-12)
        shape /= shape.sum()
        cand.append(shape)

    # uniform baseline
    cand.append(np.full(n, 1.0 / n))

    return cand


# ----------------------------------------------------------------------
def _adam_fallback(
    logits: np.ndarray,
    lam: float,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    max_iter: int = 200,
) -> np.ndarray:
    """
    Simple Adam optimiser (used when SciPy is unavailable).
    Minimises the smooth‑max surrogate for a fixed λ.
    """
    eps = 1e-15
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    for it in range(max_iter):
        a = _softmax(logits)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
    return logits


# ----------------------------------------------------------------------
def _optimize_candidate(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray | None, float]:
    """
    Optimises a single starting vector.
    Uses a continuation on the smooth‑max surrogate (L‑BFGS‑B when
    SciPy is present, otherwise a few Adam steps) followed by a short
    exponentiated‑gradient refinement.
    Returns the updated global best sequence/value.
    """
    eps = 1e-15
    logits = np.log(np.maximum(a0, eps))

    # λ schedule – from very smooth to virtually exact max
    lam_schedule = [
        0.5,
        1.0,
        2.0,
        5.0,
        10.0,
        20.0,
        50.0,
        100.0,
        200.0,
        500.0,
        1000.0,
        2000.0,
        5000.0,
        1e4,
        2e4,
        5e4,
        1e5,
    ]

    for lam in lam_schedule:
        if time.time() >= deadline:
            break

        if _opt is not None:
            # L‑BFGS‑B on the smooth‑max surrogate
            def _fun_and_grad(logits_vec, lam_val=lam):
                a = _softmax(logits_vec)
                conv = _full_convolution(a, conv_fft_len)
                max_c = conv.max()
                w = np.exp(lam_val * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0 or np.isnan(w_sum):
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum
                # smooth‑max value
                smooth_val = max_c + np.log(w_sum) / lam_val
                # gradient wrt a
                grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)
                return smooth_val, grad_log

            res = _opt.minimize(
                fun=_fun_and_grad,
                x0=logits,
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": 30, "ftol": 1e-12, "disp": False},
            )
            logits = res.x
        else:
            logits = _adam_fallback(logits, lam, n, conv_fft_len, corr_fft_len, max_iter=200)

        # Evaluate the true objective (expensive but infrequent)
        a = _softmax(logits)
        try:
            val = evaluate_sequence(a.tolist())
        except Exception:  # pragma: no cover
            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()
            sum_a = a.sum()
            val = float("inf") if sum_a < 0.01 else float(2.0 * n * max_c / (sum_a ** 2))

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            print(f"[n={n} λ={lam:.0e}] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Exact‑max sub‑gradient refinement (exponentiated‑gradient)
    if time.time() < deadline:
        a = global_best_seq if global_best_seq is not None else a0.copy()
        a = np.maximum(a, 0.0)
        a /= a.sum()
        eta = 0.6
        sub_last_improve = time.time()
        for sub_it in range(8000):
            if time.time() >= deadline:
                break
            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()
            val = float(2.0 * n * max_c / (a.sum() ** 2))
            if val < global_best_val:
                global_best_val = val
                global_best_seq = a.copy()
                sub_last_improve = time.time()
                print(f"[n={n} subgrad it={sub_it:04d}] new best {val:.6f}")

            active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
            if active.any():
                w = np.zeros_like(conv)
                w[active] = 1.0 / active.sum()
            else:
                w = np.zeros_like(conv)

            grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
            a = a * np.exp(-eta * grad_a)
            s = a.sum()
            if s < 1e-12:
                a = np.random.dirichlet(np.ones(n))
                s = a.sum()
            a /= s

            if time.time() - sub_last_improve > 10.0:
                break

    return global_best_seq, global_best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Searches for a non‑negative step‑function that minimises the
    evaluation score within the given time budget.  Returns the best
    sequence found as a list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # Try the larger grid first; extra degrees of freedom often give a
    # smaller constant in the objective.
    for n in (8192, 4096):
        if time.time() >= deadline:
            break

        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        # Prepare a warm start derived from the best solution seen so far
        warm = None
        if best_seq is not None:
            if best_seq.shape[0] == n:
                warm = best_seq.copy()
            else:
                x_old = np.linspace(0, 1, best_seq.shape[0], endpoint=False)
                x_new = np.linspace(0, 1, n, endpoint=False)
                warm = np.interp(x_new, x_old, best_seq)
                warm = np.maximum(warm, 0.0)
                if warm.sum() > 0:
                    warm = warm / warm.sum()
                else:
                    warm = None

        candidates = _initial_candidates(n, warm)

        for idx, cand in enumerate(candidates):
            if time.time() >= deadline:
                break

            a0 = np.maximum(cand, 0.0)
            s = a0.sum()
            if s < 0.01:
                a0 = np.random.dirichlet(np.ones(n))
            else:
                a0 = a0 / s

            best_seq, best_val = _optimize_candidate(
                a0,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                best_seq,
                best_val,
            )
            print(f"[n={n} cand={idx}/{len(candidates)}] best so far {best_val:.6f}")

    # Fallback – if nothing succeeded create a uniform vector.
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed domain (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return best_seq.tolist()
