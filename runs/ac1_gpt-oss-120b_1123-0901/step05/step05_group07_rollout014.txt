def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, numpy as np

    np.random.seed(seed)

    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_val = float('inf')

    # Use lengths up to 8192 if time.
    lengths = [1024, 2048, 4096, 8192]

    # adapt restarts per n
    for n in lengths:
        if time.time() > deadline: break
        print(f"=== Optimising length n={n} ===")
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))  # >= (2n-1)+(n-1)

        # number of restarts:
        restarts = 5 if n <= 2048 else 3
        if n >= 4096:
            restarts = 2

        for r in range(restarts):
            if time.time()>deadline: break

            # initial candidate
            # attempt warm start from global best_seq if same length.
            if best_seq is not None and best_seq.shape[0] == n:
                a0 = best_seq.copy()
                a0 = a0 / a0.sum()
                x = np.log(a0 + 1e-15)
                print(f"[n={n} restart={r}] warm start")
            else:
                # try to use global height_sequence_1 if present
                if 'height_sequence_1' in globals():
                    cand = np.array(height_sequence_1, dtype=np.float64)
                    if cand.shape[0] >= n:
                        cand = cand[:n]
                    else:
                        cand = np.pad(cand, (0, n-cand.shape[0]))
                    cand = np.maximum(cand, 0.0)
                    if cand.sum() < 0.01:
                        cand = np.random.dirichlet(np.ones(n))
                    else:
                        cand = cand / cand.sum()
                    x = np.log(cand + 1e-15)
                    print(f"[n={n} restart={r}] start from height_sequence_1")
                else:
                    a0 = np.random.dirichlet(np.ones(n))
                    x = np.log(a0 + 1e-15)
                    print(f"[n={n} restart={r}] random start")

            # Adam state
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1=0.9
            beta2=0.999
            eps=1e-8
            lr=0.15
            decay_every = 4000

            # Stage 1: smoothed max
            max_iters_stage1 = 80000 if n <= 2048 else 60000
            # temperature schedule
            tau_initial = 0.12
            tau_final = 5e-7

            last_improve = time.time()
            t = 0
            while time.time()<deadline and t < max_iters_stage1:
                # softmax a
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()

                # conv
                conv = full_convolution(a, fft_len=fft_len_conv)

                # temperature
                tau = max(tau_final,
                          tau_initial * (1.0 - t / max_iters_stage1))
                smooth, weights = smooth_max_and_weights(conv, tau)

                cur_val = 2.0 * n * conv.max()
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} r={r} iter={t}] new best {cur_val:.6f}")

                # gradient wrt a
                grad_a = 2.0 * cross_correlation(weights, a, fft_len=fft_len_corr)

                # backprop to logits
                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                # Adam update
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                if t % decay_every == 0:
                    lr *= 0.7

                # stagnation shake
                if time.time() - last_improve > 30.0:
                    x += (np.random.rand(n) - 0.5) * 5e-3
                    last_improve = time.time()
                    print(f"[n={n} r={r} iter={t}] shake due to stagnation")
            # Stage 2: exact max gradient refinement
            max_iters_stage2 = 20000 if n <= 2048 else 12000
            lr2 = 0.08
            t2 = 0
            last_improve2 = time.time()
            while time.time() < deadline and t2 < max_iters_stage2:
                # softmax
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()
                conv = full_convolution(a, fft_len=fft_len_conv)
                M = conv.max()
                # mask
                mask = (conv >= M - 1e-12).astype(np.float64)

                cur_val = 2.0 * n * M
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_seq = a.copy()
                    last_improve2 = time.time()
                    print(f"[n={n} r={r} stage2 iter={t2}] new best {cur_val:.6f}")

                # gradient via exact mask
                grad_a = 2.0 * cross_correlation(mask, a, fft_len=fft_len_corr)

                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                t2 += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** t2)
                v_hat = v / (1.0 - beta2 ** t2)
                x = x - lr2 * m_hat / (np.sqrt(v_hat) + eps)

                if t2 % decay_every == 0:
                    lr2 *= 0.7

                if time.time() - last_improve2 > 25.0:
                    x += (np.random.rand(n) - 0.5) * 4e-3
                    last_improve2 = time.time()
                    print(f"[n={n} r={r} stage2 iter={t2}] shake")
            # End of refinement loops
            # local greedy refinement (mass shuffle)
            if best_seq is not None and best_seq.shape[0]==n:
                a_local = best_seq.copy()
                local_iters = 5000
                for it in range(local_iters):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    # amount to move: up to a_local[i]
                    move = min(a_local[i], 0.01) * (np.random.rand()*0.6 + 0.2)
                    if move <= 0: continue
                    a_new = a_local.copy()
                    a_new[i] -= move
                    a_new[j] += move
                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = 2.0*n*conv_new.max()
                    if val_new < best_val - 1e-12:
                        best_val = val_new
                        best_seq = a_new.copy()
                        a_local = a_new
                        print(f"[local] improved to {val_new:.6f} at iter {it}")

    # final fallback
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0/n_fallback, dtype=np.float64)

    # Ensure list and non-negative
    best_seq = np.maximum(best_seq, 0.0)
    # Rescale to have sum >=0.01 (should be ~1)
    total = best_seq.sum()
    if total < 0.01:
        best_seq = np.full_like(best_seq, 0.01/len(best_seq))
        total = best_seq.sum()
    # Normalise to sum=1 to keep objective same
    best_seq = best_seq / total

    print(f"\nFinal best value (approx): {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()

"""
Propose a non‑negative step‑function (probability vector) that minimizes

        F(a) = 2·n·max(a*a) / (sum(a)^2)

The search combines three ideas:

1.  Adam optimisation on a soft‑max parametrisation with a
    temperature‑annealed smooth‑max (as in the reference code).  The
    temperature is driven to the true max for the later part of the run.

2.  A second Adam phase that uses the *exact* max‑convolution gradient
    (mask = {conv == max}) – this gives a much sharper descent when we
    are close to a local optimum.

3.  A cheap greedy / guided local refinement that moves a small amount
    of mass from indices with large gradient to those with negative
    gradient.  This often captures the last few percent of improvement.

The routine respects a global wall‑time budget (default 1000 s) and
automatically stops when the budget is exhausted.  It returns the best
sequence found (as a plain Python list).  The implementation works for
any length; we try a few lengths (1024, 2048, 4096, 8192) and keep the
overall best solution.
"""

import time
import numpy as np


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def smooth_max_and_weights(conv: np.ndarray, tau: float):
    """
    Compute a temperature‑smoothed max of `conv` and the corresponding
    softmax weights (∂smooth_max/∂conv).
    """
    c_max = conv.max()
    shifted = (conv - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    smooth = c_max + tau * np.log(sum_exp)
    weights = exp_shifted / sum_exp
    return smooth, weights


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list:
    """
    Time‑bounded optimisation that returns the best non‑negative sequence
    found so far.  The algorithm follows the three‑phase strategy described
    in the module docstring.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin

    best_seq = None          # best sequence (numpy array, sum = 1)
    best_val = float("inf")  # best objective 2·n·max(conv)

    # lengths to explore – larger n give more freedom but cost more time
    lengths = [1024, 2048, 4096, 8192]

    # ---------------------------------------------------------------------
    # Helper to obtain a starting vector of length n
    # ---------------------------------------------------------------------
    def get_starting_vector(n):
        # 1) warm start from the global best if it has the same length
        if best_seq is not None and best_seq.shape[0] == n:
            a0 = best_seq.copy()
            a0 /= a0.sum()
            return np.log(a0 + 1e-15)

        # 2) use the global height_sequence_1 if available (truncate / pad)
        if "height_sequence_1" in globals():
            cand = np.array(globals()["height_sequence_1"], dtype=np.float64)
            if cand.shape[0] >= n:
                cand = cand[:n]
            else:
                cand = np.pad(cand, (0, n - cand.shape[0]))
            cand = np.maximum(cand, 0.0)
            if cand.sum() < 0.01:
                cand = np.random.dirichlet(np.ones(n))
            else:
                cand /= cand.sum()
            return np.log(cand + 1e-15)

        # 3) pure random Dirichlet
        a0 = np.random.dirichlet(np.ones(n))
        return np.log(a0 + 1e-15)

    # ---------------------------------------------------------------------
    # Main optimisation loops
    # ---------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))  # ≥ (2n‑1)+n‑1

        # Number of restarts – more for the cheaper sizes
        if n <= 1024:
            restarts = 4
        elif n <= 2048:
            restarts = 3
        elif n <= 4096:
            restarts = 2
        else:
            restarts = 1

        for r in range(restarts):
            if time.time() > deadline:
                break

            # ---------- initialise logits ----------
            x = get_starting_vector(n)
            print(f"[n={n} restart={r}] initialised logits")

            # ---------- Adam state ----------
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.15                       # initial learning rate
            decay_every = 5000

            # ---------- stage 1 : smoothed‑max Adam ----------
            # very generous iteration budget – the loop will stop early
            max_iters_stage1 = 250000 if n <= 1024 else 150000 if n <= 2048 else 120000
            tau_initial = 0.12
            tau_final = 5e-7
            t = 0
            last_improve = time.time()

            while time.time() < deadline and t < max_iters_stage1:
                # soft‑max to obtain a probability vector (Σa = 1)
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()

                # convolution
                conv = full_convolution(a, fft_len=fft_len_conv)

                # temperature annealing
                tau = max(tau_final,
                          tau_initial * (1.0 - t / max_iters_stage1))
                _, weights = smooth_max_and_weights(conv, tau)

                cur_val = 2.0 * n * conv.max()
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} r={r} it={t}] new best {cur_val:.6f}")

                # gradient of smooth‑max (w.r.t a)
                grad_a = 2.0 * cross_correlation(weights, a, fft_len=fft_len_corr)

                # back‑propagation through the soft‑max
                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                # Adam update
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                if t % decay_every == 0:
                    lr *= 0.7

                # shake if stagnating
                if time.time() - last_improve > 30.0:
                    x += (np.random.rand(n) - 0.5) * 6e-3
                    last_improve = time.time()
                    print(f"[n={n} r={r} it={t}] shake (stagnation)")

            # ---------- stage 2 : exact‑max gradient ----------
            max_iters_stage2 = 60000 if n <= 1024 else 35000
            lr2 = 0.08
            t2 = 0
            last_improve2 = time.time()

            while time.time() < deadline and t2 < max_iters_stage2:
                # soft‑max
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()

                conv = full_convolution(a, fft_len=fft_len_conv)
                M = conv.max()
                # indicator of the active constraints (ties are allowed)
                mask = (conv >= M - 1e-12).astype(np.float64)

                cur_val = 2.0 * n * M
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_seq = a.copy()
                    last_improve2 = time.time()
                    print(f"[n={n} r={r} stage2 it={t2}] new best {cur_val:.6f}")

                grad_a = 2.0 * cross_correlation(mask, a, fft_len=fft_len_corr)

                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                t2 += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** t2)
                v_hat = v / (1.0 - beta2 ** t2)
                x = x - lr2 * m_hat / (np.sqrt(v_hat) + eps)

                if t2 % decay_every == 0:
                    lr2 *= 0.7

                if time.time() - last_improve2 > 25.0:
                    x += (np.random.rand(n) - 0.5) * 5e-3
                    last_improve2 = time.time()
                    print(f"[n={n} r={r} stage2 it={t2}] shake")

            # ---------- stage 3 : guided local mass shuffling ----------
            if best_seq is not None and best_seq.shape[0] == n:
                a_local = best_seq.copy()
                local_iters = 3000
                for it in range(local_iters):
                    if time.time() > deadline:
                        break
                    # Compute gradient for the current local vector (exact max)
                    conv = full_convolution(a_local, fft_len=fft_len_conv)
                    M = conv.max()
                    mask = (conv >= M - 1e-12).astype(np.float64)
                    grad_a = 2.0 * cross_correlation(mask, a_local, fft_len=fft_len_corr)

                    # Choose a source index among the strongest positive gradients
                    # and a sink among the most negative gradients.
                    pos_thr = np.percentile(grad_a, 85)
                    neg_thr = np.percentile(grad_a, 15)
                    pos_cands = np.where(grad_a > pos_thr)[0]
                    neg_cands = np.where(grad_a < neg_thr)[0]

                    if pos_cands.size == 0 or neg_cands.size == 0:
                        i, j = np.random.choice(n, 2, replace=False)
                    else:
                        i = np.random.choice(pos_cands)
                        j = np.random.choice(neg_cands)

                    # Move a modest fraction of mass from i to j
                    max_move = min(a_local[i], 0.009)
                    eps_move = max_move * (0.2 + 0.6 * np.random.rand())
                    if eps_move <= 0:
                        continue
                    a_new = a_local.copy()
                    a_new[i] -= eps_move
                    a_new[j] += eps_move

                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = 2.0 * n * conv_new.max()
                    if val_new < best_val - 1e-12:
                        best_val = val_new
                        best_seq = a_new.copy()
                        a_local = a_new
                        print(f"[n={n} r={r} local] improved to {val_new:.6f}")

    # ---------------------------------------------------------------------
    # Fallback in the very unlikely case we found nothing useful
    # ---------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # ---------------------------------------------------------------------
    # Normalise output (the objective is homogeneous, so any positive scaling
    # is fine – we keep sum = 1 for stability)
    # ---------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total < 0.01:                     # guard against the degenerate case
        best_seq = np.full_like(best_seq, 0.01 / best_seq.shape[0])
        total = best_seq.sum()
    best_seq = best_seq / total

    print(f"\nFinal best value (approx, 2·n·max(conv)): {best_val:.6f} "
          f"(length {len(best_seq)})")
    return best_seq.tolist()
