def _initial_candidates_n(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    cand = []
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n//4, n//2, 3*n//4):
            cand.append(np.roll(warm, shift))
    # Dirichlet random
    cand.append(np.random.dirichlet(np.ones(n)))
    # Uniform
    cand.append(np.full(n, 1.0/n))
    # Gaussian centered
    i = np.arange(n, dtype=float)
    for sigma in (n/8, n/6, n/4):
        g = np.exp(-0.5 * ((i - n/2)/sigma)**2)
        g /= g.sum()
        cand.append(g)
    # Exponential from left edge
    for alpha in (0.001, 0.002, 0.005, 0.01):
        e = np.exp(-alpha * i)
        e /= e.sum()
        cand.append(e)
        # symmetric double edge
        d = e + e[::-1]
        d /= d.sum()
        cand.append(d)
    # Power law edge: shape = ((i+1)*(n-i))**beta with beta negative
    idx = i+1
    prod = idx * (n - i)
    for beta in (-2.0, -1.5, -1.0, -0.5):
        p = np.where(prod>0, prod**beta, 0.0)
        if p.sum()>0:
            p /= p.sum()
            cand.append(p)
    # Triangular ramp (linear increase then decrease)
    mid = n//2
    ramp = np.concatenate([np.arange(1,mid+1), np.arange(mid,0,-1)][:n])
    ramp = ramp.astype(float)
    ramp /= ramp.sum()
    cand.append(ramp)
    # Cosine bump
    cos = np.cos(np.pi * (i - n/2) / n)
    cos = np.maximum(cos,0.0)
    cos /= cos.sum()
    cand.append(cos)
    return cand

def _optimize(a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val):
    # Adam phase
    eps = 1e-12
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1=0.9; beta2=0.999; eps_adam=1e-8
    lr = 0.03
    lam = 0.2
    lam_factor = 1.03
    it = 0
    last_improv = time.time()
    # Keep best local a for subgrad init
    local_best_a = a0.copy()
    local_best_val = float('inf')
    while time.time() < deadline:
        # convert logits to probability simplex via softmax
        max_log = logits.max()
        exp_log = np.exp(logits - max_log)
        a = exp_log / exp_log.sum()
        # compute conv and val
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[n={n} Adam it={it}] new global best {val:.6f}")
        if val < local_best_val:
            local_best_val = val
            local_best_a = a.copy()
            last_improv = time.time()
        # compute smooth max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1-beta1) * grad_log
        v = beta2 * v + (1-beta2) * (grad_log**2)
        it += 1
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # shake if stagnated
        if time.time() - last_improv > 5.0:
            # add small random noise to logits
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improv = time.time()

        # break iteration limit
        if it >= 300_000:
            break
    # Subgradient phase using local_best_a
    a = local_best_a.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()
    eta = 0.5
    sub_last_improv = time.time()
    kernel = np.array([0.25, 0.5, 0.25])
    for sub_it in range(12000):
        if time.time() >= deadline:
            break
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[n={n} Subgrad it={sub_it}] new global best {val:.6f}")
            sub_last_improv = time.time()
        # active set
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # exponentiated gradient
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # smoothing every 100 steps
        if (sub_it % 100) == 0:
            a = np.convolve(a, kernel, mode='same')
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 0:
                a /= a_sum

        if time.time() - sub_last_improv > 10.0:
            break
    return best_seq, best_val

"""
Propose a low‑value step function for the given evaluation metric.

The algorithm consists of three stages:

1. **Parameter sweep** – generate many simple parametric families
   (exponential, double‑exponential, power‑law, linear ramp,
   half‑sine, Gaussian) on a few grid points.  For each candidate the
   exact objective is evaluated with an FFT‑based convolution.
   The best few candidates are retained.

2. **Smooth‑max Adam optimisation** – starting from each retained
   candidate we run Adam on a log‑sum‑exp surrogate of the max
   convolution.  The optimisation works on the soft‑max (entropy)
   parametrisation of the simplex; a temperature parameter ``lam``
   grows exponentially so the surrogate converges to the true max.

3. **Exact sub‑gradient refinement** – a short exponentiated‑gradient
   phase on the true max (active‑set weights).  A light smoothing
   (3‑point binomial kernel) is applied every 100 steps to keep the
   shape regular.  The overall best sequence over all runs is returned.

The routine respects the supplied time budget (default 1000 s) and
falls back to a uniform vector if nothing is found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective value: 2·n·max(a*a)/(sum(a)**2)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _initial_candidates_n(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Generate a modest pool of diverse start vectors for size ``n``."""
    cand = []

    # warm start and a few cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # uniform
    cand.append(np.full(n, 1.0 / n))

    i = np.arange(n, dtype=np.float64)

    # exponential tails
    for alpha in (0.0005, 0.001, 0.002, 0.005, 0.01):
        exp_left = np.exp(-alpha * i)
        exp_left /= exp_left.sum()
        cand.append(exp_left)

        exp_both = exp_left + exp_left[::-1]
        exp_both = np.maximum(exp_both, 0.0)
        exp_both /= exp_both.sum()
        cand.append(exp_both)

    # power‑law edges (negative exponent)
    prod = (i + 1) * (n - i)
    for beta in (-2.5, -2.0, -1.5, -1.0, -0.5):
        shape = np.where(prod > 0, prod ** beta, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # linear ramp (monotone decreasing)
    ramp = (n - i).astype(np.float64)
    ramp /= ramp.sum()
    cand.append(ramp)

    # half‑sine (positive half‑wave)
    half_sine = np.sin(np.pi * (i + 0.5) / n)
    half_sine = np.maximum(half_sine, 0.0)
    half_sine /= half_sine.sum()
    cand.append(half_sine)

    # centred Gaussian
    for sigma in (n / 12.0, n / 8.0, n / 6.0):
        gauss = np.exp(-0.5 * ((i - n / 2) / sigma) ** 2)
        gauss = np.maximum(gauss, 0.0)
        gauss /= gauss.sum()
        cand.append(gauss)

    return cand


# ----------------------------------------------------------------------
def _adam_phase(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max (log‑sum‑exp) surrogate.
    Returns the updated global best sequence/value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05

    lam = 0.5                # start temperature
    lam_factor = 1.02

    it = 0
    last_improve = time.time()
    local_best_seq = a0.copy()
    local_best_val = float("inf")

    while time.time() < deadline:
        # ----- simplex representation -----
        max_log = logits.max()
        exp_log = np.exp(logits - max_log)
        a = exp_log / exp_log.sum()

        # ----- exact objective (for book‑keeping) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[n={n} Adam it={it:05d}] new global best {val:.6f}")

        if val < local_best_val:
            local_best_val = val
            local_best_seq = a.copy()
            last_improve = time.time()

        # ----- smooth‑max weights (log‑sum‑exp) -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, 1e6)

        # occasional shake when stagnating
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        if it > 250_000:                # safety cap
            break

    # Return the best point found by the Adam phase
    return local_best_seq, best_val


# ----------------------------------------------------------------------
def _subgrad_phase(
    a_start: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient (exponentiated‑gradient) refinement on the true
    max‑convolution.  A tiny smoothing kernel is applied every 100 steps.
    """
    a = np.maximum(a_start, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.random.dirichlet(np.ones(n))
        a_sum = a.sum()
    a = a / a_sum

    eta = 0.6
    sub_last_improve = time.time()
    smoothing_kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)

    for sub_it in range(12_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[n={n} Subgrad it={sub_it:05d}] new global best {val:.6f}")

        # active set of indices attaining the maximum
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:                     # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # mild smoothing every 100 iterations
        if (sub_it % 100) == 0:
            a = np.convolve(a, smoothing_kernel, mode="same")
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 0:
                a /= a_sum

        if time.time() - sub_last_improve > 10.0:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
def _optimize_candidate(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Full optimisation pipeline for a single starting vector.
    Returns the (possibly updated) global best sequence/value.
    """
    # Adam on smooth‑max surrogate
    a_adam, best_val = _adam_phase(
        a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
    )
    # Exact sub‑gradient refinement
    best_seq, best_val = _subgrad_phase(
        a_adam, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
    )
    return best_seq, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises ``evaluate_sequence``.  The routine respects a total
    runtime budget (default 1000 s) and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # try a few problem sizes – larger n often yields a lower bound
    size_plan = [4096, 8192, 16384]   # try larger sizes if time permits

    for n in size_plan:
        if time.time() >= deadline:
            break

        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        # obtain warm start from previous best if available
        warm_raw = globals().get("height_sequence_1")
        warm = None
        if warm_raw is not None:
            warm_arr = np.asarray(warm_raw, dtype=np.float64)
            if warm_arr.shape[0] != n:
                # linear interpolation to required length
                x_old = np.linspace(0, 1, warm_arr.shape[0], endpoint=False)
                x_new = np.linspace(0, 1, n, endpoint=False)
                warm_arr = np.interp(x_new, x_old, warm_arr)
            warm_arr = np.maximum(warm_arr, 0.0)
            if warm_arr.sum() >= 0.01:
                warm_arr /= warm_arr.sum()
                warm = warm_arr

        # ------------------------------------------------------------------
        # generate a diverse pool of seeds
        candidates = _initial_candidates_n(n, warm)

        # ------------------------------------------------------------------
        # evaluate a cheap parameter sweep before heavy optimisation:
        # keep the top 4 candidates (by exact objective) for deeper work.
        scored = []
        for cand in candidates:
            if time.time() >= deadline:
                break
            a = np.maximum(cand, 0.0)
            s = a.sum()
            if s < 0.01:
                a = np.random.dirichlet(np.ones(n))
            else:
                a /= s
            val = _objective(a, n, conv_fft_len)
            scored.append((val, a))
        scored.sort(key=lambda t: t[0])
        top_candidates = [a for _, a in scored[:4]]

        # ------------------------------------------------------------------
        # run full optimiser on the selected seeds
        for a0 in top_candidates:
            if time.time() >= deadline:
                break
            best_seq, best_val = _optimize_candidate(
                a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
            )

    # ------------------------------------------------------------------
    # final fallback
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Return plain Python list as required by the evaluation harness
    return best_seq.tolist()
