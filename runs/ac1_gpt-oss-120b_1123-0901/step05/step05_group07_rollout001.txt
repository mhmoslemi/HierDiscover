def proj_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def full_convolution(a, fft_len=None):
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv

"""
propose_candidate
-----------------

A time‑bounded hybrid optimiser for the “max‑autocorrelation” objective.
The algorithm works on several problem sizes (1024, 2048, 4096, 8192) and
combines three complementary search phases:

1. **Smooth‑max Adam** – a soft‑max (log‑sum‑exp) approximation of the
   max‑convolution is annealed from a relatively high temperature (≈0.1)
   down to ≈1e‑6.  Adam updates are performed on log‑probabilities,
   guaranteeing a non‑negative vector that stays on the probability simplex.

2. **Exact‑max projected gradient** – once the temperature is very small,
   the binary‑max (sub‑gradient) of the true max‑convolution is used.
   A plain projected‑gradient step (with projection onto the simplex) is
   applied to refine the solution further.

3. **Greedy mass‑shuffling** – a fast local‑search that repeatedly moves a
   tiny amount of mass from one coordinate to another; a move is kept only
   if it lowers the objective.  This phase discovers improvements that are
   not reachable by the smooth gradient due to its non‑convex landscape.

The optimiser is deliberately *restart‑heavy*: each length receives several
random/structured starts (Dirichlet, geometric, up‑scaled previous best,
warm‑starts).  The best sequence across all lengths and restarts is returned.
All operations respect the global 1000 s wall‑time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using FFT (real input)."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) of length len(a).
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)],   i = 0…n‑1.
    The implementation uses FFTs for O(n log n) speed.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def smooth_max_and_weights(conv: np.ndarray, tau: float):
    """
    Temperature‑smoothed max of `conv` and the corresponding softmax
    weights (∂smooth_max/∂conv).  Numerical stability is achieved by
    shifting by the maximum before exponentiation.
    """
    c_max = conv.max()
    shifted = (conv - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    smooth = c_max + tau * np.log(sum_exp)
    weights = exp_shifted / sum_exp
    return smooth, weights


def max_weights(conv: np.ndarray, eps: float = 1e-12):
    """
    Weight vector that corresponds to a sub‑gradient of the exact max.
    All entries attaining the maximum (up to `eps`) receive equal weight.
    """
    mx = conv.max()
    mask = conv >= mx - eps
    w = mask.astype(np.float64)
    if w.sum() > 0:
        w /= w.sum()
    return w


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.  Implements the classic O(n log n) algorithm.
    """
    if v.ndim != 1:
        raise ValueError("Input must be 1‑D.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all entries are ≤ 0
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax_from_logits(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list:
    """
    Return a non‑negative sequence that (empirically) minimises
        F(a) = 2·n·max(a*a) / (sum(a)^2)

    The implementation follows the hybrid strategy described in the module
    docstring.  The function respects the global time‑budget (`budget_s`)
    and prints progress information.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # slight safety margin

    best_seq = None      # numpy array (sum = 1)
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Problem sizes – we try increasingly larger lengths if time permits.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]

    # Keep a reference to the previous best length (for up‑scaling).
    prev_best = None

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # Pre‑compute FFT lengths used repeatedly.
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))   # ≥ (2n‑1)+n‑1

        # Number of independent restarts for the current length.
        # Larger n gets fewer restarts to keep the overall budget reasonable.
        if n <= 1024:
            restarts = 12
        elif n <= 2048:
            restarts = 9
        elif n <= 4096:
            restarts = 6
        else:
            restarts = 4

        for r in range(restarts):
            if time.time() > deadline:
                break

            # --------------------------------------------------------------
            # 1️⃣  Initialise a (probability vector) and logits x
            # --------------------------------------------------------------
            # Warm‑start from best of *same* length if available.
            if best_seq is not None and best_seq.shape[0] == n:
                a = best_seq.copy()
                a /= a.sum()
                x = np.log(a + 1e-15)
                print(f"[n={n} restart={r}] warm start from same‑length best")
            # Upscale from previous best (different length) when possible.
            elif prev_best is not None:
                # Linear interpolation to the new size.
                old_n = prev_best.shape[0]
                t = np.linspace(0, 1, n)
                s = np.linspace(0, 1, old_n)
                a_interp = np.interp(t, s, prev_best)
                a_interp += 1e-8 * np.random.rand(n)        # break symmetry
                a_interp = np.maximum(a_interp, 0.0)
                a = a_interp / a_interp.sum()
                x = np.log(a + 1e-15)
                print(f"[n={n} restart={r}] upscaled from length {old_n}")
            # Occasionally use a deterministic geometric start.
            elif r % 3 == 0:
                decay = 0.985
                a = np.array([decay ** i for i in range(n)], dtype=np.float64)
                a = np.maximum(a, 1e-9)
                a /= a.sum()
                x = np.log(a + 1e-15)
                print(f"[n={n} restart={r}] geometric start")
            # Otherwise random Dirichlet.
            else:
                a = np.random.dirichlet(np.ones(n))
                x = np.log(a + 1e-15)
                print(f"[n={n} restart={r}] random Dirichlet start")

            # --------------------------------------------------------------
            # 2️⃣  Adam optimisation with smooth‑max annealing
            # --------------------------------------------------------------
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8

            lr = 0.12                     # initial learning rate for logits
            decay_every = 6000           # learning‑rate decay interval

            # Number of iterations – generous but respects the global deadline.
            max_iters = (
                130000 if n <= 1024 else
                95000  if n <= 2048 else
                70000  if n <= 4096 else
                50000
            )

            # Temperature schedule (linear annealing to a tiny value).
            tau_initial = 0.12
            tau_final = 4e-7

            last_improve = time.time()
            for t in range(1, max_iters + 1):
                if time.time() > deadline:
                    break

                # ----- soft‑max → probability vector -----
                a = softmax_from_logits(x)

                # ----- convolution & (smooth) max -----
                conv = full_convolution(a, fft_len=fft_len_conv)

                # annealed temperature
                tau = max(tau_final,
                          tau_initial * (1.0 - t / max_iters))

                smooth, weights = smooth_max_and_weights(conv, tau)

                # True objective value (used for bookkeeping)
                cur_val = 2.0 * n * conv.max()
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t:06d}] "
                          f"new best {cur_val:.6f}")

                # ----- gradient of smooth‑max w.r.t. a -----
                grad_a = 2.0 * cross_correlation(weights, a, fft_len=fft_len_corr)

                # back‑propagation through the soft‑max
                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                # ----- Adam update -----
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # learning‑rate decay
                if t % decay_every == 0:
                    lr *= 0.7

                # occasional stochastic “shake” if stagnating
                if time.time() - last_improve > 30.0:
                    x += (np.random.rand(n) - 0.5) * 5e-3
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t}] shake (stagnation)")

            # --------------------------------------------------------------
            # 3️⃣  Exact‑max projected gradient refinement (few thousand steps)
            # --------------------------------------------------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a = best_seq.copy()
                # start from the already‑good point but discard logits
                fine_lr = 0.025
                fine_iters = 12000
                for ft in range(1, fine_iters + 1):
                    if time.time() > deadline:
                        break
                    conv = full_convolution(a, fft_len=fft_len_conv)
                    cur_val = 2.0 * n * conv.max()
                    if cur_val < best_val - 1e-12:
                        best_val = cur_val
                        best_seq = a.copy()
                        print(f"[fine tune] improved to {cur_val:.6f}")

                    # binary max‑weights (sub‑gradient of true max)
                    w = max_weights(conv, eps=1e-12)
                    grad_a = 2.0 * cross_correlation(w, a, fft_len=fft_len_corr)

                    # gradient step on a (sum constraint -> projection)
                    a_new = a - fine_lr * grad_a
                    a_new = project_simplex(a_new)

                    # accept if improvement, otherwise keep old a
                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = 2.0 * n * conv_new.max()
                    if val_new < cur_val - 1e-12:
                        a = a_new
                        if val_new < best_val - 1e-12:
                            best_val = val_new
                            best_seq = a_new.copy()
                            print(f"[fine tune] new best {val_new:.6f}")
                    else:
                        # decay learning rate gently to help escape plateaus
                        fine_lr *= 0.9995

                # --------------------------------------------------------------
                # 4️⃣  Greedy local mass‑shuffling (cheap hill‑climbing)
                # --------------------------------------------------------------
                a_local = best_seq.copy()
                local_iters = 3000 + (n // 256) * 400   # more steps for larger n
                for it in range(local_iters):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    # move a fraction of the smaller coordinate towards the larger
                    eps_move = min(a_local[i], 0.007) * (np.random.rand() * 0.6 + 0.2)
                    if eps_move <= 0:
                        continue
                    a_new = a_local.copy()
                    a_new[i] -= eps_move
                    a_new[j] += eps_move
                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = 2.0 * n * conv_new.max()
                    if val_new < best_val - 1e-12:
                        best_val = val_new
                        best_seq = a_new.copy()
                        a_local = a_new
                        print(f"[local refine] improved to {val_new:.6f}")

        # Store the best sequence of the current length for possible up‑scaling.
        if best_seq is not None and best_seq.shape[0] == n:
            prev_best = best_seq.copy()

    # ------------------------------------------------------------------
    # Fallback – if nothing was found (should not happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = 2.0 * n_fallback * (1.0 / n_fallback)   # = 2

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
