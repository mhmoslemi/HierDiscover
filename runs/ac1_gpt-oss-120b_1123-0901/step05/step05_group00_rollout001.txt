"""
Propose a step‑function (non‑negative vector) that minimises
    F(a) = 2·n·max(a*a) / (sum(a)^2)

Algorithm
---------
* Multi‑scale optimisation: start at modest lengths (2 k‑4 k) and
  progressively up‑sample to larger lengths (up to 65 k) – the shape
  discovered at a coarser scale guides the finer scales.
* For each length:
    – Build a diverse pool of initial vectors (uniform, up‑sampled best,
      random Dirichlet, edge‑concentrated power‑law, double‑Gaussian,
      sine‑squared bump, historic variable if present).
    – Optimise each candidate with Adam on a smooth‑max surrogate
      (log‑sum‑exp) of the convolution maximum.  The surrogate temperature
      (λ) is annealed exponentially.
    – Refine the best candidate with a sub‑gradient / exponentiated‑gradient
      scheme that works on the true maximum.
    – Perform a cheap random‑perturbation (Dirichlet) local search that
      accepts only improving moves.
* Time is monitored, and as soon as the remaining budget is low we stop.
* The best vector found (as a plain Python list) is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a*a using an FFT of (power‑of‑two) length fft_len.
    Returns the (2·n‑1)‑length convolution.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length n (the length of a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i >= 0, sum_i x_i = 1}
    (Implementation of the algorithm from Duchi et al., 2008.)
    """
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical safety: return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective value F(a) for sum(a)=1 (else denominator is used)."""
    sum_a = a.sum()
    if sum_a <= 0:
        return float("inf")
    a_norm = a / sum_a
    conv = _full_convolution(a_norm, fft_len=fft_len)
    max_c = conv.max()
    return 2.0 * n * max_c / (sum_a ** 2)

    # Target lengths (geometric progression).  The largest length is tried
    # only if time permits.
    lengths = [1024, 2048, 4096, 8192, 16384, 32768, 65536]

    # ------------------------------------------------------------------
    # Core optimisation loop over lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)   # (2n‑1)+(n‑1)

        # --------------------------------------------------------------
        # Build a rich pool of initial vectors (all are normalised)
        # --------------------------------------------------------------
        candidates = []

        # 1) Uniform
        candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # 2) Warm‑start from an exact‑size solution we have already
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.array(best_seq, dtype=np.float64))

        # 3) Upsampled version of the current global best (if shorter)
        if best_seq is not None and len(best_seq) < n:
            old = np.asarray(best_seq, dtype=np.float64)
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, old)
            if up.sum() > 0:
                up = up / up.sum()
                candidates.append(up)
            else:
                candidates.append(np.random.dirichlet(np.ones(n)))

        # 4) Historic global variable if present
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            h = np.asarray(hseq, dtype=np.float64)
            if h.shape[0] > n:
                h = h[:n]
            elif h.shape[0] < n:
                h = np.pad(h, (0, n - h.shape[0]))
            h = np.maximum(h, 0.0)
            if h.sum() > 0:
                h = h / h.sum()
                candidates.append(h)
            else:
                candidates.append(np.random.dirichlet(np.ones(n)))

        # 5) Random Dirichlet draws (two independent draws)
        candidates.append(np.random.dirichlet(np.ones(n)))
        candidates.append(np.random.dirichlet(np.ones(n)))

        # 6) Edge‑concentrated power‑law families
        idx = np.arange(1, n + 1, dtype=np.float64)
        prod = idx * (n - idx + 1.0)
        for alpha in np.linspace(-2.0, -0.5, 4):
            shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
            if shape.sum() > 0:
                shape = shape / shape.sum()
                candidates.append(shape)

        # 7) Two‑Gaussian edge shapes
        i = np.arange(n, dtype=np.float64)
        for gamma in [0.2, 0.5, 1.0, 2.0]:
            shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            shape = shape / shape.sum()
            candidates.append(shape)

        # 8) Smooth sine‑squared bump
        shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
        shape = shape / shape.sum()
        candidates.append(shape)

        # --------------------------------------------------------------
        # Optimise each candidate with Adam on a smooth‑max surrogate
        # --------------------------------------------------------------
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # Ensure positivity (tiny epsilon) and normalise
            eps = 1e-16
            a = np.maximum(a0, eps)
            a = a / a.sum()
            logits = np.log(a)                     # parameters of soft‑max

            # Adam hyper‑parameters
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2 = 0.9, 0.999
            eps_adam = 1e-8
            lr = 0.05

            # Smooth‑max temperature
            lam = 0.5
            lam_factor = 1.02

            # Iteration budget: more for small n, fewer for large n.
            max_iters = int(18000 * 1024 / n) + 3000
            last_improve = time.time()

            for it in range(1, max_iters + 1):
                # Soft‑max (ensures a stays on the simplex)
                mx = logits.max()
                exp_log = np.exp(logits - mx)
                a = exp_log / exp_log.sum()

                # Convolution and smooth‑max weight computation
                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    # fallback to uniform weights if numerical overflow occurs
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # Gradient of smooth‑max w.r.t. a
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

                # Back‑propagation through the soft‑max (standard trick)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update on logits
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                # Anneal temperature (increase λ => tighter approximation)
                lam = min(lam * lam_factor, 1e7)

                # Exact objective (sum(a)=1 => denominator = 1)
                val = 2.0 * n * max_c

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Adam it={it:05d}] new best {val:.6f}")

                # Stagnation handling: small random jitter of logits
                if time.time() - last_improve > 5.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # ----------------------------------------------------------
            # Sub‑gradient refinement on the true max (exponentiated‑gradient)
            # ----------------------------------------------------------
            a = a / a.sum()
            eta = 0.6
            sub_last_improve = time.time()
            for sub_it in range(2000):
                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad it={sub_it}] new best {val:.6f}")

                if time.time() - sub_last_improve > 5.0:
                    break
                if time.time() > deadline:
                    break

        # ------------------------------------------------------------------
        # Local random perturbation refinement for the best vector at this n
        # ------------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            a_local = best_seq.copy()
            # Parameters for the Dirichlet perturbations
            # Higher concentration -> smaller moves; we decrease it over time.
            conc_start = 200.0
            conc_end = 20.0
            max_local_iters = 5000
            for loc_it in range(max_local_iters):
                if time.time() > deadline:
                    break

                # Linear schedule for concentration (more exploration later)
                alpha = conc_start + (conc_end - conc_start) * (loc_it / max_local_iters)
                # Ensure all parameters >0 for Dirichlet
                pert = np.maximum(a_local, 1e-12) * alpha
                a_new = np.random.dirichlet(pert)

                # Compute objective (fast via FFT)
                val_new = _objective(a_new, n, conv_fft_len)
                if val_new < best_val:
                    best_val = val_new
                    best_seq = a_new.copy()
                    a_local = a_new.copy()
                    print(f"[n={n} Local perturb it={loc_it}] new best {val_new:.6f}")

                # Occasionally, after a batch of successful moves, perform a few
                # sub‑gradient EG steps to drive down the max even more.
                if (loc_it + 1) % 1000 == 0:
                    # 10 EG steps
                    a_tmp = a_local.copy()
                    eta_local = 0.5
                    for eg in range(10):
                        conv = _full_convolution(a_tmp, fft_len=conv_fft_len)
                        max_c = conv.max()
                        active = np.isclose(conv, max_c, atol=1e-12)
                        w = np.zeros_like(conv)
                        if active.any():
                            w[active] = 1.0 / active.sum()
                        grad_a = 2.0 * _cross_correlation(w, a_tmp, fft_len=corr_fft_len)
                        a_tmp = a_tmp * np.exp(-eta_local * grad_a)
                        a_tmp = a_tmp / a_tmp.sum()
                    val_tmp = _objective(a_tmp, n, conv_fft_len)
                    if val_tmp < best_val:
                        best_val = val_tmp
                        best_seq = a_tmp.copy()
                        a_local = a_tmp.copy()
                        print(f"[n={n} EG after perturb it={loc_it}] new best {val_tmp:.6f}")

    # ----------------------------------------------------------------------
    # Fallback: if nothing was discovered, return a uniform vector of moderate size
    # ----------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return as plain Python list (as required by the external evaluator)
    return best_seq.tolist()
