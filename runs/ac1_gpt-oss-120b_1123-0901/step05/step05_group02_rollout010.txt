"""
propose_candidate – a multi‑stage optimisation for the step‑function objective

The function searches for a non‑negative sequence a (sum does not matter because
the objective is scale‑invariant) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)²)

The optimisation consists of several layers:

1.   Several cheap initialisations (uniform, Dirichlet, exponential,
     power‑law, etc.).
2.   For each initial guess an Adam optimiser is run on a smooth‑max
     surrogate of the max‑convolution.  The temperature (smooth‑max sharpness)
     is increased very slowly to avoid premature “hardening”.
3.   A few sub‑gradient (exponentiated‑gradient) steps on the exact max are
     performed to fine‑tune the solution.
4.   The best sequence for a given length is fine‑tuned again with Adam but
     with a much longer horizon.
5.   Finally a cheap simulated‑annealing (SA) phase is applied to the global
     best vector in order to escape tiny local minima.

The routine respects a user supplied time‑budget (default 1000 s) and works
with lengths up to 16384 (larger lengths would be extremely costly in the
available time).  All helper functions are defined at module level – there are
no closures and the code uses only NumPy (no external solvers).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a using an FFT of length ``fft_len``.
    The result has length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact objective value from a convolution vector (sum of a is 1)."""
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    The routine respects the supplied time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Lengths we will try – larger lengths give more expressivity but are
    # slower.  The bulk of the time is spent on n=4096; we also give a
    # chance to n=8192 and n=16384 if time permits.
    lengths = [4096, 8192, 16384]

    # Pre‑compute optimal FFT sizes (powers of two) for both convolution
    # and cross‑correlation.
    fft_len_conv = {}
    fft_len_corr = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_len_conv[n] = _next_pow2(conv_len)
        # For correlation we need length conv_len + n – 1 ≤ 3·n
        fft_len_corr[n] = _next_pow2(conv_len + n - 1)

    # ------------------------------------------------------------------
    # Helper that builds a list of diverse initial guesses for a given length.
    def _initial_candidates(n: int) -> list:
        cand = []

        # Warm‑start from the global best if it has the same length.
        if best_seq is not None and len(best_seq) == n:
            cand.append(np.asarray(best_seq, dtype=np.float64))

        # Historic solution stored in the global variable (if any).
        hist = globals().get("height_sequence_1")
        if hist is not None:

            else:

            h = np.maximum(h, 0.0)
            s = h.sum()
            if s < 0.01:
                h = np.random.dirichlet(np.ones(n))
            else:
                h = h / s
            cand.append(h)

        # Pure Dirichlet (concentration = 1)
        cand.append(np.random.dirichlet(np.ones(n)))

        # Spiky Dirichlet – smaller concentration → more mass on few entries
        for conc in [0.1, 0.25, 0.5]:

        for gamma in [0.0003, 0.0007, 0.0015]:

            shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))

            x = np.arange(n, dtype=np.float64)

    # Main length loop
    for n in lengths:

        print(f"\n=== Optimising length n={n} ===")
        conv_len = fft_len_conv[n]
        corr_len = fft_len_corr[n]

        if n == 4096:

        elif n == 8192:
            max_candidates = 9
        else:
            max_candidates = 5
        candidates = _initial_candidates(n)[:max_candidates]

        # Adam hyper‑parameters (chosen for stability on long horizons)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        base_lr = 0.03          # a bit larger than before, but we will decay later
        lam_start = 0.15        # start with a fairly smooth max

        # Iterate over candidates
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # Normalise to a probability vector – this removes the scaling invariance.
            a = a0 / a0.sum()
            logits = np.log(a + 1e-15)                # soft‑max parameters

            # Iteration budget – scaled with size but capped.
            max_iters = int(32000 * 1024 / n) + 6000
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)

            # Track improvement to inject occasional random shakes.
            last_improve = time.time()
            lam = lam_start

            for t in range(1, max_iters + 1):
                # ------- soft‑max projection -------
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                # ------- smooth‑max of the convolution -------
                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                # stable soft‑max over conv (weights sum to 1)
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # ------- gradient of the surrogate -------
                grad_a = 2.0 * _cross_correlation(w, a, corr_len)

                # Back‑prop through soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # ------- Adam update -------
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                lr = base_lr / (np.sqrt(v_hat) + eps)
                logits -= lr * m_hat

                # Temperature schedule – slowly sharpen the max.
                lam = min(lam * lam_factor, 1e6)

                # Exact objective (a already sums to 1)
                val = 2.0 * n * max_c

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Adam iter={t:05d}] new best {val:.6f}")

                # Stagnation handling: add a tiny random jitter occasionally.
                if time.time() - last_improve > 7.0:
                    logits += (np.random.rand(n) - 0.5) * 8e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # ------- Sub‑gradient (exponentiated‑gradient) phase -------------
            # Run a modest number of exact‑max steps to refine the solution.
            a = a / a.sum()
            eta = 0.45
            sub_last_improve = time.time()
            max_sub_iters = int(5000 * 1024 / n) + 4000

            for sub_it in range(max_sub_iters):
                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)

                # Uniform distribution over the active indices
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                grad_a = 2.0 * _cross_correlation(w, a, corr_len)
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:                     # protect against collapse
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                val = 2.0 * n * max_c

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad iter={sub_it}] new best {val:.6f}")

                if time.time() - sub_last_improve > 6.0:
                    break
                if time.time() > deadline:
                    break

        # ------------------------------------------------------------------
        # Fine‑tuning of the still‑best vector for this length (if any)
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] fine‑tuning current best (val={best_val:.6f})")
            a = best_seq.copy()
            logits = np.log(a + 1e-15)
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.025
            lam = 0.2
            lam_factor = 1.006
            last_improve = time.time()
            fine_iters = int(48000 * 1024 / n) + 8000

            for t in range(1, fine_iters + 1):
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

                grad_a = 2.0 * _cross_correlation(w, a, corr_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                lam = min(lam * lam_factor, 1e6)

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} fine‑tune iter={t:05d}] new best {val:.6f}")

                if time.time() - last_improve > 9.0:
                    # small random shake to help escape tiny plateaus
                    logits += (np.random.rand(n) - 0.5) * 6e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

    # ----------------------------------------------------------------------
    # Simulated‑annealing polishing of the global best (if any time left).
    if best_seq is not None and time.time() < deadline:
        print("\n=== Simulated‑Annealing polishing ===")
        a = best_seq.copy()
        n = a.shape[0]
        conv_len = fft_len_conv[n]
        # work in log‑space (soft‑max) – this guarantees feasibility.
        logits = np.log(a + 1e-15)

        T = 1.0                     # initial temperature
        T_min = 1e-4
        decay = 0.9995              # exponential cooling factor
        sa_iters = 20000

        for it in range(sa_iters):
            if time.time() > deadline:
                break
            # propose a new point by jittering the logits
            sigma = 0.05 * T
            new_logits = logits + sigma * np.random.randn(n)
            mx = new_logits.max()
            exp_x = np.exp(new_logits - mx)
            new_a = exp_x / exp_x.sum()

            conv = _full_convolution(new_a, conv_len)
            val = 2.0 * n * conv.max()

            # acceptance test
            if val < best_val or np.random.rand() < np.exp(-(val - best_val) / max(T, 1e-12)):
                logits = new_logits
                best_val = val
                best_seq = new_a.copy()
                # occasional logging
                if it % 2000 == 0:
                    print(f"[SA iter={it}] new best {val:.6f}")

            # cool down
            T = max(T * decay, T_min)

    # ----------------------------------------------------------------------
    # Fallback – return a uniform vector if nothing was found (should never happen)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator.
    return best_seq.tolist()
