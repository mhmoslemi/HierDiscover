import time, numpy as np, cvxpy as cp

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0 # margin

    best_seq = None
    best_val = float('inf')
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() > deadline: break
        sqrt_sum = np.sqrt(2 * n)

        # Determine number of restarts based on remaining time
        restarts = 5 if n <= 1024 else (3 if n <= 2048 else 2)
        for r in range(restarts):
            if time.time() > deadline: break
            # initial a
            if best_seq is not None and len(best_seq) == n:
                a = np.array(best_seq, dtype=float)
                a = a / a.sum() * sqrt_sum
                print(f"n={n} restart={r} warm start")
            elif best_seq is not None and len(best_seq) < n:
                # upsample using linear interpolation
                a0 = np.array(best_seq)
                # simple repeat with interpolation
                x_old = np.linspace(0, 1, len(a0), endpoint=False)
                x_new = np.linspace(0, 1, n, endpoint=False)
                a = np.interp(x_new, x_old, a0)
                a = np.maximum(a, 0.0)
                a = a / a.sum() * sqrt_sum
                print(f"n={n} restart={r} upsampled from prev")
            else:
                a = np.random.dirichlet(np.ones(n)) * sqrt_sum
                print(f"n={n} restart={r} random start")

            # main iterative refinement
            max_iters = 30  # number of LP steps or until time out
            for it in range(max_iters):
                if time.time() > deadline:
                    break
                # compute conv of a with itself
                conv_aa = np.fft.irfft(np.fft.rfft(a, 1<<int(np.ceil(np.log2(2*n-1)))) * np.fft.rfft(a, 1<<int(np.ceil(np.log2(2*n-1)))) )[:2*n-1]
                M = conv_aa.max()
                # build and solve LP
                # define parameter f and upper bound M
                f_param = cp.Parameter(n, nonneg=True)
                f_param.value = a
                b = cp.Variable(n, nonneg=True)
                conv_fb = cp.conv(f_param, b)
                constraints = [conv_fb <= M]
                prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
                try:
                    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
                except Exception as e:
                    print(f"LP solve error: {e}")
                    break
                if prob.status not in ["optimal", "optimal_inaccurate"]:
                    print(f"LP not solved optimally status={prob.status}")
                    break
                b_opt = b.value
                if b_opt is None:
                    print("LP produced no solution")
                    break
                sum_b = b_opt.sum()
                if sum_b <= 0:
                    print("Zero sum b")
                    break
                # compute g
                g = (sqrt_sum / sum_b) * b_opt

                # line search for mixing factor t
                best_local_val = None
                best_local_a = None
                best_local_M = None
                # candidate t values
                t_candidates = [0.6, 0.4, 0.2, 0.1, 0.05, 0.02]
                for t in t_candidates:
                    a_new = (1 - t) * a + t * g
                    # compute conv anew
                    conv_new = np.fft.irfft(np.fft.rfft(a_new, 1<<int(np.ceil(np.log2(2*n-1)))) * np.fft.rfft(a_new, 1<<int(np.ceil(np.log2(2*n-1)))) )[:2*n-1]
                    M_new = conv_new.max()
                    if best_local_val is None or M_new < best_local_val:
                        best_local_val = M_new
                        best_local_a = a_new
                        best_local_M = M_new
                        best_t = t
                # if improvement, accept
                if best_local_M < M - 1e-12:
                    a = best_local_a
                    M = best_local_M
                    cur_val = 2.0 * n * M / (sqrt_sum**2)  # essentially M
                    if cur_val < best_val:
                        best_val = cur_val
                        best_seq = a.copy()
                        print(f"[n={n} restart={r} iter={it}] new best {cur_val:.6f}")
                else:
                    # no improvement; stop early
                    print(f"[n={n} restart={r} iter={it}] no improvement, break")
                    break

            # after LP loop, maybe run local random perturbation similar to earlier algorithm
            # simple mass shuffle for a fixed number of steps
            if best_seq is not None and len(best_seq) == n:
                local_a = best_seq.copy()
                for loc_it in range(2000):
                    if time.time() > deadline: break
                    i, j = np.random.choice(n, 2, replace=False)
                    move = min(local_a[i], 0.005) * (np.random.rand() * 0.6 + 0.2)
                    if move <= 0: continue
                    a_candidate = local_a.copy()
                    a_candidate[i] -= move

        best_seq = best_seq * scale

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")

"""
Propose a non‑negative step function (probability vector) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)^2)

The algorithm is based on the classic “LP‑step” described in the
statement.  For a fixed length n we keep the vector normalised to
‖a‖₁ = √(2 n).  In that normalisation the objective equals the
maximum entry of the auto‑convolution a∗a, so we minimise max(conv).

Iteration:
    1. Compute M = max(a∗a).
    2. Solve the linear program
           maximise   Σ b_j
           subject to  a∗b ≤ M   (element‑wise) ,   b ≥ 0 .
       (The convolution a∗b is linear in b, therefore the LP is
        convex and cheap to solve with cvxpy.)
    3. Rescale b to the same ℓ₁‑norm as a:
           g = (√(2 n) / Σ b) · b .
       By construction a∗g ≤ M.
    4. Take a convex combination f = (1‑t)·a + t·g and keep it if the
       new max‑convolution decreased.  A short line‑search over a few
       values of t is performed.
    5. Repeat until no improvement or time runs out.

After the LP phase a cheap random “mass‑shuffling’’ refinement is
applied to escape possible tiny plateaux.  Several restarts and a
few different lengths (1024, 2048, 4096) are tried; the best vector
found (scaled to sum = 1) is returned.

The whole routine respects a 1000 s wall‑time budget.
"""

import time
import numpy as np

# cvxpy is allowed by the rules – it provides a convenient LP interface.
import cvxpy as cp


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)‑1) using FFT.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, fft_len)[: 2 * n - 1]
    return conv


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Time‑bounded optimisation that returns the best non‑negative sequence
    discovered.  The implementation follows the LP‑based scheme explained
    in the module docstring.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # small safety margin

    best_seq = None          # best raw vector (ℓ₁‑norm = √(2n))
    best_val = float("inf")  # best objective = max(conv) in the √(2n) norm

    # Problem sizes – larger n gives a better theoretical bound.
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() > deadline:
            break

        sqrt_sum = np.sqrt(2.0 * n)          # ℓ₁‑norm we keep during optimisation
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))   # reused for all FFTs

        # Choose a modest number of restarts for each length.
        if n == 1024:
            restarts = 6
        elif n == 2048:
            restarts = 4
        else:
            restarts = 2

        for restart in range(restarts):
            if time.time() > deadline:
                break

            # ---------- initial vector a ----------
            if best_seq is not None and best_seq.shape[0] == n:
                a = best_seq.copy()
                a = a / a.sum() * sqrt_sum
                print(f"[n={n} restart={restart}] warm‑start from previous best")
            elif 'height_sequence_1' in globals():
                # use the previously found construction if available
                h = np.array(height_sequence_1, dtype=float)
                if h.shape[0] >= n:
                    a = h[:n]
                else:
                    a = np.pad(h, (0, n - h.shape[0]))
                a = np.maximum(a, 0.0)
                if a.sum() < 1e-6:
                    a = np.random.dirichlet(np.ones(n))
                a = a / a.sum() * sqrt_sum
                print(f"[n={n} restart={restart}] start from height_sequence_1")
            else:
                a = np.random.dirichlet(np.ones(n)) * sqrt_sum
                print(f"[n={n} restart={restart}] random start")

            # ---------- LP‑driven refinement ----------
            max_lp_iters = 30                     # usually enough to see progress
            for it in range(max_lp_iters):
                if time.time() > deadline:
                    break

                # current max‑convolution (M)
                conv_aa = _full_convolution(a, fft_len=fft_len)
                M = conv_aa.max()

                # ----- solve LP: maximise Σ b  subject to a∗b ≤ M , b≥0 -----
                f_param = cp.Parameter(n, nonneg=True)
                f_param.value = a
                b = cp.Variable(n, nonneg=True)

                # cvxpy provides a linear convolution expression
                conv_fb = cp.conv(f_param, b)        # length 2n‑1
                constraints = [conv_fb <= M]        # elementwise inequality
                prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

                try:
                    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
                except Exception as exc:                # safeguard – abort this restart
                    print(f"   LP solve failed ({exc}); aborting restart")
                    break

                if prob.status not in ("optimal", "optimal_inaccurate"):
                    print(f"   LP not optimal (status={prob.status}); aborting restart")
                    break

                b_opt = b.value
                if b_opt is None:
                    print("   LP returned no solution; aborting restart")
                    break

                sum_b = b_opt.sum()
                if sum_b <= 0.0:
                    print("   LP produced zero sum; aborting restart")
                    break

                # ----- build g and perform a short line‑search on mixing factor t -----
                g = (sqrt_sum / sum_b) * b_opt          # ℓ₁‑norm = √(2n)

                # try a few mixing coefficients and keep the best improvement
                t_candidates = [0.6, 0.4, 0.2, 0.1, 0.05, 0.02]
                best_M_new = M
                best_a_new = a

                for t in t_candidates:
                    a_new = (1.0 - t) * a + t * g
                    conv_new = _full_convolution(a_new, fft_len=fft_len)
                    M_new = conv_new.max()
                    if M_new < best_M_new - 1e-12:
                        best_M_new = M_new
                        best_a_new = a_new

                if best_M_new < M - 1e-12:
                    a = best_a_new
                    M = best_M_new
                    cur_val = M                               # because ‖a‖₁ = √(2n)
                    if cur_val < best_val:
                        best_val = cur_val
                        best_seq = a.copy()
                        print(f"[n={n} restart={restart} it={it}] new best {cur_val:.6f}")
                else:
                    # no improvement – stop the LP phase for this restart
                    print(f"[n={n} restart={restart} it={it}] no improvement, stop LP loop")
                    break

            # ---------- cheap random local refinement ----------
            if best_seq is not None and best_seq.shape[0] == n:
                a_loc = best_seq.copy()
                local_iters = 3000
                for _ in range(local_iters):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    # move a small fraction of a_loc[i] to a_loc[j]
                    max_move = min(a_loc[i], 0.008) * (0.2 + 0.6 * np.random.rand())
                    if max_move <= 0.0:
                        continue
                    a_new = a_loc.copy()
                    a_new[i] -= max_move
                    a_new[j] += max_move
                    conv_new = _full_convolution(a_new, fft_len=fft_len)
                    M_new = conv_new.max()
                    if M_new < best_val - 1e-12:
                        best_val = M_new
                        best_seq = a_new.copy()
                        a_loc = a_new
                        print(f"[local refine n={n}] improved to {best_val:.6f}")

    # -------------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector (objective = 2)
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=float)
        best_val = 2.0
        print("Fallback to uniform distribution")

    # Scale the final vector to sum = 1 – evaluate_sequence expects a plain list.
    best_seq = best_seq / best_seq.sum()

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
