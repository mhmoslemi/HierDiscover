"""
Propose a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The strategy is a 4‑phase hybrid search tuned for the 1000 s budget.

Phase 0 – warm‑start from a known construction (if present).
Phase 1 – fast random generation of many candidates with diverse analytic
         shapes (power‑law, exponential, mixed, “ends‑filled”).  The best
         candidate (lowest exact score) is kept as a seed.
Phase 2 – a longer Adam optimisation on the smooth log‑sum‑exp surrogate.
         The Adam runs respect the simplex (∑a = √(2 n)) via a soft‑max.
Phase 3 – projected sub‑gradient descent on the *exact* max‑convolution.
         After each projection the vector is sorted decreasingly – this
         implements the optimal (rearrangement) ordering for the centre
         convolution term.
Phase 4 – a targeted hill‑climb that always moves mass from the index
         with the largest sub‑gradient to the index with the smallest
         sub‑gradient (the direction that most reduces the current peak).
         A simple temperature schedule allows occasional uphill moves.

All phases share a global best sequence and respect the overall time
budget.  The final sequence is returned as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of two real vectors."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negatives caused by round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending sort
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max (log‑it) parametrisation
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.20,
    min_tau_factor: float = 0.001,
    alpha0: float = 0.03,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth (log‑sum‑exp) surrogate of max(conv).
    Returns the best coefficient vector encountered and its exact
    evaluation (via evaluate_sequence).
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # ----- soft‑max mapping (guarantees positivity & sum) -----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)           # stable
        p = exp_shift / np.sum(exp_shift)       # simplex probabilities
        a = target_sum * p                      # actual coefficients

        # ----- exact max of the convolution (for bookkeeping) -----
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] n={n}   new Adam best max={cur_max:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        tau = max(init_tau_factor * cur_max * (0.99 ** it),
                  min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t. a (2 · a ⊛ w_rev) -----
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # ----- back‑prop to logits (soft‑max chain rule) -----
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slowly decay the base learning rate
        alpha *= 0.9995

    # final exact evaluation (identical to the stored best_max)
    final_score = evaluate_sequence(best_a.tolist())
    return best_a, float(final_score)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the *true* max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on max(conv(a, a)) with projection onto the simplex.
    After each projection the vector is sorted decreasingly (optimal ordering).
    Returns the best sequence and its maximal‑conv value.
    """
    a = a0.copy()
    n = a.shape[0]

    # keep the best seen
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step_size = 0.1 * target_sum / n   # rough first step

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())

        # sub‑gradient of conv_k w.r.t a
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # descent step and simplex + monotone projection
        a_candidate = a - step_size * grad
        a_candidate = _project_onto_simplex(a_candidate, target_sum)
        a_candidate = np.sort(a_candidate)[::-1]   # enforce decreasing order

        val_candidate = _full_convolution_fft(a_candidate, a_candidate).max()

        # accept if we improve, otherwise shrink step
        if val_candidate < best_val - 1e-12:
            a = a_candidate
            best_a = a_candidate
            best_val = val_candidate
            step_size *= 1.02
        else:
            step_size *= 0.70

        # occasional tiny jitter
        if it % 300 == 0 and np.random.rand() < 0.15:
            jitter = np.random.rand(n) * 1e-4
            a = _project_onto_simplex(a + jitter, target_sum)
            a = np.sort(a)[::-1]

        if it % 400 == 0:
            print(
                f"[{time.time():.2f}] subgrad it={it} step={step_size:.3e} best={best_val:.6f}"
            )
        if best_val < 1.50:          # early exit threshold
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Gradient‑guided hill‑climb on the exact objective
# ----------------------------------------------------------------------
def _gradient_hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Stochastic local search that repeatedly moves a small amount of mass from the
    index with the largest sub‑gradient (i.e. the most “responsible” for the current
    peak) to the index with the smallest sub‑gradient.  A simple temperature
    schedule permits occasional uphill moves.
    Returns the best sequence encountered and its maximal‑conv value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    start = time.time()
    T0 = best_val * 0.3
    T_end = 1e-5

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())

        # sub‑gradient of the active convolution peak
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # source = index with largest *positive* gradient and non‑zero mass
        src_candidates = np.where((grad > 0) & (a > 0))[0]
        if src_candidates.size == 0:
            # fall back to a random pair
            i, j = np.random.randint(0, n, size=2)
        else:
            i = src_candidates[np.argmax(grad[src_candidates])]

            # target = index with smallest gradient (could be negative)
            j = np.argmin(grad)

        if i == j or a[i] <= 0:
            continue

        # mass to move: up to 8 % of the source amount
        delta = a[i] * np.random.rand() * 0.08
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        val_new = _full_convolution_fft(a_new, a_new).max()

        # temperature decreasing linearly with elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - frac))

        accept = False
        if val_new < best_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(val_new - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if val_new < best_val:
                best_val = val_new
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")

        # keep a tidy shape – occasionally re‑sort decreasingly
        if it % 2000 == 0:
            a = np.sort(a)[::-1]
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.50:
            break

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid global‑local search for a step‑function with a minimal evaluation
    score.  The routine respects the given time budget and combines:
       * random analytic constructions,
       * a few longer Adam runs (smooth surrogate),
       * projected sub‑gradient descent with monotone projection,
       * a gradient‑guided hill‑climb.
    It returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Warm‑start from a globally available construction, if any.
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.asarray(hs, dtype=float)
        n0 = hs_arr.shape[0]
        target0 = np.sqrt(2.0 * n0)
        hs_norm = hs_arr * (target0 / hs_arr.sum())
        score0 = evaluate_sequence(hs_norm.tolist())
        best_seq = hs_norm.copy()
        best_score = float(score0)
        print(f"[{time.time():.2f}] warm‑start seed score={best_score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – cheap random sampling of analytic families.
    # --------------------------------------------------------------
    # leave ~250 s for the later, more expensive stages
    fast_phase_deadline = deadline - 250.0
    while time.time() < fast_phase_deadline:
        # pick a length (gradually favour shorter runs as we approach the deadline)
        remaining = deadline - time.time()
        if remaining > 600:
            n = np.random.randint(1500, 3500 + 1)
        elif remaining > 300:
            n = np.random.randint(1300, 2800 + 1)
        else:
            n = np.random.randint(900, 2300 + 1)

        target_sum = np.sqrt(2.0 * n)

        # --- choose a construction pattern ---------------------------------
        pattern = np.random.choice(["dirichlet", "power", "exp", "mixed", "ends"])
        if pattern == "dirichlet":
            v = np.random.gamma(1.0, 1.0, size=n)          # Dirichlet(1)
            v = np.sort(v)[::-1]                           # decreasing
        elif pattern == "power":
            p = np.random.uniform(0.3, 1.5)
            idx = np.arange(1, n + 1, dtype=float)
            v = idx ** (-p)
        elif pattern == "exp":
            tau = np.random.uniform(0.2, 0.8) * n
            idx = np.arange(n, dtype=float)
            v = np.exp(-idx / tau)
        elif pattern == "mixed":
            p = np.random.uniform(0.3, 1.2)
            tau = np.random.uniform(0.2, 0.8) * n
            idx = np.arange(n, dtype=float)
            v = (idx + 1) ** (-p) * np.exp(-idx / tau)
        else:                           # "ends" – large values placed at the ends
            tmp = np.random.gamma(1.0, 1.0, size=n)
            tmp = np.sort(tmp)[::-1]                     # decreasing list of magnitudes
            v = np.zeros(n, dtype=float)
            left, right = 0, n - 1
            it_idx = 0
            while left <= right:
                v[left] = tmp[it_idx]
                it_idx += 1
                left += 1
                if left <= right:
                    v[right] = tmp[it_idx]
                    it_idx += 1
                    right -= 1

        # make sure we have a proper vector (non‑negative, non‑zero sum)
        v = np.maximum(v, 0.0)
        if v.sum() == 0.0:
            continue
        a = v * (target_sum / v.sum())                     # rescale to the simplex

        # quick exact evaluation (no clipping needed as values stay far below 1000)
        conv = _full_convolution_fft(a, a)
        maxb = conv.max()
        score = 2.0 * n * maxb / (a.sum() ** 2)

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time():.2f}] new random best={best_score:.6f} "
                f"(n={n}, pattern={pattern})"
            )
            if best_score < 1.503:          # strong early exit
                break

    # --------------------------------------------------------------
    # Phase 2 – a few longer Adam runs (soft‑max parametrisation)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    adam_runs = 4
    for r in range(adam_runs):
        if time.time() >= deadline:
            break
        # length: either keep the current best length or try a fresh one
        if best_seq is not None and np.random.rand() < 0.5:
            n = best_seq.shape[0]
            init_a = best_seq.copy()
        else:
            n = np.random.randint(1500, 5000 + 1)
            init_a = np.random.rand(n)

        target_sum = np.sqrt(2.0 * n)
        a0 = init_a[:n] if init_a.shape[0] >= n else np.concatenate(
            [init_a, np.random.rand(n - init_a.shape[0])]
        )
        a0 = a0 / a0.sum() * target_sum
        x0 = np.log(a0 + 1e-12)

        run_deadline = time.time() + min(30.0, remaining * 0.3)
        seq_local, score_local = _optimise_adam_softmax(
            n,
            x0,
            target_sum,
            run_deadline,
            init_tau_factor=0.15,
            min_tau_factor=0.0005,
            alpha0=0.04,
        )
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] Adam improvement → best={best_score:.6f} (run {r+1})"
            )
        remaining = deadline - time.time()
        if remaining < 30.0:
            break

    # --------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent with monotone projection
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.5)
        seq_sub, max_conv_sub = _subgradient_descent(best_seq.copy(), target_sum, sub_deadline)

        # an extra descending sort almost always reduces the centre term
        seq_sub = np.sort(seq_sub)[::-1]
        max_conv_sub = _full_convolution_fft(seq_sub, seq_sub).max()
        score_sub = 2.0 * n * max_conv_sub / (seq_sub.sum() ** 2)

        if score_sub < best_score:
            best_score = score_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] sub‑gradient improvement → best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – gradient‑guided hill‑climb (exact objective)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        hill_deadline = deadline
        seq_hill, max_conv_hill = _gradient_hillclimb(
            best_seq.copy(), target_sum, hill_deadline
        )
        score_hill = 2.0 * n * max_conv_hill / (seq_hill.sum() ** 2)
        if score_hill < best_score:
            best_score = score_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] hill‑climb improvement → best={best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback – uniform distribution (unlikely to be used)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 1200
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    print(
        f"[{time.time():.2f}] FINAL best score={best_score:.6f}, length={len(best_seq)}"
    )
    # return a plain Python list as required by the evaluator
    return best_seq.tolist()
