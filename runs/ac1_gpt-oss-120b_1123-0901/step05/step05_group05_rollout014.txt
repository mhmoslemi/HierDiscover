"""
Hybrid optimizer for the minimisation of

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The evaluation is scale‑invariant; we keep the ℓ₁‑norm
∑aᵢ = √(2 n) so the score equals max(conv(a,a)).
The method consists of

1.  quick random warm‑starts,
2.  many short Adam runs on a soft‑max parametrisation (smooth max via log‑sum‑exp),
3.  a prolonged projected sub‑gradient descent on the exact max,
4.  a temperature‑controlled stochastic hill‑climb,
5.  optional length‑extension of the best candidate and a final refinement.

All steps respect a global time‑budget (default 1000 s) and keep the vector
non‑negative, normalised to the simplex.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used everywhere)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of two real vectors."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())            # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negatives caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σw = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)                # 1‑based indices
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax parametrisation, using a smooth max
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.20,
    min_tau_factor: float = 0.001,
    alpha0: float = 0.03,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser for the log‑sum‑exp surrogate of the max convolution."""
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # ----- soft‑max mapping to the simplex (normalised sum) -----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)               # stable soft‑max
        p = exp_shift / np.sum(exp_shift)           # Σp = 1
        a = target_sum * p                           # ℓ₁‑norm = target_sum

        # ----- exact max of the convolution (for bookkeeping) -----
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] n={n}   new local best max={cur_max:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        tau = max(init_tau_factor * cur_max * (0.99 ** it),
                  min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t. a (2·a⊛w_rev) -----
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # ----- back‑propagation to logits -----
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slowly decay the base learning rate
        alpha *= 0.9995

    final_score = evaluate_sequence(best_a.tolist())
    return best_a, float(final_score)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the *true* max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on max(conv(a, a)) with simplex projection."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # start with a relatively aggressive step
    step_size = 0.3 * target_sum / n

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())               # index of the present maximum

        # sub‑gradient of the active convolution entry
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # take a step and project back onto the simplex
        a_candidate = a - step_size * grad
        a_candidate = _project_onto_simplex(a_candidate, target_sum)

        val_candidate = _full_convolution_fft(a_candidate, a_candidate).max()

        # accept improvements, otherwise shrink the step
        if val_candidate < best_val - 1e-12:
            a = a_candidate
            best_a = a_candidate
            best_val = val_candidate
            step_size *= 1.02               # slight increase after success
        else:
            step_size *= 0.70               # more aggressive reduction on failure

        # occasional tiny random jitter to keep the dynamics alive
        if it % 300 == 0 and np.random.rand() < 0.15:
            jitter = np.random.rand(n) * 1e-4
            a = _project_onto_simplex(a + jitter, target_sum)

        if it % 500 == 0:
            print(
                f"[{time.time():.2f}] subgrad it={it} step={step_size:.3e} best={best_val:.6f}"
            )
        # early exit if we already beat the target
        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (simulated annealing) on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Random local search moving a small amount of mass between two indices."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # temperature schedule – start relatively hot, cool to near zero
    T0 = best_val * 0.25
    T_end = 1e-4
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue

        # transfer at most 10 % of the source mass
        delta = a[i] * np.random.rand() * 0.10
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()

        if new_val < best_val:
            accept = True
        else:
            accept = np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12))

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")

        # occasional re‑normalisation to avoid drift due to floating point errors
        if it % 2000 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """Hybrid search returning a non‑negative step‑function that minimises the score."""
    np.random.seed(seed)
    start_time = time.time()
    margin = 0.5                         # tiny safety margin
    deadline = start_time + budget_s - margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Warm‑start from a globally supplied construction, if present.
    # --------------------------------------------------------------
    try:

        hs_arr = np.array(hs, dtype=float)
        n0 = hs_arr.shape[0]
        target0 = math.sqrt(2.0 * n0)
        hs_norm = hs_arr * (target0 / hs_arr.sum())
        score0 = evaluate_sequence(hs_norm.tolist())
        best_seq = hs_norm.copy()
        best_score = float(score0)
        print(f"[{time.time():.2f}] warm‑start seed score={best_score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 0 – quick random sampling (cheap baselines)
    # --------------------------------------------------------------
    time_for_random = min(30.0, (deadline - time.time()) * 0.1)
    while time.time() < start_time + time_for_random:
        n = np.random.randint(800, 3000)
        target = math.sqrt(2.0 * n)

        # random Dirichlet‑like start (Gamma with random concentration)
        alpha = np.random.uniform(0.2, 5.0)
        a = np.random.gamma(alpha, 1.0, size=n)
        a = a / a.sum() * target

        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = float(score)
            best_seq = a.copy()
            print(f"[{time.time():.2f}] random baseline n={n} score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – many Adam runs with diverse initialisations
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 15.0:
            break

        # allocate a modest slice for a single Adam run
        slice_time = min(30.0, remaining * 0.5)
        run_deadline = time.time() + slice_time

        # pick a problem size, biasing towards cheaper runs when time is short
        if remaining > 400:
            n = np.random.randint(1500, 3500)
        elif remaining > 200:
            n = np.random.randint(1200, 3000)
        else:
            n = np.random.randint(800, 2500)

        target_sum = math.sqrt(2.0 * n)

        # ----------- initialise a -------------
        if np.random.rand() < 0.6:
            a0 = np.random.rand(n)
        else:
            # power‑law / exponential blend
            p_exp = np.random.uniform(0.2, 1.2)
            idx = np.arange(1, n + 1, dtype=float)
            a0 = idx ** (-p_exp)
            if np.random.rand() < 0.5:
                tau_exp = np.random.uniform(0.5, 2.0) * n
                a0 *= np.exp(-idx / tau_exp)

        a0 = a0 / a0.sum() * target_sum
        x0 = np.log(a0 + 1e-12)

        # occasionally inject the current global best (trim / pad as needed)
        if best_seq is not None and np.random.rand() < 0.35:
            if best_seq.shape[0] >= n:
                a0 = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.shape[0]) * 1e-5
                a0 = np.concatenate([best_seq, pad])
            a0 = a0 / a0.sum() * target_sum
            x0 = np.log(a0 + 1e-12)

        seq_local, score_local = _optimise_adam_softmax(
            n,
            x0,
            target_sum,
            run_deadline,
            init_tau_factor=0.15,
            min_tau_factor=0.001,
            alpha0=0.05,
        )

        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL NEW BEST after Adam n={n} score={best_score:.6f}"
            )
            if best_score < 1.503:
                break

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on the exact max
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 10.0:
        n = best_seq.shape[0]
        target_sum = math.sqrt(2.0 * n)
        sub_deadline = time.time() + min(40.0, time_left * 0.6)
        seq_sub, val_sub = _subgradient_descent(best_seq.copy(), target_sum, sub_deadline)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑gradient best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – stochastic hill‑climb for fine‑tuning
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 5.0:
        n = best_seq.shape[0]
        target_sum = math.sqrt(2.0 * n)
        hill_deadline = time.time() + time_left
        seq_hill, val_hill = _hillclimb(best_seq.copy(), target_sum, hill_deadline)
        if val_hill < best_score:
            best_score = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] after hill‑climb best={best_score:.6f}")

    # --------------------------------------------------------------

        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # Return a plain Python list as required by the evaluator
    return best_seq.tolist()
