def build_candidates(n, best_seq):
    candidates = []
    # warm start from best of same length
    if best_seq is not None and len(best_seq) == n:
        candidates.append(np.asarray(best_seq, dtype=np.float64))
    # previous global variable height_sequence_1
    hseq = globals().get('height_sequence_1')
    if hseq is not None:

        if cand.shape[0] > n:
            cand = cand[:n]
        else:
            cand = np.pad(cand, (0, n - cand.shape[0]), 'constant')
        cand = np.maximum(cand, 0)
        cand = cand / cand.sum()
        candidates.append(cand)

    # edge-concentrated power law shape using index product
    idx = np.arange(1, n+1, dtype=np.float64)
    prod = idx * (n - idx + 1)

    for p in [0.5,1.0,1.5,2.0]:

def optimise_adam(a_start, n, fft_len, deadline):
    a = a_start.copy()
    # Prevent zeros causing log issue
    a = np.maximum(a, 1e-15)
    logits = np.log(a)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.04
    lam = 0.3
    lam_factor = 1.01
    # iterate
    max_iters = int(50000 * 1024 / n) + 8000
    best_local_val = float('inf')
    best_local_a = a.copy()
    last_improve = time.time()
    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # softmax
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()
        # conv
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        # smooth weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0: w = np.full_like(w, 1.0 / w.size)
        else: w /= w_sum
        # gradient
        grad_a = 2.0 * _cross_correlation(w, a, fft_len)
        # backprop through softmax
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
        lam = min(lam * lam_factor, 1e6)
        # evaluate current exact objective
        val = 2.0 * n * max_c
        if val < best_local_val:
            best_local_val = val
            best_local_a = a.copy()
            last_improve = time.time()
            # Optional print
        if time.time() - last_improve > 6.0:
            # random perturb for escape
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
    return best_local_a, best_local_val

    v = np.zeros_like(logits)

"""
Propose a non‑negative step‑function (list of floats) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

using a multi‑start strategy and a two‑phase optimisation:

* 1️⃣  Build many heuristic shapes (Dirichlet samples, edge‑concentrated
      power‑laws, two‑sided exponentials, mixed families).  Evaluate each
      shape with a *single* convolution and keep only the best few.
* 2️⃣  For the selected seeds run a smooth‑max Adam optimisation
      (soft‑max parametrisation → simplex) followed by an exact
      sub‑gradient (exponentiated‑gradient) phase and a very long
      fine‑tuning Adam run.
* 3️⃣  Finally perform a cheap random local‑search that transfers a tiny
      amount of mass between two positions while the time‑budget permits.

All heavy work is done with FFT‑based convolutions; the algorithm adapts
its iteration budget to the sequence length and respects the global time
budget (default 1000 s).  The best sequence found is returned as a plain
Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returned vector has length ``len(a)`` and equals

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    L = 1 << int(np.ceil(np.log2(w.shape[0] + n - 1)))
    fw = np.fft.rfft(w, L)
    fr = np.fft.rfft(a_rev, L)
    corr = np.fft.irfft(fw * fr, L)
    start = n - 1
    return corr[start:start + n]


def _softmax(logits: np.ndarray) -> np.ndarray:
    mx = logits.max()
    exp_x = np.exp(logits - mx)
    return exp_x / exp_x.sum()

def _build_candidates(n: int, warm_seq: np.ndarray | None) -> list[np.ndarray]:
    """Generate a diverse set of initial probability vectors of length n."""
    cand = []

    # warm start from a previously good sequence of the same length
    if warm_seq is not None and warm_seq.shape[0] == n:
        cand.append(warm_seq.copy())

    # historic global variable if present
    h = globals().get("height_sequence_1")
    if h is not None:
        h = np.asarray(h, dtype=np.float64)
        if h.shape[0] > n:
            h = h[:n]
        else:
            h = np.pad(h, (0, n - h.shape[0]))
        h = np.maximum(h, 0.0)
        if h.sum() < 0.01:
            h = np.random.dirichlet(np.ones(n))
        else:
            h /= h.sum()
        cand.append(h)

    # pure Dirichlet (flat)
    cand.append(np.random.dirichlet(np.ones(n)))

    # spiky Dirichlet – concentration < 1
    for conc in (0.15, 0.3, 0.6):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # edge‑concentrated power‑law shapes: (i+1)*(n‑i) raised to α<0
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for α in np.linspace(-3.0, -0.5, 10):
        shape = np.power(prod, α, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # two‑sided exponentials with different decay rates
    for γ in (0.0005, 0.001, 0.0025, 0.005, 0.01):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-γ * i) + np.exp(-γ * (n - 1 - i))
        shape /= shape.sum()
        cand.append(shape)

    # pure edge‑power law (no product term)
    for p in (0.5, 1.0, 1.5, 2.0):
        shape = (np.arange(1, n + 1, dtype=np.float64) ** (-p) +
                 (np.arange(1, n + 1, dtype=np.float64) ** (-p))[::-1])
        shape /= shape.sum()
        cand.append(shape)

    # random mixtures of a fast exponential and a slow power law
    for _ in range(4):
        α = np.random.uniform(-2.5, -0.5)
        γ = np.random.uniform(0.0003, 0.008)
        i = np.arange(n, dtype=np.float64)
        exp_part = np.exp(-γ * i) + np.exp(-γ * (n - 1 - i))
        power_part = np.power(prod, α, where=prod > 0, out=np.zeros_like(prod))
        mixture = 0.6 * exp_part + 0.4 * power_part
        if mixture.sum() > 0:
            mixture /= mixture.sum()
            cand.append(mixture)

    # ensure every candidate is a probability vector
    clean = []
    for c in cand:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        clean.append(c / s)
    return clean

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   fft_len: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """Adam optimisation of the smooth‑max surrogate of max(a*a)."""
    a = a0.copy()
    a = np.maximum(a, 1e-15)
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.04
    lam = 0.3               # smooth‑max temperature
    lam_factor = 1.01

    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()
    max_iters = int(50000 * 1024 / n) + 8000

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(logits)

        conv = _full_convolution(a, fft_len)
        max_c = conv.max()

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # increase temperature
        lam = min(lam * lam_factor, 1e6)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # occasional shake‑up if stuck
        if time.time() - last_improve > 8.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          fft_len: int,
                          deadline: float,
                          current_best_val: float) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on max(a*a) using exponentiated gradient."""
    a = a0.copy()
    eta = 0.55
    max_iters = 5000
    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)

        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a)

        a = a * np.exp(-eta * grad_a)        # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12:                        # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        if val < current_best_val:
            # early exit – we have already beaten the global record
            break

        if time.time() - last_improve > 8.0:
            break

    return best_a, best_val


def _fine_tune(a0: np.ndarray,
               n: int,
               fft_len: int,
               deadline: float) -> tuple[np.ndarray, float]:
    """A long, low‑learning‑rate Adam refinement."""
    a = a0.copy()
    a = np.maximum(a, 1e-15)
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.02
    lam = 0.5
    lam_factor = 1.005

    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()
    max_iters = int(120000 * 1024 / n) + 10000

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(logits)

        conv = _full_convolution(a, fft_len)
        max_c = conv.max()

        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

        grad_a = 2.0 * _cross_correlation(w, a)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        lam = min(lam * lam_factor, 1e6)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        if time.time() - last_improve > 12.0:
            logits += (np.random.rand(n) - 0.5) * 5e-3
            last_improve = time.time()

    return best_a, best_val


def _random_local_search(a_start: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a small fraction of mass from one index
    to another while time permits.  The move is accepted only if it
    improves the objective.
    """
    a = a_start.copy()
    conv = _full_convolution(a, fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    # step size (fraction of the source mass) – slowly adapts
    step_scale = 0.05

    while time.time() < deadline - 0.1:
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        delta = a[i] * step_scale * np.random.rand()
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        conv = _full_convolution(a_new, fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = a_new
            a = a_new
            # slightly increase step size when we are successful
            step_scale = min(step_scale * 1.02, 0.2)
        else:
            # otherwise slowly shrink the step
            step_scale *= 0.99

    return best_a, best_val

                break
            print(f"[n={n} seed={rank} init={init_val:.6f}]")
            # Adam on smooth‑max surrogate
            a_opt, val_opt = _adam_optimize(a0, n, fft_len, deadline)
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"  ▶ new global best after Adam: {best_val:.6f}")

            # Exact sub‑gradient (exponentiated gradient)
            a_opt, val_opt = _subgradient_optimize(a_opt, n, fft_len,
                                                   deadline, best_val)
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"  ▶ new global best after sub‑grad: {best_val:.6f}")

            # Long fine‑tune run
            a_opt, val_opt = _fine_tune(a_opt, n, fft_len, deadline)
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"  ▶ new global best after fine‑tune: {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Final cheap local‑search on the overall best sequence
    if best_seq is not None:
        n_best = best_seq.shape[0]
        fft_len_best = 1 << int(np.ceil(np.log2(2 * n_best - 1)))
        print(f"\n[final local search] best value so far {best_val:.6f}")
        best_seq, best_val = _random_local_search(best_seq, n_best,
                                                  fft_len_best, deadline)
        print(f"[final] final value {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Safety fallback – should never be needed
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return plain Python list as required by the external evaluator
    return best_seq.tolist()
