"""
Sub‑gradient + simplex projection + simulated‑annealing optimizer.

The objective for a non‑negative vector a (sum(a) = 1) is
    F(a) = 2·n·max(conv(a,a)),
where conv(a,a) is the linear convolution of a with itself.
We minimise the L∞‑norm of the auto‑convolution.

Algorithm
---------
1. Initialise a with a warm start (previous best, a triangular shape,
   a Dirichlet sample or the provided global `height_sequence_1`).
2. Repeatedly:
   * compute the convolution via FFT,
   * build a binary mask of the (approximately) maximal entries,
   * obtain the exact sub‑gradient of the max‑convolution
     (cross‑correlation of the mask with a),
   * take a gradient step on a and project back onto the probability
     simplex (Duchi et al. 2008).
   * learning‑rate is exponentially decayed; occasional random jitter
     (shake) is applied when progress stalls.
3. After the gradient phase, run a fast simulated‑annealing style
   local search that transfers a small amount of mass between two
   coordinates.  The convolution is updated incrementally using the
   analytical formula for a rank‑1 mass transfer, which costs O(n)
   per move.
4. Keep the best sequence seen over all restarts and lengths (1024,
   2048, 4096) until the wall‑time budget expires.

The whole routine respects the 1000 s limit; if no valid sequence is
found, a uniform fallback is returned.  The returned list can be fed to
`evaluate_sequence`.
"""

import time
import numpy as np


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x ≥ 0, Σ x = 1}
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list:
    """Search for a low‑value step‑function within the given time budget."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # safety margin

    best_seq = None
    best_val = float("inf")

    # candidate lengths – larger n potentially yields lower bounds
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT lengths that will be reused
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))  # ≥ (2n‑1)+n‑1

        # number of restarts for each size
        restarts = 8 if n <= 1024 else 5 if n <= 2048 else 3

        for r in range(restarts):
            if time.time() > deadline:
                break

            # --------‑ initialise a -------------------------------------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a = best_seq.copy()
                a = a / a.sum()
                print(f"[n={n} restart={r}] warm‑start from previous best")
            else:
                # try to use the provided global construction if possible
                height_seq = globals().get("height_sequence_1")
                if height_seq is not None:
                    cand = np.array(height_seq, dtype=np.float64)
                    if cand.shape[0] >= n:
                        a = cand[:n].copy()
                    else:
                        a = np.pad(cand, (0, n - cand.shape[0]), constant_values=0.0)
                    a = np.maximum(a, 0.0)
                    if a.sum() < 1e-6:
                        a = np.random.dirichlet(np.ones(n))
                    else:
                        a = a / a.sum()
                    print(f"[n={n} restart={r}] start from height_sequence_1")
                else:
                    # a simple triangular shape (often a good seed)
                    if r % 2 == 0:
                        idx = np.arange(n)
                        mid = (n - 1) / 2.0
                        a = np.maximum(0.0, 1.0 - np.abs(idx - mid) / (mid + 1))
                        a = a / a.sum()
                        print(f"[n={n} restart={r}] triangular seed")
                    else:
                        a = np.random.dirichlet(np.ones(n))
                        print(f"[n={n} restart={r}] random Dirichlet seed")

            # --------‑ hyper‑parameters --------------------------------------------
            lr = 0.12                     # initial step size
            min_lr = 5e-4                 # lower bound for learning rate
            decay_factor = 0.85
            decay_every = 4000
            max_iters = 180000 if n <= 1024 else 100000 if n <= 2048 else 70000
            last_improve = time.time()
            stagnation_limit = 30.0      # seconds without improvement before a shake

            # --------‑ gradient descent on the simplex -------------------------------
            for it in range(max_iters):
                if time.time() > deadline:
                    break

                # convolution and current objective
                conv = full_convolution(a, fft_len=fft_len_conv)
                cur_max = conv.max()
                cur_val = 2.0 * n * cur_max

                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} r={r} it={it:06d}] new best {cur_val:.6f}")

                # mask of (near) maximal entries – a tiny tolerance improves stability
                tol = 1e-12 * max(1.0, cur_max)
                mask = (conv >= cur_max - tol).astype(np.float64)

                # exact sub‑gradient of the max‑convolution
                grad = 2.0 * cross_correlation(mask, a, fft_len_corr)

                # gradient step followed by projection onto the simplex
                a_new = a - lr * grad
                a = simplex_projection(a_new)

                # learning‑rate decay
                if (it + 1) % decay_every == 0:
                    lr = max(lr * decay_factor, min_lr)

                # shake when we appear stuck
                if time.time() - last_improve > stagnation_limit:
                    jitter = (np.random.rand(n) - 0.5) * 5e-3
                    a = np.maximum(a + jitter, 0.0)
                    a = a / a.sum()
                    last_improve = time.time()
                    print(f"[n={n} r={r} it={it}] shake (stagnation)")

            # --------‑ local simulated‑annealing refinement -------------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a = best_seq.copy()
                conv = full_convolution(a, fft_len=fft_len_conv)
                cur_val = 2.0 * n * conv.max()

                # annealing schedule
                T_start = 0.02
                T_end = 1e-5
                # run until time runs out (or a reasonable iteration cap)
                sa_iter = 0
                while time.time() < deadline and sa_iter < 300000:
                    prog = sa_iter / 300000.0
                    T = T_start * (1.0 - prog) + T_end * prog

                    # propose a random mass transfer
                    i, j = np.random.choice(n, 2, replace=False)
                    eps = min(a[i], 0.15) * np.random.rand()   # move ≤15 % of source
                    if eps <= 0:
                        sa_iter += 1
                        continue

                    # incremental update of the convolution (O(n))
                    delta = np.zeros_like(conv)
                    delta[i:i + n] -= 2.0 * eps * a
                    delta[j:j + n] += 2.0 * eps * a
                    delta[2 * i] += eps * eps
                    delta[2 * j] += eps * eps
                    delta[i + j] -= 2.0 * eps * eps
                    conv_new = conv + delta

                    new_max = conv_new.max()
                    new_val = 2.0 * n * new_max

                    # Metropolis acceptance
                    accept = False
                    if new_val < cur_val:
                        accept = True
                    else:
                        if np.random.rand() < np.exp(-(new_val - cur_val) / (T + 1e-12)):
                            accept = True

                    if accept:
                        a[i] -= eps
                        a[j] += eps
                        conv = conv_new
                        cur_val = new_val
                        if new_val < best_val - 1e-12:
                            best_val = new_val
                            best_seq = a.copy()
                            print(f"[SA n={n} r={r} it={sa_iter}] improved to {new_val:.6f}")

                    # occasional re‑normalisation to counteract floating‑point drift
                    if (sa_iter + 1) % 10000 == 0:
                        a = a / a.sum()
                        conv = full_convolution(a, fft_len=fft_len_conv)

                    sa_iter += 1

    # -------------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution (objective = 2)
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = 2.0 * n_fallback * full_convolution(best_seq, fft_len=1 << int(np.ceil(np.log2(2 * n_fallback - 1)))).max()
        print("Fallback to uniform vector.")

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
