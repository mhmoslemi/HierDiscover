def _exponential_subgrad(p, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val):
    # p is half array sum 1.
    eta = 0.6
    last_improve = time.time()
    for it_sub in range(20000):
        if time.time() >= deadline: break
        # build full a:
        a = np.concatenate([p, p[::-1]])
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} subgrad it={it_sub}] new best {val:.6f}")
        # active set
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        if np.any(active):
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        grad_half = grad_a[:n//2] + grad_a[-(n//2):][::-1]
        # exponentiated gradient
        p = p * np.exp(-eta * grad_half)
        s = p.sum()
        if s < 1e-12:
            p = np.random.dirichlet(np.ones_like(p))
            s = p.sum()
        p = p / s
        if time.time() - last_improve > 5.0:
            break
    return best_seq, best_val

def _optimize_sym(n, init_half, deadline, best_seq, best_val):
    m = n // 2
    # compute FFT lengths
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # initialize logits
    eps = 1e-15
    logits = np.log(init_half + eps)
    # Adam variables
    m_adam = np.zeros_like(logits)
    v_adam = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.05

    lam = 0.1
    lam_factor = 1.01

    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < 300_000:
        # compute p via softmax
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        p = exp_log / exp_log.sum()
        a = np.concatenate([p, p[::-1]])
        # compute convolution
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # objective value
        val = float(2.0 * n * max_c / (a.sum() ** 2))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={it}] new best {val:.6f}")

        # compute smooth-max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum

        # gradient w.r.t a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # half gradient
        grad_half = grad_a[:m] + grad_a[-m:][::-1]

        # softmax gradient for logits
        avg = np.dot(p, grad_half)
        grad_log = p * (grad_half - avg)

        # Adam update
        m_adam = beta1 * m_adam + (1 - beta1) * grad_log
        v_adam = beta2 * v_adam + (1 - beta2) * (grad_log ** 2)
        m_hat = m_adam / (1 - beta1 ** (it + 1))
        v_hat = v_adam / (1 - beta2 ** (it + 1))

        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        if time.time() - last_improve > 5.0:
            # add small jitter
            logits += (np.random.rand(m) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    # Subgradient phase
    # start from p derived from current logits
    mx = logits.max()
    exp_log = np.exp(logits - mx)
    p = exp_log / exp_log.sum()
    # Possibly start from best half if more promising:
    if best_seq is not None:
        # half from best_seq
        best_half = (best_seq[:m] + best_seq[-m:][::-1]) / 2.0
        bh_sum = best_half.sum()
        if bh_sum > 0:
            p = best_half / bh_sum  # renorm to sum=1

    eta = 0.6
    sub_last_improve = time.time()
    for sub_it in range(20_000):
        if time.time() >= deadline:
            break
        a = np.concatenate([p, p[::-1]])
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c / (a.sum() ** 2))

            best_val = val
            best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[n={n} Subgrad it={sub_it}] new best {val:.6f}")

        # active set for max
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()

        if sum_p < 1e-12:

        p = p / sum_p

        if time.time() - sub_last_improve > 5.0:
            break

    return best_seq, best_val

"""
Propose a low‑score step function.

The algorithm works on a *symmetric* representation to halve the
dimensionality.  For a length‑n vector we store only the first n/2
weights p_i (p_i ≥ 0, Σp_i = 1) and mirror them:
        a = [p_0,…,p_{m‑1}, p_{m‑1},…,p_0]   (m = n//2)

Because the evaluation metric is scale‑invariant we can keep the total
mass of a equal to 2 (the sum of the mirrored halves) and never
renormalise during optimisation.

Two optimisation phases are performed for each target length:

1. **Smooth‑max Adam** – minimise a log‑sum‑exp surrogate of the
   convolution maximum.  The surrogate becomes the true max as the
   temperature λ grows.  Gradients are obtained via FFT‑based
   convolutions and the standard soft‑max ↔ logits Jacobian.

2. **Exact‑max sub‑gradient** – after the surrogate phase we switch to
   the true objective.  The gradient of the maximum convolution entry
   is obtained from an active‑set weight vector; an exponentiated‑gradient
   (multiplicative) update keeps the variables non‑negative.

The optimisation is run on a hierarchy of lengths (1024 → 2048 →
4096 → 8192 → 16384).  The best solution of a stage is linearly
interpolated to initialise the next, allowing a cheap multi‑scale
search while respecting the global time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _softmax(logits: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    mx = logits.max()
    exp = np.exp(logits - mx)
    return exp / exp.sum()


# ----------------------------------------------------------------------
def _optimize_sym(
    n: int,
    init_half: np.ndarray,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Optimise a symmetric sequence of length n.
    *init_half* is a probability vector of length n//2 (sum‑to‑1).
    Updates and returns the global best sequence/value.
    """
    m = n // 2
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ---- Adam on smooth‑max surrogate ---------------------------------
    eps = 1e-15
    logits = np.log(init_half + eps)
    m_adam = np.zeros_like(logits)
    v_adam = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.1               # temperature for log‑sum‑exp
    lam_factor = 1.01

    last_improve = time.time()
    it = 0
    max_adam_iters = 200_000

    while time.time() < deadline and it < max_adam_iters:
        # ----- map logits → probabilities → full symmetric vector -------
        p = _softmax(logits)                       # length m, sum‑to‑1
        a = np.concatenate([p, p[::-1]])          # length n, sum = 2

        # ----- convolution & exact objective (for logging) -------------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c / (a.sum() ** 2))

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={it:05d}] new best {val:.6f}")

        # ----- smooth‑max weights ---------------------------------------
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t. full a -----------------------------------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- reduce to half (mirrored) --------------------------------
        grad_half = grad_a[:m] + grad_a[-m:][::-1]

        # ----- soft‑max Jacobian (logits) -------------------------------
        avg = np.dot(p, grad_half)
        grad_log = p * (grad_half - avg)

        # ----- Adam update ---------------------------------------------
        m_adam = beta1 * m_adam + (1.0 - beta1) * grad_log
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m_adam / (1.0 - beta1 ** (it + 1))
        v_hat = v_adam / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule ------------------------------------
        lam = min(lam * lam_factor, 1e6)

        # ----- occasional jitter when stuck ---------------------------
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(m) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    # ---- Sub‑gradient phase on the true max ----------------------------
    # initialise p from the best half we have seen so far
    if global_best_seq is not None:
        best_half = (global_best_seq[:m] + global_best_seq[-m:][::-1]) / 2.0
        s = best_half.sum()
        if s > 0:
            p = best_half / s
        else:
            p = np.random.dirichlet(np.ones(m))
    else:
        p = _softmax(logits)

    eta = 0.6
    sub_last_improve = time.time()
    max_sub_iters = 30_000

    for sub_it in range(max_sub_iters):
        if time.time() >= deadline:
            break

        a = np.concatenate([p, p[::-1]])
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c / (a.sum() ** 2))

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[n={n} Subgrad it={sub_it:05d}] new best {val:.6f}")

        # ----- active set of maximal entries ----------------------------
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # ----- gradient of the true max -------------------------------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        grad_half = grad_a[:m] + grad_a[-m:][::-1]

        # ----- exponentiated‑gradient update (keeps p ≥ 0) -------------
        p = p * np.exp(-eta * grad_half)
        s = p.sum()
        if s < 1e-12:
            p = np.random.dirichlet(np.ones(m))
            s = p.sum()
        p = p / s

        if time.time() - sub_last_improve > 5.0:
            break

    return global_best_seq, global_best_val


# ----------------------------------------------------------------------
def _interpolate(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a sequence to a new length."""
    old_n = seq.shape[0]
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    return np.interp(x_new, x_old, seq)


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the
    evaluation score.  The routine respects the given time budget and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ---- optional warm‑start from the previous best -------------------
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)

    # ------------------------------------------------------------------
    # Multi‑scale schedule (increase resolution as time permits)
    stage_lengths = [1024, 2048, 4096, 8192, 16384]

    for n in stage_lengths:
        if time.time() >= deadline:
            break

        m = n // 2

        # ---- initialise half‑vector ------------------------------------
        if best_seq is not None:
            # upscale previous best
            seq_big = _interpolate(best_seq, n)
            half = (seq_big[:m] + seq_big[-m:][::-1]) / 2.0
            s = half.sum()
            if s > 0:
                half = half / s
            else:
                half = np.random.dirichlet(np.ones(m))
        elif warm_start is not None:
            # use the global warm‑start (interpolated to current size)
            seq_big = _interpolate(warm_start, n)
            half = (seq_big[:m] + seq_big[-m:][::-1]) / 2.0
            s = half.sum()
            if s > 0:
                half = half / s
            else:
                half = np.random.dirichlet(np.ones(m))
        else:
            half = np.random.dirichlet(np.ones(m))

        # ---- run the optimiser -----------------------------------------
        best_seq, best_val = _optimize_sym(
            n,
            init_half=half,
            deadline=deadline,
            global_best_seq=best_seq,
            global_best_val=best_val,
        )
        print(f"[stage n={n}] best value so far {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution if everything failed
    if best_seq is None:
        n_fallback = 4096
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return best_seq.tolist()
