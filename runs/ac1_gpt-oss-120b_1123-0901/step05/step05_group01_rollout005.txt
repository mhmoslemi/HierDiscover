    a = a0.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_val = objective(a, n, fft_len_conv)
    # step schedule: initial lr = 0.5 maybe, then decay.
    lr0 = 0.5
    for t in range(1, max_iters+1):
        if time.time() > deadline: break

        # compute conv and max
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        # active mask
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        # if no active (should not), break
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        # gradient wrt a: cross-corr
        grad = 2.0 * _cross_correlation(w, a, fft_len_corr)
        # may add gradient from denominator? Since we keep sum=1 via projection, not needed.

        # step schedule
        lr = lr0 / np.sqrt(t)
        a_new = a - lr * grad
        # project onto simplex
        a = _simplex_proj(a_new)

        # evaluate
        cur_val = objective(a, n, fft_len_conv)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

            # print progress

    return best_a, best_val

Implementation of simplex projection (Duchi et al.):

def _simplex_proj(v):
    # v can be any real vector.
    # shift to make sum=1 and non-negative.
    # returns projection onto simplex.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1 - cssv) / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‑value step‑function sequence.

The algorithm works in several stages:

1.  For a set of lengths (1024 … 16384) generate a diversified pool of
    warm‑starts (Dirichlet draws, edge‑blocks, power‑law, Gaussians, sparse
    spikes, etc.).  All candidates are normalised to sum = 1.

2.  **Smooth‑max Adam** – optimisation of a soft‑max surrogate of
    max (a∗a).  This quickly reduces the peak of the convolution while
    staying inside the probability simplex.

3.  **Exact projected‑gradient descent** – a sub‑gradient descent on the
    true objective 2·n·max(conv)/(∑a)².  The gradient of the maximum
    convolution entry is obtained by a cross‑correlation (FFT‑based).  After
    each step the iterate is projected onto the simplex (entropic mirror‑
    descent is not required; a Euclidean projection is cheap and stable).

4.  **Cheap hill‑climbing** – random pairwise mass transfers are tried;
    a move is kept if it improves the objective.  Occasionally an
    uphill move is accepted to escape tiny plateaus.

5.  The globally best sequence seen so far is refined once more with a
    longer Adam run if time permits.

All heavy work (convolution, cross‑correlation) is performed with FFTs.
The routine respects a global time‑budget (default 1000 s) and returns a
plain Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)

def _next_pow2(x: int) -> int:
    """Small utility: smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) (= n).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)²."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    # numerical safety – enforce exact sum=1
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _adam_optimize(a0: np.ndarray,
                   n: int,
                   fft_len_conv: int,
                   fft_len_corr: int,
                   deadline: float,
                   max_iters: int) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate of max (a∗a).
    Returns the best (a, value) before the deadline.
    """
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.04
    lam = 0.5
    lam_factor = 1.03

    best_a = a.copy()
    best_val = _objective(a, n, fft_len_conv)

    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # simplex via soft‑max
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # smooth‑max weights
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        ws = w.sum()
        if ws == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws

        # gradient of surrogate (by chain rule)
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        cur_val = _objective(a, n, fft_len_conv)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t:06d}] new best {cur_val:.6f}")

        # random shake if stagnating
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


def _pgd_optimize(a0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_corr: int,
                  deadline: float,
                  max_iters: int = 4000) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the exact objective.
    Uses the sub‑gradient of max (a∗a) and Euclidean simplex projection.
    """
    a = a0 / a0.sum()
    best_a = a.copy()
    best_val = _objective(a, n, fft_len_conv)

    # step‑size schedule – start relatively high then decay as 1/√t
    lr0 = 0.6
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if not active.any():
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        # sub‑gradient of the maximal convolution entry
        grad = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # step size
        lr = lr0 / np.sqrt(it)
        a_new = a - lr * grad
        # projection onto the simplex
        a = _proj_simplex(a_new)

        cur_val = _objective(a, n, fft_len_conv)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} PGD it={it}] new best {cur_val:.6f}")

        # occasional random perturbation if stuck
        if time.time() - last_improve > 4.0:
            a = _proj_simplex(a + 0.02 * (np.random.rand(n) - 0.5))
            last_improve = time.time()

    return best_a, best_val


def _hill_climb_optimize(a0: np.ndarray,
                         n: int,
                         fft_len_conv: int,
                         deadline: float,
                         init_val: float | None = None,
                         max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Simple cheap local search: move a small amount of mass between two
    random positions and keep the move if it improves the objective.
    Occasionally accepts an uphill move to escape tiny plateaus.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_conv) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2:
            continue

        # transfer up to 5 % of mass from i1 to i2
        eps = np.random.rand() * 0.05 * a[i1]
        if eps <= 0.0:
            continue
        new_a = a.copy()
        new_a[i1] -= eps
        new_a[i2] += eps
        if new_a[i1] < -1e-15:
            continue
        new_a /= new_a.sum()

        new_val = _objective(new_a, n, fft_len_conv)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            if it % 50 == 0:
                print(f"[n={n} HillClimb it={it}] new best {new_val:.6f}")
        else:
            # tiny chance to accept uphill move
            if np.random.rand() < 0.001:
                a = new_a

    return best_a, best_val


def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence by linear interpolation and renormalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new


def _generate_initial_candidates(n: int,
                                 best_seq: np.ndarray | None,
                                 hist_seq: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a diverse set of start vectors for length n.
    All candidates are non‑negative and sum to 1.
    """
    cand = []

    # warm‑starts from best known (exact or interpolated)
    if best_seq is not None:
        if best_seq.shape[0] == n:
            cand.append(best_seq.copy())
        else:
            cand.append(_interpolate_seq(best_seq, n))

    # historic sequence if supplied
    if hist_seq is not None:
        if hist_seq.shape[0] > n:
            tmp = hist_seq[:n]
        else:
            tmp = np.pad(hist_seq, (0, n - hist_seq.shape[0]))
        tmp = np.maximum(tmp, 0.0)
        tot = tmp.sum()
        if tot < 1e-12:
            tmp = np.random.dirichlet(np.ones(n))
        else:
            tmp = tmp / tot
        cand.append(tmp)

    # pure Dirichlet with various concentration parameters
    cand.append(np.random.dirichlet(np.ones(n)))                     # uniform
    cand.append(np.random.dirichlet(np.full(n, 0.3)))                # spiky
    cand.append(np.random.dirichlet(np.full(n, 0.7)))                # medium‑spiky

    # edge‑concentrated power‑law ((i+1)*(n-i))**α , α<0
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 4):
        shape = np.power(prod, alpha, where=prod > 0,
                         out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # two‑edge block shapes (rectangular windows at both ends)
    for frac in (0.03, 0.06, 0.10, 0.15):
        k = max(1, int(frac * n))
        shape = np.zeros(n)
        shape[:k] = 1.0
        shape[-k:] = 1.0
        cand.append(shape / shape.sum())

    # sparse random spikes (few non‑zero entries)
    for spikes in (2, 4, 6):
        shape = np.zeros(n)
        idxs = np.random.choice(n, spikes, replace=False)
        shape[idxs] = np.random.rand(spikes) + 0.1
        cand.append(shape / shape.sum())

    # two‑Gaussian edge shapes
    for gamma in (0.2, 0.5, 1.0):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        cand.append(shape / shape.sum())

    # three‑Gaussian interior peaks
    for sigma in (0.05, 0.12, 0.20):
        x = np.arange(n, dtype=np.float64) / (n - 1)
        peaks = sum(np.exp(-0.5 * ((x - p) / sigma) ** 2) for p in (0.25, 0.5, 0.75))
        cand.append(peaks / peaks.sum())

    # random Beta‑distribution shapes (skewed towards edges or centre)
    for a, b in ((0.5, 5.0), (5.0, 0.5), (2.0, 2.0), (0.8, 0.8)):
        t = np.arange(1, n + 1) / (n + 1)
        shape = (t ** (a - 1)) * ((1 - t) ** (b - 1))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    np.random.shuffle(cand)
    # keep a moderate number to stay within the time budget
    return cand[:20]

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    best_seq = None
    best_val = float("inf")

    # Lengths to explore – start modest, finish with the largest allowed.
    lengths = [1024, 2048, 4096, 8192, 16384]

    # Pre‑compute FFT lengths (power‑of‑two) for every n.
    conv_fft = {}
    corr_fft = {}
    for n in lengths:
        conv_len = 2 * n - 1
        conv_fft[n] = _next_pow2(conv_len)
        # cross‑correlation needs length conv_len + n - 1 ≤ 3·n
        corr_fft[n] = _next_pow2(conv_len + n - 1)

    # Load a historic solution if present.
    hist_seq = None
    if "height_sequence_1" in globals():
        tmp = globals()["height_sequence_1"]
        if isinstance(tmp, (list, tuple, np.ndarray)):
            hist_seq = np.asarray(tmp, dtype=np.float64)

    # ------------------------------------------------------------------
    # MAIN optimisation loop over the chosen lengths
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        fft_len_c = conv_fft[n]
        fft_len_x = corr_fft[n]

        # Diverse warm‑starts
        candidates = _generate_initial_candidates(n, best_seq, hist_seq)

        # Number of Adam iterations – more for small n, fewer for large n.
        max_adam_iters = int(14000 * 1024 / n) + 2500

        for cand_id, a_start in enumerate(candidates):
            if time.time() > deadline:
                break

            # ---- Phase 1 – smooth‑max Adam ----
            a_adam, val_adam = _adam_optimize(
                a_start, n, fft_len_c, fft_len_x, deadline, max_adam_iters
            )

            # ---- Phase 2 – exact projected‑gradient descent ----
            a_pgd, val_pgd = _pgd_optimize(
                a_adam, n, fft_len_c, fft_len_x, deadline,

            )

            # ---- Phase 3 – cheap hill‑climb ----
            a_hc, val_hc = _hill_climb_optimize(
                a_pgd, n, fft_len_c, deadline, init_val=val_pgd, max_iters=1500
            )

            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.copy()
                print(f"*** Global new best (n={n} cand={cand_id}) value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional long‑run refinement on the current global best
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        fft_len_c = conv_fft.get(n, _next_pow2(2 * n - 1))
        fft_len_x = corr_fft.get(n, _next_pow2(3 * n))
        print("\n=== Long‑run refinement of the global best ===")
        refined_seq, refined_val = _adam_optimize(
            best_seq, n, fft_len_c, fft_len_x, deadline,
            max_iters=int(1.6 * (14000 * 1024 / n) + 3500)
        )
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq.copy()
            print(f"*** Refined global best value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (unlikely)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list for the external evaluator.
    return best_seq.tolist()
