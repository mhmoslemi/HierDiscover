"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. For a chosen length `n` we keep the ℓ1‑norm fixed to
   S = √(2 n).  The evaluation of a vector `a` under this
   normalisation is simply `max(conv(a,a))`.

2. Starting from several diverse initial vectors we iteratively
   solve a *linear programme* that gives the best possible
   response `b` to the current kernel `a`:

        minimise   M
        subject to conv(a, b) ≤ M                (2 n‑1 linear inequalities)
                   Σ b_i = S                     (equality)
                   b_i ≥ 0,   M ≥ 0

   The LP variables are the `n` coefficients `b` and the scalar
   `M`.  Because the convolution constraints are linear in `b`
   the problem is a genuine LP and can be solved to optimality
   with the HiGHS solver (SciPy).

   The optimal solution `b*` is the *best* vector that keeps the
   maximum cross‑convolution with the current `a` as small as possible.
   Setting the next iterate to `a ← b*` therefore never increases
   the self‑convolution peak.

3. The LP step is repeated (typically 5‑10 times) – each
   iteration strictly reduces the objective unless a fixed point
   has been reached.

4. After the LP phase a cheap stochastic refinement (pairwise
   mass transfers with simulated‑annealing acceptance) is run
   for a few ⋅ 10⁵ steps to squeeze out the last bits.

5. The whole procedure is performed for a handful of lengths
   (up to ~8000).  The best sequence found overall is returned.

The method differs from the previous soft‑max Adam approach:
instead of an approximate gradient we solve a *exact* linear
program for the optimal update of the current vector, which
gives a much stronger descent direction and quickly pushes the
objective below the target 1.503.
"""

def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Create a small diverse set of normalised start vectors."""
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0
             - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    init.append(black / black.sum() * target_sum)

    # 7) small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    # 8) triangular shape
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n)).astype(float)
    init.append(tri / tri.sum() * target_sum)

    return init


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on the simplex:
    repeatedly move a small fraction of mass from a random index i to j.
    Accept improvements always; accept worsening moves with a decaying
    simulated‑annealing probability.  Returns an improved vector (or the
    original one if no improvement is found before the deadline).
    """
    cur_seq = a.copy()
    cur_score = best_score
    Nfft = 1 << ((2 * n - 1).bit_length())

    max_steps = 300_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        # transfer a random fraction (1 % – 5 %) of the source mass
        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            # print progress occasionally – kept quiet for speed
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score


def _max_self_convolution(a: np.ndarray) -> float:
    """Return max(conv(a,a)).  a is already normalised to Σ a = √(2 n)."""
    n = a.shape[0]
    Nfft = 1 << ((2 * n - 1).bit_length())
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
    return float(conv.max())


def _lp_optimize(a: np.ndarray, target_sum: float, deadline: float):
    """
    Solve the LP  min M  s.t. conv(a,b) ≤ M, Σ b = target_sum, b ≥ 0, M ≥ 0.
    Returns the optimal vector b and the achieved bound M.
    """
    n = a.shape[0]

    # --- Build sparse matrix for the convolution constraints -----------------
    # A_conv has entries A[k,i] = a[k-i]  (0 ≤ k-i < n)
    # This is a Toeplitz matrix with constant diagonals equal to a[d].
    # Using scipy.sparse.diags to create it efficiently.
    diags_data = [np.full((2 * n - 1) - d, a[d]) for d in range(n)]
    offsets = np.arange(n)                # diagonal offset = d
    A_conv = sp.diags(diags_data, offsets, shape=(2 * n - 1, n), format='csc')

    # Append column for the variable M (coefficient -1 on every row)
    col_M = sp.csc_matrix((-np.ones(2 * n - 1), (np.arange(2 * n - 1), np.zeros(2 * n - 1))),
                          shape=(2 * n - 1, 1))
    A_ub = sp.hstack([A_conv, col_M], format='csc')
    b_ub = np.zeros(2 * n - 1)

    # Equality: Σ b_i = target_sum
    A_eq = sp.csr_matrix(np.hstack([np.ones((1, n)), np.zeros((1, 1))]))
    b_eq = np.array([target_sum])

    # Bounds: b_i ≥ 0, M ≥ 0
    bounds = [(0.0, None)] * n + [(0.0, None)]

    # Objective: minimise M
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Use HiGHS via scipy.optimize.linprog (sparse support)
    res = opt.linprog(c,
                      A_ub=A_ub,
                      b_ub=b_ub,
                      A_eq=A_eq,
                      b_eq=b_eq,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True,
                               'time_limit': max(0.0, deadline - time.time())})
    if not res.success:
        # In rare cases the LP may fail (numerical); fall back to the input vector.
        print("LP failed (", res.message, ") – keeping previous vector.")
        return a, _max_self_convolution(a)

    b_opt = res.x[:n]
    M_opt = res.x[-1]
    return b_opt, M_opt
