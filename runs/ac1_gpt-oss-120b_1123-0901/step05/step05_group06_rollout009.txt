"""
Propose a non‑negative step function minimising ``evaluate_sequence``.

Algorithm
---------
* Maintain a global best sequence.
* Perform many short optimisation runs with different lengths (300 – 8000)
  and random hyper‑parameters.
* Each run uses a mirror‑descent/Adam optimiser on the log‑it (softmax)
  representation of the vector, with a temperature‑annealed soft‑max
  approximation of the max‑convolution (the objective after normalisation).
* Initialise runs from a diverse set of patterns (uniform, Dirichlet,
  exponential windows, Hann/Blackman, random perturbations) and also from
  the current global best interpolated to the required length.
* After the time budget for the Adam phase expires, a cheap stochastic
  local‑refinement (pairwise mass transfers with simulated‑annealing
  acceptance) is applied to the best sequence.
* The resulting list of heights is returned.
"""

import time
import math
import numpy as np

# Helper functions (all top‑level, no closures)

def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of convolution values.
    Returns (weights, smoothed max).
    """
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    # subtract max for numerical stability
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """Compute rev_i = Σ_k w_k * a_{k-i} (out‑of‑range ignored) using FFTs."""
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)          # reverse w before FFT
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # keep the linear part of the convolution (length = 3 n‑2)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Return max_k (a∗a)[k] using an FFT of length Nfft."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Create a small diverse set of normalised start vectors."""
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0
             - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    init.append(black / black.sum() * target_sum)

    # 7) small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    # 8) triangular shape (peak in the centre)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    if tri.sum() > 0:
        init.append(tri / tri.sum() * target_sum)

    return init


def _exp_gradient_optimize_custom(
    a0: np.ndarray,
    n: int,
    target_sum: float,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    tau_factor: float = 0.30,
    lr0: float = 0.08,
    jitter_interval: int = 3000,
):
    """
    Adam optimiser on the log‑its using a soft‑max approximation of the max‑convolution.
    Returns the best sequence seen in this run together with its exact max‑convolution value.
    """
    eps = 1e-16
    y = np.log(a0 + eps)                      # logits

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8

    # Temperature schedule (exponential decay)
    decay_rate = max_iters / 8.0

    # FFT size – must cover linear convolution of length 3 n‑2
    Nfft = 1 << ((3 * n - 2).bit_length())

    # Record the iteration of the last improvement to allow early stopping
    last_improve = 0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- current vector on the simplex -----
        logits = y - np.max(y)                # shift for stability
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()     # probability simplex
        a = target_sum * p                     # guarantees Σa = target_sum, a_i≥0

        # ----- exact (linear) convolution a*a -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- soft‑max weights for the max -----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            # hard max – distribute weight uniformly over all almost‑max indices
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ----- weighted reverse convolution (gradient of smoothed max) -----
        rev = _weighted_rev(a, w, Nfft)

        # ----- gradient w.r.t. a (2·rev) -----
        g_a = 2.0 * rev

        # ----- chain‑rule to logits (mirror‑descent on the simplex) -----
        avg = np.dot(p, g_a)                     # ⟨p, g_a⟩
        g_y = target_sum * p * (g_a - avg)       # ∂f/∂y_i

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # ----- occasional jitter to escape plateaux -----
        if jitter_interval and it % jitter_interval == 0:
            jitter = rng.normal(scale=0.01, size=n)
            y += jitter

        # ----- keep global best -----
        if M < best_score - 1e-12:
            best_score = M
            best_seq = a.copy()
            last_improve = it
            print(f"[len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

        # early‑stop if no improvement for a long stretch
        if it - last_improve > max_iters // 10:
            break

    return best_seq, best_score


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on the simplex:
    repeatedly move a small fraction of mass from a random index i to j.
    Accept improvements always; accept worsening moves with a decaying
    simulated‑annealing probability.
    """
    cur_seq = a.copy()
    cur_score = best_score

    # FFT size for fast max‑convolution (exact, no padding needed beyond 2n‑1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    max_steps = 300_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-14:
            step += 1
            continue

        # move a random fraction (1 % – 5 %) of the source mass
        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # compute new max‑convolution
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score

# Main entry point required by the harness

def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence (plain Python list) found within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load any previously discovered construction (if present)
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        base = None

    # ------------------------------------------------------------------
    # Choose a set of lengths to explore (300 – 8000)
    if base is not None:
        n0 = int(base.shape[0])
    else:
        n0 = 2100

    length_factors = [0.8, 1.0, 1.2, 1.5, 2.0, 2.5, 3.0, 4.0]
    lengths = sorted({
        max(300, min(int(n0 * f), 8000))
        for f in length_factors
    })

    # ------------------------------------------------------------------
    # Global best tracking
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Main optimisation loop – run as many independent trials as time permits
    while time.time() < deadline:
        # pick a length (randomised to diversify)
        n = int(rng.choice(lengths))
        target_sum = math.sqrt(2.0 * n)

        # generate a diverse set of initial vectors
        init_vectors = _make_initial_vectors(n, target_sum, rng)

        # inject the current global best (interpolated) if we have one
        if best_seq is not None:
            # linear interpolation of the best known vector to the new length
            old_x = np.linspace(0.0, 1.0, best_seq.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, best_seq)
            interp = np.maximum(interp, 0.0)
            interp = interp / max(interp.sum(), 1e-12) * target_sum
            init_vectors.append(interp)

        # also add the historic construction if it has a different size
        if base is not None and base.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
            interp = np.maximum(interp, 0.0)
            interp = interp / max(interp.sum(), 1e-12) * target_sum
            init_vectors.append(interp)

        # iteration budget – scaled inversely with length (more points → cheaper per step)
        max_iters = int(2_500_000 * 2100 / n)
        max_iters = max(80_000, max_iters)      # never too small

        # randomise a few hyper‑parameters for each length to increase exploration
        tau_factor = rng.uniform(0.15, 0.45)
        lr0 = rng.uniform(0.04, 0.12)
        jitter_interval = int(rng.integers(1500, 5000))

        print(f"[run] length={n}, max_iters≈{max_iters}, tau_factor={tau_factor:.3f}, lr0={lr0:.3f}")

        for idx, a0 in enumerate(init_vectors):
            if time.time() >= deadline:
                break
            print(f"  start {idx+1}/{len(init_vectors)} for length {n}")

            best_seq, best_score = _exp_gradient_optimize_custom(
                a0=a0.copy(),
                n=n,
                target_sum=target_sum,
                deadline=deadline,
                max_iters=max_iters,
                best_seq=best_seq if best_seq is not None else a0,
                best_score=best_score,
                rng=rng,
                tau_factor=tau_factor,
                lr0=lr0,
                jitter_interval=jitter_interval,
            )

    # ------------------------------------------------------------------
    # Final stochastic refinement on the globally best sequence (if time permits)
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # allocate at most half of the remaining budget to refinement, capped at 30 s
        refine_deadline = time.time() + min(remaining * 0.5, 30.0)
        best_seq, best_score = _local_refine(
            a=best_seq,
            n=best_seq.shape[0],
            deadline=refine_deadline,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Fallback – should never happen, but keep it for safety
    if best_seq is None:
        n = n0
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = float("inf")
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
