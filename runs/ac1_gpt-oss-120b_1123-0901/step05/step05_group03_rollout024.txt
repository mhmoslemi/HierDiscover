def weight_optimize(a, indices, n, L, max_iter, lr, eps):
    # a vector of length n with non-zero only at indices (weights)
    # Use exponentiated gradient descent as before.
    # We'll use weight vector w (size m) and maintain a array a with zeros else.
    m = len(indices)
    w = a[indices].copy()  # positive, sum to 1 possibly.
    for it in range(max_iter):
        # construct full a
        a_full = np.zeros(n)
        a_full[indices] = w
        # compute conv and max
        fa = np.fft.rfft(a_full, L)
        conv = np.fft.irfft(fa * fa, L)[:2*n-1]
        max_c = conv.max()
        # compute active mask for subgradient
        active = np.isclose(conv, max_c, atol=1e-12)
        if not active.any():
            break
        # weight for each convolution index: w_conv = 1/active count on active positions else 0
        w_conv = np.zeros_like(conv)
        w_conv[active] = 1.0 / active.sum()
        # gradient w.r.t a = 2 * cross-correlation of w_conv with a_full
        # compute cross-correlation using FFT: cross_corr = irfft(rfft(w_conv)*rfft(reversed a_full))
        rev_a = a_full[::-1]
        fw = np.fft.rfft(w_conv, L)
        fr = np.fft.rfft(rev_a, L)
        grad_full = 2.0 * np.fft.irfft(fw * fr, L)[:n]   # we need length? Actually cross-corr length n. We'll compute as described earlier but easier approach: conv of w_conv with reversed a = cross-correlation; Use same function earlier.
        # Actually we need gradient only at positions where a has support.
        grad = grad_full[indices]
        # Exponentiated gradient update
        w = w * np.exp(-lr * grad)
        # re-normalize
        s = w.sum()
        if s < 1e-12:
            w = np.random.dirichlet(np.ones(m))
            s = w.sum()

    # return final a vector
    a_opt = np.zeros(n)

    return a_opt

def subgradient_optimize(a, n, conv_len, corr_len, deadline, max_iters=2000, eta=0.6, patience=3.0):
    best_local_seq = a.copy()
    best_local_val = _objective(a, n, conv_len)
    last_improve = time.time()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # compute conv
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_local_val - 1e-12:
            best_local_val = val
            best_local_seq = a.copy()
            last_improve = time.time()
        # active set
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if not active.any():
            # random select a position
            w = np.zeros_like(conv)
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        # gradient
        grad = 2.0 * _cross_correlation(w, a, corr_len)
        # exponentiated gradient update
        a = a * np.exp(-eta * grad)
        # safeguard normalization
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        # occasional shake if stagnated
        if time.time() - last_improve > patience:
            # add a small random jitter
            jitter = (np.random.rand(n) - 0.5) * 1e-3
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            last_improve = time.time()
    return best_local_seq, best_local_val

"""
Propose a new step‑function sequence.

The algorithm differs from the previous Adam + sub‑gradient scheme.
It builds a random (dense) start vector, then repeatedly solves a small
linear programme:

    max    1ᵀ b
    s.t.   (f ★ b)[k] ≤ M   for all k                 ( ★ = linear convolution )
           b ≥ 0

where `f` is the current candidate and `M = max(f ★ f)`.  The LP tries to
increase the total mass while keeping the convolution below the current
worst value.  After the LP we renormalise `b` to have the same ℓ₁‑norm as
`f` and mix the two vectors (`f ← (1‑t) f + t b`).  This “best‑response”
step never increases `M`; repeating it drives the maximal convolution
towards the average 1/(2n), i.e. the theoretical lower bound of the
objective.  Several random restarts are performed until the time budget
runs out.  A final short run of the earlier exponentiated‑gradient
sub‑gradient method refines the best obtained vector.

All heavy work (FFT‑based convolutions) is done with NumPy; a linear
programme is solved with CVXPY using the ECOS solver (purely linear
constraints).  The code respects the 1000 s wall‑clock limit and returns
the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) by FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact objective value 2·n·max(a★a) / (∑a)².
    The vector `a` is assumed non‑negative.
    """
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max 1ᵀb   s.t.  (f ★ b) ≤ M ,  b ≥ 0.
    Returns the optimal b (numpy array) or None on failure.
    """
    import cvxpy as cp

    n = f.shape[0]
    b = cp.Variable(n)

    # cp.conv returns the full linear convolution (length 2·n‑1)
    conv_fb = cp.conv(f, b)

    constraints = [conv_fb <= M, b >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    # ECOS is a fast interior‑point solver for pure LPs
    try:
        prob.solve(solver="ECOS", warm_start=True, abstol=1e-9,
                    reltol=1e-9, max_iters=2000, verbose=False)
    except Exception:  # pragma: no cover – defensive
        return None

    if b.value is None:
        return None
    # Numerical noise may produce tiny negatives
    return np.maximum(b.value, 0.0)


# ----------------------------------------------------------------------
def _lp_iterate(f0: np.ndarray, n: int, conv_fft_len: int,
                max_iters: int, t: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Run the LP‑based alternating optimisation starting from `f0`.
    Returns the best vector seen and its objective value.
    """
    f = f0.copy()
    # Normalise to unit ℓ₁‑norm – the objective is scale invariant.
    if f.sum() < 1e-12:
        f = np.random.dirichlet(np.ones(n))
    f = f / f.sum()

    best_f = f.copy()
    best_val = _objective(f, n, conv_fft_len)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # current maximal convolution
        conv_ff = _full_convolution(f, conv_fft_len)
        M = conv_ff.max()

        # solve the linear programme
        b_opt = _solve_lp(f, M)
        if b_opt is None:
            break

        # renormalise b to have the same ℓ₁‑norm as the current f
        s_f = f.sum()
        s_b = b_opt.sum()
        if s_b < 1e-12:
            break
        g = b_opt * (s_f / s_b)

        # convex combination (mixing step)
        f = (1.0 - t) * f + t * g
        # keep it on the simplex
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 0:
            f /= s

        val = _objective(f, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_f = f.copy()
            print(f"[LP it={it:02d}] new best {val:.6f}")

        # modest step‑size decay to aid convergence
        t = max(t * 0.95, 0.02)

    return best_f, best_val


# ----------------------------------------------------------------------
def _subgradient_refine(a0: np.ndarray, n: int,
                        conv_fft_len: int, corr_fft_len: int,
                        deadline: float,
                        max_iters: int = 8000,
                        eta: float = 0.6,
                        patience: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the exact max‑convolution.
    Starts from `a0` (already on the simplex) and stops when the deadline
    is reached or when no improvement occurs for `patience` seconds.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    last_improve = time.time()

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            # occasional progress output
            if it % 500 == 0:
                print(f"[Subgrad it={it:04d}] new best {val:.6f}")

        # active set of indices attaining the maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # sub‑gradient of the max (chain rule)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad)

        # re‑normalise to the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # shake if we have been stagnant for a while
        if time.time() - last_improve > patience:
            jitter = (np.random.rand(n) - 0.5) * 1e-3
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises  2·n·max(a★a)/(∑a)².

    The search consists of:
      1. a few random restarts of the LP‑based “best‑response” iteration,
      2. a final refinement with the exponentiated‑gradient sub‑gradient
         method (the previous state‑of‑the‑art technique).

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Wall‑clock time budget (seconds).

    Returns
    -------
    list[float]
        The best sequence discovered (plain Python list).
    """
    np.random.seed(seed)
    # a small safety margin for the outer deadline
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # problem size – the previous best was obtained for n = 4096
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)   # 8192
    corr_fft_len = _next_pow2(3 * n - 2)   # 16384

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # optional warm start from a previously discovered sequence
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        warm = np.clip(warm, 0.0, 1000.0)
        if warm.shape[0] != n:
            if warm.shape[0] > n:
                warm = warm[:n]
            else:
                warm = np.pad(warm, (0, n - warm.shape[0]))
        if warm.sum() > 0.01:
            warm = warm / warm.sum()
            # one LP iteration starting from the warm vector
            seq, val = _lp_iterate(warm, n, conv_fft_len,
                                   max_iters=30, t=0.3, deadline=deadline)
            if val < best_val:
                best_val = val
                best_seq = seq
                print(f"[Warm‑start] LP best {val:.6f}")

    # ------------------------------------------------------------------
    # Main random‑restart loop (LP based)
    while time.time() < deadline - 1.0:
        # random support size (moderately dense – between 200 and 800 entries)
        m = np.random.randint(200, 801)

        # random support indices
        idx = np.random.choice(n, size=m, replace=False)

        # random positive weights on the support
        w = np.random.dirichlet(np.ones(m))
        a0 = np.zeros(n)
        a0[idx] = w

        # LP driven improvement
        seq, val = _lp_iterate(a0, n, conv_fft_len,
                               max_iters=30, t=0.3, deadline=deadline)
        if val < best_val:
            best_val = val
            best_seq = seq
            print(f"[Random LP] new global best {val:.6f} (support {m})")

    # ------------------------------------------------------------------
    # Final refinement with the sub‑gradient method (the previous best
    # algorithm).  This is cheap compared with the LP phase and often
    # squeezes a few extra digits out of a good candidate.
    if best_seq is not None and time.time() < deadline - 2.0:
        seq, val = _subgradient_refine(best_seq, n,
                                       conv_fft_len, corr_fft_len,
                                       deadline, max_iters=12000,
                                       eta=0.4, patience=6.0)
        if val < best_val:
            best_val = val
            best_seq = seq
            print(f"[Final sub‑grad] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded produce a uniform vector.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # For debugging you could call the provided evaluate_sequence here,
    # but it is not required for the return value.
    # print("Final score:", evaluate_sequence(best_seq.tolist()))

    return best_seq.tolist()
