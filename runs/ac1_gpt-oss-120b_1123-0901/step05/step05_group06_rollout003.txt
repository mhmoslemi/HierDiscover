"""
Propose a non‑negative step function (vector) that minimises ``evaluate_sequence``.

Algorithm
---------
1. Load the best known construction (global `height_sequence_1`) if it exists,
   otherwise start from a uniform vector.
2. Optimise the vector with a soft‑max Adam on the log‑its (as in the
   reference code).  This enforces non‑negativity and Σa = √(2 n) and quickly
   drives the max‑convolution down.
3. Refine the result with a *linear‑programming* step:
   * For the current vector f and its current peak M = max(f∗f) we solve
     ``max Σ g_i`` subject to ``f∗g ≤ M`` and ``g_i ≥ 0``.
   * The LP provides a direction that can lower the peak when mixed with
     f.  We blend f and a scaled copy of g with a few candidate step sizes
     and keep the best improvement.  Only a handful of LP iterations are
     needed.
4. Perform a cheap stochastic hill‑climbing (pairwise mass transfers with
   simulated‑annealing acceptance) to squeeze out the last bits.
5. Return the best vector found before the time budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional linear‑programming support (SciPy).  If SciPy is unavailable we
# silently skip the LP refinement and continue with the other phases.
try:
    from scipy.optimize import linprog
    _SCIPY_OK = True
except Exception:  # pragma: no cover
    _SCIPY_OK = False

# Helper utilities (no closures)

def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Return max_k (a∗a)[k] using an FFT of length Nfft."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values.
    Returns (weights, smoothed max)."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k * a_{k-i} (out‑of‑range indices ignored)
    using FFTs.  This is the gradient of the (soft‑max) max‑convolution.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)          # reverse w before FFT
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]    # linear part, length = 3 n‑2
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Create a small diverse set of normalised start vectors."""
    init = []

    # uniform
    init.append(np.full(n, target_sum / n))

    # Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0
             - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    init.append(black / black.sum() * target_sum)

    # small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    # triangular shape (peak in the centre)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    init.append(tri / tri.sum() * target_sum)

    return init


def _exp_gradient_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑its (exponentiated‑gradient) using a soft‑max
    approximation of the max‑convolution.  Returns the best vector seen in
    this run together with its exact max‑convolution value.
    """
    target_sum = math.sqrt(2.0 * n)
    eps = 1e-16
    y = np.log(a0 + eps)                     # logits

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.08
    jitter_interval = 3000

    # temperature schedule
    tau_factor = 0.30
    decay_rate = max_iters / 8.0

    Nfft = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ------- current vector (softmax of logits) -------
        logits = y - np.max(y)
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()
        a = target_sum * p

        # ------- exact convolution -------
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ------- soft‑max weights -------
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ------- weighted reverse convolution (gradient of smoothed max) -------
        rev = _weighted_rev(a, w, Nfft)

        # ------- gradient w.r.t a -------
        g_a = 2.0 * rev

        # ------- chain rule to logits (mirror‑descent on simplex) -------
        avg = np.dot(p, g_a)
        g_y = target_sum * p * (g_a - avg)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # occasional jitter
        if it % jitter_interval == 0:
            jitter = rng.normal(scale=0.01, size=n)
            y += jitter

        # keep best
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

    return best_seq, best_score


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on the simplex:
    move a small fraction of mass from a random index i to j.
    Accept improvements always; accept worsening moves with a decaying
    simulated‑annealing probability.
    """
    target_sum = a.sum()
    cur_seq = a.copy()
    cur_score = best_score
    Nfft = 1 << ((2 * n - 1).bit_length())
    max_steps = 200_000
    step = 0

    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score


def _solve_lp(f: np.ndarray, M: float):
    """Solve  max Σ g_i  subject to  f∗g ≤ M, g≥0.  Returns the optimal g or None."""
    if not _SCIPY_OK:      # pragma: no cover
        return None

    n = f.shape[0]
    rows = 2 * n - 1

    # Build the Toeplitz‑like constraint matrix A where
    # (A @ g)[k] = Σ_i f_i * g_{k-i} = (f ∗ g)[k].
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    b = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise Σ g_i  →  minimise -Σ g_i
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:                         # pragma: no cover
        return None

    if not res.success:
        return None
    return res.x


def _lp_refine(
    a: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Apply a few LP‑based refinements to a vector a.
    Each iteration solves the LP described above, rescales the solution
    to the correct ℓ₁‑norm and mixes it with a using a small step size.
    Stops when no improvement is found or the deadline is reached.
    """
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)
    Nfft = 1 << ((3 * n - 2).bit_length())

    best_seq = a.copy()
    best_score = _max_convolution(best_seq, Nfft)

    # candidate mixing factors (larger ones first – they give faster progress)
    t_vals = [0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02]

    for it in range(15):                     # at most a handful of LP solves
        if time.time() >= deadline:
            break

        M = _max_convolution(best_seq, Nfft)
        g = _solve_lp(best_seq, M)
        if g is None:
            print("[LP] linear programme failed – aborting LP phase")
            break

        sum_g = g.sum()
        if sum_g <= 1e-12:
            break

        g_scaled = g * (target_sum / sum_g)

        improved = False
        for t in t_vals:
            new_seq = (1.0 - t) * best_seq + t * g_scaled
            new_score = _max_convolution(new_seq, Nfft)
            if new_score < best_score - 1e-12:
                best_seq = new_seq
                best_score = new_score
                improved = True
                print(f"[LP] iter {it+1:2d}, t={t:.2f} → score {best_score:.6f}")
                break
        if not improved:
            # No mixing step reduced the peak – stop early.
            break

    return best_seq, best_score

        print(f"[phase 3] final max‑conv = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Convert to plain Python list as required by the harness.
    final_seq = best_seq.tolist()
    print(f"[done] best score = {best_score:.6f}, length = {len(final_seq)}")
    return final_seq
