"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. Load the previously discovered construction (``height_sequence_1``) if it
   exists; otherwise start from a uniform vector.
2. For a handful of lengths (the original size and a few enlargements) do:
   a. Interpolate the current best vector to the new length and renormalise
      to Σa = √(2 n).
   b. Repeatedly solve a linear programme
        maximise Σ g_j
        s.t.   g_j ≥ 0,
               (f * g)[k] ≤ M   (for every convolution index)
      where ``f`` is the current vector and ``M = max(f * f)``.
      The LP yields a vector ``g₀`` that can carry more mass than the
      admissible total.  Scale it down to the required sum and update
      ``f ← (1‑t) f + t g`` with a modest step size ``t``.
      After each update recompute ``M`` and keep the best vector seen.
   c. When the time budget for this length is exhausted, perform a cheap
      stochastic local refinement that moves a small fraction of mass
      between random indices (always accepting improvements, otherwise
      accepting with a simulated‑annealing probability).

The LP step is a direct implementation of the “alternating linear program”
suggested in the statement.  It reduces the maximal convolution value
significantly with only a few dozen iterations, after which the cheap
pairwise refinement squeezes out the last bits.

The whole routine respects the 1000 s budget, uses only standard
libraries (NumPy, SciPy) and returns a plain Python list.

import time
import math
import numpy as np
from scipy.optimize import linprog

# Helper functions (top‑level, no closures)

def _max_convolution(a: np.ndarray) -> float:
    """
    Compute max_k (a * a)[k] using an FFT (linear convolution).
    """
    n = a.shape[0]
    # Length of linear convolution = 2*n-1 ; choose next power‑of‑2 for FFT
    Nfft = 1 << ((2 * n - 2).bit_length())
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
    return float(conv.max())


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (shape (2*n‑1, n)) such that A @ g = f * g
    for any g ≥ 0.  Column j of A contains the vector f shifted by j.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # fill column by column – each column is a shifted copy of f
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _local_refine(
    a: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Cheap stochastic hill‑climbing on the simplex.
    Moves a random small fraction of mass from i → j.
    Accepts every improvement; otherwise accepts with a decaying SA probability.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = _max_convolution(cur)

    # FFT size for fast convolution (just for scoring)
    Nfft = 1 << ((2 * n - 2).bit_length())

    step = 0
    max_steps = 200_000
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] < 1e-12:
            step += 1
            continue

        # transfer 1‑5 % of the source mass
        frac = rng.uniform(0.01, 0.05)
        delta = cur[i] * frac
        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        # evaluate new max‑convolution
        A_fft = np.fft.rfft(new, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        if new_score < cur_score - 1e-12:
            cur, cur_score = new, new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur, cur_score = new, new_score
        step += 1
    return cur

        if base is not None and base.shape[0] == n:
            f = base.copy()
            print(f"[init] using existing sequence of length {n}")
        elif base is not None and base.shape[0] != n:
            # linear interpolation of the previous best onto the new grid
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            f = np.interp(new_x, old_x, base)
            print(f"[init] interpolated sequence to length {n}")
        else:
            # fallback: uniform start
            f = np.full(n, target_sum / n)
            print(f"[init] uniform start length {n}")

        f = np.maximum(f, 0.0)
        f = f / f.sum() * target_sum

        # initial objective
        M = _max_convolution(f)
        cur_score = M
        if cur_score < best_score:
            best_score, best_seq, best_len = cur_score, f.copy(), n
            print(f"[global] new overall best = {best_score:.6f} (len {n})")

        # ---------- LP‑based alternating optimisation ----------
        # Parameters for the alternating scheme
        max_lp_iters = 80                       # enough for a noticeable drop
        step_size = 0.35                        # blending factor (1‑t)
        lp_deadline = deadline - 20.0           # reserve time for final refinement

        for it in range(max_lp_iters):
            if time.time() >= lp_deadline:
                break

            # Build convolution matrix f * g <= M
            A = _build_conv_matrix(f)          # shape (2n‑1, n)

            # LP: maximise Σ g_j  <=> minimise –Σ g_j
            c = -np.ones(n)
            b = np.full(2 * n - 1, M)
            bounds = [(0, None)] * n

            # solve
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                          method='highs', options={'presolve': True})
            if not res.success:
                # In rare cases the LP may fail – abort the LP loop
                print(f"[lp] iteration {it} failed ({res.message}); exiting LP phase")
                break

            g = res.x
            sum_g = g.sum()
            if sum_g > 0:
                scale = target_sum / sum_g
                # Only scale down (scale ≤ 1) to stay feasible
                if scale > 1.0:
                    scale = 1.0
                g = g * scale
            else:
                # shouldn't happen, but protect against division by zero
                g = np.zeros_like(g)

            # blending update
            f = (1.0 - step_size) * f + step_size * g

            # recompute maximal convolution
            M_new = _max_convolution(f)

            # keep the best vector seen so far (exact objective = M_new)
            if M_new < best_score - 1e-12:
                best_score, best_seq, best_len = M_new, f.copy(), n
                print(f"[len {n:5d} lp {it:3d}] new best = {best_score:.6f}")

            # if progress stalls we may reduce the step size a little
            if M_new > M * (1 - 1e-4):
                step_size = max(0.05, step_size * 0.9)

            M = M_new

        # ------------------------------------------------------------------
        # After LP phase, run a brief stochastic refinement on the best of this length
        if best_seq is not None and time.time() < deadline:
            # allocate at most 10 s for the refinement of the current length
            refine_deadline = min(deadline, time.time() + 10.0)
            refined = _local_refine(best_seq if best_len == n else f,
                                    deadline=refine_deadline,
                                    rng=rng)
            refined_score = _max_convolution(refined)
            if refined_score < best_score:
                best_score, best_seq, best_len = refined_score, refined.copy(), n
                print(f"[len {n}] refined best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety net – if everything failed we fall back to a uniform vector
    if best_seq is None:
        n = n0
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = _max_convolution(best_seq)
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
