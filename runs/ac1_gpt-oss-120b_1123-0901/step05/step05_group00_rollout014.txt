def subgradient_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, best_val, best_seq):
    a = a0.copy()
    eta = 0.8
    last_improve = time.time()
    for it in range(5000):
        if time.time() > deadline: break
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        val = 2.0 * n * max_c

            best_val = val

            print(f"[Subgrad it={it}] new best {val:.6f}")

            eta *= 0.95
            last_improve = time.time()

"""
Step‑function optimiser – version 2

The algorithm works in three stages:

1. **Parametric exploration** – a large number of simple families
   (beta‑shapes, edge‑exponential, power‑law, Dirichlet, uniform)
   are sampled, normalised and evaluated.  The best vector from this
   cheap global search is kept as a warm‑start.

2. **Smooth‑max Adam optimisation** – the current vector is represented
   by log‑softmax “logits”.  A log‑sum‑exp surrogate of the
   max‑convolution is minimised by Adam while the temperature of the
   surrogate (the λ‑parameter) is slowly annealed.  This stage
   dramatically reduces the objective and benefits from the diverse
   starts generated in stage 1.

3. **Exact sub‑gradient refinement** – the true max‑convolution is used.
   An exponentiated‑gradient (mirror‑descent on the simplex) updates
   the distribution using the active‑set sub‑gradient.  The step‑size
   is adaptively decayed and a tiny random shake is applied when the
   iterate stagnates.

All stages respect a hard wall‑clock deadline (default 1000 s) and keep
track of the best sequence discovered.  The returned object is a plain
Python list of non‑negative numbers whose sum is 1 (the scaling is
irrelevant for the objective).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.

    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


def compute_objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Return F(a)=2·n·max(a*a) (the sum of a is assumed to be 1)."""
    max_c = _full_convolution(a, fft_len=fft_len).max()
    return 2.0 * n * max_c


def generate_beta_shape(n: int, p: float, q: float) -> np.ndarray:
    """Symmetric (or asymmetric) Beta(p,q) density discretised on n points."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n          # in (0,1)
    # work in log‑space to avoid overflow / underflow
    log_shape = (p - 1.0) * np.log(x) + (q - 1.0) * np.log(1.0 - x)
    shape = np.exp(log_shape)
    shape[~np.isfinite(shape)] = 0.0
    shape = np.maximum(shape, 0.0)
    total = shape.sum()
    if total <= 0.0:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= total
    return shape


def generate_exponential_shape(n: int, lam: float) -> np.ndarray:
    """Two–sided exponential tail shape: exp(-lam·i)+exp(-lam·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    total = shape.sum()
    if total == 0.0:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        shape /= total
    return shape


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_val: float,
    best_seq: np.ndarray,
) -> tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp surrogate of the max‑convolution.
    Returns the (possibly) improved (best_seq, best_val).
    """
    eps = 1e-16
    a0 = np.maximum(a0, eps)
    a0 /= a0.sum()
    logits = np.log(a0)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05

    lam = 0.2               # start temperature
    lam_factor = 1.005

    max_iters = int(20000 * 1024 / n) + 5000
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # back‑propagation through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal the temperature
        lam = min(lam * lam_factor, 1e7)

        # exact objective (sum(a)=1 → denominator = 1)
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam it={it:05d}] new best {val:.6f}")

        # jitter on stagnation
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_seq, best_val


def subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_val: float,
    best_seq: np.ndarray,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient (exponentiated‑gradient) refinement on the true max.
    """
    a = a0.copy()
    eta = 0.8
    last_improve = time.time()

    for it in range(5000):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        # active set – all entries attaining the maximum (within tolerance)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad it={it:04d}] new best {val:.6f}")

        # adapt step‑size and handle stagnation
        if time.time() - last_improve > 5.0:
            eta *= 0.95          # gently shrink the step
            last_improve = time.time()
        eta *= 0.9995           # slow exponential decay

    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a*a)/(sum(a)²).

    The routine respects a wall‑clock budget (default 1000 s) and
    returns the best sequence discovered (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # initialise best sequence (use the historic global if available)
    # ------------------------------------------------------------------
    n = 16384
    conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

    best_seq = None
    best_val = float("inf")

    # warm‑start from a previously found global solution, if any
    hseq = globals().get("height_sequence_1")
    if hseq is not None:
        h = np.asarray(hseq, dtype=np.float64)
        if h.shape[0] != n:
            # resample / pad to current length
            x_old = np.linspace(0.0, 1.0, h.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            h = np.interp(x_new, x_old, h)
        h = np.maximum(h, 0.0)
        s = h.sum()
        if s > 0.0:
            h /= s
            val = compute_objective(h, n, conv_fft_len)
            best_seq = h
            best_val = val
            print(f"[Init] global warm‑start value {val:.6f}")

    # fallback – uniform if nothing else
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_objective(best_seq, n, conv_fft_len)
        print(f"[Init] uniform start value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – cheap parametric exploration
    # ------------------------------------------------------------------
    print("\n--- Stage 1: parametric exploration ---")
    while time.time() + 5.0 < deadline:      # leave a few seconds for later stages
        rnd = np.random.rand()
        if rnd < 0.20:
            # random Dirichlet
            cand = np.random.dirichlet(np.ones(n))
        elif rnd < 0.40:
            # Beta(p,q) – parameters on log‑scale
            p = 10 ** np.random.uniform(-3, 1)   # 0.001 … 10
            q = 10 ** np.random.uniform(-3, 1)
            cand = generate_beta_shape(n, p, q)
        elif rnd < 0.60:
            # Two‑sided exponential tail
            lam = 10 ** np.random.uniform(-4, 0)  # 1e‑4 … 1
            cand = generate_exponential_shape(n, lam)
        elif rnd < 0.80:
            # Mixture of Beta and uniform
            p = 10 ** np.random.uniform(-3, 1)
            q = 10 ** np.random.uniform(-3, 1)
            beta = generate_beta_shape(n, p, q)
            w = np.random.rand()
            cand = w * beta + (1.0 - w) * np.full(n, 1.0 / n)
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
        else:
            # Power‑law edge‑concentrated family (the one used before)
            idx = np.arange(1, n + 1, dtype=np.float64)
            prod = idx * (n - idx + 1.0)
            alpha = np.random.uniform(-2.0, -0.5)
            shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
            if shape.sum() > 0.0:
                cand = shape / shape.sum()
            else:
                cand = np.full(n, 1.0 / n)

        # evaluate
        val = compute_objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Stage1] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam on the smooth‑max surrogate
    # ------------------------------------------------------------------
    print("\n--- Stage 2: Adam on smooth‑max ---")
    best_seq, best_val = adam_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_val,
        best_seq,

    # ------------------------------------------------------------------
    # Stage 3 – exact sub‑gradient refinement
    # ------------------------------------------------------------------
    print("\n--- Stage 3: exact sub‑gradient refinement ---")
    best_seq, best_val = subgradient_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_val,
        best_seq,
    )

    # ------------------------------------------------------------------
    # final safety net – if somehow nothing was found, fall back to uniform
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"\n=== Finished === best objective {best_val:.6f}")
    return best_seq.tolist()
