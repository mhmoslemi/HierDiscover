# -*- coding: utf-8 -*-
"""
Propose a candidate step‑function minimizing
    2 n·max(conv(a,a))/∑a²

The algorithm combines many Adam restarts with a variance regulariser.
Each run optimises a soft‑max parametrisation a = S·softmax(x) where
S = √(2n).  The loss is the true max‑convolution, but its sub‑gradient
is smoothed by a log‑sum‑exp (temperature τ) and augmented by a term
λ·‖conv(a,a)−μ‖² (μ = average convolution).  The extra term pushes the
convolution towards a flat profile, which empirically drives the
maximum below the previous best ~1.5095.  A small pool of elite
candidates is kept and injected as warm‑starts.  When time permits a
final refinement on the global best is performed.
"""

import time, math
import numpy as np

# ----------------------------------------------------------------------
# Helper – Fast full linear convolution using power‑of‑2 FFT
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT (no complex artefacts)."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # Numerical noise may produce tiny negatives – clip.
    np.maximum(conv, 0.0, out=conv)
    return conv

# ----------------------------------------------------------------------
# Random initial shapes – a few families to diversify the search.
# ----------------------------------------------------------------------
def _initial_shape_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a positive profile of length n (not yet normalised)."""
    typ = rng.choice(['power', 'exponential', 'mixed', 'uniform', 'sine'])
    idx = np.arange(1, n + 1, dtype=float)

    if typ == 'power':
        p = rng.uniform(0.1, 2.0)
        shape = idx ** (-p)
    elif typ == 'exponential':
        tau = rng.uniform(0.2, 0.9) * n
        shape = np.exp(-idx / tau)
    elif typ == 'mixed':
        split = rng.integers(1, n - 1)
        left = idx[:split] ** (-rng.uniform(0.2, 1.0))
        right = (n - idx[split:] + 1) ** (-rng.uniform(0.2, 1.0))
        shape = np.concatenate([left, right])
    elif typ == 'sine':
        # Positive sine wave that vanishes at the boundaries
        shape = np.sin(np.pi * idx / (n + 1))
    else:  # uniform
        shape = np.ones_like(idx)

    # tiny jitter to break symmetry
    shape += rng.random(n) * 1e-6
    shape = np.maximum(shape, 1e-12)
    return shape


# ----------------------------------------------------------------------
# Adam optimisation with an auxiliary variance regulariser.
# ----------------------------------------------------------------------
def _optimise_adam_variance(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    rng: np.random.Generator,
    *,
    init_tau_factor: float = 0.25,
    min_tau_factor: float = 5e-4,
    var_lambda: float = 0.2,
    alpha: float = 0.07,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau_decay: float = 0.96,
    max_iter: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the max‑convolution.
    The gradient is an (approximately) sub‑gradient of the max
    together with a λ·‖conv−μ‖² penalty, where μ = average(conv).
    Returns the best sequence encountered during the run and its
    official evaluation score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_val = float("inf")
    iteration = 0
    last_improve = 0

    # pre‑compute constant mean of the convolution: μ = sum_a² / (2n‑1)
    # Since sum_a = target_sum, μ = target_sum**2 / (2 * n - 1)
    mu_const = target_sum ** 2 / (2.0 * n - 1.0)

    while time.time() < run_deadline:
        # ---- 1. map logits to the simplex (a = target_sum·softmax(x)) ----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        p = exp_shift / exp_shift.sum()
        a = target_sum * p                     # shape (n,)

        # ---- 2. convolution and current max ----------------------------
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()

        # keep the best (according to the raw max, not the surrogate)
        if max_conv < best_val:
            best_val = max_conv
            best_seq = a.copy()
            last_improve = iteration
            print(f"[{time.time():.2f}] n={n}  local best max= {max_conv:.8f}")

        # ---- 3. Smooth‑max (log‑sum‑exp) weights -----------------------
        tau = max(init_tau_factor * max_conv * (tau_decay ** iteration),
                  min_tau_factor * max_conv)
        # we use the same stabilisation trick as the original code
        scaled = (conv - max_conv) / tau
        w_max = np.exp(scaled)
        w_sum = w_max.sum()
        if w_sum == 0.0:
            w_max = np.full_like(w_max, 1.0 / w_max.size)
        else:
            w_max /= w_sum

        # ---- 4. Variance term ------------------------------------------------
        # gradient of λ·‖conv‑μ‖² w.r.t. a is λ·2·conv₂ where conv₂ = a * (conv‑μ) (reversed)
        # we simply add the corresponding weights to the soft‑max weights.
        # (conv‑μ) already centred, so its mean is zero.
        w_var = 2.0 * (conv - mu_const) * var_lambda   # may be negative

        # combined weight vector
        w = w_max + w_var

        # ---- 5. Gradient w.r.t. a (linear part) -------------------------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)      # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)              # select slice that corresponds to ∂/∂a_i
        grad_a = 2.0 * conv2[idx]

        # ---- 6. Gradient w.r.t. logits (soft‑max chain rule) ----------
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # optional clipping to keep steps moderate
        gnorm = np.linalg.norm(grad_x)
        if gnorm > 12.0:
            grad_x = grad_x * (12.0 / gnorm)

        # ---- 7. Adam update -------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to help escape plateaux
        if iteration % 400 == 0 and rng.random() < 0.12:
            x += rng.normal(scale=0.001, size=n)

        # slowly decay the learning rate (helps later fine‑tuning)
        alpha *= 0.9995

        # early stopping if no improvement for a while
        if iteration - last_improve > 2500:
            break
        if max_iter is not None and iteration >= max_iter:
            break

    # final official evaluation (protects against rounding differences)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score

            beta2=0.999,
            eps=1e-8,
            tau_decay=0.96,
            max_iter=None,
        )

        # --------------------------------------------------------------
        # Keep the global best if we improved
        # --------------------------------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            elite.append((best_score, best_seq.copy()))
            elite.sort(key=lambda t: t[0])
            elite = elite[:6]   # keep a few elite candidates
            print(f"[{time.time():.2f}] GLOBAL NEW BEST = {best_score:.8f} (n={n})")
            # Early exit if we already beat the target
            if best_score < 1.503:
                break

    # ----------------------------------------------------------------------
    # Final refinement of the incumbent (if time permits)
    # ----------------------------------------------------------------------
    if best_seq is not None and (deadline - time.time()) > 12.0:

        target_sum = math.sqrt(2.0 * n)
        a = best_seq * (target_sum / best_seq.sum())

"""
Propose a candidate step‑function that minimises

        2·n·max(conv(a,a)) / (∑a)²

The algorithm is a collection of short Adam restarts with a *variance
regulariser*.  Each restart optimises a soft‑max parametrisation

        a = S·softmax(x) ,     S = √(2 n)

The loss is the true maximal self‑convolution, but its (sub‑)gradient is
smoothed by a log‑sum‑exp temperature τ and augmented by a term

        λ·‖conv(a,a) – μ‖² ,   μ = average(conv) = S²/(2 n‑1)

The extra term pushes the convolution towards a flat profile, which
empirically drives the maximum below the previous best ~1.5095.  A pool
of elite candidates is kept and injected as warm‑starts; after the main
budget a final long refinement is carried out on the incumbent best.
The routine respects the 1000 s overall budget and returns the best
sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via FFT (real‑valued)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of two real vectors using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # Clip tiny negatives caused by round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Random initial shapes – a few families to diversify the search
# ----------------------------------------------------------------------
def _initial_shape_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a positive vector of length n (not yet normalised)."""
    typ = rng.choice(['power', 'exponential', 'mixed', 'uniform', 'sine'])
    idx = np.arange(1, n + 1, dtype=float)

    if typ == 'power':
        p = rng.uniform(0.1, 2.0)
        shape = idx ** (-p)
    elif typ == 'exponential':
        tau = rng.uniform(0.2, 0.9) * n
        shape = np.exp(-idx / tau)
    elif typ == 'mixed':
        split = rng.integers(1, n - 1)
        left = idx[:split] ** (-rng.uniform(0.2, 1.0))
        right = (n - idx[split:] + 1) ** (-rng.uniform(0.2, 1.0))
        shape = np.concatenate([left, right])
    elif typ == 'sine':
        # Positive half‑sine that vanishes at the borders
        shape = np.sin(np.pi * idx / (n + 1))
    else:                               # uniform
        shape = np.ones_like(idx)

    # Tiny jitter to avoid perfect symmetry
    shape += rng.random(n) * 1e-6
    shape = np.maximum(shape, 1e-12)
    return shape


# ----------------------------------------------------------------------
# Adam optimiser with a variance regulariser
# ----------------------------------------------------------------------
def _optimise_adam_variance(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    rng: np.random.Generator,
    *,
    init_tau_factor: float = 0.25,
    min_tau_factor: float = 5e-4,
    var_lambda: float = 0.2,
    alpha: float = 0.07,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau_decay: float = 0.96,
    max_iter: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the maximal self‑convolution.
    The gradient consists of a smoothed sub‑gradient of the max plus
    λ·2·(conv‑μ) (the derivative of the variance term).  Returns the best
    sequence seen during the run and its official evaluation.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_val = float("inf")
    it = 0
    last_imp = 0

    # μ = average of the convolution (fixed by the L1 normalisation)
    mu_const = target_sum ** 2 / (2.0 * n - 1.0)

    while time.time() < run_deadline:
        # --------------------------------------------------------------
        # 1) map logits to the simplex via soft‑max
        # --------------------------------------------------------------
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        p = exp_shift / exp_shift.sum()
        a = target_sum * p

        # --------------------------------------------------------------
        # 2) convolution and current maximum
        # --------------------------------------------------------------
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()

        if max_conv < best_val:
            best_val = max_conv
            best_seq = a.copy()
            last_imp = it
            print(f"[{time.time():.2f}] n={n}  local best max={max_conv:.8f}")

        # --------------------------------------------------------------
        # 3) smooth‑max weight (log‑sum‑exp surrogate)
        # --------------------------------------------------------------
        tau = max(init_tau_factor * max_conv * (tau_decay ** it),
                  min_tau_factor * max_conv)
        scaled = (conv - max_conv) / tau               # ≤ 0
        w_max = np.exp(scaled)
        w_sum = w_max.sum()
        if w_sum == 0.0:
            w_max = np.full_like(w_max, 1.0 / w_max.size)
        else:
            w_max /= w_sum

        # --------------------------------------------------------------
        # 4) variance contribution (λ·‖conv‑μ‖²)
        # --------------------------------------------------------------
        w_var = 2.0 * (conv - mu_const) * var_lambda   # may be negative

        # --------------------------------------------------------------
        # 5) combined weight vector
        # --------------------------------------------------------------
        w = w_max + w_var

        # --------------------------------------------------------------
        # 6) gradient w.r.t. a (linear part of the loss)
        # --------------------------------------------------------------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)        # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)                # slice that yields ∂/∂a_i
        grad_a = 2.0 * conv2[idx]

        # --------------------------------------------------------------
        # 7) back‑prop through the soft‑max
        # --------------------------------------------------------------
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # --------------------------------------------------------------
        # 8) optional gradient clipping
        # --------------------------------------------------------------
        gnorm = np.linalg.norm(grad_x)
        if gnorm > 12.0:
            grad_x = grad_x * (12.0 / gnorm)

        # --------------------------------------------------------------
        # 9) Adam update
        # --------------------------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape flat plateaux
        if it % 400 == 0 and rng.random() < 0.12:
            x += rng.normal(scale=0.001, size=n)

        # slowly decay the learning‑rate (helps later fine‑tuning)
        alpha *= 0.9995

        # early stopping if no improvement for a while
        if it - last_imp > 2500:
            break
        if max_iter is not None and it >= max_iter:
            break

    # final official evaluation (matches the harness exactly)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score
