"""
Hybrid optimisation for the step‑function minimisation problem.

The objective (after normalising the sum to √(2 n)) is simply the
maximum value of the self‑convolution of the coefficient vector.
The routine combines several complementary strategies:

1.  **Soft‑max Adam** – many short independent runs with random
    initialisations, a smooth log‑sum‑exp surrogate for the max and
    an Adam optimiser on the log‑it parameters.

2.  **Projected sub‑gradient descent** – exact sub‑gradient of the
    max‑convolution (a reversed slice of the current vector) and a
    projection onto the simplex {a≥0, Σa = √(2 n)}.

3.  **Stochastic hill‑climbing** – tiny mass transfers between random
    positions, Metropolis acceptance with a linear temperature schedule.

4.  **Linear‑programming refinement** – given a current vector `a`,
    solve a LP that finds a direction `g` with maximal ℓ₁‑norm while
    keeping the convolution `a∗g` under the current peak.  A short
    mixing step reduces the peak further.

The algorithm respects the 1000 s time budget, keeps a global best
solution and falls back to a uniform construction if nothing else
is found.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# Fast full linear convolution (real FFT) – used by all stages
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of real vectors x and y."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # due to floating‑point noise a few entries can be slightly negative
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σw = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending sort
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)           # 1‑based indices
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max (log‑it) parametrisation
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.20,
    min_tau_factor: float = 0.001,
    alpha0: float = 0.03,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser with a smooth max‑surrogate (log‑sum‑exp)."""
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # ---- map logits to the simplex (a = target_sum * softmax(x)) ----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)               # stable soft‑max
        p = exp_shift / np.sum(exp_shift)           # probabilities (Σp = 1)
        a = target_sum * p

        # ---- current exact max of the convolution ----
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] n={n}  new local best max={cur_max:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp surrogate) ----
        tau = max(init_tau_factor * cur_max * (0.99 ** it),
                  min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient w.r.t. a  (2·a ⊛ w_rev) ----
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]   # length 3n‑2
        idx = (2 * n - 2) - np.arange(n)                     # positions that give Σ a_j w_{i+j}
        grad_a = 2.0 * conv2[idx]

        # ---- back‑prop to logits (chain rule) ----
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ---- Adam update ----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # tiny jitter every few hundred iterations
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slowly decay the base learning rate
        alpha *= 0.9995

    # final evaluation (exact max, which equals the evaluator output)
    final_score = best_val
    return best_a, float(final_score)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent with Euclidean projection onto the simplex."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step_size = 0.1 * target_sum / n   # initial step, roughly 1 % of the mean value

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())                # index where the max is attained

        # sub‑gradient of conv[k] w.r.t. a
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # descent step and projection
        a_candidate = a - step_size * grad
        a_candidate = _project_onto_simplex(a_candidate, target_sum)

        val_candidate = _full_convolution_fft(a_candidate, a_candidate).max()

        # accept if improvement, otherwise shrink the step size
        if val_candidate < best_val - 1e-12:
            a = a_candidate
            best_a = a_candidate
            best_val = val_candidate
            step_size *= 1.02                     # modest increase after success
        else:
            step_size *= 0.70                     # aggressive reduction on failure

        # occasional jitter to escape plateaus
        if it % 300 == 0 and np.random.rand() < 0.15:
            jitter = np.random.rand(n) * 1e-4
            a = _project_onto_simplex(a + jitter, target_sum)

        if it % 500 == 0:
            print(f"[{time.time():.2f}] subgrad it={it} step={step_size:.3e} best={best_val:.6f}")

        # early stop if already very good
        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Random local search that moves a tiny amount of mass between two indices."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # temperature schedule – start relatively hot, cool to near zero
    T0 = best_val * 0.3
    T_end = 1e-4
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        fraction = (time.time() - start) / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue

        # move at most 10 % of the mass at i
        delta = a[i] * np.random.rand() * 0.10
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()

        accept = False
        if new_val < best_val:
            accept = True
        else:
            # Metropolis criterion
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")

        # occasional re‑normalisation to avoid drift
        if it % 2000 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (single mixing step)
# ----------------------------------------------------------------------
def _lp_refine(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    One LP‑based refinement:
        maximise Σ g_j   subject to   g ≥ 0  and  a∗g ≤ M,
    where M is the current peak of a∗a.
    The optimal g is scaled to the target sum and mixed with a.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    M = best_val

    # Build the convolution matrix A (dense – still modest for n≈3000)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for col in range(n):
        A[col:col + n, col] = a

    # Linear programme: maximise Σ g   ⇔   minimise −Σ g
    c = -np.ones(n)
    bounds = [(0, None)] * n
    b_ub = np.full(rows, M)

    # Solve with HiGHS (fast interior‑point/simplex hybrid)
    res = scipy.optimize.linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method='highs',
        options={'presolve': True, 'time_limit': max(5.0, deadline - time.time())}
    )
    if not res.success:
        print(f"[{time.time():.2f}] LP refinement failed: {res.message}")
        return best_a, best_val

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return best_a, best_val

    # Scale g to the required simplex radius
    g = g * (target_sum / sum_g)

    # Try a few mixing coefficients and keep the best improvement
    for t in (0.0, 0.05, 0.15, 0.30, 0.50, 0.75, 1.0):
        a_mix = (1.0 - t) * a + t * g
        val = _full_convolution_fft(a_mix, a_mix).max()
        if val < best_val:
            best_val = val
            best_a = a_mix.copy()
            print(f"[{time.time():.2f}] LP mix t={t:.2f} improved val={best_val:.6f}")

    return best_a, best_val

# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid search for a non‑negative step function with minimal
    evaluation score.  The routine respects the time budget and
    returns the best sequence (as a plain Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # tiny safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Warm‑start from a globally available construction, if any.
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.asarray(hs, dtype=float)
        n0 = hs_arr.shape[0]
        target0 = np.sqrt(2.0 * n0)
        hs_norm = hs_arr * (target0 / hs_arr.sum())
        score0 = evaluate_sequence(hs_norm.tolist())
        best_seq = hs_norm.copy()
        best_score = float(score0)
        print(f"[{time.time():.2f}] warm‑start seed score={best_score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – many short Adam runs with diverse lengths / init.
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        # keep a fraction of the remaining budget for the later phases
        slice_time = min(12.0, remaining * 0.55)
        run_deadline = time.time() + slice_time

        # problem size – we bias towards the middle of the successful range
        if remaining > 400:
            n = np.random.randint(1800, 3500 + 1)
        elif remaining > 180:
            n = np.random.randint(1300, 2800 + 1)
        else:
            n = np.random.randint(800, 2000 + 1)

        target_sum = np.sqrt(2.0 * n)

        # initialise logits
        if np.random.rand() < 0.7:
            a0 = np.random.rand(n)
        else:
            p_exp = np.random.uniform(0.20, 0.80)
            idx = np.arange(1, n + 1, dtype=float)
            a0 = idx ** (-p_exp)
            if np.random.rand() < 0.4:
                tau_exp = np.random.uniform(0.5, 2.0) * n
                a0 *= np.exp(-idx / tau_exp)

        a0 = a0 / a0.sum() * target_sum
        x0 = np.log(a0 + 1e-12)                     # logits for the soft‑max

        # occasionally inject the current global best (trim/pad)
        if best_seq is not None and np.random.rand() < 0.35:
            if best_seq.shape[0] >= n:
                a0 = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.shape[0]) * 1e-5
                a0 = np.concatenate([best_seq, pad])
            a0 = a0 / a0.sum() * target_sum
            x0 = np.log(a0 + 1e-12)

        # run Adam optimiser
        seq_local, score_local = _optimise_adam_softmax(
            n,
            x0,
            target_sum,
            run_deadline,
            init_tau_factor=0.20,
            min_tau_factor=0.001,
            alpha0=0.03,
        )

        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] GLOBAL new best={best_score:.6f} (n={n})")
            if best_score < 1.503:               # early exit if we are already below target
                break

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (exact max).
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 8.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_end = time.time() + min(30.0, time_left * 0.5)
        seq_sub, val_sub = _subgradient_descent(best_seq.copy(),
                                                target_sum,
                                                sub_end)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑gradient best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – stochastic hill‑climbing for fine‑tuning.
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 5.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        hill_end = time.time() + time_left
        seq_hill, val_hill = _hillclimb(best_seq.copy(),
                                        target_sum,
                                        hill_end)
        if val_hill < best_score:
            best_score = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] after hill‑climb best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – linear‑programming refinement (single mixing step).
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 4.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        lp_end = time.time() + min(5.0, time_left)   # a few seconds are enough
        seq_lp, val_lp = _lp_refine(best_seq.copy(),
                                    target_sum,
                                    lp_end)
        if val_lp < best_score:
            best_score = val_lp
            best_seq = seq_lp.copy()
            print(f"[{time.time():.2f}] after LP refinement best={best_score:.6f}")

    # --------------------------------------------------------------
    # Final fallback (should never be needed).
    # --------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # Return a plain Python list as required by the evaluator.
    return best_seq.tolist()
