"""
Hybrid optimizer for the step‑function evaluation problem.

Goal
----
Find a non‑negative vector `a` (any length) that minimises

    f(a) = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv` is the ordinary linear convolution.
The score is exactly the maximal convolution value once the sum is
normalised to √(2 n), therefore we try to make the convolution as flat
as possible.

Algorithm – three stages
------------------------
1. **Coarse parametric grid** – a cheap search over a handful of
   analytic families (uniform, power‑law, exponential, cosine).  For
   several lengths (≈ 1.2k … 3k) the best candidate is kept.
2. **Soft‑max Adam refinement** – the best candidate from the grid is
   used as a warm start.  The coefficients are represented as a soft‑max
   of unconstrained logits; a log‑sum‑exp surrogate of `max(conv)`
   supplies a smooth gradient.  Adam runs for a generous time slice,
   slowly cooling both the learning‑rate and the surrogate temperature.
3. **Exact local search** – a projected sub‑gradient descent on the
   true `max(conv)` is followed by a long stochastic hill‑climb
   (Metropolis acceptance) that moves tiny amounts of mass between
   random positions while keeping the simplex constraint.

All three stages share a global best solution and respect the total
budget (default 1000 s).  The implementation is self‑contained, uses
only NumPy and the provided `evaluate_sequence`, and prints progress
messages for debugging.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)            # remove tiny negatives
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σ w = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 1️⃣  Coarse parametric grid search
# ----------------------------------------------------------------------
def _grid_search(seed: int, deadline: float):
    """
    Scan a few analytic families for several lengths.
    Returns the best vector found and its exact evaluation.
    """
    np.random.seed(seed)
    best_seq = None
    best_val = np.inf

    # lengths we try – they are cheap (FFT on ~2‑3k elements)
    lengths = [1200, 1500, 1800, 2100, 2500, 3000]

    for n in lengths:
        target_sum = np.sqrt(2.0 * n)
        # ------- uniform -------------------------------------------------
        a = np.full(n, target_sum / n)
        val = evaluate_sequence(a.tolist())
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[{time.time():.2f}] grid n={n} uniform  val={val:.6f}")

        # ------- power‑law ------------------------------------------------
        for p in np.linspace(0.0, 2.0, 13):          # 0 → uniform
            idx = np.arange(1, n + 1, dtype=float)
            raw = idx ** (-p)
            a = raw / raw.sum() * target_sum
            val = evaluate_sequence(a.tolist())
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[{time.time():.2f}] grid n={n} power p={p:.3f} val={val:.6f}")

        # ------- exponential ---------------------------------------------
        for tau in np.linspace(0.1 * n, 0.9 * n, 9):
            raw = np.exp(-np.arange(n) / tau)
            a = raw / raw.sum() * target_sum
            val = evaluate_sequence(a.tolist())
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[{time.time():.2f}] grid n={n} exp τ={tau:.0f} val={val:.6f}")

        # ------- cosine (edge‑heavy / centre‑heavy) ----------------------
        for alpha in np.linspace(-0.8, 0.8, 9):
            raw = 1.0 + alpha * np.cos(2 * np.pi * np.arange(n) / (n - 1))
            raw = np.clip(raw, 1e-12, None)
            a = raw / raw.sum() * target_sum
            val = evaluate_sequence(a.tolist())
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[{time.time():.2f}] grid n={n} cos α={alpha:.3f} val={val:.6f}")

        # ------- combined power‑law * exponential -------------------------
        for p in (0.5, 1.0, 1.5):
            for tau in (0.3 * n, 0.6 * n, 0.9 * n):
                idx = np.arange(1, n + 1, dtype=float)

                    best_val, best_seq = val, a.copy()
                    print(f"[{time.time():.2f}] grid n={n} pow×exp p={p:.1f} τ={tau:.0f} val={val:.6f}")

        if time.time() >= deadline:

# 2️⃣  Adam optimiser on the log‑softmax parametrisation

def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.20,
    min_tau_factor: float = 0.001,
    alpha0: float = 0.03,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Soft‑max Adam optimiser that minimises a smooth surrogate of
    `max(conv)`.  Returns the best exact solution found.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    # pre‑compute FFT size for the weighted convolution (reused each iter)
    size2 = 1 << ((2 * n - 2).bit_length())

    while time.time() < deadline:
        # ---- map logits → probabilities → coefficients -----------------
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)                 # stable soft‑max
        p = exp_shift / np.sum(exp_shift)            # Σp = 1
        a = target_sum * p

        # ---- exact max (for bookkeeping) ----------------------------
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] ADAM n={n} new best max={cur_max:.6f}")

        tau = max(init_tau_factor * cur_max * (0.995 ** it),
                  min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient w.r.t a (2·a ⊛ rev(w)) ------------------------
        w_rev = w[::-1]
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]   # length 3n‑2
        idx = (2 * n - 2) - np.arange(n)                     # slice picking Σ a_j w_{i+j}
        grad_a = 2.0 * conv2[idx]

        # ---- back‑prop through soft‑max ------------------------------
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ---- Adam update --------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to get out of plateaus
        if it % 250 == 0 and np.random.rand() < 0.25:
            x += np.random.randn(n) * 0.0005

        # slowly decay the base learning rate
        alpha *= 0.9995

    # final exact evaluation of the best candidate we have seen
    final_score = evaluate_sequence(best_a.tolist())
    return best_a, float(final_score)


# ----------------------------------------------------------------------
# 3️⃣  Projected sub‑gradient descent on the true `max(conv)`
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent for the non‑smooth objective.
    The step size adapts automatically.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step_size = 0.05 * target_sum / n           # start modest

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())                  # index of the current max

        # sub‑gradient of the active convolution entry
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # descent step + projection onto the simplex
        a_candidate = a - step_size * grad
        a_candidate = _project_onto_simplex(a_candidate, target_sum)

        val_candidate = _full_convolution_fft(a_candidate, a_candidate).max()
        if val_candidate < best_val - 1e-12:
            a = a_candidate
            best_a = a_candidate
            best_val = val_candidate
            step_size *= 1.03                # slight boost after success
        else:
            step_size *= 0.70                # back‑off on failure

        if it % 400 == 0:
            print(
                f"[{time.time():.2f}] subgrad it={it:5d} step={step_size:.2e} best={best_val:.6f}"
            )
        if best_val < 1.48:                   # early stop – a comfortable margin
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# 4️⃣  Long stochastic hill‑climbing (Metropolis)
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style local search.
    Moves a random fraction of mass from i → j and accepts
    with the Metropolis rule.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # temperature schedule – start warm and cool linearly
    T_start = best_val * 0.4
    T_end = 1e-5
    start = time.time()
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T_start * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0.0:
            continue

        # move a tiny slice of mass
        delta = a[i] * np.random.rand() * 0.08   # at most 8 % of a[i]
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()

        accept = False
        if new_val < best_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hill it={it:7d} new best={best_val:.6f}")

        # occasional re‑normalisation (numerical drift)
        if it % 2500 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.48:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs):
    """
    Returns a list of non‑negative heights that (hopefully) beats the
    current best score 1.503.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # safety margin

    # ------------------------------------------------------------------
    # (A)  Warm‑start from a known construction if it exists
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        if hs_arr.size > 0:
            n0 = hs_arr.shape[0]
            target0 = np.sqrt(2.0 * n0)
            hs_norm = hs_arr * (target0 / hs_arr.sum())
            score0 = evaluate_sequence(hs_norm.tolist())
            best_seq = hs_norm.copy()
            best_score = float(score0)
            print(f"[{time.time():.2f}] warm‑start seed score={best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # (B)  Coarse parametric grid search (fast, cheap)
    # ------------------------------------------------------------------
    grid_deadline = start_time + 0.20 * (deadline - start_time)   # ~20 % of budget
    if time.time() < grid_deadline:
        seq_grid, score_grid = _grid_search(seed + 1, grid_deadline)
        if score_grid < best_score:
            best_score = score_grid
            best_seq = seq_grid.copy()
            print(f"[{time.time():.2f}] after grid best={best_score:.6f}")

    # ------------------------------------------------------------------
    # (C)  Adam refinement – give the most promising length a long run
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        # allocate half of the remaining time to a focused Adam run
        adam_deadline = time.time() + 0.45 * (deadline - time.time())
        # initialise logits from the current best
        x0 = np.log(best_seq + 1e-12)
        seq_adam, score_adam = _optimise_adam_softmax(
            n, x0, target_sum, adam_deadline,
            init_tau_factor=0.25,
            min_tau_factor=0.002,
            alpha0=0.04,
        )
        if score_adam < best_score:
            best_score = score_adam
            best_seq = seq_adam.copy()
            print(f"[{time.time():.2f}] after Adam best={best_score:.6f}")

    # ------------------------------------------------------------------
    # (D)  Projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_deadline = time.time() + 0.20 * (deadline - time.time())
        seq_sub, val_sub = _subgradient_descent(best_seq.copy(), target_sum, sub_deadline)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑grad best={best_score:.6f}")

    # ------------------------------------------------------------------
    # (E)  Final long hill‑climb
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        hill_deadline = deadline
        seq_hill, val_hill = _hillclimb(best_seq.copy(), target_sum, hill_deadline)
        if val_hill < best_score:
            best_score = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] final hill‑climb best={best_score:.6f}")

    # ------------------------------------------------------------------
    # (F)  Fallback – uniform (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # The evaluator expects a plain Python list.
    print(f"[{time.time():.2f}] returning length={best_seq.shape[0]} score={best_score:.6f}")
    return best_seq.tolist()
