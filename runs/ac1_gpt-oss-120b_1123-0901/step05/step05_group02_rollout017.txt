import time, numpy as np

def compute_obj(a):
    n = len(a)
    conv = np.convolve(a, a)
    max_c = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_c / (sum_a**2)
```

```python
"""
Propose a step‑function (non‑negative list) that minimises
    F(a)=2·n·max(a*a)/(sum a)^2

This implementation builds a set of diverse initial shapes, refines each
candidate with Adam on a smooth approximation of the L∞‑norm, then
performs a simulated‑annealing permutation search to further lower the
max convolution.  It respects a global time budget and returns the best
sequence found.
"""
import time
import numpy as np

def _next_pow2(x):
    return 1 << int(np.ceil(np.log2(x)))

def _conv_fft_len(n):
    return _next_pow2(2 * n - 1)

def _full_convolution_fft(a, fft_len):
    """Return linear convolution a*a using FFT of given length."""
    # a is 1‑d np.ndarray of float64
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv

def _cross_corr_fft(w, a, fft_len):
    """
    Compute linear cross‑correlation (w ★ a). Both inputs are length‑n
    arrays; the result has length n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

def _objective(a, fft_len):
    """
    Exact objective value 2·n·max(conv)/(sum^2) for a non‑negative array a.
    """
    n = a.shape[0]
    conv = _full_convolution_fft(a, fft_len)
    max_c = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_c / (sum_a * sum_a)

def _adam_optimize(a0, fft_len_conv, fft_len_corr, deadline):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best array found (normalised to sum=1) and its objective.
    """
    n = a0.shape[0]
    # Normalise to sum 1
    a = a0 / a0.sum()
    # Log‑softmax parameters
    logits = np.log(a + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.3
    lam_factor = 1.015

    best_a = a.copy()
    best_val = _objective(a, fft_len_conv)

    last_improve = time.time()
    # iteration budget roughly proportional to n
    max_iters = int(25000 * 1024 / n) + 5000

    for t in range(1, max_iters + 1):
        # soft‑max to stay on simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution_fft(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max (chain rule through convolution)
        grad_a = 2.0 * _cross_corr_fft(w, a, fft_len_corr)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log * grad_log)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        lam = min(lam * lam_factor, 1e6)

        # exact objective (max_c is exact max of convolution)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={t}] new best {val:.6f}")

        # small random shake if stuck
        if time.time() - last_improve > 8.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    # final pass: return best array (still sum‑1)
    return best_a, best_val

def _permute_anneal(a0, fft_len_conv, deadline):
    """
    Simulated‑annealing over permutations of a0.
    Returns best permutation found and its objective.
    """
    n = a0.shape[0]
    cur = a0.copy()
    cur_val = _objective(cur, fft_len_conv)

    best_seq = cur.copy()
    best_val = cur_val

    # temperature schedule: start relatively high, decay linearly
    T0 = 0.1
    max_iters = 200000  # arbitrary large, will stop earlier by time

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # propose a simple swap
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # perform swap
        cur[i], cur[j] = cur[j], cur[i]
        new_val = _objective(cur, fft_len_conv)

        if new_val < cur_val:
            accept = True
        else:
            # probabilistic acceptance
            t = T0 * (1.0 - it / max_iters)
            prob = np.exp(-(new_val - cur_val) / max(t, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur.copy()
                # optional print
                print(f"[Perm n={n} iter={it}] new best {new_val:.6f}")
        else:
            # revert swap
            cur[i], cur[j] = cur[j], cur[i]

        # occasional random shake to escape plateaus
        if it % 5000 == 0 and time.time() - deadline > -5:
            # Not needed
            pass

    return best_seq, best_val

def _initial_candidates(n):
    """
    Produce a list of initial numpy arrays for length n.
    All candidates are normalised to sum 1 before returning.
    """
    candidates = []

    # Dirichlet(1) – uniform random on simplex
    candidates.append(np.random.dirichlet(np.ones(n)))

    # Concentrated Dirichlet (α < 1)
    for concentration in [0.2, 0.5, 0.8]:
        candidates.append(np.random.dirichlet(np.full(n, concentration)))

    # Edge‑concentrated power‑law ((i+1)*(n-i))**α, α<0
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.1, 9):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # Two‑sided exponential
    for gamma in [0.0005, 0.001, 0.002, 0.005]:
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # Slightly smoothed tent (rise/fall)
    left = np.linspace(0, 1, n // 2, endpoint=False)
    right = left[::-1]
    tent = np.concatenate([left, right])
    tent = np.maximum(tent, 0)
    tent /= tent.sum()
    candidates.append(tent)

    # Add small noise versions of the best known global (if any)
    # This will be handled at top level
    return candidates

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with low evaluate_sequence() score.
    Uses Adam optimisation on a smooth‑max surrogate followed by a
    permutation annealing phase.  The function respects the supplied time
    budget and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq = None
    best_val = float("inf")

    # Choose lengths – longer vectors give more freedom but cost more.
    lengths = [2048, 4096, 8192]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        fft_len_conv = _conv_fft_len(n)
        # for cross‑correlation we need length at least 3·n
        fft_len_corr = _next_pow2(fft_len_conv + n)

        # Build initial candidates
        candidates = _initial_candidates(n)

        # Warm‑start from best sequence if it already has the right length.
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.asarray(best_seq, dtype=np.float64))

        # Global historic solution (if any)
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            cand = np.asarray(hseq, dtype=np.float64)
            if cand.shape[0] != n:
                # truncate or pad
                if cand.shape[0] > n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s < 0.01:
                cand = np.random.dirichlet(np.ones(n))
            else:
                cand = cand / s
            candidates.append(cand)

        # Iterate over candidates
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # Normalise to simplex
            a0 = a0 / a0.sum()
            # Adam optimisation
            adam_seq, adam_val = _adam_optimize(a0, fft_len_conv, fft_len_corr, deadline)
            if adam_val < best_val:
                best_val = adam_val
                best_seq = adam_seq.copy()
                print(f"[n={n} cand={cand_id}] Adam global best {adam_val:.6f}")

            # Permutation annealing – start from Adam result
            perm_seq, perm_val = _permute_anneal(adam_seq, fft_len_conv, deadline)
            if perm_val < best_val:
                best_val = perm_val
                best_seq = perm_seq.copy()
                print(f"[n={n} cand={cand_id}] Perm global best {perm_val:.6f}")

            # Optional short exponentiated‑gradient refinement (sub‑gradient)
            # Use the same code as before but with fewer iterations
            a = perm_seq.copy()
            eta = 0.55
            sub_last = time.time()
            max_sub_iters = 2000
            for sub_it in range(max_sub_iters):
                if time.time() > deadline:
                    break
                conv = _full_convolution_fft(a, fft_len_conv)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)
                grad_a = 2.0 * _cross_corr_fft(w, a, fft_len_corr)
                a = a * np.exp(-eta * grad_a)
                # normalise
                s = a.sum()
                if s < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[n={n} cand={cand_id} Subgrad iter={sub_it}] new best {val:.6f}")
                # early stop if no improvement for a while
                if time.time() - sub_last > 8.0:
                    break
                sub_last = time.time()
        # End of candidates loop

        # Final fine‑tuning for this length (if we have a best of this length)
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] Fine‑tuning global best (val={best_val:.6f})")
            ft_seq, ft_val = _adam_optimize(best_seq, fft_len_conv, fft_len_corr, deadline)
            if ft_val < best_val:
                best_val = ft_val
                best_seq = ft_seq.copy()
                print(f"[n={n}] Fine‑tune improved to {ft_val:.6f}")

    # If nothing found, fall back to uniform vector of length 2048
    if best_seq is None:
        n0 = 2048

"""
Propose a step‑function (non‑negative list) that approximately minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

The search proceeds in three phases for each chosen length *n*:

1. **Diverse initialisations** – Dirichlet samples, edge‑concentrated
   power‑laws, two‑sided exponentials, a small “tent” shape and, if
   available, the historic best solution.

2. **Adam optimisation** of a smooth‑max surrogate of the convolution
   peak.  The parameters live on the probability simplex via a soft‑max
   (logits).  Temperature *λ* grows exponentially so the surrogate
   approaches the true ℓ∞‑norm.

3. **Permutation annealing** – the values themselves are kept fixed,
   but their order is optimised by a simulated‑annealing swap walk.
   This reduces the maximum of the (exact) linear convolution, which
   is the only term that matters for the final score.

A tiny sub‑gradient (exponentiated‑gradient) polish is run after the
permutation phase, and a final fine‑tune (Adam) is performed on the
global best of the current length.  The routine respects a global time
budget (default 1000 s) and returns the best sequence discovered as a
plain Python list.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << int(np.ceil(np.log2(x)))


def _conv_fft_len(n: int) -> int:
    """FFT length for a linear convolution of two length‑n vectors."""
    return _next_pow2(2 * n - 1)


def _full_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of prescribed length."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the
    smooth‑max.  Both inputs have length *n* and the result is length *n*.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, fft_len: int) -> float:
    """Exact evaluation 2·n·max(conv)/(sum²) for a non‑negative array a."""
    n = a.shape[0]
    conv = _full_convolution_fft(a, fft_len)
    max_c = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_c / (sum_a * sum_a)


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   fft_len_conv: int,
                   fft_len_corr: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a smooth‑max surrogate.
    Returns the best array (normalised to sum = 1) and its exact objective.
    """
    n = a0.shape[0]
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)                     # soft‑max parameters

    # Adam state
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.3                # temperature for smooth‑max
    lam_factor = 1.015

    best_a = a.copy()
    best_val = _objective(a, fft_len_conv)

    last_improve = time.time()
    max_iters = int(25000 * 1024 / n) + 5000   # ≈10 k for n≈4 k

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ---- simplex via soft‑max ------------------------------------
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ---- smooth‑max of the convolution -------------------------
        conv = _full_convolution_fft(a, fft_len_conv)
        max_c = conv.max()
        # avoid overflow in the exponential
        w = np.exp(np.clip(lam * (conv - max_c), -700, 700))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient (through the linear convolution) -------------
        grad_a = 2.0 * _cross_corr_fft(w, a, fft_len_corr)

        # ---- back‑prop through the soft‑max ------------------------
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update -------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log * grad_log)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- raise temperature --------------------------------------
        lam = min(lam * lam_factor, 1e6)

        # ---- exact objective (max_c is the true maximum) ----------
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={t:05d}] new best {val:.6f}")

        # ---- shake if stuck -----------------------------------------
        if time.time() - last_improve > 8.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    # return the best we have seen (still normalised)
    return best_a, best_val


# ----------------------------------------------------------------------
def _permute_anneal(a0: np.ndarray,
                    fft_len_conv: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing over permutations of a0.
    The multiset of values is kept unchanged; only their order varies.
    """
    n = a0.shape[0]
    cur = a0.copy()
    cur_val = _objective(cur, fft_len_conv)

    best_seq = cur.copy()
    best_val = cur_val

    # Linear temperature schedule (starts at 0.1 and ends at ~0)
    T0 = 0.1
    max_iters = 200_000               # large upper bound; time stops earlier

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # propose swap
        cur[i], cur[j] = cur[j], cur[i]
        new_val = _objective(cur, fft_len_conv)

        if new_val < cur_val:
            accept = True
        else:
            # probabilistic acceptance
            temp = T0 * (1.0 - it / max_iters)
            prob = np.exp(-(new_val - cur_val) / max(temp, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur.copy()
                print(f"[Perm n={n} iter={it:06d}] new best {new_val:.6f}")
        else:
            # revert swap
            cur[i], cur[j] = cur[j], cur[i]

        # occasional forced shake to avoid dead‑ends (very cheap)
        if it % 5000 == 0 and time.time() - deadline > -5.0:
            pass

    return best_seq, best_val


# ----------------------------------------------------------------------
def _initial_candidates(n: int) -> list[np.ndarray]:
    """
    Build a modest pool of diverse starting points for length *n*.
    All vectors are normalised to sum = 1 before being returned.
    """
    candidates = []

    # Dirichlet(1) – uniform on the simplex
    candidates.append(np.random.dirichlet(np.ones(n)))

    # Concentrated Dirichlet (α < 1) – produces spikier vectors
    for conc in (0.2, 0.5, 0.8):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # Edge‑concentrated power‑law ((i+1)*(n‑i))**α, α<0
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.1, 9):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # Two‑sided exponential decay
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # Simple “tent” shape (linear rise then fall)
    left = np.linspace(0, 1, n // 2, endpoint=False)
    right = left[::-1]
    tent = np.concatenate([left, right])
    tent = np.maximum(tent, 0)
    tent /= tent.sum()
    candidates.append(tent)

    return candidates


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a safety margin

    best_seq = None
    best_val = float("inf")

    # Lengths to explore – larger n gives more freedom but costs more.
    lengths = [2048, 4096, 8192]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        fft_len_conv = _conv_fft_len(n)
        # cross‑correlation length needs at most conv_len + n – 1
        fft_len_corr = _next_pow2(2 * n - 1 + n)

        # Gather starting points.
        candidates = _initial_candidates(n)

        # Warm‑start from the globally best solution (if of same size).
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.asarray(best_seq, dtype=np.float64))

        # Historic solution, if the environment provides one.
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            cand = np.asarray(hseq, dtype=np.float64)
            if cand.shape[0] > n:
                cand = cand[:n]
            elif cand.shape[0] < n:
                cand = np.pad(cand, (0, n - cand.shape[0]))
            cand = np.maximum(cand, 0.0)
            if cand.sum() < 0.01:
                cand = np.random.dirichlet(np.ones(n))
            else:
                cand = cand / cand.sum()
            candidates.append(cand)

        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # -----------------------------------------------------------
            # 1) Adam on smooth‑max
            # -----------------------------------------------------------
            adam_seq, adam_val = _adam_optimize(
                a0, fft_len_conv, fft_len_corr, deadline
            )
            if adam_val < best_val:
                best_val = adam_val
                best_seq = adam_seq.copy()
                print(f"[n={n} cand={cand_id}] Adam global best {adam_val:.6f}")

            # -----------------------------------------------------------
            # 2) Permutation annealing (keeps the multiset of numbers)
            # -----------------------------------------------------------
            perm_seq, perm_val = _permute_anneal(
                adam_seq, fft_len_conv, deadline
            )
            if perm_val < best_val:
                best_val = perm_val
                best_seq = perm_seq.copy()
                print(f"[n={n} cand={cand_id}] Perm global best {perm_val:.6f}")

            # -----------------------------------------------------------
            # 3) Tiny exponentiated‑gradient (sub‑gradient) polish
            # -----------------------------------------------------------
            a = perm_seq.copy()
            eta = 0.55
            sub_last = time.time()
            max_sub_iters = 2000
            for sub_it in range(max_sub_iters):
                if time.time() > deadline:
                    break
                conv = _full_convolution_fft(a, fft_len_conv)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)
                grad_a = 2.0 * _cross_corr_fft(w, a, fft_len_corr)
                a = a * np.exp(-eta * grad_a)

                s = a.sum()
                if s < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[n={n} cand={cand_id} Subgrad iter={sub_it}] new best {val:.6f}")

                if time.time() - sub_last > 8.0:
                    break
                sub_last = time.time()

        # -----------------------------------------------------------------
        # Fine‑tuning of the *global* best for the current length
        # -----------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] Fine‑tuning global best (val={best_val:.6f})")
            ft_seq, ft_val = _adam_optimize(
                best_seq, fft_len_conv, fft_len_corr, deadline
            )
            if ft_val < best_val:
                best_val = ft_val
                best_seq = ft_seq.copy()
                print(f"[n={n}] Fine‑tune improved to {ft_val:.6f}")

    # --------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (should never happen)
    # --------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Convert to a plain Python list as required by the external evaluator.
    return best_seq.tolist()
