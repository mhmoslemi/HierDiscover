def generate_initial_candidates(n):
    candidates = []
    # Uniform
    candidates.append(np.full(n, 1.0/n))
    # Random Dirichlet (conc=1, conc=0.3)
    candidates.append(np.random.dirichlet(np.ones(n)))
    candidates.append(np.random.dirichlet(np.full(n, 0.3)))
    # Exponential shapes with gamma values list
    for gamma in [0.0005, 0.001, 0.002, 0.004, 0.008]:
        i = np.arange(n, dtype=float)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n-1-i))
        shape /= shape.sum()
        candidates.append(shape)
    # Power law shapes:
    idx = np.arange(n, dtype=float)+1
    prod = idx * (n - idx + 1)
    for alpha in np.linspace(-2.0, -0.2, 9):
        shape = np.power(prod, alpha, where=prod>0, out=np.zeros_like(prod))
        if shape.sum()>0:
            shape /= shape.sum()
            candidates.append(shape)
    # Sinusoid shapes:
    for p in [1,2,3]:
        shape = np.sin(np.pi * np.arange(1, n+1) / (n+1))**p
        shape = np.maximum(shape,0)
        shape /= shape.sum()
        candidates.append(shape)
    # Cosine shapes
    shape = (1 + np.cos(2*np.pi*np.arange(n)/n))/2
    shape /= shape.sum()
    candidates.append(shape)
    # Tent shape:
    left = np.linspace(0,1,n//2,endpoint=False)
    right = left[::-1]
    tent = np.concatenate([left,right])[:n]
    tent/=tent.sum()
    candidates.append(tent)
    # Slightly random Dirichlet with small concentration:
    candidates.append(np.random.dirichlet(np.full(n,0.1)))
    return candidates

"""
Propose a step‑function (non‑negative list) that approximately minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

by a multi‑stage search:
  * a rich set of analytic initial shapes,
  * Adam optimisation on a smooth‑max surrogate of the convolution peak,
  * exponentiated‑gradient sub‑gradient optimisation on the exact max,
  * a cheap stochastic local‑search (Gaussian perturbations + short Adam rolls).
The routine respects a time budget (default 1000 s) and returns the best
sequence discovered as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities --------------------------------------------------------
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT of length ``fft_len`` (a power of two).
    Returns an array of length ``2*len(a)-1``.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    The result has length len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective (sum(a) is assumed to be 1)."""
    conv = _full_convolution(a, fft_len)
    return 2.0 * n * conv.max()


def _generate_initial_candidates(n: int) -> list[np.ndarray]:
    """
    Produce a diverse collection of start vectors of length ``n``.
    All candidates are normalised to have sum = 1.
    """
    candidates = []

    # 1) Uniform (baseline)
    candidates.append(np.full(n, 1.0 / n))

    # 2) Random Dirichlet (several concentrations)
    candidates.append(np.random.dirichlet(np.ones(n)))                # conc = 1
    candidates.append(np.random.dirichlet(np.full(n, 0.3)))           # more spiky
    candidates.append(np.random.dirichlet(np.full(n, 0.8)))           # smoother

    # 3) Two‑sided exponential families
    for gamma in [0.0005, 0.001, 0.002, 0.004, 0.008]:
        i = np.arange(n, dtype=float)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # 4) Power‑law shapes  a_i ∝ (i+1)(n‑i)^α  with negative α
    idx = np.arange(1, n + 1, dtype=float)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 9):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # 5) Sinusoidal families
    for p in [1, 2, 3]:
        shape = np.sin(np.pi * np.arange(1, n + 1) / (n + 1)) ** p
        shape = np.maximum(shape, 0)
        shape /= shape.sum()
        candidates.append(shape)

    # 6) Cosine bump
    shape = (1.0 + np.cos(2.0 * np.pi * np.arange(n) / n)) / 2.0
    shape /= shape.sum()
    candidates.append(shape)

    # 7) Tent (linear rise then fall)
    left = np.linspace(0.0, 1.0, n // 2, endpoint=False)
    right = left[::-1]
    tent = np.concatenate([left, right])[:n]
    if tent.sum() > 0:
        tent /= tent.sum()
        candidates.append(tent)

    # 8) Slightly “noisy’’ Dirichlet with very low concentration
    candidates.append(np.random.dirichlet(np.full(n, 0.05)))

    # 9) Warm start from a globally stored best of same length, if available
    hseq = globals().get("height_sequence_1")
    if isinstance(hseq, (list, np.ndarray)):
        arr = np.asarray(hseq, dtype=float)
        if arr.shape[0] > n:
            arr = arr[:n]
        elif arr.shape[0] < n:
            arr = np.pad(arr, (0, n - arr.shape[0]), constant_values=0.0)
        arr = np.maximum(arr, 0.0)
        if arr.sum() > 1e-9:
            arr /= arr.sum()
            candidates.append(arr)

    return candidates


# ----------------------------------------------------------------------
# Optimisation primitives -------------------------------------------------
# ----------------------------------------------------------------------
def _adam_optimize(
    logits: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    max_iters: int,
    lr: float,
    lam_init: float,
    lam_factor: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate.
    Returns the best vector (probability simplex) seen and its objective.
    """
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lam = lam_init
    best_val = float("inf")
    best_a = None
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # ---- simplex via soft‑max ----
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ---- convolution and smooth‑max weights ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the surrogate ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- back‑prop through the soft‑max ----
        avg = np.dot(a, grad_a)                     # ⟨a,∇⟩
        grad_log = a * (grad_a - avg)               # ∂L/∂logits

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, 1e6)

        # ---- objective (exact, because sum(a)=1) ----
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # ---- occasional random shake to escape plateaus ----
        if time.time() - last_improve > 8.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    return best_a if best_a is not None else a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    max_iters: int,
    eta: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient on the exact max‑convolution.
    Returns the best sequence seen and its objective.
    """
    a = a0.copy()
    best_val = float("inf")
    best_a = None
    last_improve = time.time()

    for it in range(max_iters):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # active set (within numerical tolerance)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.full_like(conv, 1.0 / conv.size)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # early stop if stagnant
        if time.time() - last_improve > 8.0:
            break
        if time.time() > deadline:
            break

    return best_a if best_a is not None else a, best_val


def _random_local_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    steps: int,
    sigma: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic refinement: Gaussian perturbations + renormalisation.
    Accepts only improving moves (hill‑climbing).
    """
    a = a0.copy()
    best_val = _objective(a, n, conv_fft_len)
    best_a = a.copy()
    last_improve = time.time()

    for _ in range(steps):
        if time.time() > deadline:
            break

        # propose a perturbation on the log‑space (more scale‑invariant)
        log_a = np.log(a + 1e-15)
        log_a += np.random.randn(n) * sigma
        a_prop = np.exp(log_a)
        a_prop = np.maximum(a_prop, 0.0)
        s = a_prop.sum()
        if s < 1e-12:
            continue
        a_prop /= s

        val = _objective(a_prop, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a_prop.copy()
            a = a_prop.copy()
            last_improve = time.time()
            # gradually shrink the perturbation scale when we improve
            sigma *= 0.98

        # if we have not improved for a while, shrink sigma aggressively
        if time.time() - last_improve > 6.0:
            sigma *= 0.9
            last_improve = time.time()

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # Lengths to explore – larger lengths give more expressive power but cost more.
    lengths = [2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)    # sufficient for cross‑correlation

        # --------------------------------------------------------------
        # Assemble initial candidates
        # --------------------------------------------------------------
        cand_list = _generate_initial_candidates(n)

        # --------------------------------------------------------------
        # Main optimisation loop over candidates
        # --------------------------------------------------------------
        for cid, cand in enumerate(cand_list):
            if time.time() > deadline:
                break

            # Normalise to simplex (sum == 1) – this does not affect the objective.
            a0 = cand.astype(np.float64)
            a0 = np.maximum(a0, 0.0)
            if a0.sum() < 1e-9:
                a0 = np.random.dirichlet(np.ones(n))
            a0 /= a0.sum()

            logits = np.log(a0 + 1e-15)

            # ---- Adam (smooth‑max) ----
            max_iters_adam = int(20000 * 1024 / n) + 3000
            a_adam, val_adam = _adam_optimize(
                logits=logits.copy(),
                n=n,
                conv_fft_len=conv_fft_len,
                corr_fft_len=corr_fft_len,
                max_iters=max_iters_adam,
                lr=0.07,
                lam_init=0.4,
                lam_factor=1.015,
                deadline=deadline,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n} cand={cid} Adam] new global best {best_val:.6f}")

            # ---- Sub‑gradient (exponentiated gradient) ----
            max_iters_sub = int(4000 * 1024 / n) + 500
            a_sub, val_sub = _subgradient_optimize(
                a0=a_adam,
                n=n,
                conv_fft_len=conv_fft_len,
                corr_fft_len=corr_fft_len,
                max_iters=max_iters_sub,
                eta=0.55,
                deadline=deadline,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[n={n} cand={cid} Subgrad] new global best {best_val:.6f}")

            # ---- Stochastic local refinement ----
            steps_local = int(3000 * 1024 / n) + 500
            a_loc, val_loc = _random_local_search(
                a0=a_sub,
                n=n,
                conv_fft_len=conv_fft_len,
                steps=steps_local,
                sigma=0.02,
                deadline=deadline,
            )
            if val_loc < best_val:
                best_val = val_loc
                best_seq = a_loc.copy()
                print(f"[n={n} cand={cid} Local] new global best {best_val:.6f}")

        # --------------------------------------------------------------
        # Fine‑tune the *current* global best for this length (if any)
        # --------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] fine‑tuning the best of this length (val={best_val:.6f})")
            logits = np.log(best_seq + 1e-15)
            # tighter Adam run
            fine_iters = int(40000 * 1024 / n) + 8000
            a_fine, val_fine = _adam_optimize(
                logits=logits,
                n=n,
                conv_fft_len=conv_fft_len,
                corr_fft_len=corr_fft_len,
                max_iters=fine_iters,
                lr=0.045,
                lam_init=0.6,
                lam_factor=1.01,
                deadline=deadline,
            )
            if val_fine < best_val:
                best_val = val_fine
                best_seq = a_fine.copy()
                print(f"[n={n} fine‑tune] new global best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Convert to plain Python list (required by external evaluator)
    return best_seq.tolist()
