"""
Optimiser for the max‑auto‑convolution bound.

The algorithm works in three stages:

1. **Fast analytical sweep** – for each problem size we generate a
   large collection of hand‑crafted families (power‑law edge shapes,
   double‑exponential edges, sine‑powered bumps, random Dirichlet draws,
   uniform, warm‑starts).  Every candidate is evaluated with a cheap FFT‑based
   bound and the best few are remembered.

2. **Heavy local optimisation** – the retained candidates are refined by
   two phases:
      * Adam on a smooth‑max surrogate (log‑sum‑exp) of the
        convolution maximum.  The temperature is annealed aggressively.
      * A sub‑gradient / exponentiated‑gradient phase that works on the
        true max using an active‑set weight vector.

   Hyper‑parameters (iteration budgets, learning rates, temperature schedule)
   are tuned aggressively for the allotted wall‑clock time.

3. **Simple stochastic hill‑climbing** – after the deterministic refinements
   a handful of tiny random mass transfers are tried; any improvement is
   kept.

All work respects the global time budget (default 1000 s).  The best
sequence found is returned as a plain Python list, ready for
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities – identical to the reference implementation.
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.

    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


def _fast_bound(a: np.ndarray, conv_fft_len: int) -> float:
    """
    Compute the evaluation bound 2·n·max(a*a) (the denominator is 1 because
    we keep a normalised to sum(a)=1).  Uses the same FFT routine as the
    official evaluator for consistency.
    """
    n = a.shape[0]
    conv = _full_convolution(a, fft_len=conv_fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence(...).  The routine obeys the given time budget and
    returns the best sequence discovered (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # slight safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Lengths – we start small (cheap) and finish with the largest size we
    # can afford.  Larger sizes usually give a slightly smaller bound.
    # ------------------------------------------------------------------
    lengths = [4096, 8192, 16384]   # 16384 is the size used in the reference run

    # ------------------------------------------------------------------
    # Analytic families to sweep (global, reused for every length)
    # ------------------------------------------------------------------
    # power‑law edge shapes: a_i ∝ (i·(n-i+1))^α
    alpha_grid = np.linspace(-2.5, -0.2, 23)      # fine grid
    # double‑exponential edge shapes: a_i ∝ exp(-γ·i²)+exp(-γ·(n-1-i)²)
    gamma_grid = np.logspace(-2, 1, 13)
    # sine‑power bumps: sin(...)^p
    p_grid = [1, 2, 3, 4, 5, 6]

    # ------------------------------------------------------------------
    # Main loop over lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

        # --------------------------------------------------------------
        # Build a big, heterogeneous set of start vectors
        # --------------------------------------------------------------
        start_vectors = []

        # (a) uniform
        start_vectors.append((np.full(n, 1.0 / n, dtype=np.float64), "uniform"))

        # (b) warm‑start from a previous solution of exactly this size
        if best_seq is not None and len(best_seq) == n:
            start_vectors.append((np.asarray(best_seq, dtype=np.float64), "global_best"))

        # (c) up‑sample a shorter best (linear interpolation)
        if best_seq is not None and len(best_seq) < n:
            old = np.asarray(best_seq, dtype=np.float64)
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, old)
            up = up / up.sum() if up.sum() > 0 else np.random.dirichlet(np.ones(n))
            start_vectors.append((up, "upsampled"))

        # (d) historic global variable, if present
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            h = np.asarray(hseq, dtype=np.float64)
            if h.shape[0] > n:
                h = h[:n]
            elif h.shape[0] < n:
                h = np.pad(h, (0, n - h.shape[0]))
            h = np.maximum(h, 0.0)
            h = h / h.sum() if h.sum() > 0 else np.random.dirichlet(np.ones(n))
            start_vectors.append((h, "global_historic"))

        # (e) random Dirichlet draws – a spread of concentration values
        for _ in range(25):
            conc = np.random.uniform(0.2, 5.0)   # 0.2 → sparse, 5 → close to uniform
            vec = np.random.dirichlet(np.full(n, conc))
            start_vectors.append((vec, "rand_dir"))

        # (f) power‑law edge families
        idx = np.arange(1, n + 1, dtype=np.float64)
        prod = idx * (n - idx + 1.0)
        for alpha in alpha_grid:
            shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
            if shape.sum() > 0:
                shape = shape / shape.sum()
                start_vectors.append((shape, f"pow_{alpha:.2f}"))

        # (g) double‑exponential edge families
        i = np.arange(n, dtype=np.float64)
        for gamma in gamma_grid:
            shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            shape = shape / shape.sum()
            start_vectors.append((shape, f"exp_{gamma:.2e}"))

        # (h) sine‑power bumps
        base = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        for p in p_grid:
            shape = base ** p
            shape = shape / shape.sum()
            start_vectors.append((shape, f"sine_{p}"))

        # --------------------------------------------------------------
        # Quick scan – keep the few most promising candidates
        # --------------------------------------------------------------
        promising = []                # list of (value, vector, description)
        for vec, descr in start_vectors:
            if time.time() > deadline:
                break
            # small epsilon avoids pathological zero rows
            a = np.maximum(vec, 1e-16)
            a = a / a.sum()
            val = _fast_bound(a, conv_fft_len)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[{descr} n={n}] new global best {val:.6f}")
            promising.append((val, a, descr))

        # keep the top‑K for heavy optimisation
        promising.sort(key=lambda t: t[0])
        top_k = 5
        top_candidates = [(a, descr) for _, a, descr in promising[:top_k]]

        # --------------------------------------------------------------
        # Heavy optimiser – Adam on smooth‑max + exact sub‑gradient
        # --------------------------------------------------------------
        for cand_id, (a0, descr) in enumerate(top_candidates):
            if time.time() > deadline:
                break

            # Initialise from a0 (already normalised)
            a = np.maximum(a0, 1e-16)
            a = a / a.sum()
            logits = np.log(a)

            # Adam hyper‑parameters
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2 = 0.9, 0.999
            eps_adam = 1e-8
            lr = 0.05

            # smooth‑max temperature
            lam = 0.5
            lam_factor = 1.02

            last_improve = time.time()

            # iteration budget – larger n gets fewer but still plenty
            max_iters = int(40000 * 1024 / n) + 5000

            for it in range(1, max_iters + 1):
                # soft‑max (parameter → probability simplex)
                mx = logits.max()
                exp_log = np.exp(logits - mx)
                a = exp_log / exp_log.sum()

                # convolution and smooth‑max weighting
                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # gradient w.r.t. a (smooth surrogate)
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

                # back‑prop through the soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update on the logits
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                # anneal the temperature
                lam = min(lam * lam_factor, 1e7)

                # exact objective (sum(a) = 1 → denominator = 1)
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} {descr} Adam it={it:05d}] new best {val:.6f}")

                # stagnation handling – tiny random shake
                if time.time() - last_improve > 5.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # ------------------------------------------------------
            # Sub‑gradient refinement on the true max (active‑set step)
            # ------------------------------------------------------
            a = a / a.sum()
            eta = 0.6
            sub_last = time.time()
            for sub_it in range(3000):
                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)
                # exponentiated‑gradient step (keeps a non‑negative)
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:                     # re‑initialise if we collapsed
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last = time.time()
                    print(f"[n={n} {descr} Subgrad it={sub_it:04d}] new best {val:.6f}")

                if time.time() - sub_last > 5.0:
                    break
                if time.time() > deadline:
                    break

        # --------------------------------------------------------------
        # Tiny stochastic hill‑climbing around the current best for this n
        # --------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            a = best_seq.copy()
            for _ in range(2000):
                if time.time() > deadline:
                    break
                i, j = np.random.randint(0, n, size=2)
                if i == j:
                    continue
                eps = np.random.rand() * 5e-4          # very small mass transfer
                if a[i] < eps:
                    continue
                a_new = a.copy()
                a_new[i] -= eps
                a_new[j] += eps
                # normalise is not needed – sum unchanged
                val = _fast_bound(a_new, conv_fft_len)
                if val < best_val:
                    best_val = val
                    best_seq = a_new.copy()
                    a = a_new
                    print(f"[n={n} hill‑climb] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was ever discovered (unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return plain Python list as required by the external evaluator
    return best_seq.tolist()
