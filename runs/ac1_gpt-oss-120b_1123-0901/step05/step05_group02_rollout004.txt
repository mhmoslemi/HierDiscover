"""
improved_step_search.py

Search for a non‑negative step function (list of heights) that minimises

        F(a) = 2·n·max(conv(a,a)) / (sum(a)²)

where `conv` is the linear convolution.  The objective is homogeneous, so the
search works with probability vectors (sum = 1) and finally returns the
unnormalised vector – any scaling yields the same score.

The algorithm is a hybrid of several optimisation ideas:

*   A rich pool of initial shapes (random Dirichlet, power‑law edges,
    double‑sided exponentials, Hann/Blackman windows, Gaussian, sin‑powers,
    mixtures and up‑sampled previously found solutions).

*   Phase‑1 – Adam optimiser on a smooth‑max surrogate of ``max(conv)``.
    The temperature (smooth‑max sharpness) grows quickly (factor 1.04) so
    the surrogate quickly focuses on the true maximum.

*   Phase‑2 – Exponentiated‑gradient (mirror‑descent) on the exact
    ``max(conv)`` using the sub‑gradient formed by the active shifts.

*   Phase‑3 – A short simulated‑annealing / random‑shake refinement that
    moves mass between two coordinates while preserving the simplex constraint.

*   For each length we run a fine‑tuning Adam pass on the current global
    best of that length.

Time is limited by the ``budget_s`` argument (default 1000 s).  The code
frequently checks the remaining budget and aborts gracefully.  All
operations run on the CPU and use only NumPy and the standard library.

The function ``propose_candidate`` complies with the required signature
and returns a plain ``list[float]``.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Return the next power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length `fft_len`."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # convolution length = 2*len(a)-1, slice the useful part
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Returned vector has length len(a).  Implements

        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence that minimises the evaluation function.
    The function respects the given time budget and returns the best sequence
    discovered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin for the caller

    best_seq = None          # best numpy array found so far
    best_val = float("inf")  # best value of 2·n·max(conv)

    # ------------------------------------------------------------------
    # Search lengths – larger lengths give more expressive power,
    # but FFT work grows ~n·log n.
    lengths = [2048, 4096, 8192, 16384]

    # Pre‑compute FFT lengths (powers of two) for all considered n.
    fft_len_conv = {}
    fft_len_corr = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_len_conv[n] = _next_pow2(conv_len)
        # cross‑correlation needs conv_len + n - 1 ≤ 3·n
        fft_len_corr[n] = _next_pow2(conv_len + n - 1)

    # ------------------------------------------------------------------
    # Helper: evaluate the current objective (sum of a is assumed to be 1)
    def _objective(a: np.ndarray, n: int) -> float:
        """Return 2·n·max(conv(a,a)) for a probability vector."""
        conv = _full_convolution(a, fft_len_conv[n])
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_len_fft = fft_len_conv[n]
        corr_len_fft = fft_len_corr[n]

        # ------------------------------------------------------------------
        # Build a diverse set of initial candidates (numpy arrays, sum=1)
        candidates = []

        # Warm‑start from a previously found solution of the same length.
        if best_seq is not None and len(best_seq) == n:
            candidates.append(best_seq.astype(np.float64))

        # Upsample a previously found solution of a different length.
        if best_seq is not None and len(best_seq) != n:
            old = best_seq.astype(np.float64)
            old_x = np.linspace(0, 1, old.shape[0])
            new_x = np.linspace(0, 1, n)
            up = np.interp(new_x, old_x, old)
            up = np.maximum(up, 0.0)
            if up.sum() < 0.01:
                up = np.random.dirichlet(np.ones(n))
            else:
                up /= up.sum()
            candidates.append(up)

        # ---- Random Dirichlet families -------------------------------------------------
        candidates.append(np.random.dirichlet(np.ones(n)))          # uniform Dirichlet
        for concentration in (0.2, 0.5, 0.8):                       # spiky Dirichlet
            candidates.append(
                np.random.dirichlet(np.full(n, concentration))
            )
        # Random mixtures of two Dirichlets
        for _ in range(2):
            w = np.random.rand()
            d1 = np.random.dirichlet(np.ones(n))
            d2 = np.random.dirichlet(np.full(n, 0.3))
            mix = w * d1 + (1 - w) * d2
            candidates.append(mix / mix.sum())

        # ---- Deterministic shapes -------------------------------------------------------
        idx = np.arange(n, dtype=np.float64)
        # Power‑law edge shapes (mass near the ends)
        prod = (idx + 1) * (n - idx)
        for alpha in np.linspace(-2.0, -0.2, 9):
            shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
            if shape.sum() > 0:
                candidates.append(shape / shape.sum())

        # Double‑sided exponential
        for gamma in (0.0005, 0.001, 0.002, 0.005):
            shape = np.exp(-gamma * idx) + np.exp(-gamma * (n - 1 - idx))
            candidates.append(shape / shape.sum())

        # Hann window (raised cosine)
        shape = 0.5 - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1))
        candidates.append(shape / shape.sum())

        # Blackman window
        shape = (
            0.42
            - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1))
            + 0.08 * np.cos(4.0 * np.pi * idx / (n - 1))
        )
        candidates.append(shape / shape.sum())

        # Gaussian bell centred in the middle
        sigma = n / 8.0
        shape = np.exp(
            -0.5 * ((idx - (n - 1) / 2.0) / sigma) ** 2
        )
        candidates.append(shape / shape.sum())

        # Sin‑power families
        for p in (0.5, 1.0, 2.0, 3.0):
            shape = np.sin(np.pi * (idx + 0.5) / n) ** p
            candidates.append(shape / shape.sum())

        # Smoothed tent (linear rise then fall, then convolve with a tiny kernel)
        left = np.linspace(0, 1, n // 2, endpoint=False)
        right = left[::-1]
        tent = np.concatenate([left, right])
        # slight smoothing via moving average of width 3
        kernel = np.ones(3) / 3.0
        tent = np.convolve(tent, kernel, mode='same')
        tent = np.maximum(tent, 0)
        candidates.append(tent / tent.sum())

        # ------------------------------------------------------------------
        # Optimise each candidate
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # Normalise to a probability vector (sum == 1)
            a = a0 / a0.sum()
            logits = np.log(a + 1e-15)       # soft‑max parameters (unconstrained)

            # -----------------------------------------------------------------
            # Phase 1 – Adam on smooth‑max surrogate
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.06
            lam = 0.3                     # temperature for smooth‑max
            lam_factor = 1.04             # rapid sharpening
            last_improve = time.time()
            # iteration budget scaled with problem size
            max_iters = int(50000 * 1024 / n) + 8000

            for t in range(1, max_iters + 1):
                # ----- soft‑max mapping -----
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                # ----- convolution and smooth‑max weights -----
                conv = _full_convolution(a, conv_len_fft)
                max_c = conv.max()
                # w_i ∝ exp(lam·(conv_i - max_c))  (numerically stable)
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # ----- gradient of the surrogate -----
                grad_a = 2.0 * _cross_correlation(w, a, corr_len_fft)

                # back‑prop through the soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # ----- Adam update -----
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # ----- temperature schedule -----
                lam = min(lam * lam_factor, 1e6)

                # ----- evaluate exact objective (sum(a)==1) -----
                val = 2.0 * n * max_c

                # ----- store global best -----
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Adam iter={t:05d}] new best {val:.6f}")

                # ----- occasional random shake to escape plateaus -----
                if time.time() - last_improve > 8.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # -----------------------------------------------------------------
            # Phase 2 – Exponentiated gradient on exact max (sub‑gradient)
            a = a / a.sum()
            eta = 0.55                     # step size
            sub_last_improve = time.time()
            max_sub_iters = 5000

            for sub_it in range(max_sub_iters):
                conv = _full_convolution(a, conv_len_fft)
                max_c = conv.max()
                # active set where conv == max_c (within tiny tolerance)
                active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                grad_a = 2.0 * _cross_correlation(w, a, corr_len_fft)

                # exponentiated‑gradient update (mirror descent w.r.t. KL)
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:               # prevent collapse
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                conv = _full_convolution(a, conv_len_fft)
                max_c = conv.max()
                val = 2.0 * n * max_c

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad iter={sub_it}] new best {val:.6f}")

                if time.time() - sub_last_improve > 8.0:
                    break
                if time.time() > deadline:
                    break

        # ------------------------------------------------------------------
        # Fine‑tuning of the *global* best for this length (if any)
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] fine‑tuning the current best (val={best_val:.6f})")
            a = best_seq.copy()
            logits = np.log(a + 1e-15)
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.04
            lam = 0.5
            lam_factor = 1.01
            last_improve = time.time()
            fine_iters = int(40000 * 1024 / n) + 5000

            for t in range(1, fine_iters + 1):
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                conv = _full_convolution(a, conv_len_fft)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

                grad_a = 2.0 * _cross_correlation(w, a, corr_len_fft)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                lam = min(lam * lam_factor, 1e6)

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} fine‑tune iter={t:05d}] new best {val:.6f}")

                if time.time() - last_improve > 10.0:
                    # small random shake before convergence
                    logits += (np.random.rand(n) - 0.5) * 5e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

    # ------------------------------------------------------------------
    # Global random‑shake / simulated‑annealing refinement (if time permits)
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\n=== Global simulated‑annealing refinement ===")
        a = best_seq.copy()
        n = a.shape[0]
        conv_len_fft = fft_len_conv[n]
        corr_len_fft = fft_len_corr[n]

        cur_val = _objective(a, n)
        best_local_val = cur_val
        best_local_seq = a.copy()

        # annealing schedule: exponential decay of temperature
        T_start = 0.02
        T_end = 1e-6
        total_steps = 3000
        start_time = time.time()

        for step in range(total_steps):
            if time.time() > deadline:
                break
            t = step / total_steps
            T = T_start * (T_end / T_start) ** t   # geometric cooling

            # propose a mass transfer between two distinct indices
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            if a[i] <= 0.0:
                continue
            # transfer a random fraction (up to 30%) of a[i] to a[j]
            delta = np.random.rand() * 0.3 * a[i]
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            # evaluate new candidate
            conv = _full_convolution(a_new, conv_len_fft)
            val_new = 2.0 * n * conv.max()

            # Metropolis acceptance
            if val_new < cur_val or np.random.rand() < np.exp(-(val_new - cur_val) / max(T, 1e-12)):
                a = a_new
                cur_val = val_new
                if val_new < best_local_val:
                    best_local_val = val_new
                    best_local_seq = a_new.copy()
                    print(f"[SA step={step:04d}] new best {val_new:.6f}")

        if best_local_val < best_val:
            best_val = best_local_val
            best_seq = best_local_seq.copy()
            print(f"=== Post‑SA improvement: {best_val:.6f} ===")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (should not happen)
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Return as plain list (the caller will feed it to evaluate_sequence)
    return best_seq.tolist()
