def max_convolution(a):
    n = a.shape[0]
    fft_len = 1 << int(np.ceil(np.log2(2*n-1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa*fa, fft_len)[:2*n-1]
    return conv.max()

"""
Step‑function optimiser – enhanced exploration.

The optimiser builds a rich set of analytically‑parameterised candidate
distributions (exponential tails, power‑law edges, symmetric Beta,
two‑Gaussian edges, edge‑blocks, Dirichlet draws, and warm‑starts from
previous bests).  All candidates are normalised to a probability simplex,
their objective
    F(a) = 2·n·max(a∗a) / (sum(a)²)
is evaluated cheaply with an FFT based convolution.

The best few candidates are refined with a two‑phase gradient method:
    * Adam on a smooth‑max surrogate (log‑sum‑exp) to obtain a good region
      in the high‑dimensional space.
    * A few hundred exponentiated‑gradient sub‑gradient steps on the true
      max‑convolution to finish the optimisation.

Time is split over a cascade of lengths
[1024, 2048, 4096, 8192, 16384, 32768]; the former stages act as warm‑starts
for the larger problems.  The routine respects the wall‑clock budget
(budget_s seconds) and always returns the best sequence discovered.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Small helper returning the next power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a∗a using an FFT of given length."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.

    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


def _max_convolution(a: np.ndarray) -> float:
    """Return max entry of a∗a (linear convolution)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, fft_len)
    return conv.max()


def _objective(a: np.ndarray) -> float:
    """Exact objective for a – assumes a is already non‑negative."""
    n = a.shape[0]
    max_c = _max_convolution(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Analytic families of candidates
def _gen_exponential(n: int, c: float) -> np.ndarray:
    """Symmetric exponential tail: exp(-c·i) + exp(-c·(n‑1‑i))."""
    idx = np.arange(n, dtype=np.float64)
    shape = np.exp(-c * idx) + np.exp(-c * (n - 1 - idx))
    shape /= shape.sum()
    return shape


def _gen_powerlaw(n: int, alpha: float) -> np.ndarray:
    """Edge‑concentrated power‑law: (i·(n‑i‑1))**alpha, α<0."""
    i = np.arange(1, n + 1, dtype=np.float64)
    prod = i * (n - i + 1.0)
    shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
    if shape.sum() == 0:
        shape = np.random.dirichlet(np.ones(n))
    else:
        shape /= shape.sum()
    return shape


def _gen_beta(n: int, a: float) -> np.ndarray:
    """Symmetric Beta( a , a ) density on the discrete grid."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    shape = np.power(x, a - 1.0) * np.power(1.0 - x, a - 1.0)
    shape = np.where(shape > 0, shape, 0.0)
    if shape.sum() == 0:
        shape = np.random.dirichlet(np.ones(n))
    else:
        shape /= shape.sum()
    return shape


def _gen_two_gaussian(n: int, sigma: float) -> np.ndarray:
    """Two symmetric Gaussian bumps placed at the ends."""
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-0.5 * (i / sigma) ** 2) + np.exp(-0.5 * ((n - 1 - i) / sigma) ** 2)
    shape /= shape.sum()
    return shape


def _gen_edge_block(n: int, k: int) -> np.ndarray:
    """Uniform block of size k at each edge, zero elsewhere."""
    if k <= 0 or k * 2 > n:
        k = max(1, n // 10)
    val = 1.0 / (2 * k)
    a = np.zeros(n, dtype=np.float64)
    a[:k] = val
    a[-k:] = val
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimizes
    F(a)=2·n·max(a∗a)/(sum(a)²).

    The routine respects ``budget_s`` seconds of wall‑clock time and
    returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a tiny safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Length cascade – small problems give cheap warm‑starts for larger ones.
    lengths = [1024, 2048, 4096, 8192, 16384, 32768]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        # --------------------------------------------------------------
        # Build a diverse pool of initial candidates (raw, not yet normalised)
        raw_cands = []

        # uniform
        raw_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

        # warm‑start from an exact‑size previous solution
        if best_seq is not None and len(best_seq) == n:
            raw_cands.append(np.asarray(best_seq, dtype=np.float64))

        # up‑sample a shorter best (linear interpolation)
        if best_seq is not None and len(best_seq) < n:
            old = np.asarray(best_seq, dtype=np.float64)
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, old)
            raw_cands.append(up)

        # historic global variable, if present
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            h = np.asarray(hseq, dtype=np.float64)
            if h.shape[0] > n:
                h = h[:n]
            elif h.shape[0] < n:
                h = np.pad(h, (0, n - h.shape[0]))
            raw_cands.append(h)

        # random Dirichlet draw
        raw_cands.append(np.random.dirichlet(np.ones(n)))

        # ----- analytic families -------------------------------------------------
        # symmetric exponential tails
        for c in np.logspace(-2, 0.5, 6):          # c ≈ 0.01 … 3
            raw_cands.append(_gen_exponential(n, c))

        # power‑law edge families (α negative)
        for alpha in np.linspace(-2.0, -0.5, 5):
            raw_cands.append(_gen_powerlaw(n, alpha))

        # symmetric Beta densities
        for a_param in np.linspace(0.3, 4.0, 5):
            raw_cands.append(_gen_beta(n, a_param))

        # two‑Gaussian edges
        for sigma_frac in np.linspace(0.005, 0.15, 5):
            sigma = sigma_frac * n
            raw_cands.append(_gen_two_gaussian(n, sigma))

        # edge blocks – few large entries at the two ends
        for k_frac in np.linspace(0.001, 0.03, 5):
            k = max(1, int(round(k_frac * n)))
            raw_cands.append(_gen_edge_block(n, k))

        # --------------------------------------------------------------
        # Normalise candidates, evaluate, and keep the best few.
        evaluated = []   # list of (value, candidate) tuples
        for idx, cand in enumerate(raw_cands):
            if time.time() > deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s < 1e-12:
                continue
            cand = cand / s
            val = _objective(cand)
            evaluated.append((val, cand))
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                print(f"[n={n} init cand={idx}] new best {val:.6f}")

        # --------------------------------------------------------------
        # Refine the top candidates with a two‑phase gradient optimiser.
        evaluated.sort(key=lambda t: t[0])
        top_k = min(3, len(evaluated))
        for top_idx in range(top_k):
            if time.time() > deadline:
                break
            init_val, init_cand = evaluated[top_idx]
            # ----- Adam on smooth‑max surrogate ---------------------------------
            eps = 1e-16
            a = np.maximum(init_cand, eps)
            a = a / a.sum()
            logits = np.log(a + eps)

            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2 = 0.9, 0.999
            eps_adam = 1e-8
            lr = 0.05

            lam = 0.5
            lam_factor = 1.01
            last_improve = time.time()
            # iteration budget – longer for smaller n
            max_iters = int(25000 * 1024 / n) + 3000

            best_local_a = a.copy()
            best_local_val = init_val

            for it in range(1, max_iters + 1):
                if time.time() > deadline:
                    break

                # soft‑max → simplex
                mx = logits.max()
                exp_log = np.exp(logits - mx)
                a = exp_log / exp_log.sum()

                # convolution and smooth‑max weights
                conv = _full_convolution(a, conv_fft_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # gradient of smooth‑max w.r.t. a
                grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

                # back‑prop through soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                # anneal temperature
                lam = min(lam * lam_factor, 1e7)

                # exact objective (denominator = 1 after normalisation)
                val = 2.0 * n * max_c
                if val < best_local_val:
                    best_local_val = val
                    best_local_a = a.copy()
                    last_improve = time.time()
                    if val < best_val:
                        best_val = val
                        best_seq = a.copy()
                        print(f"[n={n} cand={top_idx} Adam it={it}] new best {val:.6f}")

                # stagnation handling – tiny random shake
                if time.time() - last_improve > 5.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-3
                    last_improve = time.time()

            # ----- Sub‑gradient refinement on the true max -----------------------
            a = best_local_a.copy()
            eta = 0.6
            sub_last = time.time()
            for sub_it in range(2000):
                if time.time() > deadline:
                    break
                conv = _full_convolution(a, conv_fft_len)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
                a = a * np.exp(-eta * grad_a)
                s = a.sum()

                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()

                val = 2.0 * n * max_c
                if val < best_val:

                    best_seq = a.copy()
                    sub_last = time.time()

                if time.time() - sub_last > 5.0:

    # Fallback – uniform vector if nothing was discovered (unlikely)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
