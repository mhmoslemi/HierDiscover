"""
Propose a non‑negative step‑function (list of heights) that drives the
evaluation score

        F(a) = 2·n·max(a*a) / (sum(a)²)

as low as possible.

The algorithm builds a set of *symmetric* candidate shapes (uniform,
Beta‑type, sin‑power, Gaussian‑like, Dirichlet‑based) for each
problem size, then refines them with two phases:

1. **Adam on a smooth‑max surrogate** – the surrogate is the
   soft‑max of the convolution entries, controlled by a temperature
   parameter that is increased slowly, turning the smooth‑max into the
   true max.

2. **Exponentiated‑gradient (sub‑gradient) phase** on the exact
   maximum, which quickly drives the solution to a local optimum.

All updates are projected onto the probability simplex and onto the
space of symmetric vectors (a_i = a_{n‑1‑i}), because the objective is
invariant under this symmetry.  The routine respects a global time
budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _fft_len_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x (fast for NumPy FFTs)."""
    return 1 << int(np.ceil(np.log2(x)))


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using a real FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the
    smooth‑max.  Returns a vector of length len(a) equal to the
    discrete convolution of w with the reversed a.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Project onto the subspace a_i = a_{n-1-i}."""
    return (v + v[::-1]) / 2.0


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    # Use the classic algorithm based on sorting.
    if v.min() >= 0 and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    # Ensure exact sum == 1 (tiny numerical drift)
    w /= w.sum()
    return w


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact evaluation of 2·n·max(conv) (sum of a is assumed = 1)."""
    conv = _full_convolution(a, fft_len)
    return 2.0 * n * conv.max()

    candidates.append(np.full(n, 1.0 / n))

    # 2. Symmetric Dirichlet (sample half, mirror)
    half = n // 2
    for _ in range(3):
        if n % 2 == 0:
            half_vec = np.random.dirichlet(np.ones(half))
            a = np.concatenate([half_vec, half_vec[::-1]])
        else:
            half_vec = np.random.dirichlet(np.ones(half))
            mid = np.random.rand()
            a = np.concatenate([half_vec, [mid], half_vec[::-1]])
        a /= a.sum()
        candidates.append(a)

    # 3. Power‑law product shape  ((i+1)*(n-i))**α , α>0 (central mass)
    idx = np.arange(n, dtype=np.float64) + 1.0          # 1 … n
    prod = idx * (n - idx + 1.0)                       # (i+1)*(n-i)
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # 4. Symmetric sin‑power families
    x = (np.arange(n, dtype=np.float64) + 0.5) / n          # 0..1
    sinus = np.sin(np.pi * x)                               # zero at edges
    for p in [1, 2, 3, 4]:
        shape = np.power(sinus, p)
        shape /= shape.sum()
        candidates.append(shape)

    # 5. Gaussian‑like (discrete) shapes centred in the middle
    centre = (n - 1) / 2.0
    for sigma in [n / 8.0, n / 6.0, n / 4.0]:
        shape = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
        shape /= shape.sum()
        candidates.append(shape)

    # 6. Two‑sided exponential (already used in the reference)
    for gamma in [0.001, 0.002, 0.004, 0.008]:
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # 7. Small perturbations of the previous global best (if any)
    if globals().get("height_sequence_1") is not None:
        best = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if best.shape[0] != n:
            # resize by simple truncation/padding (zero padding preserves symmetry)
            if best.shape[0] > n:
                best = best[:n]
            else:
                pad = np.zeros(n - best.shape[0])
                best = np.concatenate([best, pad])
        best = np.maximum(best, 0.0)
        best /= best.sum()
        # a few jittered copies
        for _ in range(2):
            jitter = best + 0.01 * (np.random.rand(n) - 0.5)
            jitter = np.maximum(jitter, 0.0)
            jitter /= jitter.sum()
            candidates.append(jitter)

    return candidates


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Run a time‑bounded search for a low‑score step function.
    Returns the best sequence (plain Python list) found before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_val = float("inf")

    # Lengths we will explore – larger lengths tend to give smaller scores.
    lengths = [8192, 16384]

    # Pre‑compute FFT lengths for each n (next power of two ≥ 2n‑1).
    fft_len = {n: _fft_len_pow2(2 * n - 1) for n in lengths}

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = fft_len[n]
        # cross‑correlation needs length at most 3n for safety
        corr_fft_len = _fft_len_pow2(3 * n)

        # -------------------------------------------------------------
        # Build initial candidates (all symmetric, normalised)
        candidates = _initial_candidates(n)

        # -------------------------------------------------------------
        # Optimise each candidate independently
        for cid, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # Ensure exact simplex and symmetry before starting
            a = _project_to_simplex(_symmetrize(a0))

            # Use a log‑softmax parametrisation (logits) to stay on the simplex
            logits = np.log(a + 1e-15)

            # ----- Adam on a smooth‑max surrogate -----
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.08               # slightly larger step size
            lam = 0.2               # temperature start
            lam_factor = 1.025

            last_improve = time.time()
            # iteration budget roughly 15 k for n≈8 k, scaled inversely with n
            max_iters = int(30000 * 8192 / n) + 3000

            for t in range(1, max_iters + 1):
                # soft‑max → probability simplex (automatically non‑negative)
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()
                a = _symmetrize(a)          # enforce symmetry after each step

                # convolution and smooth‑max weights
                conv = _full_convolution(a, conv_fft_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # gradient of the smooth‑max w.r.t. a
                grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

                # back‑prop through soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update (logits live in ℝ)
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # increase temperature → smoother‑max approaches true max
                lam = min(lam * lam_factor, 1e6)

                # exact objective (since a sums to 1)
                val = 2.0 * n * max_c

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cid} Adam iter={t:05d}] new best {val:.6f}")

                # restart if stagnant for a while
                if time.time() - last_improve > 7.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # ----- Exponentiated‑gradient (sub‑gradient) on the *exact* max -----
            a = _project_to_simplex(_symmetrize(a))   # ensure clean state
            eta = 0.55
            sub_last_improve = time.time()
            max_sub_iters = 6000

            for sub_it in range(max_sub_iters):
                conv = _full_convolution(a, conv_fft_len)
                max_c = conv.max()
                # indices where the maximum is attained (within tolerance)
                active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)

                w = np.zeros_like(conv)
                if active.any():
                    w[active] = 1.0 / active.sum()   # uniform over active set

                grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

                # exponentiated‑gradient step
                a = a * np.exp(-eta * grad_a)
                a = _project_to_simplex(_symmetrize(a))

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last_improve = time.time()
                    print(f"[n={n} cand={cid} Subgrad iter={sub_it}] new best {val:.6f}")

                # early‑stop this sub‑gradient run if no progress
                if time.time() - sub_last_improve > 6.0:
                    break
                if time.time() > deadline:
                    break

        # -------------------------------------------------------------
        # Fine‑tune the *global* best of this length (if we have one)
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] fine‑tuning global best (val={best_val:.6f})")
            a = best_seq.copy()
            logits = np.log(a + 1e-15)
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.045
            lam = 0.5
            lam_factor = 1.015
            last_improve = time.time()
            fine_iters = int(50000 * 8192 / n) + 5000

            for t in range(1, fine_iters + 1):
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()
                a = _symmetrize(a)

                conv = _full_convolution(a, conv_fft_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

                grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                lam = min(lam * lam_factor, 1e6)

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} fine‑tune iter={t:05d}] new best {val:.6f}")

                if time.time() - last_improve > 9.0:
                    # small random shake to escape possible plateaus
                    logits += (np.random.rand(n) - 0.5) * 5e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

    # ----------------------------------------------------------------------
    # Fallback – return a uniform vector if nothing was found (should never happen)
    if best_seq is None:
        fallback_n = 4096
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # expose the best sequence for possible warm‑starts in a future run
    globals()["height_sequence_1"] = best_seq.copy()

    # Convert to plain Python list as required by the external evaluator
    return best_seq.tolist()
