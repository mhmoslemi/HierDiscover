def project_simplex(v, S):
    # v may be any real numbers
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (array of heights) that minimizes

    2 * n * max(conv(a,a)) / (sum(a)**2)

where ``conv`` is the full linear convolution.  The objective is equivalent to
minimising the peak of the a‑periodic autocorrelation under a fixed ℓ¹ norm.

The algorithm mixes several ideas:

* **Diverse random initial shapes** – power‑law, exponential, Gaussian,
  triangular and uniform profiles provide many basins of attraction.
* **Adam optimisation on a soft‑max parametrisation** – the vector ``x``
  stores log‑probabilities; ``a = S·softmax(x)`` guarantees non‑negativity
  and the exact required sum ``S = sqrt(2 n)``.
* **Smooth‑max surrogate** – a temperature‑scaled log‑sum‑exp on the
  convolution values supplies a differentiable approximation of the true
  ``max``.  The temperature is decayed slowly so that the optimiser first
  follows a smooth landscape and later refines the true peak.
* **Multiple short restarts** – each run gets a modest time slice; the best
  candidates are kept and occasionally used as warm‑starts for new runs.
* **Final local refinement** – a longer Adam run on the current global best,
  followed by a cheap hill‑climbing phase that swaps tiny amounts of mass
  between random positions.

The procedure respects the 1000 s budget, prints progress information and
returns the best sequence (as a plain Python list) found so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – full linear convolution using a power‑of‑2 FFT
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of two real vectors using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)             # eliminate tiny negatives
    return conv


# ----------------------------------------------------------------------
# Random shape generator (diverse families)
# ----------------------------------------------------------------------
def _initial_shape_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a positive vector of length n using a random family."""
    typ = rng.choice(
        ["power", "exponential", "gaussian", "triangular", "mixed", "uniform"]
    )
    idx = np.arange(1, n + 1, dtype=float)

    if typ == "power":
        p = rng.uniform(0.1, 2.5)
        shape = idx ** (-p)
    elif typ == "exponential":
        tau = rng.uniform(0.2, 0.9) * n
        shape = np.exp(-idx / tau)
    elif typ == "gaussian":
        mu = rng.uniform(0, n - 1)
        sigma = rng.uniform(0.1, 0.5) * n + 1e-9
        shape = np.exp(-0.5 * ((idx - 1 - mu) / sigma) ** 2)
    elif typ == "triangular":
        peak = rng.integers(1, n - 1)
        left = np.linspace(0.0, 1.0, peak + 1)[1:]          # exclude the zero at start
        right = np.linspace(1.0, 0.0, n - peak)[1:]        # exclude the duplicated peak
        shape = np.concatenate([left, right])
    elif typ == "mixed":
        p = rng.uniform(0.2, 1.5)
        tau = rng.uniform(0.3, 0.8) * n
        shape = (idx ** (-p)) * np.exp(-idx / tau)
    else:  # uniform
        shape = np.ones_like(idx)

    # tiny jitter to break perfect symmetry
    shape += rng.random(n) * 1e-6
    shape = np.maximum(shape, 1e-12)
    return shape


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max parametrisation
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
    *,
    init_tau_factor: float = 0.35,
    min_tau_factor: float = 5e-5,
    tau_decay: float = 0.985,
    alpha: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    max_iter: int | None = None,
    grad_clip: float = 5.0,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best sequence (numpy array) and its official score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_val = float("inf")
    iteration = 0
    last_improve = 0

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1. Soft‑max mapping (ensures non‑negativity and correct sum)
        # --------------------------------------------------------------
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)            # stabilise
        p = exp_shift / exp_shift.sum()
        a = target_sum * p                         # current candidate

        # --------------------------------------------------------------
        # 2. Convolution and smooth‑max preparation
        # --------------------------------------------------------------
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()

        # Keep the best (according to the true objective, not the surrogate)
        if max_conv < best_val:
            best_val = max_conv
            best_seq = a.copy()
            last_improve = iteration
            print(f"[{time.time():.2f}] n={n}  new local max_conv = {max_conv:.9f}")

        # --------------------------------------------------------------
        # 3. Log‑sum‑exp weights (smooth max)
        # --------------------------------------------------------------
        tau = max(init_tau_factor * max_conv * (tau_decay ** iteration),
                  min_tau_factor * max_conv)
        scaled = (conv - max_conv) / tau            # ≤ 0
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # 4. Gradient w.r.t. a (conv is quadratic ⇒ linear gradient)
        # --------------------------------------------------------------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)   # length = 2n‑1
        idx = (2 * n - 2) - np.arange(n)           # slice picking the right positions
        grad_a = 2.0 * conv2[idx]

        # --------------------------------------------------------------
        # 5. Gradient w.r.t. logits x (chain rule through soft‑max)
        # --------------------------------------------------------------
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # optional gradient clipping
        gnorm = np.linalg.norm(grad_x)
        if gnorm > grad_clip:
            grad_x = grad_x * (grad_clip / gnorm)

        # --------------------------------------------------------------
        # 6. Adam update
        # --------------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if iteration % 400 == 0 and rng.random() < 0.12:
            x += rng.normal(scale=0.001, size=n)

        # decay learning rate very slowly
        alpha *= 0.9995

        # early stopping if no improvement for many iterations
        if iteration - last_improve > 3000:
            break

        if max_iter is not None and iteration >= max_iter:
            break

    # ------------------------------------------------------------------
    # Final legal evaluation of the best candidate we have seen.
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Tiny hill‑climbing local improvement (mass swaps)
# ----------------------------------------------------------------------
def _hill_climb(seq: np.ndarray, rng: np.random.Generator,
                time_deadline: float, max_moves: int = 2000) -> np.ndarray:
    """
    Simple stochastic local search: pick two indices i,j and move a small
    amount of mass from i to j, accepting the move if the true objective
    improves.  Operates directly on the sequence (which already satisfies
    the sum constraint).
    """
    n = seq.size
    best = seq.copy()
    best_score = evaluate_sequence(best.tolist())
    move_scale = best.sum() * 0.005  # initial move size ≈0.5 % of total mass

    moves = 0
    while time.time() < time_deadline and moves < max_moves:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = rng.normal(scale=move_scale)
        # ensure non‑negativity of source
        if best[i] - delta < 0:
            delta = best[i] * 0.5
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # evaluate
        val = evaluate_sequence(cand.tolist())
        if val < best_score:
            best = cand
            best_score = val
            # gradually shrink move scale when we find improvements
            move_scale *= 0.95
        else:
            # occasionally increase scale to explore larger steps
            if rng.random() < 0.02:
                move_scale *= 1.05
        moves += 1
    return best
