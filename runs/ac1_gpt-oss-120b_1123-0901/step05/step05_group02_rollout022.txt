"""
Propose a non‑negative step‑function (list of heights) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

where a*a is the linear convolution (numpy.convolve).

The search combines several ideas that complement the previous
gradient‑descent implementation:

* a richer pool of deterministic initial shapes (Dirichlet, power‑law,
  exponential, sine, cosine, Gaussian and truncated‑arcsine);
* a two‑stage optimisation – first an Adam optimiser on a smooth‑max
  surrogate, then an exact exponentiated‑gradient (sub‑gradient) phase;
* aggressive temperature growth for the smooth‑max and adaptive learning
  rates that scale with the problem size;
* a “global‑fine‑tune’’ pass that re‑optimises the current best sequence
  for the longest length that has been explored;
* a small amount of random shaking when progress stalls.

All heavy work (convolutions and cross‑correlations) is performed with
FFT‑based routines that run in O(n log n).  The routine respects a time
budget (default 1000 s) and returns the best sequence discovered as a
plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based linear convolution (a*a) and the cross‑correlation needed for
# the gradient of the smooth‑max surrogate.
def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, conv_len: int) -> float:
    """Exact evaluation of the objective (no clamping, sum is the raw sum)."""
    c = _conv_fft(a, conv_len)
    return 2.0 * n * c.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
def _build_initial_candidates(n: int) -> list[np.ndarray]:
    """
    Produce a diversified list of starting points for the optimisation.
    All candidates are normalised to sum to 1 (the scaling does not affect
    the objective, but a fixed sum makes the optimiser more stable).
    """
    cand = []

    # 1) warm‑start from a previously found solution (if any)
    global height_sequence_1
    if "height_sequence_1" in globals():
        h = np.asarray(height_sequence_1, dtype=np.float64)
        if h.shape[0] > n:
            h = h[:n]
        else:
            h = np.pad(h, (0, n - h.shape[0]))
        h = np.maximum(h, 0.0)
        if h.sum() < 1e-8:
            h = np.random.dirichlet(np.ones(n))
        else:
            h = h / h.sum()
        cand.append(h)

    # 2) pure Dirichlet (concentration = 1)
    cand.append(np.random.dirichlet(np.ones(n)))

    # 3) spiky Dirichlet (concentration < 1)
    for conc in (0.2, 0.4, 0.6, 0.8):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 4) over‑dispersed Dirichlet (concentration > 1) – almost uniform
    for conc in (1.5, 2.0, 3.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 5) edge‑concentrated power‑law shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 9):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape = shape / shape.sum()
            cand.append(shape)

    # 6) two‑sided exponential decay
    for gamma in (0.0008, 0.0015, 0.003, 0.006):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape = shape / shape.sum()
        cand.append(shape)

    # 7) smooth tent (linear rise then fall)
    left = np.linspace(0, 1, n // 2, endpoint=False)
    right = left[::-1]
    tent = np.concatenate([left, right])
    if tent.shape[0] < n:   # odd length – add a centre point
        tent = np.concatenate([tent, [0.0]])
    tent = np.maximum(tent, 0)
    tent = tent / tent.sum()
    cand.append(tent)

    # 8) cosine window (a single bump)
    i = np.arange(n, dtype=np.float64)
    coswin = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    coswin = coswin / coswin.sum()
    cand.append(coswin)

    # 9) half‑sine window (zero at both ends, peak in the centre)
    sinwin = np.sin(np.pi * i / (n - 1))
    sinwin = sinwin / sinwin.sum()
    cand.append(sinwin)

    # 10) discrete Gaussian (σ scaled with n)
    sigma = n / 8.0
    gauss = np.exp(-0.5 * ((i - (n - 1) / 2) / sigma) ** 2)
    gauss = gauss / gauss.sum()
    cand.append(gauss)

    # 11) truncated arcsine (inverse square‑root edges)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    arcs = arcs / arcs.sum()
    cand.append(arcs)

    # 12) random sparse combinatorial pattern (≈ sqrt(n) spikes)
    m_spikes = int(np.sqrt(2 * n))
    sparse = np.zeros(n, dtype=np.float64)
    indices = np.random.choice(n, size=m_spikes, replace=False)
    sparse[indices] = 1.0
    if sparse.sum() > 0:
        sparse = sparse / sparse.sum()
        cand.append(sparse)

    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5     # safety margin

    best_seq = None
    best_val = float("inf")

    # Lengths to explore – larger lengths give a larger “search space’’
    # but also a higher potential for a lower bound.
    lengths = [1024, 2048, 4096, 8192]

    # Pre‑compute the FFT lengths (next power of two) for each n.
    fft_len_conv = {}
    fft_len_corr = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_len_conv[n] = 1 << int(np.ceil(np.log2(conv_len)))
        fft_len_corr[n] = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_len = fft_len_conv[n]
        corr_len = fft_len_corr[n]

        # ----- build a diversified list of initial candidates ---------------
        candidates = _build_initial_candidates(n)

        # ----- optimise each candidate --------------------------------------
        for cid, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # Normalise to a probability vector (sum = 1) and work in the
            # soft‑max (logits) domain which automatically respects the simplex.
            a = a0 / a0.sum()
            logits = np.log(a + 1e-15)

            # ---- Adam on a smooth‑max surrogate of the true max -------------
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8

            # Learning‑rate adapts mildly with the problem size.
            lr = 0.08 * np.sqrt(2048.0 / n)

            # Temperature (smooth‑max) schedule.
            lam = 0.2
            lam_factor = 1.018    # fairly aggressive

            last_improve = time.time()
            # iteration budget grows for smaller n – we give a bit more slack
            max_iters = int(50000 * 1024 / n) + 5000

            for t in range(1, max_iters + 1):
                # soft‑max => simplex
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                # convolution and smooth‑max weights
                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # gradient of the surrogate (linear in w)
                grad_a = 2.0 * _corr_fft(w, a, corr_len)

                # back‑prop through the soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # temperature update (approach the true max)
                lam = min(lam * lam_factor, 1e6)

                # exact objective (sum(a) == 1)
                val = 2.0 * n * max_c

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cid} Adam iter={t:05d}] new best {val:.6f}")

                # Stagnation handling – a gentle random shake.
                if time.time() - last_improve > 7.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # ---- Exact max: exponentiated‑gradient (sub‑gradient) phase -----
            a = a / a.sum()
            eta = 0.55                     # step‑size for EG
            sub_last_improve = time.time()
            max_sub_iters = 6000

            for sub_it in range(max_sub_iters):
                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                # indices where the max is attained (allow a tiny tolerance)
                active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)

                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                grad_a = 2.0 * _corr_fft(w, a, corr_len)

                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:                     # protect against collapse
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                val = 2.0 * n * max_c

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last_improve = time.time()
                    print(f"[n={n} cand={cid} Subgrad iter={sub_it}] new best {val:.6f}")

                # early abort if no improvement for a while
                if time.time() - sub_last_improve > 8.0:
                    break
                if time.time() > deadline:
                    break

        # ---------- fine‑tune the current global best for this length -----
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] fine‑tuning the current best (val={best_val:.6f})")
            a = best_seq.copy()
            logits = np.log(a + 1e-15)

            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.045 * np.sqrt(2048.0 / n)
            lam = 0.4
            lam_factor = 1.012
            last_improve = time.time()
            fine_iters = int(60000 * 1024 / n) + 5000

            for t in range(1, fine_iters + 1):
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                conv = _conv_fft(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

                grad_a = 2.0 * _corr_fft(w, a, corr_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                lam = min(lam * lam_factor, 1e6)

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} fine‑tune iter={t:05d}] new best {val:.6f}")

                if time.time() - last_improve > 10.0:
                    # small random perturbation to escape shallow plateaus
                    logits += (np.random.rand(n) - 0.5) * 5e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

    # ------------------------------------------------------------------
    # Fallback – a simple uniform vector (should never be needed).
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list – the external evaluator expects a list.
    return best_seq.tolist()
