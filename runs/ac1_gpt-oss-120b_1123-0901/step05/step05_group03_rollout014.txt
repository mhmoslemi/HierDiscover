def _optimize_blocks(m, seg_len, n, conv_fft_len, corr_fft_len, deadline, init_w, global_best_val):
    # returns best_w, best_val
    eps = 1e-15
    if init_w is not None:
        logits = np.log(init_w + eps)
    else:
        logits = np.log(np.random.dirichlet(np.ones(m)))
    # Adam states
    m_adam = np.zeros_like(logits)
    v_adam = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.02
    best_w = None
    best_val_local = float('inf')
    last_improve = time.time()
    iter = 0
    # Loop while time
    while time.time() < deadline and iter < 300000:
        # compute block probabilities
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        w_blocks = exp_log / exp_log.sum()
        # expand to full a
        a = np.repeat(w_blocks / seg_len, seg_len)  # length n
        # compute conv
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c  # sum a = 1
        if val < best_val_local:
            best_val_local = val
            best_w = w_blocks.copy()
            last_improve = time.time()
            # could also update global best externally, but not needed
            print(f"[blocks={m} iter={iter:06d}] new block best {val:.6f}")
        # compute smooth max weights
        w_smooth, _ = _smooth_weights(conv, lam)
        # gradient w.r.t a
        grad_a = 2.0 * _cross_correlation(w_smooth, a, corr_fft_len)
        # compute gradient w.r.t block masses (summing over each segment)
        grad_w = np.add.reduceat(grad_a, np.arange(0, n, seg_len)) / seg_len
        # gradient w.r.t logits using mirror map (softmax)
        avg = np.dot(w_blocks, grad_w)
        grad_logits = w_blocks * (grad_w - avg)
        # Adam update
        m_adam = beta1 * m_adam + (1 - beta1) * grad_logits
        v_adam = beta2 * v_adam + (1 - beta2) * (grad_logits ** 2)
        m_hat = m_adam / (1 - beta1 ** (iter + 1))
        v_hat = v_adam / (1 - beta2 ** (iter + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # update lam
        lam = min(lam * lam_factor, 1e6)
        # stagnation shake
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(m) - 0.5) * 1e-2
            last_improve = time.time()
        iter += 1
        if iter % 50000 == 0:
            # check time
            if time.time() >= deadline:
                break
    # Return best block distribution
    if best_w is None:
        best_w = np.exp(logits - np.max(logits))
        best_w = best_w / best_w.sum()
    return best_w, best_val_local

def _subgradient_refine(a, n, conv_fft_len, corr_fft_len, deadline, current_best_val):
    # a sum may not be 1; ensure normalization.
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s
    best_a = a.copy()
    best_val = current_best_val
    # compute initial objective at start maybe
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    val = 2.0 * n * max_c
    if val < best_val:
        best_val = val
        best_a = a.copy()
    eta = 0.6
    last_improve = time.time()
    for it in range(12000):
        if time.time() >= deadline:
            break
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[subgrad refine iter={it:05d}] new best {val:.6f}")
        # active set approx:
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        if active.any():
            w_active = np.zeros_like(conv)
            w_active[active] = 1.0 / active.sum()
        else:
            w_active = np.zeros_like(conv)
        grad_a = 2.0 * _cross_correlation(w_active, a, corr_fft_len)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        if time.time() - last_improve > 5.0:
            # stagnated: random shake
            a = a + (np.random.rand(n) - 0.5) * 1e-3
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            last_improve = time.time()
    return best_a, best_val

"""
Propose a low‑value step‑function.

The algorithm works in three stages:

1. **Hierarchical block optimisation** –  The vector of length `n=4096`
   is represented by a small number `m` of constant blocks
   (m = 64 → 256 → 1024).  On the block level we optimise a smooth
   surrogate of the objective  
         F(a)=2·n·max(a*a)   (the sum is normalised to 1)  
   using Adam on the log‑soft‑max parametrisation of the block masses.
   Convolutions and cross‑correlations are computed with FFTs, so each
   iteration is O(n log n) even for the full‑resolution vector.

2. **Exact sub‑gradient refinement** –  The best block‑wise solution of a
   stage is expanded to the full vector and refined with an
   exponentiated‑gradient update on the true max‑convolution (no smoothing)
   until progress stalls.

3. **Optional up‑sampling** –  If time permits the final distribution is
   linearly interpolated to length 8192 and a short sub‑gradient pass is
   performed there as well.

The routine respects the supplied time budget (default 1000 s) and
maintains a global best sequence.  It starts from a previous solution
contained in the global variable `height_sequence_1` when available,
otherwise from a random Dirichlet start.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _smooth_weights(conv: np.ndarray, lam: float):
    """Log‑sum‑exp smooth‑max weights."""
    cmax = conv.max()
    w = np.exp(lam * (conv - cmax))            # max exponent = 0
    s = w.sum()
    if s == 0 or np.isnan(s):
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w = w / s
    return w, cmax


# ----------------------------------------------------------------------
def _optimize_blocks(
    m: int,
    seg_len: int,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_blocks: np.ndarray | None,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smoothed‑max surrogate for a block‑wise representation.
    Returns the best block vector (sums to 1) and its objective value.
    """
    eps = 1e-15
    if init_blocks is not None:
        logits = np.log(init_blocks + eps)
    else:
        logits = np.log(np.random.dirichlet(np.ones(m)))

    m_adam = np.zeros_like(logits)
    v_adam = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.02

    best_w = None
    best_val = float("inf")
    last_improve = time.time()
    it = 0

    # main loop ---------------------------------------------------------
    while time.time() < deadline and it < 300_000:
        # ---- probabilities on the simplex (soft‑max) ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        w_blocks = exp_log / exp_log.sum()                     # ∑ w = 1

        # ---- expand to full vector (constant per block) ----
        a = np.repeat(w_blocks / seg_len, seg_len)              # length n, ∑ a = 1

        # ---- convolution & exact objective ----------------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                                   # sum(a)=1 ⇒ denominator =1
        if val < best_val:
            best_val = val
            best_w = w_blocks.copy()
            last_improve = time.time()
            print(f"[blocks={m:04d} it={it:06d}] new best {val:.6f}")

        # ---- smooth‑max weights ----------------------------
        w_smooth, _ = _smooth_weights(conv, lam)

        # ---- gradient of the surrogate --------------------
        grad_a = 2.0 * _cross_correlation(w_smooth, a, corr_fft_len)

        # ---- map gradient back to block variables ----------
        # sum of gradients inside each block, divided by block length
        grad_w = np.add.reduceat(grad_a, np.arange(0, n, seg_len)) / seg_len

        # ---- mirror‑descent (soft‑max) gradient -----------
        avg = np.dot(w_blocks, grad_w)
        grad_log = w_blocks * (grad_w - avg)

        # ---- Adam update ---------------------------------
        m_adam = beta1 * m_adam + (1.0 - beta1) * grad_log
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m_adam / (1.0 - beta1 ** (it + 1))
        v_hat = v_adam / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule -------------------------
        lam = min(lam * lam_factor, 1e6)

        # ---- occasional random shake when stuck ----------
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(m) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    # fallback – use the final logits if nothing was better
    if best_w is None:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        best_w = exp_log / exp_log.sum()
        best_val = 2.0 * n * _full_convolution(
            np.repeat(best_w / seg_len, seg_len), conv_fft_len
        ).max()

    return best_w, best_val


# ----------------------------------------------------------------------
def _subgradient_refine(
    a: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    current_best: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the exact max‑convolution.
    Returns an improved vector (still sum‑normalised) and its value.
    """
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = current_best
    eta = 0.6
    last_improve = time.time()

    for it in range(12_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[subgrad it={it:05d}] new best {val:.6f}")

        # ---- active set (indices attaining the max) ----
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # ---- gradient of the exact max -----------------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- exponentiated‑gradient step ----------------
        a = a * np.exp(-eta * grad_a)
        total = a.sum()
        if total < 1e-12:                     # degenerated – re‑randomise
            a = np.random.dirichlet(np.ones(n))
            total = a.sum()
        a = a / total

        # ---- stagnation shake -------------------------
        if time.time() - last_improve > 5.0:
            a += (np.random.rand(n) - 0.5) * 1e-3
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Returns the best sequence found within
    the allotted time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Global best tracking
    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Fixed problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (previous solution) if available
    warm_global = globals().get("height_sequence_1")
    warm_arr = None
    if warm_global is not None:
        warm_arr = np.asarray(warm_global, dtype=np.float64)
        if warm_arr.shape[0] > n:
            warm_arr = warm_arr[:n]
        elif warm_arr.shape[0] < n:
            warm_arr = np.pad(warm_arr, (0, n - warm_arr.shape[0]))
        warm_arr = np.maximum(warm_arr, 0.0)
        if warm_arr.sum() >= 0.01:
            warm_arr = warm_arr / warm_arr.sum()

    # ------------------------------------------------------------------
    # Hierarchical block optimisation (coarse → fine)
    block_counts = [64, 256, 1024]          # 64·64 = 4096, 256·16 = 4096, …
    prev_full = None                       # full‑resolution vector from previous stage

    for m in block_counts:
        if time.time() >= deadline:
            break
        seg_len = n // m
        assert n % m == 0

        # initialise block masses:
        init_blocks = None
        if prev_full is not None:
            # aggregate the refined full vector from previous stage
            init_blocks = np.add.reduceat(prev_full, np.arange(0, n, seg_len))
            init_blocks = init_blocks / init_blocks.sum()
        elif warm_arr is not None:
            init_blocks = np.add.reduceat(warm_arr, np.arange(0, n, seg_len))
            init_blocks = init_blocks / init_blocks.sum()

        # ----- block‑level Adam optimisation -----
        best_blocks, _ = _optimize_blocks(
            m,
            seg_len,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            init_blocks,
        )

        # ----- expand to full vector & exact refinement -----
        full_a = np.repeat(best_blocks / seg_len, seg_len)
        full_a, best_val_stage = _subgradient_refine(
            full_a, n, conv_fft_len, corr_fft_len, deadline, best_val
        )

        # ----- update global best -----
        if best_val_stage < best_val:
            best_val = best_val_stage
            best_seq = full_a.copy()
            print(f"*** Global best improved to {best_val:.6f} at block size {m}")

        # use this refined vector as starting point for the next stage
        prev_full = full_a

    # ------------------------------------------------------------------
    # Optional up‑sampling to a larger length if time permits
    if time.time() + 12 < deadline and best_seq is not None:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best to the larger grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        a_big, best_val_big = _subgradient_refine(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_val,
        )
        if best_val_big < best_val:
            best_val = best_val_big
            best_seq = a_big.copy()
            print(f"*** Upsampled improvement: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator's allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # expose the result for the next run (optional)
    globals()["height_sequence_1"] = best_seq.tolist()

    return best_seq.tolist()
