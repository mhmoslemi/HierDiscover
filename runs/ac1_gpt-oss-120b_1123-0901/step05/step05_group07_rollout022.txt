Hence a simple deterministic construction with a moderate

import time
import numpy as np

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list:
    """
    Build polynomial‑ramp sequences a_i = 1000·(i/(n‑1))^p,
    evaluate them with ``evaluate_sequence`` and return the best one.
    """
    # No randomness is needed, but we honour the signature.
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # safety margin

    best_seq = None

    exponents = [1, 2, 3, 4, 5, 6, 8, 10, 12]
