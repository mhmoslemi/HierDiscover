"""
Propose a step‑function (non‑negative list) that minimizes

        F(a) = 2·n·max(a*a) / (sum(a)²)

using a multi‑stage optimisation:
  1.  Build a diverse set of initial candidates (uniform, Dirichlet,
      monotone power‑law / exponential, linear, two‑sided exponential,
      warm‑starts from the previous best, etc.).
  2.  For each candidate run a smooth‑max Adam optimiser (soft‑max
      parametrisation of the simplex) with a slowly rising temperature.
  3.  Refine the result with an exact sub‑gradient (exponentiated‑gradient)
      phase.
  4.  After all candidates of a given length are examined, fine‑tune the
      current global best with a longer Adam run.
  5.  Finally perform a cheap simulated‑annealing walk around the best
      solution to escape possible shallow local minima.

The routine respects a user‑supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list.  All helper
functions are defined at top level (no closures) and only NumPy and
the standard library are used.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using the given FFT length."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # keep only the first 2·len(a)‑1 terms
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the
    smooth‑max. Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, conv_len: int) -> float:
    """
    Exact evaluation of the objective 2·n·max(a*a) / (sum(a)²).
    The sum is always 1 in our optimisation, so the denominator can be omitted.
    """
    conv = _full_convolution(a, conv_len)
    max_c = conv.max()
    return 2.0 * n * max_c


def _candidate_pool(n: int, rng: np.random.Generator, best_seq: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a diverse list of starting vectors for a given length.
    All vectors are normalised to sum to one.
    """
    cand = []

    # warm‑start from a previous best of the same size
    if best_seq is not None and best_seq.shape[0] == n:
        cand.append(best_seq.copy())

    # previous global solution (height_sequence_1) – adapt size if needed
    h = globals().get("height_sequence_1")
    if isinstance(h, (list, tuple, np.ndarray)):
        h_arr = np.asarray(h, dtype=np.float64)
        if h_arr.size > n:
            h_arr = h_arr[:n]
        else:
            h_arr = np.pad(h_arr, (0, n - h_arr.size))
        h_arr = np.maximum(h_arr, 0.0)
        if h_arr.sum() < 1e-8:
            h_arr = rng.dirichlet(np.ones(n))
        else:
            h_arr = h_arr / h_arr.sum()
        cand.append(h_arr)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # Dirichlet families
    cand.append(rng.dirichlet(np.ones(n)))                 # concentration 1
    for conc in (0.2, 0.5, 0.8, 0.05):
        cand.append(rng.dirichlet(np.full(n, conc)))

    # linear decreasing (a “ramp”)
    ramp = np.arange(n, 0, -1, dtype=np.float64)
    cand.append(ramp / ramp.sum())

    # power‑law decreasing shapes a_i ∝ (i+1)^{-α}
    for alpha in (0.5, 1.0, 1.5, 2.0):
        pw = np.arange(1, n + 1, dtype=np.float64) ** (-alpha)
        cand.append(pw / pw.sum())

    # monotone exponential decays a_i ∝ exp(-γ·i)
    for gamma in (0.0003, 0.0007, 0.0015, 0.003):
        exp_decay = np.exp(-gamma * np.arange(n, dtype=np.float64))
        cand.append(exp_decay / exp_decay.sum())

    # two‑sided exponential (already used in the reference)
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        i = np.arange(n, dtype=np.float64)
        sym = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        cand.append(sym / sym.sum())

    # mixture of two random Dirichlets (adds multimodality)
    d1 = rng.dirichlet(np.ones(n))
    d2 = rng.dirichlet(np.full(n, 0.1))
    mix = 0.6 * d1 + 0.4 * d2
    cand.append(mix / mix.sum())

    # keep the pool reasonably small
    return cand[:20]


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a non‑negative sequence with a low evaluation score.

    The function respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # a small safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Lengths to explore – we start small (fast) and progress to a larger
    # length, carrying the best solution forward.
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        conv_len = _next_pow2(2 * n - 1)
        corr_len = _next_pow2(3 * n)          # safe upper bound for correlation

        # ------------------------------------------------------------------
        # Build a set of diverse initial guesses
        candidates = _candidate_pool(n, rng, best_seq)

        # ------------------------------------------------------------------
        # Optimise each candidate separately
        for cid, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # --- initialise simplex via soft‑max logits -----------------
            a = a0 / a0.sum()
            logits = np.log(a + 1e-15)

            # Adam hyper‑parameters
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.05
            lam = 0.3                # start temperature for soft‑max
            lam_factor = 1.015

            # iteration budget grows for larger n, but we keep a hard cap
            max_iters = int(18000 * 1024 / n) + 6000

            last_improve = time.time()

            for t in range(1, max_iters + 1):
                # simplex projection via soft‑max
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                # convolution and smooth‑max weights
                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

                # gradient of the smooth‑max surrogate
                grad_a = 2.0 * _cross_correlation(w, a, corr_len)

                # back‑prop through the soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # temperature schedule – get closer to the true max
                lam = min(lam * lam_factor, 1e6)

                # exact objective (a sum is 1)
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cid} Adam iter={t:05d}] new best {val:.6f}")

                # occasional random shake to leave plateaus
                if time.time() - last_improve > 8.0:
                    logits += (rng.random(n) - 0.5) * 1e-2
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # ----- Sub‑gradient (exact max) phase -----------------------
            a = a / a.sum()
            eta = 0.55
            sub_last_improve = time.time()
            max_sub_iters = 4000

            for sub_it in range(max_sub_iters):
                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)

                if not active.any():
                    w = np.zeros_like(conv)
                else:
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()

                grad_a = 2.0 * _cross_correlation(w, a, corr_len)

                a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
                s = a.sum()
                if s < 1e-12:                   # avoid collapse
                    a = rng.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last_improve = time.time()
                    print(f"[n={n} cand={cid} Subgrad iter={sub_it}] new best {val:.6f}")

                if time.time() - sub_last_improve > 8.0:
                    break
                if time.time() > deadline:
                    break

        # ------------------------------------------------------------------
        # Fine‑tune the *global* best for this length (if we have one)
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] fine‑tuning the current best (val={best_val:.6f})")
            a = best_seq.copy()
            logits = np.log(a + 1e-15)
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.04
            lam = 0.5
            lam_factor = 1.01
            last_improve = time.time()
            fine_iters = int(50000 * 1024 / n) + 8000

            for t in range(1, fine_iters + 1):
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

                grad_a = 2.0 * _cross_correlation(w, a, corr_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                lam = min(lam * lam_factor, 1e6)

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} fine‑tune iter={t:05d}] new best {val:.6f}")

                if time.time() - last_improve > 10.0:
                    # small random nudge to escape any tiny plateau
                    logits += (rng.random(n) - 0.5) * 5e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

    # ----------------------------------------------------------------------
    # Final stochastic “annealing” walk around the best solution found.
    if best_seq is not None and time.time() < deadline:
        n = len(best_seq)
        conv_len = _next_pow2(2 * n - 1)
        corr_len = _next_pow2(3 * n)

        a = best_seq.copy()
        val_cur = _objective(a, n, conv_len)
        T = 0.05                     # initial temperature
        T_decay = 0.9999
        anneal_iters = 20000

        print(f"\n[Annealing] starting value: {val_cur:.6f}")
        for it in range(anneal_iters):
            if time.time() > deadline:
                break

            # propose a neighbour by moving a small amount of mass
            delta = rng.normal(scale=0.002, size=n)
            a_new = a + delta
            a_new = np.maximum(a_new, 0.0)
            if a_new.sum() == 0:
                a_new = rng.dirichlet(np.ones(n))
            else:
                a_new /= a_new.sum()

            val_new = _objective(a_new, n, conv_len)

            if val_new < val_cur or rng.random() < np.exp(-(val_new - val_cur) / T):
                a, val_cur = a_new, val_new
                if val_cur < best_val:
                    best_val = val_cur
                    best_seq = a.copy()
                    print(f"[Anneal iter={it}] new best {best_val:.6f}")

            T = max(T * T_decay, 1e-6)

    # ----------------------------------------------------------------------
    # Fallback – uniform vector if nothing was discovered (should never happen)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return as plain Python list for the external evaluator
    return best_seq.tolist()
