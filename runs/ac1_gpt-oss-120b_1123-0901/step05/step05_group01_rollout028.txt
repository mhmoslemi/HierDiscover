def _anneal_optimize(a0, n, fft_len_c, deadline, cur_best_val):
    a = a0.copy()
    best_a = a.copy()
    best_val = cur_best_val
    temperature = 0.1
    decay = 0.9995
    # compute initial objective
    obj = _objective(a, n, fft_len_c)
    # while time left
    while time.time() < deadline:
        # propose random mass transfer between random indices
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            continue
        # propose move: take delta fraction of a[i]
        delta = (np.random.rand() * 0.1) * a[i]  # up to 10% of i's mass
        if delta == 0:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        if new_a[i] < 0:
            continue
        # renormalise
        new_a /= new_a.sum()
        new_val = _objective(new_a, n, fft_len_c)
        diff = new_val - obj
        if diff < 0 or np.random.rand() < np.exp(-diff / temperature):
            a = new_a
            obj = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[Anneal] new best {best_val:.6f}")
        temperature *= decay
        # occasional reset of temperature if too low
        if temperature < 1e-8:
            temperature = 0.1
    return best_a, best_val
```

"""
Propose a step‑function height sequence that minimises

        F(a) = 2·n·max(a∗a) / (sum a)²

using a multi‑stage optimiser with richer initial families,
longer Adam runs and a final simulated‑annealing polishing.
The routine respects a 1000 s budget (≈ 0.5 s safety margin).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers (same as the reference implementation)
def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)−1) using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) = n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)²."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence by linear interpolation and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new

def _generate_initial_candidates(n: int,
                                 best_seq: np.ndarray | None,
                                 hist_seq: np.ndarray | None,
                                 max_cand: int = 20) -> list[np.ndarray]:
    """
    Produce a diverse set of starting points for length n.
    All candidates are non‑negative and sum to 1.
    The pool includes the previous global best, a few classic analytic
    shapes and many random Dirichlet draws.
    """
    cand = []

    # 1) Warm‑start from the current global best (exact length)
    if best_seq is not None and best_seq.shape[0] == n:
        cand.append(best_seq.copy())
    # 2) Warm‑start by interpolating the current best to this length
    if best_seq is not None and best_seq.shape[0] != n:
        cand.append(_interpolate_seq(best_seq, n))

    # 3) Historic sequence (if available)
    if hist_seq is not None:
        if hist_seq.shape[0] > n:
            tmp = hist_seq[:n]
        else:
            tmp = np.pad(hist_seq, (0, n - hist_seq.shape[0]))
        tmp = np.maximum(tmp, 0.0)
        tot = tmp.sum()
        if tot < 1e-12:
            tmp = np.random.dirichlet(np.ones(n))
        else:
            tmp = tmp / tot
        cand.append(tmp)

    # 4) Pure random Dirichlet (uniform concentration)
    cand.append(np.random.dirichlet(np.ones(n)))

    # 5) Low‑concentration Dirichlet – produces spiky starters
    for conc in (0.2, 0.5, 1.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 6) Monotone decreasing exponential families
    for gamma in (0.001, 0.003, 0.01, 0.03, 0.1):
        shape = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = shape / shape.sum()
        cand.append(shape)

    # 7) Monotone decreasing power‑law families
    for alpha in (0.5, 1.0, 1.5, 2.0):
        idx = np.arange(1, n + 1, dtype=np.float64)
        shape = idx ** (-alpha)
        shape = shape / shape.sum()
        cand.append(shape)

    # 8) Linear ramp (high→low)
    shape = np.linspace(1.0, 0.0, n, dtype=np.float64)
    shape = shape / shape.sum()
    cand.append(shape)

    # 9) Quadratic ramp (high→low, smoother)
    shape = 1.0 - (np.arange(n, dtype=np.float64) / (n - 1)) ** 2
    shape = shape / shape.sum()
    cand.append(shape)

    # 10) Cosine ramp (high→low)
    shape = np.cos(np.linspace(0, np.pi / 2, n, dtype=np.float64))
    shape = shape / shape.sum()
    cand.append(shape)

    # 11) Half‑cosine “bell’’ centred at the left edge
    shape = np.cos(np.linspace(0, np.pi, n, dtype=np.float64)) ** 2
    shape = np.maximum(shape, 0.0)
    shape = shape / shape.sum()
    cand.append(shape)

    # 12) Two‑Gaussian edge shapes (kept from the reference)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        cand.append(shape / shape.sum())

    # 13) Three‑peak Gaussian mixture (kept from the reference)
    for sigma in (0.05, 0.10, 0.20):
        x = np.arange(n, dtype=np.float64) / (n - 1)
        peaks = sum(np.exp(-0.5 * ((x - p) / sigma) ** 2) for p in (0.25, 0.5, 0.75))
        cand.append(peaks / peaks.sum())

    # Shuffle and truncate
    np.random.shuffle(cand)
    return cand[:max_cand]

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   fft_len_conv: int,
                   fft_len_corr: int,
                   deadline: float,
                   max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate (exponential weighting).
    Returns the best (a, value) encountered before the deadline.
    """
    # start from a proper simplex point
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5               # temperature for the smooth‑max
    lam_factor = 1.03

    best_a = a.copy()
    best_val = _objective(a, n, fft_len_conv)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ----- simplex projection via soft‑max -----
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ----- smooth‑max weights -----
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        ws = w.sum()
        if ws == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws

        # ----- gradient of the surrogate (via correlation) -----
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # ----- back‑prop through the soft‑max -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, 1e7)

        cur_val = _objective(a, n, fft_len_conv)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t:06d}] new best {cur_val:.6f}")

        # ----- occasional random shake if idle -----
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


def _subgrad_optimize(a0: np.ndarray,
                      n: int,
                      fft_len_conv: int,
                      fft_len_corr: int,
                      deadline: float,
                      init_val: float | None = None,
                      max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient (exponentiated‑gradient) optimiser on the true max.
    """
    a = a0 / a0.sum()
    if init_val is None:
        best_val = _objective(a, n, fft_len_conv)
    else:
        best_val = init_val
    best_a = a.copy()
    last_improve = time.time()
    eta = 0.6               # step‑size for exponentiated gradient

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # safeguard against collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        cur_val = _objective(a, n, fft_len_conv)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it}] new best {cur_val:.6f}")

        if time.time() - last_improve > 4.0:
            break

    return best_a, best_val


def _hill_climb_optimize(a0: np.ndarray,
                         n: int,
                         fft_len_conv: int,
                         deadline: float,
                         init_val: float | None = None,
                         max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Simple cheap hill climbing: move a small amount of mass between two indices.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_conv) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2:
            continue

        # move up to 5 % of the mass from i1 to i2
        eps = np.random.rand() * 0.05 * a[i1]
        if eps <= 0.0:
            continue
        new_a = a.copy()
        new_a[i1] -= eps
        new_a[i2] += eps
        if new_a[i1] < -1e-15:
            continue
        new_a /= new_a.sum()

        new_val = _objective(new_a, n, fft_len_conv)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            if it % 50 == 0:
                print(f"[n={n} HillClimb it={it}] new best {new_val:.6f}")
        else:
            # occasional uphill move to avoid tiny plateaus
            if np.random.rand() < 0.001:
                a = new_a

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     fft_len_conv: int,
                     deadline: float,
                     cur_best_val: float) -> tuple[np.ndarray, float]:
    """
    Very cheap simulated‑annealing polish that perturbs the current best.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = cur_best_val
    cur_val = cur_best_val
    temperature = 0.1
    decay = 0.99995

    while time.time() < deadline:
        # propose a random mass transfer
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = (np.random.rand() * 0.05) * a[i]   # up to 5 % of i's mass
        if eps == 0:
            continue
        new_a = a.copy()
        new_a[i] -= eps
        new_a[j] += eps
        if new_a[i] < 0:
            continue
        new_a /= new_a.sum()
        new_val = _objective(new_a, n, fft_len_conv)

        diff = new_val - cur_val
        if diff < 0 or np.random.rand() < np.exp(-diff / max(temperature, 1e-12)):
            a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
                print(f"[Anneal] new best {best_val:.6f}")

        temperature *= decay
        if temperature < 1e-8:
            temperature = 0.1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # slight safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Lengths to explore – we concentrate the bulk of the budget on the
    # largest practical size (16384) but still scan a few smaller sizes.
    lengths = [16384, 8192, 4096, 2048, 1024]

    # Pre‑compute FFT lengths (nearest power‑of‑two) for speed.
    conv_fft = {}
    corr_fft = {}
    for n in lengths:
        conv_len = 2 * n - 1
        conv_fft[n] = 1 << int(np.ceil(np.log2(conv_len)))
        # correlation needs conv_len + n - 1 ≤ 3·n
        corr_fft[n] = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

    # Load an historic solution if available (height_sequence_1)
    hist_seq = None
    if "height_sequence_1" in globals():
        tmp = globals()["height_sequence_1"]
        if isinstance(tmp, (list, tuple, np.ndarray)):
            hist_seq = np.asarray(tmp, dtype=np.float64)

    # ------------------------------------------------------------------
    # MAIN optimisation loop
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        fft_len_c = conv_fft[n]
        fft_len_x = corr_fft[n]

        # number of candidate starts – more for the biggest size
        max_candidates = 30 if n >= 8192 else 18
        candidates = _generate_initial_candidates(n, best_seq, hist_seq, max_candidates)

        # Adam iteration budget – longer runs for larger n
        max_adam_iters = int(20000 * 1024 / n) + 3000

        for cand_id, a_start in enumerate(candidates):
            if time.time() > deadline:
                break

            # ---- Phase 1 – smooth‑max Adam ----
            a_adam, val_adam = _adam_optimize(
                a_start, n, fft_len_c, fft_len_x, deadline, max_iters=max_adam_iters
            )

            # ---- Phase 2 – exact sub‑gradient ----
            a_sub, val_sub = _subgrad_optimize(
                a_adam, n, fft_len_c, fft_len_x, deadline,
                init_val=val_adam, max_iters=3000
            )

            # ---- Phase 3 – cheap hill‑climb ----
            a_hc, val_hc = _hill_climb_optimize(
                a_sub, n, fft_len_c, deadline, init_val=val_sub, max_iters=3000
            )

            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.copy()
                print(f"*** Global new best (n={n} cand={cand_id}) value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final annealing refinement on the current global best (if time permits)
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        fft_len_c = conv_fft.get(n, 1 << int(np.ceil(np.log2(2 * n - 1))))
        print("\n=== Final simulated‑annealing refinement ===")
        best_seq, best_val = _anneal_optimize(best_seq, n, fft_len_c, deadline, best_val)
        print(f"*** Refined global best value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (unlikely)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Convert to plain Python list for the external evaluator.
    return best_seq.tolist()
