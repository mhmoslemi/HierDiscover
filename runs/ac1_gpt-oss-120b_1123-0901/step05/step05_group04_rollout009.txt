def proj_simplex(v, z):
    # project onto simplex sum = z, nonnegative
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)

def full_convolution_fft(a):
    a = np.asarray(a, dtype=float)
    n = a.size
    size = 1 << ((2*n - 1 - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    # eliminate tiny negative: conv = np.maximum(conv, 0.0)
    np.maximum(conv, 0.0, out=conv)
    return conv
```

def subgradient_refine(a, target_sum, deadline, step_factor=0.01):
    a = a.copy()
    n = a.size
    best = a.copy()
    best_score = evaluate_sequence(a.tolist())
    while time.time() < deadline:
        conv = full_convolution_fft(a)
        # find indices with maximum value (could be many)
        max_val = conv.max()
        ks = np.flatnonzero(conv == max_val)
        # pick one index randomly
        k = ks[np.random.randint(len(ks))]
        # compute gradient
        grad = np.zeros_like(a)
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        # for i in [lo..hi], j = k-i in range.
        # grad[i] = 2*a[k-i]
        idx = np.arange(lo, hi+1)
        grad[idx] = 2.0 * a[k - idx]
        # step size per iteration
        # adapt step size relative to target_sum
        eta = step_factor * target_sum / (np.linalg.norm(grad) + 1e-12)
        a_new = a - eta * grad
        # project onto simplex
        a_new = project_simplex(a_new, target_sum)
        # ensure non-negative (projected already)
        # compute new score
        score = evaluate_sequence(a_new.tolist())
        if score < best_score - 1e-9:
            best_score = score
            best = a_new.copy()
        # optionally accept even if not improved to escape? maybe not.
        a = a_new
        # early stop if no improvement for many iterations
        # Could check after 500 iterations no improvement then break
        # But time controlled.
    return best, best_score

def _project_onto_simplex(v: np.ndarray, z: float) -> np.ndarray:
    n = v.shape[0]
    # Sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n+1)

        return np.full_like(v, z / n)

    w = np.maximum(v - theta, 0)
    return w

def _subgradient_refine(a: np.ndarray, target_sum: float, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    a = np.clip(a, 0.0, 1000.0)
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_score = _compute_score(best_a)
    n = a.shape[0]
    no_improve = 0
    while time.time() < deadline:
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        ks = np.flatnonzero(conv == max_val)
        k = ks[rng.integers(len(ks))]
        # compute gradient
        grad = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]
        # step size: set to a fraction of target_sum / norm_grad
        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0:
            break
        eta = 0.4 * target_sum / grad_norm  # step factor
        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, target_sum)
        # Clip to [0,1000]
        a_new = np.clip(a_new, 0.0, 1000.0)
        score = _compute_score(a_new)
        if score < best_score - 1e-9:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        a = a_new
        if no_improve > 3000:
            break
    return best_a, best_score

def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    rng: np.random.Generator,
    init_tau_factor: float = 0.25,
    min_tau_factor: float = 5e-4,
    alpha: float = 0.07,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau_decay: float = 0.96,
    max_iter: int | None = None,
) -> tuple[np.ndarray, float]:
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    best_seq = None
    best_val = float('inf')
    iteration = 0
    last_improve = 0
    while time.time() < run_deadline:
        # Softmax mapping
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        p = exp_shift / exp_shift.sum()
        a = target_sum * p

        # Convolution
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        if max_conv < best_val:
            best_val = max_conv
            best_seq = a.copy()
            last_improve = iteration
            # print progress
            print(f"[{time.time():.2f}] n={n} iter={iteration} local best={max_conv:.8f}")

        # Smooth max weights
        tau = max(init_tau_factor * max_conv * (tau_decay ** iteration),
                  min_tau_factor * max_conv)
        scaled = (conv - max_conv) / tau  # non-positive
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient wrt a
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # Gradient wrt logits x (chain rule)
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # Clip gradient norm
        gnorm = np.linalg.norm(grad_x)
        if gnorm > 10.0:
            grad_x = grad_x * (10.0 / gnorm)

        # Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter
        if iteration % 300 == 0 and rng.random() < 0.15:
            x += rng.normal(size=n) * 0.001

        # decay learning rate
        alpha *= 0.9997

        if iteration - last_improve > 2500:
            break
        if max_iter is not None and iteration >= max_iter:
            break

    # final score via internal compute
    final_score = _compute_score(best_seq) if best_seq is not None else float('inf')
    return best_seq, final_score

"""
Search for a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm mixes several ideas:

*   A large pool of random initial shapes (power‑law, exponential, mixed,
    uniform) and occasional warm‑starts from previously good candidates.
*   A soft‑max Adam optimiser that minimises a smooth approximation of the
    max‑convolution.  The temperature (τ) is annealed aggressively so that the
    optimisation eventually follows a true sub‑gradient of the L∞ objective.
*   After each Adam run a cheap sub‑gradient refinement on the exact
    max‑convolution (the true L∞ norm) is performed – this tightens the bound
    further.
*   All iterates are projected onto the simplex
    {a_i ≥ 0, Σ a_i = √(2 n)} to keep the normalisation required by the
    evaluator.  Projection is done with the standard O(n log n) algorithm.
*   The search stops when the time budget expires (≈ 1000 s) and returns the
    best sequence seen.  The final sequence is a plain Python list, ready for
    the outer harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Numerical helpers
# ----------------------------------------------------------------------


def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a power‑of‑2 FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)            # remove tiny negatives
    return conv


def _compute_score(seq: np.ndarray) -> float:
    """
    Fast proxy for evaluate_sequence when Σ a_i = √(2 n):
    the returned value is exactly max(conv(a, a)).
    """
    conv = _full_convolution_fft(seq, seq)
    return float(conv.max())


def _project_onto_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
    S = {x ≥ 0, Σ x_i = z}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – fall back to uniform distribution
        return np.full_like(v, z / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Random initial shapes (all are later normalised to √(2 n))
# ----------------------------------------------------------------------


def _initial_shape_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a positive vector of length n with a random profile."""
    typ = rng.choice(['power', 'exponential', 'mixed', 'uniform'])
    idx = np.arange(1, n + 1, dtype=float)

    if typ == 'power':
        p = rng.uniform(0.1, 2.0)               # exponent
        shape = idx ** (-p)
    elif typ == 'exponential':
        tau = rng.uniform(0.2, 0.9) * n
        shape = np.exp(-idx / tau)
    elif typ == 'mixed':
        split = rng.integers(1, n - 1)
        left = idx[:split] ** (-rng.uniform(0.2, 1.0))
        right = (n - idx[split:] + 1) ** (-rng.uniform(0.2, 1.0))
        shape = np.concatenate([left, right])
    else:  # uniform
        shape = np.ones_like(idx)

    # tiny jitter to avoid exact symmetries
    shape += rng.random(n) * 1e-6
    shape = np.maximum(shape, 1e-12)
    return shape

        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape flat plateaux
        if iteration % 300 == 0 and rng.random() < 0.15:
            x += rng.normal(size=n) * 0.001

        # slowly decay learning rate
        alpha *= 0.9997

        # early stop if we have not improved for a while
        if iteration - last_improve > 2500:
            break
        if max_iter is not None and iteration >= max_iter:
            break

    final_score = _compute_score(best_seq) if best_seq is not None else float("inf")
    return best_seq, final_score


# ----------------------------------------------------------------------
# Sub‑gradient refinement on the exact L∞ objective
# ----------------------------------------------------------------------


def _subgradient_refine(
    a: np.ndarray,
    target_sum: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on   max(conv(a, a)).
    The iterate is always projected onto the simplex Σ a_i = target_sum,
    a_i ≥ 0.  Returns the best sequence seen and its internal max‑conv value.
    """
    a = np.clip(a, 0.0, 1000.0)
    a = a * (target_sum / a.sum())
    best_a = a.copy()
    best_val = _compute_score(best_a)

    n = a.shape[0]
    stagnant = 0

    while time.time() < deadline:
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        # pick a random index among those attaining the maximum
        ks = np.flatnonzero(conv == max_val)
        k = ks[rng.integers(len(ks))]

        # gradient of conv[k] w.r.t. a (2 * a_{k-i} where defined)
        grad = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * a[k - idx]

        gnorm = np.linalg.norm(grad)
        if gnorm == 0.0:
            break
        # step size proportional to target_sum / ‖grad‖
        eta = 0.4 * target_sum / gnorm
        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, target_sum)
        a_new = np.clip(a_new, 0.0, 1000.0)

        val_new = _compute_score(a_new)
        if val_new < best_val - 1e-9:
            best_val = val_new
            best_a = a_new.copy()
            stagnant = 0
        else:
            stagnant += 1

        a = a_new
        if stagnant > 3000:                         # give up if stuck
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the outer harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the given
    evaluation function.  The routine respects a ``budget_s`` time limit
    (≈ 1000 s) and returns the best sequence discovered as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.1                # safety margin

    best_seq = None
    best_score = float("inf")
    top_candidates = []  # list of (score, seq) – keep a handful of elites

    # --------------------------------------------------------------
    # Warm‑start from a previously known construction, if available.
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        hs_arr = hs_arr * (np.sqrt(2 * hs_arr.size) / hs_arr.sum())
        sc = _compute_score(hs_arr)
        if sc < best_score:
            best_score = sc
            best_seq = hs_arr.copy()
            top_candidates.append((sc, hs_arr.copy()))
            print(f"[{time.time():.2f}] Warm‑start global = {sc:.8f} (n={hs_arr.size})")
    except Exception:
        pass  # no global seed present

    # ----------------------------------------------------------------
    # Helper to update the global best with a new candidate
    # ----------------------------------------------------------------
    def _update_global(seq: np.ndarray):
        nonlocal best_seq, best_score, top_candidates
        seq = np.clip(seq, 0.0, 1000.0)
        n = seq.size
        target = np.sqrt(2.0 * n)
        seq = seq * (target / seq.sum())
        score = _compute_score(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] GLOBAL NEW BEST = {score:.8f} (n={n})")
        top_candidates.append((score, seq.copy()))
        top_candidates.sort(key=lambda x: x[0])
        if len(top_candidates) > 10:
            top_candidates = top_candidates[:10]

    # --------------------------------------------------------------
    # Main restart loop – each iteration receives a small time slice.
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # Choose problem size (larger early on, smaller later for speed)
        if remaining > 300:
            n = rng.integers(2000, 4000 + 1)
        elif remaining > 120:
            n = rng.integers(1200, 2500 + 1)
        else:
            n = rng.integers(800, 1800 + 1)

        # ----------------------------------------------------------
        # Build an initial shape (occasionally from a good previous candidate)
        # ----------------------------------------------------------
        if top_candidates and rng.random() < 0.45:
            src_seq = top_candidates[rng.integers(len(top_candidates))][1]
            if src_seq.size >= n:
                init = src_seq[:n].copy()
            else:
                pad = rng.random(n - src_seq.size) * 1e-6
                init = np.concatenate([src_seq, pad])
        else:
            init = _initial_shape_random(n, rng)

        # ----------------------------------------------------------
        # Quick evaluation of the raw shape (cheap way to catch lucky draws)
        # ----------------------------------------------------------
        if rng.random() < 0.12:
            _update_global(init)

        # ----------------------------------------------------------
        # Prepare Adam optimiser (soft‑max parametrisation)
        # ----------------------------------------------------------
        target_sum = np.sqrt(2.0 * n)
        init = init / init.sum() * target_sum
        x0 = np.log(init + 1e-12)

        # Allocate a time slice for this optimisation run
        slice_time = min(12.0, remaining * 0.55)
        run_deadline = min(deadline, time.time() + slice_time)

        seq_local, score_local = _optimise_adam_softmax(
            n,
            x0,
            target_sum,
            run_deadline,
            rng,
            init_tau_factor=0.25,
            min_tau_factor=5e-4,
            alpha=0.07,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            tau_decay=0.96,
            max_iter=None,
        )

        if seq_local is not None:
            _update_global(seq_local)

        # ----------------------------------------------------------
        # Optional sub‑gradient polishing of the Adam result
        # ----------------------------------------------------------
        if seq_local is not None and (deadline - time.time()) > 4.0:
            refine_deadline = min(deadline, time.time() + min(4.0, remaining * 0.2))
            seq_refined, score_refined = _subgradient_refine(
                seq_local, target_sum, refine_deadline, rng
            )
            if score_refined < best_score:
                _update_global(seq_refined)

        # early exit if we already beat the target (1.503)
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Final polishing on the overall best (time permitting)
    # ------------------------------------------------------------------
    if best_seq is not None and (deadline - time.time()) > 8.0:
        n = best_seq.size
        target_sum = np.sqrt(2.0 * n)
        final_deadline = deadline
        seq_refined, score_refined = _subgradient_refine(
            best_seq, target_sum, final_deadline, rng
        )
        if score_refined < best_score:
            best_seq = seq_refined.copy()
            best_score = score_refined
            print(f"[{time.time():.2f}] FINAL REFINEMENT improved best to {best_score:.8f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that no candidate was generated.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # The outer harness expects a plain Python list.
    return best_seq.tolist()
