def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_val = float('inf')
    # Choose lengths to explore in order of increasing size
    lengths = [1024, 2048, 4096, 8192, 16384]
    for n in lengths:
        if time.time() > deadline: break
        conv_len = _next_pow2(2 * n - 1)
        corr_len = _next_pow2(3 * n)  # enough for cross-corr (2n-1 + n -1)
        # Build candidate init list
        candidates = []
        # if best_seq has same length -> warm start
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.array(best_seq, dtype=np.float64))
        # generate N random Dirichlet seeds (some spiky, some uniform)
        for conc in [0.2, 0.5, 1.0]:
            for _ in range(2):
                cand = np.random.dirichlet(np.full(n, conc))
                candidates.append(cand)
        # many random exponentials
        for gamma in [0.001, 0.002, 0.005, 0.01]:
            i = np.arange(n, dtype=np.float64)
            cand = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
            cand = cand / cand.sum()
            candidates.append(cand)
        # power-law edge shapes
        idx = np.arange(1, n+1, dtype=np.float64)
        prod = idx * (n - idx + 1)
        for alpha in np.linspace(-2, -0.2, 5):
            shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
            if shape.sum() > 0:

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x ≥ 1)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a using an FFT of length ``fft_len``.
    Returns an array of length 2*len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Output length = len(a).  Implements

        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)]

    where n = len(a) and w has length 2·n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

            lam = 0.1
            lam_factor = 1.015
            max_lam = 1e6
            last_improve = time.time()
            fine_iters = int(50000 * 1024 / n) + 10000

            for it in range(1, fine_iters + 1):
                if time.time() > deadline:
                    break
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum != 0.0:
                    w /= w_sum
                else:
                    w = np.full_like(w, 1.0 / w.size)

                grad_a = 2.0 * _cross_correlation(w, a, corr_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                lam = min(lam * lam_factor, max_lam)

                val = 2.0 * n * max_c
                if val < best_val - 1e-12:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} fine‑tune it={it:05d}] new best {val:.6f}")

                if time.time() - last_improve > 12.0:
                    logits += (np.random.rand(n) - 0.5) * 5e-3
                    last_improve = time.time()

    # ----------------------------------------------------------------------
    # Final random perturbation stage (if time permits)
    # ----------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        n = len(best_seq)
        conv_len = _next_pow2(2 * n - 1)
        corr_len = _next_pow2(3 * n)
        print("[final perturb] attempting small random shakes")
        for trial in range(5):
            if time.time() > deadline:
                break
            logits = np.log(best_seq + 1e-15)
            logits += np.random.randn(n) * 0.02      # moderate noise
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.04
            lam = 0.05
            lam_factor = 1.03
            max_lam = 1e5
            for it in range(2000):
                if time.time() > deadline:
                    break
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()
                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                w = w / w_sum if w_sum != 0.0 else np.full_like(w, 1.0 / w.size)

                grad_a = 2.0 * _cross_correlation(w, a, corr_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** (it + 1))
                v_hat = v / (1.0 - beta2 ** (it + 1))
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                lam = min(lam * lam_factor, max_lam)

                val = 2.0 * n * max_c
                if val < best_val - 1e-12:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[final perturb trial={trial} it={it}] new best {val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – uniform vector (should never be needed)
    # ----------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list for the external evaluator
    return best_seq.tolist()
