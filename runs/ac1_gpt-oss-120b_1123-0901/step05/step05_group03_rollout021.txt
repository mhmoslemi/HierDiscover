"""
Hybrid optimiser for the step‑function problem.

The routine builds a diverse pool of initial probability vectors,
optimises each of them with a long Adam run on a smooth‑max proxy,
refines the result with an exponentiated‑gradient (sub‑gradient) phase,
and finally performs a cheap simulated‑annealing hill‑climb.
If time permits the best candidate is interpolated to a larger size
(8192) and the whole pipeline is repeated – larger n often yields a
lower ‖a∗a‖∞ value.  The code respects the 1000 s budget and prints
intermediate best scores.
"""

import time
import numpy as np
from typing import List, Tuple, Optional

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector with the same length as a.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _cost(a: np.ndarray,
          n: int,
          conv_fft_len: int) -> float:
    """
    Exact objective 2·n·max(a∗a) / (sum(a)²).  Assumes a is non‑negative.
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s <= 0.0:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray,
                n: int,
                conv_fft_len: int,
                corr_fft_len: int,
                deadline: float,
                best_seq: Optional[np.ndarray],
                best_val: float) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation of the log‑sum‑exp (smooth‑max) surrogate.
    Returns the best sequence/value found during this phase.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07

    lam = 0.5          # temperature for the smooth‑max
    lam_factor = 1.05

    last_improve = time.time()
    it = 0
    max_it = 500_000

    while time.time() < deadline and it < max_it:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # current objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c / (a.sum() ** 2))

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule – converge to true max
        lam = min(lam * lam_factor, 1e8)

        # occasional jitter if we are stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    # Return the best sequence encountered (may be a0 if nothing improved)
    return best_seq, best_val


def _subgradient_phase(a0: np.ndarray,
                       n: int,
                       conv_fft_len: int,
                       corr_fft_len: int,
                       deadline: float,
                       best_seq: Optional[np.ndarray],
                       best_val: float) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient phase on the exact max.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    else:
        a /= s

    eta = 0.6
    it = 0
    max_it = 30_000
    last_improve = time.time()

    while time.time() < deadline and it < max_it:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c / (a.sum() ** 2))

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it:05d}] new best {val:.6f}")

        # active set where the max is attained
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient step (keeps non‑negativity)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:          # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 5.0:
            break
        it += 1

    return best_seq, best_val


def _annealing_phase(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     deadline: float,
                     best_seq: Optional[np.ndarray],
                     best_val: float) -> Tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climb that moves a tiny amount of mass
    between two coordinates.  Starts from the best point obtained so far.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    # parameters
    max_it = 50_000
    step_factor = 0.2          # maximal fraction of the donor mass to move
    T0 = 0.002                # initial temperature for uphill moves

    it = 0
    start_time = time.time()
    while time.time() < deadline and it < max_it:
        # linearly decreasing temperature
        remaining = max(deadline - time.time(), 0.0)
        T = T0 * (remaining / (deadline - start_time + 1e-12))

        i, j = np.random.choice(n, 2, replace=False)
        if a[i] <= 0.0:
            it += 1
            continue
        delta = a[i] * step_factor * np.random.rand()
        if delta <= 0.0:
            it += 1
            continue

        a_prop = a.copy()
        a_prop[i] -= delta
        a_prop[j] += delta

        # keep strict feasibility
        if a_prop[i] < 0.0:
            it += 1
            continue

        val_cur = float(2.0 * n *
                        _full_convolution(a, conv_fft_len).max() /
                        (a.sum() ** 2))
        val_new = float(2.0 * n *
                        _full_convolution(a_prop, conv_fft_len).max() /
                        (a_prop.sum() ** 2))

        if val_new < best_val:
            best_val = val_new
            best_seq = a_prop.copy()
            print(f"[Anneal n={n} it={it:05d}] new best {val_new:.6f}")

        # Accept if improvement or with Metropolis probability
        if val_new < val_cur or np.random.rand() < np.exp(-(val_new - val_cur) / (T + 1e-12)):
            a = a_prop

        it += 1

    return best_seq, best_val


def _initial_candidates(n: int, warm: Optional[np.ndarray]) -> List[np.ndarray]:
    """
    Assemble a modest pool of diverse start vectors.
    """
    cand = []

    # warm start and a few cyclical shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # power‑law shapes (heavy tails)
    idx = np.arange(n, dtype=np.float64) + 1.0
    for p in np.linspace(-2.0, -0.5, 4):
        shape = idx ** p
        shape = shape / shape.sum()
        cand.append(shape)

    # two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.2, 0.5, 1.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape = shape / shape.sum()
        cand.append(shape)

    # sinusoidal (single half‑wave)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape = shape / shape.sum()
    cand.append(shape)

    # uniform baseline
    cand.append(np.full(n, 1.0 / n))

    return cand


def _optimise_one_candidate(a0: np.ndarray,
                            n: int,
                            conv_fft_len: int,
                            corr_fft_len: int,
                            deadline: float,
                            best_seq: Optional[np.ndarray],
                            best_val: float) -> Tuple[np.ndarray, float]:
    """
    Full optimisation pipeline for a single start vector.
    """
    # Adam on smooth‑max surrogate
    best_seq, best_val = _adam_phase(a0, n, conv_fft_len,
                                     corr_fft_len, deadline,
                                     best_seq, best_val)

    # exact‑max sub‑gradient refinement
    best_seq, best_val = _subgradient_phase(best_seq, n,
                                            conv_fft_len,
                                            corr_fft_len,
                                            deadline,
                                            best_seq, best_val)

    # cheap simulated‑annealing hill‑climb
    best_seq, best_val = _annealing_phase(best_seq, n,
                                          conv_fft_len,
                                          deadline,
                                          best_seq, best_val)

    return best_seq, best_val


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function (probability vector) with a low
    evaluation score.  The routine respects the time budget and returns the
    best sequence it found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq: Optional[np.ndarray] = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # primary optimisation on a moderate size (4096)
    n0 = 4096
    conv_fft0 = _next_pow2(2 * n0 - 1)
    corr_fft0 = _next_pow2(3 * n0 - 2)

    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n0:
            warm = warm[:n0]
        elif warm.shape[0] < n0:
            warm = np.pad(warm, (0, n0 - warm.shape[0]))
        warm = np.clip(warm, 0.0, None)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    candidates = _initial_candidates(n0, warm)

    # optimise each candidate (stop early if budget exhausted)
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        a0 = np.clip(cand, 0.0, None)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n0))
        else:
            a0 = a0 / a0.sum()
        print(f"--- Optimising candidate {cid + 1}/{len(candidates)} (n={n0})")
        best_seq, best_val = _optimise_one_candidate(
            a0, n0, conv_fft0, corr_fft0, deadline, best_seq, best_val
        )

    # ------------------------------------------------------------------
    # If time permits, upscale to a larger grid (8192) and repeat.
    if time.time() + 10 < deadline:
        n_big = 8192
        conv_fft_big = _next_pow2(2 * n_big - 1)
        corr_fft_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # Linear interpolation onto the finer grid
            x_old = np.linspace(0.0, 1.0, n0, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.clip(a_big, 0.0, None)
            a_big = a_big / a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print(f"--- Upscaling to n={n_big} for final refinement")
        best_seq, best_val = _optimise_one_candidate(
            a_big, n_big, conv_fft_big, corr_fft_big, deadline,
            best_seq, best_val
        )

    # ------------------------------------------------------------------
    # fallback – uniform if nothing succeeded
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # final clipping to respect the evaluator's bounds
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Finished. Best value ≈ {best_val:.6f}")
    return best_seq.tolist()
