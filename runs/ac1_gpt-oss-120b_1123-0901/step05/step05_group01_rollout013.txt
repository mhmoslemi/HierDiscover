"""
Simple random‑search / hill‑climb optimiser for the ``evaluate_sequence``
objective.

The algorithm works as follows:

1.  Initialise a set of candidate sequences (uniform, edge‑biased,
    and a few Dirichlet draws).  All candidates are normalised to have
    unit sum – the objective is homogeneous, so any scaling gives the
    same score.
2.  Evaluate each candidate with ``evaluate_sequence`` and keep the best.
3.  Starting from the current best, perform a cheap hill‑climbing local
    search: repeatedly transfer a small amount of mass between two random
    positions and accept the move if the objective improves.
4.  If a global variable ``height_sequence_1`` is present, it is used as an
    additional warm‑start.
5.  The search respects a total time budget (default 1000 s).  If the
    budget expires early the best sequence found so far is returned.

The implementation is deliberately lightweight: it only uses NumPy and the
provided ``evaluate_sequence`` function, and it makes no use of closures or
lambda expressions, satisfying the competition constraints.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _evaluate(seq: np.ndarray) -> float:
    """Thin wrapper converting a NumPy array to a plain list for the evaluator."""
    return evaluate_sequence(seq.tolist())  # ``evaluate_sequence`` is provided globally


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """
    Generate a modest collection of diverse start‑up vectors of length ``n``.
    Every candidate is non‑negative and normalised to have sum = 1.
    """
    candidates = []

    # 1. uniform distribution – the theoretical optimum for the pure problem
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    candidates.append(uniform)

    # 2. warm‑start from the best sequence seen so far (if it has the right size)
    if warm is not None:
        if warm.shape[0] == n:
            candidates.append(warm.copy())
        else:
            # linearly interpolate / truncate to the required length
            x_old = np.linspace(0.0, 1.0, warm.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            interp = np.interp(x_new, x_old, warm)
            interp = np.maximum(interp, 0.0)
            if interp.sum() < 1e-12:
                interp = np.random.dirichlet(np.ones(n))
            else:
                interp /= interp.sum()
            candidates.append(interp)

    # 3. a few edge‑biased shapes (inverse‑root and double‑exponential)
    i = np.arange(n, dtype=np.float64) + 1.0
    inv_root = 1.0 / np.sqrt(i) + 1.0 / np.sqrt(n - i + 1.0)
    inv_root = np.maximum(inv_root, 0.0)
    inv_root /= inv_root.sum()
    candidates.append(inv_root)

    exp_edge = np.exp(-0.5 * (i - 1) ** 2) + np.exp(-0.5 * (n - i) ** 2)
    exp_edge = np.maximum(exp_edge, 0.0)
    exp_edge /= exp_edge.sum()
    candidates.append(exp_edge)

    # 4. a handful of random Dirichlet samples
    for _ in range(5):
        cand = np.random.dirichlet(np.ones(n))
        candidates.append(cand)

    # keep the list short so that the search stays within the time budget
    return candidates[:12]


def _hill_climb(a: np.ndarray,
                deadline: float,
                max_iters: int = 3000,
                step_scale: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Very cheap local optimiser.

    At each iteration a random pair of indices (i, j) is chosen,
    a small amount of probability mass is moved from ``i`` to ``j``,
    and the move is kept if the objective improves.
    """
    best = a.copy()
    best_val = _evaluate(best)

    n = a.shape[0]

    for it in range(max_iters):
        if time.time() > deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # transfer up to ``step_scale`` of the source mass to the target
        delta = step_scale * a[i] * np.random.rand()
        if delta <= 0.0:
            continue

        new = best.copy()
        new[i] -= delta
        new[j] += delta

        # keep the vector on the simplex
        if new[i] < 0.0:
            continue
        new_sum = new.sum()
        if new_sum == 0.0:
            continue
        new /= new_sum

        val = _evaluate(new)
        if val < best_val:
            best_val = val
            best = new
            # occasional progress output – useful when the budget is large
            if it % 200 == 0:
                print(f"[hill‑climb] iter={it:05d} new best={best_val:.6f}")

    return best, best_val


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence that minimises

        F(a) = 2·n·max(a∗a) / (∑a)²

    The function respects a total runtime budget (default 1000 s) and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # a small safety margin

    # ------------------------------------------------------------------
    # Load the historic solution, if present – it may already be very good.
    warm_start = None
    if "height_sequence_1" in globals():
        tmp = globals()["height_sequence_1"]
        if isinstance(tmp, (list, tuple, np.ndarray)):
            arr = np.asarray(tmp, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0:
                warm_start = np.maximum(arr, 0.0)
                if warm_start.sum() > 0.0:
                    warm_start /= warm_start.sum()

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Try a few different lengths – short lengths are cheap, long lengths
    # give the algorithm more freedom.
    lengths = [512, 1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Exploring length n={n} ===")
        candidates = _initial_candidates(n, warm_start)

        # ----------------------------------------------------------------
        # Evaluate every start‑up candidate and keep the best.
        for idx, cand in enumerate(candidates):
            if time.time() > deadline:
                break

            val = _evaluate(cand)
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                print(f"[n={n} init {idx}] new global best {best_val:.6f}")

        # ----------------------------------------------------------------
        # Refine the current global best (if it has the same length) with a
        # cheap hill‑climb.  For a different length we interpolate.
        if best_seq is not None:
            if best_seq.shape[0] != n:
                # simple linear interpolation to the current length
                x_old = np.linspace(0.0, 1.0, best_seq.shape[0])
                x_new = np.linspace(0.0, 1.0, n)
                seq = np.interp(x_new, x_old, best_seq)
                seq = np.maximum(seq, 0.0)
                if seq.sum() < 1e-12:
                    seq = np.random.dirichlet(np.ones(n))
                else:
                    seq /= seq.sum()
            else:
                seq = best_seq.copy()

            # hill‑climb on this length
            refined, val_refined = _hill_climb(seq, deadline, max_iters=2000)
            if val_refined < best_val:
                best_val = val_refined
                best_seq = refined.copy()
                print(f"[n={n} hill‑climb] new global best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded (should never happen) return a uniform
    # vector of moderate size.
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = _evaluate(best_seq)

    print(f"\n*** Final best value {best_val:.6f} with length {best_seq.shape[0]} ***")
    return best_seq.tolist()
