import time, numpy as np

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0

    # Determine n: try larger n if time.
    n = 8192
    if time.time() + 0.5 > deadline:
        n = 4096
    # compute FFT lengths
    fft_len = _next_pow2(2*n - 1)
    corr_fft_len = _next_pow2(3*n - 2)  # for cross-correlation length required

    best_val = float('inf')
    best_seq = None

    # Define function to evaluate
    def evaluate(a):
        nonlocal best_val, best_seq
        val = compute_val(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"New best val {val:.6f} at time {time.time():.2f}")
        return val

    # Parameter grid search
    # Exponential shape
    for alpha in np.linspace(0.990, 0.9999, 200):
        if time.time() >= deadline:
            break
        a = np.exp(np.log(alpha) * np.arange(n))
        a = a / a.sum()
        evaluate(a)

    # Symmetric exponential
    for alpha in np.linspace(0.990, 0.9999, 150):
        if time.time() >= deadline:
            break
        i = np.arange(n)
        mid = (n-1)/2
        a = np.exp(np.log(alpha) * np.abs(i - mid))
        a = a / a.sum()
        evaluate(a)

    # Power law negative
    for p in np.linspace(-4.0, -0.2, 200):
        if time.time() >= deadline:
            break
        i = np.arange(1, n+1)
        a = i**p
        a = a / a.sum()
        evaluate(a)

    # Symmetric power law:
    for p in np.linspace(-4.0, -0.2, 150):
        if time.time() >= deadline:
            break
        i = np.arange(1, n+1)
        rev = (n - i + 1)**p
        a = i**p + rev
        a = a / a.sum()
        evaluate(a)

    # Gaussian shape
    for gamma in np.linspace(0.0005, 0.5, 150):
        if time.time() >= deadline:
            break
        i = np.arange(n)
        mid = (n-1)/2
        a = np.exp(-gamma * (i - mid)**2)
        a = a / a.sum()
        evaluate(a)

    for _ in range(30):
        if time.time() >= deadline:
            break
        a = np.random.dirichlet(np.ones(n))

    if best_seq is not None:
        # Use Adam on smooth max for a few thousand iterations, respecting deadline

    best_seq = np.clip(best_seq, 0.0, 1000.0)

def _adam_refine(a0, n, conv_fft_len, corr_fft_len, deadline, best_val):
    a = a0.copy()
    # parameters
    lr = 0.03
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lam = 0.5
    lam_factor = 1.1
    it = 0
    last_improve = time.time()
    while time.time() < deadline - 0.5 and it < 50000:
        # softmax param
        logits = np.log(a + 1e-15)
        # compute a via softmax
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        # compute conv
        conv = _full_convolution_fft(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam refine iter={it}] new best {val:.6f}")
        # smooth weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # gradient of logits
        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** (it+1))
        v_hat = v / (1 - beta2 ** (it+1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
        lam = min(lam * lam_factor, 1e7)
        # optional random shake
        if time.time() - last_improve > 10.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()
        it += 1
        # compute a again from logits each iteration? Actually we recomputed a from logits at top; but updating logits directly modifies a after next iteration. Fine.

    # after loop, compute final a from logits
    logits = np.log(best_seq + 1e-15) if best_seq is not None else np.log(a0 + 1e-15)
    mx = logits.max()
    exp_log = np.exp(logits - mx)
    a = exp_log / exp_log.sum()
    return a, best_val

"""
Propose a step‑function (non‑negative vector) that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)²)

where * denotes the linear convolution.
The algorithm works in three phases:

1. **Fast parametric sweep** – a collection of simple analytic families
   (exponential, symmetric exponential, power‑law, symmetric power‑law,
   Gaussian) is sampled on a dense grid.  Every candidate is evaluated
   with an FFT based convolution, the best one is kept.

2. **Optional warm‑start** – if a previous solution exists in the global
   variable ``height_sequence_1`` it is interpolated to the working length
   and added to the pool.

3. **Local refinement** – the best candidate from phase 1 is refined by
   a smooth‑max Adam optimisation followed by a few sub‑gradient steps.
   Both phases respect the overall time budget (default 1000 s).

The implementation avoids any closures and uses only NumPy (FFT) and
standard Python libraries, so it can be executed in the evaluation
environment.  The returned list is guaranteed to be non‑negative and
conforms to the limits imposed by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than *x*."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using an FFT of size *fft_len*."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Result has length len(a) and equals
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1 .
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective_val(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact value of F(a) (sum(a) is assumed > 0)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
def _adam_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    start_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smooth‑max surrogate of the max‑convolution.
    Returns the refined vector and the corresponding exact value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5          # initial temperature (smooth‑max)
    lam_factor = 1.02

    best_seq = a0.copy()
    best_val = start_val
    last_improve = time.time()
    it = 0

    # --------------------------------------------------------------- Adam loop
    while time.time() < deadline:
        # ----- soft‑max mapping (keeps a on the simplex)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- exact objective (for tracking)
        val = _objective_val(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam iter={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- gradient w.r.t. logits (chain rule through soft‑max)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule – slowly turn the surrogate into a true max
        lam = min(lam * lam_factor, 1e6)

        # ----- occasional random “shake” when we stall
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        if it > 200_000:        # hard safety stop
            break

    # ------------------------------------------------------- Sub‑gradient phase
    a = best_seq.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    eta = 0.6
    sub_last_improve = time.time()
    for sub_it in range(8_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = _objective_val(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[Subgrad iter={sub_it:04d}] new best {val:.6f}")

        # active set – indices attaining the maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                 # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 5.0:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
def _generate_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a modest pool of structured candidates.
    Each vector is normalised to have L1‑norm 1.
    """
    cand = []

    # 1) warm start (previous best) and a few rotated versions
    if warm is not None:
        cand.append(warm.copy())
        shifts = (n // 4, n // 2, 3 * n // 4)
        for sh in shifts:
            cand.append(np.roll(warm, sh))

    # 2) pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # 3) exponential families
    #    a_i = α^i
    for lam in np.linspace(0.5e-4, 3.0e-4, 70):               # λ = -log α
        alpha = np.exp(-lam)
        a = np.exp(-lam * np.arange(n))
        a /= a.sum()
        cand.append(a)

    #    symmetric exponential: a_i = α^{|i-mid|}
    mid = (n - 1) / 2.0
    for lam in np.linspace(0.5e-4, 3.0e-4, 60):
        a = np.exp(-lam * np.abs(np.arange(n) - mid))
        a /= a.sum()
        cand.append(a)

    # 4) power‑law families: a_i = (i+1)^p  (p < 0)
    idx = np.arange(1, n + 1, dtype=np.float64)
    for p in np.linspace(-4.0, -0.2, 80):
        a = idx ** p
        a /= a.sum()
        cand.append(a)

    #   symmetric power‑law: a_i = (i+1)^p + (n-i)^p
    rev = (n - idx + 1) ** p
    a = idx ** p + rev
    a /= a.sum()
    cand.append(a)

    # 5) Gaussian family centred in the middle
    for gamma in np.linspace(1e-4, 0.4, 70):
        a = np.exp(-gamma * (np.arange(n) - mid) ** 2)
        a /= a.sum()
        cand.append(a)

    # Ensure all candidates are non‑negative and have non‑zero sum
    filtered = []
    for v in cand:
        v = np.clip(v, 0.0, 1000.0)
        if v.sum() > 0.01:
            filtered.append(v / v.sum())
    return filtered


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a low‑scoring non‑negative step function.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.6          # small safety margin

    # ------------------------------------------------------------------
    # Choose a working length.
    # 8192 works well within the time budget and allows a value close to the theoretical lower bound 1.
    n = 8192
    # If the deadline is already too close, fall back to 4096.
    if time.time() + 5.0 > deadline:
        n = 4096

    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start – lift a previous solution if it exists.
    warm = None
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size > 0:
            # interpolate (linear) to the current length
            x_old = np.linspace(0, 1, prev.size, endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            warm = np.interp(x_new, x_old, prev)
            warm = np.clip(warm, 0.0, 1000.0)
            if warm.sum() > 0.01:
                warm = warm / warm.sum()
            else:
                warm = None

    # ------------------------------------------------------------------
    # Fast parametric sweep
    candidates = _generate_candidates(n, warm)

    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        val = _objective_val(cand, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[grid {idx:03d}] new best {val:.6f}")

    # ------------------------------------------------------------------
    # If we still have time, run a more expensive local refinement on the best vector.
    if best_seq is not None and time.time() + 10.0 < deadline:
        print("Starting local Adam refinement...")
        best_seq, best_val = _adam_refine(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_val,
        )
        print(f"Refinement finished – best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – a uniform vector (should never be needed).
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return best_seq.tolist()
