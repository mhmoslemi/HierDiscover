def adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline):
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05
    lam = 1.0
    lam_factor = 1.02
    max_iters = int(40000 * 1024 / n) + 2000
    best_local_val = float('inf')
    best_local_a = a.copy()
    last_improve = time.time()
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # softmax
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        # compute weight
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum if w_sum>0 else np.ones_like(w) / w.size
        # grad wrt a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)
        # grad wrt logits via softmax
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # anneal lam
        lam = min(lam * lam_factor, 1e8)
        # evaluation
        val = 2.0 * n * max_c
        if val < best_local_val:
            best_local_val = val
            best_local_a = a.copy()
            last_improve = time.time()
        # stagnation detection
        if time.time() - last_improve > 4.0:
            # add jitter
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()
    return best_local_a, best_local_val

def subgrad_refine(a, n, conv_fft_len, corr_fft_len, deadline):
    a = a / a.sum()
    best_val = 2.0 * n * _full_convolution(a, conv_fft_len).max()
    best_a = a.copy()
    eta = 0.45 # step size
    last_improve = time.time()
    for it in range(3000):
        if time.time() > deadline:
            break
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
        # optional decreasing eta
        eta = eta * 0.9995
        if time.time() - last_improve > 5.0:
            break
    return best_a, best_val

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5
    best_seq = None
    best_val = float('inf')
    # lengths from small to large (use 2048 as minimal to get good shape)
    lengths = [2048, 4096, 8192, 16384, 32768]  # optionally 65536 if time remaining
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Length {n} ===")
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))
        # generate candidates
        candidates = []
        # uniform
        candidates.append(np.full(n, 1.0/n))
        # warm restart from best (if same length)
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.asarray(best_seq))
        # upsample previous best (if exists and different length)
        if best_seq is not None and len(best_seq) != n:
            old = np.asarray(best_seq)
            x_old = np.linspace(0,1,len(old))
            x_new = np.linspace(0,1,n)
            up = np.interp(x_new, x_old, old)
            candidates.append(up / up.sum())
        # historic variable
        hseq = globals().get('height_sequence_1')
        if hseq is not None:
            h = np.asarray(hseq, dtype=np.float64)
            if len(h) != n:
                if len(h) > n:
                    h = h[:n]
                else:
                    h = np.pad(h, (0, n-len(h)))
            candidates.append(np.maximum(h, 0))
        # random Dirichlet draws (multiple)
        for _ in range(3):
            candidates.append(np.random.dirichlet(np.ones(n)))
        # beta-power families (symmetric)
        idx = np.arange(1, n+1, dtype=np.float64)
        prod = idx * (n - idx + 1.0)
        for alpha in [-2.5, -2.0, -1.5, -1.0, -0.5]:
            shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
            if shape.sum() > 0:
                shape = shape / shape.sum()
                candidates.append(shape)
        # two-gaussian edge shapes
        i = np.arange(n, dtype=np.float64)
        for gamma in [0.1, 0.3, 0.6, 1.0]:
            shape = np.exp(-gamma * (i**2)) + np.exp(-gamma * ((n-1 - i)**2))
            shape = shape / shape.sum()
            candidates.append(shape)
        # sine-squared bump
        shape = np.sin(np.pi * (np.arange(n)+0.5)/n)**2
        candidates.append(shape/shape.sum())
        # mixture of uniform and edge shape
        edge = np.exp(-0.5 * (i**2)) + np.exp(-0.5 * ((n-1-i)**2))
        edge /= edge.sum()
        mix = 0.5* np.full(n, 1.0/n) + 0.5* edge
        candidates.append(mix / mix.sum())
        # process candidates
        for cand_id, cand in enumerate(candidates):
            if time.time() > deadline:
                break
            try:
                a_opt, val_opt = adam_optimize(cand, n, conv_fft_len, corr_fft_len, deadline)
                # Sub-gradient refinement
                a_ref, val_ref = subgrad_refine(a_opt, n, conv_fft_len, corr_fft_len, deadline)
                if val_ref < best_val:
                    best_val = val_ref
                    best_seq = a_ref.copy()
                    print(f"[n={n} cand={cand_id}] new global best {best_val:.6f}")
                # Also keep a_opt if better than current global:
                if val_opt < best_val:
                    best_val = val_opt
                    best_seq = a_opt.copy()
                    print(f"[n={n} cand={cand_id}] new global best (adam) {best_val:.6f}")
            except Exception as e:
                print(f"[n={n} cand={cand_id}] exception {e}")
        # after processing candidates for this length, do local jitter around best for same length
        if best_seq is not None and len(best_seq) == n:
            for jitter_id in range(5):
                if time.time() > deadline:
                    break
                logits = np.log(np.maximum(best_seq, 1e-12))
                logits += np.random.normal(scale=0.02, size=n)  # small perturb
                a0 = np.exp(logits - np.max(logits))
                a0 = a0 / a0.sum()
                a_opt, val_opt = adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline)
                a_ref, val_ref = subgrad_refine(a_opt, n, conv_fft_len, corr_fft_len, deadline)
                if val_ref < best_val:
                    best_val = val_ref
                    best_seq = a_ref.copy()
                    print(f"[n={n} jitter={jitter_id}] new global best {best_val:.6f}")
    # fallback
    if best_seq is None:
        best_seq = np.full(2048, 1.0/2048)
    return best_seq.tolist()

"""
Propose a step‑function (non‑negative vector) that minimises

    F(a) = 2 * n * max(a * a) / (sum(a)**2)

where * denotes linear convolution.  The objective is homogeneous, so we fix
sum(a)=1 and minimise 2·n·max(conv).  The optimiser works on a cascade of
lengths, creates a diverse set of seeds, and refines each seed with two
phases:

1. Adam on a smooth‑max surrogate of the convolution maximum (log‑sum‑exp
   with a slowly increasing temperature).  The vector is parametrised by
   logits → soft‑max, guaranteeing non‑negativity and unit sum.

2. Exact sub‑gradient refinement on the true ℓ∞‑norm using an
   exponentiated‑gradient update.

After processing a length we jitter the current global best (if its length
matches) to escape possible local minima.  The routine respects the supplied
wall‑clock budget (default 1000 s) and returns the best sequence found as a
plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a*a using a power‑of‑two FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) required for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]  for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max surrogate.
    Returns (best_a, best_val) where best_val = 2*n*max(conv) for the best
    iterate discovered before the deadline.
    """
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05

    lam = 1.0               # smooth‑max temperature
    lam_factor = 1.02

    # iteration budget scaled with vector length
    max_iters = int(40000 * 1024 / n) + 2000
    best_val = float("inf")
    best_a = a.copy()
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ----- soft‑max to enforce simplex constraints -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- convolution and smooth‑max weights -----
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum if w_sum > 0 else np.full_like(w, 1.0 / w.size)

        # ----- gradient of smooth‑max w.r.t. a -----
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # ----- back‑propagation through soft‑max -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature annealing -----
        lam = min(lam * lam_factor, 1e8)

        # ----- objective evaluation (exact, because sum(a)=1) -----
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # ----- stagnation handling -----
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Exact ℓ∞ sub‑gradient refinement using an exponentiated‑gradient update.
    Returns (best_a, best_val) after at most 3000 iterations or when the
    deadline is reached.
    """
    a = a0 / a0.sum()
    best_val = float("inf")
    best_a = a.copy()
    eta = 0.45
    last_improve = time.time()

    for it in range(3000):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                      # fallback to a random distribution
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        eta *= 0.9995                       # slowly decay step size

        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the wall‑clock budget (seconds) and
    returns the best sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    best_seq = None
    best_val = float("inf")

    # Process increasingly larger lengths (smaller ones are cheap warm‑starts)
    lengths = [2048, 4096, 8192, 16384, 32768]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

        # --------------------------------------------------------------
        # Build a heterogeneous pool of seeds
        # --------------------------------------------------------------
        candidates = []

        # 1) uniform
        candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # 2) warm‑start from a previous solution of the exact size
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.asarray(best_seq, dtype=np.float64))

        # 3) up‑sample / down‑sample a previous best of different size
        if best_seq is not None and len(best_seq) != n:
            old = np.asarray(best_seq, dtype=np.float64)
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, old)
            if up.sum() > 0:
                up = up / up.sum()
            else:
                up = np.random.dirichlet(np.ones(n))
            candidates.append(up)

        # 4) historic global variable if present
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            h = np.asarray(hseq, dtype=np.float64)
            if h.shape[0] > n:
                h = h[:n]
            elif h.shape[0] < n:
                h = np.pad(h, (0, n - h.shape[0]))
            h = np.maximum(h, 0.0)
            if h.sum() > 0:
                h = h / h.sum()
            else:
                h = np.random.dirichlet(np.ones(n))
            candidates.append(h)

        # 5) a few random Dirichlet draws
        for _ in range(3):
            candidates.append(np.random.dirichlet(np.ones(n)))

        # 6) symmetric power‑law shapes (edge‑concentrated)
        idx = np.arange(1, n + 1, dtype=np.float64)
        prod = idx * (n - idx + 1.0)
        for alpha in [-2.5, -2.0, -1.5, -1.0, -0.5]:
            shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
            if shape.sum() > 0:
                shape = shape / shape.sum()
                candidates.append(shape)

        # 7) two‑Gaussian edge shapes
        i = np.arange(n, dtype=np.float64)
        for gamma in [0.1, 0.3, 0.6, 1.0]:
            shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            shape = shape / shape.sum()
            candidates.append(shape)

        # 8) smooth sine‑squared bump
        shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
        shape = shape / shape.sum()
        candidates.append(shape)

        # 9) mixture of uniform and edge Gaussian
        edge = np.exp(-0.5 * (i ** 2)) + np.exp(-0.5 * ((n - 1 - i) ** 2))
        edge = edge / edge.sum()
        mix = 0.5 * np.full(n, 1.0 / n) + 0.5 * edge
        mix = mix / mix.sum()
        candidates.append(mix)

        # Limit the number of candidates for large n to keep within budget
        max_cand = 12 if n <= 8192 else 6
        candidates = candidates[:max_cand]

        # --------------------------------------------------------------
        # Optimise every candidate
        # --------------------------------------------------------------
        for cand_id, cand in enumerate(candidates):
            if time.time() > deadline:
                break

            try:
                # Phase 1: Adam on smooth‑max surrogate
                a_adam, val_adam = _adam_optimize(
                    cand, n, conv_fft_len, corr_fft_len, deadline
                )

                # Phase 2: exact sub‑gradient refinement
                a_ref, val_ref = _subgradient_refine(
                    a_adam, n, conv_fft_len, corr_fft_len, deadline
                )

                # Keep the best among both phases
                if val_ref < best_val:
                    best_val = val_ref
                    best_seq = a_ref.copy()
                    print(
                        f"[n={n} cand={cand_id}] new global best {best_val:.6f}"
                    )
                elif val_adam < best_val:
                    best_val = val_adam
                    best_seq = a_adam.copy()
                    print(
                        f"[n={n} cand={cand_id}] new global best (Adam) {best_val:.6f}"
                    )
            except Exception as exc:
                print(f"[n={n} cand={cand_id}] exception: {exc}")

        # --------------------------------------------------------------
        # Small random jitters around the current best of this length
        # --------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            for jitter_id in range(5):
                if time.time() > deadline:
                    break
                logits = np.log(np.maximum(best_seq, 1e-12))
                logits += np.random.normal(scale=0.02, size=n)
                a0 = np.exp(logits - np.max(logits))
                a0 = a0 / a0.sum()

                a_adam, val_adam = _adam_optimize(
                    a0, n, conv_fft_len, corr_fft_len, deadline
                )
                a_ref, val_ref = _subgradient_refine(
                    a_adam, n, conv_fft_len, corr_fft_len, deadline
                )
                if val_ref < best_val:
                    best_val = val_ref
                    best_seq = a_ref.copy()
                    print(
                        f"[n={n} jitter={jitter_id}] new global best {best_val:.6f}"
                    )

    # ------------------------------------------------------------------
    # Fallback safety net
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return as plain Python list (required by the external evaluator)
    return best_seq.tolist()
