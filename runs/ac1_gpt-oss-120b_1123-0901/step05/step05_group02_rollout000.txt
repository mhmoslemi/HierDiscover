"""
Optimisation of a non‑negative step function a that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

The algorithm explores several lengths (2048 … 16384) and many
initial shapes (Dirichlet, beta‑law, sinusoid, exponential, mixtures,
and perturbations of the current best).  For each candidate a two
stages are performed:

  1. Adam on a smooth‑max surrogate of max(a*a) (soft‑max with
     temperature λ that is slowly increased).  The parameters are the
     logits of a soft‑max, guaranteeing feasibility (non‑negative,
     sum = 1).

  2. Exact sub‑gradient descent on the true max using a Euclidean
     projection onto the simplex (the classic projection of Duchi et
     al.).  The step size decays as η / √t.

The best sequence found for any length is kept globally; after all
lengths a final fine‑tuning of that global best is performed with a
longer Adam run.  All operations use FFTs for O(n log n) convolution
and cross‑correlation.  The routine respects a user‑supplied time
budget (default 1000 s) and returns a plain Python list suitable for
the external `evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, Σx=1}.
    Implementation of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("vector required")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    best_seq = _global_best_seq
    best_val = _global_best_val

    # ------------------------------------------------------------------
    # Lengths to explore – larger n gives more freedom but is slower.
    lengths = [2048, 4096, 8192, 16384]

    # Pre‑compute power‑of‑two FFT lengths for each n.
    fft_len_conv = {}
    fft_len_corr = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_len_conv[n] = _next_pow2(conv_len)
        # cross‑correlation needs up to conv_len + n – 1 ≤ 3·n
        fft_len_corr[n] = _next_pow2(conv_len + n - 1)

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_len = fft_len_conv[n]
        corr_len = fft_len_corr[n]

        # --------------------- build a list of initial candidates ----------
        candidates = []

        # 1) Warm‑start from global best if it matches current length
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.asarray(best_seq, dtype=np.float64))

        # 2) Warm‑start from historic variable if present (height_sequence_1)
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            cand = np.asarray(hseq, dtype=np.float64)
            if cand.shape[0] > n:
                cand = cand[:n]
            else:
                cand = np.pad(cand, (0, n - cand.shape[0]))
            cand = np.clip(cand, 0.0, None)
            s = cand.sum()
            if s < 0.01:
                cand = np.random.dirichlet(np.ones(n))
            else:
                cand = cand / s
            candidates.append(cand)

        # 3) Random Dirichlet (concentration drawn uniformly)
        for concentration in [0.3, 0.7, 1.0, 1.5]:
            candidates.append(np.random.dirichlet(np.full(n, concentration)))

        # 4) Random Dirichlet with concentration randomly sampled
        for _ in range(3):
            conc = 10 ** np.random.uniform(-2, 1)  # between 0.01 and 10
            candidates.append(np.random.dirichlet(np.full(n, conc)))

        # 5) Beta‑law shapes a_i ∝ (i+1)^{α}·(n-i)^{β}
        for _ in range(5):
            alpha = np.random.uniform(-2, 2)
            beta = np.random.uniform(-2, 2)
            idx = np.arange(n, dtype=np.float64) + 1.0
            shape = np.power(idx, alpha, where=idx > 0, out=np.zeros_like(idx))
            shape *= np.power(n - idx + 1.0, beta, where=(n - idx + 1) > 0,
                              out=np.zeros_like(idx))
            if shape.sum() > 0:
                shape /= shape.sum()
                candidates.append(shape)

        # 6) Sinusoidal shape (positive half‑sine)
        sin_shape = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
        sin_shape = np.maximum(sin_shape, 0.0)
        sin_shape /= sin_shape.sum()
        candidates.append(sin_shape)

        # 7) Two‑sided exponential shape with random decay
        for gamma in np.random.uniform(0.0003, 0.006, size=4):
            i = np.arange(n, dtype=np.float64)
            exp_shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
            exp_shape /= exp_shape.sum()
            candidates.append(exp_shape)

        # 8) Small random perturbation of the current global best (if any)
        if best_seq is not None:
            for _ in range(2):
                pert = best_seq.copy()
                # transfer a tiny amount of mass between random positions
                for __ in range(5):
                    i, j = np.random.randint(0, n, size=2)
                    eps = np.random.uniform(0, 1e-3)
                    if pert[i] >= eps:
                        pert[i] -= eps
                        pert[j] += eps
                pert = _project_simplex(pert)
                candidates.append(pert)

        # ------------------------------------------------------------------
        # Helper to compute objective (sum normalized to 1, so denominator=1)
        def _obj(a_vec: np.ndarray) -> float:
            conv = _full_convolution(a_vec, conv_len)
            return 2.0 * n * conv.max()

        # ------------------------------------------------------------------
        # Optimise every candidate
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # Ensure we start from a valid probability vector
            a = a0 / a0.sum()
            logits = np.log(a + 1e-15)

            # ----- Phase 1: Adam on smooth‑max surrogate -----
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.07
            lam = 0.2                     # temperature
            lam_factor = 1.018
            last_improve = time.time()

            # iteration budget scaled with n (more for smaller n)
            max_iters = int(26000 * 1024 / n) + 4000

            for t in range(1, max_iters + 1):
                # soft‑max → simplex
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                # convolution
                conv = _full_convolution(a, conv_len)
                max_c = conv.max()

                # smooth‑max weights
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # gradient of surrogate (using cross‑correlation)
                grad_a = 2.0 * _cross_correlation(w, a, corr_len)

                # back‑prop through the soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # increase temperature (smooth‑max to true max)
                lam = min(lam * lam_factor, 1e6)

                # exact objective (since sum(a)=1)
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    _global_best_seq = best_seq
                    _global_best_val = best_val
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Adam iter={t:05d}] new best {val:.6f}")

                # simple stagnation handling – random shake
                if time.time() - last_improve > 7.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # ----- Phase 2: Exact sub‑gradient with Euclidean projection -----
            # initialise from the last a of the Adam run
            a = a / a.sum()
            eta0 = 0.6
            sub_last_improve = time.time()
            max_sub_iters = 9000

            for sub_it in range(1, max_sub_iters + 1):
                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                # active set (indices where conv is (almost) maximal)
                active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.full_like(conv, 1.0 / conv.size)

                grad_a = 2.0 * _cross_correlation(w, a, corr_len)

                # step size decays as 1/sqrt(t)
                eta = eta0 / np.sqrt(sub_it)
                a_new = a - eta * grad_a
                a = _project_simplex(a_new)

                # recompute objective
                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                val = 2.0 * n * max_c

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    _global_best_seq = best_seq
                    _global_best_val = best_val
                    sub_last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad iter={sub_it}] new best {val:.6f}")

                # stagnation break
                if time.time() - sub_last_improve > 9.0:
                    break
                if time.time() > deadline:
                    break

        # ------------------------------------------------------------------
        # Fine‑tuning of the current global best for this length (if applicable)
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] fine‑tuning global best (val={best_val:.6f})")
            a = best_seq.copy()
            logits = np.log(a + 1e-15)
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.045
            lam = 0.5
            lam_factor = 1.012
            last_improve = time.time()
            fine_iters = int(42000 * 1024 / n) + 6000

            for t in range(1, fine_iters + 1):
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum != 0.0:
                    w /= w_sum
                else:
                    w = np.full_like(w, 1.0 / w.size)

                grad_a = 2.0 * _cross_correlation(w, a, corr_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                lam = min(lam * lam_factor, 1e6)

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    _global_best_seq = best_seq
                    _global_best_val = best_val
                    last_improve = time.time()
                    print(f"[n={n} fine‑tune iter={t:05d}] new best {val:.6f}")

                if time.time() - last_improve > 10.0:
                    logits += (np.random.rand(n) - 0.5) * 5e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

    # ------------------------------------------------------------------
    # Final fallback – uniform vector (should never be needed)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = _obj(best_seq)

    print(f"\n=== Search completed: best value = {best_val:.6f} (length {len(best_seq)}) ===")
    return best_seq.tolist()
