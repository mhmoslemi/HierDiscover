def _optimize_subgradient(n, a0, target_sum, deadline, rng):
    a = np.maximum(a0, 1e-12) maybe.
    Project onto simplex: a = project_onto_simplex(a, target_sum)
    best_val = np.inf
    best_seq = a.copy()
    iter = 0
    last_improve_iter = 0
    step_initial = target_sum * 0.1 (maybe).
    while time.time() < deadline:
        iter += 1
        # conv
        conv = _full_convolution_fft(a, a)  # length 2n-1
        max_val = conv.max()
        if max_val < best_val:
            best_val = max_val
            best_seq = a.copy()
            last_improve_iter = iter
        # grad
        k = conv.argmax()
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2 * a[j[mask]]
        if k % 2 == 0:
            center = k // 2
            if 0 <= center < n:
                grad[center] = a[center]  # correct
        # step size
        step = step_initial / np.sqrt(iter)
        v = a - step * grad
        # project onto simplex
        a = _project_onto_simplex(v, target_sum)
        # occasional jitter to escape
        if iter % 500 == 0 and rng.random() < 0.2:
            a += rng.normal(scale=step*0.1, size=n)
            a = _project_onto_simplex(a, target_sum)
        # early stop if no improvement for many iters
        if iter - last_improve_iter > 5000:
            break
        # to avoid infinite loops
        if iter >= 20000:
            break
    # final evaluation
    score = evaluate_sequence(best_seq.tolist())
    return best_seq, score

"""
Propose a non‑negative step‑function sequence that minimises

    2 * n * max(conv(a,a)) / (sum(a)**2)

The implementation uses a simple projected sub‑gradient method on the
exact max‑convolution (instead of a smooth surrogate).  The algorithm
works on the simplex {a_i ≥ 0, Σ a_i = √(2 n)} by projecting each
gradient step onto this simplex.  A few seconds of jitter are added
periodically to escape plateaux.  Multiple random restarts of different
lengths are performed within the time budget, and the best sequence
found is returned.  If a previous construction (global variable
``height_sequence_1``) is available it is used as a warm‑start.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – full linear convolution using a power‑of‑2 FFT
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of two real vectors using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative values caused by numerical noise
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Projection of a vector onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum s."""
    # If already feasible, return unchanged (saves work for small vectors)
    if v.min() >= 0 and np.isclose(v.sum(), s):
        return v
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    # Find the last index where the condition holds
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: ensure exact sum (tiny adjustments)
    diff = s - w.sum()
    if abs(diff) > 1e-12:
        # distribute the residual uniformly over positive entries
        pos = w > 0
        if pos.any():
            w[pos] += diff / pos.sum()
        else:
            w += diff / n
    # Clip any possible overflow (>1000) because the outer evaluator caps it.
    np.minimum(w, 1000.0, out=w)
    return w


# ----------------------------------------------------------------------
# Gradient of the max‑convolution term (sub‑gradient)
# ----------------------------------------------------------------------
def _max_conv_subgradient(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Compute a sub‑gradient of max(conv(a,a)) at the current maximum index.
    The gradient w.r.t. a_i is:
        2 * a_{k-i}            if i != k-i
        a_i                    if i == k-i
    where k = max_idx (0‑based) and the pair (i, k-i) lies inside the vector.
    """
    n = a.shape[0]
    i = np.arange(n)
    j = max_idx - i
    mask = (j >= 0) & (j < n)

    grad = np.zeros_like(a)
    # generic case: i != j
    grad[mask] = 2.0 * a[j[mask]]
    # diagonal case (i == j) appears when max_idx is even and centre inside range
    if max_idx % 2 == 0:
        centre = max_idx // 2
        if 0 <= centre < n:
            grad[centre] = a[centre]    # replace the 2*a[centre] by a[centre]
    return grad


# ----------------------------------------------------------------------
# Simple projected sub‑gradient optimiser
# ----------------------------------------------------------------------
def _optimise_subgradient(
    n: int,
    a0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Optimise a by projected sub‑gradient descent on max(conv(a,a)).
    Returns the best sequence discovered and its official evaluation score.
    """
    # start from a feasible point on the simplex
    a = _project_onto_simplex(a0, target_sum)

    best_seq = a.copy()
    # Because target_sum = sqrt(2n) the objective is simply max(conv)
    best_val = float("inf")
    iteration = 0
    last_improve = 0

    # Step‑size schedule parameters
    step0 = target_sum * 0.1   # initial step (10 % of total mass)

    while time.time() < run_deadline:
        iteration += 1

        # full convolution and current max
        conv = _full_convolution_fft(a, a)          # length = 2n‑1
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_seq = a.copy()
            last_improve = iteration
            print(f"[{time.time():.2f}] n={n}  subgrad best = {cur_max:.8f}")

        # sub‑gradient of the active entry
        k = int(conv.argmax())
        grad = _max_conv_subgradient(a, k)

        # step size (decreasing like 1/√t)
        step = step0 / np.sqrt(iteration)

        # gradient step followed by projection onto simplex
        a_candidate = a - step * grad
        a = _project_onto_simplex(a_candidate, target_sum)

        # occasional jitter to avoid stagnation
        if iteration % 500 == 0 and rng.random() < 0.2:
            jitter = rng.normal(scale=step * 0.2, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)

        # early stop if stagnant for a long time
        if iteration - last_improve > 5000:
            break
        if iteration >= 20000:   # safety cap
            break

    # final official evaluation (the evaluator also caps values at 1000)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Random initial shape generators (all positive, later scaled)
# ----------------------------------------------------------------------
def _initial_shape_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random positive profile of length n."""
    typ = rng.choice(['power', 'exponential', 'linear', 'quadratic', 'sine', 'uniform'])
    idx = np.arange(n, dtype=float) + 1.0

    if typ == 'power':
        p = rng.uniform(0.1, 2.0)
        shape = idx ** (-p)
    elif typ == 'exponential':
        tau = rng.uniform(0.2, 0.9) * n
        shape = np.exp(-idx / tau)
    elif typ == 'linear':
        shape = np.linspace(1.0, 0.0, n)
    elif typ == 'quadratic':
        shape = (1.0 - (idx - 1) / (n - 1)) ** 2
    elif typ == 'sine':
        # half‑sine wave, strictly positive
        shape = np.sin(np.pi * (idx - 0.5) / n)
    else:  # uniform
        shape = np.ones(n, dtype=float)

    # small jitter to break exact symmetries
    shape += rng.random(n) * 1e-6
    shape = np.maximum(shape, 1e-12)
    return shape
