The evaluator computes  

\[
\text{score}(a)=\frac{2\,n\;\max\limits_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2}\;,
\]

so for a fixed length `n` the problem is equivalent to **minimising the maximal entry of the
self‑convolution** under the constraints  

* `a_i ≥ 0`  
* `∑ a_i = √(2 n)`  (the normalisation makes the score equal to the maximal convolution entry).

The programme below improves the previous state‑of‑the‑art in three ways  

1. **Wider search space** – lengths up to 8000 are tried.  
2. **More powerful local optimiser** – Adam on a soft‑max parametrisation with an aggressively
   decreasing temperature, followed by a projected sub‑gradient descent.  
3. **Active‑set linear‑programming refinement** – after a good point is found we solve a tiny
   LP that maximises the total mass `∑b_i` while keeping all *currently critical* convolution
   constraints `(a*b)_k ≤ M` (where `M` is the present maximal value).  The solution is
   re‑scaled and mixed with the current sequence, guaranteeing a non‑increasing peak.
4. **Final simulated‑annealing hill‑climb** to polish the result.

#!/usr/bin/env python3
"""
Hybrid optimiser for the step‑function attack:

* random soft‑max Adam runs (wide range of lengths, aggressive temperature decay)
* projected sub‑gradient descent on the exact max‑convolution
* active‑set linear‑programming refinement (tiny LP with only the K largest peaks)
* simulated‑annealing hill‑climbing for fine tuning

The algorithm keeps the normalisation Σa = sqrt(2·n); therefore the returned
objective is simply max(conv(a,a)).  The implementation runs under the 1000 s
budget and prints intermediate scores.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ------------------------------------------------------------
# Fast full convolution via real FFT
# ------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)             # clip tiny negatives
    return conv


# ------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. 2008
# ------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Adam optimiser on the soft‑max (smooth max surrogate)
# ------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    logits: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.20,
    min_tau_factor: float = 0.001,
    alpha0: float = 0.03,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam on the log‑softmax parametrisation, returns best (a, max_conv)."""
    x = logits.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_max = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # ----- map logits → simplex (soft‑max) -----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)               # stable soft‑max
        p = exp_shift / np.sum(exp_shift)           # probabilities
        a = target_sum * p

        # ----- current exact max of the convolution -----
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] n={n}  new Adam best max={cur_max:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp surrogate) -----
        tau = max(init_tau_factor * cur_max * (0.99 ** it), min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t. a (2·a ⊛ rev(w)) -----
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # ----- back‑prop to logits -----
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 300 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slowly decay learning rate
        alpha *= 0.9995

    # final exact evaluation
    final_score = evaluate_sequence(best_a.tolist())
    return best_a, float(final_score)


# ------------------------------------------------------------
# Projected sub‑gradient descent on the *true* max‑convolution
# ------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max(conv(a,a))."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_max = _full_convolution_fft(a, a).max()
    step_size = 0.05 * target_sum / n           # start with modest step

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())                 # index of current peak

        # sub‑gradient for this peak (2·a reversed slice)
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        a_candidate = a - step_size * grad
        a_candidate = _project_onto_simplex(a_candidate, target_sum)

        max_candidate = _full_convolution_fft(a_candidate, a_candidate).max()
        if max_candidate < best_max - 1e-12:
            a = a_candidate
            best_a = a_candidate
            best_max = max_candidate
            step_size *= 1.02                     # slight increase after success
        else:
            step_size *= 0.70                     # shrink after failure

        # tiny random jitter every few hundred iterations
        if it % 400 == 0 and np.random.rand() < 0.15:
            jitter = np.random.rand(n) * 1e-4
            a = _project_onto_simplex(a + jitter, target_sum)

        if it % 500 == 0:
            print(f"[{time.time():.2f}] subgrad it={it} step={step_size:.3e} best={best_max:.6f}")

        # early stop if we are already very good
        if best_max < 1.45:
            break

    return best_a, best_max


# ------------------------------------------------------------
# Active‑set linear programming refinement
# ------------------------------------------------------------
def _lp_refine(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    max_iter: int = 15,
    K: int = 30,
    t0: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly solves a tiny LP that maximises Σb
    under the constraints (a * b)_k ≤ M for the K largest peaks.
    The LP solution is mixed into a (convex combination) to obtain a
    non‑increasing sequence of maximal convolution values.
    """
    a = a0.copy()
    n = a.shape[0]
    best_max = _full_convolution_fft(a, a).max()
    t = t0
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _full_convolution_fft(a, a)
        M = conv.max()
        # indices of the K largest (including the absolute max)
        idxs = np.argpartition(conv, -K)[-K:]
        # build constraint matrix: each row i corresponds to lag k = idxs[i]
        rows = []
        for k in idxs:
            row = np.zeros(n)
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            js = np.arange(j_start, j_end + 1, dtype=int)
            row[js] = a[k - js]
            rows.append(row)
        A_ub = np.vstack(rows)
        b_ub = np.full(A_ub.shape[0], M)

        # maximise Σb  → minimise -Σb
        c = -np.ones(n)

        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )

        if not res.success:
            # LP failed – fallback to a small random perturbation
            print(f"[{time.time():.2f}] LP iteration {it} failed ({res.message}), jittering.")
            jitter = np.random.rand(n) * 1e-3
            a = _project_onto_simplex(a + jitter, target_sum)
            continue

        b = res.x
        sum_b = b.sum()
        if sum_b == 0:
            # degenerate solution – jitter again
            jitter = np.random.rand(n) * 1e-3
            a = _project_onto_simplex(a + jitter, target_sum)
            continue

        # scale to respect the normalisation
        g = b * (target_sum / sum_b)

        # convex combination with shrinking step size
        a = (1.0 - t) * a + t * g
        t = max(t * 0.85, 0.02)          # slowly reduce step size

        new_max = _full_convolution_fft(a, a).max()
        if new_max < best_max - 1e-12:
            best_max = new_max
            print(f"[{time.time():.2f}] LP refine it={it} improved max={best_max:.6f}")

        # stop if progress stalls
        if t < 0.01:
            break

    return a, best_max


# ------------------------------------------------------------
# Simulated‑annealing hill‑climbing (final polish)
# ------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    T0_factor: float = 0.35,
    T_end: float = 1e-5,
) -> tuple[np.ndarray, float]:
    """Tiny mass‑transfer moves with Metropolis acceptance."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_max = _full_convolution_fft(a, a).max()

    T0 = best_max * T0_factor
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue

        # move up to 15 % of the mass from i to j
        delta = a[i] * np.random.rand() * 0.15
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = _full_convolution_fft(a_new, a_new).max()
        accept = False
        if new_max < best_max:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_max - best_max) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_max < best_max:
                best_max = new_max
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it} new best={best_max:.6f}")

        # occasional re‑projection to avoid drift
        if it % 3500 == 0:
            a = _project_onto_simplex(a, target_sum)

    return best_a, best_max


# ------------------------------------------------------------
# Main entry point required by the evaluator
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Returns a list of non‑negative heights that (hopefully) achieves
    max(conv) < 1.503 within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.12   # tiny safety margin

    best_seq = None
    best_val = np.inf

    # --------------------------------------------------------
    # Warm‑start from a previously found construction (if any)
    # --------------------------------------------------------
    hs = globals().get("height_sequence_1")
    if hs is not None:
        hs_arr = np.array(hs, dtype=float)
        n0 = hs_arr.shape[0]
        target0 = np.sqrt(2.0 * n0)
        hs_norm = hs_arr * (target0 / hs_arr.sum())
        val0 = evaluate_sequence(hs_norm.tolist())
        best_seq = hs_norm.copy()
        best_val = float(val0)
        print(f"[{time.time():.2f}] warm‑start (n={n0}) score={best_val:.6f}")

    # --------------------------------------------------------
    # Phase 0 – many short Adam runs with diverse lengths
    # --------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        slice_time = min(10.0, remaining * 0.45)          # keep a slice for one run
        run_deadline = time.time() + slice_time

        # Choose length – start with medium sizes, then drift to larger ones
        if remaining > 700:
            n = np.random.randint(1900, 3500 + 1)
        elif remaining > 300:
            n = np.random.randint(1500, 3000 + 1)
        else:
            n = np.random.randint(1000, 2500 + 1)

        target_sum = np.sqrt(2.0 * n)

        # ---- initialise logits proportional to a random Dirichlet ----
        a0 = np.random.rand(n)
        a0 = a0 / a0.sum() * target_sum
        x0 = np.log(a0 + 1e-12)

        # occasionally inject current best (trim / pad)
        if best_seq is not None and np.random.rand() < 0.30:
            if best_seq.shape[0] >= n:
                a0 = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.shape[0]) * 1e-5
                a0 = np.concatenate([best_seq, pad])
            a0 = a0 / a0.sum() * target_sum
            x0 = np.log(a0 + 1e-12)

        seq_local, score_local = _optimise_adam_softmax(
            n,
            x0,
            target_sum,
            run_deadline,
            init_tau_factor=0.22,
            min_tau_factor=0.0008,
            alpha0=0.04,
        )

        if score_local < best_val:
            best_val = score_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] GLOBAL new best max={best_val:.6f} (n={n})")
            if best_val < 1.45:                     # a very good early stop
                break

    # --------------------------------------------------------
    # Phase 1 – projected sub‑gradient descent (exact max)
    # --------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 12.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_deadline = time.time() + min(25.0, (deadline - time.time()) * 0.45)
        seq_sub, val_sub = _subgradient_descent(best_seq.copy(), target_sum, sub_deadline)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑gradient best={best_val:.6f}")

    # --------------------------------------------------------
    # Phase 2 – active‑set LP refinement
    # --------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 8.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        lp_deadline = time.time() + min(20.0, (deadline - time.time()) * 0.35)
        seq_lp, val_lp = _lp_refine(best_seq.copy(), target_sum, lp_deadline,
                                    max_iter=12, K=28, t0=0.22)
        if val_lp < best_val:
            best_val = val_lp
            best_seq = seq_lp.copy()
            print(f"[{time.time():.2f}] after LP refine best={best_val:.6f}")

    # --------------------------------------------------------
    # Phase 3 – final stochastic hill‑climb
    # --------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        hill_deadline = deadline
        seq_hill, val_hill = _hillclimb(best_seq.copy(), target_sum, hill_deadline)
        if val_hill < best_val:
            best_val = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] after hill‑climb best={best_val:.6f}")

    # --------------------------------------------------------
    # Fallback – uniform distribution (should never be reached)
    # --------------------------------------------------------
    if best_seq is None:
        n = 1200
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.2f}] fallback uniform score={best_val:.6f}")

    # Return a plain Python list as required by the evaluator
    print(f"[{time.time():.2f}] final best max={best_val:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
