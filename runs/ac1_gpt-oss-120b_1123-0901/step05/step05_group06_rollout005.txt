def _pgd_optimize(a0, n, deadline, max_iters, best_seq, best_score, rng):
    target_sum = sqrt(2 n)
    a = a0.copy()
    # maybe ensure sum target
    a = a * target_sum / a.sum()
    # step size schedule: initial lr = target_sum * 0.1?
    for it in range(max_iters):
        if time.time() > deadline:
            break

        # compute conv via fft
        Nfft = 1 << ((3*n-2).bit_length())
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[:2*n-1]
        M = conv.max()
        # track best
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD] n {n} iter {it} new best {best_score}")

        # compute active set
        eps_active = 1e-12
        mask = conv >= M - eps_active
        cnt = mask.sum()
        if cnt == 0:
            # fallback
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = mask.astype(float) / cnt

        # compute rev gradient
        rev = _weighted_rev(a, w, Nfft)
        grad = 2 * rev

        # gradient step
        # adapt step size: lr = step0 / math.sqrt(it+1)
        lr0 = 0.1 * target_sum  # approximate magnitude; tune.
        lr = lr0 / math.sqrt(it+1)

        a_new = a - lr * grad

        # project onto simplex
        a = _project_simplex(a_new, target_sum)

        # occasional jitter
        if it % 2000 == 0:
            # potentially random reset of a few coordinates
            jitter = rng.normal(scale=0.01*target_sum/n, size=n)
            a = np.maximum(a + jitter, 0)
            a = _project_simplex(a, target_sum)

    return best_seq, best_score

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

The algorithm combines two complementary optimisation schemes:

1. **Projected Gradient Descent (PGD)** on the exact max‑convolution.
   The sub‑gradient of the objective is obtained from the active
   (max‑attaining) convolution indices and the reverse convolution is
   computed via a single FFT per iteration.  After each gradient step the
   vector is projected onto the simplex {a_i ≥ 0, Σ a_i = √(2 n)}.
   An annealing schedule for the step size and occasional random jitter
   help escape plateaux.

2. **Exponentiated‑gradient Adam** (the previous approach) is kept as a
   second, independent launch.  The best result from both methods is
   retained.

Finally a cheap stochastic refinement (mass transfers with simulated
annealing) is applied to the global best sequence.

The whole search respects the 1000 s time budget, dynamically
allocating iterations to the different phases.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities ------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0, Σ x_i = target_sum}.
    Implements the algorithm of Duchi et al. (2008) in O(n log n).
    """
    if target_sum <= 0:
        return np.zeros_like(v)

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)

    w = np.maximum(v - theta, 0.0)
    return w


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute the weighted reverse convolution needed for the sub‑gradient:

        rev_i = Σ_k w_k * a_{k-i}   (indices outside the array are ignored)

    The calculation is performed with FFTs (O(N log N)).
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)   # reverse w before FFT
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)

    # keep the linear part of the convolution
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]   # length = 3 n‑2
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Exact max of a * a (linear convolution) using one FFT."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Generate a small, diverse set of feasible start vectors."""
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0
             - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    init.append(black / black.sum() * target_sum)

    # 7) small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    # 8) triangular shape (peak in the centre)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    init.append(tri / tri.sum() * target_sum)

    # 9) sparse random spikes (helps explore highly non‑uniform shapes)
    k = max(2, min(15, n // 100))
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k)
    vals = vals / vals.sum() * target_sum
    sparse = np.zeros(n)
    sparse[idx] = vals
    init.append(sparse)

    return init

def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected Gradient Descent on the exact max‑convolution.
    Returns the best vector observed and its exact objective value.
    """
    target_sum = math.sqrt(2.0 * n)

    # Initialise on the simplex
    a = a0.copy()
    a = a * target_sum / a.sum()

    # FFT size – must accommodate linear convolution of length 3 n‑2
    Nfft = 1 << ((3 * n - 2).bit_length())

    # Simple decreasing step‑size schedule
    step0 = 0.001 * target_sum          # base step (≈ 0.05 % of total mass)
    eps_active = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- convolution & current max -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- keep global best -----
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[PGD len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

        # ----- active set (hard max) -----
        mask = conv >= M - eps_active
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = mask.astype(float) / float(cnt)

        # ----- weighted reverse convolution (sub‑gradient) -----
        # reuse A_fft for both conv and rev
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[: len(w) + n - 1]
        idx_start = 2 * n - 2
        rev = conv_rev[idx_start - np.arange(n)]

        g = 2.0 * rev                     # ∇_a max(a∗a)

        # ----- gradient step on the simplex -----
        lr = step0 / math.sqrt(it)        # simple annealing
        a_tmp = a - lr * g
        a = _project_onto_simplex(a_tmp, target_sum)

        # ----- occasional jitter to avoid stagnation -----
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.01 * target_sum / n, size=n)
            a = _project_onto_simplex(a + jitter, target_sum)

    return best_seq, best_score


def _exp_gradient_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑its (exponentiated‑gradient) using a soft‑max
    approximation of the max‑convolution (identical to the previous
    implementation, but re‑using the FFT of a to avoid a duplicate FFT).
    Returns the best vector seen in this run together with its exact
    objective value.
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise logits from the start vector (add tiny epsilon to avoid log(0))
    eps = 1e-16
    y = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.08                      # base learning rate for logits
    jitter_interval = 3000

    # temperature (soft‑max) schedule
    tau_factor = 0.30               # τ₀ = τ_factor·M
    decay_rate = max_iters / 8.0    # exponential decay factor

    # FFT size – must be large enough for linear conv of length 3 n‑2
    Nfft = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- current vector (softmax of logits) -----
        logits = y - np.max(y)                 # shift for stability
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()      # lies on the probability simplex
        a = target_sum * p                     # respects Σa = target_sum, a_i≥0

        # ----- FFT of a (re‑used) -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- keep global best -----
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[Adam len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

        # ----- soft‑max weights for the max -----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
        else:
            maxc = conv.max()
            ex = np.exp((conv - maxc) / tau)
            sum_ex = ex.sum()
            w = ex / sum_ex

        # ----- weighted reverse convolution (gradient of smoothed max) -----
        W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
        conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
        conv_rev = conv_rev[: len(w) + n - 1]
        idx_start = 2 * n - 2
        rev = conv_rev[idx_start - np.arange(n)]

        # ----- gradient wrt a (2·rev) -----
        g_a = 2.0 * rev

        # ----- chain rule to logits (mirror‑descent on the simplex) -----
        avg = np.dot(p, g_a)                 # ⟨p, g_a⟩
        g_y = target_sum * p * (g_a - avg)   # ∂f/∂y_i

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # occasional jitter
        if it % jitter_interval == 0:
            jitter = rng.normal(scale=0.01, size=n)
            y += jitter

    return best_seq, best_score


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on the simplex:
    repeatedly move a small fraction of mass from a random index i to j.
    Accept improvements always; accept worsening moves with a decaying
    simulated‑annealing probability.
    """
    target_sum = a.sum()                     # already √(2 n)
    cur_seq = a.copy()
    cur_score = best_score

    # FFT size for fast max‑convolution
    Nfft = 1 << ((2 * n - 1).bit_length())

    # annealing schedule
    max_steps = 200_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        # transfer a random fraction (1 % – 5 %) of the source mass
        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # compute new max‑convolution
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        # deterministic improvement
        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (plain Python list) found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load any previously discovered construction if available
    base = None
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Choose a single promising length (the historically good size)
    if base is not None:
        n0 = int(base.shape[0])
    else:
        n0 = 2100               # fallback from earlier runs

    # Explore a small set of nearby lengths – larger sizes often give similar
    # values but cost more time.
    length_factors = [0.9, 1.0, 1.1, 1.2]
    lengths = sorted({max(300, min(int(n0 * f), 8000)) for f in length_factors})

    best_seq = None
    best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # ---- generate diverse starting points --------------------------------
        init_vectors = _make_initial_vectors(n, target_sum, rng)

        if base is not None and base.shape[0] != n:
            # linear interpolation of the previous best to the new size
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
            interp = np.maximum(interp, 0.0)
            interp = interp / interp.sum() * target_sum
            init_vectors.append(interp)

        # ---- per‑start optimisation -----------------------------------------
        # iteration budget: more for small n, never less than 150 k
        max_iters = int(2_500_000 * 2100 / n)
        max_iters = max(150_000, max_iters)

        print(f"[run] length={n}, max_iters≈{max_iters}, target_sum={target_sum:.3f}")

        for idx, a0 in enumerate(init_vectors):
            if time.time() >= deadline:
                break
            print(f"  start {idx+1}/{len(init_vectors)} for length {n}")

            # Run Adam‑based exponentiated gradient optimisation
            best_seq, best_score = _exp_gradient_optimize(
                a0=a0.copy(),
                n=n,
                deadline=deadline,
                max_iters=max_iters,
                best_seq=best_seq if best_seq is not None else a0,
                best_score=best_score,
                rng=rng,
            )

            # Run Projected Gradient Descent from the same start (independent)
            best_seq, best_score = _pgd_optimize(
                a0=a0.copy(),
                n=n,
                deadline=deadline,
                max_iters=max_iters,
                best_seq=best_seq,
                best_score=best_score,
                rng=rng,
            )

    # ------------------------------------------------------------------
    # Local stochastic refinement on the global best (if time permits)
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # give at most half the remaining budget to refinement (capped at 30 s)
        refine_deadline = time.time() + min(remaining * 0.5, 30.0)
        best_seq, best_score = _local_refine(
            a=best_seq,
            n=best_seq.shape[0],
            deadline=refine_deadline,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Fallback if everything failed (unlikely)
    if best_seq is None:
        n = n0
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = float('inf')
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
