"""
Enhanced step‑function optimiser.

The optimiser builds a diverse pool of seeds (uniform, Dirichlet,
power‑law edge shapes, edge‑Gaussian shapes and a sine‑squared bump) and
optimises each seed with a long Adam phase on a smooth‑max surrogate of the
ℓ∞‑norm of the self‑convolution.  Most of the wall‑clock budget is spent on a
medium size problem (n≈8192); this gives cheap but powerful gradient steps.
After a best candidate is found it is up‑sampled to the full length
16384 and refined again.  Finally a cheap stochastic local‑search (random
perturbations + a few Adam steps) is performed until the time budget is
exhausted.  The routine respects the 1000 s budget, prints progress and
returns a plain Python list of non‑negative heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based, power‑of‑two lengths)
# ----------------------------------------------------------------------
def _fft_len(at_least: int) -> int:
    """Smallest power of two ≥ at_least."""
    return 1 << int(np.ceil(np.log2(at_least)))


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = _fft_len(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


# ----------------------------------------------------------------------
# Objective helper
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact value of 2·n·max(a*a)/(sum(a)²).
    a must be non‑negative; the function does not normalise it.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    lam_start: float = 0.5,
    lam_factor: float = 1.01,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps_adam: float = 1e-8,
    stagnation_secs: float = 5.0,
) -> tuple[np.ndarray, float]:
    """
    Perform Adam optimisation starting from a0 (already normalised to sum=1).
    Returns the best vector seen and its exact objective value.
    """
    a0 = np.maximum(a0, 1e-16)                # avoid log(0)
    a0 = a0 / a0.sum()
    logits = np.log(a0)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    conv_fft_len = _fft_len(2 * n - 1)
    corr_fft_len = _fft_len(3 * n - 2)

    lam = lam_start
    best_a = None
    best_val = float("inf")
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        # ---- soft‑max (project onto simplex) ----------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()            # a is on the simplex

        # ---- convolution and smooth‑max weights --------
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))       # log‑sum‑exp trick
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of smooth‑max w.r.t. a ----------
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # ---- back‑propagation through the soft‑max -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update on logits --------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature annealing --------------------
        lam = min(lam * lam_factor, 1e7)

        # ---- objective (exact) -----------------------
        val = 2.0 * n * max_c          # sum(a)=1 ⇒ denominator = 1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} it={it:05d}] new best {val:.6f}")

        # ---- stagnation handling ---------------------
        if time.time() - last_improve > stagnation_secs:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

        # ---- budget check ----------------------------
        if time.time() > deadline:
            break

    # If the inner loop finished without improving, fall back to the last a
    if best_a is None:
        best_a = a.copy()
        best_val = _objective(best_a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Sub‑gradient refinement on the true max (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _subgrad_refine(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2000,
    eta: float = 0.6,
    stagnation_secs: float = 5.0,
) -> tuple[np.ndarray, float]:
    """
    Perform a few exponentiated‑gradient steps on the true ℓ∞‑norm.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() == 0.0:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    conv_fft_len = _fft_len(2 * n - 1)
    corr_fft_len = _fft_len(3 * n - 2)

    best_a = a.copy()
    best_val = _objective(a)
    last_improve = time.time()

    for it in range(max_iters):
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} sub‑grad it={it}] new best {val:.6f}")

        if time.time() - last_improve > stagnation_secs:
            break
        if time.time() > deadline:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Stochastic local search around the current best
# ----------------------------------------------------------------------
def _local_search(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_trials: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly perturb the current best vector and run a short Adam phase.
    Accepts improvements and gradually shrinks the perturbation scale.
    """
    best_a = a0.copy()
    best_val = _objective(best_a)
    sigma = 0.2                     # initial perturbation amplitude
    trial = 0

    while time.time() < deadline and trial < max_trials:
        trial += 1
        # generate a random perturbation (additive Gaussian on logits)
        pert = np.random.randn(n) * sigma
        a_pert = best_a + pert
        a_pert = np.maximum(a_pert, 0.0)
        if a_pert.sum() == 0.0:
            a_pert = np.random.dirichlet(np.ones(n))
        a_pert /= a_pert.sum()

        # a brief Adam run (few hundred iterations)
        max_it = max(200, int(800 * 1024 / n))
        a_opt, val_opt = _adam_optimize(
            a_pert,
            n,
            deadline,
            max_iters=max_it,
            lam_start=0.3,
            lam_factor=1.02,
            lr=0.03,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            sigma = 0.2                 # reset amplitude after improvement
            print(f"[n={n} local‑search trial={trial}] improvement {val_opt:.6f}")
        else:
            sigma *= 0.99               # slowly reduce the step size

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a*a)/(sum(a)²).

    The routine respects the given wall‑clock budget and returns the best
    sequence discovered (as a plain Python list).  The method combines
    a rich set of initial families, long Adam optimisation on a smooth‑max
    surrogate, a short exponentiated‑gradient refinement and a final
    stochastic local‑search.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # tiny safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – coarse work on an intermediate size (n≈8192)
    # ------------------------------------------------------------------
    # we concentrate most of the budget on this size; earlier (smaller)
    # sizes are cheap warm‑starts.
    work_lengths = [256, 512, 1024, 2048, 4096, 8192]

    for n in work_lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = _fft_len(2 * n - 1)
        corr_fft_len = _fft_len(3 * n - 2)

        # --------------------------------------------------------------
        # Build a small but diverse pool of initial candidates
        # --------------------------------------------------------------
        cand_pool = []

        # 1) uniform
        cand_pool.append(np.full(n, 1.0 / n, dtype=np.float64))

        # 2) random Dirichlet
        cand_pool.append(np.random.dirichlet(np.ones(n)).astype(np.float64))

        # 3) power‑law edge families (α negative)
        idx = np.arange(1, n + 1, dtype=np.float64)
        prod = idx * (n - idx + 1.0)
        for alpha in np.linspace(-2.0, -0.4, 5):
            shape = np.where(prod > 0, prod ** alpha, 0.0)
            if shape.sum() > 0:
                shape /= shape.sum()
                cand_pool.append(shape.astype(np.float64))

        # 4) two‑Gaussian edge shapes
        i = np.arange(n, dtype=np.float64)
        for gamma in [0.1, 0.3, 0.8, 1.5]:
            shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            shape /= shape.sum()
            cand_pool.append(shape.astype(np.float64))

        # 5) smooth sine‑squared bump
        shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
        shape /= shape.sum()
        cand_pool.append(shape.astype(np.float64))

        # optional warm‑start from a previous global best of the same length
        if best_seq is not None and len(best_seq) == n:
            cand_pool.append(np.asarray(best_seq, dtype=np.float64))

        # --------------------------------------------------------------
        # Optimise each candidate with a long Adam phase + sub‑gradient clean‑up
        # --------------------------------------------------------------
        for cid, a0 in enumerate(cand_pool):
            if time.time() > deadline:
                break

            # estimate iteration budget (more for smaller n)
            max_iters = int(25000 * 1024 / n) + 3000   # a bit larger than before

            a_opt, val_opt = _adam_optimize(
                a0,
                n,
                deadline,
                max_iters=max_iters,
                lam_start=0.5,
                lam_factor=1.01,
                lr=0.05,
            )
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"==> Global best improved to {val_opt:.6f} (len={n})")

            # short sub‑gradient refinement
            a_ref, val_ref = _subgrad_refine(
                a_opt,
                n,
                deadline,
                max_iters=1500,
                eta=0.6,
            )
            if val_ref < best_val:
                best_val = val_ref
                best_seq = a_ref.copy()
                print(f"==> Global best (post‑subgrad) {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – up‑sample to the full length (16384) and fine‑tune
    # ------------------------------------------------------------------
    final_n = 16384
    if best_seq is not None and len(best_seq) < final_n:
        # linear interpolation to the larger grid
        x_old = np.linspace(0.0, 1.0, len(best_seq))
        x_new = np.linspace(0.0, 1.0, final_n)
        a_up = np.interp(x_new, x_old, best_seq)
        a_up = np.maximum(a_up, 0.0)
        if a_up.sum() == 0.0:
            a_up = np.random.dirichlet(np.ones(final_n))
        a_up /= a_up.sum()
    elif best_seq is not None and len(best_seq) == final_n:
        a_up = best_seq.copy()
    else:
        a_up = np.full(final_n, 1.0 / final_n, dtype=np.float64)

    if time.time() < deadline:
        # a moderate Adam run on the full size
        max_iters_full = int(12000 * 1024 / final_n) + 4000   # ≈ 5000
        a_up_opt, val_up_opt = _adam_optimize(
            a_up,
            final_n,
            deadline,
            max_iters=max_iters_full,
            lam_start=0.4,
            lam_factor=1.015,
            lr=0.04,
        )
        if val_up_opt < best_val:
            best_val = val_up_opt
            best_seq = a_up_opt.copy()
            print(f"==> After up‑sample Adam best {val_up_opt:.6f}")

        # a short sub‑gradient clean‑up
        a_up_ref, val_up_ref = _subgrad_refine(
            a_up_opt,
            final_n,
            deadline,
            max_iters=1200,
            eta=0.55,
        )
        if val_up_ref < best_val:
            best_val = val_up_ref
            best_seq = a_up_ref.copy()
            print(f"==> After up‑sample sub‑grad best {val_up_ref:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – stochastic local search until the clock runs out
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a_final, val_final = _local_search(best_seq, final_n, deadline)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final.copy()
            print(f"==> After local search best {val_final:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Convert to plain Python list as required by the external evaluator
    return best_seq.tolist()
