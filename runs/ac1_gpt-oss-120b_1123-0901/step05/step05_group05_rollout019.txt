def _simulated_annealing_optimize(n, deadline, init_seq=None):
    # initial a:
    if init_seq is not None:
        a = np.array(init_seq, dtype=float)
        # ensure length n
        if a.shape[0] != n:
            a = np.resize(a, n)   # but careful; just cut/pad
    else:
        # Dirichlet with random concentration
        alpha = np.random.uniform(0.1, 5.0)
        a = np.random.gamma(alpha, 1.0, size=n)
    # normalize sum to 1.0
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0/n)
    else:
        a = a / a_sum

    # compute initial conv and score
    conv = _full_convolution_fft(a, a)
    max_conv = conv.max()
    cur_score = 2.0 * n * max_conv  # sum=1 => denominator = 1
    # But need to divide by sum**2 = 1, so same.

    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    total_time = deadline - start_time
    if total_time <= 0:
        return best_a, best_score

    # parameters
    init_temp_factor = 0.4   # relative to initial objective
    T0 = max(cur_score * init_temp_factor, 1e-8)
    Tmin = max(cur_score * 0.001, 1e-12)
    init_move_scale = 0.5
    min_move_scale = 0.001

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        # temperature schedule exponential decay
        T = T0 * (1 - frac) + Tmin
        move_scale = init_move_scale * (1 - frac) + min_move_scale

        # choose random i with a[i] > 0
        i = np.random.randint(n)
        # if a[i] small, maybe skip selecting
        if a[i] < 1e-12:
            continue
        # choose random j != i
        j = np.random.randint(n-1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * move_scale
        if delta <= 0:
            continue

        # compute candidate conv
        conv_candidate = conv.copy()
        # perform updates
        conv_candidate[i:i+n] -= 2.0 * delta * a
        conv_candidate[j:j+n] += 2.0 * delta * a
        conv_candidate[2*i] += delta * delta
        conv_candidate[2*j] += delta * delta
        conv_candidate[i + j] -= 2.0 * delta * delta
        # clip negative:
        # using np.maximum
        np.maximum(conv_candidate, 0.0, out=conv_candidate)

        # compute new score
        max_conv_cand = conv_candidate.max()
        new_score = 2.0 * n * max_conv_cand  # sum=1

        # acceptance
        if new_score < cur_score:
            accept = True
        else:
            # probability
            accept = np.random.rand() < np.exp(-(new_score - cur_score) / (T + 1e-12))

        if accept:
            # update a and conv
            a[i] -= delta
            a[j] += delta
            # Numerical error: maybe a[i] becomes negative due to rounding; clip:
            if a[i] < 0:
                a[i] = 0.0
            conv = conv_candidate
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                # optionally log
                print(f"[{time.time():.2f}] SA n={n} new best={best_score:.6f}")

        # occasional perturbation to avoid stagnation
        if np.random.rand() < 0.001:
            # random jitter
            jitter = (np.random.rand(n) - 0.5) * 1e-6
            a += jitter
            # ensure nonnegativity and renormalize
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum == 0:
                a = np.full(n, 1.0/n)
            else:
                a /= a_sum
            conv = _full_convolution_fft(a, a)
            cur_score = 2.0 * n * conv.max()
            # maybe update best
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    # final evaluation using evaluate_sequence for exactness
    final_score = evaluate_sequence(best_a.tolist())
    return best_a, float(final_score)
