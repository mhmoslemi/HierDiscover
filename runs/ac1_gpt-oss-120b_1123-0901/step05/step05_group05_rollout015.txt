"""
candidate_proposer.py

Hybrid optimisation for the step‑function problem

    minimise  2 n · max(conv(a,a)) / (∑a)²

The optimiser works in three stages:

1. **Exploratory Adam runs** – the coefficients are represented through a
   soft‑max of unconstrained logits.  A smooth log‑sum‑exp surrogate
   of the max‑convolution is minimised with Adam.  Many short runs with
   different lengths (2000 – 8000) and diverse initialisations are
   performed; the best result of each run is kept.

2. **Adaptive projected sub‑gradient descent** – starting from the best
   Adam candidate, the exact sub‑gradient of the *true* max‑convolution
   is computed.  A back‑tracking line‑search is used to guarantee a
   monotone decrease, and each step is projected back onto the simplex
   `{a_i ≥ 0, ∑a_i = √(2 n)}` with the O(n log n) algorithm of Duchi et al.
   This stage usually removes the remaining few hundredths of the objective.

3. **Fine‑grained stochastic hill‑climbing** – tiny mass transfers between
   random positions are accepted if they lower the peak or with a
   Metropolis probability that slowly cools.  This refines the solution
   to the final tolerance.

All three stages share a global best solution.  The code respects a
user‑supplied time budget (default 1000 s) and returns a plain Python
list that can be fed to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)            # eliminate tiny negatives
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {w ≥ 0, Σw = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection of v onto the simplex of radius s (Duchi et al. 2008)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimisation of the smooth‑max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    logits: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.20,
    min_tau_factor: float = 0.001,
    alpha0: float = 0.03,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the soft‑max parametrisation."""
    x = logits.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # ----- map logits → simplex -----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)                 # stable soft‑max
        p = exp_shift / np.sum(exp_shift)             # probabilities (Σp = 1)
        a = target_sum * p

        # ----- exact max‑convolution (for logging only) -----
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] n={n}   Adam best max={cur_max:.6f}")

        # ----- smooth‑max surrogate (log‑sum‑exp) -----
        tau = max(init_tau_factor * cur_max * (0.99 ** it),
                  min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # ----- gradient wrt a (2 · a ⊛ w_rev) -----
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)               # positions matching Σ a_j w_{i+j}
        grad_a = 2.0 * conv2[idx]

        # ----- back‑prop to logits -----
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slowly decay the base learning rate
        alpha *= 0.9995

    # final exact evaluation of the best soft‑max point
    final_score = evaluate_sequence(best_a.tolist())
    return best_a, float(final_score)


# ----------------------------------------------------------------------
# Adaptive projected sub‑gradient descent on the *true* max‑convolution
# ----------------------------------------------------------------------
def _adaptive_subgradient(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent with a back‑tracking line‑search.
    Returns the best sequence encountered and its exact evaluation.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # initial step size – roughly a few percent of the average entry
    step_size = 0.5 * target_sum / n

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())               # shift with maximal value

        # sub‑gradient of conv_k w.r.t a
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # try a few back‑tracking steps
        improved = False
        eta = step_size
        for _ in range(10):
            a_candidate = a - eta * grad
            a_candidate = _project_onto_simplex(a_candidate, target_sum)

            new_val = _full_convolution_fft(a_candidate, a_candidate).max()
            if new_val < best_val - 1e-12:
                a = a_candidate
                best_a = a_candidate.copy()
                best_val = new_val
                improved = True
                break
            eta *= 0.5          # shrink step if no improvement

        if not improved:
            step_size *= 0.7    # aggressive reduction on failure
        else:
            step_size *= 1.02   # modest increase after a successful step

        # occasional jitter to stay away from plateaus
        if it % 300 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(n) * 1e-4
            a = _project_onto_simplex(a + jitter, target_sum)

        # low‑frequency progress report
        if it % 500 == 0:
            print(
                f"[{time.time():.2f}] subgrad it={it} step={step_size:.3e} best={best_val:.6f}"
            )
        # early stop if we already beat the target
        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Local stochastic search that transfers a tiny amount of mass between
    two indices while respecting the simplex constraint.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # temperature schedule
    T0 = best_val * 0.3
    T_end = 1e-4
    start = time.time()
    it = 0

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue

        # move at most 10 % of the source mass
        delta = a[i] * np.random.rand() * 0.10
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()

        accept = False
        if new_val < best_val:
            accept = True
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")

        # periodic re‑normalisation to avoid numerical drift
        if it % 2000 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val
