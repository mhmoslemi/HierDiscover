"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Length selection** – try a handful of lengths up to 8000 (including the
   historic length 2100).  The total time budget is split evenly between the
   lengths.

2. **Diverse starts** – for each length a set of simple analytic shapes is
   generated (uniform, Dirichlet, exponential, cosine‑U, power‑law, etc.) as
   well as a few low‑alpha Dirichlet draws to obtain sparse candidates.
   If a historic construction is available it is interpolated to the current
   size and added to the pool.

3. **Smoothed‑max Adam on the simplex** – each start vector is optimised with
   an exponentiated‑gradient (soft‑max) parametrisation, i.e. we keep logits
   `y` and use `p = softmax(y)` to obtain a probability vector.
   The objective is the *maximum* entry of the linear convolution `a*a`.
   A temperature `τ` gives a smooth soft‑max approximation; `τ` decays
   exponentially during the run.  The gradient of the smoothed max is the
   weighted reverse convolution, computed with three‑FFT’s (O(n log n)).
   Adam updates are applied to the logits.  Hyper‑parameters (initial step‑
   size, τ‑factor, jitter frequency) are randomly sampled for each run to
   increase diversity.

4. **Global best tracking** – whenever a run finds a lower peak the vector
   and its peak value are stored.

5. **Stochastic refinement** – after the Adam phase a cheap hill‑climbing /
   simulated‑annealing routine repeatedly moves a small fraction of mass from
   a random source index to a random destination index.  Improvements are
   accepted always; worsening moves are accepted with a linearly cooling
   temperature.

6. **Return** – the best vector (as a plain Python list) found before the
   deadline is returned.  Since the sum is normalised to ``sqrt(2 n)`` the
   returned objective value is exactly the peak of the convolution, i.e.
   ``evaluate_sequence`` returns that number.

The implementation is fully self‑contained, uses only NumPy and the
provided ``evaluate_sequence`` function and respects the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers ---------------------------------------------------

def _conv_max(a: np.ndarray, Nfft: int) -> float:
    """Maximum value of the linear convolution a*a using an FFT of length Nfft."""
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _softmax_weights(conv: np.ndarray, tau: float):
    """Smooth max via soft‑max on the convolution values."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    s = ex.sum()
    w = ex / s
    # smooth approximation of max
    L = maxc + tau * math.log(s)
    return w, L


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k * a_{k-i}  (out‑of‑range indices ignored)
    using FFTs.  This is the gradient of the smoothed max–convolution.
    """
    A = np.fft.rfft(a, n=Nfft)
    W_rev = np.fft.rfft(w[::-1], n=Nfft)          # reverse w before FFT
    conv_rev = np.fft.irfft(W_rev * A, n=Nfft)
    conv_rev = conv_rev[: w.size + a.size - 1]    # linear part, length = 3n‑2
    idx_start = 2 * a.size - 2
    rev = conv_rev[idx_start - np.arange(a.size)]
    return rev


# ----------------------------------------------------------------------
# Initial vector generator ----------------------------------------------

def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Return a list of diverse start vectors (all normalised to ``target_sum``)."""
    init = []

    # 0) uniform
    init.append(np.full(n, target_sum / n))

    # 1) Dirichlet (α = 1)
    g = rng.gamma(1.0, 1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 2) Sparse Dirichlet (α in [0.1, 0.5])
    alpha = rng.uniform(0.1, 0.5)
    g = rng.gamma(alpha, 1.0, size=n)
    if g.sum() > 0:
        init.append(g / g.sum() * target_sum)

    # 3) Exponential decay (forward)
    rate = rng.uniform(0.001, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) Exponential decay (reverse)
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Cosine‑U shape   a_i ∝ 1 + cos(π i/(n‑1))  (large at ends, small in centre)
    u = 1.0 + np.cos(np.pi * np.arange(n) / (n - 1))
    init.append(u / u.sum() * target_sum)

    # 6) Power‑law   a_i ∝ i^{‑p} + (n‑i)^{‑p}
    p = rng.uniform(0.2, 2.0)
    idx = np.arange(1, n + 1, dtype=float)
    pw = (idx ** (-p) + idx[::-1] ** (-p))
    init.append(pw / pw.sum() * target_sum)

    # 7) Small random perturbation of uniform
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.02 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    pert = pert / pert.sum() * target_sum
    init.append(pert)

    return init


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max (smoothed) max‑convolution -------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the logits of a probability vector.
    Returns the best vector (as a NumPy array) seen in this run and its
    exact peak value (i.e. max(a*a)).
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise logits from start (add epsilon to avoid log(0))
    eps = 1e-16
    y = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8

    # hyper‑parameters – randomised per run for extra exploration
    lr0 = rng.uniform(0.05, 0.15)            # base learning rate for logits
    tau_factor = rng.uniform(0.2, 0.5)      # τ₀ = factor·M
    jitter_interval = int(rng.integers(1500, 5000))

    # exponential decay schedule for τ
    decay_rate = max_iters / 8.0

    # FFT length for the 3‑n‑2 reverse‑convolution
    Nfft = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- probability simplex via soft‑max of logits ----
        logits = y - np.max(y)
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()
        a = target_sum * p                     # guaranteed Σa = target_sum, a_i≥0

        # ---- exact convolution to obtain the current peak ----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ---- keep the best ever seen ----
        if M < best_score - 1e-12:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:9d}] new best = {best_score:.8f}")

        # ---- soft‑max weights for the smoothed max ----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            # exact sub‑gradient (average over all active indices)
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
        else:
            w, _ = _softmax_weights(conv, tau)

        # ---- weighted reverse convolution (gradient of smoothed max) ----
        rev = _weighted_rev(a, w, Nfft)

        # ---- gradient w.r.t. a (2·rev) ----
        g_a = 2.0 * rev

        # ---- chain rule to logits (mirror‑descent on simplex) ----
        avg = np.dot(p, g_a)                 # ⟨p, g_a⟩
        g_y = target_sum * p * (g_a - avg)   # ∂/∂y_i

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)          # simple decay of the step‑size
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # ---- occasional jitter to escape plateaux ----
        if jitter_interval > 0 and it % jitter_interval == 0:
            y += rng.normal(scale=0.01, size=n)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Local stochastic refinement (pairwise mass transfers) ---------------

def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    cur_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Hill‑climbing with simulated‑annealing acceptance on the simplex."""
    target_sum = a.sum()                     # already √(2 n)
    cur_seq = a.copy()
    best_seq = a.copy()
    best_score = cur_score

    # FFT size for fast max‑convolution (only forward FFT needed)
    Nfft = 1 << ((2 * n - 1).bit_length())

    # annealing schedule
    max_steps = 250_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        # move a random fraction (1 %–10 %) of the source mass
        frac = rng.uniform(0.01, 0.10)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # recompute peak
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        # deterministic improvement
        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"[refine] NEW BEST = {best_score:.8f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.02 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return best_seq, best_score
