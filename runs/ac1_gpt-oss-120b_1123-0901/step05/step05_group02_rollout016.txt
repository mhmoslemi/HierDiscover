The code below searches for a non‑negative step‑function (a probability vector) that minimises  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2}\; .
\]

The objective is invariant under scaling, therefore every candidate is normalised to a
probability vector (`sum = 1`).  
The optimiser works in three stages

1. **Rich initialisation** – several deterministic shapes (power‑law, exponential,
   Gaussian, tent) and many random Dirichlet draws are generated.
2. **Smooth‑max Adam** – the max‑convolution is approximated by a soft‑max
   (`w = softmax(λ·(conv‑max))`).  Adam updates are performed on the
   log‑soft‑max parameters (`logits`).  The temperature `λ` is increased very
   slowly (`λ←λ·1.006`) so the optimiser sees a gradually sharpening surrogate
   of the true max.
3. **Exact‑max sub‑gradient phase** – after the smooth phase the true max is used.
   An exponentiated‑gradient (mirror‑descent) step is taken on the simplex,
   followed by a few iterations of a Euclidean projected‑gradient descent
   (the gradient of the max is simply a shifted copy of the current vector).

All phases keep track of the best objective value ever seen.  
When time runs out a final **random‑walk annealing** is performed around the
current best solution to try to escape tiny local minima.

The implementation respects the 1000 s budget, prints progress, and finally
returns the best sequence as a plain Python list – exactly what the external
evaluator expects.

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
step_function_search.py

Search for a non‑negative step‑function (probability vector) minimising

    F(a) = 2·n·max(a*a) / (sum(a)²)

The algorithm uses:
  * a large pool of diverse initial candidates,
  * a smooth‑max Adam optimiser on the log‑soft‑max (logits) representation,
  * an exact‑max exponentiated‑gradient (mirror descent) phase,
  * a short projected‑gradient descent (Euclidean) refinement,
  * a final random‑walk annealing on the best solution.

It fits into the allowed 1000 s budget and returns the best sequence found
as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities --------------------------------------------------
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper – next power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a using an FFT of length ``fft_len``.
    The result has length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _softmax(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = logits.max()
    exp_x = np.exp(logits - mx)
    return exp_x / exp_x.sum()


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) ----------------------------------------
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Projects a vector onto the probability simplex {x ≥ 0, Σx = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("project_to_simplex expects a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective ------------------------------------------------------------
# ----------------------------------------------------------------------


def _objective(a: np.ndarray, n: int, conv_len: int) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(conv)  .
    """
    conv = _full_convolution(a, conv_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Candidate factories ----------------------------------------------------
# ----------------------------------------------------------------------


def _make_candidates(n: int, rng: np.random.Generator) -> list:
    """
    Build a list of diverse starting points (all normalised to sum‑1).
    """
    cand = []

    # Warm‑start placeholder – will be filled later by the caller.
    # Uniform distribution (always useful)
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Pure random Dirichlet (α = 1)
    cand.append(rng.dirichlet(np.ones(n)))

    # Spiky Dirichlet (α < 1) – tends to concentrate mass
    for alpha in (0.2, 0.5, 0.8):
        cand.append(rng.dirichlet(np.full(n, alpha)))

    # Power‑law edge shape: (i+1)*(n-i) raised to a negative exponent
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.1, 13):
        shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Two‑sided exponential
    for gamma in (0.0005, 0.001, 0.002, 0.005):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        cand.append(shape)

    # Gaussian centred in the middle
    for sigma in (n / 8.0, n / 12.0, n / 20.0):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-0.5 * ((i - (n - 1) / 2.0) / sigma) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Tent shape (linear rise then fall)
    left = np.linspace(0, 1, n // 2, endpoint=False)
    right = left[::-1]
    tent = np.concatenate([left, right])
    tent = np.maximum(tent, 0.0)
    tent /= tent.sum()
    cand.append(tent)

    # Slightly smoothed sinusoid (always non‑negative)
    sin_shape = np.sin(np.pi * np.arange(n) / (n - 1))
    sin_shape = np.maximum(sin_shape, 0.0)
    sin_shape /= sin_shape.sum()
    cand.append(sin_shape)

    # Ensure every candidate is a proper probability vector
    clean = []
    for a in cand:
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = rng.dirichlet(np.ones(n))
        else:
            a = a / s
        clean.append(a)
    return clean

def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    max_iters: int,
    deadline: float,
) -> tuple:
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns (best_a, best_val, last_a, last_logit).
    """
    a = a0.copy()
    logits = np.log(a + 1e-15)  # initialise logits for the soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05                     # a bit smaller than the original code
    lam = 0.3                     # start temperature
    lam_factor = 1.006            # slower increase → smoother surrogate longer

    best_a = a.copy()
    best_val = float("inf")
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # --- soft‑max to stay on the simplex ----------
        a = _softmax(logits)

        # --- convolution and smooth‑max weights --------
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- gradient of the smooth‑max  -------------
        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        # --- back‑prop through the soft‑max ----------
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # --- Adam update -----------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # --- temperature schedule --------------------
        lam = min(lam * lam_factor, 1e6)

        # --- objective (exact) -----------------------
        val = 2.0 * n * max_c          # sum(a)=1 → denominator disappears

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t:05d}] new best {val:.6f}")

        # --- occasional random shake to escape stall --
        if time.time() - last_improve > 8.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    # final a from logits (ensures we return a feasible point)
    a = _softmax(logits)
    return best_a, best_val, a, logits


def _expgrad_exact_max(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    max_iters: int,
    deadline: float,
) -> tuple:
    """
    Exponentiated‑gradient on the exact max‑convolution.
    Returns (best_a, best_val, last_a).
    """
    a = a0.copy()
    eta = 0.55                    # step‑size
    best_a = a.copy()
    best_val = float("inf")
    last_improve = time.time()

    for it in range(max_iters):
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        # active set of indices attaining the maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        # exponentiated‑gradient step (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:               # protect against collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} EG iter={it}] new best {val:.6f}")

        if time.time() - last_improve > 8.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val, a


def _projected_gradient_descent(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    max_iters: int,
    deadline: float,
) -> tuple:
    """
    Euclidean projected‑gradient descent on the exact max.
    Gradient of max‑convolution is simply a shifted copy of ``a``.
    Returns (best_a, best_val).
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        k_max = int(conv.argmax())               # index of the maximal entry

        # gradient of max‑convolution: 2 * a shifted by s = k_max-(n-1)
        shift = k_max - (n - 1)
        g = np.zeros_like(a)
        if shift >= 0:
            g[: n - shift] = 2.0 * a[shift:]
        else:
            g[-shift:] = 2.0 * a[: n + shift]

        # step size decays like 1/√t
        step = 0.2 / np.sqrt(t)

        a = a - step * g
        a = _project_to_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} PGD iter={t}] new best {val:.6f}")

        if time.time() - last_improve > 8.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val


def _random_walk_annealing(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    deadline: float,
    start_temp: float = 0.02,
) -> tuple:
    """
    Very light simulated‑annealing on the simplex.
    Small Gaussian perturbations of the logits are proposed.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, conv_len)
    logit = np.log(a + 1e-15)

    # schedule parameters
    total = max(1.0, deadline - time.time())
    while time.time() < deadline:
        # linear cooling
        remaining = deadline - time.time()
        temp = start_temp * (remaining / total)

        # propose a perturbation in log‑space
        proposal_logit = logit + np.random.normal(scale=0.05, size=n)
        a_prop = _softmax(proposal_logit)

        val = _objective(a_prop, n, conv_len)

        if val < best_val or np.random.rand() < np.exp(-(val - best_val) / max(temp, 1e-12)):
            # accept
            a = a_prop
            logit = proposal_logit
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[n={n} Anneal] new best {val:.6f}")

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence that minimises the evaluation function.
    The routine respects the supplied time budget (default 1000 s) and returns
    the best sequence discovered as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # a safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Lengths to explore – larger lengths give more freedom but cost more.
    # We start with 4096 (the length that previously gave 1.509) and then,
    # if time allows, try 8192.
    # ------------------------------------------------------------------
    lengths = [4096, 8192]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_len = _next_pow2(2 * n - 1)
        corr_len = _next_pow2(3 * n)  # enough for cross‑correlation

        # ------------------------------------------------------------------
        # Build candidate pool (warm‑start from the globally best if its length matches)
        candidates = _make_candidates(n, rng)

        if best_seq is not None and len(best_seq) == n:
            candidates.insert(0, np.asarray(best_seq, dtype=np.float64))

        # ------------------------------------------------------------------
        # Optimise every candidate
        for cid, cand in enumerate(candidates):
            if time.time() > deadline:
                break

            a_init = cand / cand.sum()
            # ----------- Phase 1 – Adam on smooth‑max --------------------
            max_iters_adam = int(30000 * 1024 / n) + 8000   # a bit more than before
            best_a_adam, best_val_adam, a_mid, logits_mid = _adam_smooth_max(
                a_init,
                n,
                conv_len,
                corr_len,
                max_iters_adam,
                deadline,
            )

            # keep global best if improved
            if best_val_adam < best_val:
                best_val = best_val_adam
                best_seq = best_a_adam.copy()

            # ----------- Phase 2 – Exact‑max exponentiated gradient ------
            max_iters_eg = 6000
            best_a_eg, best_val_eg, a_mid2 = _expgrad_exact_max(
                a_mid,
                n,
                conv_len,
                corr_len,
                max_iters_eg,
                deadline,
            )
            if best_val_eg < best_val:
                best_val = best_val_eg
                best_seq = best_a_eg.copy()

            # ----------- Phase 3 – Projected gradient descent -----------
            max_iters_pgd = int(5000 * 1024 / n) + 500
            best_a_pgd, best_val_pgd = _projected_gradient_descent(
                a_mid2,
                n,
                conv_len,
                max_iters_pgd,
                deadline,
            )
            if best_val_pgd < best_val:
                best_val = best_val_pgd
                best_seq = best_a_pgd.copy()

            # ----------- Phase 4 – Light annealing around current best ---
            # Only if we still have a comfortable time window.
            if time.time() + 30.0 < deadline:
                best_a_ann, best_val_ann = _random_walk_annealing(
                    best_seq,
                    n,
                    conv_len,
                    deadline - 5.0,   # small buffer for the outer loop
                )
                if best_val_ann < best_val:
                    best_val = best_val_ann
                    best_seq = best_a_ann.copy()

        # ------------------------------------------------------------------
        # End of length ``n`` – a brief fine‑tune on the global best for this n
        # ------------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] final fine‑tune (current best {best_val:.6f})")
            # One more Adam pass with a much slower temperature increase
            max_iters_fine = int(50000 * 1024 / n) + 10000
            _, _, a_final, _ = _adam_smooth_max(
                best_seq,
                n,
                conv_len,
                corr_len,
                max_iters_fine,
                deadline,
            )
            final_val = _objective(a_final, n, conv_len)
            if final_val < best_val:
                best_val = final_val
                best_seq = a_final.copy()

    # ----------------------------------------------------------------------
    # Fallback – uniform vector (should never be needed)
    # ----------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 4096
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = _objective(best_seq, fallback_n, _next_pow2(2 * fallback_n - 1))
        print(f"Fallback used – objective {best_val:.6f}")

    print(f"\n=== SEARCH FINISHED ===\nBest objective found: {best_val:.6f}")
    # Return as a plain Python list for the external evaluator
    return best_seq.tolist()
```
