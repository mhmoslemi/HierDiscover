"""
Propose a candidate step‑function that minimises

    F(a) = 2·n·max(a∗a) / (sum(a)²)

The algorithm is an enhanced version of the previous Adam + sub‑gradient
pipeline:

* a richer pool of initial shapes (edge‑power, exponential edge,
  double‑Gaussian, sinusoid, uniform, warm‑start);
* for each start we run a long Adam phase on a smooth‑max surrogate,
  with a modest learning‑rate schedule and an aggressive temperature
  increase;
* this is followed by a much longer exponentiated‑gradient sub‑gradient
  phase on the true max;
* after the main pass we perform cheap random perturbations of the best
  sequence and a few extra sub‑gradient steps to escape shallow minima;
* finally, if time permits we up‑sample the current best to a larger
  dimension (8192) and repeat the whole optimisation.

All steps respect the time budget (default 1000 s) and keep track of the
overall best sequence/value.  The returned list is ready for
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using an FFT of size fft_len."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # Only the first 2*len(a)-1 entries are relevant.
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) required for the gradient.
    Returns a vector of length len(a) where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)].
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp smooth‑max surrogate.
    Returns the (possibly) improved global best.
    """
    # hyper‑parameters (slightly tuned wrt the previous version)
    lr = 0.07
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = 0.2                 # start temperature
    lam_factor = 1.03
    max_iter = 300_000

    logits = np.log(a0 + 1e-15)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        # ----- simplex from logits (stable soft‑max) -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- convolution & true objective (for monitoring) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={it:06d}] new best {val:.9f}")

        # ----- smooth‑max weights -----
        # w_i ∝ exp(lam * (conv_i - max_c))
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient wrt a (chain rule through soft‑max) -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)                # ⟨a,∇⟩ term from soft‑max Jacobian
        grad_log = a * (grad_a - avg)          # gradient w.r.t. logits

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, 1e6)

        # ----- occasional jitter when stuck -----
        if time.time() - last_improve > 4.0:
            jitter = (np.random.rand(n) - 0.5) * 1e-2
            logits += jitter
            last_improve = time.time()

        it += 1

    # Return the current best (might be unchanged)
    return best_seq, best_val


def _subgrad_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the exact max‑convolution.
    """
    a = np.maximum(a0, 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    else:
        a /= a_sum

    eta = 0.7                    # step‑size for EG (tuned)
    max_sub_iters = 20_000
    last_improve = time.time()

    for it in range(max_sub_iters):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} iter={it:05d}] new best {val:.9f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # re‑normalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # stagnation detection
        if time.time() - last_improve > 5.0:
            break

    return best_seq, best_val


def _optimize_one(a0: np.ndarray,
                  n: int,
                  conv_fft_len: int,
                  corr_fft_len: int,
                  deadline: float,
                  best_seq: np.ndarray | None,
                  best_val: float) -> tuple[np.ndarray, float]:
    """
    Full optimisation for a single start vector:
    1) Adam on smooth‑max.
    2) Sub‑gradient on the exact max.
    Returns updated global best.
    """
    best_seq, best_val = _adam_optimize(
        a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
    )
    best_seq, best_val = _subgrad_optimize(
        best_seq if best_seq is not None else a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a diverse pool of start vectors (all on the simplex).
    """
    cand = []

    # warm start and simple rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # pure Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # power‑law edge shapes: a_i ∝ (i·(n-i+1))^α
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-3.0, -0.4, 8):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # exponential edge decay: a_i ∝ exp(-β·dist_to_edge)
    dist_edge = np.minimum(np.arange(n) + 1, n - np.arange(n))
    for beta in (0.1, 0.3, 0.6, 1.0, 2.0):
        shape = np.exp(-beta * (dist_edge - 1))
        shape /= shape.sum()
        cand.append(shape)

    # two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.1, 0.3, 0.7, 1.5):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # sinusoidal half‑wave (positive)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _random_perturb(a: np.ndarray, scale: float = 0.02) -> np.ndarray:
    """Small random perturbation followed by projection onto the simplex."""
    n = a.shape[0]
    # select a few coordinates to perturb
    k = max(2, int(0.01 * n))            # ~1 % of entries
    idx = np.random.choice(n, size=k, replace=False)
    delta = (np.random.rand(k) - 0.5) * scale
    a2 = a.copy()
    a2[idx] += delta
    a2 = np.maximum(a2, 0.0)
    s = a2.sum()
    if s < 1e-12:
        a2 = np.random.dirichlet(np.ones(n))
    else:
        a2 /= s
    return a2


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Primary optimisation at n = 4096 (fits comfortably in memory)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # retrieve warm start if available
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_arr = np.asarray(warm_start, dtype=np.float64)
        if warm_arr.shape[0] != n:
            # resize by simple interpolation / truncation
            if warm_arr.shape[0] > n:
                warm_arr = warm_arr[:n]
            else:
                warm_arr = np.pad(warm_arr, (0, n - warm_arr.shape[0]))
        warm_arr = np.maximum(warm_arr, 0.0)
        if warm_arr.sum() >= 0.01:
            warm_arr = warm_arr / warm_arr.sum()
        else:
            warm_arr = None
    else:
        warm_arr = None

    candidates = _initial_candidates(n, warm_arr)

    # optimise each candidate (more time allocated to the first few)
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break

        # ensure a valid start on the simplex
        a0 = np.maximum(cand, 0.0)
        s = a0.sum()
        if s < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / s

        # run the two‑phase optimiser
        best_seq, best_val = _optimize_one(
            a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
        )

    # ------------------------------------------------------------------
    # Local random‑perturbation search around the current best.
    # This stage is cheap (no FFT length change) and can squeeze out a
    # few more digits before the deadline.
    if best_seq is not None:
        while time.time() + 0.3 < deadline:
            pert = _random_perturb(best_seq, scale=0.025)
            # a short sub‑gradient refinement (max 1500 iters)
            best_seq, best_val = _subgrad_optimize(
                pert,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                best_seq,
                best_val,
            )
            # break if we spent a second without any improvement
            if time.time() - deadline > -1.0:
                break

    # ------------------------------------------------------------------
    # If time permits, upscale to a larger size (8192) and repeat.
    if time.time() + 12.0 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation onto the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("[Info] Upscaling to n =", n_big)
        best_seq, best_val = _optimize_one(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )

        # one final cheap perturbation loop at the bigger resolution
        if best_seq is not None:
            while time.time() + 0.3 < deadline:
                pert = _random_perturb(best_seq, scale=0.02)
                best_seq, best_val = _subgrad_optimize(
                    pert,
                    n_big,
                    conv_fft_len_big,
                    corr_fft_len_big,
                    deadline,
                    best_seq,
                    best_val,
                )
                if time.time() >= deadline:
                    break

    # ------------------------------------------------------------------
    # Fallback: if nothing succeeded, return a uniform vector.
    if best_seq is None:
        print("[Warning] No valid sequence found – returning uniform.")
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final logging
    print(f"[Result] Best value = {best_val:.9f} with length = {best_seq.shape[0]}")
    return best_seq.tolist()
