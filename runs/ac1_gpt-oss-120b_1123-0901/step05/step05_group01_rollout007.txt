"""
Propose a non‑negative step‑function sequence with a low evaluation score.

Algorithm overview
------------------
1. **Length schedule** – Test a handful of lengths (4096 … 65536).  
2. **Diverse warm‑starts** – For each length generate a small pool of
   candidates:
   * the best sequence found so far (or interpolated to the new size),
   * a historic sequence if available,
   * a random Dirichlet draw,
   * a parametric sweep over several families (edge‑concentrated
     power‑laws, inverse‑root, edge Gaussians, etc.).
3. **Optimization pipeline** – Each candidate is refined by three stages:
   * *Adam* on a smooth‑max surrogate of the max‑convolution (soft‑max
     re‑parametrisation guarantees non‑negativity and unit sum),
   * *Exact sub‑gradient* (exponentiated‑gradient) on the true max,
   * *Hill‑climb* random pairwise mass transfers.
4. **Global‑best refinement** – If time permits a long Adam run on the
   current global best.
5. **Return** – the best sequence (plain Python list) found before the
   deadline.  All internal evaluations use an FFT based convolution to
   stay fast for large n; the final score is obtained via
   ``evaluate_sequence`` (provided by the grader).

The code respects the 1000 s time budget, prints progress, and has no
file or network I/O.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT convolution utilities
def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)−1) using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Objective helpers (scale‑invariant, sum is kept equal to 1 internally)
def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)² (sum a ≈ 1)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)

# Candidate generation
def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence by linear interpolation and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new


def _param_sweep_candidates(n: int) -> list[np.ndarray]:
    """Generate a set of parametric shapes (edge‑concentrated, inverse‑root, etc.)."""
    cand = []
    idx = np.arange(n, dtype=np.float64)
    prod = (idx + 1) * (n - idx)        # (i+1)*(n-i)

    # Edge‑concentrated power‑law ((i+1)*(n-i))**α , α < 0
    for alpha in np.linspace(-5.0, -0.2, 15):
        shape = np.power(prod, alpha, where=prod > 0,
                         out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # Inverse‑root edge shapes (i+1)**‑α + (n‑i)**‑α
    for alpha in np.linspace(0.3, 2.0, 12):
        shape = np.power(idx + 1, -alpha) + np.power(n - idx, -alpha)
        cand.append(shape / shape.sum())

    # Two‑Gaussian edge shapes
    for gamma in (0.5, 1.0, 2.0, 3.0, 5.0):
        shape = np.exp(-gamma * (idx ** 2)) + np.exp(-gamma * ((n - 1 - idx) ** 2))
        cand.append(shape / shape.sum())

    # Three symmetric Gaussian peaks (0.25,0.5,0.75)
    for sigma in (0.07, 0.12, 0.2):
        x = idx / (n - 1)
        peaks = sum(np.exp(-0.5 * ((x - p) / sigma) ** 2) for p in (0.25, 0.5, 0.75))
        cand.append(peaks / peaks.sum())

    return cand


def _generate_initial_candidates(n: int,
                                 best_seq: np.ndarray | None,
                                 hist_seq: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a modestly sized, diverse set of starting points for length n.
    All returned arrays are non‑negative and sum to 1.
    """
    cand = []

    # Warm‑start from the global best (exact length)
    if best_seq is not None:
        if best_seq.shape[0] == n:
            cand.append(best_seq.copy())
        else:
            cand.append(_interpolate_seq(best_seq, n))

    # Historic sequence if available
    if hist_seq is not None:
        if hist_seq.shape[0] > n:
            tmp = hist_seq[:n].copy()
        else:
            tmp = np.pad(hist_seq, (0, n - hist_seq.shape[0]))
        tmp = np.maximum(tmp, 0.0)
        tot = tmp.sum()
        if tot < 1e-12:
            tmp = np.random.dirichlet(np.ones(n))
        else:
            tmp = tmp / tot
        cand.append(tmp)

    # Pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # Parametric sweep (edges, inverse‑root, gaussians, multi‑peak)
    cand.extend(_param_sweep_candidates(n))

    # Shuffle and keep a limited number (to respect budget)
    np.random.shuffle(cand)
    return cand[:12]

# Optimisation stages
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   fft_len_conv: int,
                   fft_len_corr: int,
                   deadline: float,
                   max_iters: int) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a smooth‑max surrogate.
    Returns the best (a, value) encountered before the deadline.
    """
    # keep sum = 1 via soft‑max on logits
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.02                     # a bit smaller than before – stable for large n
    lam = 2.0                     # start with a moderate smooth‑max temperature
    lam_factor = 1.02
    max_lam = 1e7

    best_a = a.copy()
    best_val = _objective(a, n, fft_len_conv)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # simplex projection via soft‑max
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # smooth‑max weights for the convolution entries
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, max_lam)

        cur_val = _objective(a, n, fft_len_conv)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t:06d}] new best {cur_val:.6f}")

        # occasional random shake if stuck for >4 s
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


def _subgrad_optimize(a0: np.ndarray,
                      n: int,
                      fft_len_conv: int,
                      fft_len_corr: int,
                      deadline: float,
                      init_val: float | None = None,
                      max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient (exponentiated‑gradient) optimisation on the true max.
    """
    a = a0 / a0.sum()
    if init_val is None:
        best_val = _objective(a, n, fft_len_conv)
    else:
        best_val = init_val
    best_a = a.copy()
    last_improve = time.time()
    eta = 0.6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        # active set: indices achieving the maximum (within tolerance)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient step keeps a ≥ 0
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        cur_val = _objective(a, n, fft_len_conv)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it}] new best {cur_val:.6f}")

        # stop early if no improvement for a while
        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val


def _hill_climb_optimize(a0: np.ndarray,
                         n: int,
                         fft_len_conv: int,
                         deadline: float,
                         init_val: float | None = None,
                         max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: move a small amount of mass between two coordinates.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_conv) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2:
            continue

        # move up to 5 % of the mass from i1 to i2
        eps = np.random.rand() * 0.05 * a[i1]
        if eps <= 0.0:
            continue

        new_a = a.copy()
        new_a[i1] -= eps
        new_a[i2] += eps
        if new_a[i1] < -1e-15:
            continue
        new_a /= new_a.sum()

        new_val = _objective(new_a, n, fft_len_conv)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            last_improve = time.time()
            if it % 50 == 0:
                print(f"[n={n} HillClimb it={it}] new best {new_val:.6f}")
        else:
            # occasional uphill move to escape tiny plateaus
            if np.random.rand() < 0.001:
                a = new_a

        if time.time() - last_improve > 8.0:
            break

    return best_a, best_val


def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         fft_len_conv: int,
                         deadline: float,
                         init_val: float | None = None,
                         max_moves: int = 5000) -> tuple[np.ndarray, float]:
    """
    Simple SA that swaps a small amount of probability mass between two random sites.
    Very cheap – only a few thousand moves.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_conv) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val

    temperature = 0.1
    decay = 0.9995

    for mv in range(max_moves):
        if time.time() > deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move a random fraction (≤2 %) of mass from i to j
        eps = np.random.rand() * 0.02 * a[i]
        if eps <= 0.0:
            continue
        new_a = a.copy()
        new_a[i] -= eps
        new_a[j] += eps
        if new_a[i] < -1e-15:
            continue
        new_a /= new_a.sum()
        new_val = _objective(new_a, n, fft_len_conv)

        delta = new_val - cur_val
        if delta < 0 or np.exp(-delta / temperature) > np.random.rand():
            a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
        temperature = max(temperature * decay, 1e-6)

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    best_seq = None          # np.ndarray, sum ≈ 1
    best_val = float("inf")

    # Lengths – start modest and finish with the largest affordable ones.
    lengths = [4096, 8192, 16384, 32768, 65536]

    # Pre‑compute FFT sizes (next power of two) for each length.
    conv_fft = {}
    corr_fft = {}
    for n in lengths:
        conv_len = 2 * n - 1
        conv_fft[n] = 1 << int(np.ceil(np.log2(conv_len)))
        # correlation length ≤ 3·n
        corr_fft[n] = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

    # Load historic solution if present (global variable from previous run)
    hist_seq = None
    if "height_sequence_1" in globals():
        tmp = globals()["height_sequence_1"]
        if isinstance(tmp, (list, tuple, np.ndarray)):
            hist_seq = np.asarray(tmp, dtype=np.float64)

    # Main loop over lengths -------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        fft_len_c = conv_fft[n]
        fft_len_x = corr_fft[n]

        # Build a modest, diverse pool of warm‑starts.
        candidates = _generate_initial_candidates(n, best_seq, hist_seq)

        # Number of Adam iterations – longer for smaller n.
        max_adam_iters = int(18000 * 1024 / n) + 3000

        for cand_id, a_start in enumerate(candidates):
            if time.time() > deadline:
                break

            # ----- Phase 1 – smooth‑max Adam -----
            a_adam, val_adam = _adam_optimize(
                a_start, n, fft_len_c, fft_len_x, deadline, max_adam_iters
            )

            # ----- Phase 2 – exact sub‑gradient -----
            a_sub, val_sub = _subgrad_optimize(
                a_adam, n, fft_len_c, fft_len_x, deadline, init_val=val_adam
            )

            # ----- Phase 3 – cheap hill‑climb -----
            a_hc, val_hc = _hill_climb_optimize(
                a_sub, n, fft_len_c, deadline, init_val=val_sub
            )

            # ----- Optional tiny SA refinement (quick) -----
            a_sa, val_sa = _simulated_annealing(
                a_hc, n, fft_len_c, deadline, init_val=val_hc, max_moves=2000
            )

            # Choose the best among the three refinements
            final_a, final_val = a_sa, val_sa
            if val_hc < final_val:
                final_a, final_val = a_hc, val_hc
            if val_sub < final_val:
                final_a, final_val = a_sub, val_sub
            if val_adam < final_val:
                final_a, final_val = a_adam, val_adam

            # Update global best
            if final_val < best_val:
                best_val = final_val
                best_seq = final_a.copy()
                print(f"*** Global new best (n={n} cand={cand_id}) value={best_val:.6f}")

    # ----------------------------------------------------------------------
    # Long‑run refinement on the current global best (if time left)
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        fft_len_c = conv_fft.get(n, 1 << int(np.ceil(np.log2(2 * n - 1))))
        fft_len_x = corr_fft.get(n, 1 << int(np.ceil(np.log2(3 * n))))
        print("\n=== Long‑run refinement of the global best ===")
        refined_seq, refined_val = _adam_optimize(
            best_seq, n, fft_len_c, fft_len_x, deadline,
            max_iters=int(2.5 * (18000 * 1024 / n) + 5000)
        )
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq.copy()
            print(f"*** Refined global best value={best_val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – uniform vector (should never happen)
    if best_seq is None:
        fallback_n = 4096
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Convert to plain Python list for the grader.
    final_list = best_seq.tolist()

    # Quick sanity check – print the official evaluation (uses the grader’s function).
    try:
        from __main__ import evaluate_sequence  # will be present in the execution env
        official_score = evaluate_sequence(final_list)
        print(f"\nOfficial evaluation score: {official_score:.6f}")
    except Exception:
        # The grader will call evaluate_sequence anyway; ignore any import issues here.
        pass

    return final_list
