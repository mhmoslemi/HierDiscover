assistantfinal**Solution Overview**

The goal is to minimise  

\[
\text{score}(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2},
\qquad a_i\ge 0 .
\]

Because the score is invariant under scaling we can (and do) keep the
\(L_1\)-norm fixed to  

\[
\sum_i a_i = \sqrt{2n},
\]

so the score is simply the largest entry of the (linear) convolution
\(a*a\).

The programme works in three stages

1. **Fast parametric search** – we generate candidates from several
   analytically‑motivated families (`geom`, `power`, `mixed`,
   `exp_pow`, …).  For a given length `n` the whole family is sampled
   densely (randomised log‑space for the parameters).  The candidate
   with the smallest convolution peak is stored as the current best.

2. **Refinement with Adam on a soft‑max parametrisation** – the best
   candidate is taken as a warm start.  The coefficients are written as a
   soft‑max of unrestricted logits; a smooth surrogate of the
   “max‑convolution” (log‑sum‑exp) gives a differentiable objective.
   Adam with a slowly decaying learning‑rate explores the neighbourhood
   and often pushes the peak well below the value obtained by the pure
   parametric search.

3. **Exact fine‑tuning** – a short projected sub‑gradient descent on the
   true (non‑smooth) peak, followed by a very cheap stochastic
   hill‑climb, finishes the optimisation.

All stages respect the global time budget (`budget_s`).  The programme
keeps the globally best sequence at any moment and prints progress
messages; finally it returns the best sequence found as a plain Python
list.

```python
"""
Hybrid optimiser for the step‑function problem.

1.  Randomly sample a few analytically motivated families
    (geometric, power‑law, mixed exponential‑power, stretched‑exponential)
    for many different lengths n.  The convolution peak is obtained with
    a fast FFT‑based convolution – this gives a good initial guess
    quickly.

2.  Refine the best guess with Adam optimisation on the soft‑max
    parametrisation.  A log‑sum‑exp surrogate replaces the hard max,
    making the objective smooth while still encouraging a low peak.

3.  Finish with a short projected sub‑gradient descent on the exact
    max‑convolution and a tiny stochastic hill‑climb.

The routine respects the supplied time budget (default 1000 s) and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full convolution via FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors x and y using an FFT of power‑of‑2 size."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)            # eliminate tiny negatives
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex { w ≥ 0, Σ w = s }
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex with radius s (Duchi et al. 2008)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # descending sort
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)            # 1‑based indices
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1               # zero‑based index of the last positive entry
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Evaluation of the exact objective (max of the convolution)
# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray) -> float:
    """Return max_k (a*a)[k] for a non‑negative vector a."""
    return _full_convolution_fft(a, a).max()


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max (logits) parametrisation
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    logits: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    alpha0: float = 0.02,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    init_tau_factor: float = 0.15,
    min_tau_factor: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max–convolution.
    Returns (best_vector, best_exact_score).
    """
    x = logits.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # map logits to the simplex (a = target_sum * softmax(x))
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        p = exp_shift / np.sum(exp_shift)
        a = target_sum * p

        # exact peak (for bookkeeping)
        cur_max = _conv_max(a)
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] Adam n={n} new best max={cur_max:.6f}")

        # ----- smooth‑max surrogate (log‑sum‑exp) -----
        tau = max(init_tau_factor * cur_max * (0.99 ** it), min_tau_factor * cur_max)
        w = np.exp((np.convolve(a, a) - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # gradient w.r.t. a  (2 · a ⊛ w_rev)
        w_rev = w[::-1]
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # back‑prop through the soft‑max
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # decay learning rate very slowly
        alpha *= 0.9995

    # final exact evaluation of the best candidate
    final_score = _conv_max(best_a) if best_a is not None else np.inf
    return best_a, float(final_score)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on max(a*a) with simplex projection."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _conv_max(a)
    step_size = 0.1 * target_sum / n   # rough 1 % of the average entry

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())                     # index of current maximum

        # sub‑gradient of the active convolution entry
        grad = np.zeros_like(a)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # descent step and projection onto the simplex
        a_candidate = a - step_size * grad
        a_candidate = _project_onto_simplex(a_candidate, target_sum)

        val_candidate = _conv_max(a_candidate)

        if val_candidate < best_val - 1e-12:
            a = a_candidate
            best_a = a_candidate
            best_val = val_candidate
            step_size *= 1.02          # slightly increase after success
        else:
            step_size *= 0.70          # shrink after failure

        # occasional jitter to keep the search alive
        if it % 300 == 0 and np.random.rand() < 0.15:
            jitter = np.random.rand(n) * 1e-4
            a = _project_onto_simplex(a + jitter, target_sum)

        if it % 500 == 0:
            print(f"[{time.time():.2f}] sub‑grad it={it} step={step_size:.3e} best={best_val:.6f}")

        if best_val < 1.503:          # early exit if we already beat the target
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny stochastic hill‑climb for final polishing
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Random local search that moves a tiny amount of mass from one index to another.
    Accepts improving moves; otherwise uses a Metropolis criterion with a cooling schedule.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _conv_max(a)

    T0 = best_val * 0.3
    T_end = 1e-4
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue

        delta = a[i] * np.random.rand() * 0.10    # at most 10 % of the source mass
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _conv_max(a_new)

        accept = False
        if new_val < best_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")

        if it % 2000 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Generators for simple parametric families
# ----------------------------------------------------------------------
def _make_candidate(n: int, kind: str, params: tuple[float, ...]) -> np.ndarray:
    """Build a non‑negative vector of length n from the supplied family."""
    i = np.arange(n, dtype=float)

    if kind == "geom":                     # a_i = r**i
        r = params[0]
        a = r ** i
    elif kind == "geom_rev":               # a_i = r**(n-1-i)
        r = params[0]
        a = r ** (n - 1 - i)
    elif kind == "power":                  # a_i = (i+1)^(-p)
        p = params[0]
        a = (i + 1) ** (-p)
    elif kind == "mixed":                  # a_i = (i+1)^(-p) * r**i
        r, p = params
        a = (i + 1) ** (-p) * (r ** i)
    elif kind == "exp":                    # a_i = exp(-λ i)
        lam = params[0]
        a = np.exp(-lam * i)
    elif kind == "exp_pow":                # a_i = exp(-λ (i+1)^α)
        lam, alpha = params
        a = np.exp(-lam * (i + 1) ** alpha)
    else:                                  # fallback – uniform (should never be needed)
        a = np.ones_like(i)

    # avoid a completely zero vector
    a = np.maximum(a, 1e-15)

    # normalise to the required L1‑norm
    target = np.sqrt(2.0 * n)
    a = a * (target / a.sum())
    return a

        if cur_max < best_score * 0.98 and remaining > 30.0:
            # initialise logits from the candidate
            logits0 = np.log(a + 1e-12)
            sub_dead = time.time() + min(12.0, remaining * 0.25)
            seq_adam, val_adam = _optimise_adam_softmax(
                n,
                logits0,
                np.sqrt(2.0 * n),
                sub_dead,
                alpha0=0.015,
                init_tau_factor=0.12,
            )
            if val_adam < best_score:
                best_score = val_adam
                best_seq = seq_adam.copy()
                print(f"[{time.time():.2f}] Adam‑refine improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – long Adam optimisation starting from the current best
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 60.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        logits0 = np.log(best_seq + 1e-12)
        seq_adam, val_adam = _optimise_adam_softmax(
            n, logits0, target_sum, deadline,
            alpha0=0.02,
            init_tau_factor=0.15,
            min_tau_factor=0.001,
        )
        if val_adam < best_score:
            best_score = val_adam
            best_seq = seq_adam.copy()
            print(f"[{time.time():.2f}] post‑Adam new best={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – projected sub‑gradient descent (exact max)
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 15.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_end = time.time() + min(20.0, time_left * 0.5)
        seq_sub, val_sub = _subgradient_descent(best_seq.copy(), target_sum, sub_end)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑grad best={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – final stochastic hill‑climb
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 5.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        hill_end = time.time() + time_left
        seq_hill, val_hill = _hillclimb(best_seq.copy(), target_sum, hill_end)
        if val_hill < best_score:
            best_score = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] after hill‑climb best={best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # --------------------------------------------------------------
    if best_seq is None:
        n = 2000
        target = np.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)
        best_score = 2.0
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # Return the best sequence as a plain Python list
    return best_seq.tolist()
```
