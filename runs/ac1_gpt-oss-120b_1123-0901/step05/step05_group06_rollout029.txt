def _sparse_anneal(n, k, target_sum, deadline, rng):
    # initialize positions equally spaced
    pos = np.linspace(0, n-1, k, dtype=int)
    # ensure uniqueness, adjust if duplicates due to rounding
    # add random positions to fill
    # assign equal mass
    a = np.zeros(n)

    # compute best score
    best_conv = _max_conv(a)
    best_a = a.copy()
    best_pos = pos.copy()
    
    start = time.time()
    # schedule temperature decay over time
    init_temp = best_conv
    while time.time() < deadline:
        # propose move: select random index in support and new position
        i_idx = rng.integers(k)
        old_pos = pos[i_idx]
        delta = rng.integers(-5,6)  # random move step
        new_pos = int(old_pos + delta)
        if new_pos < 0 or new_pos >= n:
            continue

        if new_pos in pos:
            continue
        # propose new vector
        a_new = a.copy()
        a_new[old_pos] = 0.0
        a_new[new_pos] = mass
        new_conv = _max_conv(a_new)
        # accept if better, else SA

def _max_convolution_dense(a: np.ndarray) -> float:
    """Return max(a * a) using FFT (linear convolution)."""
    n = a.shape[0]
    Nfft = 1 << ((2 * n - 1).bit_length())
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]

    best_score = _max_convolution_sparse(pos, mass, n)

    # current state (might move away from best)
    cur_pos = pos.copy()
    cur_score = best_score

    init_temp = best_score
    # move range (max index displacement)
    max_step = max(1, n // 100)     # about 1 % of length

        # pick a random element of the support to move
        i = rng.integers(k)

        # deterministic improvement
        if new_score < cur_score:

            cur_score = new_score

            if new_score < best_score:

def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau

def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """Weighted reverse convolution needed for the smooth‑max gradient."""
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev

def _exp_gradient_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑its with a soft‑max approximation of the max‑convolution.
    """
    target_sum = math.sqrt(2.0 * n)

    eps = 1e-16
    y = np.log(a0 + eps)

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.08
    jitter_interval = 2000

    tau_factor = 0.30
    decay_rate = max_iters / 8.0

    Nfft = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        logits = y - np.max(y)
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()
        a = target_sum * p

        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        rev = _weighted_rev(a, w, Nfft)

        g_a = 2.0 * rev
        avg = np.dot(p, g_a)
        g_y = target_sum * p * (g_a - avg)

        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        if it % jitter_interval == 0:
            jitter = rng.normal(scale=0.01, size=n)
            y += jitter

        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[adam n={n} iter {it}] new best = {best_score:.6f}")

    return best_seq, best_score

def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on the simplex.
    Moves a small fraction of mass from i to j, accepts improvements
    always and accepts worsening moves with a temperature that cools
    linearly.
    """
    target_sum = a.sum()
    cur_seq = a.copy()
    cur_score = best_score

    Nfft = 1 << ((2 * n - 1).bit_length())
    max_steps = 200_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score

"""

The algorithm consists of three stages:

1. **Sparse simulated‑annealing** – we treat the vector as a probability
   distribution (its sum is fixed).  For a support of size `k` and equal
   masses the convolution becomes the distribution of the sum of two
   independent draws.  By moving the support positions we can drastically
   lower the peak of that distribution while keeping the total mass
   constant.  This phase is extremely cheap because the convolution is
   computed by counting ordered pair sums rather than by FFT.

2. **Soft‑max Adam refinement** – the best dense vector from stage 1 is
   refined with the exponentiated‑gradient optimiser used in the previous
   state‑of‑the‑art code.  The optimiser works on the log‑its (softmax)
   parametrisation, automatically respects non‑negativity and the
   `Σa = √(2 n)` normalisation, and uses a smooth approximation of the
   max‑convolution that is annealed to the true max.

3. **Pairwise stochastic hill‑climbing** – a cheap local search that
   repeatedly transfers a small fraction of mass between two indices.
   Improvements are always accepted; worsening moves are accepted with a
