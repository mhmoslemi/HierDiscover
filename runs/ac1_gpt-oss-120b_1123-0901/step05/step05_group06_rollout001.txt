def _frank_wolfe_optimize(a0, target_sum, deadline, max_iters, best_score, best_seq, rng):
    n = len(a0)
    f = a0.copy()
    # ensure sum = target_sum
    f = f * (target_sum / f.sum())
    M = _max_convolution(f)
    cur_score = 2 * n * M / (target_sum * target_sum)
    if cur_score < best_score:
        best_score = cur_score
        best_seq = f.copy()
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # Solve LP
        b = _solve_lp_max_sum_under_conv_bound(f, M)
        if b is None or b.sum() < 1e-12:

        b = b * (target_sum / b.sum())

        t = 0.5 / (1 + it)  # decreasing

        # recompute M and score

        if cur_score < best_score:
            best_score = cur_score
            best_seq = f.copy()
    return best_seq, best_score

def _solve_lp_max_sum_under_conv_bound(f, M):
    n = len(f)
    rows = 2*n - 1
    # Build A_ub (rows x n)
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f
    # objective: maximize sum(b) -> minimize -ones @ b
    c = -np.ones(n)
    ub = np.full(rows, M, dtype=np.float64)  # RHS
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs', options={'presolve': True, 'disp': False})
    if res.success:
        return res.x
    else:
        # fallback: return None
        return None

def _refine_sequence(seq, deadline, rng):
    n = len(seq)
    target_sum = seq.sum()
    cur_seq = seq.copy()
    # compute current max conv
    Nfft = 1 << ((2*n-1).bit_length())
    A_fft = np.fft.rfft(cur_seq, n=Nfft)
    conv = np.fft.irfft(A_fft*A_fft, n=Nfft)[:2*n-1]
    cur_max = float(conv.max())
    cur_score = 2 * n * cur_max / (target_sum**2)
    step = 0

"""
Propose a non‑negative step function minimizing ``evaluate_sequence``.
Algorithm
---------
1. Explore a few problem sizes (n≈500‒2500).  For each size generate several
   diverse initial vectors (uniform, Dirichlet, exponential, windowed, etc.).
2. Run a Frank‑Wolfe style optimisation:
   * Let f be the current vector (ℓ₁‑norm = √(2 n)).
   * Compute M = max (f∗f).
   * Solve the linear program
         maximise   Σ b_j
         subject to (f∗b)_k ≤ M   for all k,
                    b_j ≥ 0.
     The LP yields a feasible direction b that tries to spread the mass
     while respecting the current convolution bound.
   * Rescale b to the required ℓ₁‑norm and update
         f ← (1‑t) f + t b   with a decreasing step size t.
   * Keep the best vector (smallest objective) seen during the run.
3. After the global best vector has been found, apply a cheap stochastic
   refinement: repeatedly transfer a small fraction of mass from a random
   source index to a random target index, accepting improvements always
   and worsening moves with a simulated‑annealing probability.  This
   tightening step often yields the final few percent of improvement.
All constraints (non‑negative entries, ℓ₁‑norm) are enforced explicitly.
The routine obeys the 1000 s time budget by checking the deadline at
each iteration.
"""

from __future__ import annotations

import time
import math
import numpy as np
from scipy.optimize import linprog  # type: ignore

# ----------------------------------------------------------------------
# Utility functions ------------------------------------------------------


def _max_convolution(a: np.ndarray) -> float:
    """
    Compute max_k (a ∗ a)[k] using an FFT of appropriate length.
    """
    n = a.shape[0]
    Nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
    return float(conv.max())


def _make_initial_vectors(
    n: int, target_sum: float, rng: np.random.Generator
) -> list[np.ndarray]:
    """
    Produce a small diverse set of normalised start vectors for a given length.
    All vectors satisfy Σ a_i = target_sum and a_i ≥ 0.
    """
    init: list[np.ndarray] = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) triangular bump (peak in the centre)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    if tri.sum() > 0:
        init.append(tri / tri.sum() * target_sum)

    return init


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the linear convolution matrix C such that (C @ b) = f ∗ b.
    Shape is (2·n‑1, n) for a vector f of length n.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    C = np.zeros((rows, n), dtype=np.float64)
    # each column i contains f shifted by i positions
    for i in range(n):
        C[i : i + n, i] = f
    return C


def _solve_lp_max_sum_under_conv_bound(
    f: np.ndarray, M: float
) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   Σ b_j
        subject to (f ∗ b)_k ≤ M   for every k,
                   b_j ≥ 0.
    Returns the optimal b (may be all zeros) or None on failure.
    """
    n = f.shape[0]
    # Build constraint matrix for (f ∗ b) ≤ M
    A = _build_conv_matrix(f)
    b_ub = np.full(A.shape[0], M, dtype=np.float64)

    # Objective: maximise Σ b → minimise -Σ b
    c = -np.ones(n, dtype=np.float64)

    # Use scipy's high‑performance HiGHS solver
    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True, "quiet": True},
    )
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Frank‑Wolfe optimisation ----------------------------------------------


def _frank_wolfe_optimize(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Frank‑Wolfe style optimisation for a given start vector.
    Returns the best vector found and its objective value
    (the ``evaluate_sequence`` score, i.e. 2·n·max(conv)/(sum²)).
    """
    n = a0.shape[0]
    # initialise feasible point on the simplex
    f = a0.astype(np.float64)
    f = f * (target_sum / f.sum())

    # current best
    cur_M = _max_convolution(f)
    best_f = f.copy()
    best_score = 2.0 * n * cur_M / (target_sum * target_sum)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # --- linear optimisation sub‑problem ---------------------------------
        b = _solve_lp_max_sum_under_conv_bound(f, cur_M)
        if b is None or b.sum() < 1e-12:
            # LP failed or yielded no mass – cannot improve further
            break

        # rescale b to the required ℓ₁‑norm
        b = b * (target_sum / b.sum())

        # step size (standard FW schedule: 2/(it+2))
        t = 2.0 / (it + 2.0)

        # update on the simplex
        f = (1.0 - t) * f + t * b

        # evaluate new max‑convolution
        cur_M = _max_convolution(f)
        cur_score = 2.0 * n * cur_M / (target_sum * target_sum)

        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            print(
                f"[FW n={n:5d} iter={it:6d}] new global best = {best_score:.6f}"
            )

    return best_f, best_score


# ----------------------------------------------------------------------
# Stochastic local refinement --------------------------------------------


def _refine_sequence(
    seq: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Cheap stochastic hill‑climbing: repeatedly move a small fraction of mass
    from a random source index to a random target index.  Accept improvements
    always; accept worsening moves with a decaying simulated‑annealing
    probability.  Returns the improved vector and its objective value.
    """
    n = seq.shape[0]
    target_sum = seq.sum()  # should equal √(2 n)
    cur_seq = seq.copy()

    # Initialise current objective
    Nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(cur_seq, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
    cur_max = float(conv.max())
    cur_score = 2.0 * n * cur_max / (target_sum * target_sum)

    step = 0
    max_steps = 500_000  # generous budget – stopped by deadline

    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        # transfer 1–5 % of the mass from i to j
        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # compute new max‑convolution
        A = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
        new_max = float(conv.max())
        new_score = 2.0 * n * new_max / (target_sum * target_sum)

        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_max = new_max
            cur_score = new_score
            # occasional progress report
            if step % 50_000 == 0:
                print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_max = new_max
                cur_score = new_score
        step += 1

    return cur_seq, cur_score

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
