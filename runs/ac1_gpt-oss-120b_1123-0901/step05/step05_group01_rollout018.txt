def simulated_annealing(a0, n, fft_len_c, time_budget):
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_c)
    best_a = a.copy()
    best_val = cur_val
    t_start = time.time()
    # set temperature schedule parameters
    # e.g., T_start = 0.01 * cur_val maybe
    # T_end = 1e-6
    # We'll compute total steps maybe 200k, so compute ratio.

    # Determine total steps based on remaining time left.
    steps = int( (time_budget - time.time()) / 0.0005 ) # approximate conv time

    for step in range(steps):
        if time.time() > time_budget:
            break
        # propose move
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # choose eps mass proportionally to a[i]
        # adapt epsilon based on temperature maybe
        max_move = 0.1 * a[i]  # 10% of mass
        eps = np.random.rand() * max_move
        # ensure non-negative
        if a[i] - eps < 0:
            eps = a[i]
        new_a = a.copy()
        new_a[i] -= eps
        new_a[j] += eps
        new_a /= new_a.sum()  # re-normalize (should stay 1)
        new_val = _objective(new_a, n, fft_len_c)
        # acceptance
        T = max(T_end, T_start * ((steps - step) / steps)**2) # schedule decreasing quadratically
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
        # occasional reheat: if no improvement for 200 steps, increase T temporarily
    return best_a, best_val
```

"""
Propose a non‑negative step‑function (list of floats) with a low
evaluation score F = 2·n·max(conv(a,a)) / (sum a)².

The search proceeds in several stages within the 1000 s budget:

1. **Diverse warm‑starts** – Dirichlet random, edge‑focused power‑laws,
   inverse‑root “arcsine” shape, mixtures of Gaussians, random sparse
   supports, etc.  All candidates are normalised to sum‑1.
2. **Smooth‑max Adam** – optimiser on the log‑sum‑exp surrogate of the
   max‑convolution.  The sequence is represented by soft‑max logits so
   the simplex constraint is satisfied automatically.
3. **Exact sub‑gradient (exponentiated‑gradient)** – works on the true
   max‑convolution and keeps the iterates on the probability simplex.
4. **Cheap hill‑climbing** – random mass transfers between two coordinates.
5. **Global best refinement** – a longer Adam run on the best sequence found.
6. **Simulated‑annealing fine‑tuning** – many small random moves accepted
   according to a temperature schedule; this stage exploits the remaining
   time budget and often squeezes a few extra digits out of the bound.

All heavy work uses FFT‑based convolutions (O(n log n)).  The algorithm
respects the time limit and returns a plain Python list of non‑negative
floats.
"""

import time
import numpy as np
from typing import List, Tuple, Optional

# ----------------------------------------------------------------------
# FFT based convolution utilities
def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)−1 using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) = n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)²."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Helper utilities
def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence by linear interpolation and re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new


def _arcsine_shape(n: int) -> np.ndarray:
    """Arcsine‑type density ∝ 1/√(x(1‑x)) discretised on n points."""
    i = np.arange(n, dtype=np.float64) + 0.5          # mid‑point indices
    shape = 1.0 / np.sqrt(i * (n - i))
    shape = np.maximum(shape, 0.0)
    total = shape.sum()
    if total < 1e-12:
        shape = np.random.dirichlet(np.ones(n))
    else:
        shape = shape / total
    return shape


def _product_powerlaw_shape(n: int, al: float, ar: float) -> np.ndarray:
    """Shape ∝ (i+1)^{-al}·(n-i)^{-ar}."""
    i = np.arange(n, dtype=np.float64) + 1.0
    shape = (i ** (-al)) * ((n - i + 1.0) ** (-ar))
    total = shape.sum()
    if total < 1e-12:
        shape = np.random.dirichlet(np.ones(n))
    else:
        shape = shape / total
    return shape


def _generate_initial_candidates(n: int,
                                 best_seq: Optional[np.ndarray],
                                 hist_seq: Optional[np.ndarray]) -> List[np.ndarray]:
    """
    Produce a modestly sized, diverse set of starting points for length n.
    All returned arrays are non‑negative and sum to 1.
    """
    cand: List[np.ndarray] = []

    # 1) warm‑starts from the global best (exact length or interpolated)
    if best_seq is not None and best_seq.shape[0] == n:
        cand.append(best_seq.copy())
    if best_seq is not None and best_seq.shape[0] != n:
        cand.append(_interpolate_seq(best_seq, n))

    # 2) historic sequence, if any
    if hist_seq is not None:
        if hist_seq.shape[0] > n:
            tmp = hist_seq[:n]
        else:
            tmp = np.pad(hist_seq, (0, n - hist_seq.shape[0]))
        tmp = np.maximum(tmp, 0.0)
        tot = tmp.sum()
        if tot < 1e-12:
            tmp = np.random.dirichlet(np.ones(n))
        else:
            tmp = tmp / tot
        cand.append(tmp)

    # 3) pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # 4) edge‑concentrated power‑law shapes ((i+1)*(n-i))**α , α<0
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 5):
        shape = np.power(prod, alpha, where=prod > 0,
                         out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # 5) two‑Gaussian edge shapes
    for gamma in (0.2, 0.5, 1.0, 2.0):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape = shape / shape.sum()
        cand.append(shape)

    # 6) inverse‑root edge shapes (i+1)**‑α + (n‑i)**‑α
    for alpha in (0.5, 0.8, 1.0):
        i = np.arange(n, dtype=np.float64) + 1.0
        shape = np.power(i, -alpha) + np.power(n - i + 1.0, -alpha)
        shape = shape / shape.sum()
        cand.append(shape)

    # 7) arcsine‑type shape (uniform sum distribution in continuous limit)
    cand.append(_arcsine_shape(n))

    # 8) random product‑power‑law shapes (different left/right exponents)
    for _ in range(3):
        al = np.random.uniform(0.1, 2.0)
        ar = np.random.uniform(0.1, 2.0)
        cand.append(_product_powerlaw_shape(n, al, ar))

    # 9) three‑Gaussian peaks (fixed positions) and random‑peak mixtures
    x = np.arange(n, dtype=np.float64) / (n - 1)
    for sigma in (0.05, 0.10, 0.20):
        peaks = sum(np.exp(-0.5 * ((x - p) / sigma) ** 2) for p in (0.25, 0.5, 0.75))
        cand.append(peaks / peaks.sum())
    # random‑peak mixtures
    for _ in range(2):
        num_peaks = np.random.randint(2, 5)
        mix = np.zeros_like(x)
        for __ in range(num_peaks):
            mu = np.random.rand()
            sig = np.random.uniform(0.02, 0.25)
            mix += np.exp(-0.5 * ((x - mu) / sig) ** 2)
        cand.append(mix / mix.sum())

    # 10) sparse binary support (≈√n points, equal mass)
    k = max(2, int(np.sqrt(n)))
    idxs = np.random.choice(n, size=k, replace=False)
    sparse = np.zeros(n, dtype=np.float64)
    sparse[idxs] = 1.0 / k
    cand.append(sparse)

    # Shuffle and keep a limited number
    np.random.shuffle(cand)
    return cand[:15]   # keep a modest size to stay within the budget

# Optimisation primitives
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   fft_len_conv: int,
                   fft_len_corr: int,
                   deadline: float,
                   max_iters: int) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate (log‑sum‑exp).
    Returns the best (a, value) encountered before the deadline.
    """
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.03

    best_a = a.copy()
    best_val = _objective(a, n, fft_len_conv)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # simplex via soft‑max
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # smooth‑max weights (log‑sum‑exp)
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        ws = w.sum()
        if ws == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws

        # gradient of the surrogate (exact)
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        cur_val = _objective(a, n, fft_len_conv)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t:06d}] new best {cur_val:.6f}")

        # random shake if stuck for a while
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


def _subgrad_optimize(a0: np.ndarray,
                      n: int,
                      fft_len_conv: int,
                      fft_len_corr: int,
                      deadline: float,
                      init_val: Optional[float] = None,
                      max_iters: int = 2000) -> Tuple[np.ndarray, float]:
    """
    Exact sub‑gradient (exponentiated‑gradient) optimisation on the true max.
    """
    a = a0 / a0.sum()
    best_a = a.copy()
    if init_val is None:
        best_val = _objective(a, n, fft_len_conv)
    else:
        best_val = init_val
    last_improve = time.time()
    eta = 0.6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        cur_val = _objective(a, n, fft_len_conv)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it}] new best {cur_val:.6f}")

        if time.time() - last_improve > 4.0:
            break

    return best_a, best_val


def _hill_climb_optimize(a0: np.ndarray,
                         n: int,
                         fft_len_conv: int,
                         deadline: float,
                         init_val: Optional[float] = None,
                         max_iters: int = 2000) -> Tuple[np.ndarray, float]:
    """
    Simple cheap local search – move a small fraction of mass between two
    random coordinates and keep the move if it improves the objective.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_conv) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2:
            continue

        eps = np.random.rand() * 0.05 * a[i1]      # up to 5 % of mass at i1
        if eps <= 0.0:
            continue
        new_a = a.copy()
        new_a[i1] -= eps
        new_a[i2] += eps
        if new_a[i1] < -1e-15:
            continue
        new_a /= new_a.sum()

        new_val = _objective(new_a, n, fft_len_conv)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            if it % 50 == 0:
                print(f"[n={n} HillClimb it={it}] new best {new_val:.6f}")
        else:
            # occasional uphill move to escape tiny plateaus
            if np.random.rand() < 0.001:
                a = new_a

    return best_a, best_val


def _simulated_annealing_optimize(a0: np.ndarray,
                                  n: int,
                                  fft_len_conv: int,
                                  deadline: float) -> Tuple[np.ndarray, float]:
    """
    Simulated‑annealing fine‑tuner.
    Randomly transfers a fraction of mass between two positions,
    accepting worsening moves with probability exp(‑Δ/T).
    The temperature decays exponentially.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_conv)
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    # Rough estimate of the number of iterations we can afford
    est_it = max(2000, int((deadline - start) / 0.0004))
    for it in range(est_it):
        if time.time() > deadline:
            break

        # exponential temperature schedule
        frac = it / est_it
        T = max(1e-8, cur_val * 0.05 * np.exp(-5.0 * frac))

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # propose moving a fraction of mass from i to j
        max_move = min(0.5, a[i])                # never move more than half of the source mass
        eps = np.random.rand() * max_move * (0.5 + 0.5 * (1.0 - frac))
        if eps <= 0.0:
            continue
        new_a = a.copy()
        new_a[i] -= eps
        new_a[j] += eps
        # sum stays exactly 1, no re‑normalisation needed

        new_val = _objective(new_a, n, fft_len_conv)
        delta = new_val - cur_val
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()

    print(f"[n={n} SA] finished – best value {best_val:.6f}")
    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> List[float]:
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: Optional[np.ndarray] = None
    best_val = float("inf")

    # Lengths to explore – start modest, finish with the largest allowed.
    lengths = [1024, 2048, 4096, 8192, 16384]

    # Pre‑compute FFT lengths for convolution and correlation.
    conv_fft = {}
    corr_fft = {}
    for n in lengths:
        conv_len = 2 * n - 1
        conv_fft[n] = 1 << int(np.ceil(np.log2(conv_len)))
        corr_fft[n] = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

    # Load a historic solution if it exists.
    hist_seq: Optional[np.ndarray] = None
    if "height_sequence_1" in globals():
        tmp = globals()["height_sequence_1"]
        if isinstance(tmp, (list, tuple, np.ndarray)):
            hist_seq = np.asarray(tmp, dtype=np.float64)

    # ------------------------------------------------------------------
    # MAIN optimisation loop over the chosen lengths
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        fft_len_c = conv_fft[n]
        fft_len_x = corr_fft[n]

        candidates = _generate_initial_candidates(n, best_seq, hist_seq)

        # Adam iteration budget shrinks with n (as in the reference code)
        max_adam_iters = int(15000 * 1024 / n) + 2000

        for cand_id, a_start in enumerate(candidates):
            if time.time() > deadline:
                break

            # ---- Phase 1 – smooth‑max Adam ----
            a_adam, val_adam = _adam_optimize(
                a_start, n, fft_len_c, fft_len_x, deadline, max_adam_iters
            )

            # ---- Phase 2 – exact sub‑gradient ----
            a_sub, val_sub = _subgrad_optimize(
                a_adam, n, fft_len_c, fft_len_x, deadline, init_val=val_adam
            )

            # ---- Phase 3 – cheap hill‑climb ----
            a_hc, val_hc = _hill_climb_optimize(
                a_sub, n, fft_len_c, deadline, init_val=val_sub
            )

            # Update global best
            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.copy()
                print(f"*** Global new best (n={n} cand={cand_id}) value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional long‑run refinement on the global best (if time permits)
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        fft_len_c = conv_fft.get(n,
                                 1 << int(np.ceil(np.log2(2 * n - 1))))
        fft_len_x = corr_fft.get(n,
                                 1 << int(np.ceil(np.log2(3 * n))))
        print("\n=== Long‑run refinement of the global best ===")
        refined_seq, refined_val = _adam_optimize(
            best_seq, n, fft_len_c, fft_len_x, deadline,
            max_iters=int(1.5 * (15000 * 1024 / n) + 3000)
        )
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq.copy()
            print(f"*** Refined global best value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing fine‑tuning if any time remains
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        fft_len_c = conv_fft.get(n,
                                 1 << int(np.ceil(np.log2(2 * n - 1))))
        print("\n=== Simulated‑annealing refinement (final stage) ===")
        best_seq, best_val = _simulated_annealing_optimize(
            best_seq, n, fft_len_c, deadline
        )
        print(f"*** After SA best value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (unlikely)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Ensure we return a plain Python list (as required by the external evaluator)
    return best_seq.tolist()
