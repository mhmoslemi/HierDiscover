        if time.time() >= deadline: break

        # compute a = target_sum * p
        a = target_sum * p

        # compute convolution using FFT
        Nfft = 1 << ((2*n-1).bit_length())
        A = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A*A, n=Nfft)[:2*n-1]
        M = conv.max()

        # keep best
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print...

        # compute gradient: subgradient of max
        # mask indices
        epsilon = 1e-12
        mask = conv >= M - epsilon
        cnt = mask.sum()
        w = mask.astype(float) / cnt

        # weighted reverse convolution to get rev (size n)
        rev = _weighted_rev(a, w, Nfft)

        grad_a = 2 * rev  # shape n

        # gradient w.r.t p 
        # a = target_sum * p => grad_p = target_sum * grad_a
        grad_p = target_sum * grad_a

        # exponentiated gradient update
        # step size schedule: eta = eta0 / sqrt(t)
        eta0 = 0.2
        eta = eta0 / math.sqrt(t)

        # compute update
        # p = p * exp(-eta * grad_p)
        # to avoid overflow, we can clip grad_p
        update_factor = np.exp(-eta * grad_p)
        p = p * update_factor
        # renorm:
        sum_p = p.sum()
        if sum_p == 0:
            # reinitialize to uniform
            p = np.full(n, 1.0/n)
        else:
            p = p / sum_p

    return best_seq, best_score

"""...
Propose candidate using exponentiated gradient descent and incremental local refinement.

import time, math, numpy as np

# helper functions

def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev

def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())

def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    init = []

    # uniform
    init.append(np.full(n, target_sum / n))

    # Dirichlet with various concentration
    for alpha in (0.5, 1.0, 2.0, 5.0):
        g = rng.gamma(alpha, 1.0, size=n)
        init.append(g / g.sum() * target_sum)

    # exponential decay
    rate = rng.uniform(0.001, 0.01)
    exp_vec = np.exp(-rate * np.arange(n))
    init.append(exp_vec / exp_vec.sum() * target_sum)

    # reversed exponential
    init.append(exp_vec[::-1] / exp_vec.sum() * target_sum)

    # Gaussian window
    sigma = rng.uniform(0.1, 0.3) * n
    x = np.arange(n)
    gauss = np.exp(-0.5 * ((x - (n-1)/2) / sigma) ** 2)
    init.append(gauss / gauss.sum() * target_sum)

    # Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # triangular
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    init.append(tri / tri.sum() * target_sum)

    return init

def _exp_grad_optimize(p0: np.ndarray, n: int, target_sum: float, max_iters: int,
                       deadline: float, best_seq: np.ndarray, best_score: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent on simplex) minimizing max(a*a).
    p0 is a probability vector (sum 1). Returns best exact vector (a) and its max.
    """
    # copy probability
    p = p0.copy()
    # FFT size for 2n-1 convolution
    Nfft = 1 << ((2 * n - 1).bit_length())
    # schedule for learning rate
    eta0 = 0.5

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        a = target_sum * p
        # compute convolution
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:8d}] new best max={M:.6f}")

        # subgradient of max via uniform weight over active positions
        eps = 1e-12
        active = conv >= M - eps
        cnt = active.sum()
        w = active.astype(float) / max(cnt, 1)

        # weighted reverse convolution to get rev
        rev = _weighted_rev(a, w, Nfft)

        # gradient w.r.t a
        grad_a = 2.0 * rev
        # gradient w.r.t probabilities p (since a = target_sum * p)
        grad_p = target_sum * grad_a

        # learning rate schedule (sqrt decay)
        eta = eta0 / math.sqrt(it)

        # exponentiated update; cap magnitude to avoid overflow
        # Clip grad to a reasonable range
        clip = 10.0 / eta
        g_clip = np.clip(grad_p, -clip, clip)
        factor = np.exp(-eta * g_clip)
        p = p * factor
        # renormalize
        sum_p = p.sum()
        if sum_p == 0 or not np.isfinite(sum_p):
            p = np.full(n, 1.0 / n)
        else:
            p /= sum_p

    return best_seq, best_score

def _incremental_refine(a: np.ndarray, conv: np.ndarray, max_steps: int,
                        deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Fast hill‑climbing that moves a small fraction of mass between two indices.
    The convolution is updated incrementally, costing O(n) per step.
    Returns improved a and its max convolution value.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_conv = conv.copy()
    cur_max = float(cur_conv.max())
    # Annealing schedule
    for step in range(max_steps):
        if time.time() >= deadline:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur_a[i] < 1e-12:
            continue
        # move a random fraction (1‑3 %)
        frac = rng.uniform(0.01, 0.03)
        delta = cur_a[i] * frac
        if delta <= 0.0:
            continue

        # apply incremental update using original array before change
        # shift contributions
        # will use copy of cur_a for reading; avoid affecting later slices
        a_before = cur_a
        # update convolution in place
        # subtract contribution from i
        cur_conv[i:i+n] -= 2.0 * delta * a_before
        # add contribution from j
        cur_conv[j:j+n] += 2.0 * delta * a_before
        # delta^2 contributions
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j
        cur_conv[idx2i] += delta * delta
        cur_conv[idx2j] += delta * delta
        cur_conv[idxij] -= 2.0 * delta * delta

        # update a
        cur_a[i] -= delta
        cur_a[j] += delta

        new_max = float(cur_conv.max())
        # accept if improves or via annealing
        if new_max < cur_max - 1e-12:
            cur_max = new_max
        else:
            # temperature decreasing linearly
            T = max(0.001 * cur_max, 1e-6) * (1.0 - step / max_steps)
            prob = math.exp(-(new_max - cur_max) / max(T, 1e-12))
            if rng.random() < prob:
                cur_max = new_max
            else:
                # revert changes
                # we need to undo update; easiest: recompute from scratch (cost O(n log n))
                # for simplicity, reject move and revert a and conv
                # recompute
                cur_a[i] += delta
                cur_a[j] -= delta
                # full recompute conv
                Nfft = 1 << ((2 * n - 1).bit_length())
                A_fft = np.fft.rfft(cur_a, n=Nfft)
                cur_conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
                cur_max = float(cur_conv.max())
        # optional progress
        if step % 50000 == 0 and step > 0:
            print(f"[refine] step {step} current max={cur_max:.6f}")

    return cur_a, cur_max

            interp = np.interp(new_x, old_x, base)
            interp = np.maximum(interp, 0.0)
            interp = interp / interp.sum() * target_sum
            init_vectors.append(interp)

                deadline=deadline,
                best_seq=best_seq if best_seq is not None else vec,
                best_score=best_score,
                rng=rng,
            )
            print(f"  completed start {idx+1}/{len(init_vectors)} for n={n}")

    # If we have a candidate, perform a final fast local refinement
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0:
            # recompute convolution for the best sequence
            n = best_seq.shape[0]
            Nfft = 1 << ((2 * n - 1).bit_length())
            A_fft = np.fft.rfft(best_seq, n=Nfft)
            conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
            max_steps = int(300_000 * (2100 / n))
            max_steps = max(50_000, max_steps)
            print(f"[final refinement] length={n}, steps≈{max_steps}")
            best_seq, final_max = _incremental_refine(
                a=best_seq,
                conv=conv,
                max_steps=max_steps,
                deadline=deadline,
                rng=rng,
            )
            best_score = final_max

    # Fallback to uniform if nothing found
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        print("[fallback] using uniform sequence")

    print(f"[done] best max convolution = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()

"""Propose a step‑function that minimises ``evaluate_sequence``.

The algorithm works in two stages:

1. **Exponentiated‑gradient descent on the probability simplex**  
   – Initialise many diverse probability vectors (uniform, Dirichlet,
     exponential, Gaussian, Hann, triangular).  
   – At each iteration compute the exact self‑convolution via an FFT,
     take the maximal entry and obtain a sub‑gradient (uniform weight on
     all active positions).  
   – Update the probabilities by a mirror‑descent step
     ``p_i ← p_i·exp(-η·∇_i)`` and renormalise.  
   This quickly drives the max‑convolution down while respecting the
   non‑negativity and Σa = √(2 n) constraints.

2. **Fast stochastic local refinement**  
   – Starting from the best vector found, repeatedly transfer a small
     fraction of mass from one index to another.  
   – The convolution is updated **incrementally** (O(n) per move) using the
     analytical rank‑1 update, so many cheap moves can be tried.  
   – Improvements are always accepted; worsening moves are accepted with a
     linearly‑decreasing simulated‑annealing probability.

Both stages are time‑limited; the remaining budget is spent on the final
refinement.  The returned list contains non‑negative numbers and the
objective value is evaluated by ``evaluate_sequence`` outside this module.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)

def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k·a_{k‑i} (indices outside [0,n) are ignored)
    using FFTs.  The result has length len(a) (= n).
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)      # reverse w before FFT
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # keep linear part (length = len(w)+len(a)-1 = 3·n‑2)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2                # index of the first element of rev
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Create a small but diverse pool of starting vectors (already normalised)."""
    init = []

    # 0) uniform
    init.append(np.full(n, target_sum / n))

    # 1) Dirichlet with several concentration parameters
    for alpha in (0.5, 1.0, 2.0, 5.0):
        g = rng.gamma(alpha, 1.0, size=n)
        init.append(g / g.sum() * target_sum)

    # 2) exponential decay
    rate = rng.uniform(0.001, 0.01)
    exp_vec = np.exp(-rate * np.arange(n))
    init.append(exp_vec / exp_vec.sum() * target_sum)

    # 3) reversed exponential
    init.append(exp_vec[::-1] / exp_vec.sum() * target_sum)

    # 4) Gaussian window (σ proportional to n)
    sigma = rng.uniform(0.1, 0.3) * n
    x = np.arange(n)
    gauss = np.exp(-0.5 * ((x - (n - 1) / 2) / sigma) ** 2)
    init.append(gauss / gauss.sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) triangular shape
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    init.append(tri / tri.sum() * target_sum)

    return init


def _exp_grad_optimize(p0: np.ndarray,
                       n: int,
                       target_sum: float,
                       max_iters: int,
                       deadline: float,
                       best_seq: np.ndarray,
                       best_score: float,
                       rng: np.random.Generator):
    """
    Exponentiated‑gradient (mirror descent on the simplex) minimising the
    maximal entry of the self‑convolution.  Returns the best exact vector
    encountered and its max‑convolution value.
    """
    # initialise probability vector (sum 1)
    p = p0.copy()

    # FFT size – must be ≥ 3·n‑2 for the weighted reverse convolution
    Nfft = 1 << ((3 * n - 2).bit_length())

    eta0 = 0.5                     # initial step size for the exponentiated update

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        a = target_sum * p                     # actual coefficients (Σa = target_sum)

        # ----- exact self‑convolution (full) -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # keep globally best vector
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:8d}] new best max={M:.6f}")

        # ----- sub‑gradient of the max‑convolution -----
        eps = 1e-12
        active = conv >= M - eps                # Boolean mask of active maxima
        cnt = active.sum()
        w = active.astype(float) / max(cnt, 1)  # uniform weight on the active set

        # weighted reverse convolution gives the gradient w.r.t. a
        rev = _weighted_rev(a, w, Nfft)
        grad_a = 2.0 * rev                       # ∂max/∂a_i = 2·rev_i

        # gradient w.r.t. the probability simplex (a = target_sum·p)
        grad_p = target_sum * grad_a

        # ----- exponentiated‑gradient update (mirror descent on the simplex) -----
        eta = eta0 / math.sqrt(it)              # 1/√t decay
        # clip the gradient to keep exp(·) numerically stable
        clip_val = 10.0 / max(eta, 1e-12)
        g_clipped = np.clip(grad_p, -clip_val, clip_val)
        factor = np.exp(-eta * g_clipped)
        p = p * factor
        s = p.sum()
        if s == 0 or not np.isfinite(s):
            p = np.full(n, 1.0 / n)
        else:
            p /= s

    return best_seq, best_score


def _incremental_refine(a: np.ndarray,
                        conv: np.ndarray,
                        max_steps: int,
                        deadline: float,
                        rng: np.random.Generator):
    """
    Fast stochastic hill‑climbing that moves a small fraction of mass between
    two indices while updating the convolution incrementally (O(n) per step).
    Returns an improved vector and its maximal convolution entry.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_conv = conv.copy()
    cur_max = float(cur_conv.max())

    for step in range(max_steps):
        if time.time() >= deadline:
            break

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or cur_a[i] < 1e-12:
            continue

        # transfer a random fraction (1‑3 %) of the source mass
        frac = rng.uniform(0.01, 0.03)
        delta = cur_a[i] * frac
        if delta <= 0.0:
            continue

        # ----- apply the rank‑1 update to the convolution (in‑place) -----
        # note: we need the *old* a values for the shift contributions
        a_old = cur_a

        # remove contribution of the departing mass
        cur_conv[i:i + n] -= 2.0 * delta * a_old
        # add contribution of the arriving mass
        cur_conv[j:j + n] += 2.0 * delta * a_old
        # delta² terms (indices are always inside the array)
        cur_conv[2 * i] += delta * delta
        cur_conv[2 * j] += delta * delta
        cur_conv[i + j] -= 2.0 * delta * delta

        # update the vector itself
        cur_a[i] -= delta
        cur_a[j] += delta

        new_max = float(cur_conv.max())

        # ----- acceptance test -----
        if new_max < cur_max - 1e-12:
            cur_max = new_max
        else:
            # annealing temperature (linear decay)
            T = max(0.001 * cur_max, 1e-6) * (1.0 - step / max_steps)
            prob = math.exp(-(new_max - cur_max) / max(T, 1e-12))
            if rng.random() < prob:
                cur_max = new_max
            else:
                # reject: revert to the previous state (full recompute)
                cur_a[i] += delta
                cur_a[j] -= delta
                Nfft = 1 << ((2 * n - 1).bit_length())
                A_fft = np.fft.rfft(cur_a, n=Nfft)
                cur_conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
                cur_max = float(cur_conv.max())

        if step % 100_000 == 0 and step > 0:
            print(f"[refine] step {step:7d} current max={cur_max:.6f}")

    return cur_a, cur_max
