    a = a0.copy()
    best_val = init_val or _objective(a,...)
    best_a = a.copy()
    eta = 0.1  # initial step size

    for it in range(...):
        if time.time()>deadline: break
        conv = _full_convolution(a,...)
        max_c = conv.max()
        active = (conv == max_c)
        w = np.where(active, 1.0/active.sum(), 0.0)
        grad = 2.0 * _cross_correlation(w, a, fft_len_corr)
        # step direction negative gradient
        # backtracking line search
        step = eta
        for ls in range(10):
            a_candidate = a - step * grad
            a_candidate = simplex_projection(a_candidate)
            val_candidate = _objective(a_candidate, n, fft_len_conv)
            if val_candidate < best_val:
                # accept
                a = a_candidate
                best_val = val_candidate
                best_a = a_candidate.copy()
                break
            else:

"""
Propose a non‑negative step‑function (a list of floats) that minimizes

        F(a) = 2·n·max(a∗a) / (sum a)²

The optimizer is a multi‑stage search:

1.  Diverse warm‑starts (Dirichlet, edge‑power‑law, inverse‑root,
    Gaussian mixtures, arcsine‑type shapes, etc.).
2.  Adam on a smooth‑max surrogate (exponential soft‑max of the
    convolution).  The soft‑max temperature is increased gradually.
3.  Exact projected‑gradient descent on the true max‑convolution.
    The gradient is obtained from the active set of indices where the
    convolution attains its maximum and each gradient step is projected
    onto the probability simplex.
4.  A cheap hill‑climbing local search (mass transfer between two
    positions) for final polishing.
5.  If time permits, a long‑run refinement of the best sequence using
    a slower Adam schedule.

All phases respect a global deadline (default 1000 s).  The code
contains no closures or filesystem access and can be run as a drop‑in
replacement for the previous `propose_candidate` implementation.
"""

import time
import numpy as np

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)−1) using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) = n."""
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)²."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x ≥ 0, sum x = 1}."""
    # Implementation of the algorithm from Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence by linear interpolation and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new


def _generate_initial_candidates(
    n: int,
    best_seq: np.ndarray | None,
    hist_seq: np.ndarray | None,
) -> list[np.ndarray]:
    """Produce a diverse set of starting points for length n.
    All returned arrays are non‑negative and sum to 1."""
    cand = []

    # 1) Warm‑starts from the current global best (exact length & interpolated)
    if best_seq is not None:
        if best_seq.shape[0] == n:
            cand.append(best_seq.copy())
        else:
            cand.append(_interpolate_seq(best_seq, n))

    # 2) Historic solution if supplied
    if hist_seq is not None:
        if hist_seq.shape[0] > n:
            tmp = hist_seq[:n]
        else:
            tmp = np.pad(hist_seq, (0, n - hist_seq.shape[0]))
        tmp = np.maximum(tmp, 0.0)
        tot = tmp.sum()
        if tot < 1e-12:
            tmp = np.random.dirichlet(np.ones(n))
        else:
            tmp = tmp / tot
        cand.append(tmp)

    # 3) Pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # 4) Edge‑concentrated power‑law shapes  ((i+1)*(n-i))**α , α<0
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-3.0, -0.2, 7):
        shape = np.power(prod, alpha, where=prod > 0,
                         out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # 5) Inverse‑root / arcsine–type shapes  (i+1)**‑p + (n‑i)**‑p
    for p in (0.4, 0.5, 0.6, 0.8, 1.0):
        i = np.arange(n, dtype=np.float64) + 1.0
        shape = np.power(i, -p) + np.power(n - i + 1.0, -p)
        cand.append(shape / shape.sum())

    # 6) Two‑Gaussian edge shapes
    for gamma in (0.1, 0.2, 0.5, 1.0):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        cand.append(shape / shape.sum())

    # 7) Three‑Gaussian interior peaks
    for sigma in (0.04, 0.07, 0.12):
        x = np.arange(n, dtype=np.float64) / (n - 1)
        peaks = sum(np.exp(-0.5 * ((x - p) / sigma) ** 2) for p in (0.25, 0.5, 0.75))
        cand.append(peaks / peaks.sum())

    # 8) Mixture of edge‑power‑law and uniform
    for alpha in (-1.0, -0.7):
        edge = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        mix = 0.7 * edge + 0.3 * np.ones_like(edge)
        cand.append(mix / mix.sum())

    # 9) Randomised perturbations of the arcsine shape
    base = 1.0 / np.sqrt(np.maximum(idx * (n - idx + 1), 1e-12))
    for _ in range(3):
        pert = base * (1.0 + 0.2 * (np.random.rand(n) - 0.5))
        pert = np.maximum(pert, 0.0)
        cand.append(pert / pert.sum())

    np.random.shuffle(cand)
    # keep a reasonable number to stay inside the time budget
    return cand[:18]

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_c: int,
    fft_len_x: int,
    deadline: float,
    max_iters: int,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the exponential‑smooth‑max surrogate.
    Returns the best (a, value) encountered before the deadline.
    """
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.04
    lam = 0.5                # temperature of the smooth‑max
    lam_factor = 1.04

    best_a = a.copy()
    conv = _full_convolution(a, fft_len_c)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ----- simplex via soft‑max -----
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ----- smooth‑max weights -----
        conv = _full_convolution(a, fft_len_c)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        ws = w.sum()
        if ws == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws

        # ----- gradient of the surrogate -----
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_x)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        cur_val = 2.0 * n * max_c / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t:06d}] new best {cur_val:.6f}")

        # occasional random shake if stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 8e-3
            last_improve = time.time()

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_c: int,
    fft_len_x: int,
    deadline: float,
    max_iters: int = 2500,
    init_val: float | None = None,
) -> tuple[np.ndarray, float]:
    """
    Projected‑gradient descent on the *exact* max‑convolution objective.
    Uses the true active‑set sub‑gradient, a back‑tracking line‑search
    and Euclidean projection onto the simplex.
    """
    a = a0 / a0.sum()
    if init_val is None:
        best_val = _objective(a, n, fft_len_c)
    else:
        best_val = init_val
    best_a = a.copy()

    eta = 0.15               # initial step size
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_c)
        max_c = conv.max()
        # active set (weights uniform over all maximisers)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, fft_len_x)  # ∇ max(conv)

        # ----- line‑search -----
        step = eta
        for _ in range(12):
            cand = a - step * grad
            cand = _simplex_projection(cand)
            val = _objective(cand, n, fft_len_c)
            if val < best_val:
                a = cand
                best_val = val
                best_a = cand.copy()
                last_improve = time.time()
                step *= 1.02            # slightly increase after success
                break
            else:
                step *= 0.5
        else:
            # no improvement after line‑search – shrink step
            step *= 0.5

        eta = step

        # early‑exit if nothing improved for several seconds
        if time.time() - last_improve > 6.0:
            break

    return best_a, best_val


def _hill_climb_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_c: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Simple cheap local search: move a small amount of mass between two random
    coordinates and keep the move if it improves the objective.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_c) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2:
            continue

        # move up to 5 % of the probability mass from i1 to i2
        eps = np.random.rand() * 0.05 * a[i1]
        if eps <= 0.0:
            continue
        new_a = a.copy()
        new_a[i1] -= eps
        new_a[i2] += eps
        if new_a[i1] < -1e-15:
            continue
        new_a /= new_a.sum()

        new_val = _objective(new_a, n, fft_len_c)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            last_improve = time.time()
            if it % 50 == 0:
                print(f"[n={n} HillClimb it={it}] new best {new_val:.6f}")
        else:
            # occasional uphill move to escape tiny plateaus
            if np.random.rand() < 0.001:
                a = new_a

        if time.time() - last_improve > 5.0:
            # no progress for a while – stop early
            break

    return best_a, best_val

        fft_len_c = conv_fft[n]
        fft_len_x = corr_fft[n]

        candidates = _generate_initial_candidates(n, best_seq, hist_seq)

        # number of Adam iterations scales down with length
        max_adam_iters = int(15000 * 1024 / n) + 2000

        for cand_id, a_start in enumerate(candidates):
            if time.time() > deadline:
                break

            # ---- Phase 1 – smooth‑max Adam ----
            a_adam, val_adam = _adam_optimize(
                a_start, n, fft_len_c, fft_len_x, deadline, max_adam_iters
            )

            # ---- Phase 2 – exact projected‑gradient descent ----
            a_pgd, val_pgd = _pgd_optimize(
                a_adam, n, fft_len_c, fft_len_x, deadline,
                max_iters=1800, init_val=val_adam
            )

            # ---- Phase 3 – cheap hill‑climb ----
            a_hc, val_hc = _hill_climb_optimize(
                a_pgd, n, fft_len_c, deadline, init_val=val_pgd
            )

            # Update global best if improved.
            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.copy()
                print(f"*** Global new best (n={n} cand={cand_id}) value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional long‑run refinement on the global best (if any time left)
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        fft_len_c = conv_fft.get(n, 1 << int(np.ceil(np.log2(2 * n - 1))))
        fft_len_x = corr_fft.get(n, 1 << int(np.ceil(np.log2(3 * n))))
        print("\n=== Long‑run refinement of the global best ===")
        # Longer Adam run with a slower learning rate and more iterations.
        refined_seq, refined_val = _adam_optimize(
            best_seq, n, fft_len_c, fft_len_x, deadline,
            max_iters=int(1.5 * (15000 * 1024 / n) + 3000)
        )
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq.copy()
            print(f"*** Refined global best value={best_val:.6f}")

        # Final projected‑gradient polishing
        refined_seq2, refined_val2 = _pgd_optimize(
            best_seq, n, fft_len_c, fft_len_x, deadline,
            max_iters=2500, init_val=best_val
        )
        if refined_val2 < best_val:
            best_val = refined_val2
            best_seq = refined_seq2.copy()
            print(f"*** Final PGD polish improved value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (unlikely)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator.
    print(f"\nBest value achieved: {best_val:.6f} with length {best_seq.shape[0]}")
    return best_seq.tolist()
