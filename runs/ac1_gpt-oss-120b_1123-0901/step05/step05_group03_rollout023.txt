def solve_lp_for_g(f, M, deadline):
    n = len(f)
    col = np.concatenate([f, np.zeros(n-1)])
    row = np.zeros(n)
    A = scipy.linalg.toeplitz(col, row)  # shape (2n-1, n)
    b = np.full(2 * n - 1, M)
    c = -np.ones(n)  # maximize sum(g)
    # call linprog
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        raise ValueError(f"LP failed: {res.message}")
    return res.x

    # precompute symmetrical pairs list for convenience
    half = n // 2

        i = np.random.randint(0, half)

        # choose delta as up to 10% of source

        new_seq[j] += delta

            best = new_seq

    best_seq = None
    best_val = float('inf')

    conv_fft_len = _next_pow2(2*n - 1)
    corr_fft_len = _next_pow2(3*n - 2)

    warm_start = globals().get('height_sequence_1')
    if warm_start is not None:
        warm = np.asarray(warm_start, dtype=np.float64)
        if warm.shape[0] != n:
            if warm.shape[0] > n:
                warm = warm[:n]
            else:
                warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm /= warm.sum()
        else:
            warm = None
    else:
        warm = None

    # initial candidates: warm (and rotations), random dirichlet, some param shapes
    candidates = []
    if warm is not None:
        candidates.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    candidates.append(np.random.dirichlet(np.ones(n)))
    # power-law edge shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.5, -0.5, 7):
        shape = np.where(prod > 0, prod**alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)
    # double-Gaussian edges
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.1, 0.2, 0.5, 1.0):

        shape /= shape.sum()

    shape = np.maximum(shape, 0.0)

    candidates.append(shape)

    # Optimize each candidate using deep optimizer
    for cid, cand in enumerate(candidates):

            a0 = np.random.dirichlet(np.ones(n))

            a0 = a0 / s

                                            deadline, best_seq, best_val)

            small_seq /= small_seq.sum()

"""
Hybrid search for a low‑score step‑function.

The algorithm combines three ideas:

1. Adam optimisation on a smooth‑max (log‑sum‑exp) surrogate of the
   objective, followed by an exponentiated‑gradient sub‑gradient phase.
   This reproduces the successful strategy from the previous run but
   with a few tuned hyper‑parameters and an early‑stop based on the
   global deadline.

2. A cheap linear‑programming “mass‑inflation’’ step (LP‑refine).  For a
   given distribution `f` we solve the linear program
       maximise  Σ g_j
       subject   f * g ≤ M   (M = max(f*f)),   g ≥ 0
   The solution `g` is mixed with `f` (convex combination) and the
   result is re‑normalised.  Repeating this step can lower the peak
   of the self‑convolution while keeping the total mass large.

3. A final greedy hill‑climbing phase that swaps tiny amounts of mass
   between symmetric positions.  The move is accepted only when it
   strictly improves the objective; many cheap iterations are performed
   while time permits.

All steps respect the 1000 s time budget.  The best sequence found is
returned as a plain list of non‑negative floats (the caller will invoke
`evaluate_sequence` to obtain the score).
"""

import time
import numpy as np
import scipy.linalg
import scipy.optimize

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a (full convolution)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, fft_len)
    return float(conv.max())


def _objective_val(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.shape[0]
    max_c = _max_convolution(a)
    s = a.sum()
    if s < 1e-12:  # safeguard – should never happen
        return float("inf")
    return float(2.0 * n * max_c / (s ** 2))


# ----------------------------------------------------------------------
def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp surrogate followed by an exponentiated‑gradient
    sub‑gradient phase.  The routine updates the global best sequence/value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                         # slightly larger than before
    lam = 0.5                         # temperature for smooth‑max
    lam_factor = 1.02

    last_improve = time.time()
    it = 0

    while time.time() < deadline:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for monitoring)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:05d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # occasional shake when stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        if it > 300_000:          # safety bound
            break

    # ------------------------------------------------------------------
    # Sub‑gradient (exponentiated‑gradient) phase – runs until time runs out
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()

    eta = 0.6
    sub_last_improve = time.time()
    for sub_it in range(30_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[Subgrad n={n} it={sub_it:05d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 8.0:
            break

    return global_best_seq, global_best_val


# ----------------------------------------------------------------------
def _solve_lp_for_g(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve:
        maximise   Σ g_j
        subject to  (f * g)[k] ≤ M   for all k = 0..2·n-2
                    g_j ≥ 0
    Returns the optimal g (density on the simplex).
    """
    n = f.shape[0]
    # Build convolution matrix A where A[k, j] = f_{k-j} if valid else 0.
    col = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
    row = np.zeros(n, dtype=np.float64)
    A = scipy.linalg.toeplitz(col, row)          # shape (2n-1, n)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)            # maximise Σ g_j  → minimise -Σ g_j
    bounds = [(0.0, None)] * n

    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                 method="highs", options={"presolve": True})
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return np.maximum(res.x, 0.0)                # safety


def _lp_refine(f: np.ndarray, max_steps: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Iteratively mix f with the LP solution g (mass‑inflation) while time permits.
    Returns the best sequence found and its objective value.
    """
    best_f = f.copy()
    best_val = _objective_val(best_f)

    for step in range(max_steps):
        if time.time() >= deadline:
            break
        M = _max_convolution(f)
        try:
            g = _solve_lp_for_g(f, M)
        except Exception as exc:
            print(f"[LP step={step}] failed: {exc}")
            break

        t = 0.30                     # mixing weight
        new_f = (1.0 - t) * f + t * g
        new_f = np.maximum(new_f, 0.0)
        s = new_f.sum()
        if s < 1e-12:
            new_f = np.random.dirichlet(np.ones_like(new_f))
            s = new_f.sum()
        new_f /= s

        val = _objective_val(new_f)
        if val < best_val:
            best_val = val
            best_f = new_f.copy()
            print(f"[LP refine step={step}] new best {val:.6f}")

        # move forward
        f = new_f

    return best_f, best_val


# ----------------------------------------------------------------------
def _hill_climb(seq: np.ndarray, max_iters: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple greedy local search: move a tiny amount of mass from one symmetric
    position to another, keeping the total sum constant.  Only improvements
    are kept.
    """
    n = seq.shape[0]
    half = n // 2
    best = seq.copy()
    best_val = _objective_val(best)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = np.random.randint(0, half)
        j = np.random.randint(0, half)
        if i == j:
            continue

        # transfer up to 10 % of the source mass
        delta = best[i] * np.random.rand() * 0.10
        if delta <= 0.0:
            continue
        if best[i] < delta:
            delta = best[i]

        new_seq = best.copy()
        new_seq[i] -= delta
        new_seq[n - 1 - i] -= delta
        new_seq[j] += delta
        new_seq[n - 1 - j] += delta

        new_seq = np.maximum(new_seq, 0.0)
        s = new_seq.sum()
        if s < 1e-12:
            continue
        new_seq /= s

        val = _objective_val(new_seq)
        if val < best_val:
            best_val = val
            best = new_seq
            if it % 2000 == 0:
                print(f"[HillClimb it={it}] new best {val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and returns the best sequence.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Primary optimisation size (the previous best used 4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Build initial candidate pool (diverse shapes)
    warm = None
    height_seq_1 = globals().get("height_sequence_1")
    if height_seq_1 is not None:
        warm = np.asarray(height_seq_1, dtype=np.float64)
        if warm.shape[0] != n:
            if warm.shape[0] > n:
                warm = warm[:n]
            else:
                warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm /= warm.sum()
        else:
            warm = None

    candidates: list[np.ndarray] = []

    if warm is not None:
        candidates.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    # random Dirichlet
    candidates.append(np.random.dirichlet(np.ones(n)))

    # power‑law edge shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.5, -0.5, 7):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.1, 0.2, 0.5, 1.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # sinusoidal positive half‑wave
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # ------------------------------------------------------------------
    # Optimise each candidate with the deep optimiser
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        s = a0.sum()
        if s < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / s

        best_seq, best_val = _deep_optimize(
            a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
        )
        print(f"[candidate {cid}] best so far {best_val:.6f}")

    # ------------------------------------------------------------------
    # LP refinement on a down‑sampled version if time permits
    if time.time() < deadline - 30.0:
        n_small = 1024
        if best_seq is not None:
            # simple block averaging to down‑sample
            factor = n // n_small
            small_seq = best_seq[: factor * n_small].reshape(n_small, factor).mean(axis=1)
        else:
            small_seq = np.random.dirichlet(np.ones(n_small))
        small_seq = np.maximum(small_seq, 0.0)
        small_seq /= small_seq.sum()

        # a few LP refinement steps
        small_seq, small_val = _lp_refine(small_seq, max_steps=8, deadline=deadline)

        # interpolate back to the original length
        x_old = np.linspace(0.0, 1.0, n_small, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n, endpoint=False)
        up_seq = np.interp(x_new, x_old, small_seq)
        up_seq = np.maximum(up_seq, 0.0)
        up_seq /= up_seq.sum()

        # optionally run deep optimiser again on the up‑sampled candidate
        if time.time() < deadline - 10.0:
            best_seq, best_val = _deep_optimize(
                up_seq, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
            )
        else:
            if best_seq is None or small_val < best_val:
                best_seq, best_val = up_seq, small_val

    # ------------------------------------------------------------------
    # Final greedy local search (hill climbing) if we still have time
    if best_seq is not None and time.time() < deadline - 5.0:
        max_hc_iters = 200_000
        best_seq, best_val = _hill_climb(best_seq, max_iters=max_hc_iters, deadline=deadline)
        print(f"[HillClimb] final best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to allowed range (non‑negative, ≤1000) before returning
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return best_seq.tolist()
