def _subgrad_optimize(a0, n, target_sum, deadline, max_iters, best_seq, best_score, rng):
    y = np.log(a0+1e-16)

    m, v = zeros, zeros
    beta1,beta2 =0.9,0.999
    epsadam = 1e-8
    lr0 = 0.08
    Nfft = 1 << ((3*n - 2).bit_length())
    for it in range(1,max_iters+1):
        if time.time() >= deadline:
            break
        logits = y - np.max(y)
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()
        a = target_sum * p

        # convolution and max
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[:2*n-1]
        M = conv.max()
        # compute mask of max indices (tolerance)
        mask = conv >= M - 1e-12
        cnt = mask.sum()
        if cnt == 0:
            w = np.ones_like(conv) / conv.size
        else:
            w = mask.astype(float) / cnt

        # gradient
        rev = _weighted_rev(a, w, Nfft)
        g_a = 2.0 * rev

        # gradient wrt logits
        avg = np.dot(p, g_a)
        g_y = target_sum * p * (g_a - avg)

        # Adam update
        m = beta1 * m + (1-beta1) * g_y
        v = beta2 * v + (1-beta2) * (g_y * g_y)
        m_hat = m / (1-beta1**it)
        v_hat = v / (1-beta2**it)
        lr = lr0 / (1 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # jitter occasional every 2000 iterations
        if it % 2000 == 0:
            y += rng.normal(scale=0.01, size=n)

        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

    return best_seq, best_score

def _gradient_mass_transfer_refine(a, deadline, rng):

    target_sum = a.sum()

    cur_score = float(np.max(np.fft.irfft(np.fft.rfft(cur_seq, n=1 << ((2*n - 1).bit_length())) * np.fft.rfft(cur_seq, n=1 << ((2*n - 1).bit_length())) )[:2*n-1]))

    # maybe compute conv each time

    step = 0
    max_steps = 300_000
    while time.time() < deadline and step < max_steps:
        # compute gradient at current
        A_fft = np.fft.rfft(cur_seq, n=1 << ((3*n - 2).bit_length()))   # Wait we need for rev length? For gradient we just need rev for max indices.
        conv = np.fft.irfft(A_fft * A_fft, n=1 << ((3*n - 2).bit_length()))[:2*n-1]
        M = conv.max()
        mask = conv >= M - 1e-12
        w = mask.astype(float) / max(mask.sum(), 1)
        rev = _weighted_rev(cur_seq, w, 1 << ((3*n - 2).bit_length()))
        g_a = 2.0 * rev
        # Choose source index i = argmax(g) (positive gradient -> increasing max), target j = argmin(g) (negative gradient)
        i = int(np.argmax(g_a))
        j = int(np.argmin(g_a))
        if i == j or cur_seq[i] < 1e-15:
            # random fallback
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                step += 1
                continue
        # transfer small fraction
        frac = rng.uniform(0.02, 0.1)  # moderate fraction
        delta = cur_seq[i] * frac
        if delta <= 0:
            step += 1
            continue
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # recompute max conv
        A_fft2 = np.fft.rfft(new_seq, n=1 << ((2*n - 1).bit_length()))
        conv2 = np.fft.irfft(A_fft2 * A_fft2, n=1 << ((2*n - 1).bit_length()))[:2*n-1]
        new_score = float(conv2.max())
        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine-grad] improved to {cur_score:.6f} step {step}")
        else:
            # optionally perform simulated annealing acceptance using temperature dec schedule
            T = max(cur_score * 0.01, 1e-5) * (1 - step / max_steps)
            if rng.random() < math.exp(-(new_score - cur_score) / max(T,1e-12)):
                cur_seq = new_seq
                cur_score = new_score
        step += 1
    return cur_seq, cur_score

"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.

Algorithm
---------
* Initialise a diverse pool of start vectors (uniform, Dirichlet, exponential,
  Hann/Blackman windows …).  If a historic construction ``height_sequence_1``
  exists, also interpolate it to the current length.
* Optimise each start with *hard‑max* sub‑gradient descent on the log‑its
  (exponentiated‑gradient / mirror‑descent).  For a given vector a the
  objective is  max_k (a∗a)[k]; its sub‑gradient is
      g = 2·rev   with   rev_i = Σ_{k∈Active} a_{k-i},
  where *Active* are the indices attaining the maximum.  The gradient is
  back‑propagated to the logits and updated by Adam.
  The hard‑max (instead of a soft‑max) makes each iteration cheaper
  (only one FFT + O(n) work) so we can afford many more updates.
* After the global budget is exhausted, a cheap *gradient‑directed mass‑transfer*
  refinement is performed: we repeatedly move a small fraction of mass from an
  index with large gradient contribution to one with a small (or negative)
  contribution, accepting only improving moves (or occasional simulated‑annealing
  steps).
* The best vector found (as a plain list) is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all are top‑level (no closures)

def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k * a_{k-i} (out‑of‑range indices ignored)
    using FFTs.  Length of ``w`` must be 2·n‑1 (full convolution length).
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)          # reverse w before FFT
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]    # linear part, length = 3 n‑2
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Return max_k (a∗a)[k] using an FFT of length Nfft."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Create a small diverse set of normalised start vectors."""
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0
             - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    init.append(black / black.sum() * target_sum)

    # 7) small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    # 8) triangular shape (peak in the centre)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    init.append(tri / tri.sum() * target_sum)

    return init

# Core optimisation primitives

def _hardmax_optimize(
    a0: np.ndarray,
    n: int,
    target_sum: float,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑its using a *hard* max (sub‑gradient) of the
    convolution.  Returns the best vector seen in this run together with its
    exact objective value (i.e. the maximal convolution entry).
    """
    eps = 1e-16
    y = np.log(a0 + eps)                # logits

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.08
    jitter_interval = 3000

    # FFT size for the linear convolution (needs room for length 3 n‑2)
    Nfft = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ------- current point on the simplex -------
        logits = y - np.max(y)                     # stability shift
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()
        a = target_sum * p

        # ------- exact convolution -------
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ------- active set (hard max) -------
        # treat all indices within 1e-12 of the maximum as active
        mask = conv >= M - 1e-12
        active = np.where(mask)[0]
        cnt_active = active.size

        # ------- sub‑gradient wrt a -------
        if cnt_active <= 5:        # cheap O(n) case
            rev = np.zeros_like(a)
            for k in active:
                low = max(0, k - (n - 1))
                high = min(k, n - 1)
                idx = np.arange(low, high + 1)
                rev[low:high + 1] += a[k - idx]
            rev /= cnt_active
        else:                      # fallback to FFT when many active indices
            w = np.zeros_like(conv)
            w[active] = 1.0 / cnt_active
            rev = _weighted_rev(a, w, Nfft)

        g_a = 2.0 * rev                     # ∇_a max(a∗a)

        # ------- chain rule to logits (mirror‑descent on the simplex) -------
        avg = np.dot(p, g_a)
        g_y = target_sum * p * (g_a - avg)

        # ------- Adam update -------
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # occasional jitter to shake out plateaux
        if it % jitter_interval == 0:
            y += rng.normal(scale=0.01, size=n)

        # keep global best
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:9d}] new best = {best_score:.6f}")

    return best_seq, best_score


def _gradient_mass_transfer_refine(
    a: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple deterministic refinement that repeatedly moves a small fraction of
    mass from the index contributing most to the current max to the index
    contributing least (according to the sub‑gradient).  Accepts only improving
    moves (with an occasional simulated‑annealing fallback).
    """
    n = a.shape[0]
    cur_seq = a.copy()
    target_sum = cur_seq.sum()
    # helper for fast max‑convolution
    def max_conv(x):
        Nfft = 1 << ((2 * n - 1).bit_length())
        A_fft = np.fft.rfft(x, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        return float(conv.max()), conv

    cur_score, cur_conv = max_conv(cur_seq)

    max_steps = 300_000
    step = 0
    while time.time() < deadline and step < max_steps:
        # recompute active set and sub‑gradient for the current sequence
        M, conv = cur_score, cur_conv
        mask = conv >= M - 1e-12
        active = np.where(mask)[0]
        cnt_active = active.size

        # sub‑gradient wrt a (same code as in optimizer, but without FFT
        # for rev when possible)
        if cnt_active <= 5:
            rev = np.zeros_like(cur_seq)
            for k in active:
                low = max(0, k - (n - 1))
                high = min(k, n - 1)
                idx = np.arange(low, high + 1)
                rev[low:high + 1] += cur_seq[k - idx]
            rev /= cnt_active
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / cnt_active
            Nfft = 1 << ((3 * n - 2).bit_length())
            rev = _weighted_rev(cur_seq, w, Nfft)

        g_a = 2.0 * rev

        # indices that increase / decrease the max the most
        i_src = int(np.argmax(g_a))   # move *away* from this index
        i_dst = int(np.argmin(g_a))   # move *towards* this index
        if i_src == i_dst or cur_seq[i_src] < 1e-15:
            # fallback to a random pair
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst or cur_seq[i_src] < 1e-15:
                step += 1
                continue

        frac = rng.uniform(0.02, 0.08)                 # modest transfer
        delta = cur_seq[i_src] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i_src] -= delta
        new_seq[i_dst] += delta

        new_score, _ = max_conv(new_seq)

        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            cur_conv = _
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
                cur_conv = _
        step += 1

    return cur_seq, cur_score
