import numpy as np, time, math

def _full_convolution(a, fft_len=None):
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2*n-1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa*fa, fft_len)[:2*n-1]
    return conv

def _cross_correlation(w, a, fft_len):
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw*fr, fft_len)
    start = n-1
    return corr[start:start+n]

def _softmax(logits):
    mx = np.max(logits)
    exp = np.exp(logits - mx)
    return exp / exp.sum()

def propose_candidate(seed=0, budget_s=1000):
    """Search for a non‑negative step function that minimises the evaluation bound.

    The search works on a cascade of lengths (starting at 65536 if time permits)
    and optimises each candidate with a long Adam phase on a smooth‑max
    surrogate, followed by an exponentiated‑gradient sub‑gradient refinement.
    Several families of initial points are tried, including a random‑phase
    Fourier construction that often yields flatter self‑convolutions.
    The best sequence found before the wall‑clock deadline is returned as a
    plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2

    best_seq = None
    best_val = float('inf')
    # Determine max length we can afford based on remaining time. Roughly assume
    # 0.1 s per 10 000 iterations on a length of 32768.
    # We will try up to 65536 if enough time; otherwise fall back to 16384.
    # We'll keep a simple list.
    lengths = [16384, 32768, 65536]
    # We'll later prune based on time.

    # Helper to evaluate and maybe update global best.
    def maybe_update(a):
        nonlocal best_seq, best_val
        # Evaluate objective using our own fast evaluation: 2*n*max(conv) (since sum assumed 1)
        n = a.shape[0]
        conv = _full_convolution(a)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[UPDATE length={n}] new best {val:.6f}")
        return val

    for n in lengths:
        # Check time before starting
        if time.time() > deadline:
            break

        # If n is too big for remaining time estimate:
        # we compute approximate iteration budget per candidate later.
        # Proceed anyway, hope time.

        print(f"\n=== Optimising length {n} ===")
        fft_len = 1 << int(np.ceil(np.log2(2*n-1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3*n-2)))

        # Build initial candidates list
        cand_list = []

        # Uniform
        cand_list.append(np.full(n, 1.0/n, dtype=np.float64))

        # Warm‑start from previous best if same length
        if best_seq is not None and len(best_seq) == n:
            cand_list.append(np.array(best_seq, dtype=np.float64))

        # Upsample previous best if shorter
        if best_seq is not None and len(best_seq) < n:
            old = np.asarray(best_seq, dtype=np.float64)
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, old)
            up = up.clip(min=0)
            if up.sum() == 0:
                up = np.random.dirichlet(np.ones(n))
            else:
                up = up / up.sum()
            cand_list.append(up)

        # Historic global variable if present
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            h = np.asarray(hseq, dtype=np.float64)
            if h.shape[0] > n:
                h = h[:n]
            elif h.shape[0] < n:
                h = np.pad(h, (0, n - h.shape[0]), mode='constant')
            h = np.maximum(h, 0.0)
            if h.sum() == 0:
                h = np.random.dirichlet(np.ones(n))
            else:
                h = h / h.sum()
            cand_list.append(h)

        # Random Dirichlet
        cand_list.append(np.random.dirichlet(np.ones(n)))

        # Random Fourier phase construction
        # Pick random phases, compute real IFFT, take absolute value
        random_phase = np.exp(1j * np.random.uniform(0, 2*np.pi, fft_len))
        # Force symmetry to get real signal: conjugate symmetry
        # We can generate half and mirror
        # Simpler: generate real vector via iFFT of random complex numbers then take absolute
        # Use direct IFFT
        a_phase = np.fft.ifft(random_phase).real
        # Trim or pad to length n
        if a_phase.shape[0] >= n:
            a_phase = a_phase[:n]
        else:
            a_phase = np.pad(a_phase, (0, n - a_phase.shape[0]), mode='constant')
        a_phase = np.abs(a_phase)
        if a_phase.sum() == 0:
            a_phase = np.random.dirichlet(np.ones(n))
        else:
            a_phase = a_phase / a_phase.sum()
        cand_list.append(a_phase)

        # Two‑Gaussian edge shape (as before)
        i = np.arange(n, dtype=np.float64)
        for gamma in [0.3, 0.6, 1.2]:
            shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            shape = shape / shape.sum()
            cand_list.append(shape)

        # Sine‑squared bump
        shape = np.sin(np.pi * (np.arange(n)+0.5) / n) ** 2
        shape = shape / shape.sum()
        cand_list.append(shape)

        # Remove duplicates (optional)
        # Use a simple set of hashable approximations to avoid identical candidates
        # Not needed.

        # Now optimise each candidate
        for cand_id, a0 in enumerate(cand_list):
            if time.time() > deadline:
                break

            # ensure non‑negative and normalised
            a = np.maximum(a0, 1e-16)
            a = a / a.sum()
            logits = np.log(a + 1e-20)   # add epsilon to avoid log(0)

            # Adam parameters
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2 = 0.9, 0.999
            eps_adam = 1e-8
            lr = 0.08

            # Smooth‑max temperature schedule
            lam = 0.1
            lam_factor = 1.015  # slower increase -> smoother optimisation
            max_iters = int(15000 * 1024 / n) + 4000   # more iterations for smaller n

            last_improve = time.time()

            for it in range(1, max_iters+1):
                # softmax mapping
                a = _softmax(logits)

                # compute convolution
                conv = _full_convolution(a, fft_len=fft_len)
                max_c = conv.max()

                # Smooth‑max weighting
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w.fill(1.0 / w.size)
                else:
                    w /= w_sum

                # gradient of smooth‑max w.r.t a
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

                # backprop through softmax
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update
                m = beta1 * m + (1 - beta1) * grad_log
                v = beta2 * v + (1 - beta2) * (grad_log ** 2)
                m_hat = m / (1 - beta1 ** it)
                v_hat = v / (1 - beta2 ** it)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                # anneal temperature
                lam = min(lam * lam_factor, 1e7)

                # evaluate objective (exact)
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Adam it={it}] new best {val:.6f}")

                # stagnation detection
                if time.time() - last_improve > 6.0:
                    # small random perturbation of logits
                    logits += (np.random.rand(n) - 0.5) * 1e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # Sub‑gradient refinement with exponentiated gradient
            a = a / a.sum()
            eta = 0.5
            sub_last = time.time()
            for sub_it in range(3000):
                conv = _full_convolution(a, fft_len=fft_len)
                max_c = conv.max()
                # active set
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:  # numerical collapse
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad it={sub_it}] new best {val:.6f}")

                if time.time() - sub_last > 4.0:
                    break
                if time.time() > deadline:
                    break

    # fallback to uniform if nothing found
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0/fallback_n, dtype=np.float64)

    # Convert to list of Python floats
    return best_seq.tolist()

"""
Step‑function optimiser (new version).

The search works on a cascade of problem sizes (1024 – 65536).  For each size
a diverse pool of start points is built – uniform, the current best,
upsampled older solutions, random Dirichlet draws, a random‑phase Fourier
construction and a few analytical families.  Each start is refined by

* Adam on a smooth‑max surrogate of the true objective
  (the surrogate becomes the exact max once the temperature λ is large);
* a short exponentiated‑gradient sub‑gradient phase on the true ℓ∞‑norm.

The Adam phase is run for many more iterations than the previous
implementation (the budget scales with 1/n) and uses a slower temperature
schedule, giving the optimizer more time to escape plateaus.  The best
sequence found before the wall‑clock deadline is returned as a plain Python
list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level – no closures)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a via a power‑of‑two FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.

    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0 … n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


def _softmax(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → probability simplex)."""
    mx = np.max(logits)
    exp_shift = np.exp(logits - mx)
    return exp_shift / exp_shift.sum()


def _random_phase_candidate(n: int, fft_len: int) -> np.ndarray:
    """
    Generate a candidate by drawing random unit‑magnitude phases in the
    frequency domain, inverting with an IFFT, taking the absolute value and
    normalising.  The construction often yields a fairly flat self‑convolution.
    """
    # rfft size is fft_len//2+1 complex numbers
    phases = np.exp(1j * np.random.uniform(0.0, 2.0 * np.pi,
                                          size=fft_len // 2 + 1))
    a_full = np.fft.irfft(phases, n=fft_len)          # real vector length fft_len
    a = a_full[:n]                                    # keep first n entries
    a = np.abs(a)
    s = a.sum()
    if s == 0:
        a = np.random.dirichlet(np.ones(n))
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the bound
    ``evaluate_sequence`` returns.  The routine respects the given wall‑clock
    budget and returns the best sequence discovered (plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2               # tiny safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Length schedule – we try increasing sizes while time permits.
    # ------------------------------------------------------------------
    lengths = [16384, 32768, 65536]   # start with a medium size, grow if time allows

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        # FFT lengths needed for the two convolutions used in the optimiser
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

        # --------------------------------------------------------------
        # Build a diverse set of initial candidates
        # --------------------------------------------------------------
        candidates = []

        # 0 – uniform distribution
        candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # 1 – reuse the current global best if it has the same length
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.asarray(best_seq, dtype=np.float64))

        # 2 – up‑sample a shorter best (linear interpolation)
        if best_seq is not None and len(best_seq) < n:
            old = np.asarray(best_seq, dtype=np.float64)
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, old)
            up = np.maximum(up, 0.0)
            if up.sum() == 0:
                up = np.random.dirichlet(np.ones(n))
            else:
                up = up / up.sum()
            candidates.append(up)

        # 3 – historic global variable, if present
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            h = np.asarray(hseq, dtype=np.float64)
            if h.shape[0] > n:
                h = h[:n]
            elif h.shape[0] < n:
                h = np.pad(h, (0, n - h.shape[0]), mode="constant")
            h = np.maximum(h, 0.0)
            if h.sum() == 0:
                h = np.random.dirichlet(np.ones(n))
            else:
                h = h / h.sum()
            candidates.append(h)

        # 4 – random Dirichlet draw
        candidates.append(np.random.dirichlet(np.ones(n)))

        # 5 – random‑phase Fourier construction (new ingredient)
        candidates.append(_random_phase_candidate(n, conv_fft_len))

        # 6 – two‑Gaussian edge shapes (different widths)
        i = np.arange(n, dtype=np.float64)
        for gamma in (0.3, 0.6, 1.2):
            shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            shape = shape / shape.sum()
            candidates.append(shape)

        # 7 – smooth sine‑squared bump
        shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
        shape = shape / shape.sum()
        candidates.append(shape)

        # --------------------------------------------------------------
        # Optimise each candidate
        # --------------------------------------------------------------
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # initialise on the simplex (logits for soft‑max)
            a = np.maximum(a0, 1e-16)
            a = a / a.sum()
            logits = np.log(a + 1e-20)

            # ------- Adam on the smooth‑max surrogate -----------------
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2 = 0.9, 0.999
            eps_adam = 1e-8
            lr = 0.08

            lam = 0.10                 # start temperature (smooth‑max)
            lam_factor = 1.015         # slow annealing
            last_improve = time.time()

            # iteration budget grows when the problem is smaller
            max_iters = int(15000 * 1024 / n) + 4000

            for it in range(1, max_iters + 1):
                # map logits → probability simplex
                a = _softmax(logits)

                # convolution and smooth‑max weights
                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w = w / w_sum

                # gradient of the smooth‑max surrogate
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

                # back‑propagation through soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update on the logits
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                # anneal temperature
                lam = min(lam * lam_factor, 1e7)

                # exact objective (sum(a)=1 → denominator = 1)
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Adam it={it:05d}] new best {val:.6f}")

                # stagnation handling – tiny random shake of logits
                if time.time() - last_improve > 6.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # ------- Sub‑gradient refinement on the true max ----------
            a = a / a.sum()
            eta = 0.5
            sub_last = time.time()
            for sub_it in range(3000):
                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:                     # catastrophic collapse – restart locally
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad it={sub_it:04d}] new best {val:.6f}")

                if time.time() - sub_last > 4.0:   # stop if no improvement for a few seconds
                    break
                if time.time() > deadline:
                    break

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return as plain Python list as required by the external evaluator
    return best_seq.tolist()
