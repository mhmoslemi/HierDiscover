def _subgradient_descent(a0, n, max_iter, time_remaining, rng):
   a = a0.copy()
   best_seq = a.copy()
   best_val = np.inf
   for it in range(1, max_iter+1):
        # compute conv
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        if max_conv < best_val:
            best_val = max_conv
            best_seq = a.copy()
        # check time
        if time.time() > deadline: break
        # gradient
        # find indices where conv == max (within epsilon)
        eps = 1e-12
        max_mask = conv >= max_conv - eps
        # get corresponding shifts (indices of conv)
        shift_indices = np.where(max_mask)[0]
        # pick random shift among them
        k = rng.choice(shift_indices)
        # compute gradient g
        if k == 0:
            g = 2 * a
        else:
            g = np.zeros_like(a)
            g[k:] += a[:-k]
            g[:-k] += a[k:]
        # step size
        lr = 0.5 / np.sqrt(it)
        # update a
        a = a - lr * g
        # projection onto simplex
        a = _project_to_simplex(a)
        # maybe add jitter small random noise
        if it % 200 == 0 and rng.random() < 0.2:
            noise = rng.normal(scale=0.001, size=n)
            a = a + noise
            a = np.clip(a, 0, None)
            a = _project_to_simplex(a)
   return best_seq, best_val

def _project_to_simplex(v):
    """Project vector v onto the probability simplex {w: w >= 0, sum w = 1}."""
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _full_convolution_fft(a, b):
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    fa = np.fft.rfft(a, size)

    conv = np.fft.irfft(fa * fb, size)[:n]

def _lp_refine(f, rows, cols, max_iter, t, time_deadline):
    n = len(f)
    target_sum = np.sqrt(2.0 * n)
    # scale f sum to target
    f = f * (target_sum / f.sum())
    best_f = f.copy()
    best_val = evaluate_sequence(f.tolist())
    for it in range(max_iter):
        if time.time() >= time_deadline:
            break
        # compute M = max conv of f with itself
        conv_f = _full_convolution_fft(f, f)
        M = conv_f.max()
        # build A matrix data using current f
        data = np.tile(f, n)  # length n*n
        A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        # set up LP: max sum g_j subject to A @ g <= M, g >=0
        # Linprog solves min; set c = -1.
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n,
                      method='highs', options={'presolve': True})
        if not res.success:
            break
        g0 = res.x
        sum_g = g0.sum()
        if sum_g == 0:
            break
        g = g0 * (target_sum / sum_g)
        # mix
        f = (1 - t) * f + t * g
        # sometimes jitter to break symmetry
        # evaluate
        val = evaluate_sequence(f.tolist())
        if val < best_val:
            best_val = val
            best_f = f.copy()
        # optionally reduce t gradually
        t = max(t * 0.9, 0.01)  # shrink mixing parameter
    return best_f, best_val

"""
Hybrid optimisation for the step‑function bound

We treat the objective 2·n·max(conv(a,a))/(sum(a)²) as
    val = 2·n·max(conv)        (after normalising ∑a = 1, because the
                               expression is homogeneous).

The search consists of three stages:

1. **Projected sub‑gradient descent** – minimise the true max‑convolution.
   At each iteration we:
   * compute the full convolution via a power‑of‑2 FFT,
   * locate the shift k that attains the current maximum,
   * build the exact sub‑gradient of conv_k (a simple slice operation),
   * take a diminishing‑step Adam‑like update and project back onto the
     probability simplex (non‑negative, unit sum).

2. **Linear‑programming refinement** – given the current vector f,
   solve the LP
        maximise   1ᵀ g
        s.t.       f * g ≤ M   (M = max(conv(f,f)))
                    g ≥ 0
   and mix the solution g back into f with a small mixing factor t.
   Repeating this step monotonically reduces the bound.

3. **Random restarts & warm‑starts** – we run the above pipeline on many
   randomly generated initial shapes (power‑law, exponential, mixed) and
   also reuse the best construction found so far.  The overall best
   sequence is returned (scaled to the canonical sum √(2 n)).

All heavy work (FFT, projection, LP) is vectorised and uses the
high‑performance SciPy “highs” LP solver.  The routine respects the
global time budget (default 1000 s) and prints progress messages.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via an FFT of power‑of‑2 size."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative artefacts
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simplex projection (L2‑projection onto {w≥0, ∑w=1})
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Projects v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of a single convolution entry
# ----------------------------------------------------------------------
def _grad_of_conv_at_shift(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of conv_k = (a*a)[k] w.r.t. a."""
    n = a.shape[0]
    g = np.zeros_like(a)
    if k < n:
        # indices 0 .. k receive the reversed slice a[k::-1]
        g[: k + 1] = 2.0 * a[k :: -1]
    else:
        # k >= n: the active window starts at start = k-(n-1)
        start = k - (n - 1)
        g[start:] = 2.0 * a[start:][::-1]
    return g


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    n: int,
    max_iter: int,
    run_deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for ``2·n·max(conv(a,a))``.
    The vector is kept on the unit simplex (∑a = 1) – this makes the
    objective equal to ``2·n·max(conv)``.
    Returns the best sequence seen and its objective value.
    """
    a = _project_to_simplex(a0.copy())
    best_seq = a.copy()
    best_val = float("inf")
    it = 0
    last_improve = 0

    while it < max_iter and time.time() < run_deadline:
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()
        val = 2.0 * n * max_conv               # because ∑a = 1
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = it
            print(f"[{time.time():.2f}]  sub‑grad iter {it:5d}  val = {val:.8f}")

        # sub‑gradient of the active shift
        k_star = int(conv.argmax())
        grad = _grad_of_conv_at_shift(a, k_star)

        # diminishing step size (simple 1/√t schedule)
        step = 0.5 / np.sqrt(it + 1)
        a = a - step * grad

        # keep feasibility
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a)

        it += 1
        if it - last_improve > 2000:          # stagnation guard
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “LP‑step” described in the statement)
# ----------------------------------------------------------------------
def _precompute_A_indices(n: int):
    """Pre‑compute row / column indices for the Toeplitz constraint matrix."""
    rows = np.concatenate([np.arange(j, j + n) for j in range(n)], dtype=np.int32)
    cols = np.concatenate([np.full(n, j, dtype=np.int32) for j in range(n)], dtype=np.int32)
    return rows, cols


def _lp_refine(
    f: np.ndarray,
    rows: np.ndarray,
    cols: np.ndarray,
    max_lp_iter: int,
    t_mix: float,
    time_deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Perform the LP‑based improvement described in the problem statement.
    ``f`` is assumed to be already scaled to the canonical sum √(2 n).
    Returns an improved sequence and its official evaluation score.
    """
    n = f.shape[0]
    target_sum = np.sqrt(2.0 * n)
    f = f * (target_sum / f.sum())          # enforce the canonical normalisation

    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    for lp_it in range(max_lp_iter):
        if time.time() >= time_deadline:
            break

        # current bound M = max(conv(f,f))
        conv_f = _full_convolution_fft(f, f)
        M = conv_f.max()

        # build the constraint matrix A (size (2n‑1) × n)
        data = np.tile(f, n)                 # each column j contains f[0..n‑1]
        A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

        # maximise 1ᵀ g  ⇔ minimise -1ᵀ g
        c = -np.ones(n)
        b = np.full(2 * n - 1, M)

        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": 5.0},
        )
        if not res.success:
            # LP failed – abort further refinements
            break

        g0 = res.x
        sum_g = g0.sum()
        if sum_g == 0.0:
            break
        # rescale to the required total mass
        g = g0 * (target_sum / sum_g)

        # mixture step (monotone decrease of the bound)
        f = (1.0 - t_mix) * f + t_mix * g

        # official evaluation (includes clipping etc.)
        score = evaluate_sequence(f.tolist())
        if score < best_score:
            best_score = score
            best_seq = f.copy()
            print(
                f"[{time.time():.2f}]   LP iter {lp_it:2d}  improved score = {score:.8f}"
            )

        # shrink the mixing factor slowly – helps convergence
        t_mix = max(t_mix * 0.94, 0.01)

    return best_seq, best_score

        if sc < best_score:
            best_score = sc
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] warm‑start global score = {sc:.8f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Main optimisation loop – many short restarts.
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # pick a problem size (mid‑range lengths work well)
        n = int(rng.integers(1500, 3000 + 1))

        # ----------------------------------------------------------
        # Build an initial shape (occasionally warm‑start from best)
        # ----------------------------------------------------------
        if best_seq is not None and rng.random() < 0.35:
            src = best_seq
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = rng.random(n - src.size) * 1e-6
                init = np.concatenate([src, pad])
        else:
            typ = rng.choice(["power", "exponential", "mixed"])
            idx = np.arange(1, n + 1, dtype=float)
            if typ == "power":
                p = rng.uniform(0.1, 2.0)
                init = idx ** (-p)
            elif typ == "exponential":
                tau = rng.uniform(0.3, 0.8) * n
                init = np.exp(-idx / tau)
            else:  # mixed
                split = rng.integers(1, n - 1)
                left = idx[:split] ** (-rng.uniform(0.2, 1.0))
                right = (n - idx[split:] + 1) ** (-rng.uniform(0.2, 1.0))
                init = np.concatenate([left, right])

        init = np.maximum(init, 1e-12)
        init = init / init.sum()                       # unit simplex

        # ----------------------------------------------------------
        # Sub‑gradient descent (fast local optimisation)
        # ----------------------------------------------------------
        time_slice = min(20.0, remaining * 0.45)
        sub_deadline = min(deadline, time.time() + time_slice)
        seq_local, val_local = _subgradient_descent(
            init,
            n,
            max_iter=8000,
            run_deadline=sub_deadline,
        )
        # official evaluation
        score_local = evaluate_sequence(seq_local.tolist())
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL NEW BEST (sub‑grad) n={n} score={score_local:.8f}"
            )
            if best_score < 1.503:
                break

        # ----------------------------------------------------------
        # LP refinement – only if we still have time left.
        # ----------------------------------------------------------
        if time.time() < deadline - 5.0:
            if n not in A_cache:
                A_cache[n] = _precompute_A_indices(n)
            rows, cols = A_cache[n]
            lp_time = min(25.0, deadline - time.time())
            seq_refined, score_refined = _lp_refine(
                best_seq if best_seq is not None else seq_local,
                rows,
                cols,
                max_lp_iter=12,
                t_mix=0.18,
                time_deadline=time.time() + lp_time,
            )
            if score_refined < best_score:
                best_score = score_refined
                best_seq = seq_refined.copy()
                print(
                    f"[{time.time():.2f}] GLOBAL NEW BEST (LP) n={n} score={score_refined:.8f}"
                )
                if best_score < 1.503:
                    break

    # ------------------------------------------------------------------
    # Fallback – should never be needed but guarantees a valid return.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n)

    # Scale the final answer to the canonical sum √(2 n) (optional but tidy)
    n = best_seq.size
    target_sum = np.sqrt(2.0 * n)
    best_seq = best_seq * (target_sum / best_seq.sum())

    return best_seq.tolist()
