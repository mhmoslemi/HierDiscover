def propose_candidate(seed=0, budget_s=1000):

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1

    best_seq = None; best_val = inf

    # list of lengths in descending order (big to small)
    lengths = [16384, 8192, 4096, 2048, 1024]  # choose maybe 5 lengths.

    time_per_len = {} # dynamic allocation: allocate equal fractions of remaining time; but we can just use while loop with time check.

    for n in lengths:
        if time.time() > deadline: break
        print(f"Optimizing length {n}")
        fft_len = 1 << int(np.ceil(np.log2(2*n-1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3*n-2)))  # maybe larger

        # generate initial candidates
        candidates = []
        # Add best from previous same length if any
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.array(best_seq))
        # Add warm start from truncated or padded best_seq of other length (downsample or upsample)
        if best_seq is not None:
            # if best_seq shorter, pad zeros; if longer, subsample
            a = np.array(best_seq)
            if len(a) < n:
                a = np.pad(a, (0, n - len(a)))
            elif len(a) > n:
                a = a[:n]  # simple truncation
            a = a / a.sum()
            candidates.append(a)
        # Add random Dirichlet possibilities
        for conc in [0.3, 0.7, 1.0]:
            candidates.append(np.random.dirichlet(np.full(n, conc)))
        # Add power-law edge concentrated shape
        i = np.arange(n)
        prod = (i+1)*(n-i)
        for alpha in np.linspace(-2, -0.2, 5):
            shape = np.where(prod>0, np.power(prod, alpha), 0)
            if shape.sum()>0:
                shape = shape / shape.sum()
                candidates.append(shape)

        # Optimize each candidate via combined ADAM+PGD+SA
        for cand_id, a_init in enumerate(candidates):
            if time.time() > deadline: break
            a = a_init.astype(np.float64)
            if a.sum() == 0:
                a = np.random.dirichlet(np.ones(n))
            a = a / a.sum()
            # Parameterize via logits for simplex
            logits = np.log(a + 1e-15)
            # ADAM settings
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)

"""
Propose a non‑negative step‑function (list of heights) that minimizes

    F(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The search works on several scales (n = 512 … 16384).  For each length we
* generate a diverse set of seeds (random Dirichlet, power‑law edges,
  two‑sided exponentials, a tent and warm‑starts from previously found
  solutions),
* optimise each seed with three stages
    – Adam on a smooth‑max surrogate,
    – exponentiated‑gradient / projected‑gradient descent on the exact max,
    – a short simulated‑annealing walk,
* keep the best sequence seen overall.

Time‑budget handling is strict: the routine stops as soon as the
deadline is reached and returns the best sequence discovered so far.
All heavy work is performed with FFTs (convolution / cross‑correlation)
and the sequence is kept on the probability simplex, which makes the
objective identical to ``evaluate_sequence`` (the sum is forced to 1, the
values are non‑negative and well below the 1000‑clamp).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the
    smooth‑max.  The result has length ``len(a)``.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implements the O(n log n) algorithm from
    Duchi et al., 2008.
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to length ``new_n`` and normalisation."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.arange(old_n, dtype=np.float64)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=np.float64)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s < 1e-12:
        up = np.full(new_n, 1.0 / new_n, dtype=np.float64)
    else:
        up = up / s
    return up


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.2   # safety margin

    best_seq = None          # best array found so far (already normalised)
    best_val = float("inf")  # best objective value

    # ------------------------------------------------------------------
    # Length schedule – start small, finish with the largest
    lengths = [512, 1024, 2048, 4096, 8192, 16384]

    # store the best sequence for the last processed size (for up‑sampling)
    best_per_len = {}

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n = {n} ===")

        # FFT lengths (powers of two)
        conv_len = 2 * n - 1
        fft_len_conv = _next_pow2(conv_len)
        fft_len_corr = _next_pow2(conv_len + n - 1)

        # --------------------------------------------------------------
        # Build a pool of initial candidates for this length
        candidates = []

        # 1. Warm‑start from the best solution of the previous size,
        #    up‑sampled or down‑sampled to the current length.
        if best_seq is not None:
            cand = _upsample(best_seq, n)
            candidates.append(cand)

        # 2. Random Dirichlet draws (different concentrations)
        for conc in [0.3, 0.7, 1.0]:
            candidates.append(rng.dirichlet(np.full(n, conc)))

        # 3. Edge‑concentrated power‑law shapes
        idx = np.arange(n, dtype=np.float64) + 1.0
        prod = idx * (n - idx + 1.0)
        for alpha in np.linspace(-2.0, -0.2, 5):
            shape = np.where(prod > 0,
                             np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod)),
                             0.0)
            if shape.sum() > 0:
                shape = shape / shape.sum()
                candidates.append(shape)

        # 4. Two‑sided exponential
        i = np.arange(n, dtype=np.float64)
        for gamma in [0.001, 0.002, 0.004, 0.008]:
            shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
            shape = shape / shape.sum()
            candidates.append(shape)

        # 5. Tent (linear rise then fall)
        half = n // 2
        left = np.linspace(0, 1, half, endpoint=False)
        right = left[::-1]
        tent = np.concatenate([left, right])
        if tent.shape[0] < n:
            tent = np.append(tent, 0.0)
        tent = np.maximum(tent, 0.0)
        tent = tent / tent.sum()
        candidates.append(tent)

        # --------------------------------------------------------------
        # Optimise every candidate
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # Normalise to a probability vector (sum = 1)
            a = a0.astype(np.float64)
            if a.sum() < 1e-12:
                a = rng.dirichlet(np.ones(n))
            else:
                a /= a.sum()

            # Parameterisation via logits (soft‑max = simplex)
            logits = np.log(a + 1e-15)

            # ----------------------------------------------------------
            # Phase 1 – Adam on a smooth‑max surrogate
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.08                     # learning rate (a bit larger)
            lam = 0.25                    # initial temperature
            lam_factor = 1.03            # quicker temperature increase

            # iteration budget that shrinks with n
            max_iters = int(25000 * 1024 / n) + 5000
            last_improve = time.time()

            for t in range(1, max_iters + 1):
                if time.time() > deadline:
                    break

                # ----- soft‑max (logits → probabilities) -----
                mx = logits.max()
                exp_logits = np.exp(logits - mx)
                a = exp_logits / exp_logits.sum()

                # ----- convolution and smooth‑max weights -----
                conv = _full_convolution(a, fft_len_conv)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # ----- gradient of the smooth‑max surrogate -----
                grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

                # ----- back‑prop through soft‑max -----
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # ----- Adam update -----
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # ----- temperature schedule -----
                lam = min(lam * lam_factor, 1e6)

                # ----- exact objective (sum = 1) -----
                obj = 2.0 * n * max_c
                if obj < best_val:
                    best_val = obj
                    best_seq = a.copy()
                    best_per_len[n] = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Adam it={t:05d}] new best {obj:.6f}")

                # ----- stagnation handling -----
                if time.time() - last_improve > 8.0:
                    logits += (rng.random(n) - 0.5) * 5e-3
                    last_improve = time.time()

            # ----------------------------------------------------------
            # Phase 2 – Exponentiated‑gradient / projected gradient on exact max
            a = a / a.sum()
            eta = 0.65                     # starting step size for EG
            sub_last = time.time()
            sub_iters = int(6000 * 1024 / n) + 2000

            for sub_it in range(sub_iters):
                if time.time() > deadline:
                    break
                conv = _full_convolution(a, fft_len_conv)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12)
                # uniform weight over all active positions
                w = np.zeros_like(conv)
                if active.any():
                    w[active] = 1.0 / active.sum()
                grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

                # exponentiated‑gradient step (keeps us on the simplex)
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:                     # avoid collapse
                    a = rng.dirichlet(np.ones(n))
                else:
                    a = a / s

                # evaluate
                conv = _full_convolution(a, fft_len_conv)
                obj = 2.0 * n * conv.max()
                if obj < best_val:
                    best_val = obj
                    best_seq = a.copy()
                    best_per_len[n] = a.copy()
                    sub_last = time.time()
                    print(f"[n={n} cand={cand_id} EG it={sub_it}] new best {obj:.6f}")

                # decay step size slowly
                eta *= 0.995

                if time.time() - sub_last > 6.0:
                    break

            # ----------------------------------------------------------
            # Phase 3 – Simulated‑annealing fine‑tuning (tiny mass moves)
            a_sa = a.copy()
            cur_conv = _full_convolution(a_sa, fft_len_conv)
            cur_val = 2.0 * n * cur_conv.max()
            T = 0.2 * cur_val                 # initial temperature
            sa_iters = int(12000 * 1024 / n) + 3000

            for sa_it in range(sa_iters):
                if time.time() > deadline:
                    break
                # propose a transfer of ε from i → j
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                eps_move = rng.random() * 0.008   # ≤ 0.8 % of total mass
                eps_move = min(eps_move, a_sa[i])  # cannot go negative
                a_new = a_sa.copy()
                a_new[i] -= eps_move
                a_new[j] += eps_move

                conv_new = _full_convolution(a_new, fft_len_conv)
                new_val = 2.0 * n * conv_new.max()
                delta = new_val - cur_val

                if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
                    a_sa = a_new
                    cur_val = new_val
                    if cur_val < best_val:
                        best_val = cur_val
                        best_seq = a_sa.copy()
                        best_per_len[n] = a_sa.copy()
                        print(f"[n={n} SA it={sa_it}] new best {cur_val:.6f}")

                # geometric cooling
                T *= 0.9995
                if T < 1e-9:
                    break

        # --------------------------------------------------------------
        # (optional) final fine‑tune of the global best for this length
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] final fine‑tune (current best {best_val:.6f})")
            a = best_seq.copy()
            logits = np.log(a + 1e-15)
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.04
            lam = 0.5
            lam_factor = 1.01
            last_imp = time.time()
            fine_iters = int(30000 * 1024 / n) + 5000

            for t in range(1, fine_iters + 1):
                if time.time() > deadline:
                    break
                mx = logits.max()
                exp_logits = np.exp(logits - mx)
                a = exp_logits / exp_logits.sum()

                conv = _full_convolution(a, fft_len_conv)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

                grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                m = beta1 * m + (1 - beta1) * grad_log
                v = beta2 * v + (1 - beta2) * (grad_log ** 2)
                m_hat = m / (1 - beta1 ** t)
                v_hat = v / (1 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                lam = min(lam * lam_factor, 1e6)

                obj = 2.0 * n * max_c
                if obj < best_val:
                    best_val = obj
                    best_seq = a.copy()
                    best_per_len[n] = a.copy()
                    last_imp = time.time()
                    print(f"[n={n} fine it={t:05d}] new best {obj:.6f}")

                if time.time() - last_imp > 10.0:
                    logits += (rng.random(n) - 0.5) * 5e-3
                    last_imp = time.time()

    # ------------------------------------------------------------------
    # Fallback – uniform distribution if nothing was found (should not happen)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return as a plain Python list
    return best_seq.tolist()
