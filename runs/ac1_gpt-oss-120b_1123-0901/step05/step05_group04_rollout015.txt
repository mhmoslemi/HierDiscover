def propose_candidate(seed=42, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    best_seq = None
    best_score = float("inf")
    top_candidates = [] # store (score, seq)

    while time.time() < deadline:
        # pick n
        n = rng.integers(800, 2000+1)
        target_sum = np.sqrt(2*n)

        # generate init shape
        shape_type = rng.choice(['uniform','power','exponential'])
        if shape_type == 'uniform':
            a = np.full(n, target_sum/n)
        elif shape_type == 'power':
            p = rng.uniform(0.1, 2.0)
            idx = np.arange(1, n+1)
            a = idx**(-p)
        elif shape_type == 'exponential':
            tau = rng.uniform(0.2,0.9)*n
            idx = np.arange(1,n+1)
            a = np.exp(-idx/tau)
        # add jitter
        a += rng.random(n) * 1e-6
        a = np.maximum(a, 1e-12)

        # optional quick local search using Adam for few steps (say 100)

            top_candidates.sort(key=lambda x:x[0])
            top_candidates = top_candidates[:5]
            print(f"[{time.time():.2f}] New global best start {score:.8f} n={n}")

        # If n <= 1500 and enough time, do LP refinement
        if n <= 1500 and (deadline - time.time()) > 5.0:

def lp_refine(seq, n, target_sum, deadline):
    f = seq.copy()
    t = 0.2
    best_f = f.copy()
    best_M = np.max(np.convolve(f, f))
    best_score = 2 * n * best_M / (target_sum**2)

    while time.time() < deadline:
        M = best_M
        # Build A (dense)
        rows = 2*n - 1
        A = np.zeros((rows, n), dtype=float)
        for i in range(n):
            A[i:i+n, i] = f[i]
        # Solve LP
        c = -np.ones(n)
        b_ub = np.full(rows, M, dtype=float)
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                    bounds=[(0, None)]*n,
                                    method='highs')
        if not res.success:
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break
        b *= (target_sum / sum_b)

        # convex combination
        f_new = (1 - t) * f + t * b
        conv_new = np.convolve(f_new, f_new)
        M_new = conv_new.max()

        if M_new < M - 1e-9:
            f = f_new
            best_M = M_new
            best_f = f.copy()
            # possibly increase t a bit
            t = min(0.5, t * 1.05)
        else:
            # no improvement, reduce t
            t *= 0.5
            if t < 1e-4:
                break
        # compute score
        cur_score = 2 * n * best_M / (target_sum**2)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
        # early exit if score < 1.503
        if cur_score < 1.503:
            break
        # optional break after few iteration
        if np.isclose(t, 0.0):
            break
    # final evaluation using evaluate_sequence for rounding
    final_score = evaluate_sequence(best_f.tolist())
    return best_f, final_score

def _lp_refine(seq, n, target_sum, deadline, rng):
    import numpy as np, time, scipy.optimize
    f = seq.copy()
    # initial M
    conv_f = np.convolve(f, f)
    M = conv_f.max()
    best_f = f.copy()
    best_M = M
    best_score = 2 * n * M / (target_sum ** 2)
    t = 0.2
    iteration = 0
    while time.time() < deadline:
        # Build constraint matrix A (dense)
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=float)
        for i in range(n):
            A[i:i+n, i] = f[i]
        # Solve LP: maximize sum(b) ↔ minimize -sum(b)
        c = -np.ones(n)
        b_ub = np.full(rows, M, dtype=float)
        bounds = [(0, None)] * n
        # Provide a small time limit to the solver to keep within deadline
        # HiGHS respects 'time_limit' option in seconds.
        time_left = max(0.0, deadline - time.time())
        opts = {'presolve': True}
        if time_left > 0:
            opts['time_limit'] = time_left
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                    bounds=bounds, method='highs',
                                    options=opts)
        if not res.success:
            # LP failed – give up on this candidate
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break
        # rescale b to the target sum
        b *= (target_sum / sum_b)

        # convex combination
        f_new = (1.0 - t) * f + t * b
        conv_new = np.convolve(f_new, f_new)
        M_new = conv_new.max()

        # decide acceptance
        if M_new < M - 1e-12:
            f = f_new
            M = M_new
            # enlarge step size a little to accelerate
            t = min(0.5, t * 1.05)
        else:
            # step too aggressive – shrink
            t *= 0.5
            if t < 1e-5:
                break

        iteration += 1
        cur_score = 2 * n * M / (target_sum ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
        # early exit if good enough
        if cur_score < 1.503:
            break
        # also stop after a reasonable number of inner iterations
        if iteration >= 15:
            break
    # final evaluation through the provided function (handles clipping)
    final_score = evaluate_sequence(best_f.tolist())
    return best_f, final_score

"""
search_step_function.py

Search for a non‑negative step function minimizing

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm has two stages.

1️⃣  Random parametric search  
    – Choose a length n (800 … 2500) and a profile type (power‑law,
      exponential, mixed or uniform).  The profile is normalised to
      Σa = √(2 n) and evaluated with the provided ``evaluate_sequence``.
    – The best few candidates are kept.

2️⃣  Linear‑programming refinement  
    – For a candidate f compute M = max(conv(f, f)).
    – Solve the LP  

          maximise   Σ b_j
          subject to b ≥ 0
                     f * b ≤ M                (entry‑wise)

      (the convolution is linear in b).  The LP is solved with
      ``scipy.optimize.linprog`` (HiGHS).  The solution g₀ is rescaled to
      the canonical sum √(2 n) → g.
    – Update the shape by a convex combination  

          f ← (1‑t) f + t g ,   t ≈ 0.2

      This step provably never increases the maximum convolution.
      The process is repeated a few times (≤15) or until the time budget
      runs out.

The best sequence found before the 1000 s deadline is returned.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Helper – generate a random shape
# ----------------------------------------------------------------------
def _random_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a positive vector of length n of a random profile."""
    typ = rng.choice(['power', 'exponential', 'mixed', 'uniform'])
    idx = np.arange(1, n + 1, dtype=float)

    if typ == 'power':
        p = rng.uniform(0.05, 5.0)           # exponent
        shape = idx ** (-p)
    elif typ == 'exponential':
        tau = rng.uniform(0.05, 1.0) * n     # decay length
        shape = np.exp(-idx / tau)
    elif typ == 'mixed':
        split = rng.integers(1, n - 1)
        p_left = rng.uniform(0.05, 5.0)
        p_right = rng.uniform(0.05, 5.0)
        left = idx[:split] ** (-p_left)
        right = (n - idx[split:] + 1) ** (-p_right)
        shape = np.concatenate([left, right])
    else:                                   # uniform
        shape = np.ones_like(idx)

    # tiny jitter to break perfect symmetry
    shape += rng.random(n) * 1e-9
    shape = np.maximum(shape, 1e-15)
    return shape


# ----------------------------------------------------------------------
# Linear‑programming refinement (the classic alternating projection step)
# ----------------------------------------------------------------------
def _lp_refine(seq: np.ndarray,
               n: int,
               target_sum: float,
               deadline: float,
               rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform a few iterations of the LP‑based refinement described in the
    statement.  Returns the refined sequence and its official score.
    """
    f = seq.astype(float).copy()
    # ensure exact normalisation
    f *= target_sum / f.sum()

    # initial objective value
    conv_f = np.convolve(f, f)
    M = conv_f.max()
    best_f = f.copy()
    best_M = M
    best_score = 2.0 * n * M / (target_sum ** 2)

    # step size for the convex combination
    t = 0.2
    iteration = 0

    while time.time() < deadline:
        # --------------------------------------------------------------
        # Build the dense convolution matrix A  (size (2n‑1) × n)
        # --------------------------------------------------------------
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=float)
        for i in range(n):
            A[i:i + n, i] = f[i]               # column i has f[i] on rows i … i+n‑1

        # --------------------------------------------------------------
        # Solve   maximise Σ b   ⇔ minimise   -Σ b
        # --------------------------------------------------------------
        c = -np.ones(n)                         # minimise –sum(b)
        b_ub = np.full(rows, M, dtype=float)    # right‑hand side vector
        bounds = [(0.0, None)] * n

        # give HiGHS a tiny time limit so we never exceed the global budget
        time_left = max(0.0, deadline - time.time())
        opts = {'presolve': True}
        if time_left > 0.0:
            opts['time_limit'] = time_left

        res = scipy.optimize.linprog(c,
                                     A_ub=A,
                                     b_ub=b_ub,
                                     bounds=bounds,
                                     method='highs',
                                     options=opts)

        if not res.success:
            # LP failed – abort the refinement for this candidate
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            break

        # --------------------------------------------------------------
        # Rescale to the canonical sum √(2 n)
        # --------------------------------------------------------------
        b *= (target_sum / sum_b)

        # --------------------------------------------------------------
        # Convex combination f ← (1‑t) f + t b
        # --------------------------------------------------------------
        f_new = (1.0 - t) * f + t * b
        conv_new = np.convolve(f_new, f_new)
        M_new = conv_new.max()

        # --------------------------------------------------------------
        # Acceptance test
        # --------------------------------------------------------------
        if M_new < M - 1e-12:
            # improvement – keep the new point and slightly enlarge step size
            f = f_new
            M = M_new
            t = min(0.5, t * 1.05)
        else:
            # no improvement – shrink the step size
            t *= 0.5
            if t < 1e-5:
                break

        iteration += 1

        cur_score = 2.0 * n * M / (target_sum ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
            best_M = M

        # early stop if we already beat the target
        if cur_score < 1.503:
            break

        # safety net – do not run forever
        if iteration >= 15:
            break

    # final official evaluation (handles clipping etc.)
    final_score = evaluate_sequence(best_f.tolist())
    return best_f, final_score

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the evaluation
    function.  The routine runs for at most ``budget_s`` seconds and returns
    the best sequence discovered.

    Strategy
    --------
    * Phase‑1 – cheap random parametric search over power‑law,
      exponential, mixed and uniform profiles.  Keep the five best
      candidates.
    * Phase‑2 – for the small‑n candidates (n ≤ 1500) run a
      linear‑programming refinement that repeatedly solves

          maximise Σ b  subject to  b ≥ 0 ,  f * b ≤ M

      and updates the shape by a convex combination.
    * The globally best sequence (or a fall‑back uniform one) is returned.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.2          # tiny safety margin

    best_seq = None
    best_score = float('inf')
    top = []                                   # list of (score, seq, n)

    # ------------------------------------------------------------------
    # Warm‑start from a known construction, if present.
    # ------------------------------------------------------------------
    try:
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        hs = np.clip(hs, 0.0, 1000.0)
        sc = evaluate_sequence(hs.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = hs.copy()
            top.append((sc, hs.copy(), hs.size))
            print(f"[{time.time():.2f}] Warm‑start global = {sc:.8f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 1 – random parametric search (≈ 55 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start + budget_s * 0.55
    while time.time() < phase1_deadline:
        n = int(rng.integers(800, 2500 + 1))
        target_sum = np.sqrt(2.0 * n)

        shape = _random_shape(n, rng)
        seq = shape / shape.sum() * target_sum

        # direct evaluation (the cheap way)
        score = evaluate_sequence(seq.tolist())

        # update global best
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] New global best (random) = {score:.8f} n={n}")

        # maintain a shortlist of the five most promising candidates
        if len(top) < 5:
            top.append((score, seq.copy(), n))

    # ------------------------------------------------------------------
    # Phase 2 – LP refinement of the best small‑n candidates
    # ------------------------------------------------------------------
    # keep only candidates with reasonable size for the dense LP
    small_candidates = [(s, seq, n) for (s, seq, n) in top if n <= 1500]
    small_candidates.sort(key=lambda x: x[0])

        if time.time() >= deadline - 4.0:      # leave a tiny margin
            break

        target_sum = np.sqrt(2.0 * n)
        # ensure exact normalisation before the refinement
        seq = seq * (target_sum / seq.sum())

        # allocate a slice of the remaining budget for this candidate
        remaining = deadline - time.time()
        slice_deadline = time.time() + min(remaining * 0.4, 20.0)

        refined_seq, refined_score = _lp_refine(seq, n, target_sum,
                                                slice_deadline, rng)

        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[{time.time():.2f}] NEW BEST after LP (rank {rank}) = {refined_score:.8f} n={n}")

        # stop early if we already reached the target bound
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – should never be needed but guarantees a valid return
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # The harness expects a plain Python list.
    return best_seq.tolist()
