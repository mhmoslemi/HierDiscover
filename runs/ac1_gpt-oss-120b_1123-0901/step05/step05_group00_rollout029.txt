def _optimize_candidate(a0, typ, n, conv_fft_len, corr_fft_len, max_iters, deadline):
    """
    Optimize a single candidate.
    typ: 'full' or 'sym'.
    a0: initial numpy array.
    Returns (best_seq_full, best_val)
    """
    eps = 1e-16
    # Initialize logits
    if typ == 'full':
        a = np.maximum(a0, eps)
        a = a / a.sum()
        logits = np.log(a)
    else:
        # sym: a0 is half vector (size m)
        # make half unscaled probabilities sum to 1
        half = np.maximum(a0, eps)
        half = half / half.sum()
        logits = np.log(half)

    # Adam parameters
    mexp = np.zeros_like(logits)
    vexp = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05

    lam = 0.5
    lam_factor = 1.01
    last_improve = time.time()
    best_local_seq = None
    best_local_val = float('inf')
    # iterative loop
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # compute probabilities
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        probs = exp_log / exp_log.sum()

        if typ == 'full':
            a = probs
            a_full = a
        else:
            # sym case
            half_unscaled = probs
            half = 0.5 * half_unscaled
            a_full = np.concatenate([half, half[::-1]])
        # convolution
        conv = _full_convolution(a_full, fft_len=conv_fft_len)
        max_c = conv.max()
        # compute w for smooth max
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient in full space
        grad_full = 2.0 * _cross_correlation(w, a_full, fft_len=corr_fft_len)
        # gradient wrt logits
        if typ == 'full':
            avg = np.dot(a_full, grad_full)
            grad_log = a_full * (grad_full - avg)
        else:
            # combine gradient for half
            m_half = n // 2
            grad_first = grad_full[:m_half]
            grad_rev = grad_full[-1:-m_half-1:-1]
            grad_half_combined = grad_first + grad_rev
            grad_half_unscaled = 0.5 * grad_half_combined
            avg = np.dot(probs, grad_half_unscaled)
            grad_log = probs * (grad_half_unscaled - avg)

        # Adam update
        mexp = beta1 * mexp + (1 - beta1) * grad_log
        vexp = beta2 * vexp + (1 - beta2) * (grad_log ** 2)
        m_hat = mexp / (1 - beta1 ** it)
        v_hat = vexp / (1 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal lambda
        lam = min(lam * lam_factor, 1e7)

        # compute objective value (exact)
        val = 2.0 * n * max_c
        if val < best_local_val:
            best_local_val = val
            best_local_seq = a_full.copy()
            last_improve = time.time()
            # optional print minimal?
            # skip printing to reduce output

"""
Symmetry‑aware exponential‑gradient optimiser.

The optimiser works on the probability simplex (the sum of the coefficients is
fixed to 1, which is harmless because the objective is homogeneous).  For each
target length n we generate a diverse pool of start points, some of which are
forced to be symmetric (a_i = a_{n-1‑i}).  Every start point is first refined
with Adam on a log‑sum‑exp (smooth‑max) surrogate of the true ℓ∞‑norm of the
self‑convolution.  Afterwards an exact sub‑gradient method (exponentiated
gradient) finishes the optimisation.  The algorithm scales the number of
iterations with the problem size, uses a stagnation detector with a tiny random
shake, and finally performs a deep refinement on the best global sequence.
All sequences are kept non‑negative and are returned as plain Python lists.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a * a using a power‑of‑two FFT.
    Returns the full convolution of length 2·len(a)‑1.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of
    max(a*a) with respect to a.

    Returns a vector of length len(a) where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(len(a)-1)]   for i = 0…len(a)-1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Core optimisation for a single candidate
# ----------------------------------------------------------------------


def _optimize_candidate(
    a0: np.ndarray,
    kind: str,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    max_iters: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Optimise one start point.

    Parameters
    ----------
    a0 : np.ndarray
        Initial array.  For ``kind == 'full'`` it has length n,
        for ``kind == 'sym'`` it has length n//2 (the left half).
    kind : {'full', 'sym'}
        Whether the optimisation is performed on the full vector
        (no symmetry enforced) or on a symmetric model.
    n : int
        Full length of the step function.
    conv_fft_len, corr_fft_len : int
        FFT lengths pre‑computed for the current n.
    max_iters : int
        Number of Adam iterations.
    deadline : float
        Unix time after which the optimisation must stop.

    Returns
    -------
    best_seq : np.ndarray
        The best full‑length sequence found (sums to 1).
    best_val : float
        Corresponding evaluation value.
    """
    eps = 1e-16
    m = n // 2                         # half length, used only when kind=='sym'

    # initialise logits (log‑softmax parameters)
    if kind == "full":
        a = np.maximum(a0, eps)
        a = a / a.sum()
        logits = np.log(a + eps)
    else:  # symmetric model
        half = np.maximum(a0, eps)
        half = half / half.sum()
        logits = np.log(half + eps)

    # Adam hyper‑parameters
    mexp = np.zeros_like(logits)
    vexp = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05

    # smooth‑max temperature
    lam = 0.5
    lam_factor = 1.01

    best_local_seq = None
    best_local_val = float("inf")
    last_improve = time.time()

    # ------------------------------------------------------------------
    # 1) Adam on the log‑sum‑exp surrogate
    # ------------------------------------------------------------------
    for it in range(1, max_iters + 1):
        # ---- deadline guard ------------------------------------------------
        if time.time() > deadline:
            break

        # soft‑max -> probability simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        probs = exp_log / exp_log.sum()          # sum = 1

        if kind == "full":
            a_full = probs
        else:  # symmetric
            half_unscaled = probs
            half = 0.5 * half_unscaled
            a_full = np.concatenate([half, half[::-1]])

        # convolution of the current full vector
        conv = _full_convolution(a_full, fft_len=conv_fft_len)
        max_c = conv.max()

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. the full vector
        grad_full = 2.0 * _cross_correlation(w, a_full, fft_len=corr_fft_len)

        # back‑propagation through the soft‑max
        if kind == "full":
            avg = np.dot(a_full, grad_full)
            grad_log = a_full * (grad_full - avg)
        else:
            # combine the two symmetric contributions
            grad_left = grad_full[:m]
            grad_right = grad_full[-1:-m - 1:-1]
            grad_half_combined = grad_left + grad_right
            # chain rule for the 0.5 scaling
            grad_half_unscaled = 0.5 * grad_half_combined
            avg = np.dot(probs, grad_half_unscaled)
            grad_log = probs * (grad_half_unscaled - avg)

        # Adam update on the logits
        mexp = beta1 * mexp + (1.0 - beta1) * grad_log
        vexp = beta2 * vexp + (1.0 - beta2) * (grad_log ** 2)
        m_hat = mexp / (1.0 - beta1 ** it)
        v_hat = vexp / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal the temperature (approach the true max)
        lam = min(lam * lam_factor, 1e7)

        # exact objective (the denominator is 1 because we keep sum=1)
        val = 2.0 * n * max_c
        if val < best_local_val:
            best_local_val = val
            best_local_seq = a_full.copy()
            last_improve = time.time()

        # stagnation detection – tiny random shake of the logits
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(logits.shape[0]) - 0.5) * 1e-3
            last_improve = time.time()

    # ------------------------------------------------------------------
    # 2) Exact sub‑gradient refinement (exponentiated gradient)
    # ------------------------------------------------------------------
    if best_local_seq is None:               # fallback (should never happen)
        if kind == "full":
            best_local_seq = np.full(n, 1.0 / n)
        else:
            half = np.full(m, 0.5 / m)
            best_local_seq = np.concatenate([half, half[::-1]])

    a_full = best_local_seq.copy()
    a_full = a_full / a_full.sum()           # guarantee normalisation

    eta = 0.6                                 # step size for EG
    sub_last = time.time()
    for sub_it in range(2000):
        if time.time() > deadline:
            break

        conv = _full_convolution(a_full, fft_len=conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_full = 2.0 * _cross_correlation(w, a_full, fft_len=corr_fft_len)

        # exponentiated‑gradient update (keeps positivity)
        a_full = a_full * np.exp(-eta * grad_full)

        # renormalise (project onto the simplex)
        s = a_full.sum()
        if s < 1e-12:
            a_full = np.random.dirichlet(np.ones(n))
            s = a_full.sum()
        a_full = a_full / s

        conv = _full_convolution(a_full, fft_len=conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_local_val:
            best_local_val = val
            best_local_seq = a_full.copy()
            sub_last = time.time()

        if time.time() - sub_last > 5.0:
            break

    return best_local_seq, best_local_val

        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        print("Fallback to uniform vector")

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
