"""
Optimises a non‑negative step‑function sequence to minimise

        F(a) = 2·n·max(a∗a) / (sum a)²

The algorithm is a refined version of the multi‑stage scheme that was
previously used:

1.  Generate a diverse set of warm‑starts (random Dirichlet, edge‑heavy,
    Beta‑shaped, multi‑Gaussian, etc.).
2.  Run a smooth‑max Adam optimiser (more iterations and a slower
    temperature schedule).
3.  Refine with an exact exponentiated‑gradient (sub‑gradient) optimiser
    and a decreasing step‑size.
4.  Perform *gradient‑guided* hill‑climbing: the sub‑gradient tells which
    coordinates increase the objective the most – we move a small amount
    of mass from the “worst” index to the “best’’ one.
5.  Keep the best sequence across several lengths; the longest length
    (up to 32768) receives a final long‑run refinement.

All operations respect the simplex (non‑negative, sum‑to‑1) constraint,
use FFT‑based convolutions for speed and are bound by the global time
budget (default 1000 s).  The function ``propose_candidate`` returns the
best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# Utility FFT helpers
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)−1) using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) = n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)² (with sum a ≈ 1)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)

def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence by linear interpolation and renormalise to sum=1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new


def _generate_initial_candidates(
    n: int,
    best_seq: np.ndarray | None,
    hist_seq: np.ndarray | None,
) -> list[np.ndarray]:
    """
    Produce a modestly sized, diverse set of starting points for length n.
    All returned arrays are non‑negative and sum to 1.
    """
    cand = []

    # Warm‑start from global best (exact length)
    if best_seq is not None and best_seq.shape[0] == n:
        cand.append(best_seq.copy())
    # Interpolated warm‑start
    if best_seq is not None and best_seq.shape[0] != n:
        cand.append(_interpolate_seq(best_seq, n))

    # Historic sequence (if present)
    if hist_seq is not None:
        if hist_seq.shape[0] > n:
            tmp = hist_seq[:n]
        else:
            tmp = np.pad(hist_seq, (0, n - hist_seq.shape[0]))
        tmp = np.maximum(tmp, 0.0)
        tot = tmp.sum()
        if tot < 1e-12:
            tmp = np.random.dirichlet(np.ones(n))
        else:
            tmp = tmp / tot
        cand.append(tmp)

    # Pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # Edge‑concentrated power‑law shapes ((i+1)*(n-i))**α , α<0
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 5):
        shape = np.power(prod, alpha, where=prod > 0,
                         out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # Two‑Gaussian edge shapes
    for gamma in (0.2, 0.5, 1.0, 2.0):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        cand.append(shape / shape.sum())

    # Inverse‑root edge shapes (i+1)**‑α + (n‑i)**‑α
    for alpha in (0.5, 0.8, 1.0):
        i = np.arange(n, dtype=np.float64) + 1.0
        shape = np.power(i, -alpha) + np.power(n - i + 1.0, -alpha)
        cand.append(shape / shape.sum())

    # Three Gaussian peaks (0.25,0.5,0.75) with various widths
    for sigma in (0.05, 0.10, 0.20):
        x = np.arange(n, dtype=np.float64) / (n - 1)
        peaks = sum(np.exp(-0.5 * ((x - p) / sigma) ** 2) for p in (0.25, 0.5, 0.75))
        cand.append(peaks / peaks.sum())

    # Symmetric Beta‑like shapes ( (i+1)/n )**p * (1-(i)/n)**q
    for p, q in ((0.5, 0.5), (1.0, 1.0), (1.5, 1.5), (2.0, 2.0)):
        i = np.arange(n, dtype=np.float64) + 0.5
        shape = (i / n) ** p * (1 - (i - 1) / n) ** q
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # Shuffle and keep a limited number (12) to stay within budget
    np.random.shuffle(cand)
    return cand[:12]

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    max_iters: int,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best (a, value) encountered before the deadline.
    """
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.04  # slightly faster temperature rise

    best_a = a.copy()
    conv = _full_convolution(a, fft_len_conv)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # Soft‑max projection onto simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # Smooth‑max weights
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        ws = w.sum()
        if ws == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= ws

        # Gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # Back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # Temperature schedule
        lam = min(lam * lam_factor, 1e7)

        cur_val = 2.0 * n * max_c / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t:06d}] new best {cur_val:.6f}")

        # Small random shake after a few seconds without improvement
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


def _subgrad_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient optimisation (exponentiated‑gradient) on the true max.
    Uses a decreasing step‑size η_t = η0 / sqrt(t).
    """
    a = a0 / a0.sum()
    best_a = a.copy()
    if init_val is None:
        conv = _full_convolution(a, fft_len_conv)
        best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    else:
        best_val = init_val

    last_improve = time.time()
    eta0 = 0.6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if not active.any():
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        eta = eta0 / math.sqrt(it)
        # keep exponent bounded to avoid overflow
        exp_term = np.exp(np.clip(-eta * grad_a, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        cur_val = 2.0 * n * max_c / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it}] new best {cur_val:.6f}")

        if time.time() - last_improve > 4.0:
            break

    return best_a, best_val


def _gradient_hill_climb(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Directed hill‑climbing using the sub‑gradient to choose a source
    (high positive gradient) and a sink (most negative gradient) and
    moving a small amount of mass between them.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_conv) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if not active.any():
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        # Sub‑gradient (direction of increase)
        grad = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # If all gradients ≤ 0 we are (almost) optimal
        if grad.max() <= 1e-12:
            break

        src = int(np.argmax(grad))          # reduce this coordinate
        dst = int(np.argmin(grad))          # increase this coordinate

        # Amount to move: up to a fraction of the source mass
        eps_max = 0.08 * a[src]             # 8 % of source (tunable)
        if eps_max <= 0.0:
            # source already empty → pick random pair as fallback
            i1, i2 = np.random.randint(0, n, size=2)
            src, dst = i1, i2
            eps_max = 0.05 * a[src] if a[src] > 0 else 0.0

        if eps_max <= 0.0:
            # nothing to move, stop
            break

        # Try a few candidate epsilons and keep the best improvement
        best_eps = 0.0
        best_eps_val = cur_val
        for factor in (0.2, 0.5, 1.0):
            eps = eps_max * factor
            new_a = a.copy()
            new_a[src] -= eps
            new_a[dst] += eps
            if new_a[src] < -1e-15:
                continue
            # Renormalise only to suppress rounding drift
            new_a = new_a / new_a.sum()
            new_val = _objective(new_a, n, fft_len_conv)
            if new_val < best_eps_val:
                best_eps_val = new_val
                best_eps = eps

        if best_eps > 0.0:
            a[src] -= best_eps
            a[dst] += best_eps
            a = a / a.sum()
            cur_val = best_eps_val
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_a = a.copy()
                last_improve = time.time()
                if it % 50 == 0:
                    print(f"[n={n} GradHC it={it}] new best {cur_val:.6f}")

        else:
            # No improving move found – try a cheap random perturbation
            i1, i2 = np.random.randint(0, n, size=2)
            if i1 == i2:
                continue
            eps = 0.02 * a[i1]
            if eps <= 0:
                continue
            new_a = a.copy()
            new_a[i1] -= eps
            new_a[i2] += eps
            if new_a[i1] < -1e-15:
                continue
            new_a = new_a / new_a.sum()
            new_val = _objective(new_a, n, fft_len_conv)
            if new_val < cur_val:
                a = new_a
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_a = a.copy()
                    last_improve = time.time()

        if time.time() - last_improve > 5.0:
            # stagnation
            break

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # Lengths – we start small and end with a fairly large length.
    lengths = [1024, 2048, 4096, 8192, 16384, 32768]

    # Pre‑compute FFT lengths for conv / corr for each n.
    fft_conv = {}
    fft_corr = {}
    for n in lengths:
        conv_len = 2 * n - 1
        corr_len = conv_len + n - 1
        fft_conv[n] = _next_pow2(conv_len)
        fft_corr[n] = _next_pow2(corr_len)

    # Load historic solution if any.
    hist_seq = None
    if "height_sequence_1" in globals():
        tmp = globals()["height_sequence_1"]
        if isinstance(tmp, (list, tuple, np.ndarray)):
            hist_seq = np.asarray(tmp, dtype=np.float64)

    # ------------------------------------------------------------------
    # MAIN optimisation loop over lengths
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        fft_len_c = fft_conv[n]
        fft_len_x = fft_corr[n]

        # Generate candidates
        candidates = _generate_initial_candidates(n, best_seq, hist_seq)

        # Adaptive iteration budgets (more work on smaller n, still enough on large n)
        max_adam_iters = max(2000, int(18000 * 1024 / n) + 3000)
        max_subgrad_iters = max(800, int(3000 * 1024 / n) + 1200)
        max_hc_iters = max(1500, int(4000 * 1024 / n) + 2000)

        for cand_id, a_start in enumerate(candidates):
            if time.time() > deadline:
                break

            # ---- Phase 1 – smooth‑max Adam ----
            a_adam, val_adam = _adam_optimize(
                a_start, n, fft_len_c, fft_len_x, deadline, max_adam_iters
            )

            # ---- Phase 2 – exact sub‑gradient (exponentiated‑gradient) ----
            a_sub, val_sub = _subgrad_optimize(
                a_adam, n, fft_len_c, fft_len_x, deadline, init_val=val_adam,
                max_iters=max_subgrad_iters
            )

            # ---- Phase 3 – gradient‑guided hill‑climbing ----
            a_hc, val_hc = _gradient_hill_climb(
                a_sub, n, fft_len_c, fft_len_x, deadline, init_val=val_sub,
                max_iters=max_hc_iters
            )

            # Update global best if improved.
            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.copy()
                print(f"*** Global new best (n={n} cand={cand_id}) value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final long‑run refinement on the current global best (if time permits)
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        fft_len_c = fft_conv.get(n, _next_pow2(2 * n - 1))
        fft_len_x = fft_corr.get(n, _next_pow2(3 * n - 2))
        print("\n=== Final long‑run refinement of global best ===")
        # Use a longer Adam run with a slower learning rate.
        refined_seq, refined_val = _adam_optimize(
            best_seq, n, fft_len_c, fft_len_x, deadline,
            max_iters=int(1.5 * (18000 * 1024 / n) + 6000)
        )
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq.copy()
            print(f"*** Refined global best value={best_val:.6f}")

        # Additional gradient hill‑climb with many iterations (if time left)
        if time.time() < deadline - 2.0:
            extra_hc_iters = int(8000 * 1024 / n) + 4000
            final_seq, final_val = _gradient_hill_climb(
                best_seq, n, fft_len_c, fft_len_x, deadline,
                init_val=best_val,
                max_iters=extra_hc_iters
            )
            if final_val < best_val:
                best_val = final_val
                best_seq = final_seq.copy()
                print(f"*** Final hill‑climb improvement value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (unlikely)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return plain Python list as required.
    print(f"\n=== Finished. Best value ≈ {best_val:.6f} with length {best_seq.shape[0]} ===")
    return best_seq.tolist()
