    best = a

    for it in range(max_iter):
        M = max(conv(a))

            best = a_new
        # else might still progress? but keep new a anyway

    return best, best_val

def _lp_refine(a0, n, fft_len_conv, deadline, max_iter=10, t=0.5):
    a = a0.copy()
    # ensure sum to 1
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a, n, fft_len_conv)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # compute M
        conv = _full_convolution(a, fft_len_conv)
        M = conv.max()
        # build constraint matrix
        A_ub = _build_conv_ub_matrix(a) # function defined
        b_ub = np.full(2*n - 1, M * (1 + 1e-9), dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if not res.success:
            # fallback: break or random jitter
            # optionally jitter a small random perturbation
            a = a + np.random.randn(n) * 1e-4
            a = np.maximum(a, 0)
            a /= a.sum()
            continue
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break
        g = g0 / sum_g0
        # mixing
        a = (1 - t) * a + t * g
        # renormalize
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
        else:
            a /= s
        val = _objective(a, n, fft_len_conv)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # maybe reduce t adaptively?
        else:
            # optionally reduce t
            pass
    return best_a, best_val

"""
Multi‑stage optimiser for the step‑function attack.

1.  Search for a good shape on a modest size (n≈1024) using an exact linear
    programme: maximise the ℓ₁‑norm of a partner vector g under the
    constraint (f∗g)≤max(f∗f).  After normalising g we move a small step
    towards it (convex combination).  A few iterations of this “LP‑mix”
    already produce a structure with a markedly smaller max‑convolution.

2.  Interpolate the best small‑size shape to the target length
    (up to 16384) and finish the optimisation with the smooth‑max Adam
    optimiser, an exact sub‑gradient (exponentiated‑gradient) pass and a
    cheap hill‑climber – the same refinement pipeline that was used in the
    reference implementation.

All steps stop as soon as the overall time budget (default 1000 s) is
exhausted.  The function returns a plain ``list`` of non‑negative floats
ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import spdiags

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)‑1 using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) = n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(a∗a) / (sum a)²."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Simple helpers
def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence by linear interpolation and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new


def _generate_initial_candidates(
    n: int, best_seq: np.ndarray | None, hist_seq: np.ndarray | None
) -> list[np.ndarray]:
    """Generate a modestly sized, diverse set of starting points (sum=1)."""
    cand = []

    # warm start from the current global best (exact length)
    if best_seq is not None and best_seq.shape[0] == n:
        cand.append(best_seq.copy())
    # warm start by interpolating the global best
    if best_seq is not None and best_seq.shape[0] != n:
        cand.append(_interpolate_seq(best_seq, n))

    # historic sequence if we have one
    if hist_seq is not None:
        if hist_seq.shape[0] > n:
            tmp = hist_seq[:n]
        else:
            tmp = np.pad(hist_seq, (0, n - hist_seq.shape[0]))
        tmp = np.maximum(tmp, 0.0)
        tot = tmp.sum()
        if tot < 1e-12:
            tmp = np.random.dirichlet(np.ones(n))
        else:
            tmp = tmp / tot
        cand.append(tmp)

    # pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # edge‑concentrated power‑law shapes ((i+1)*(n-i))**α , α<0
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 5):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # two‑Gaussian edge shapes
    for gamma in (0.2, 0.5, 1.0, 2.0):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        cand.append(shape / shape.sum())

    # inverse‑root edge shapes
    for alpha in (0.5, 0.8, 1.0):
        i = np.arange(n, dtype=np.float64) + 1.0
        shape = np.power(i, -alpha) + np.power(n - i + 1.0, -alpha)
        cand.append(shape / shape.sum())

    # three Gaussian peaks
    for sigma in (0.05, 0.10, 0.20):
        x = np.arange(n, dtype=np.float64) / (n - 1)
        peaks = sum(np.exp(-0.5 * ((x - p) / sigma) ** 2) for p in (0.25, 0.5, 0.75))
        cand.append(peaks / peaks.sum())

    np.random.shuffle(cand)
    return cand[:8]          # keep a modest number to respect the budget


# ----------------------------------------------------------------------
# Linear‑programming based “LP‑mix” refinement
def _build_conv_ub_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the sparse matrix A (shape (2n‑1)×n) such that
    A @ g = f * g  (linear convolution) for any g.
    """
    n = f.shape[0]
    m = 2 * n - 1
    # each diagonal -i (i=0..n‑1) contains the constant f[i]
    data = np.broadcast_to(f[:, None], (n, n)).astype(np.float64)   # shape (n,n)
    offsets = np.arange(0, -n, -1)                                 # 0, -1, … , -(n‑1)
    A = spdiags(data, offsets, m, n, format="csr")
    return A


def _lp_refine(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iter: int = 5,
    step: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Perform a few iterations of the LP‑mix scheme.
    Returns the best (a, objective) found before the deadline.
    """
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a, n, fft_len)

    for it in range(max_iter):
        if time.time() > deadline:
            break

        # current bound M = max(f*f)
        conv = _full_convolution(a, fft_len)
        M = conv.max()

        # construct linear constraints A @ g ≤ M
        A = _build_conv_ub_matrix(a)
        b = np.full(2 * n - 1, M * (1.0 + 1e-9), dtype=np.float64)

        # maximise sum(g)  → minimise –sum(g)
        res = linprog(
            -np.ones(n, dtype=np.float64),
            A_ub=A,
            b_ub=b,
            bounds=(0, None),
            method="highs",
            options={"presolve": True},
        )

        if not res.success:
            # fallback – inject a tiny random perturbation and continue
            a += (np.random.rand(n) - 0.5) * 1e-4
            a = np.maximum(a, 0.0)
            a /= a.sum()
            continue

        g0 = np.maximum(res.x, 0.0)
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            break

        # normalise to keep the same ℓ₁‑norm as a
        g = g0 / sum_g0

        # convex combination and projection onto simplex
        a = (1.0 - step) * a + step * g
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
        else:
            a = a / s

        val = _objective(a, n, fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # optionally shrink the step size when we make progress
            step = min(step * 1.05, 0.9)

    return best_a, best_val


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser (identical to the reference implementation)
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    max_iters: int,
) -> tuple[np.ndarray, float]:
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.03

    best_a = a.copy()
    best_val = _objective(a, n, fft_len_conv)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # simplex via soft‑max
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # smooth‑max weights
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        ws = w.sum()
        if ws == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws

        # surrogate gradient
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        cur_val = _objective(a, n, fft_len_conv)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t:06d}] new best {cur_val:.6f}")

        # occasional random shake when stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val

# Exact sub‑gradient (exponentiated gradient)
def _subgrad_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    a = a0 / a0.sum()
    best_a = a.copy()
    if init_val is None:
        best_val = _objective(a, n, fft_len_conv)
    else:
        best_val = init_val
    last_improve = time.time()
    eta = 0.6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        cur_val = _objective(a, n, fft_len_conv)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it}] new best {cur_val:.6f}")

        if time.time() - last_improve > 4.0:
            break

    return best_a, best_val

# Simple hill‑climb local search
def _hill_climb_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_conv) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2:
            continue

        # move up to 5 % of the mass from i1 to i2
        eps = np.random.rand() * 0.05 * a[i1]
        if eps <= 0.0:
            continue
        new_a = a.copy()
        new_a[i1] -= eps
        new_a[i2] += eps
        if new_a[i1] < -1e-15:
            continue
        new_a /= new_a.sum()

        new_val = _objective(new_a, n, fft_len_conv)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            if it % 50 == 0:
                print(f"[n={n} HillClimb it={it}] new best {new_val:.6f}")
        else:
            # very occasional uphill move to escape plateaus
            if np.random.rand() < 0.001:
                a = new_a

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣ LP‑mix stage on a modest size (n≈1024)
    small_n = 1024
    fft_len_small = 1 << int(np.ceil(np.log2(2 * small_n - 1)))

    print("\n=== LP‑mix stage (n=1024) ===")
    candidates = _generate_initial_candidates(small_n, best_seq, None)

    for cid, cand in enumerate(candidates):
        if time.time() > deadline:
            break
        a_start = np.asarray(cand, dtype=np.float64)

        # a few LP‑mix iterations (fast, effectful)
        a_lp, val_lp = _lp_refine(
            a_start,
            small_n,
            fft_len_small,
            deadline,
            max_iter=6,      # a handful of rounds is enough
            step=0.5,
        )

        # brief Adam / sub‑gradient / hill‑climb to polish the small vector
        a_adam, val_adam = _adam_optimize(
            a_lp,
            small_n,
            fft_len_small,
            fft_len_small,
            deadline,
            max_iters=1500,
        )
        a_sub, val_sub = _subgrad_optimize(
            a_adam,
            small_n,
            fft_len_small,
            fft_len_small,
            deadline,
            init_val=val_adam,
            max_iters=1500,
        )
        a_hc, val_hc = _hill_climb_optimize(
            a_sub,
            small_n,
            fft_len_small,
            deadline,
            init_val=val_sub,
            max_iters=1500,
        )

        if val_hc < best_val:
            best_val = val_hc
            best_seq = a_hc.copy()
            print(f"*** New global best after candidate {cid} → {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Upscale and refine on progressively larger lengths
    final_lengths = [2048, 4096, 8192, 16384]

    for n in final_lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Upscaling to n={n} ===")
        # interpolate the current best to the new length
        a = _interpolate_seq(best_seq, n)

        fft_len_c = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_x = 1 << int(np.ceil(np.log2(3 * n)))   # safe over‑estimate for correlation

        # Adam – a longer run for the large vector
        max_adam_iters = int(15000 * 1024 / n) + 2500
        a_adam, val_adam = _adam_optimize(
            a, n, fft_len_c, fft_len_x, deadline, max_iters=max_adam_iters
        )
        a_sub, val_sub = _subgrad_optimize(
            a_adam, n, fft_len_c, fft_len_x, deadline, init_val=val_adam, max_iters=2000
        )
        a_hc, val_hc = _hill_climb_optimize(
            a_sub, n, fft_len_c, deadline, init_val=val_sub, max_iters=2000
        )

        if val_hc < best_val:
            best_val = val_hc
            best_seq = a_hc.copy()
            print(f"*** Improved global best at n={n} → {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Final long‑run Adam refinement on the best sequence (if time permits)
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        fft_len_c = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_x = 1 << int(np.ceil(np.log2(3 * n)))
        print("\n=== Final long‑run Adam refinement ===")
        refined_seq, refined_val = _adam_optimize(
            best_seq,
            n,
            fft_len_c,
            fft_len_x,
            deadline,
            max_iters=int(1.5 * (15000 * 1024 / n) + 3000),
        )
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq.copy()
            print(f"*** Refined global best value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (unlikely)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    return best_seq.tolist()
