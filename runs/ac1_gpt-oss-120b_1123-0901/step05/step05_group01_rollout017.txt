import time, numpy as np

def _next_pow2(x):
    return 1 << (x-1).bit_length()

def _full_convolution(a, fft_len):
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2*n-1]

def _cross_correlation(w, a, fft_len):
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n-1
    return corr[start:start+n]

def _objective(a, n, fft_len):
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_c / (s*s)  # sum=1 => 2*n*max_c

def _interpolate_seq(seq, new_n):
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0., 1., old_n)
    x_new = np.linspace(0., 1., new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new

def _adam_optimize_sym(a0, n, fft_len_c, fft_len_corr, deadline, max_iters):
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)  # shape (n,)
    # enforce symmetry on logits initially
    logits = (logits + logits[::-1]) / 2.0

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.5
    lam_factor = 1.03

    best_a = a.copy()
    conv = _full_convolution(a, fft_len_c)
    best_val = 2.0 * n * conv.max() / (a.sum()**2)  # sum=1
    last_improve = time.time()

    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break

        # softmax to produce a
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()
        # enforce symmetry
        a = (a + a[::-1]) / 2.0
        # re‑normalise (should already sum to 1)
        a = a / a.sum()

        # smooth‑max weights
        conv = _full_convolution(a, fft_len_c)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        ws = w.sum()
        if ws == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / ws

        # gradient of surrogate wrt a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # backprop through softmax (logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # enforce symmetry on logits again
        logits = (logits + logits[::-1]) / 2.0

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        cur_val = 2.0 * n * max_c / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t}] new best {cur_val:.6f}")

        if time.time() - last_improve > 4.0:
            # random shake
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
    return best_a, best_val

def _subgrad_optimize_sym(a0, n, fft_len_c, fft_len_corr, deadline, init_val=None, max_iters=2000):
    a = a0 / a0.sum()
    a = (a + a[::-1]) / 2.0
    a = a / a.sum()
    best_a = a.copy()
    if init_val is None:
        conv = _full_convolution(a, fft_len_c)
        best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    else:
        best_val = init_val
    last_improve = time.time()
    eta = 0.6

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_c)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        a = a * np.exp(-eta * grad_a)
        # enforce symmetry
        a = (a + a[::-1]) / 2.0
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        conv = _full_convolution(a, fft_len_c)
        cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it}] new best {cur_val:.6f}")

        if time.time() - last_improve > 4.0:
            break

    return best_a, best_val

def _hill_climb_optimize_sym(a0, n, fft_len_c, deadline, init_val=None, max_iters=2000):
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_c) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break

        # pick two positions (respect symmetry)
        i1 = np.random.randint(0, n//2 + 1)  # include middle if odd
        # choose counterpart i1c = n-1-i1
        # pick another position i2 distinct
        i2 = np.random.randint(0, n//2 + 1)
        if i2 == i1:
            continue

        # transfer some mass from i1 to i2 (both sides)
        # total transferable from both symmetric sites
        total_mass_i1 = a[i1] + (a[n-1-i1] if i1 != n-1-i1 else 0.0)
        if total_mass_i1 <= 0:
            continue
        eps = np.random.rand() * 0.05 * total_mass_i1

        # create new array copy
        new_a = a.copy()
        # subtract from i1 and its mirror
        new_a[i1] -= eps/2
        if i1 != n-1-i1:
            new_a[n-1-i1] -= eps/2
        # add to i2 and its mirror
        new_a[i2] += eps/2
        if i2 != n-1-i2:
            new_a[n-1-i2] += eps/2
        # ensure non-negatives
        if new_a[i1] < -1e-12 or new_a[n-1-i1] < -1e-12:
            continue
        # renorm
        s = new_a.sum()
        if s <= 0:
            continue
        new_a = new_a / s

        new_val = _objective(new_a, n, fft_len_c)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            if it % 50 == 0:
                print(f"[n={n} HillClimb it={it}] new best {new_val:.6f}")
        else:
            # occasionally accept uphill to escape freezing
            if np.random.rand() < 0.001:
                a = new_a

    return best_a, best_val

2.  Pick the few most promising candidates (smallest current objective)
    and refine each with three optimisation phases:
      • Adam on a smooth‑max surrogate of the convolution (log‑softmax
        parametrisation, symmetric logits, temperature schedule).
      • Exact sub‑gradient (exponentiated‑gradient) on the true max.
      • Cheap hill‑climbing that moves a tiny amount of mass between two
        positions and re‑symmetrises.
3.  Keep the globally best sequence found.
4.  If time remains, run a long Adam refinement on the global best.

All steps respect a global time budget (default 1000 s) and only
return a plain Python list of floats, ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation of w with a (used for gradients).
    Returns a length‑n vector.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)^2 (sum a ≈ 1)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence by linear interpolation and re‑normalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new


# ----------------------------------------------------------------------
def _generate_candidates(
    n: int,
    best_seq: np.ndarray | None,
    hist_seq: np.ndarray | None,
) -> list[np.ndarray]:
    """Produce a modest but diverse pool of symmetric, normalised candidates."""
    cand = []

    # 0) warm‑starts from previous best
    if best_seq is not None and best_seq.shape[0] == n:
        cand.append(best_seq.copy())
    if best_seq is not None:
        cand.append(_interpolate_seq(best_seq, n))

    # 1) historic sequence (if any)
    if hist_seq is not None:
        if hist_seq.shape[0] > n:
            tmp = hist_seq[:n]
        else:
            tmp = np.pad(hist_seq, (0, n - hist_seq.shape[0]))
        tmp = np.maximum(tmp, 0.0)
        tot = tmp.sum()
        if tot < 1e-12:
            tmp = np.random.dirichlet(np.ones(n))
        else:
            tmp = tmp / tot
        cand.append(tmp)

    # 2) a few random Dirichlet draws
    for _ in range(3):
        cand.append(np.random.dirichlet(np.ones(n)))

    # 3) aggressive edge‑concentrated power‑law shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)                 # (i+1)*(n‑i)
    for alpha in np.linspace(-6.0, -0.3, 12):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape = shape / shape.sum()
            cand.append(shape)

    # 4) exponential edge shapes   a_i ∝ e^{‑β i}+e^{‑β (n‑1‑i)}
    for beta in (0.3, 0.6, 1.0, 1.5, 2.5, 4.0):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-beta * i) + np.exp(-beta * (n - 1 - i))
        shape = shape / shape.sum()
        cand.append(shape)

    # 5) inverse‑power edge shapes   (i+1)^{‑p}+(n‑i)^{‑p}
    for p in (0.8, 1.2, 1.5, 2.0, 2.5):
        i = np.arange(1, n + 1, dtype=np.float64)
        shape = np.power(i, -p) + np.power(n - i + 1.0, -p)
        shape = shape / shape.sum()
        cand.append(shape)

    # 6) three‑peak Gaussian mixture (peaks at 0.2,0.5,0.8)
    xs = np.arange(n, dtype=np.float64) / (n - 1)
    for sigma in (0.04, 0.07, 0.10):
        peaks = sum(np.exp(-0.5 * ((xs - mu) / sigma) ** 2) for mu in (0.2, 0.5, 0.8))
        peaks = peaks / peaks.sum()
        cand.append(peaks)

    # 7) random mixtures of an edge‑Gaussian and a uniform component
    for _ in range(2):
        w = np.random.rand()
        edge = np.exp(-0.5 * ((xs - 0.0) / 0.07) ** 2) + np.exp(-0.5 * ((xs - 1.0) / 0.07) ** 2)
        mix = w * edge + (1.0 - w) * np.ones_like(xs)
        mix = mix / mix.sum()
        cand.append(mix)

    # ---- always enforce non‑negativity & normalisation ----
    cand = [np.maximum(c, 0.0) for c in cand]
    cand = [
        c / c.sum() if c.sum() > 0.0 else np.random.dirichlet(np.ones(n))
        for c in cand
    ]

    # keep a reasonable number (the optimizer is expensive)
    if len(cand) > 30:
        np.random.shuffle(cand)
        cand = cand[:30]
    return cand


# ----------------------------------------------------------------------
def _adam_optimize_sym(
    a0: np.ndarray,
    n: int,
    fft_len_c: int,
    fft_len_corr: int,
    deadline: float,
    max_iters: int,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max surrogate.
    Symmetry (a_i = a_{n‑1‑i}) is enforced by averaging the logits each step.
    """
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)
    # enforce initial symmetry
    logits = (logits + logits[::-1]) / 2.0

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.5
    lam_factor = 1.03

    best_a = a.copy()
    conv = _full_convolution(a, fft_len_c)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max → a (lies on the simplex)
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()
        # enforce symmetry on the probability vector
        a = (a + a[::-1]) / 2.0
        a = a / a.sum()

        # smooth‑max weights over the convolution entries
        conv = _full_convolution(a, fft_len_c)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        ws = w.sum()
        w = w / ws if ws != 0.0 else np.full_like(w, 1.0 / w.size)

        # gradient of the surrogate wrt a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through the soft‑max (gradients w.r.t. logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # symmetry enforcement on logits
        logits = (logits + logits[::-1]) / 2.0

        # temperature schedule for the smooth‑max
        lam = min(lam * lam_factor, 1e7)

        cur_val = 2.0 * n * max_c / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t:06d}] new best {cur_val:.6f}")

        # occasional random shake to escape stagnation
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


def _subgrad_optimize_sym(
    a0: np.ndarray,
    n: int,
    fft_len_c: int,
    fft_len_corr: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient optimisation (exponentiated‑gradient) with symmetry.
    """
    a = a0 / a0.sum()
    a = (a + a[::-1]) / 2.0
    a = a / a.sum()
    best_a = a.copy()
    if init_val is None:
        conv = _full_convolution(a, fft_len_c)
        best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    else:
        best_val = init_val
    last_improve = time.time()
    eta = 0.6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_c)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient step (preserves non‑negativity)
        a = a * np.exp(-eta * grad_a)
        a = (a + a[::-1]) / 2.0   # enforce symmetry
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        cur_val = 2.0 * n * _full_convolution(a, fft_len_c).max() / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it:06d}] new best {cur_val:.6f}")

        if time.time() - last_improve > 4.0:
            break

    return best_a, best_val


def _hill_climb_optimize_sym(
    a0: np.ndarray,
    n: int,
    fft_len_c: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: move a small amount of mass between two random
    coordinates, keep the move if it improves the exact objective.
    Symmetry is re‑imposed after every trial.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_c) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2:
            continue

        # transfer up to 5 % of the mass at i1 towards i2
        eps = np.random.rand() * 0.05 * a[i1]
        if eps <= 0.0:
            continue

        new_a = a.copy()
        new_a[i1] -= eps
        new_a[i2] += eps

        # re‑symmetrise and renormalise
        new_a = (new_a + new_a[::-1]) / 2.0
        new_a = new_a / new_a.sum()

        new_val = _objective(new_a, n, fft_len_c)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            if it % 100 == 0:
                print(f"[n={n} HillClimb it={it:06d}] new best {new_val:.6f}")
        else:
            # rare uphill jumps to avoid tiny plateaus
            if np.random.rand() < 0.001:
                a = new_a

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function sequence that minimises the
    evaluation function.  Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # 0.5 s safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # Lengths to explore – start modest, finish with the largest allowed.
    lengths = [512, 1024, 2048, 4096, 8192, 16384]

    # Pre‑compute FFT sizes for convolution and cross‑correlation.
    fft_len_conv = {}
    fft_len_corr = {}
    for n in lengths:
        fft_len_conv[n] = _next_pow2(2 * n - 1)
        fft_len_corr[n] = _next_pow2(2 * n - 1 + n - 1)   # 3·n‑2 ≈

    # Historical solution (if the notebook supplied one).
    hist_seq = None
    if "height_sequence_1" in globals():
        tmp = globals()["height_sequence_1"]
        if isinstance(tmp, (list, tuple, np.ndarray)):
            hist_seq = np.asarray(tmp, dtype=np.float64)

    # ------------------------------------------------------------------
    # MAIN optimisation loop over the chosen lengths
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        fft_c = fft_len_conv[n]
        fft_corr = fft_len_corr[n]

        # ------------------------------------------------------------------
        # 1) generate a pool of diverse start points (all symmetric)
        candidates = _generate_candidates(n, best_seq, hist_seq)

        # ------------------------------------------------------------------
        # 2) cheap evaluation of the pool, keep the most promising few
        scores = [_objective(c, n, fft_c) for c in candidates]
        order = np.argsort(scores)
        top_k = min(5, len(candidates))          # number of candidates to fully refine
        chosen = [candidates[i] for i in order[:top_k]]
        print(f"  evaluated {len(candidates)} candidates, best preliminary value {scores[order[0]]:.6f}")

        # ------------------------------------------------------------------
        # 3) Refine each chosen candidate with three phases
        #   (Adam → exact sub‑gradient → hill‑climb)
        max_adam_iters = int(20000 * 1024 / n) + 2000   # more iterations for smaller n
        for cid, a_start in enumerate(chosen):
            if time.time() > deadline:
                break

            print(f"  -- candidate {cid+1}/{top_k} (init value {scores[order[cid]]:.6f})")
            a_adam, val_adam = _adam_optimize_sym(
                a_start, n, fft_c, fft_corr, deadline, max_adam_iters
            )
            a_sub, val_sub = _subgrad_optimize_sym(
                a_adam, n, fft_c, fft_corr, deadline, init_val=val_adam
            )
            a_hc, val_hc = _hill_climb_optimize_sym(
                a_sub, n, fft_c, deadline, init_val=val_sub
            )

            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.copy()
                print(f"*** Global new best (n={n}, cand={cid}) value={best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional long‑run refinement on the current global best
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        fft_c = fft_len_conv.get(n, _next_pow2(2 * n - 1))
        fft_corr = fft_len_corr.get(n, _next_pow2(3 * n - 2))
        print("\n=== Long‑run refinement of the global best ===")
        refined_seq, refined_val = _adam_optimize_sym(
            best_seq,
            n,
            fft_c,
            fft_corr,
            deadline,
            max_iters=int(1.5 * (20000 * 1024 / n) + 3000),
        )
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq.copy()
            print(f"*** Refined global best value={best_val:.6f}")

    # ------------------------------------------------------------------
    # 5) Fallback (should never happen)
    if best_seq is None:
        print("Fallback: using uniform vector.")
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator.
    print(f"\nFinished – best objective ≈ {best_val:.6f}")
    return best_seq.tolist()
