from scipy.optimize import linprog
from scipy import sparse

def _lp_improve(f, M, t, deadline, max_time_per_lp=10.0):
    n = len(f)
    # Build A_ub matrix
    rows = []
    cols = []
    data = []
    for i in range(n):
        # row index start = i, end = i+n
        # iterate over k = i..i+n-1
        # Column i gets value f[i] for each row k
        rows.extend(range(i, i+n))
        cols.extend([i]*n)
        data.extend([f[i]]*n)
    A = sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # objective to maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    # bounds for variables: (0, None)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback: return f unchanged
        return f
    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        return f
    g = b / sum_b  # normalized to sum=1 (since f sum =1)
    # combine
    f_new = (1-t) * f + t * g
    return f_new
```

def _lp_refine(a, n, M, t, deadline, max_iters=5):
    # Solve LP to get b that maximizes sum b under conv(f,b) <= M.
    # We'll repeat up to max_iters but break if time exceeded.
    best_a = a.copy()
    for i in range(max_iters):
        if time.time() > deadline:
            break
        # build sparse constraint matrix for current a
        rows = []
        cols = []
        data = []
        for col in range(n):
            start = col
            end = col + n
            rows.extend(range(start, end))
            cols.extend([col]*n)
            # each entry = a[col]
            data.extend([a[col]]*n)
        A = sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        # bounds
        bounds = [(0, None)]*n
        # RHS vector
        b_ub = np.full(2*n-1, M)
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break
        g = b / sum_b
        a = (1-t) * a + t * g
        # recompute M after update
        conv = _full_convolution(a, 1 << int(np.ceil(np.log2(2*n-1))))
        M = conv.max()
        best_a = a.copy()
        # maybe reduce t each iteration
        t *= 0.5
    return best_a
```

def _generate_initial_candidates(n, best_seq, hist_seq):
    cand = []
    # best_seq direct and interpolated
    if best_seq is not None:
        if best_seq.shape[0] == n:
            cand.append(best_seq.copy())
        else:
            cand.append(_interpolate_seq(best_seq, n))
    # hist_seq similar
    if hist_seq is not None:
        if hist_seq.shape[0] > n:
            tmp = hist_seq[:n]
        else:
            tmp = np.pad(hist_seq, (0, n - hist_seq.shape[0]))
        tmp = np.maximum(tmp, 0.0)
        tot = tmp.sum()
        if tot < 1e-12:
            tmp = np.random.dirichlet(np.ones(n))
        else:
            tmp = tmp / tot
        cand.append(tmp)

    # random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # ... existing families

    # Beta distributions
    x = np.linspace(0, 1, n)
    for p in np.linspace(0.2, 5, 4):
        for q in np.linspace(0.2, 5, 4):
            shape = (x ** p) * ((1 - x) ** q)
            if shape.sum() > 0:
                cand.append(shape / shape.sum())

    # random convex combos of existing candidates
    base = cand.copy()
    for _ in range(5):
        a, b = np.random.choice(len(base), size=2, replace=False)
        lam = np.random.rand()
        combo = lam * base[a] + (1 - lam) * base[b]
        combo = combo / combo.sum()
        cand.append(combo)

    # trim list
    np.random.shuffle(cand)
    max_cand = 20 if n <= 1024 else 12
    return cand[:max_cand]

"""
Propose a step‑function sequence that minimises
    F(a) = 2·n·max(a∗a) / (sum a)² .

Algorithm
---------

* **Diversified warm‑starts** – Dirichlet draws, power‑law edges,
  Gaussian peaks, Beta‑shaped densities and random convex combinations.
  All candidates are projected onto the probability simplex and made
  symmetric (a = (a + a[::-1])/2).

* **Heavy Adam optimisation** on a smooth‑max surrogate.
  The temperature (lam) is increased exponentially, the learning rate
  decays slowly and after each soft‑max step we re‑symmetrise the vector.
  A large number of iterations is spent on the *small* lengths
  (512‑2048) where each iteration is cheap.

* **Exact sub‑gradient (exponentiated‑gradient) descent** on the true
  max‑convolution, again with a symmetry projection.

* **Local hill‑climbing** – random mass transfers between two bins.

* **Linear‑programming refinement** – with the current sequence `f`
  we solve the LP
      maximise   Σ b_i
      subject to b ≥ 0 , (f * b)_k ≤ M   (∀k),
  where M = max(f∗f).  The optimal `b` is renormalised to unit sum and
  mixed with `f` (mixing factor t decreasing from 0.5).  A few LP steps
  dramatically tighten the bound.

* **Interpolation to larger lengths** and a final cheap Adam run.

The whole routine respects a 1000 s wall‑clock budget and returns the
best sequence found as a plain Python list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy import sparse

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using an FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) required for the gradient."""
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)²."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce a_i = a_{n-1-i} and re‑normalise."""
    a = (a + a[::-1]) * 0.5
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones_like(a))
    else:
        a = a / s
    return a


def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence by linear interpolation and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new


def _generate_initial_candidates(n: int,
                                 best_seq: np.ndarray | None,
                                 hist_seq: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a modestly sized, diverse set of starting points for length n.
    All returned arrays are non‑negative and sum to 1, and are symmetric.
    """
    cand = []

    # Warm‑start from the global best (exact or interpolated)
    if best_seq is not None:
        if best_seq.shape[0] == n:
            cand.append(best_seq.copy())
        else:
            cand.append(_interpolate_seq(best_seq, n))

    # Historic sequence, if any
    if hist_seq is not None:
        if hist_seq.shape[0] > n:
            tmp = hist_seq[:n]
        else:
            tmp = np.pad(hist_seq, (0, n - hist_seq.shape[0]))
        tmp = np.maximum(tmp, 0.0)
        tot = tmp.sum()
        if tot < 1e-12:
            tmp = np.random.dirichlet(np.ones(n))
        else:
            tmp = tmp / tot
        cand.append(tmp)

    # Pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # Edge‑concentrated power‑law shapes ((i+1)*(n-i))**α , α<0
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 5):
        shape = np.power(prod, alpha, where=prod > 0,
                         out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # Two‑Gaussian edge shapes
    for gamma in (0.2, 0.5, 1.0, 2.0):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        cand.append(shape / shape.sum())

    # Inverse‑root edge shapes
    for alpha in (0.5, 0.8, 1.0):
        i = np.arange(n, dtype=np.float64) + 1.0
        shape = np.power(i, -alpha) + np.power(n - i + 1.0, -alpha)
        cand.append(shape / shape.sum())

    # Three Gaussian peaks
    for sigma in (0.05, 0.10, 0.20):
        x = np.arange(n, dtype=np.float64) / (n - 1)
        peaks = sum(np.exp(-0.5 * ((x - p) / sigma) ** 2) for p in (0.25, 0.5, 0.75))
        cand.append(peaks / peaks.sum())

    # Beta‑like families (x^p (1-x)^q)
    x = np.linspace(0.0, 1.0, n)
    for p in np.linspace(0.2, 5.0, 4):
        for q in np.linspace(0.2, 5.0, 4):
            shape = (x ** p) * ((1.0 - x) ** q)
            if shape.sum() > 0:
                cand.append(shape / shape.sum())

    # Random convex combinations of already generated shapes
    base = cand.copy()
    rng = np.random.default_rng()
    for _ in range(5):
        a, b = rng.choice(len(base), size=2, replace=False)
        lam = rng.random()
        combo = lam * base[a] + (1.0 - lam) * base[b]
        combo = combo / combo.sum()
        cand.append(combo)

    # Symmetrise, normalise and keep a limited number
    cand = [_symmetrize(c) for c in cand]
    np.random.shuffle(cand)
    max_cand = 20 if n <= 1024 else 12
    return cand[:max_cand]

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   fft_len_conv: int,
                   fft_len_corr: int,
                   deadline: float,
                   max_iters: int) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best (a, value) encountered before the deadline.
    The iterate is forced to stay symmetric.
    """
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5          # smooth‑max temperature
    lam_factor = 1.05  # a little slower increase than before

    best_a = a.copy()
    conv = _full_convolution(a, fft_len_conv)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # simplex via soft‑max
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # enforce symmetry after the soft‑max
        a = _symmetrize(a)

        # smooth‑max weights
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        ws = w.sum()
        if ws == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule (smooth‑max becomes sharper)
        lam = min(lam * lam_factor, 1e7)

        cur_val = 2.0 * n * max_c / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t:06d}] new best {cur_val:.6f}")

        # occasional random shake when stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


def _subgrad_optimize(a0: np.ndarray,
                      n: int,
                      fft_len_conv: int,
                      fft_len_corr: int,
                      deadline: float,
                      init_val: float | None = None,
                      max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient optimisation (exponentiated‑gradient) on the true max.
    The iterate stays symmetric.
    """
    a = a0 / a0.sum()
    best_a = a.copy()
    if init_val is None:
        conv = _full_convolution(a, fft_len_conv)
        best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    else:
        best_val = init_val
    last_improve = time.time()
    eta = 0.6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad_a)

        # project back onto simplex and enforce symmetry
        a = _symmetrize(a)

        conv = _full_convolution(a, fft_len_conv)
        cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it}] new best {cur_val:.6f}")

        if time.time() - last_improve > 4.0:
            break

    return best_a, best_val


def _hill_climb_optimize(a0: np.ndarray,
                         n: int,
                         fft_len_conv: int,
                         deadline: float,
                         init_val: float | None = None,
                         max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Simple cheap local search: move a small amount of mass between two bins.
    The sequence is symmetrised after each accepted move.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_conv) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2:
            continue

        # transfer up to 5 % of mass from i1 to i2
        eps = np.random.rand() * 0.05 * a[i1]
        if eps <= 0.0:
            continue
        new_a = a.copy()
        new_a[i1] -= eps
        new_a[i2] += eps
        if new_a[i1] < -1e-15:
            continue
        new_a = _symmetrize(new_a)
        new_a /= new_a.sum()

        new_val = _objective(new_a, n, fft_len_conv)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            if it % 50 == 0:
                print(f"[n={n} HillClimb it={it}] new best {new_val:.6f}")
        else:
            # occasional uphill move to escape tiny plateaus
            if np.random.rand() < 0.001:
                a = new_a

    return best_a, best_val


def _lp_refine(a: np.ndarray,
               n: int,
               max_conv: float,
               deadline: float,
               max_steps: int = 3,
               init_mix: float = 0.5) -> np.ndarray:
    """
    Linear‑programming refinement described in the statement.
    Repeatedly solves:
        maximise Σ b_i   s.t.  b ≥ 0 ,  (a * b)_k ≤ max_conv  (∀k)
    The obtained b is normalised to unit sum and mixed into a.
    The iterate is kept symmetric.
    """
    t = init_mix
    for step in range(max_steps):
        if time.time() > deadline:
            break

        # Build sparse matrix for the constraints (a * b)_k ≤ max_conv
        rows = []
        cols = []
        data = []
        for col in range(n):
            start = col
            end = col + n
            rows.extend(range(start, end))
            cols.extend([col] * n)
            data.extend([a[col]] * n)
        A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        b_ub = np.full(2 * n - 1, max_conv)
        c = -np.ones(n)          # minimise –Σb_i  → maximise Σb_i
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            break

        b = res.x
        sb = b.sum()
        if sb < 1e-12:
            break
        g = b / sb                # renormalise to unit sum
        g = _symmetrize(g)

        # convex combination
        a = (1.0 - t) * a + t * g
        a = _symmetrize(a)
        a /= a.sum()

        # update bound for next LP
        conv = _full_convolution(a, 1 << int(np.ceil(np.log2(2 * n - 1))))
        max_conv = conv.max()

        # shrink mixing factor
        t *= 0.5

    return a

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    best_seq = None
    best_val = float("inf")

    # 1. Heavily optimise on modest lengths where each iteration is cheap.
    # --------------------------------------------------------------
    small_lengths = [512, 1024, 2048]          # many iterations, many starts
    for n in small_lengths:

            break
        print(f"\n=== Optimising small length n={n} ===")

        # candidate pool

        for cid, a_start in enumerate(candidates):

    print(f"\n=== Finished. Best value found: {best_val:.6f} (length {best_seq.shape[0]}) ===")
    return best_seq.tolist()
