import numpy as np, time, math, random

def full_convolution(a, fft_len=None):
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2*n-1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa*fa, fft_len)[:2*n-1]
    return conv

def cross_correlation(w, a, fft_len):
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw*fr, fft_len)
    start = n-1
    return corr[start:start+n]

def smooth_max_and_weights(conv, tau):
    cmax = conv.max()
    shifted = (conv - cmax) / tau
    # shift ensures numerical stability
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    smooth = cmax + tau * np.log(sum_exp)
    weights = exp_shifted / sum_exp
    return smooth, weights

"""
Propose a non‑negative step‑function (probability vector) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)^2)

The algorithm is an enhanced version of the previous Adam‑based optimiser.
Key ideas
* Use several problem sizes (1024…8192). Larger n gives more freedom.
* Much longer optimisation budgets per length and many restarts.
* Exponential temperature annealing for the smooth‑max (log‑sum‑exp) and a
  final “hard‑max” fine‑tuning phase.
* Adam with a modest learning‑rate decay and occasional random jitter when
  progress stalls.
* After gradient descent a cheap greedy mass‑shuffling refinement is run
  for many iterations.
* When possible the current best solution is up‑sampled/down‑sampled to serve
  as a warm‑start for a larger length.

The function respects the global 1000 s wall‑time budget, prints progress,
and finally returns the best sequence found (as a plain Python list).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT‑based linear convolution and cross‑correlation)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def smooth_max_and_weights(conv: np.ndarray, tau: float):
    """
    Temperature‑smoothed max of `conv` and the corresponding softmax weights.
    Returns (smooth_max, weights) where weights sum to 1 and
    ∂smooth_max/∂conv = weights.
    """
    cmax = conv.max()
    shifted = (conv - cmax) / tau          # stability shift
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    smooth = cmax + tau * np.log(sum_exp)
    weights = exp_shifted / sum_exp
    return smooth, weights


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list:
    """
    Time‑bounded optimisation that returns the best non‑negative sequence
    found so far.  The algorithm follows the strategy described in the
    module docstring.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None          # best probability vector (sum = 1)
    best_val = float("inf")  # best objective (2·n·max(conv))

    # ------------------------------------------------------------------
    # Problem sizes – larger n usually allows a smaller bound.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT lengths (padded to next power of two for speed)
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))   # ≥ (2n‑1)+(n‑1)

        # Number of restarts and iteration budget – give larger problems
        # more work because they have a larger potential improvement.
        if n <= 1024:
            restarts = 18
            max_iters = 180_000
        elif n <= 2048:
            restarts = 12
            max_iters = 200_000
        elif n <= 4096:
            restarts = 8
            max_iters = 150_000
        else:  # n == 8192
            restarts = 5
            max_iters = 120_000

        for r in range(restarts):
            if time.time() > deadline:
                break

            # --------------------------------------------------------------
            # Initialise logits (log‑softmax parameters) – several strategies
            # --------------------------------------------------------------
            if best_seq is not None and best_seq.shape[0] == n:
                # Warm start from the current best of the same length
                a0 = np.maximum(best_seq, 0.0)
                a0_sum = a0.sum()
                if a0_sum == 0.0:
                    a0 = np.random.dirichlet(np.ones(n))
                else:
                    a0 = a0 / a0_sum
                x = np.log(a0 + 1e-15)
                init_tag = "warm"
            elif 'height_sequence_1' in globals():
                # Upsample / down‑sample the historic construction
                base = np.array(height_sequence_1, dtype=np.float64)
                if base.shape[0] != n:
                    # Linear interpolation to the required length
                    src_idx = np.linspace(0, base.shape[0] - 1, base.shape[0])
                    dst_idx = np.linspace(0, base.shape[0] - 1, n)
                    base = np.interp(dst_idx, src_idx, base)
                base = np.maximum(base, 0.0)
                if base.sum() < 0.01:
                    a0 = np.random.dirichlet(np.ones(n))
                else:
                    a0 = base / base.sum()
                x = np.log(a0 + 1e-15)
                init_tag = "height_seq"
            else:
                # Purely random start
                a0 = np.random.dirichlet(np.ones(n))
                x = np.log(a0 + 1e-15)
                init_tag = "random"

            print(f"[n={n} restart={r}] init: {init_tag}")

            # --------------------------------------------------------------
            # Adam state
            # --------------------------------------------------------------
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.15                     # initial learning rate
            decay_interval = 4000        # learning‑rate decay schedule

            # Temperature schedule (exponential annealing)
            tau_initial = 0.20
            tau_final = 1e-7

            t = 0
            last_improve = time.time()

            # --------------------------------------------------------------
            # Main Adam loop (smooth‑max)
            # --------------------------------------------------------------
            while time.time() < deadline and t < max_iters:
                # soft‑max → probability vector (sum = 1, all ≥0)
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()

                # convolution of a with itself
                conv = full_convolution(a, fft_len=fft_len_conv)

                # annealed temperature
                tau = max(tau_final,
                          tau_initial * (tau_final / tau_initial) ** (t / max_iters))
                smooth, weights = smooth_max_and_weights(conv, tau)

                # exact objective (for bookkeeping) – sum(a)=1 ⇒ F = 2·n·max(conv)
                cur_val = 2.0 * n * conv.max()

                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t:06d}] new best {cur_val:.6f}")

                # gradient of smooth‑max w.r.t. a
                grad_a = 2.0 * cross_correlation(weights, a, fft_len=fft_len_corr)

                # back‑propagation through the soft‑max
                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                # Adam update
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # learning‑rate decay
                if t % decay_interval == 0:
                    lr *= 0.85

                # stagnation handling – small random jitter in log‑space
                if time.time() - last_improve > 35.0:
                    x += (np.random.rand(n) - 0.5) * 5e-3
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t}] shake (stagnation)")

            # --------------------------------------------------------------
            # Fine‑tuning with the true (hard) max – very low temperature
            # --------------------------------------------------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a = best_seq.copy()
                x = np.log(a + 1e-15)
                m = np.zeros(n)
                v = np.zeros(n)
                lr = 0.02
                fine_iters = 20_000
                for ft in range(fine_iters):
                    if time.time() > deadline:
                        break
                    # recompute a from logits (still a probability vector)
                    max_x = np.max(x)
                    exp_x = np.exp(x - max_x)
                    a = exp_x / exp_x.sum()

                    conv = full_convolution(a, fft_len=fft_len_conv)
                    cur_val = 2.0 * n * conv.max()
                    if cur_val < best_val - 1e-12:
                        best_val = cur_val
                        best_seq = a.copy()
                        print(f"[fine n={n} iter={ft}] improved to {cur_val:.6f}")

                    # hard‑max gradient: indicator at the maximal shift
                    k_star = np.argmax(conv)
                    w = np.zeros_like(conv)
                    w[k_star] = 1.0
                    grad_a = 2.0 * cross_correlation(w, a, fft_len=fft_len_corr)

                    inner = np.dot(a, grad_a)
                    grad_x = a * (grad_a - inner)

                    # Adam step
                    m = beta1 * m + (1.0 - beta1) * grad_x
                    v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                    m_hat = m / (1.0 - beta1 ** (ft + 1))
                    v_hat = v / (1.0 - beta2 ** (ft + 1))
                    x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

            # --------------------------------------------------------------
            # Cheap greedy refinement – mass shuffling
            # --------------------------------------------------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a_local = best_seq.copy()
                local_iters = 5_000
                for it in range(local_iters):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    # move a random fraction (up to 20 %) of the source mass
                    frac = np.random.rand() * 0.2
                    eps_move = frac * a_local[i]
                    if eps_move <= 0.0:
                        continue
                    a_new = a_local.copy()
                    a_new[i] -= eps_move
                    a_new[j] += eps_move
                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = 2.0 * n * conv_new.max()
                    if val_new < best_val - 1e-12:
                        best_val = val_new
                        best_seq = a_new.copy()
                        a_local = a_new
                        print(f"[local n={n} iter={it}] improved to {val_new:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – uniform vector (objective = 2) – should never be needed.
    # ----------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = 2.0

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
