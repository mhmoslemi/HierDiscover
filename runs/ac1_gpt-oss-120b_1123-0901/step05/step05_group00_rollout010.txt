        print(f"\n=== Optimising length {n} ===")
        conv_fft_len = 1 << int(np.ceil(np.log2(2*n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3*n - 2)))

        # Build initial candidates:
        candidates = []

        # uniform

        # if best_seq of same length: add
        if best_seq is not None and len(best_seq)==n:

        # upsample previous best (if any)
        if best_seq is not None and len(best_seq) < n:

            x_old = np.linspace(0.0, 1.0, old.shape[0])

            up = np.interp(x_new, x_old, old)
            up = np.maximum(up, 0.0)
            if up.sum()>0:
                up = up / up.sum()
            else:
                up = np.random.dirichlet(np.ones(n))
            candidates.append(up)

        # parameter families add random param shapes
        # Determine number of random param samples based on time left and n
        remaining = deadline - time.time()
        # approximate cost per param = O(n log n) ~ maybe 0.0005s for n=8192, 0.001 for 16384
        # compute allowed samples:
        cost_per = 0.0007 * n / 8192 # scale

        # but limit for huge n:
        max_samples = min(max_samples, 2000)  # cap

            # ensure nonneg, normalize

            s = shape.sum()
            if s == 0:

            candidates.append(shape)

        # power-law exponents grid:

            idx = np.arange(1, n+1, dtype=np.float64)

                v_hat = v / (1.0 - beta2 ** it)

                # anneal lam
                lam = min(lam * lam_factor, 1e7)

                # compute objective (normalized sum=1)
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Adam it={it}] new best {val:.6f}")

                # stagnation:
                if time.time() - last_improve > 5.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # Sub-gradient refinement
            a = a / a.sum()
            eta = 0.6
            sub_last = time.time()
            for sub_it in range(2000):
                conv = np.fft.irfft(np.fft.rfft(a, conv_fft_len) ** 2, conv_fft_len)[:2*n-1]
                max_c = conv.max()
                # active set mask
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)
                grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
                a = a * np.exp(-eta * grad_a)
                s = a.sum()

                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                # evaluate objective
                conv = np.fft.irfft(np.fft.rfft(a, conv_fft_len) ** 2, conv_fft_len)[:2*n-1]
                max_c = conv.max()
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad it={sub_it}] new best {val:.6f}")

                if time.time() - sub_last > 5.0:
                    break
                if time.time() > deadline:
                    break

    # If unable to find any (unlikely)
    if best_seq is None:
        best_seq = np.full(2048, 1.0/2048)
        best_val = compute_obj(best_seq, 2048)

    print(f"\nFinal best value: {best_val:.6f} with length {len(best_seq)}")
    return best_seq.tolist()

"""
Propose a high‑quality step‑function (non‑negative vector) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

The algorithm works on a cascade of sizes (1024 … 16384).  For every size it
generates a large and diverse pool of start vectors:

    * uniform vector,
    * the current best solution (and an up‑sampled version of a shorter best),
    * random Dirichlet draws,
    * several analytical families (power‑law, two‑Gaussian edges,
      sine‑squared bump) and, most importantly,
    * *random mixtures of two symmetric Beta distributions* – a flexible
      family that can produce both edge‑heavy and centre‑heavy shapes.

All candidates are evaluated quickly (exact max of the linear convolution)
and the best 20 are refined by a two‑phase optimisation:

    1. Adam on a smooth‑max surrogate (log‑sum‑exp) with a slowly
       annealed temperature λ; the vector is kept in the probability simplex
       via a soft‑max parametrisation.
    2. A few thousand steps of an exponentiated‑gradient sub‑gradient
       method on the true max‑convolution.

Time‑budget checks guarantee termination before the 1000 s limit.
The best sequence found (as a plain Python list) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are defined at module level (no closures)
# ----------------------------------------------------------------------


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a using an FFT of length ``fft_len`` (power of two).
    Returns the full convolution of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


def _beta_mixture(n: int, w: float, a: float, b: float) -> np.ndarray:
    """
    Symmetric Beta‑mixture on a regular grid of size ``n``.
    The density is   w·Beta(a,a) + (1‑w)·Beta(b,b)   (both symmetric).
    Returned vector is normalised to sum = 1.
    """
    idx = np.arange(n, dtype=np.float64)
    x = (idx + 0.5) / n                     # points in (0,1)
    # Beta density (ignoring normalising constant – we normalise later)
    shape_a = np.power(x, a - 1.0) * np.power(1.0 - x, a - 1.0)
    shape_b = np.power(x, b - 1.0) * np.power(1.0 - x, b - 1.0)
    shape = w * shape_a + (1.0 - w) * shape_b
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s == 0.0:
        # fall back to uniform if something went wrong
        return np.full(n, 1.0 / n, dtype=np.float64)
    return shape / s


def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """
    Exact objective for a normalised vector ``a`` (sum = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c)

# Main entry point required by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the wall‑clock budget and returns the best
    sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Process a cascade of lengths – small sizes are cheap warm‑starts.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        # FFT lengths (next power of two)
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))   # (2n‑1)+(n‑1)

        # --------------------------------------------------------------
        # Build a diverse pool of initial candidates
        # --------------------------------------------------------------
        candidates = []

        # 1) uniform vector
        candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # 2) warm‑start from previous best of exactly the same size
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.asarray(best_seq, dtype=np.float64))

        # 3) up‑sample a shorter best (linear interpolation)
        if best_seq is not None and len(best_seq) < n:
            old = np.asarray(best_seq, dtype=np.float64)
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, old)
            up = np.maximum(up, 0.0)
            if up.sum() > 0.0:
                up = up / up.sum()
            else:
                up = np.random.dirichlet(np.ones(n))
            candidates.append(up)

        # 4) historic global variable, if present
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            h = np.asarray(hseq, dtype=np.float64)
            if h.shape[0] > n:
                h = h[:n]
            elif h.shape[0] < n:
                h = np.pad(h, (0, n - h.shape[0]))
            h = np.maximum(h, 0.0)
            s = h.sum()
            if s > 0.0:
                h = h / s
            else:
                h = np.random.dirichlet(np.ones(n))
            candidates.append(h)

        # 5) a few random Dirichlet draws
        for _ in range(3):
            candidates.append(np.random.dirichlet(np.ones(n)))

        # 6) power‑law edge‑concentrated family (i·(n‑i+1))^{α}
        idx = np.arange(1, n + 1, dtype=np.float64)
        prod = idx * (n - idx + 1.0)
        for alpha in np.linspace(-2.0, -0.5, 4):
            shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
            if shape.sum() > 0.0:
                shape = shape / shape.sum()
                candidates.append(shape)

        # 7) two‑Gaussian edge shapes
        i = np.arange(n, dtype=np.float64)
        for gamma in (0.2, 0.5, 1.0, 2.0):
            shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            shape = shape / shape.sum()
            candidates.append(shape)

        # 8) smooth sine‑squared bump
        shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
        shape = shape / shape.sum()
        candidates.append(shape)

        # --------------------------------------------------------------
        # Random mixtures of two symmetric Beta distributions
        # --------------------------------------------------------------
        # The number of random mixtures is limited by the remaining time.
        # A rough estimate for the cost of one evaluation (FFT) is
        #   ≈ 0.5 µs·n   (very small for n≤16384)
        # We therefore allocate a generous budget for these cheap candidates.
        remaining = deadline - time.time()
        # Cost per evaluation grows roughly linearly with n; we use a factor.
        cost_est = 1e-6 * n   # seconds per evaluation (very liberal)
        max_mix = max(5, int(remaining // (cost_est * 2)))
        max_mix = min(max_mix, 2000)      # hard cap

        for _ in range(max_mix):
            w = np.random.rand()
            a_alpha = np.random.uniform(0.05, 2.0)
            b_alpha = np.random.uniform(0.05, 2.0)
            shape = _beta_mixture(n, w, a_alpha, b_alpha)
            candidates.append(shape)

        # --------------------------------------------------------------
        # Quick evaluation of all candidates – keep the best few
        # --------------------------------------------------------------
        evaluated = []
        for a in candidates:
            val = _objective(a, conv_fft_len)
            evaluated.append((val, a))
            if val < best_val:
                best_val = val
                best_seq = a.copy()
        evaluated.sort(key=lambda x: x[0])
        top_k = 20
        top_candidates = [a for _, a in evaluated[:top_k]]
        print(f"Pre‑screen best for n={n}: {best_val:.6f}")

        # --------------------------------------------------------------
        # Optimise every top candidate (Adam → sub‑gradient)
        # --------------------------------------------------------------
        for cand_id, a0 in enumerate(top_candidates):
            if time.time() > deadline:
                break

            # initialise logits (soft‑max parametrisation)
            eps = 1e-16
            a = np.maximum(a0, eps)
            a = a / a.sum()
            logits = np.log(a)

            # Adam hyper‑parameters
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2 = 0.9, 0.999
            eps_adam = 1e-8
            lr = 0.05

            lam = 0.5                # temperature for smooth‑max
            lam_factor = 1.01
            last_improve = time.time()
            # iteration budget – roughly 30 k·1024 / n + 4 k
            max_iters = int(30000 * 1024 / n) + 4000

            for it in range(1, max_iters + 1):
                # soft‑max → probability vector on the simplex
                mx = logits.max()
                exp_log = np.exp(logits - mx)
                a = exp_log / exp_log.sum()

                # convolution of a with itself
                conv = _full_convolution(a, conv_fft_len)
                max_c = conv.max()

                # smooth‑max weights (log‑sum‑exp surrogate)
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # gradient of the surrogate w.r.t. a
                grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

                # back‑propagation through the soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update on logits
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                # anneal temperature
                lam = min(lam * lam_factor, 1e7)

                # exact objective (sum(a)=1 ⇒ denominator = 1)
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Adam it={it:05d}] new best {val:.6f}")

                # stagnation handling (tiny random shake)
                if time.time() - last_improve > 5.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # ----------------------------------------------------------
            # Sub‑gradient refinement on the true max (exact objective)
            # ----------------------------------------------------------
            a = a / a.sum()
            eta = 0.6
            sub_last = time.time()
            for sub_it in range(2000):
                conv = _full_convolution(a, conv_fft_len)
                max_c = conv.max()
                # active set (indices where the max is attained)
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                # evaluate exact objective
                conv = _full_convolution(a, conv_fft_len)
                max_c = conv.max()
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad it={sub_it}] new best {val:.6f}")

                if time.time() - sub_last > 5.0:
                    break
                if time.time() > deadline:
                    break

    # ------------------------------------------------------------------
    # Fallback – in the extremely unlikely case that nothing was found.
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = _objective(best_seq, 1 << int(np.ceil(np.log2(2 * fallback_n - 1))))

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
