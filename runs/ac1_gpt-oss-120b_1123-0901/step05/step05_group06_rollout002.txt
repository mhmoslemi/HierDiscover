def build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        start = max(0, k-(n-1))
        end = min(k, n-1)
        # j indices: start...end
        # coefficient for b_j = a_{k-j}
        # For each j, coeff = a[k-j]
        # So ramp over j
        for j in range(start, end+1):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    return csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

"""
Hybrid optimisation for the auto‑convolution L∞ problem.

The algorithm combines three ideas:
1. A fast Adam optimiser on the soft‑max approximation of the max
   convolution (the same technique that gave the previous 1.5098 bound).
2. A linear‑programming “partner” step (as described in the problem
   statement) that finds a sequence g with maximal ℓ₁‑norm while keeping
   f∗g ≤ max(f∗f).  A convex combination of f and g strictly reduces the
   max‑convolution after only a few iterations.
3. A cheap stochastic mass‑transfer refinement that fine‑tunes the final
   solution.

The search explores a few different lengths (the historic 2100 and a few
scaled sizes) while respecting the overall 1000 s budget.  The best
sequence found is returned as a plain Python list.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper utilities ------------------------------------------------------


def _max_convolution(a: np.ndarray) -> float:
    """Return max_k (a∗a)[k] using an FFT (O(n log n))."""
    n = a.shape[0]
    Nfft = 1 << ((2 * n - 1).bit_length())
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
    return float(conv.max())


def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k * a_{k-i} (out‑of‑range ignored) using FFTs.
    This is the gradient of the (soft‑max) smoothed max‑convolution.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]    # linear part
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Create a small, diverse set of normalised start vectors."""
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0
             - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    init.append(black / black.sum() * target_sum)

    # 7) small random perturbation of uniform
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    # 8) triangular (peak in centre)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    init.append(tri / tri.sum() * target_sum)

    return init


def _exp_gradient_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑its (exponentiated‑gradient) using a soft‑max
    approximation of the max‑convolution.  Returns the best vector seen in
    this run together with its exact max‑convolution.
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise logits from the start vector
    eps = 1e-16
    y = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.08
    jitter_interval = 2500

    tau_factor = 0.30
    decay_rate = max_iters / 8.0
    Nfft = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current probability simplex via softmax
        logits = y - np.max(y)
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()
        a = target_sum * p

        # exact self‑convolution
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # soft‑max weighting
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # gradient via weighted reverse convolution
        rev = _weighted_rev(a, w, Nfft)
        g_a = 2.0 * rev

        # chain rule into logits (mirror‑descent on simplex)
        avg = np.dot(p, g_a)
        g_y = target_sum * p * (g_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # occasional jitter to escape plateaux
        if it % jitter_interval == 0:
            y += rng.normal(scale=0.01, size=n)

        # keep the best ever
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:8d}] new best M = {best_score:.6f}")

    return best_seq, best_score


def _build_conv_matrix(f: np.ndarray):
    """
    Build a sparse matrix A of shape (2n‑1, n) such that A @ b = f∗b.
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        for j in range(start, end + 1):
            rows.append(k)
            cols.append(j)
            data.append(f[k - j])
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_step(f: np.ndarray, M: float, target_sum: float):
    """
    Solve the linear programme:
        maximise   sum(b)
        s.t.       b ≥ 0,
                   f ∗ b ≤ M  (element‑wise),
    then return the normalised partner g (sum = target_sum).
    Returns None on failure.
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)
    ub = np.full(2 * n - 1, M, dtype=float)
    c = -np.ones(n)                 # minimise –sum(b)  → maximise sum(b)
    bounds = [(0, None)] * n

    res = opt.linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    if not res.success:
        return None
    b = res.x
    if b.sum() < 1e-12:
        return None
    g = target_sum * b / b.sum()
    return g


def _lp_refine(
    f: np.ndarray,
    target_sum: float,
    max_iters: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly compute a partner g via LP and replace f by a convex
    combination (1‑t)·f + t·g that lowers max(f∗f).  Stops after
    max_iters or when no improvement is observed.
    """
    best_f = f.copy()
    best_M = _max_convolution(best_f)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        M = best_M
        g = _lp_step(best_f, M, target_sum)
        if g is None:
            break

        # try several mixing factors and keep the best improvement
        candidate_f = best_f
        candidate_M = best_M
        for t in (0.5, 0.3, 0.2, 0.1, 0.05, 0.02):
            cand = (1.0 - t) * best_f + t * g
            cand_M = _max_convolution(cand)
            if cand_M < candidate_M - 1e-12:
                candidate_f = cand
                candidate_M = cand_M

        if candidate_M < best_M - 1e-12:
            best_f = candidate_f
            best_M = candidate_M
            print(f"[LP iter {it}] improved M → {best_M:.6f}")
        else:
            # no further improvement – exit early
            break

    return best_f, best_M


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing on the simplex with a simple simulated‑annealing
    acceptance rule.
    """
    cur_seq = a.copy()
    cur_score = best_score
    Nfft = 1 << ((2 * n - 1).bit_length())
    max_steps = 200_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue
        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        conv = np.fft.irfft(np.fft.rfft(new_seq, n=Nfft) ** 2, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())
        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1
    return cur_seq, cur_score
