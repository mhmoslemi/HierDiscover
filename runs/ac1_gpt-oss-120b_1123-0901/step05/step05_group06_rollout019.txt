def project_onto_simplex(v, s):
    # ensure nonnegative simplex sum s
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v, s):
    # Project onto simplex {x >=0, sum x = s}
    # Input v: 1D array
    n = v.shape[0]
    # Sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n+1)
    rho = np.where(rho_arr > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def gradient_descent_optimize(a0, n, target_sum, deadline, max_iters, rng):
    a = a0.copy()
    # ensure sum
    a = a / a.sum() * target_sum
    Nfft = 1 << ((2*n - 1).bit_length())
    best_seq = a.copy()
    best_score = max_convolution(a)
    # We'll compute M each iteration anyway.
    # schedule: lr0 = 0.5 (or maybe 0.4)
    lr0 = 0.4
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # Compute convolution
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[:2*n-1]
        M = float(conv.max())
        if M < best_score:
            best_score = M
            best_seq = a.copy()
        # subgradient
        mask = conv >= M - 1e-12
        cnt = mask.sum()
        w = mask.astype(float) / max(cnt, 1)
        rev = weighted_rev(a, w, Nfft)
        grad = 2.0 * rev
        # step size
        lr = lr0 * (1.0 - it / max_iters)
        a = a - lr * grad
        # project onto simplex
        a = simplex_projection(a, target_sum)
        # optional jitter every some steps
        if it % 5000 == 0 and it > 0:
            # small random perturbation
            delta = rng.normal(scale=0.001 * target_sum / n, size=n)
            a = a + delta
            a = simplex_projection(a, target_sum)
    return best_seq, best_score

"""
A fast projected‑subgradient optimiser for the step‑function problem.

The objective is   f(a) = max (a * a)   (the maximal entry of the
linear convolution of a with itself).  The evaluation routine
``evaluate_sequence`` returns exactly this value after normalising
the vector to Σa = √(2 n).  The optimiser works directly on the
constrained simplex:

* a_i ≥ 0
* Σ a_i = √(2 n)

The method:
  1. Build several diverse starting vectors (uniform, Dirichlet,
     sinusoidal, interpolated previous best, etc.).
  2. For each start run a projected sub‑gradient descent on the
     exact max‑convolution.  The sub‑gradient is obtained by a
     weighted reverse convolution (FFT‑based, O(n log n)).
  3. After each iteration the point is projected onto the
     probability simplex with the required total mass.
  4. Keep the overall best vector seen.
  5. Finish with a cheap stochastic pairwise‐mass‑transfer
     refinement (simulated‑annealing) to squeeze out the last bits.

The whole search respects the 1000 s budget; the main loop stops as
soon as the deadline is reached.  The algorithm differs from the
previous ADAM/soft‑max approach by using the true sub‑gradient of
the non‑smooth objective and a direct projection onto the feasible
set, which often finds a lower value in the same amount of time.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top‑level, no closures
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex
        {x | x_i ≥ 0, Σ x_i = s } .
    Returned array has the same shape as v.
    """
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # find the largest rho where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k * a_{k-i}
    (out‑of‑range indices are ignored) using FFTs.
    Length of w is 2·len(a)‑1, i.e. the size of a*a.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)      # reverse w before FFT
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # keep the linear part of the convolution (length = 3 n‑2)
    n = a.shape[0]
    conv_rev = conv_rev[: 3 * n - 2]
    idx_start = 2 * n - 2
    rev = conv_rev[idx_start - np.arange(n)]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Return max_k (a∗a)[k] using an FFT of length Nfft."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climbing on the simplex:
    repeatedly move a small fraction of mass from a random index i
    to j.  Improvements are always accepted; deteriorations are
    accepted with a (linear) simulated‑annealing schedule.
    """
    target_sum = a.sum()
    cur_seq = a.copy()
    cur_score = best_score
    Nfft = 1 << ((2 * n - 1).bit_length())

    max_steps = 200_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        # move a random 1 %‑5 % of the source mass
        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # recompute max‑convolution (FFT based)
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1
    return cur_seq, cur_score


def _make_initial_vectors(
    n: int, target_sum: float, rng: np.random.Generator, base: np.ndarray | None
) -> list[np.ndarray]:
    """
    Create a modestly diverse pool of start vectors for a given length.
    """
    init = []

    # 1) Uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (γ(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) Sinusoidal half‑wave
    xs = np.linspace(0.0, np.pi, n)
    s = np.sin(xs)
    init.append(s / s.sum() * target_sum)

    # 4) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 5) Random Gaussian mixture (2‑5 components)
    for _ in range(3):
        k = rng.integers(2, 6)
        comp = np.zeros(n)
        for __ in range(k):
            loc = rng.uniform(0, n)
            sigma = rng.uniform(0.04 * n, 0.15 * n)
            idx = np.arange(n)
            comp += np.exp(-0.5 * ((idx - loc) / sigma) ** 2)
        if comp.sum() > 0:
            init.append(comp / comp.sum() * target_sum)

    # 6) Interpolated previous best (if any) and length differs
    if base is not None and base.shape[0] != n:
        old_x = np.linspace(0.0, 1.0, base.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        interp = np.interp(new_x, old_x, base)
        interp = np.maximum(interp, 0.0)
        interp = interp / interp.sum() * target_sum
        init.append(interp)

    return init


def _subgradient_optimize(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Returns the best vector seen and its exact max‑convolution value.
    """
    n = a0.shape[0]
    a = a0.copy()
    a = a / a.sum() * target_sum          # ensure feasibility

    # FFT size must accommodate the 3 n‑2 linear convolution used in
    # the weighted reverse step.
    Nfft = 1 << ((3 * n - 2).bit_length())

    best_seq = a.copy()
    best_score = _max_convolution(a, Nfft)

    # step‑size schedule: α / sqrt(t)
    alpha = 0.6

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- current max convolution and its argmax mask ----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

        # Uniform weight over all indices attaining the max (sub‑gradient)
        mask = conv >= M - 1e-12
        w = mask.astype(float) / max(mask.sum(), 1)

        # ---- sub‑gradient of the max‑convolution ----
        rev = _weighted_rev(a, w, Nfft)
        grad = 2.0 * rev                     # ∂ max(a∗a) / ∂a

        # ---- projected gradient step ----
        lr = alpha / math.sqrt(it)
        a = a - lr * grad
        a = _simplex_projection(a, target_sum)

        # ---- occasional jitter to escape plateaux ----
        if it % 3000 == 0:
            jitter = rng.normal(scale=0.001 * target_sum / n, size=n)
            a = a + jitter
            a = _simplex_projection(a, target_sum)

    return best_seq, best_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence (plain Python list)
    found within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a small safety margin

    # ------------------------------------------------------------------
    # Try to load the previously discovered construction (if any)
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        base = None

    # ------------------------------------------------------------------
    # Choose a single length to focus the effort.
    # The historic best used n=2100; we keep that as the default.
    if base is not None:
        n0 = base.shape[0]
    else:
        n0 = 2100

    # Upper bound for the search (cannot go beyond 8000 as per problem statement)
    max_n = min(8000, int(1.5 * n0))
    n = max_n

    target_sum = math.sqrt(2.0 * n)

    print(f"[setup] using length = {n}, target_sum = {target_sum:.3f}")

    # ------------------------------------------------------------------
    # Build a modest set of diverse starts
    init_vectors = _make_initial_vectors(n, target_sum, rng, base)

    # Global best trackers
    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Run sub‑gradient optimisation from each start.  The iteration
    # budget per start is generous; the loop will break as soon as the
    # global deadline is reached.
    for idx, a0 in enumerate(init_vectors):
        if time.time() >= deadline:
            break
        print(f"[run] start {idx+1}/{len(init_vectors)} for length {n}")

        # Very generous iteration limit – actual loop may stop early.
        max_iters = 2_500_000

        seq, score = _subgradient_optimize(
            a0=a0,
            target_sum=target_sum,
            deadline=deadline,
            max_iters=max_iters,
            rng=rng,
        )

        if score < best_score:
            best_score = score
            best_seq = seq

    # ------------------------------------------------------------------
    # Final cheap stochastic refinement (if time permits)
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        refine_deadline = time.time() + min(remaining * 0.5, 20.0)
        best_seq, best_score = _local_refine(
            a=best_seq,
            n=best_seq.shape[0],
            deadline=refine_deadline,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Fallback (very unlikely to be taken)
    if best_seq is None:
        print("[fallback] using uniform sequence")
        best_seq = np.full(n, target_sum / n)

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
