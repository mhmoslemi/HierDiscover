def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    best_val = float('inf')
    best_seq = None

    lengths = [4096, 8192, 16384]  # produce fine increments

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"Optimizing length {n}")
        conv_fft_len = 1 << int(np.ceil(np.log2(2*n-1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3*n-2)))
        candidates = build_candidates(n, best_seq)

        # Determine iteration budgets based on n
        max_iters_adam = int(20000 * 4096 / n) + 2000   # roughly constant overall cost
        max_iters_sub = int(5000 * 4096 / n) + 2000

        for idx, init in enumerate(candidates):
            if time.time() > deadline:
                break
            a = np.maximum(init, 1e-12)
            a = a / a.sum()
            logits = np.log(a)

            # Adam
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2 = 0.9, 0.999
            lr = 0.03
            lam = 0.5
            lam_factor = 1.005   # slower increase
            last_update = time.time()

            for it in range(max_iters_adam):
                # compute a from logits via softmax
                mx = logits.max()
                exp_log = np.exp(logits - mx)
                a = exp_log / exp_log.sum()

                conv = _full_convolution(a, conv_fft_len)
                max_c = conv.max()
                # smoothmax weights
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

                # Back-propagate through softmax
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                m = beta1*m + (1-beta1)*grad_log
                v = beta2*v + (1-beta2)*(grad_log**2)
                m_hat = m / (1-beta1**(it+1))
                v_hat = v / (1-beta2**(it+1))
                logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

                lam = min(lam * lam_factor, 1e7)

                val = 2.0 * n * max_c   # denominator = 1

                if val < best_val - 1e-8:
                    best_val = val
                    best_seq = a.copy()
                    last_update = time.time()
                    print(f"[n={n} cand={idx} Adam it={it}] new best {val:.6f}")

                # stagnation
                if time.time() - last_update > 5.0:
                    logits += (np.random.rand(n)-0.5)*1e-3
                    last_update = time.time()

                if time.time() > deadline:
                    break

            # Subgradient refinement
            a = a / a.sum()
            eta = 0.6
            last_sub = time.time()
            sub_iters = max_iters_sub
            for sub_it in range(sub_iters):
                conv = _full_convolution(a, conv_fft_len)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)
                grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
                a = a * np.exp(-eta * grad_a)
                # maintain positivity, no zeros
                s = a.sum()
                if s < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                val = 2.0 * n * max_c
                if val < best_val - 1e-8:
                    best_val = val
                    best_seq = a.copy()
                    last_sub = time.time()
                    print(f"[n={n} cand={idx} Subgrad it={sub_it}] new best {val:.6f}")

                if time.time() - last_sub > 5.0:
                    break
                if time.time() > deadline:
                    break

    # Final fine-tuning via hill climbing on the best (largest length)
    if best_seq is not None and len(best_seq) == lengths[-1]:
        n = len(best_seq)
        conv_fft_len = 1 << int(np.ceil(np.log2(2*n-1)))
        a = best_seq.copy()
        best_val_local = best_val
        # hill climbing steps
        step = max(1e-4, 0.001)  # relative mass moved each step
        while time.time() < deadline:
            # pick two distinct indices
            i, j = np.random.choice(n, 2, replace=False)
            if a[i] < step:
                continue
            a_new = a.copy()
            a_new[i] -= step
            a_new[j] += step
            # renormalize (but sum is unchanged)
            conv = _full_convolution(a_new, conv_fft_len)
            val = 2.0 * n * conv.max()
            if val < best_val_local:
                best_val_local = val
                a = a_new
                best_seq = a_new.copy()
                best_val = val
                print(f"[HillClimb] improved to {val:.6f}")
            # optionally adapt step size
        # after loop, best_seq updated

    # fallback uniform if none
    if best_seq is None:
        n = 2048
        best_seq = np.full(n, 1.0/n)

    return best_seq.tolist()

"""
propose_candidate
-----------------

Search for a non‑negative step function (a probability vector) that
minimises

    F(a) = 2·n·max(a*a) / (sum(a)^2)

The search is performed in three stages:

1. **Diverse initial families** – for each target length we build a pool
   of candidates from simple analytic families (uniform, up‑sampled
   previous best, power‑law edges, Beta distributions, mixtures,
   two‑Gaussian edges, sine‑squared bump) and random Dirichlet draws.

2. **Smooth‑max Adam optimisation** – each candidate is optimised on the
   log‑sum‑exp surrogate of the ℓ∞‑norm.  The iterate is kept on the
   probability simplex by a soft‑max over logits.  A slowly increasing
   temperature (log‑sum‑exp “λ”) yields a smooth approximation early on
   and a tight max later.  Adam’s adaptive learning‑rate speeds up
   convergence.

3. **Exact sub‑gradient refinement** – a few exponentiated‑gradient
   steps on the true max convolution are performed to clean up the
   solution.

4. **Fine‑tuning** – the best vector for the largest length is further
   refined with a high‑temperature Adam run.

The routine respects a wall‑clock budget (default 1000 s) and returns
the best sequence found as a plain Python list.
"""

import time
import numpy as np


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using an FFT of length a power of two."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.

    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) pdf on n points (mid‑point rule)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid zeros when α or β < 1
    eps = 1e-16
    x = np.clip(x, eps, 1 - eps)
    shape = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    if shape.sum() == 0:
        shape = np.random.dirichlet(np.ones(n))
    else:
        shape = shape / shape.sum()
    return shape


def _build_candidates(n: int, global_best: np.ndarray) -> list:
    """
    Construct a heterogeneous list of initial probability vectors of length n.
    """
    candidates = []

    # 1. uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2. previous best of same length
    if global_best is not None and global_best.shape[0] == n:
        candidates.append(global_best.copy())

    # 3. up‑sample a shorter best (linear interpolation)
    if global_best is not None and global_best.shape[0] < n:
        old = global_best
        x_old = np.linspace(0.0, 1.0, old.shape[0])
        x_new = np.linspace(0.0, 1.0, n)
        up = np.interp(x_new, x_old, old)
        if up.sum() > 0:
            up = up / up.sum()
        else:
            up = np.random.dirichlet(np.ones(n))
        candidates.append(up)

    # 4. random Dirichlet
    candidates.append(np.random.dirichlet(np.ones(n)))

    # 5. edge‑concentrated power‑law families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-0.5, -4.0, 6):
        shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            candidates.append(shape)

    # 6. Beta family (symmetric & asymmetric)
    beta_params = [0.2, 0.4, 0.8, 1.2, 2.0]
    for a in beta_params:
        for b in beta_params:
            shape = _beta_shape(n, a, b)
            candidates.append(shape)

    # 7. mixtures of edge‑shape and uniform
    edge = np.where(prod > 0, np.power(prod, -2.0), 0.0)
    edge = edge / edge.sum()
    uniform = np.full(n, 1.0 / n)
    for p in [0.3, 0.5, 0.7]:
        mix = p * edge + (1.0 - p) * uniform
        mix = mix / mix.sum()
        candidates.append(mix)

    # 8. two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in [0.2, 0.5, 1.0, 2.0]:
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape = shape / shape.sum()
        candidates.append(shape)

    # 9. smooth sine‑squared bump
    bump = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    bump = bump / bump.sum()
    candidates.append(bump)

    # 10. a handful of extra random Dirichlet draws for extra diversity
    for _ in range(2):
        candidates.append(np.random.dirichlet(np.ones(n)))

    return candidates


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    objective.  The function respects the given time budget and returns the
    best sequence encountered as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin for the outer harness

    best_seq = None                # numpy array (probability vector)
    best_val = float("inf")

    # Process lengths from coarse to fine – a coarse solution can be
    # up‑sampled for the next stage.
    lengths = [4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

        # Build initial candidates (list of numpy arrays)
        candidates = _build_candidates(n, best_seq)

        # Iteration budgets (more for small n, less for large n)
        if n <= 2048:
            max_iters_adam = 12000
            max_iters_sub = 6000
        elif n <= 4096:
            max_iters_adam = 8000
            max_iters_sub = 4000
        elif n <= 8192:
            max_iters_adam = 5000
            max_iters_sub = 2500
        else:
            max_iters_adam = 3000
            max_iters_sub = 1500

        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # initialise simplex via soft‑max logits
            a = np.maximum(a0, 1e-16)
            a = a / a.sum()
            logits = np.log(a)

            # ----- Adam on smooth‑max surrogate --------------------
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2 = 0.9, 0.999
            lr = 0.03
            lam = 0.5                      # temperature for log‑sum‑exp
            lam_factor = 1.005
            last_improve = time.time()

            for it in range(1, max_iters_adam + 1):
                # soft‑max → simplex
                mx = logits.max()
                exp_log = np.exp(logits - mx)
                a = exp_log / exp_log.sum()

                # convolution
                conv = _full_convolution(a, conv_fft_len)
                max_c = conv.max()

                # smooth‑max weights
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # gradient w.r.t. a
                grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

                # gradient through the soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)
                logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

                # anneal temperature – slowly approach true max
                lam = min(lam * lam_factor, 1e7)

                # exact objective (sum(a) = 1)
                val = 2.0 * n * max_c

                if val < best_val - 1e-9:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Adam it={it:05d}] new best {val:.6f}")

                # stagnation handling – tiny random shake
                if time.time() - last_improve > 5.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # ----- Exact sub‑gradient refinement -------------------
            a = a / a.sum()
            eta = 0.6
            sub_last = time.time()
            for sub_it in range(max_iters_sub):
                conv = _full_convolution(a, conv_fft_len)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                val = 2.0 * n * max_c
                if val < best_val - 1e-9:
                    best_val = val
                    best_seq = a.copy()
                    sub_last = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad it={sub_it}] new best {val:.6f}")

                if time.time() - sub_last > 5.0:
                    break
                if time.time() > deadline:
                    break

    # ----------------------------------------------------------------------
    # Fine‑tuning on the best vector of the largest length (if we have one)
    # ----------------------------------------------------------------------
    if best_seq is not None and len(best_seq) == lengths[-1]:
        n = len(best_seq)
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))

        a = best_seq.copy()
        logits = np.log(np.maximum(a, 1e-16))
        m = np.zeros_like(logits)
        v = np.zeros_like(logits)
        beta1, beta2 = 0.9, 0.999
        lr = 0.015
        lam = 1e6                # very high – approximates true max
        last_improve = time.time()

        for it in range(1, 2501):
            # soft‑max reconstruction
            mx = logits.max()
            exp_log = np.exp(logits - mx)
            a = exp_log / exp_log.sum()

            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()

            # smooth‑max weights (practically a single active entry)
            w = np.exp(lam * (conv - max_c))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

            # back‑prop through soft‑max
            avg = np.dot(a, grad_a)
            grad_log = a * (grad_a - avg)

            # Adam step
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            val = 2.0 * n * max_c
            if val < best_val - 1e-9:
                best_val = val
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[Fine‑tune it={it}] new best {val:.6f}")

            # occasional tiny random shake to escape flat plateaus
            if time.time() - last_improve > 4.0:
                logits += (np.random.rand(n) - 0.5) * 5e-4
                last_improve = time.time()

            if time.time() > deadline:
                break

    # Fallback – uniform vector if nothing was found (should never happen)
    # ----------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
