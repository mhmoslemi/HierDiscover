"""
Non‑negative step‑function optimizer (candidate generator).

The algorithm combines three ideas that complement the earlier
soft‑max/Adam approach:

1. **Monotone exponential start** – each restart begins from a power‑law
   shape a_i ∝ (i+1)^{-p} (p sampled randomly).  Such shapes are known to
   reduce the self‑convolution peak because large mass at the left hand
   side is paired with tiny mass on the right side.

2. **Smooth‑max Adam optimisation** – exactly as in the reference code we
   minimise a temperature‑annealed smooth approximation of
   max(conv(a,a)).  Gradients are obtained via FFT based cross‑correlation
   and Adam with learning‑rate decay.

3. **Targeted mass‑shift & sorting moves** – when progress stalls we
   identify the convolution index k where the peak occurs, damp the
   contributing coefficients and redistribute the removed mass to the
   remaining coordinates (or simply sort the vector in decreasing order).
   These cheap non‑gradient updates often escape flat regions and give a
   further drop in the objective.

The search is run for a few different lengths (1024, 2048, 4096) and a
fixed wall‑time budget (default 1000 s).  The best sequence found so far
is kept globally; warm‑starts from a previous best (or from the provided
`height_sequence_1`) are used when possible.  The function returns the
best list of floats; the external `evaluate_sequence` will be called by
the grader.

"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution utilities (linear, not circular)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def smooth_max_and_weights(conv: np.ndarray, tau: float):
    """
    Compute a temperature‑smoothed max of `conv` and the corresponding
    softmax weights (∂smooth_max/∂conv).
    """
    # shift for numerical stability
    c_max = conv.max()
    shifted = (conv - c_max) / max(tau, 1e-12)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    smooth = c_max + tau * np.log(sum_exp)
    weights = exp_shifted / sum_exp
    return smooth, weights


# ----------------------------------------------------------------------
# Helper: evaluate objective locally (identical to evaluate_sequence
# except that we do not perform the clipping that the grader does – we
# keep all entries inside [0,1000] ourselves).
# ----------------------------------------------------------------------
def local_objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) / (sum(a)**2)  (no clipping)."""
    a = np.clip(a, 0.0, 1000.0)
    n = a.shape[0]
    conv = full_convolution(a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
# Targeted mass‑shift move – reduces coefficients that actively
# participate in the current maximal convolution entry.
# ----------------------------------------------------------------------
def mass_shift_move(a: np.ndarray, conv: np.ndarray, delta: float = 0.02) -> np.ndarray:
    """
    Identify the active indices for the current convolution peak,
    shrink them by a factor (1‑delta) and redistribute the removed mass
    uniformly to the remaining coordinates.  Returns a new probability
    vector (still sums to one).
    """
    n = a.shape[0]
    kmax = int(conv.argmax())
    # active region: indices i with a partner j = kmax-i inside the array
    start = max(0, kmax - (n - 1))
    end = min(n - 1, kmax)
    active = np.arange(start, end + 1, dtype=int)

    if active.size == 0:
        return a

    a_new = a.copy()
    removed = a_new[active] * delta
    a_new[active] -= removed
    total_removed = removed.sum()

    # redistribute uniformly to the complementary set
    rest = np.setdiff1d(np.arange(n), active, assume_unique=True)
    if rest.size == 0:
        a_new += total_removed / n
    else:
        a_new[rest] += total_removed / rest.size

    # renormalise to keep sum == 1
    a_new /= a_new.sum()
    return a_new

                cand = np.array(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.clip(cand, 0.0, 1000.0)
                if cand.sum() < 0.01:
                    cand = np.random.dirichlet(np.ones(n))
                else:
                    cand /= cand.sum()
                x = np.log(cand + 1e-15)
                print(f"[n={n} restart={r}] start from height_sequence_1")
            else:
                # power‑law seed a_i ∝ (i+1)^{-p}
                p = np.random.uniform(0.3, 2.5)
                a0 = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
                a0 = np.clip(a0, 0.0, 1000.0)
                a0 /= a0.sum()
                # add a tiny uniform floor to avoid exact zeros
                a0 = 0.999 * a0 + 0.001 / n
                a0 /= a0.sum()
                x = np.log(a0 + 1e-15)
                print(f"[n={n} restart={r}] power‑law start p={p:.3f}")

            # Adam state
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)

            t = 0
            last_improve = time.time()
            lr = lr_initial

            # temperature schedule for smooth‑max
            tau_initial = 0.15
            tau_final = 5e-7

            while time.time() < deadline and t < max_iters:
                # --------------------------------------------------------------
                # soft‑max -> probability vector (sum = 1)
                # --------------------------------------------------------------
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()                # a_i ≥ 0, Σa = 1

                # --------------------------------------------------------------
                # convolution & smooth‑max
                # --------------------------------------------------------------
                conv = full_convolution(a, fft_len=fft_len_conv)

                # annealed temperature
                tau = max(tau_final, tau_initial * (1.0 - t / max_iters))
                smooth, weights = smooth_max_and_weights(conv, tau)

                # exact objective (for bookkeeping)
                cur_val = 2.0 * n * conv.max()
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t:06d}] new best {cur_val:.6f}")

                # --------------------------------------------------------------
                # gradient of smooth‑max w.r.t. a   (∂smooth/∂a = 2·(weights ★ a))
                # --------------------------------------------------------------
                grad_a = 2.0 * cross_correlation(weights, a, fft_len=fft_len_corr)

                # back‑propagation through the soft‑max
                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                # --------------------------------------------------------------
                # Adam update
                # --------------------------------------------------------------
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)

                # learning‑rate decay (geometric)
                if t % decay_every == 0:
                    lr *= decay_factor
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # --------------------------------------------------------------
                # Stagnation handling – mass‑shift / sorting / random jitter
                # --------------------------------------------------------------
                if time.time() - last_improve > 30.0:
                    # try a mass‑shift move
                    a_shift = mass_shift_move(a, conv, delta=0.03)
                    # also try a sorted version (descending)
                    a_sorted = np.sort(a)[::-1]

                    # evaluate both candidates
                    val_shift = 2.0 * n * full_convolution(a_shift, fft_len=fft_len_conv).max()
                    val_sorted = 2.0 * n * full_convolution(a_sorted, fft_len=fft_len_conv).max()

                    # pick the best among current / shift / sorted
                    best_candidate = a
                    best_candidate_val = cur_val
                    if val_shift < best_candidate_val:
                        best_candidate = a_shift
                        best_candidate_val = val_shift
                    if val_sorted < best_candidate_val:
                        best_candidate = a_sorted
                        best_candidate_val = val_sorted

                    if best_candidate_val + 1e-12 < cur_val:
                        # accept the improvement and reset logits accordingly
                        best_seq = best_candidate.copy()
                        best_val = best_candidate_val
                        x = np.log(best_candidate + 1e-15)
                        print(f"[n={n} restart={r} iter={t}] stagnation move ↓ {best_candidate_val:.6f}")
                        last_improve = time.time()
                    else:
                        # if none helped, apply a small random jitter to logits
                        jitter = (np.random.rand(n) - 0.5) * 0.004
                        x += jitter
                        last_improve = time.time()
                        print(f"[n={n} restart={r} iter={t}] jitter (no improvement)")

            # ------------------------------------------------------------------
            # Local greedy refinement: tiny random mass shuffles
            # ------------------------------------------------------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a_local = best_seq.copy()
                refine_iters = 4000
                for it in range(refine_iters):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    # move a small fraction from i to j
                    move = min(a_local[i], 0.006) * (0.2 + 0.6 * np.random.rand())
                    if move <= 0:
                        continue
                    a_new = a_local.copy()
                    a_new[i] -= move
                    a_new[j] += move
                    # keep the vector normalized (sum = 1)
                    a_new /= a_new.sum()
                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = 2.0 * n * conv_new.max()
                    if val_new < best_val - 1e-12:
                        best_val = val_new
                        best_seq = a_new.copy()
                        a_local = a_new
                        print(f"[n={n} refine] improved to {val_new:.6f}")
            # early exit if we already beat the target
            if best_val <= 1.503:
                print(f"Target reached (value={best_val:.6f}); terminating early.")
                return best_seq.tolist()

    # ----------------------------------------------------------------------
    # Fallback – uniform vector (objective = 2) if nothing was found
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(1024, 1.0 / 1024.0, dtype=np.float64)
        best_val = local_objective(best_seq)

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
