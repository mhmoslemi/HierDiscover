    n=len(a)
    # w length 2n-1
    # conv of w with reversed a: (w * rev(a))[i] for i=0..2n-2, but we want only n entries.
    rev_a = a[::-1]
    # Use FFT convolution: conv = np.fft.irfft(np.fft.rfft(w, fft_len) * np.fft.rfft(rev_a, fft_len), fft_len)
    # Then take slice starting at n-1 to n-1+n-1 = 2n-2? Actually earlier code used start = a.shape[0] - 1.

    corr_full = np.fft.irfft(fa*fb, fft_len)

"""
Hybrid LP‑guided descent + exponentiated‑gradient refinement.

The algorithm works on a fixed length ‑‑ by default 4096 elements – and
tries to minimize

        F(a) = 2·n·max(a∗a) / (sum(a)²)

which is exactly the value returned by ``evaluate_sequence`` (the sum is
normalised to 1, so the denominator disappears).

* Phase 1 (LP‑step):
  For the current distribution f we compute M = max(f∗f) and solve the
  linear program

        maximise   Σ b_j
        subject to  (f∗b)_i ≤ M    for all i,
                    b_j ≥ 0.

  The constraints are linear in b, therefore a sparse matrix for the
  convolution can be built once and reused.  The solution b typically
  has Σb > 1, i.e. we can add extra mass without raising the current
  max‑convolution.  After normalising b to a probability vector g we
  perform a tiny line‑search over the mixture (1−t)·f + t·g and keep the
  best improvement.

* Phase 2 (refinement):
  Starting from the best vector found in phase 1 we run a pure
  sub‑gradient / exponentiated‑gradient scheme on the exact max‑convolution.
  The sub‑gradient is obtained from the active set of maximal convolution
  coefficients and the update is

        a ← a·exp(−η·grad)   (followed by renormalisation).

Both phases respect a global time budget (default 1000 s).  If a warm‑start
(`height_sequence_1`) is present it is used, otherwise a random Dirichlet
vector is taken as initial point.

The routine returns a plain list of non‑negative floats ready for
``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two not less than *x*."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*b using FFT (returns full length)."""
    fa = np.fft.rfft(a, fft_len)
    fb = np.fft.rfft(b, fft_len)
    conv = np.fft.irfft(fa * fb, fft_len)
    # keep only the needed part
    L = a.shape[0] + b.shape[0] - 1
    return conv[:L]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) returning a vector of length len(a).
    It satisfies (w ★ a)[i] = Σ_k w[k]·a[i+k−(n−1)].
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr_full = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr_full[start:start + a.shape[0]]


def _conv_matrix_structure(n: int):
    """
    Build the index arrays (rows, cols) for the sparse convolution matrix
    C with entries C[i, j] = 1   iff   i∈[j, j+n‑1].
    The actual data (the coefficients f_j) are supplied later.
    """
    length = n * n
    rows = np.empty(length, dtype=np.int32)
    cols = np.empty(length, dtype=np.int32)
    pos = 0
    for j in range(n):
        rows[pos:pos + n] = np.arange(j, j + n, dtype=np.int32)
        cols[pos:pos + n] = j
        pos += n
    return rows, cols


def _solve_lp_direction(f: np.ndarray, M: float,
                       rows: np.ndarray, cols: np.ndarray) -> np.ndarray | None:
    """
    Solve the LP that finds a direction b maximizing Σb_j under
    (f∗b)_i ≤ M.  Returns the optimal b (may be all zeros) or None on failure.
    """
    n = f.shape[0]
    data = np.repeat(f, n)                # length n*n
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)                       # maximise Σb → minimise –Σb
    b_ub = np.full(2 * n - 1, M + 1e-12, dtype=np.float64)  # tiny slack
    # SciPy’s HiGHS handles sparse matrices efficiently.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 works well in the published experiments.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if available)
    warm = globals().get('height_sequence_1')
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            a = warm / warm.sum()
        else:
            a = np.random.dirichlet(np.ones(n))
    else:
        a = np.random.dirichlet(np.ones(n))

    # initialise bookkeeping
    best_seq = a.copy()
    best_val = 2.0 * n * _full_convolution(a, a, conv_fft_len).max()   # sum(a)=1 ⇒ denominator =1
    print(f"[init] objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Pre‑compute the sparse index pattern for the convolution matrix.
    rows_idx, cols_idx = _conv_matrix_structure(n)

    # ------------------------------------------------------------------
    # Phase 1 – LP guided updates
    lp_iter = 0
    while time.time() < deadline and lp_iter < 20:
        conv_a = _full_convolution(a, a, conv_fft_len)
        M = conv_a.max()
        # Solve LP for a feasible direction
        b = _solve_lp_direction(a, M, rows_idx, cols_idx)
        if b is None:
            print("[LP] linear program failed – terminating LP phase")
            break
        sum_b = b.sum()
        if sum_b <= 1e-12:
            # No useful direction found
            print("[LP] trivial LP solution (all zeros) – terminating")
            break

        # Normalise the LP solution to obtain a probability vector g
        g = b / sum_b

        # Pre‑compute the three convolutions needed for the line‑search
        conv_g   = _full_convolution(g, g, conv_fft_len)
        conv_ag  = _full_convolution(a, g, conv_fft_len)

        # Simple line‑search over t ∈ {0.0, 0.1, …, 1.0}
        best_t = 0.0
        best_max = M
        for t in np.linspace(0.0, 1.0, 11)[1:]:          # skip t=0
            # mixture convolution: (1‑t)²·conv_a + 2·t·(1‑t)·conv_ag + t²·conv_g
            mix_conv = (1.0 - t) ** 2 * conv_a + 2.0 * t * (1.0 - t) * conv_ag + t ** 2 * conv_g
            cur_max = mix_conv.max()
            if cur_max < best_max - 1e-15:
                best_max = cur_max
                best_t = t

        if best_t == 0.0:
            # No improving step found – stop LP phase
            print(f"[LP] no improvement at iteration {lp_iter}")
            break

        # Update the distribution
        a = (1.0 - best_t) * a + best_t * g
        a = np.maximum(a, 0.0)
        a /= a.sum()

        cur_val = 2.0 * n * best_max
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[LP iter {lp_iter:02d}] new best = {best_val:.6f}")

        lp_iter += 1

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient refinement on the exact max
    eg_iter = 0
    while time.time() < deadline and eg_iter < 20000:
        conv = _full_convolution(a, a, conv_fft_len)
        max_c = conv.max()

        # active set of maximal entries (sub‑gradient)
        active = np.isclose(conv, max_c, atol=1e-12)
        if not active.any():
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        # gradient of the (exact) max‑convolution
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps non‑negativity)
        eta = 0.6
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[EG iter {eg_iter:05d}] new best = {best_val:.6f}")

        eg_iter += 1

        # early stop if we have not improved for a while
        if time.time() > deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # Optional up‑scaling to a longer sequence if time permits.
    if time.time() + 30 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        # linear interpolation of the best sequence found so far
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # simple EG refinement on the larger grid (fewer iterations)
        a = a_big
        corr_fft_len_big = _next_pow2(3 * n_big - 2)
        for _ in range(8000):
            if time.time() >= deadline:
                break
            conv = _full_convolution(a, a, conv_fft_len_big)
            max_c = conv.max()
            active = np.isclose(conv, max_c, atol=1e-12)
            if not active.any():
                break
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
            grad = 2.0 * _cross_correlation(w, a, corr_fft_len_big)
            eta = 0.6
            a = a * np.exp(-eta * grad)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            cur_val = 2.0 * n_big * max_c
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[big EG] new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final safety – clip to the domain allowed by ``evaluate_sequence``.
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return best_seq.tolist()
