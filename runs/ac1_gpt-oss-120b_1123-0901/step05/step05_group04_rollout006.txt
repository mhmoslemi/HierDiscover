def propose_candidate(seed=0, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    best_seq = None
    best_score = inf

    # Warm start search using global height_sequence_1

        a0 = np.array(hs, dtype=float)

        score0 = evaluate_sequence(a0.tolist())
        if score0 < best_score:
            best_score = score0
            best_seq = a0
    except Exception:
        pass

    # restarts loop:
    while time.time() < deadline:
        remaining = deadline - time.time()
        # Choose n.
        n = rng.integers(1500, 2500) # moderate.
        # Build initial a
        if best_seq is not None and best_seq.size == n and rng.random() < 0.4:
            a_init = best_seq.copy()
        else:
            # random shape as earlier code
            typ = rng.choice(['power','exponential','mixed','uniform'])
            idx = np.arange(1, n+1, dtype=float)
            if typ == 'power':

                a_init = idx**(-p)

                tau = rng.uniform(0.2,0.8)*n

"""
Propose a non‑negative step‑function (height sequence) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is a hybrid of two complementary heuristics:

1. **Linear‑programming based alternating improvement** (inspired by the
   “g‑update” described in the statement).  For a current candidate `a`
   we compute M = max(conv(a,a)).  A linear program then finds a vector
   `g ≥ 0` that maximises its ℓ₁‑norm while respecting the linear
   constraints `conv(a, g) ≤ M`.  Mixing `a` and `g` (`a←(1‑t)a + t g`)
   with a decreasing step size strictly reduces the maximal convolution
   value.

2. **Short Adam‑softmax refinements** – after the LP phase a few Adam
   optimisation steps on a soft‑max parametrisation are performed to
   polish the solution and escape any remaining plateaux.

Multiple random restarts with varying lengths `n` are performed
within the allotted time budget (≈1000 s).  The best sequence found is
returned as a plain Python list.  The implementation falls back to the
Adam‑only path if SciPy is unavailable.
"""

def _initial_shape_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a positive random profile of length ``n`` (before normalisation)."""
    typ = rng.choice(['power', 'exponential', 'mixed', 'uniform'])
    idx = np.arange(1, n + 1, dtype=float)

    if typ == 'power':
        p = rng.uniform(0.1, 2.0)
        shape = idx ** (-p)
    elif typ == 'exponential':
        tau = rng.uniform(0.2, 0.9) * n
        shape = np.exp(-idx / tau)
    elif typ == 'mixed':
        split = rng.integers(1, n - 1)
        left = idx[:split] ** (-rng.uniform(0.2, 1.0))
        right = (n - idx[split:] + 1) ** (-rng.uniform(0.2, 1.0))
        shape = np.concatenate([left, right])
    else:  # uniform
        shape = np.ones_like(idx)

    # tiny jitter to avoid perfect symmetry
    shape += rng.random(n) * 1e-6
    shape = np.maximum(shape, 1e-12)
    return shape


def _solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  max   1ᵀg
    s.t.    conv(a,g) ≤ M   (2n‑1 linear constraints)
            g ≥ 0
    Returns the optimal vector ``g`` (may be all zeros if the solver fails).
    """
    if not _SCIPY_OK:               # pragma: no cover
        return np.zeros_like(a)

    n = a.shape[0]
    # Build the dense convolution matrix: conv(a,g) = T(a) @ g
    # First column = a padded with zeros, first row = [a₀, 0,…,0]
    col = np.concatenate([a, np.zeros(n - 1)])
    row = np.concatenate([a[:1], np.zeros(n - 1)])
    A = toeplitz(col, row)          # shape (2n‑1, n), dense but still modest

    # Linear programme data
    c = -np.ones(n)                 # minimise -sum(g)  ⇔ maximise sum(g)
    ub = np.full(2 * n - 1, M)

    # Solve with HiGHS (fast interior‑point+simplex hybrid)
    res = linprog(
        c=c,
        A_ub=A,
        b_ub=ub,
        bounds=[(0.0, None)] * n,
        method='highs',
        options={'presolve': True},
    )
    if not res.success:
        # Fallback – return the current a (feasible) so the outer loop can continue.
        return np.zeros_like(a)
    return res.x


def _lp_iterative_improve(a0: np.ndarray,
                          target_sum: float,
                          deadline: float,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Starting from ``a0`` (already normalised to ``target_sum``) perform a
    sequence of LP‑updates (the “g‑step”) with a simple convex combination.
    Returns the best sequence found and its evaluation score.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = evaluate_sequence(a.tolist())

    # step size schedule – start fairly aggressive and then decay
    t = 0.5
    iteration = 0
    last_improve_iter = 0

    while time.time() < deadline:
        # 1) compute current max‑convolution
        conv = np.convolve(a, a)
        M = conv.max()

        # 2) solve LP for g respecting conv(a,g) ≤ M
        g = _solve_lp_max_sum(a, M)

        # sanity check – if LP failed to produce a positive vector, abort
        s = g.sum()
        if s <= 1e-12:
            break

        # 3) scale g to meet the canonical sum
        g = g * (target_sum / s)

        # 4) convex update
        a_new = (1.0 - t) * a + t * g
        # renormalise (numerical drift)
        a_new = a_new * (target_sum / a_new.sum())

        # 5) evaluate
        score_new = evaluate_sequence(a_new.tolist())

        # keep global best
        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            last_improve_iter = iteration
            print(f"[{time.time():.2f}] LP improvement: score={best_score:.8f} (n={len(a)})")

        # prepare for next iteration
        a = a_new
        iteration += 1
        t = max(0.05, t * 0.96)           # slowly shrink step size

        # early stop if no improvement for a while
        if iteration - last_improve_iter > 30:
            break

    return best_seq, best_score

def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using an FFT of size power‑of‑2 (real input)."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)      # strip tiny negatives
    return conv


def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    rng: np.random.Generator,
    *,
    init_tau_factor: float = 0.25,
    min_tau_factor: float = 5e-4,
    alpha: float = 0.07,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau_decay: float = 0.96,
    max_iter: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a smoothed‑max surrogate (the same routine that
    produced the previous best result).  Returns the best sequence and
    its official score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_val = float("inf")
    iteration = 0
    last_improve = 0

    while time.time() < run_deadline:
        # ---- 1. map logits → simplex via soft‑max (norm = target_sum) ----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        p = exp_shift / exp_shift.sum()
        a = target_sum * p

        # ---- 2. convolution & smooth‑max weighting --------------------
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()

        if max_conv < best_val:
            best_val = max_conv
            best_seq = a.copy()
            last_improve = iteration
            print(f"[{time.time():.2f}] Adam local best = {best_val:.8f}")

        # smooth‑max via log‑sum‑exp
        tau = max(init_tau_factor * max_conv * (tau_decay ** iteration),
                  min_tau_factor * max_conv)
        scaled = (conv - max_conv) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient wrt a (derivative of Σ w_k·conv_k)
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)      # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # chain rule through soft‑max
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # optional clipping
        gnorm = np.linalg.norm(grad_x)
        if gnorm > 10.0:
            grad_x = grad_x * (10.0 / gnorm)

        # ---- 6. Adam update -----------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # jitter occasionally
        if iteration % 300 == 0 and rng.random() < 0.15:
            x += rng.normal(size=n) * 0.001

        alpha *= 0.9997

        # early stop on stagnation
        if iteration - last_improve > 2000:
            break
        if max_iter is not None and iteration >= max_iter:
            break

    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score

            a = best_seq * (target_sum_adam / best_seq.sum())
            x0 = np.log(a + 1e-12)
            seq_adam, score_adam = _optimise_adam_softmax(
                n_adam,
                x0,
                target_sum_adam,
                extra_deadline,
                rng,
                init_tau_factor=0.12,
                min_tau_factor=1e-5,
                alpha=0.04,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
                tau_decay=0.94,
                max_iter=None,
            )
            if score_adam < best_score:
                best_score = score_adam
                best_seq = seq_adam.copy()
                top_candidates.append((best_score, best_seq.copy()))
                top_candidates.sort(key=lambda z: z[0])
                top_candidates = top_candidates[:5]
                print(f"[{time.time():.2f}] GLOBAL NEW BEST (Adam) = {best_score:.8f}")

    # ----------------------------------------------------------------------
    # Final refinement on the incumbent (if time permits)
    # ----------------------------------------------------------------------
    if best_seq is not None and (deadline - time.time()) > 8.0:
        n = best_seq.size
        target_sum = math.sqrt(2.0 * n)
        a = best_seq * (target_sum / best_seq.sum())
        x0 = np.log(a + 1e-12)
        run_deadline = deadline
        seq_ref, score_ref = _optimise_adam_softmax(
            n,
            x0,
            target_sum,
            run_deadline,
            rng,
            init_tau_factor=0.08,
            min_tau_factor=5e-5,
            alpha=0.03,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            tau_decay=0.93,
            max_iter=None,
        )
        if score_ref < best_score:
            best_score = score_ref
            best_seq = seq_ref.copy()
            print(f"[{time.time():.2f}] REFINEMENT improved best to {best_score:.8f}")

    # ----------------------------------------------------------------------
    # Fallback (should never be needed)
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list as required by the outer harness
    return best_seq.tolist()
