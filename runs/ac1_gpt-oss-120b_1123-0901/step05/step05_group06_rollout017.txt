"""
Propose a non‑negative step function minimising ``evaluate_sequence``.

Algorithm
---------
1. **Parameter sweep** – generate many simple parametric shapes
   (Beta‑distribution, Gaussian, powered sine) for a handful of lengths.
   The shapes are normalised to Σa = √(2 n) and evaluated.
   The best vector from this cheap global search is kept.

2. **Refinement with Adam** – starting from the best vector (and a few
   jittered variants) run a soft‑max Adam optimiser on the log‑its.
   This optimisation respects the simplex constraint a≥0, Σa = √(2 n)
   and directly minimises the smoothed max‑convolution.

3. **Stochastic hill‑climbing** – a final simulated‑annealing style
   mass‑transfer loop moves a small fraction of the mass between random
   positions, always accepting improvements and sometimes worsening
   moves, to squeeze out the remaining bits.

The whole routine respects the 1000 s budget, falls back to a uniform
vector if nothing is found, and returns a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top‑level, no closures

def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values.
    Returns (weights, smoothed max)."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """Gradient of the (soft‑max) max‑convolution (reverse convolution)."""
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]          # linear part
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Exact max of a∗a using FFT."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Create a modest but diverse set of normalised start vectors."""
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0
             - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    init.append(black / black.sum() * target_sum)

    # 7) small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    pert = pert / pert.sum() * target_sum
    init.append(pert)

    # 8) triangular shape (peak in the centre)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    init.append(tri / tri.sum() * target_sum)

    return init


def _exp_gradient_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑its using a soft‑max approximation of the
    max‑convolution.  Returns the best vector seen in this run together
    with its exact max‑convolution value.
    """
    target_sum = math.sqrt(2.0 * n)

    eps = 1e-16
    y = np.log(a0 + eps)                # logits

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.08

    # temperature schedule
    tau_factor = 0.35                # slightly higher initial temperature
    decay_rate = max_iters / 7.0

    # FFT size for linear conv of length 3 n‑2
    Nfft = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- current probability simplex ----
        logits = y - np.max(y)               # shift for stability
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()
        a = target_sum * p

        # ---- exact convolution and its max ----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ---- soft‑max weights ----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ---- gradient wrt a (2·rev) ----
        rev = _weighted_rev(a, w, Nfft)
        g_a = 2.0 * rev

        # ---- chain rule to logits (mirror‑descent on simplex) ----
        avg = np.dot(p, g_a)
        g_y = target_sum * p * (g_a - avg)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # occasional jitter to escape plateaux
        if it % 2500 == 0:
            y += rng.normal(scale=0.02, size=n)

        # keep global best
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[Adam] len={n:5d} it={it:8d} → new best {best_score:.6f}")

    return best_seq, best_score


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style stochastic hill‑climbing on the simplex.
    Moves a small random fraction of mass from i → j, accepting
    improvements always and occasional uphill moves.
    """
    target_sum = a.sum()                     # already √(2 n)
    cur_seq = a.copy()
    cur_score = best_score

    Nfft = 1 << ((2 * n - 1).bit_length())
    max_steps = 300_000
    step = 0

    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        # transfer a modest fraction (1 %–10 %)
        frac = rng.uniform(0.01, 0.10)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[Refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated annealing acceptance
            T = max(0.02 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score

# Parametric families for the cheap global search

def _gen_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on (0,1) sampled at mid‑points."""
    x = (np.arange(n) + 0.5) / n
    y = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    y = np.maximum(y, 0.0)
    return y


def _gen_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n)
    center = (n - 1) / 2.0
    y = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    return y


def _gen_sine_pow(n: int, p: float) -> np.ndarray:
    i = np.arange(n)
    y = np.sin(np.pi * (i + 0.5) / n) ** p
    return np.maximum(y, 0.0)
