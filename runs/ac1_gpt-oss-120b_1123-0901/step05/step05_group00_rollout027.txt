Propose candidate step‑function.
The method works in three phases:
1. Random search over several analytic families (beta, power‑law,

2. The best shape is up‑sampled to the target length (16384) and refined
   by a long Adam optimisation on a smooth‑max surrogate of the max
   convolution entry.  Symmetry of the vector is enforced throughout.
3. An exact sub‑gradient descent with exponentiated‑gradient updates
   finishes the optimisation.  Small random perturbations are tried at
   the end if time permits.
The routine respects the given wall‑clock budget and returns the best
vector (as a plain Python list) found.
"""

import time
import numpy as np
import random

def _next_pow_two(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _fft_len_for_conv(n: int) -> int:
    """Length of zero‑padded FFT to compute a linear convolution of length 2n‑1."""
    return _next_pow_two(2 * n - 1)

def _fft_len_for_corr(n: int) -> int:
    """Length for cross‑correlation w★a (needs length (2n‑1)+(n‑1))."""
    return _next_pow_two(3 * n - 2)

def _compute_objective(a: np.ndarray) -> float:
    """
    Exact objective F(a)=2·n·max(a*a)/(sum(a)²) for a non‑negative vector.
    The vector is normalised inside the function.
    """
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    n = a.shape[0]
    fft_len = _fft_len_for_conv(n)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    max_c = conv.max()
    return float(2.0 * n * max_c)

def _smoothmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Return a probability vector proportional to exp(lam·(conv−max(conv)))."""
    max_c = conv.max()
    w = np.exp(lam * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) of two vectors of length n.
    Returns a vector of length n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

def _upsample(seq: np.ndarray, target_len: int) -> np.ndarray:
    """Linear interpolation from len(seq) to target_len keeping symmetry."""
    x_old = np.linspace(0.0, 1.0, seq.shape[0])
    x_new = np.linspace(0.0, 1.0, target_len)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0.0)
    if up.sum() == 0.0:
        up = np.full_like(up, 1.0 / up.size)
    else:
        up = up / up.sum()
    # enforce symmetry
    up = (up + up[::-1]) / 2.0
    up /= up.sum()
    return up

The method works in three phases:

1. **Random analytic search** (n≈4096).  
   A diverse pool of families (uniform, Dirichlet, Beta,
   edge‑concentrated power‑law, cosine‑/sine‑powered, double‑Gaussian) is
   sampled.  Each candidate is normalised, symmetrised and evaluated
   exactly.  The best shape is retained.

2. **Upsampling + Adam optimisation** (target n=16384).  
   The best shape is linearly interpolated to the full length, turned into
   symmetric logits and refined by Adam on a smooth‑max surrogate of the
   convolution maximum.  Symmetry of the logits (hence of the vector) is
   enforced at every step.

3. **Exact sub‑gradient refinement**.  
   An exponentiated‑gradient update using the true active set of maximal
   convolution entries flattens the autocorrelation further.  Small random
   perturbations are tried if time permits.

    return 1 << (x - 1).bit_length()

    """Length of zero‑padded FFT for a linear convolution of size 2·n‑1."""
    return _next_pow_two(2 * n - 1)

    """Length of zero‑padded FFT for the cross‑correlation w★a (length 3·n‑2)."""
    return _next_pow_two(3 * n - 2)

    inside the function; a must be non‑negative.

    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _upsample(seq: np.ndarray, target_len: int) -> np.ndarray:
    """Linear interpolation from len(seq) to target_len, preserving symmetry."""
    x_old = np.linspace(0.0, 1.0, seq.shape[0])
    x_new = np.linspace(0.0, 1.0, target_len)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0.0)
    if up.sum() == 0.0:

    else:
        up = up / up.sum()

    return up
