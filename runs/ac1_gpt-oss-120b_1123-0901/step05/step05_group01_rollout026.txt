def _random_piecewise_constant(n):
    k = np.random.randint(2, 6)
    block_lengths = np.random.dirichlet(np.ones(k)) * n
    block_lengths = np.maximum(np.round(block_lengths).astype(int), 1)

            diff -= 1
        else:

                block_lengths[idx] -= 1
                diff += 1

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_val = float('inf')
    hist_seq = None
    if 'height_sequence_1' in globals():
        tmp = globals()['height_sequence_1']
        if isinstance(tmp, (list, tuple, np.ndarray)):
            hist_seq = np.asarray(tmp, dtype=np.float64)

    # define list of lengths
    lengths = [256, 512, 1024, 2048, 4096, 8192]
    # compute fft lengths per n
    conv_fft = {}
    corr_fft = {}
    for n in lengths:
        conv_len = 2*n - 1
        conv_fft[n] = 1 << int(np.ceil(np.log2(conv_len)))
        corr_fft[n] = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")
        fft_len_c = conv_fft[n]
        fft_len_x = corr_fft[n]

        candidates = _generate_initial_candidates(n, best_seq, hist_seq)

        # Determine iteration budgets, scaled with n
        max_adam_iters = int(18000 * 1024 / n) + 2500   # slightly higher than before
        max_subgrad_iters = 4000
        max_hc_iters = 3000

        for idx, a_start in enumerate(candidates):
            if time.time() > deadline:
                break
            # Adam
            a_adam, val_adam = _adam_optimize(a_start, n, fft_len_c, fft_len_x,
                                              deadline, max_adam_iters)
            # subgrad
            a_sub, val_sub = _subgrad_optimize(a_adam, n, fft_len_c, fft_len_x,
                                               deadline, init_val=val_adam,

                                                init_val=val_sub,
                                                max_iters=max_hc_iters)
            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.copy()
                print(f"*** Global new best (n={n}, cand={idx}) value={best_val:.6f}")

    # final refinement
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        fft_len_c = conv_fft.get(n, 1 << int(np.ceil(np.log2(2 * n - 1))))

"""
Search for a low‑value non‑negative step‑function sequence.

Algorithm
---------
1. Generate a diverse set of warm‑starts for a given length:
   - the current global best (interpolated if needed),
   - a historic solution if supplied,
   - uniform vector,
   - several random Dirichlet draws (plain, sorted, permuted),
   - power‑law edge concentrations,
   - two‑Gaussian edge shapes,
   - inverse‑root edge shapes,
   - sinusoidal half‑wave, triangular window,
   - a few random piece‑wise‑constant blocks.
2. For every candidate run three optimisation phases:
   a) **Adam** on a smooth log‑sum‑exp surrogate of the max‑convolution.
   b) **Exact sub‑gradient** descent on the true max‑convolution
      (exponentiated‑gradient update that keeps the variables non‑negative).
   c) **Cheap hill‑climb** that transfers a small amount of mass between
      two random positions.
3. Keep the best sequence across all explored lengths.
4. If time permits, perform a final longer Adam refinement (and a final
   sub‑gradient pass) on the global best.
All steps respect a global deadline (default 1000 s) and the function
returns a plain Python list as required by the external evaluator.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, length‐power‑of‑2 for speed)
def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear full convolution a*a (length 2·len(a)‑1) using FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) = n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)²."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence by linear interpolation and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new


# ----------------------------------------------------------------------
# Additional candidate generators
def _random_piecewise_constant(n: int) -> np.ndarray:
    """
    Random piece‑wise‑constant shape: pick 2‑5 blocks, assign random heights.
    The resulting vector is normalised to sum‑to‑one.
    """
    k = np.random.randint(2, 6)               # number of blocks
    # block lengths from a Dirichlet distribution, at least 1 element each
    lengths = np.random.dirichlet(np.ones(k)) * n
    lengths = np.maximum(np.round(lengths).astype(int), 1)
    diff = n - lengths.sum()
    # fix rounding errors
    while diff != 0:
        idx = np.random.randint(k)
        if diff > 0:
            lengths[idx] += 1
            diff -= 1
        else:
            if lengths[idx] > 1:
                lengths[idx] -= 1
                diff += 1
    heights = np.random.rand(k) + 1e-3
    seq = np.concatenate([np.full(l, h) for l, h in zip(lengths, heights)])
    seq = seq[:n]                               # safety trim
    seq = np.maximum(seq, 0.0)
    seq /= seq.sum()
    return seq

def _generate_initial_candidates(
    n: int,
    best_seq: np.ndarray | None,
    hist_seq: np.ndarray | None,
) -> list[np.ndarray]:
    """
    Produce a diverse set of warm‑starts for length n.
    All returned arrays are non‑negative and sum to 1.
    """
    cand = []

    # 0) warm‑start from the global best (exact length)
    if best_seq is not None and best_seq.shape[0] == n:
        cand.append(best_seq.copy())
    # 1) warm‑start by interpolating the best to this length
    if best_seq is not None and best_seq.shape[0] != n:
        cand.append(_interpolate_seq(best_seq, n))

    # 2) historic sequence (if any)
    if hist_seq is not None:
        if hist_seq.shape[0] >= n:
            tmp = hist_seq[:n]
        else:
            tmp = np.pad(hist_seq, (0, n - hist_seq.shape[0]))
        tmp = np.maximum(tmp, 0.0)
        s = tmp.sum()
        if s < 1e-12:
            tmp = np.random.dirichlet(np.ones(n))
        else:
            tmp = tmp / s
        cand.append(tmp)

    # 3) uniform vector (the classic baseline)
    cand.append(np.full(n, 1.0 / n))

    # 4) random Dirichlet draws (plain, sorted, permuted)
    for _ in range(3):
        cand.append(np.random.dirichlet(np.ones(n)))
    d = np.random.dirichlet(np.ones(n))
    np.random.shuffle(d)
    cand.append(d)                                            # permuted
    cand.append(np.sort(d)[::-1])                             # descending
    cand.append(np.sort(d))                                   # ascending

    # 5) edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 4):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # 6) two‑Gaussian edge shapes
    for gamma in (0.2, 0.5, 1.0, 2.0):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        cand.append(shape / shape.sum())

    # 7) inverse‑root edge shapes
    for alpha in (0.5, 0.8, 1.0):
        i = np.arange(n, dtype=np.float64) + 1.0
        shape = i ** (-alpha) + (n - i + 1.0) ** (-alpha)
        cand.append(shape / shape.sum())

    # 8) sinusoidal half‑wave window (zero at the edges)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    shape = np.sin(np.pi * x) ** 2
    cand.append(shape / shape.sum())

    # 9) triangular window (peak in the centre)
    shape = 1.0 - np.abs(2.0 * x - 1.0)
    shape = np.maximum(shape, 0.0)
    cand.append(shape / shape.sum())

    # 10) random piece‑wise‑constant blocks
    for _ in range(3):
        cand.append(_random_piecewise_constant(n))

    # shuffle and keep a limited number (≈20)
    np.random.shuffle(cand)
    final = []
    for c in cand:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        final.append(c / s)
        if len(final) >= 20:
            break
    return final

# Optimisers
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    max_iters: int,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate (log‑sum‑exp).
    Returns the best (sequence, value) encountered before the deadline.
    """
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.03

    best_a = a.copy()
    conv = _full_convolution(a, fft_len_conv)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # simplex projection via soft‑max (logits → probabilities)
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # smooth‑max weights (log‑sum‑exp)
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        ws = w.sum()
        if ws == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws

        # gradient of the surrogate (chain rule through the soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # increase temperature – smoother -> sharper approximation
        lam = min(lam * lam_factor, 1e7)

        cur_val = 2.0 * n * max_c / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t:06d}] new best {cur_val:.6f}")

        # random shake if stuck for a while
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


def _subgrad_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent (exponentiated‑gradient) on the true max.
    """
    a = a0 / a0.sum()
    best_a = a.copy()
    if init_val is None:
        conv = _full_convolution(a, fft_len_conv)
        best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    else:
        best_val = init_val
    last_improve = time.time()
    eta = 0.6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        conv = _full_convolution(a, fft_len_conv)
        cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it}] new best {cur_val:.6f}")

        # stop early if no improvement for a few seconds
        if time.time() - last_improve > 4.0:
            break

    return best_a, best_val


def _hill_climb_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    deadline: float,
    init_val: float | None = None,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Simple cheap local search: repeatedly shift a small amount of mass
    between two random coordinates and keep the move if it improves the objective.
    """
    a = a0.copy()
    cur_val = (
        _objective(a, n, fft_len_conv) if init_val is None else init_val
    )
    best_a = a.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2:
            continue

        # move up to 5 % of the mass from i1 to i2
        eps = np.random.rand() * 0.05 * a[i1]
        if eps <= 0.0:
            continue
        new_a = a.copy()
        new_a[i1] -= eps
        new_a[i2] += eps
        if new_a[i1] < -1e-15:
            continue
        new_a /= new_a.sum()

        new_val = _objective(new_a, n, fft_len_conv)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            if it % 50 == 0:
                print(f"[n={n} HillClimb it={it}] new best {new_val:.6f}")
        else:
            # tiny probability to accept uphill move – helps escape plateaus
            if np.random.rand() < 0.001:
                a = new_a

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best sequence found as a plain Python list.
    The routine respects the supplied time budget (seconds).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_val = float("inf")

    # optional historic solution
    hist_seq = None
    if "height_sequence_1" in globals():
        tmp = globals()["height_sequence_1"]
        if isinstance(tmp, (list, tuple, np.ndarray)):
            hist_seq = np.asarray(tmp, dtype=np.float64)

    # lengths to explore – start modest, finish with the largest feasible.
    lengths = [256, 512, 1024, 2048, 4096, 8192]

    # pre‑compute FFT lengths for each n
    conv_fft = {}
    corr_fft = {}
    for n in lengths:
        conv_len = 2 * n - 1
        conv_fft[n] = 1 << int(np.ceil(np.log2(conv_len)))
        corr_fft[n] = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

    # ------------------------------------------------------------------
    # MAIN optimisation loop over the chosen lengths
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        fft_len_c = conv_fft[n]
        fft_len_x = corr_fft[n]

        # generate a modest but diverse set of warm‑starts
        candidates = _generate_initial_candidates(n, best_seq, hist_seq)

        # iteration budgets (scale roughly with 1/n)
        max_adam_iters = int(18000 * 1024 / n) + 2500   # a bit more than the reference
        max_subgrad_iters = 4000
        max_hc_iters = 3000

        for cand_id, a_start in enumerate(candidates):
            if time.time() > deadline:
                break

            # ---- Phase 1 – smooth‑max Adam ----
            a_adam, val_adam = _adam_optimize(
                a_start, n, fft_len_c, fft_len_x, deadline, max_adam_iters
            )

            # ---- Phase 2 – exact sub‑gradient descent ----
            a_sub, val_sub = _subgrad_optimize(
                a_adam,
                n,
                fft_len_c,
                fft_len_x,
                deadline,
                init_val=val_adam,
                max_iters=max_subgrad_iters,
            )

            # ---- Phase 3 – cheap hill‑climb local search ----
            a_hc, val_hc = _hill_climb_optimize(
                a_sub,
                n,
                fft_len_c,
                deadline,
                init_val=val_sub,
                max_iters=max_hc_iters,
            )

            # keep global best if improved
            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.copy()
                print(f"*** Global new best (n={n} cand={cand_id}) value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional long‑run refinement on the global best (if any time left)
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        fft_len_c = conv_fft.get(
            n, 1 << int(np.ceil(np.log2(2 * n - 1)))
        )
        fft_len_x = corr_fft.get(
            n, 1 << int(np.ceil(np.log2(3 * n)))
        )
        print("\n=== Final refinement of the global best ===")
        # A longer Adam run with a slower learning rate and more iterations.
        refined_seq, refined_val = _adam_optimize(
            best_seq,
            n,
            fft_len_c,
            fft_len_x,
            deadline,
            max_iters=int(1.5 * max_adam_iters + 3000),
        )
        if refined_val < best_val:
            best_val, best_seq = refined_val, refined_seq.copy()
            print(f"*** Refined global best after Adam: {best_val:.6f}")

        # final sub‑gradient pass
        if time.time() < deadline:
            refined_seq2, refined_val2 = _subgrad_optimize(
                best_seq,
                n,
                fft_len_c,
                fft_len_x,
                deadline,
                init_val=best_val,
                max_iters=4000,
            )
            if refined_val2 < best_val:
                best_val, best_seq = refined_val2, refined_seq2.copy()
                print(f"*** Refined global best after sub‑grad: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (highly unlikely)
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = _objective(best_seq, fallback_n, conv_fft[fallback_n])

    print(f"\nFinal best value: {best_val:.6f}")
    # Return a plain list as required by the external evaluator.
    return best_seq.tolist()
