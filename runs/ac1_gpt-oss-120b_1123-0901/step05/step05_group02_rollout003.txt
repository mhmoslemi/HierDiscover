def cross_corr(w, a, fft_len):
    n = len(a)
    rev_a = a[::-1]
    # compute with zero padding
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    # slice
    start = n - 1
    return corr[start:start+n]

"""
Propose a step‑function (non‑negative list) that (approximately) minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

where “a*a” is the linear convolution (numpy.convolve).

The algorithm works in two stages:

1. **Smooth‑max Adam optimisation** –  
   The max in the objective is replaced by a differentiable soft‑max
   (log‑sum‑exp).  Adam is applied to the soft‑max surrogate while
   keeping the coefficients on the probability simplex (logits → soft‑max).

2. **Simulated‑annealing refinement** –  
   Starting from the best point found by Adam, we perform a cheap
   multiplicative‑noise random walk (log‑normal perturbations) with
   an exponential cooling schedule.  The true (non‑smoothed) objective
   is evaluated after each move, and uphill moves are occasionally
   accepted according to a temperature parameter.

If time permits after the first optimisation, the current best sequence
is interpolated to a longer length (up to 8192) and the whole procedure
is repeated, exploiting the extra degrees of freedom of a larger
representation.

All heavy work (convolution and the gradient of the soft‑max) is done
with FFTs for speed.  The routine respects the supplied time budget and
returns a plain Python list that can be fed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_corr(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) for a probability vector a."""
    conv = _conv_fft(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _adam_optimize(
    init_a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    best_seq: np.ndarray,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best sequence found in this run together with its exact value.
    """
    # initialise logits from the starting vector (add a tiny jitter)
    eps = 1e-15
    logits = np.log(init_a + eps) + np.random.randn(n) * 0.05

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    # Adam hyper‑parameters
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.02

    # soft‑max temperature schedule
    lam = 0.5          # start “smoothness”
    lam_factor = 1.04  # increase factor per iteration
    max_lam = 1e6

    # iteration budget – generous, but we also respect the global deadline
    max_iters = 20000

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max → probability simplex
        mx = logits.max()
        exp_logits = np.exp(logits - mx)
        a = exp_logits / exp_logits.sum()

        # convolution & smooth‑max weights
        conv = _conv_fft(a, fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective
        grad_a = 2.0 * _cross_corr(w, a, fft_len)

        # back‑prop through the soft‑max (mirror‑descent on the simplex)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on the logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase the temperature (make the surrogate approach the true max)
        lam = min(lam * lam_factor, max_lam)

        # exact objective (no smoothing)
        val = 2.0 * n * max_c
        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            print(f"Adam iter {t:05d} – new best {val:.6f}")

        # occasional jitter if we’re stuck for a while
        if t % 500 == 0:
            # nothing – the loop will automatically continue
            pass

    return best_seq, best_val


def _anneal_refine(
    init_a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    best_seq: np.ndarray,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing / evolutionary refinement.
    Multiplicative log‑normal perturbations keep the vector on the simplex.
    """
    a = init_a.copy()
    val = best_val

    # a few auxiliary objects
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # temperature starts as a small fraction of the current value
    temperature = 0.02 * val
    cooling = 0.9995

    # noise level (relative std‑dev of the multiplicative factor)
    sigma = 0.02
    sigma_min = 0.0005
    sigma_decay = 0.9999

    while time.time() < deadline:
        # propose a multiplicative perturbation (log‑normal)
        noise = np.random.randn(n) * sigma
        a_prop = a * np.exp(noise)

        # occasionally blend a bit with the uniform distribution
        if np.random.rand() < 0.05:
            alpha = np.random.rand() * 0.1
            a_prop = (1 - alpha) * a_prop + alpha * uniform

        # re‑normalise to the simplex
        s = a_prop.sum()
        if s <= 0.0:
            continue
        a_prop /= s

        # exact objective
        conv = _conv_fft(a_prop, fft_len)
        max_c = conv.max()
        val_prop = 2.0 * n * max_c

        delta = val_prop - val
        if delta < 0 or np.exp(-delta / max(temperature, 1e-12)) > np.random.rand():
            # accept
            a = a_prop
            val = val_prop
            if val < best_val - 1e-12:
                best_val = val
                best_seq = a.copy()
                print(f"SA step – new best {val:.6f}")

        # cool down
        temperature = max(temperature * cooling, 1e-9)
        sigma = max(sigma * sigma_decay, sigma_min)

    return best_seq, best_val


def _optimize_once(
    a0: np.ndarray,
    n: int,
    time_budget: float,
    global_best_seq: np.ndarray,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run one full optimisation round (Adam + SA) on a vector of length n.
    The function respects the supplied *time_budget* seconds.
    """
    deadline = time.time() + time_budget

    fft_len = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 1) Adam phase
    best_seq, best_val = _adam_optimize(
        init_a=a0,
        n=n,
        fft_len=fft_len,
        deadline=deadline,
        best_seq=global_best_seq,
        best_val=global_best_val,
    )

    # --------------------------------------------------------------
    # 2) Simulated‑annealing refinement
    best_seq, best_val = _anneal_refine(
        init_a=best_seq,
        n=n,
        fft_len=fft_len,
        deadline=deadline,
        best_seq=best_seq,
        best_val=best_val,
    )

    return best_seq, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence that minimises the evaluation function.
    The routine respects a time budget (default 1000 s) and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    # safety margin – we stop a bit early to avoid abrupt termination
    deadline = time.time() + budget_s - 0.5

    # ------------------------------------------------------------------
    # Starting point: use the previously stored sequence if it exists,
    # otherwise fall back to a random Dirichlet vector of length 4096.
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple)):
        a0 = np.asarray(prev, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s < 1e-12:
            a0 = np.random.dirichlet(np.ones(len(a0)))
        else:
            a0 /= s
        n = a0.shape[0]
        print(f"Using previous sequence of length {n}")
    else:
        n = 4096
        a0 = np.random.dirichlet(np.ones(n))
        print(f"No previous sequence – starting from random length {n}")

    # ------------------------------------------------------------------
    # Global best tracking (across possible up‑sampling steps)
    best_seq = a0.copy()
    best_val = _objective(best_seq, n, _next_pow2(2 * n - 1))
    print(f"Initial objective value: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – optimise the current length
    # Allocate roughly 60 % of the remaining budget to the first length.
    remaining = deadline - time.time()
    phase1_budget = min(remaining * 0.6, 600.0)  # cap to keep some time for up‑sampling
    best_seq, best_val = _optimize_once(
        best_seq, n, phase1_budget, best_seq, best_val
    )

    # ------------------------------------------------------------------
    # Phase 2 – if time permits, upscale to a longer length (max 8192)
    while time.time() < deadline:
        if n >= 8192:
            break  # already at our maximum practical size
        # decide new size (double, but never exceed 8192)
        new_n = min(8192, n * 2)

        # linear interpolation to the larger grid
        x_old = np.linspace(0.0, 1.0, n)
        x_new = np.linspace(0.0, 1.0, new_n)
        a_up = np.interp(x_new, x_old, best_seq)
        a_up = np.maximum(a_up, 0.0)
        a_up /= a_up.sum()

        print(f"\n--- Upscaling: {n} → {new_n} ---")
        n = new_n
        best_seq, best_val = _optimize_once(
            a_up, n, deadline - time.time(), best_seq, best_val
        )
        # loop will break automatically when time runs out or n hits 8192

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain list)
    best_seq = np.maximum(best_seq, 0.0)
    s = best_seq.sum()
    if s == 0.0:
        best_seq = np.full(best_seq.shape, 1.0 / best_seq.shape[0])
    else:
        best_seq = best_seq / s

    print(f"\nFinal best value: {best_val:.6f} (length {best_seq.shape[0]})")
    return best_seq.tolist()
