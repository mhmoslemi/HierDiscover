            # optimization loop
            max_iters = 200000 if n <= 1024 else 150000 if n <= 2048 else 100000
            lr0 = 0.5 if n <= 512 else 0.2
            t = 0
            best_local_val = float('inf')

            while time.time() < deadline and t < max_iters:

                if cur_val < best_val - 1e-12:

                max_val = conv.max()

def project_simplex(v):
    # returns projection onto simplex {x >=0, sum = 1}
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def subgradient_descent(a, n, deadline):
    conv = full_convolution(a)
    M = conv.max()
    best_local = a.copy()
    best_local_val = 2*n * M
    lr0 = 0.1
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = full_convolution(a)
        M = conv.max()
        # check improvement
        val = 2.0 * n * M
        if val < best_local_val - 1e-12:
            best_local_val = val
            best_local = a.copy()
        # gradient w.r.t. max entries
        mask = conv >= M - 1e-12
        w = mask.astype(float)
        w /= w.sum()
        grad = 2.0 * cross_correlation(w, a, fft_len_corr)
        # step size
        lr = lr0 / np.sqrt(it + 1)
        a_new = a - lr * grad
        a = project_simplex(a_new)
        it += 1
        # occasional random shake if no improvement for some time
        # we could maintain time of last improvement
    return best_local, best_local_val

def local_refine(a, n, deadline):
    best_a = a.copy()
    best_val = 2*n * full_convolution(a).max()
    while time.time() < deadline:
        i, j = np.random.choice(n, 2, replace=False)
        if a[i] <= 0: continue
        eps_move = min(a[i], 0.01) * np.random.rand() * 0.5
        a_new = best_a.copy()
        a_new[i] -= eps_move
        a_new[j] += eps_move
        a_new = project_simplex(a_new)  # sum stable but ensure non-neg
        val_new = 2*n * full_convolution(a_new).max()
        if val_new < best_val:
            best_val = val_new
            best_a = a_new
    return best_a, best_val
```

"""
Hybrid optimization for the *max‐convolution* objective

We minimise
    F(a) = 2·n·max(conv(a,a)) / (sum(a)**2)
on the probability simplex (a_i ≥ 0, Σ a_i = 1).  The routine works in three
stages for a given length *n*:

1.  **Adam on a softmax parametrisation** – a smooth log‑sum‑exp
    approximation of the max is used so that a true gradient is available.
    The temperature τ is annealed from ~0.2 down to 1e‑7.

2.  **Exact sub‑gradient refinement** – once the iterates are close to a
    local optimum we switch to the true sub‑gradient of the max (weights are
    placed on the currently maximal convolution entries).  After each step the
    iterate is projected back onto the simplex.

3.  **Cheap mass‑shuffling hill‑climb** – a few thousand random moves of a
    small amount of mass between two coordinates are accepted only if they
    improve the objective.

The optimiser is run on a schedule of increasing problem sizes
(256 → 512 → 1024 → 2048 → 4096).  The best sequence seen so far is always kept
and warm‑started when a larger size is tackled (linear interpolation of the
previous result).  The whole search respects the 1000 s wall‑time limit and
returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------------- #
# Helper FFT based convolutions
# ----------------------------------------------------------------------------- #

def fft_len_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << int(np.ceil(np.log2(x)))


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = fft_len_pow2(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def smooth_max_and_weights(conv: np.ndarray, tau: float):
    """
    Temperature‑smoothed max of ``conv`` and the corresponding softmax
    weights (the derivative of the smooth max w.r.t. the entries).
    """
    c_max = conv.max()
    shifted = (conv - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    smooth = c_max + tau * np.log(sum_exp)
    weights = exp_shifted / sum_exp
    return smooth, weights


# ----------------------------------------------------------------------------- #
# Simplex projection (Euclidean projection onto {x ≥ 0, Σx = 1})
# ----------------------------------------------------------------------------- #

def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector `v` onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.shape[0]

    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------------- #
# Small utilities
# ----------------------------------------------------------------------------- #

def interpolate(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a probability vector `a` to length `new_n`.
    The result is re‑normalised to sum to one.
    """
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    t_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    t_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    a_new = np.interp(t_new, t_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-12:
        a_new = np.random.dirichlet(np.ones(new_n))
    else:
        a_new /= s
    return a_new


def objective(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (Σa = 1):
        F(a) = 2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------------- #
# Optimisation primitives
# ----------------------------------------------------------------------------- #

def adam_optimize(a0: np.ndarray, n: int, deadline: float,
                  max_iters: int = None) -> tuple:
    """
    Adam optimiser on the log‑softmax parametrisation with a smoothed max.
    Returns the best vector found and its objective value.
    """
    # FFT lengths
    fft_len_conv = fft_len_pow2(2 * n - 1)
    fft_len_corr = fft_len_pow2(3 * n)

    # Initialise logits
    a0 = a0 / a0.sum()
    x = np.log(a0 + 1e-15)

    # Adam state
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.2
    decay_every = 6000

    # iteration limits
    if max_iters is None:
        if n <= 512:
            max_iters = 250_000
        elif n <= 1024:
            max_iters = 180_000
        elif n <= 2048:
            max_iters = 120_000
        else:
            max_iters = 100_000

    best_a = None
    best_val = float("inf")
    last_improve = time.time()
    tau_initial = 0.2
    tau_final = 1e-7

    for t in range(max_iters):
        if time.time() > deadline:
            break

        # ----- soft‑max → probability vector -----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ----- convolution and smooth max -----
        conv = full_convolution(a, fft_len=fft_len_conv)
        tau = max(tau_final, tau_initial * (1.0 - t / max_iters))
        _, weights = smooth_max_and_weights(conv, tau)

        # ----- true objective (for bookkeeping) -----
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={t:05d}] new best {cur_val:.6f}")

        # ----- gradient of smooth max w.r.t. a -----
        grad_a = 2.0 * cross_correlation(weights, a, fft_len=fft_len_corr)

        # back‑prop through soft‑max
        inner = np.dot(a, grad_a)
        grad_x = a * (grad_a - inner)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** (t + 1))
        v_hat = v / (1.0 - beta2 ** (t + 1))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # learning‑rate decay
        if (t + 1) % decay_every == 0:
            lr *= 0.7

        # ----- stagnation handling -----
        if time.time() - last_improve > 30.0:
            x += (np.random.rand(n) - 0.5) * 4e-3
            last_improve = time.time()
            print(f"[Adam n={n} iter={t}] shake (stagnation)")

    # If we never improved, return the last a
    if best_a is None:
        best_a = a
        best_val = 2.0 * n * full_convolution(best_a, fft_len=fft_len_conv).max()
    return best_a, best_val


def subgradient_optimize(a0: np.ndarray, n: int, deadline: float,
                         max_iters: int = None) -> tuple:
    """
    Exact sub‑gradient descent on the true max (no smoothing).
    Projection onto the simplex is performed after each update.
    Returns the best vector encountered and its objective value.
    """
    fft_len_corr = fft_len_pow2(3 * n)

    a = a0 / a0.sum()
    best_a = a.copy()
    best_val = objective(a)
    last_improve = time.time()

    if max_iters is None:
        max_iters = 120_000 if n <= 1024 else 80_000

    for t in range(max_iters):
        if time.time() > deadline:
            break

        conv = full_convolution(a)
        cur_max = conv.max()
        cur_val = 2.0 * n * cur_max
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} iter={t:05d}] new best {cur_val:.6f}")

        # weights on all maximal indices (sub‑gradient of max)
        mask = conv >= cur_max - 1e-12
        w = mask.astype(float)
        w /= w.sum()                      # distribution over the active set

        # gradient of the weighted max
        grad = 2.0 * cross_correlation(w, a, fft_len=fft_len_corr)

        # step size (diminishing)
        lr = 0.15 / np.sqrt(t + 1)

        # simple momentum can help (heavy‑ball style)
        if t == 0:
            mom = np.zeros_like(a)
        mom = 0.7 * mom + (1.0 - 0.7) * grad

        a_new = a - lr * mom
        a = project_simplex(a_new)

        # occasional random jitter to escape plateaus
        if time.time() - last_improve > 40.0:
            jitter = (np.random.rand(n) - 0.5) * 5e-3
            a = project_simplex(a + jitter)
            last_improve = time.time()
            print(f"[Subgrad n={n} iter={t}] shake (stagnation)")

    return best_a, best_val


def local_refine(a0: np.ndarray, n: int, deadline: float,
                 max_moves: int = 3000) -> tuple:
    """
    Very cheap hill‑climbing: move a tiny amount of mass between two
    randomly chosen coordinates and keep the move if it improves the
    objective.  The procedure stops when the time budget expires or when
    `max_moves` attempts have been made.
    """
    a = a0.copy()
    best_val = objective(a)

    for _ in range(max_moves):
        if time.time() > deadline:
            break
        i, j = np.random.choice(n, 2, replace=False)
        if a[i] <= 0.0:
            continue
        # move a fraction of a[i] (between 20% and 80% of a[i] but never >0.01)
        step = min(a[i], 0.01) * (0.2 + 0.6 * np.random.rand())
        a_candidate = a.copy()
        a_candidate[i] -= step
        a_candidate[j] += step
        a_candidate = project_simplex(a_candidate)   # keep on the simplex
        val = objective(a_candidate)
        if val < best_val - 1e-12:
            best_val = val
            a = a_candidate
            # print only occasionally to avoid flood
            print(f"[Local refine n={n}] improved to {val:.6f}")

    return a, best_val

# Main entry point

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine works on several problem sizes, uses a warm‑start
    from previously discovered best vectors (or from the global variable
    `height_sequence_1` if present) and returns the best sequence found
    before the 1000 s wall‑time limit expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # Problem sizes – we start small (fast) and then go larger.
    sizes = [256, 512, 1024, 2048, 4096]

    for n in sizes:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # ----- initial vector -------------------------------------------------
        if best_seq is not None and best_seq.shape[0] == n:
            a0 = best_seq.copy()
        elif best_seq is not None and best_seq.shape[0] < n:
            a0 = interpolate(best_seq, n)
        elif 'height_sequence_1' in globals():
            cand = np.array(height_sequence_1, dtype=np.float64)
            if cand.shape[0] >= n:
                a0 = cand[:n]
            else:
                a0 = np.pad(cand, (0, n - cand.shape[0]))
            a0 = np.maximum(a0, 0.0)
            if a0.sum() < 0.01:
                a0 = np.random.dirichlet(np.ones(n))
            else:
                a0 = a0 / a0.sum()
        else:
            a0 = np.random.dirichlet(np.ones(n))

        # ----- Adam warm‑up ----------------------------------------------------
        a_adam, val_adam = adam_optimize(a0, n, deadline)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Global] new best after Adam (n={n}) : {val_adam:.6f}")

        # ----- Exact sub‑gradient refinement -----------------------------------
        a_sub, val_sub = subgradient_optimize(best_seq, n, deadline)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Global] new best after sub‑gradient (n={n}) : {val_sub:.6f}")

        # ----- Local greedy mass shuffling ------------------------------------
        a_loc, val_loc = local_refine(best_seq, n, deadline)
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Global] new best after local refine (n={n}) : {val_loc:.6f}")

    # -------------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector (objective = 2)
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
