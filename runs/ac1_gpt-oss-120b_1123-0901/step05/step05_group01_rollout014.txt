"""
Propose a low‑value step‑function sequence.

The search consists of three stages:

1. **Diverse warm‑starts** – for every chosen length we create many
   analytically‑generated shapes (power‑law edges, inverse‑root,
   edge Gaussians, random Fourier‑series, sparse support…) and a few
   pure Dirichlets.  Every candidate is normalised to sum 1 and evaluated
   with a fast FFT‑based convolution.

2. **Refinement** – the most promising candidates (5 per length) are
   refined with three complementary optimisers:
      * Adam on a smooth‑max surrogate of the maximum convolution
        (gradients obtained via a cross‑correlation).
      * Exact sub‑gradient (exponentiated‑gradient) on the true max.
      * Cheap hill‑climbing that swaps a tiny amount of mass between
        two positions.
   All three use the same FFT lengths that are pre‑computed for speed.

3. **Late‑time annealing** – the current global best is repeatedly
   jittered (≤1 % random perturbation) and re‑optimised with Adam.
   This consumes any remaining time and often squeezes a few extra
   percent out of the objective.

The whole routine respects a 1000 s budget, constantly checking the
deadline.  The returned list contains non‑negative floats and is ready
to be fed to ``evaluate_sequence``.
"""

import time

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)−1) via FFT of size *fft_len*."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) (the usual gradient of a⋆a w.r.t. a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact evaluation of the objective 2·n·max(a*a) / (sum a)²."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    # the caller guarantees s > 0 and a ≥ 0
    return 2.0 * n * max_c / (s * s)

    lam_factor = 1.04  # warm‑up speed

    best_a = a.copy()
    best_val = _objective(a, n, fft_len_conv)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max projection
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # smooth‑max weights for the current λ
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (see derivation in the description)
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through the soft‑max (∇_log a)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on the logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # increase temperature
        lam = min(lam * lam_factor, 1e7)

        cur_val = _objective(a, n, fft_len_conv)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t:06d}] new best {cur_val:.6f}")

        # occasional random shake if we stalled
        if time.time() - last_improve > 3.5:
            logits += (np.random.rand(n) - 0.5) * 5e-3
            last_improve = time.time()

    return best_a, best_val


def _subgrad_optimize(a0: np.ndarray,
                      n: int,
                      fft_len_conv: int,
                      fft_len_corr: int,
                      deadline: float,
                      init_val: float | None = None,
                      max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient (exponentiated‑gradient) optimisation on the true max.
    """
    a = a0 / a0.sum()
    best_a = a.copy()
    best_val = _objective(a, n, fft_len_conv) if init_val is None else init_val
    last_improve = time.time()
    eta = 0.6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        # uniform weight on all active indices
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        cur_val = _objective(a, n, fft_len_conv)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it}] new best {cur_val:.6f}")

        # stop early after a few seconds of stagnation
        if time.time() - last_improve > 4.0:
            break

    return best_a, best_val


def _hill_climb_optimize(a0: np.ndarray,
                         n: int,
                         fft_len_conv: int,
                         deadline: float,
                         init_val: float | None = None,
                         max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Simple cheap local search: move a tiny amount of mass between two
    random coordinates and keep the move if it improves the objective.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_conv) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2:
            continue

        # transfer up to 5 % of the mass from i1 to i2
        eps = np.random.rand() * 0.05 * a[i1]
        if eps <= 0.0:
            continue
        new_a = a.copy()
        new_a[i1] -= eps
        new_a[i2] += eps
        if new_a[i1] < -1e-15:
            continue
        new_a /= new_a.sum()

        new_val = _objective(new_a, n, fft_len_conv)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            if it % 50 == 0:
                print(f"[n={n} HillClimb it={it}] new best {new_val:.6f}")
        else:
            # occasional uphill move to escape tiny plateaus
            if np.random.rand() < 0.001:
                a = new_a

    return best_a, best_val


# ----------------------------------------------------------------------
def _make_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Produce a diverse set of start‑up shapes for length *n*."""
    cand = []

    # 1) pure Dirichlet (uniformly random)
    cand.append(rng.dirichlet(np.ones(n)))

    # 2) edge‑concentrated power‑law ( (i+1)*(n-i) )**α , α negative
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in rng.uniform(-5.0, -0.2, size=5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # 3) inverse‑root edge shapes (i+1)**‑α + (n‑i)**‑α
    for alpha in rng.uniform(0.4, 2.0, size=5):
        i = np.arange(n, dtype=np.float64) + 1.0
        shape = np.power(i, -alpha) + np.power(n - i + 1.0, -alpha)
        cand.append(shape / shape.sum())

    # 4) two edge Gaussians
    for gamma in rng.uniform(0.1, 2.0, size=4):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        cand.append(shape / shape.sum())

    # 5) random Fourier‑series exponentials (always positive)
    K = max(3, n // 15)          # a handful of frequencies
    for _ in range(4):
        freqs = rng.integers(1, K + 1, size=K)
        coeff_cos = rng.normal(scale=0.7, size=K)
        coeff_sin = rng.normal(scale=0.7, size=K)
        x = np.linspace(0.0, 1.0, n, endpoint=False)
        s = np.zeros_like(x)
        for f, c, d in zip(freqs, coeff_cos, coeff_sin):
            s += c * np.cos(2 * np.pi * f * x) + d * np.sin(2 * np.pi * f * x)
        shape = np.exp(s - np.max(s))        # shift for numerical stability
        cand.append(shape / shape.sum())

    # 6) sparse support (few non‑zero positions)
    for _ in range(3):
        k = rng.integers(int(0.01 * n), int(0.15 * n) + 1)
        idxs = rng.choice(n, size=k, replace=False)
        vals = rng.random(k) + 0.1               # avoid exact zeros
        shape = np.zeros(n)
        shape[idxs] = vals
        cand.append(shape / shape.sum())

    rng.shuffle(cand)
    # keep a reasonably sized pool
    return cand[:30]

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function sequence that minimises
    `evaluate_sequence`.  The routine respects the given time budget and
    returns the best sequence discovered as a Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    best_seq = None
    best_val = float("inf")

    # lengths to explore – start modest, finish with the largest feasible.
    lengths = [2048, 4096, 8192, 16384, 32768]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT lengths (next power of two) – pre‑computed once per n
        fft_len_conv = _next_pow2(2 * n - 1)
        fft_len_corr = _next_pow2(3 * n)          # safe for correlation

        # --------------------------------------------------------------
        # Stage 1 – generate many cheap candidates and keep the top few
        raw_cands = _make_candidates(n, rng)

        scored = []
        for cand in raw_cands:
            if time.time() > deadline:
                break
            val = _objective(cand, n, fft_len_conv)
            scored.append((val, cand))
        scored.sort(key=lambda x: x[0])
        top_cands = [c for _, c in scored[:5]]    # keep the 5 best for deep work

        # --------------------------------------------------------------
        # Stage 2 – deep refinement of the shortlist
        for idx, a_start in enumerate(top_cands):
            if time.time() > deadline:
                break

            # smooth‑max Adam
            max_adam_iters = int(18000 * 1024 / n) + 3000
            a_adam, val_adam = _adam_optimize(
                a_start, n, fft_len_conv, fft_len_corr, deadline, max_adam_iters
            )

            # exact sub‑gradient
            a_sub, val_sub = _subgrad_optimize(
                a_adam, n, fft_len_conv, fft_len_corr, deadline,
                init_val=val_adam, max_iters=2500
            )

            # cheap hill‑climb
            a_hc, val_hc = _hill_climb_optimize(
                a_sub, n, fft_len_conv, deadline,
                init_val=val_sub, max_iters=2500
            )

            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.copy()
                print(f"*** Global best updated: n={n} cand={idx} val={best_val:.6f}")

        # --------------------------------------------------------------
        # Stage 3 – late‑time annealing on the current global best
        if best_seq is not None and time.time() < deadline:
            # number of extra annealing rounds depends on remaining budget
            while time.time() < deadline - 5.0:     # keep a few seconds margin
                # small random perturbation (≤1 % per coordinate)
                pert = best_seq * (1.0 + 0.01 * (rng.random(n) - 0.5))
                pert = np.maximum(pert, 0.0)
                pert /= pert.sum()

                # a modest Adam run to settle the perturbation
                a_ref, val_ref = _adam_optimize(
                    pert, n, fft_len_conv, fft_len_corr, deadline,
                    max_iters=int(5000 * 1024 / n) + 800
                )
                if val_ref < best_val:
                    best_val = val_ref
                    best_seq = a_ref.copy()
                    print(f"*** Annealing improvement: val={best_val:.6f}")
                else:
                    # occasional acceptance of a slightly worse move – simulated annealing style
                    if rng.random() < 0.03:
                        best_seq = a_ref.copy()
                        best_val = val_ref

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded (highly unlikely)
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = _objective(best_seq, n_fallback, _next_pow2(2 * n_fallback - 1))
        print("Fallback uniform sequence used")

    # Return plain Python list
    return best_seq.tolist()
