"""
Propose a non‑negative step‑function minimizing

    F(a) = 2·n·max(a∗a) / (sum a)²

The algorithm follows a multi‑stage scheme:

1.  Progressive lengths: start from a very small sequence, find a good shape,
    then interpolate it to larger lengths (256 → … → 16384).
2.  A large, diverse set of warm‑starts:
    - uniform, Dirichlet (α∈{0.1,0.5,1,2}),
    - Beta‑distribution shapes,
    - power‑law edge‑heavy families,
    - Gaussian edge / interior bumps,
    - random piecewise‑linear profiles.
3.  For each warm‑start run three refinements:
    * Smooth‑max Adam (softmax surrogate of the max, temperature ↑).
    * Exact sub‑gradient with exponentiated‑gradient updates (mirror‑descent).
    * Cheap stochastic hill‑climb that moves a small amount of mass between two
      random coordinates.
4.  Keep the globally best sequence (lowest objective) and, if time permits,
    perform a long‑run Adam refinement on that sequence.

All heavy lifting (convolution, cross‑correlation) is done with FFTs
so that even n=16384 is fast (≈5·10⁵ flops per evaluation).  The code
honours the 1000 s budget and prints progress.
"""

# FFT‑based convolution utilities
def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)−1) using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) = n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)² (sum a is assumed >0)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    return 2.0 * n * max_c / (s * s)

# Normalisation helpers
def _normalise(a: np.ndarray) -> np.ndarray:
    """Scale a to have non‑negative entries and unit sum."""
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total < 1e-15:
        a = np.random.dirichlet(np.ones_like(a))
        total = a.sum()
    return a / total


def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence by linear interpolation and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    return _normalise(new)

def _generate_initial_candidates(n: int,
                                 best_seq: np.ndarray | None,
                                 hist_seq: np.ndarray | None) -> list[np.ndarray]:

    cand = []

    # 1. warm‑starts from previously found solutions
    if best_seq is not None:

        cand.append(_normalise(tmp))

    cand.append(np.full(n, 1.0 / n))                     # uniform

        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 3. Beta‑distribution shaped families (edge‑heavy or centre‑heavy)
    x = np.linspace(0.0, 1.0, n)
    for a_param, b_param in [(0.3, 0.3), (0.5, 1.5), (1.5, 0.5), (0.7, 0.7), (0.2, 2.0)]:
        shape = (x ** (a_param - 1)) * ((1 - x) ** (b_param - 1))
        if shape.sum() > 0:
            cand.append(_normalise(shape))

    # 4. Power‑law edge shape  (i+1)*(n-i) ** alpha with negative alpha
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 6):
        shape = np.power(prod, alpha, where=prod > 0,
                         out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(_normalise(shape))

    # 5. Two‑sided exponential decay from the borders
    for lam in (0.002, 0.005, 0.01, 0.02):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        cand.append(_normalise(shape))

    # 6. One or several Gaussian bumps (edges, centre, three‑peak)
    for sigma in (0.05, 0.10, 0.20):
        i = np.arange(n, dtype=np.float64) / (n - 1)
        # edge bumps
        shape = np.exp(-0.5 * ((i) / sigma) ** 2) + np.exp(-0.5 * ((1 - i) / sigma) ** 2)
        cand.append(_normalise(shape))
        # centre bump
        shape = np.exp(-0.5 * ((i - 0.5) / sigma) ** 2)
        cand.append(_normalise(shape))
        # three‑peak bump
        peaks = sum(np.exp(-0.5 * ((i - p) / sigma) ** 2) for p in (0.25, 0.5, 0.75))
        cand.append(_normalise(peaks))

    # 7. Random piecewise‑linear profiles
    rng = np.random.default_rng()
    for _ in range(4):
        k = rng.integers(3, 8)          # number of breakpoints
        xs = np.sort(rng.random(k) * (n - 1))
        ys = rng.random(k)            # heights in [0,1)
        # enforce endpoints = 0 to keep shape edge‑light if desired
        xs = np.concatenate(([0], xs, [n - 1]))
        ys = np.concatenate(([0.0], ys, [0.0]))
        new = np.interp(np.arange(n), xs, ys)
        cand.append(_normalise(new))

    # 8. Very sparse Dirichlet (small concentration → few large entries)
    for conc in (0.01, 0.05, 0.1):
        cand.append(_normalise(np.random.dirichlet(np.full(n, conc))))

    # Shuffle and keep a moderate number (≈30) for speed
    rng.shuffle(cand)
    return cand[:30]

def _adam_smooth_max(a0: np.ndarray,

            w /= w_sum

        # Gradient of the smooth objective (before simplex)
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # Back‑prop through soft‑max (mirror‑gradient)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule – slowly make the surrogate sharper
        lam = min(lam * lam_factor, 1e6)

        cur_val = _objective(a, n, fft_len_conv)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t:06d}] new best {cur_val:.6f}")

        # occasional random jitter if stuck
        if time.time() - last_improve > 3.5:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


def _expgrad_subgrad(a0: np.ndarray,
                    n: int,
                    fft_len_conv: int,
                    fft_len_corr: int,
                    deadline: float,
                    init_val: float | None = None,
                    max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient (exponentiated‑gradient / mirror descent) on the true max.
    """
    a = _normalise(a0)
    best_a = a.copy()
    if init_val is None:
        best_val = _objective(a, n, fft_len_conv)
    else:
        best_val = init_val

    eta = 0.6           # base step size – empirically stable
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        # weight vector for active constraints (uniform over them)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # sub‑gradient of max(conv) wrt a
        grad = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient update (keeps non‑negative)
        a = a * np.exp(-eta * grad)

        # renormalise to simplex
        a_sum = a.sum()
        if a_sum < 1e-15:                     # numerical safety
            a = np.random.dirichlet(np.ones(n))
        else:
            a /= a_sum

        cur_val = _objective(a, n, fft_len_conv)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} ExpGrad it={it}] new best {cur_val:.6f}")

        # stop early if no improvement for a while
        if time.time() - last_improve > 4.0:
            break

        # optional step‑size decay
        if it % 500 == 0:
            eta *= 0.85

    return best_a, best_val


def _hill_climb(a0: np.ndarray,
                n: int,
                fft_len_conv: int,
                deadline: float,
                init_val: float | None = None,
                max_iters: int = 2500) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: move a small amount of mass between
    two coordinates and keep the change if it improves the objective.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_conv) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val
    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # transfer up to 4 % of mass from i to j
        eps = rng.random() * 0.04 * a[i]
        if eps <= 0.0:
            continue
        new_a = a.copy()
        new_a[i] -= eps
        new_a[j] += eps
        if new_a[i] < -1e-15:
            continue
        new_a /= new_a.sum()

        new_val = _objective(new_a, n, fft_len_conv)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            if it % 70 == 0:
                print(f"[n={n} HillClimb it={it}] new best {new_val:.6f}")
        else:
            # occasional uphill move to avoid tiny plateaus
            if rng.random() < 0.001:
                a = new_a

    return best_a, best_val

# Main entry point
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of floats) with a low
    evaluation score.  The function respects the given time budget and
    returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    best_seq = None
    best_val = float("inf")

    # Attempt a progressive ladder of lengths – finishing with the largest.
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]

    # Pre‑compute FFT lengths (next power of two) for each n.
    fft_conv = {}
    fft_corr = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_conv[n] = 1 << int(np.ceil(np.log2(conv_len)))
        # correlation needs conv_len + n - 1 ≤ 3·n, which is always true:
        corr_len = conv_len + n - 1
        fft_corr[n] = 1 << int(np.ceil(np.log2(corr_len)))

    # Load historic solution if present.
    hist_seq = None
    if "height_sequence_1" in globals():
        tmp = globals()["height_sequence_1"]
        if isinstance(tmp, (list, tuple, np.ndarray)):
            hist_seq = np.asarray(tmp, dtype=np.float64)

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_fft = fft_conv[n]
        corr_fft = fft_corr[n]

        # generate a pool of diverse starts for this length
        candidates = _generate_initial_candidates(n, best_seq, hist_seq)

        # schedule: smaller n gets many iterations, larger n fewer (still enough)
        max_adam_iters = int(20000 * 256 / n) + 1500   # roughly ~20k it at n=256

        for cand_id, cand in enumerate(candidates):
            if time.time() > deadline:
                break

            # 1. Adam on smooth‑max surrogate
            a_adam, val_adam = _adam_smooth_max(
                cand, n, conv_fft, corr_fft, deadline, max_adam_iters)

            # 2. Exact sub‑gradient via exponentiated‑gradient
            a_sub, val_sub = _expgrad_subgrad(
                a_adam, n, conv_fft, corr_fft, deadline, init_val=val_adam)

            # 3. Cheap hill‑climb local refinement
            a_hc, val_hc = _hill_climb(
                a_sub, n, conv_fft, deadline, init_val=val_sub)

            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.copy()
                print(f"*** Global new best (n={n} cand={cand_id}) value={best_val:.6f}")

    # ------------------------------------------------------------------
    # Final long‑run refinement on the global best, if time permits.
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        conv_fft = fft_conv.get(n, 1 << int(np.ceil(np.log2(2 * n - 1))))
        corr_fft = fft_corr.get(n, 1 << int(np.ceil(np.log2(3 * n))))
        print("\n=== Long‑run refinement of the global best ===")
        refined_seq, refined_val = _adam_smooth_max(
            best_seq, n, conv_fft, corr_fft, deadline,
            max_iters=int(1.5 * (20000 * 256 / n) + 3000))
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq.copy()
            print(f"*** Refined global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback (should never happen)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return as plain Python list, as expected by the external evaluator.
    return best_seq.tolist()
