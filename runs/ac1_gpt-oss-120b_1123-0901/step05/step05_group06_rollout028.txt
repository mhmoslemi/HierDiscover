    cur_score = compute_score(cur)

    Nfft = 1 << ((2*n-1).bit_length())
    while time.time() < deadline:
        # random move type: transfer or swap
        i,j = rng.integers(0,n, size=2)
        if i==j: continue
        # decide move: transfer fraction of mass from i to j or swap
        if rng.random() < 0.5:
            # transfer move
            if cur[i] < 1e-12: continue

    deadline = start + budget_s - 5

    best_seq = None
    best_score = float('inf')
    best_n = None

    # Use previously discovered base if defined.
    base = None
    try:
        base = np.array(height_sequence_1, dtype=float)
    except Exception:
        pass

    # Choose lengths
    lengths = [500, 1000, 1500, 2100, 3000, 4000, 6000, 8000]  # ensures diverse

    # allocate time per length: maybe equal split.
    # We'll compute per-length budget as remaining / len(lengths)
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        # build init list
        init_vectors = _make_initial_vectors(n, target_sum, rng)
        if base is not None:
            # interpolate base to n
            base_interp = np.interp(np.linspace(0,1, n), np.linspace(0,1, base.shape[0]), base)
            base_interp = np.maximum(base_interp,0)
            base_interp = base_interp / base_interp.sum() * target_sum
            init_vectors.append(base_interp)

        # compute per-length budget: allocate remaining time slice
        remaining = deadline - time.time()
        per_len_budget = remaining * 0.2  # allocate up to 20% for this length
        len_deadline = time.time() + per_len_budget
        # for each init, run refinement
        for idx, vec in enumerate(init_vectors):
            if time.time() >= len_deadline:
                break
            # ensure vec sum = target_sum
            vec = np.maximum(vec, 0.0)
            vec = vec / vec.sum() * target_sum

            seq, score = local_refine_fast(vec, n, len_deadline, rng)
            if score < best_score:
                best_seq = seq.copy()
                best_score = score
                best_n = n
                print(f"[new global best] len {n} from init {idx} score {best_score:.6f}")

    # maybe additional refine on best overall with remaining time
    if best_seq is not None and time.time() < deadline:

        # fallback uniform
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        print("[fallback] uniform")
    return best_seq.tolist()

def _incremental_refine(a, n, deadline, rng):
    target_sum = a.sum()
    L = 2 * n - 1
    conv = np.convolve(a, a)  # shape L
    cur_max = conv.max()
    start = time.time()
    total_time = deadline - start
    step = 0
    while time.time() < deadline:
        # Determine decay factor for delta fraction
        progress = (time.time() - start) / total_time if total_time > 0 else 1.0
        max_frac = 0.5 * (1 - progress) + 0.02  # decays from 0.5 to 0.02
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

            new_max = conv_swapped.max()

                conv = conv_swapped
                cur_max = new_max

        # Transfer
        if a[i] < 1e-12:
            continue

"""
Fast stochastic hill‑climbing for the max‑self‑convolution problem.

The evaluation function is homogeneous:
    score = 2·n·max(a∗a) / (sum(a)²)
If we keep sum(a) = √(2 n) the denominator equals the numerator
and the score is simply max(a∗a).  Therefore the algorithm works on a
probability vector p (p_i ≥ 0, Σp_i = 1) and maintains the total mass
explicitly – a cheap O(n) update of the full convolution is used after
every elementary move.

* Initialise a small diverse pool of lengths (500 … 8000) and several
  simple starting shapes (uniform, Dirichlet, exponential, etc.).
* For each start run a long simulated‑annealing / hill‑climbing loop:
    – choose a source index i and a destination j,
    – transfer a random fraction of the source mass to the destination,
    – update the linear convolution in‑place with two slice operations
      plus the three quadratic correction terms (derived from
      (a+Δ)*(a+Δ) = a*a + 2·Δ*a + Δ*Δ),
    – accept every improving move; accept worsening moves with a
      temperature that decays linearly with the elapsed time.
    – every few thousand steps re‑normalise the vector to avoid drift.
* Keep the globally best sequence and finish with a short extra
  refinement on the remaining budget.
All operations are vectorised NumPy code, so millions of inexpensive
iterations fit easily into the 1000 s budget.
"""
import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Helper: create a few diverse start vectors for a given length
def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Γ(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    exp_vec = np.exp(-rate * np.arange(n))
    init.append(exp_vec / exp_vec.sum() * target_sum)

    # 4) reversed exponential
    rev = exp_vec[::-1]
    init.append(rev / rev.sum() * target_sum)

    # 5) Hann window (smooth bump – gives a nice baseline)
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) small random perturbation of uniform
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.02 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    return init


# ----------------------------------------------------------------------
# Fast incremental refinement
def _incremental_refine(a0: np.ndarray,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator):
    """
    Stochastic hill‑climbing with an O(n) convolution update.
    Returns the improved vector and its exact objective value
    (max convolution, which equals the evaluate_sequence score when
    Σa = √(2 n)).
    """
    target_sum = a0.sum()                     # should be √(2·n) but we keep it exact
    a = a0.copy()
    L = 2 * n - 1
    conv = np.convolve(a, a)                  # linear convolution, length L
    cur_max = conv.max()
    start = time.time()
    total_time = max(deadline - start, 1e-9)

    steps = 0
    while time.time() < deadline:
        # linear schedule for the maximum transferred fraction
        progress = (time.time() - start) / total_time
        max_frac = 0.5 * (1.0 - progress) + 0.02   # from 0.5 → 0.02

        # pick source / destination
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # occasional pure swap (helps escaping flat regions)
        if rng.random() < 0.07:
            if a[i] == a[j]:
                continue
            a_sw = a.copy()
            a_sw[i], a_sw[j] = a_sw[j], a_sw[i]
            conv_sw = np.convolve(a_sw, a_sw)
            new_max = conv_sw.max()
            if new_max < cur_max:
                a = a_sw
                conv = conv_sw
                cur_max = new_max
            continue

        # ---- mass transfer move ----
        if a[i] < 1e-12:
            continue
        delta = a[i] * rng.uniform(0.01, max_frac)   # transfer a fraction of the source mass

        # incremental convolution update (copy, then slice ops)
        conv_new = conv.copy()
        # linear part: 2·Δ·a added/subtracted at the two shifted windows
        conv_new[i:i + n] -= 2.0 * delta * a
        conv_new[j:j + n] += 2.0 * delta * a
        # quadratic correction terms (derived analytically)
        conv_new[i + j] -= 2.0 * delta * delta
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta

        new_max = conv_new.max()

        # ---- acceptance ----
        if new_max + 1e-12 < cur_max:          # strict improvement
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            cur_max = new_max
        else:
            # simulated‑annealing acceptance with a linearly decaying temperature
            T = cur_max * 0.02 * max(0.0, 1.0 - progress) + 1e-9
            prob = math.exp(-(new_max - cur_max) / T)
            if rng.random() < prob:
                a[i] -= delta
                a[j] += delta
                conv = conv_new
                cur_max = new_max

        steps += 1

        # re‑normalise occasionally to keep the sum exact (prevents drift)
        if steps % 5000 == 0:
            sum_err = a.sum() - target_sum
            if abs(sum_err) > 1e-12:
                a *= target_sum / a.sum()
                conv = np.convolve(a, a)
                cur_max = conv.max()

    return a, cur_max
