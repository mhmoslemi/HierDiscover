            shift = np.random.randint(0, n - L + 1)
            a = np.roll(a, shift)

"""
Propose a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in three phases:

1. **Random sparse constructions** –  
   Pick a length *n* and a small number *k* of active
   positions (k ≈ α·√n).  Put the same weight on the *k*
   positions so that the total ℓ1‑norm is √(2n).  Because
   only O(√n) entries are non‑zero the number of ordered
   pairs that contribute to any convolution index is O(1),
   which drives the maximum convolution towards the theoretical
   lower bound 1.

2. **Simple hill‑climbing** –  
   Starting from the best sparse vector, repeatedly move a
   random amount of mass from one index to another (the move
   respects the simplex constraint).  Accept every improving
   move, otherwise accept with a temperature‑controlled
   Metropolis probability.  This refines the candidate without
   any gradients.

3. **Fallback** – a uniform vector is returned only if nothing
   else was found (should never happen).

All heavy lifting (full linear convolution) is performed with a
real FFT.  The routine respects the supplied time budget
(`budget_s`).  The returned list is a plain Python list so the
outside evaluator can call `evaluate_sequence` on it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # tiny negative values appear because of rounding – clip them
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Project a vector onto the simplex {w ≥ 0, Σw = s}
# (Duchi et al. 2008) – used by the hill‑climber to keep the sum
# fixed after a jitter.
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending sort
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)           # 1‑based indices
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact evaluation of the objective (identical to evaluate_sequence)
# ----------------------------------------------------------------------
def _score_of(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq))/sum(seq)² – the value the
    competition evaluator would produce."""
    n = seq.shape[0]
    conv = _full_convolution_fft(seq, seq)
    max_conv = conv.max()
    s = seq.sum()
    # the evaluator returns np.inf for sum < 0.01 – we never get there
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Random sparse construction generator
# ----------------------------------------------------------------------
def _random_sparse_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Build a candidate with O(√n) non‑zero entries.
    The number of active entries is k = round(alpha·√n) with a
    random factor α∈[1.3,2.2].
    All active entries share the same weight so that the ℓ1‑norm
    equals √(2·n).
    """
    target = np.sqrt(2.0 * n)

    # choose how many positions are non‑zero
    sqrt_n = np.sqrt(n)
    alpha = rng.uniform(1.3, 2.2)                 # controls sparsity
    k = max(2, int(round(alpha * sqrt_n)))
    k = min(k, n)                                # guard

    # random distinct indices
    idx = rng.choice(n, size=k, replace=False)

    a = np.zeros(n, dtype=np.float64)
    # uniform weight on the k positions
    a[idx] = target / k

    # with a small probability we perturb the weights a little
    if rng.rand() < 0.15:
        # Dirichlet distribution over the active coordinates
        w = rng.dirichlet(np.ones(k))
        a[idx] = w * target

    return a


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing (exact objective)
# ----------------------------------------------------------------------
def _hillclimb(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Local search that moves a small amount of mass between two
    coordinates while keeping the simplex constraint."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score_of(a)

    # temperature schedule – start relatively hot, cool to near zero
    T0 = best_val * 0.3
    T_end = 1e-4
    start = time.time()
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] <= 0:
            continue

        # move up to 20 % of the mass at i
        delta = a[i] * rng.random() * 0.20
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _score_of(a_new)

        if new_val < best_val - 1e-12:
            best_val = new_val
            best_a = a_new.copy()
            a = a_new
            print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                a = a_new

        # occasional re‑projection to avoid numerical drift
        if it % 2000 == 0:
            target = np.sqrt(2.0 * n)
            a = _project_onto_simplex(a, target)

        if best_val < 1.503:
            break

    return best_a, best_val

    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    best_seq = None
    best_score = np.inf

    if best_seq is not None and time.time() < deadline - 5.0:

        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")
