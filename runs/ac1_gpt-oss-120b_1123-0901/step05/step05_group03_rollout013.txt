def cross_corr_fft(w, a, fft_len):
    a_rev = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(a_rev, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)  # length fft_len
    # Compute slice: start = len(a) - 1, end = start + len(a)
    start = len(a) - 1
    return corr[start:start + len(a)]
```

"""
Propose a candidate step‑function (non‑negative sequence) that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)²)

The algorithm mixes three ideas:

1. **Entropy‑mirror Adam** on a smooth‑max surrogate (log‑sum‑exp) to obtain a
   good local optimum on the simplex.  The surrogate temperature is increased
   gradually so the optimizer ends up minimising the true max‑convolution.

2. **Explicit sub‑gradient phase** (exponentiated‑gradient updates) that works
   directly on the exact max‑convolution; this refines the Adam solution.

3. **Local random‑pair perturbations** (simulated‑annealing style) that move a
   tiny amount of mass between two coordinates.  This cheap hill‑climbing
   often pushes the objective a few‑hundredths lower – enough to cross the
   1.503 target.

The routine respects the given time budget (default 1000 s) and returns the
best sequence found as a plain Python list.  If a previous best solution is
available via the global variable `height_sequence_1` it is used as a warm
start; otherwise a diverse pool of random and handcrafted initialisations is
tried.  All heavy work uses FFT so that each objective evaluation is O(n log n).

"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    a_rev = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(a_rev, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _objective_val(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective 2·n·max(a∗a) / (sum(a)²) – sum is assumed ≈ 1."""
    conv = _full_convolution_fft(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate followed by an exponentiated‑gradient
    sub‑gradient phase.  The routine respects the global deadline and updates
    (global_best_seq, global_best_val) with the best sequence/value ever seen.
    """
    # ---- Adam on smoothed max ------------------------------------------------
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04                                     # a bit smaller than before

    lam = 0.5                                     # start temperature
    lam_factor = 1.03
    max_lam = 1e6

    # cross‑correlation length
    corr_fft_len = _next_pow2(3 * n - 2)

    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        # simplex point from logits
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution & exact objective (useful for logging)
        conv = _full_convolution_fft(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:05d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a  (chain rule through soft‑max)
        grad_a = 2.0 * _cross_correlation_fft(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule – slowly turn the surrogate into a true max
        lam = min(lam * lam_factor, max_lam)

        # occasional shake when stagnating
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        if it > 250_000:                     # safety bound
            break

    # ---- Sub‑gradient phase on the exact max -------------------------------
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()

    eta = 0.6
    sub_last_improve = time.time()
    for sub_it in range(12_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution_fft(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[Subgrad n={n} it={sub_it:05d}] new best {val:.6f}")

        # active set (indices where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation_fft(w, a, corr_fft_len)

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid numerical collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # stop if no progress for a while
        if time.time() - sub_last_improve > 5.0:
            break

    return global_best_seq, global_best_val


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of start vectors."""
    cand = []

    # warm start and simple rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # power‑law edge focussed (α < 0)
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # sinusoidal half‑wave
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # triangular (linear ramp up‑down)
    tri = np.minimum(np.arange(n) + 1, n - np.arange(n))
    tri = np.maximum(tri, 0.0)
    tri = tri.astype(np.float64)
    tri /= tri.sum()
    cand.append(tri)

    # arcsine‑type (∝ 1/√(x(1‑x))) – we truncate the singularities
    x = (np.arange(1, n + 1) - 0.5) / n
    arcs = 1.0 / np.sqrt(np.clip(x * (1 - x), 1e-12, None))
    arcs /= arcs.sum()
    cand.append(arcs)

    # uniform baseline
    cand.append(np.full(n, 1.0 / n))

    return cand


# ----------------------------------------------------------------------
def _local_refine(a: np.ndarray, deadline: float, cur_best: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a small amount of mass from one index to another.
    Accept only improvements (optional simulated‑annealing could be added).
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_val = cur_best
    it = 0
    while time.time() < deadline - 0.05:
        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # amount to shift – a modest fraction of the source mass
        eps = best_seq[i] * np.random.uniform(1e-4, 5e-3)
        if eps <= 0 or eps > best_seq[i]:
            continue
        cand = best_seq.copy()
        cand[i] -= eps
        cand[j] += eps
        # keep simplex (sum unchanged) and stay non‑negative
        if cand[i] < 0:
            continue
        val = _objective_val(cand, n, _next_pow2(2 * n - 1))
        if val < best_val:
            best_seq = cand
            best_val = val
            print(f"[local refine it={it:06d}] new best {best_val:.6f}")
            # reset stagnation counter
            it = 0
        else:
            it += 1
        # every few thousand tries check the time budget
        if it % 20000 == 0 and time.time() >= deadline - 0.05:
            break
    return best_seq, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a low‑objective step‑function under the
    given time budget.  Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --- primary length -----------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)

    # warm start from the global variable if present
    warm_start = globals().get("height_sequence_1")
    warm = None
    if warm_start is not None:
        warm = np.asarray(warm_start, dtype=np.float64)
        if warm.shape[0] != n:
            # simple linear interpolation to the required size
            x_old = np.linspace(0, 1, warm.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            warm = np.interp(x_new, x_old, warm)
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    candidates = _initial_candidates(n, warm)

    # add a few extra random Dirichlet seeds to increase diversity
    for _ in range(6):
        candidates.append(np.random.dirichlet(np.ones(n)))

    # ----------------------------------------------------------------------
    # optimise each candidate (the loop respects the global deadline)
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break

        a0 = np.maximum(cand, 0.0)
        s = a0.sum()
        if s < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / s

        best_seq, best_val = _deep_optimize(
            a0, n, conv_fft_len, deadline, best_seq, best_val
        )

        # early exit if we have already beaten the target
        if best_val < 1.503:
            print(f"Target already reached (val={best_val:.6f}) – stopping.")
            break

    # ----------------------------------------------------------------------
    # Final cheap local refinements on the best vector found so far
    if best_seq is not None and time.time() < deadline - 5.0:
        best_seq, best_val = _local_refine(best_seq, deadline, best_val)

    # ----------------------------------------------------------------------
    # If time permits, try a larger grid (8192) by linear interpolation.
    # Larger n often lowers the bound a little further.
    if time.time() + 5 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        if best_seq is not None:
            x_old = np.linspace(0, 1, best_seq.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        # a quick but still thorough optimisation on the larger grid
        best_seq_big, best_val_big = _deep_optimize(
            a_big, n_big, conv_fft_len_big, deadline, None, float("inf")
        )
        if best_val_big < best_val:
            best_seq, best_val = best_seq_big, best_val_big

    # ----------------------------------------------------------------------
    # Fallback – uniform vector if nothing else succeeded.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # enforce the evaluator's clipping (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # final sanity check (optional)
    # print(f"Final best value ≈ {best_val:.6f}")

    return best_seq.tolist()
