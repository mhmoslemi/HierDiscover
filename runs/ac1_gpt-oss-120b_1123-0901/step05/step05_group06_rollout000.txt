"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.
-----------------------------------------------------------------------

Algorithm Overview
------------------
* **Multi‑scale optimisation** – start from the best known construction,
  gradually increase the length (up to 8000) by interpolation.
* **Adam on the log‑its** – the vector is represented as a softmax of
  unconstrained logits, guaranteeing non‑negativity and the exact sum
  ``√(2 n)``.
* **Smooth max approximation** – the max of the autocorrelation
  ``a∗a`` is approximated by a temperature‑softened soft‑max.
  Its gradient is obtained by a weighted reverse convolution using FFTs.
* **Adaptive schedule** – the temperature decays exponentially,
  the learning rate follows Adam’s bias‑corrected schedule and
  occasional jitter is added to escape plateaux.
* **Diverse initialisations** – for each length a small pool of
  analytically‑motivated vectors (uniform, Dirichlet, exponential,
  cosine, Gaussian, edge‑biased) together with the interpolated
  previous best are optimised.
* **Final stochastic refinement** – after the global best has been
  discovered, a cheap simulated‑annealing hill‑climber performs many
  pairwise mass‑transfer moves to squeeze the objective a little more.

The code is deliberately written with only top‑level functions (no
closures) and respects the 1000 s time budget.  It returns a plain Python
list that can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, tau: float):
    """
    Compute a soft‑max weight vector for the entries of ``conv``.
    Returns (weights, smoothed value L) where

        L = tau * log( sum_i exp(conv_i / tau) )

    The implementation is numerically stable.
    """
    if tau <= 0.0:
        # exact max
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, conv.max()
    maxc = conv.max()
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    L = maxc + tau * np.log(sum_ex)
    return w, L


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute the weighted reverse convolution needed for the gradient.
    For each i, rev[i] = Σ_k w[k] * a[k - i] (out‑of‑range indices ignored).
    The implementation uses FFTs for O(n log n) speed.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    # reverse the weight vector before FFT
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # keep the linear part of the convolution (length = n + len(w) - 1 = 3n-2)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _conv_max(a: np.ndarray, Nfft: int) -> float:
    """
    Exact maximum of the linear convolution a * a, computed via FFT.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """
    Produce a modest set of diverse, normalised start vectors for length ``n``.
    Each vector sums to ``target_sum`` and contains only non‑negative entries.
    """
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.02)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential (mass at the end)
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) half‑cosine window (smooth bump)
    cos = 0.5 * (1 - np.cos(np.pi * np.arange(n) / (n - 1)))
    init.append(cos / cos.sum() * target_sum)

    # 6) Gaussian‑like bump (σ chosen so the support fits)
    sigma = n / 6.0
    gauss = np.exp(-0.5 * ((np.arange(n) - n / 2) / sigma) ** 2)
    init.append(gauss / gauss.sum() * target_sum)

    # 7) edge‑biased (two spikes at the ends)
    edge = np.zeros(n)
    edge[0] = edge[-1] = 0.5
    init.append(edge / edge.sum() * target_sum)

    # 8) small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    if pert.sum() == 0:
        pert = np.full(n, target_sum / n)
    init.append(pert / pert.sum() * target_sum)

    return init


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑its (exponentiated‑gradient) using a soft‑max
    approximation of the max‑convolution.
    Returns the best vector seen in this run together with its exact objective.
    """
    target_sum = math.sqrt(2.0 * n)
    eps = 1e-16
    y = np.log(a0 + eps)          # logits (unconstrained)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.07                     # base learning rate for logits

    # Temperature schedule parameters
    tau_factor = 0.35              # initial τ = τ_factor * M
    decay_rate = max_iters / 6.0   # exponential decay denominator

    # FFT size – must be a power of two ≥ 3 n‑2
    Nfft = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- probability simplex (softmax) -----
        logits = y - np.max(y)                 # improve numerical stability
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()      # Σ p = 1, p_i ≥ 0
        a = target_sum * p                     # Σ a = target_sum

        # ----- exact autocorrelation (linear conv) -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- soft‑max weights for the max -----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-8:
            # exact max: uniform weight over all almost‑max entries
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ----- gradient wrt a (2·rev) -----
        rev = _weighted_rev(a, w, Nfft)
        g_a = 2.0 * rev

        # ----- chain rule to logits (mirror‑descent on simplex) -----
        avg = np.dot(p, g_a)                 # ⟨p, g_a⟩
        g_y = target_sum * p * (g_a - avg)   # ∂f/∂y_i

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # occasional jitter to escape shallow plateaux
        if it % 4000 == 0:
            jitter = rng.normal(scale=0.015, size=n)
            y += jitter

        # ----- keep global best -----
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:9d}] new best = {best_score:.6f}")

    return best_seq, best_score


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing on the simplex with simulated‑annealing.
    Small fractions of mass are moved from a random source index to a random
    target index (or occasionally redistributed among three indices).
    """
    cur_seq = a.copy()
    cur_score = best_score
    target_sum = cur_seq.sum()          # already √(2 n)

    # FFT size for fast max‑convolution
    Nfft = 1 << ((2 * n - 1).bit_length())

    # simulated‑annealing schedule
    max_steps = 250_000
    step = 0
    while time.time() < deadline and step < max_steps:
        # ---- propose a move -------------------------------------------------
        i = rng.integers(0, n)
        if cur_seq[i] < 1e-12:
            step += 1
            continue
        j = rng.integers(0, n)
        if i == j:
            step += 1
            continue

        # transfer a random fraction (1%–6%) of the source mass
        frac = rng.uniform(0.01, 0.06)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # occasionally perform a three‑way redistribution (to increase diversity)
        if rng.random() < 0.02:
            k = rng.integers(0, n)
            if k != i and k != j and new_seq[k] > 0.0:
                extra = new_seq[k] * rng.uniform(0.01, 0.04)
                new_seq[k] -= extra
                new_seq[j] += extra

        # ----- evaluate -----
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        # ----- Metropolis acceptance -----
        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # temperature decays linearly
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score

        step += 1

    return cur_seq, cur_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (as a plain Python list) found within the time budget.
    """
    # ------------------------------------------------------------------
    # Setup RNG and time limits
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # 5 s safety margin

    # ------------------------------------------------------------------
    # Load a previously discovered construction if it exists
    # ------------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        base = None

    # ------------------------------------------------------------------
    # Choose a schedule of lengths to explore
    # ------------------------------------------------------------------
    if base is not None:
        n0 = int(base.shape[0])
    else:
        n0 = 2100               # historic fallback

    # length factors (up to 8000, never smaller than 300)
    factors = [1.0, 1.5, 2.0, 2.5, 3.5, 5.0]
    lengths = sorted({max(300, min(int(n0 * f), 8000)) for f in factors})

    # ------------------------------------------------------------------
    # Global best tracking
    # ------------------------------------------------------------------
    best_seq = None          # numpy array
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Optimisation over each chosen length
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # Build initial pool
        init_vectors = _make_initial_vectors(n, target_sum, rng)

        # Add interpolated previous best (if any) for cross‑scale transfer
        if base is not None:
            if base.shape[0] != n:
                old_x = np.linspace(0.0, 1.0, base.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                interp = np.interp(new_x, old_x, base)
                interp = np.maximum(interp, 0.0)
                if interp.sum() == 0:
                    interp = np.full(n, target_sum / n)
                else:
                    interp = interp / interp.sum() * target_sum
                init_vectors.append(interp)

        # Determine a sensible iteration budget per length.
        # The total work is roughly proportional to 1/n, so we keep more
        # iterations for short vectors.
        max_iters = int(3_000_000 * 2100 / n)
        max_iters = max(150_000, min(max_iters, 600_000))

        print(f"[run] length={n}, max_iters≈{max_iters}, target_sum={target_sum:.3f}")

        for idx, a0 in enumerate(init_vectors):
            if time.time() >= deadline:
                break
            print(f"  start {idx + 1}/{len(init_vectors)} for length {n}")

            best_seq, best_score = _adam_optimize(
                a0=a0.copy(),
                n=n,
                deadline=deadline,
                max_iters=max_iters,
                best_seq=best_seq if best_seq is not None else a0,
                best_score=best_score,
                rng=rng,
            )

        # Update ``base`` so that the next (larger) length can reuse the
        # current best as its interpolation seed.
        base = best_seq

    # ------------------------------------------------------------------
    # Final stochastic refinement (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # allocate up to half of the remaining budget to refinement,
        # but not more than ~20 s (the refinement is cheap)
        refine_deadline = time.time() + min(remaining * 0.5, 20.0)
        best_seq, best_score = _local_refine(
            a=best_seq,
            n=best_seq.shape[0],
            deadline=refine_deadline,
            best_score=best_score,
            rng=rng,
        )

    # ------------------------------------------------------------------
    # Fallback in the highly unlikely case nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = n0
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = float('inf')
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
