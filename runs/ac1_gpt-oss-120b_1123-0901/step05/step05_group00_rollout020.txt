def adam_optimize(a_init, n, conv_fft_len, corr_fft_len, max_iters, deadline):
    eps = 1e-16
    a = np.maximum(a_init, eps)
    a = a / a.sum()
    logits = np.log(a)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1=0.9; beta2=0.999; eps_adam=1e-8; lr=0.05
    lam = 0.5
    lam_factor = 1.01
    best_val_local = np.inf
    best_a_local = a.copy()
    last_improve = time.time()
    for it in range(max_iters):
        if time.time() > deadline: break
        # compute softmax
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        # conv
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        # compute smooth max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient w.r.t a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)
        # grad through softmax
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it+1))
        v_hat = v / (1.0 - beta2 ** (it+1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # anneal
        lam = min(lam * lam_factor, 1e7)
        val = 2.0 * n * max_c
        if val < best_val_local:
            best_val_local = val
            best_a_local = a.copy()
            last_improve = time.time()
            # also update global best via global variables? We'll handle outside.
        # stagnation 
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()
    # Return best instance
    return best_a_local, best_val_local

def subgradient_optimize(a_init, n, conv_fft_len, corr_fft_len, max_iters, deadline):
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_val = 2.0 * n * _full_convolution(a, fft_len=conv_fft_len).max()
    eta = 0.6
    last_improve = time.time()
    for it in range(max_iters):
        if time.time() > deadline: break
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
        if time.time() - last_improve > 5.0:
            break
    return best_a, best_val

"""
Step‑function optimiser – enhanced version.

The objective is                     F(a)=2·n·max(a*a)/(∑a)² .
Because the function is 0‑homogeneous we may restrict to probability
vectors (∑a=1).  The search works on a cascade of lengths, starting
with short vectors (fast cheap runs) and ending with a long vector
(up‑to 16384 entries).

Main ideas
----------
* Diverse families of initial points (uniform, up‑sampled best,
  Dirichlet draws, edge‑concentrated power‑laws, twin Gaussians,
  sine‑squared bump, random mixtures).
* Adam optimisation on a smooth‑max surrogate (log‑sum‑exp) of the
  convolution peak.  The temperature λ is annealed rapidly (×1.01)
  until it is essentially a true max.
* An exact sub‑gradient phase on the true max after Adam.
* After a full pass for a given length we perform a number of
  “local‑refinement” runs: start from the current best vector,
  perturb the logits with Gaussian noise and run a short Adam +
  sub‑gradient refinement.  This helps to escape shallow basins.
* Time management – the routine respects the wall‑clock budget and
  dynamically short‑circuits when the deadline approaches.

The code is deliberately self‑contained, uses only NumPy and the
standard library, and returns the best sequence found as a plain
Python list (exactly what the external evaluator expects).
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.

    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   max_iters: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate.
    Returns the best vector (probability simplex) and its objective value.
    """
    eps = 1e-16
    a = np.maximum(a0, eps)
    a = a / a.sum()
    logits = np.log(a)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05

    lam = 0.5               # initial temperature for log‑sum‑exp
    lam_factor = 1.01

    best_a = a.copy()
    conv = _full_convolution(a, fft_len=conv_fft_len)
    max_c = conv.max()
    best_val = 2.0 * n * max_c
    last_improve = time.time()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        # back‑propagation through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal temperature
        lam = min(lam * lam_factor, 1e7)

        # evaluate exact objective (denominator = 1 because ∑a=1)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            # (debug output is handled by the caller)

        # stagnation handling – tiny random shake
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          max_iters: int,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution value.
    Returns the best vector found and its objective value.
    """
    a = a0.copy()
    a = a / a.sum()
    best_a = a.copy()
    conv = _full_convolution(a, fft_len=conv_fft_len)
    max_c = conv.max()
    best_val = 2.0 * n * max_c
    eta = 0.6
    last_improve = time.time()

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _full_convolution(a, fft_len=conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        if time.time() - last_improve > 5.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Candidate factories
# ----------------------------------------------------------------------
def _make_initial_candidates(n: int, best_seq: np.ndarray) -> list[np.ndarray]:
    """
    Produce a heterogeneous list of start‑vectors for a given length.
    All vectors are normalised to sum to one.
    """
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # warm‑start from exact‑size best
    if best_seq is not None and len(best_seq) == n:
        candidates.append(best_seq.copy())

    # up‑sample a shorter best (linear interpolation)
    if best_seq is not None and len(best_seq) < n:
        old = np.asarray(best_seq, dtype=np.float64)
        x_old = np.linspace(0.0, 1.0, old.shape[0])
        x_new = np.linspace(0.0, 1.0, n)
        up = np.interp(x_new, x_old, old)
        if up.sum() > 0:
            up = up / up.sum()
        else:
            up = np.random.dirichlet(np.ones(n))
        candidates.append(up)

    # historic global variable, if present
    hseq = globals().get("height_sequence_1")
    if hseq is not None:
        h = np.asarray(hseq, dtype=np.float64)
        if h.shape[0] > n:
            h = h[:n]
        elif h.shape[0] < n:
            h = np.pad(h, (0, n - h.shape[0]), "constant")
        h = np.maximum(h, 0.0)
        if h.sum() > 0:
            h = h / h.sum()
        else:
            h = np.random.dirichlet(np.ones(n))
        candidates.append(h)

    # random Dirichlet
    candidates.append(np.random.dirichlet(np.ones(n)))

    # edge‑concentrated power‑law families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 4):
        shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            candidates.append(shape)

    # two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in [0.2, 0.5, 1.0, 2.0]:
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape = shape / shape.sum()
        candidates.append(shape)

    # smooth sine‑squared bump
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    shape = shape / shape.sum()
    candidates.append(shape)

    # random mixtures of edge‑Gaussian and uniform
    for _ in range(3):
        p = np.random.rand()
        gamma = np.random.uniform(0.1, 2.0)
        edge = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        edge = edge / edge.sum()
        mix = p * edge + (1.0 - p) * (np.full(n, 1.0 / n))
        mix = mix / mix.sum()
        candidates.append(mix)

    return candidates


# ----------------------------------------------------------------------
# Main entry point – the function that will be called externally
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a*a)/(sum(a)²).

    The routine respects the given wall‑clock budget and returns the best
    sequence discovered (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # small safety margin

    best_seq = None          # will hold a NumPy array (sum = 1)
    best_val = float("inf")

    # Lengths from cheap to expensive.  Small lengths give quick warm‑starts.
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))   # (2n‑1)+(n‑1)

        # ------------------------------------------------------------------
        # Build initial pool – we prune it for the very largest n to stay fast.
        # ------------------------------------------------------------------
        candidates = _make_initial_candidates(n, best_seq)

        # Limit the number of candidates for large n (memory / time)
        if n >= 8192:
            # keep a representative subset (uniform, upsample, dirichlet,
            # a couple of power‑laws, a couple of Gaussians)
            idx_keep = [0, 1, 2, 3, 4, 5, 6, 7]  # just a small sample
            candidates = [candidates[i] for i in idx_keep if i < len(candidates)]

        # ------------------------------------------------------------------
        # Optimise every starting point
        # ------------------------------------------------------------------
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # Ensure a proper probability vector
            a0 = np.maximum(a0, 0.0)
            if a0.sum() < 1e-12:
                a0 = np.random.dirichlet(np.ones(n))
            a0 = a0 / a0.sum()

            # iteration budget – larger n receives fewer iterations
            max_iters_adam = int(25000 * 1024 / n) + 2000
            max_iters_sub = 2000

            # ----- Adam on smooth‑max -----
            a_adam, val_adam = _adam_optimize(
                a0, n, conv_fft_len, corr_fft_len,
                max_iters_adam, deadline
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n} cand={cand_id} Adam] new best {val_adam:.6f}")

            # ----- Exact sub‑gradient refinement -----
            a_sub, val_sub = _subgradient_optimize(
                a_adam, n, conv_fft_len, corr_fft_len,
                max_iters_sub, deadline
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[n={n} cand={cand_id} Subgrad] new best {val_sub:.6f}")

        # ------------------------------------------------------------------
        # Local refinement of the current best vector for this length.
        # This is a cheap “hill‑climbing” phase that repeatedly perturbs
        # the logits of the best vector and performs a short Adam run.
        # ------------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            while time.time() < deadline - 1.0:
                # perturb logits with Gaussian noise
                noisy = np.log(best_seq + 1e-16) + np.random.normal(scale=0.12, size=n)
                a_start = np.exp(noisy - noisy.max())
                a_start = a_start / a_start.sum()

                # short Adam + sub‑gradient
                a_ref, val_ref = _adam_optimize(
                    a_start, n, conv_fft_len, corr_fft_len,
                    max_iters=2000, deadline=deadline
                )
                a_ref, val_ref = _subgradient_optimize(
                    a_ref, n, conv_fft_len, corr_fft_len,
                    max_iters=500, deadline=deadline
                )

                if val_ref < best_val:
                    best_val = val_ref
                    best_seq = a_ref.copy()
                    print(f"[n={n} perturbed] new best {val_ref:.6f}")
                else:
                    # stop after a few consecutive non‑improvements
                    break

    # ----------------------------------------------------------------------
    # Fallback – uniform vector if nothing was discovered (should never happen)
    # ----------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Convert to plain Python list as required by the external evaluator
    return best_seq.tolist()
