"""
Propose a non‑negative step‑function (probability vector) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)**2)

The algorithm is a longer and more aggressive version of the soft‑max Adam
optimiser used previously.  Main ideas

* Work on the probability simplex by parametrising a = softmax(x).
* Use a temperature‑annealed smooth‑max for most of the run, then switch to
  the exact max (sub‑gradient) once the temperature becomes very small.
* Multiple restarts and a schedule of increasing problem sizes (1024, 2048,
  4096).  The best sequence found for a given length is kept globally.
* After the main Adam phase a fine‑tuning phase (very low temperature) with a
  smaller learning rate is performed.
* Finally, a cheap “mass‑shuffling’’ greedy refinement moves tiny amounts of
  probability between random coordinates and accepts the move only if the
  objective improves.
* Simple stagnation handling: when no improvement for ~30 s a small random
  jitter is added to the logits.

import time
import numpy as np


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def smooth_max_and_weights(conv: np.ndarray, tau: float):
    """
    Compute a temperature‑smoothed max of `conv` and the corresponding
    softmax weights (∂smooth_max/∂conv).
    """
    c_max = conv.max()
    shifted = (conv - c_max) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    smooth = c_max + tau * np.log(sum_exp)
    weights = exp_shifted / sum_exp
    return smooth, weights

                          tau_initial * (1.0 - t / max_iters))

                if tau > 1e-5:
                    # smooth max (log‑sum‑exp) – gives a smooth gradient
                    _, weights = smooth_max_and_weights(conv, tau)
                else:
                    # exact max – use sub‑gradient of the true max
                    max_c = conv.max()
                    eps_eq = max_c * 1e-12 + 1e-15
                    idx = np.where(conv >= max_c - eps_eq)[0]
                    weights = np.zeros_like(conv)
                    if idx.size:
                        weights[idx] = 1.0 / idx.size

                # ----- gradient of (smooth) max w.r.t. a -----
                # grad_a = ∂max/∂a = 2·(weights ★ a)
                grad_a = 2.0 * cross_correlation(weights, a, fft_len=fft_len_corr)

                # back‑propagation through the soft‑max
                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                # ----- Adam update -------------------------------------------------
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # learning‑rate decay (geometric)
                if t % decay_every == 0:
                    lr *= 0.7

                # ----- bookkeeping --------------------------------------------------
                cur_val = 2.0 * n * conv.max()
                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t:06d}] new best {cur_val:.6f}")

                # ----- stagnation handling -----------------------------------------
                if time.time() - last_improve > 30.0:
                    # small random jitter in log‑space
                    x += (np.random.rand(n) - 0.5) * 5e-3
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t}] shake (stagnation)")

            # ----- fine‑tuning at very low temperature ------------------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a = best_seq.copy()
                x = np.log(a + 1e-15)
                m = np.zeros(n)
                v = np.zeros(n)
                lr_fine = 0.02
                fine_iters = 20000
                ft = 0
                while time.time() < deadline and ft < fine_iters:
                    ft += 1
                    # temperature at the end is essentially zero → exact max
                    max_x = np.max(x)
                    exp_x = np.exp(x - max_x)
                    a = exp_x / exp_x.sum()
                    conv = full_convolution(a, fft_len=fft_len_conv)

                    max_c = conv.max()
                    eps_eq = max_c * 1e-12 + 1e-15
                    idx = np.where(conv >= max_c - eps_eq)[0]
                    weights = np.zeros_like(conv)
                    if idx.size:
                        weights[idx] = 1.0 / idx.size

                    grad_a = 2.0 * cross_correlation(weights, a, fft_len=fft_len_corr)
                    inner = np.dot(a, grad_a)
                    grad_x = a * (grad_a - inner)

                    # Adam step with smaller learning rate
                    m = beta1 * m + (1 - beta1) * grad_x
                    v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
                    m_hat = m / (1 - beta1 ** ft)
                    v_hat = v / (1 - beta2 ** ft)
                    x = x - lr_fine * m_hat / (np.sqrt(v_hat) + eps)

                    cur_val = 2.0 * n * conv.max()
                    if cur_val < best_val - 1e-12:
                        best_val = cur_val
                        best_seq = a.copy()
                        print(f"[n={n} fine {ft}] improved to {cur_val:.6f}")

            # ----- cheap greedy refinement (mass shuffling) -------------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a_local = best_seq.copy()
                local_iters = 5000
                for it in range(local_iters):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    # move a modest fraction of a_local[i] to a_local[j]
                    move_frac = np.random.rand() * 0.5 + 0.2   # 0.2 … 0.7
                    eps_move = min(a_local[i] * move_frac, a_local[i])
                    if eps_move <= 0:
                        continue
                    a_new = a_local.copy()
                    a_new[i] -= eps_move
                    a_new[j] += eps_move
                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = 2.0 * n * conv_new.max()
                    if val_new < best_val - 1e-12:
                        best_val = val_new
                        best_seq = a_new.copy()
                        a_local = a_new
                        print(f"[n={n} local] improved to {val_new:.6f}")

    # -------------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector (objective = 2)
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    # convert to plain Python list as required by the evaluator
    return best_seq.tolist()
