"""
propose_candidate
-----------------

Searches for a non‑negative step‑function (probability vector) that
minimises

        F(a) = 2·n·max(a∗a) / (∑a)²

using a fast “parameter sweep → long Adam + sub‑gradient → brief
simulated‑annealing” pipeline.  The sweep explores a rich family of
U‑shaped (edge‑concentrated) distributions – power‑law, cosine‑power,
Gaussian‑edges and mixtures – with a fine grid of exponents.  The best
candidate is then refined by a high‑iteration Adam optimiser on a
smooth‑max surrogate followed by an exact‑max exponentiated‑gradient
phase (identical to the previous state‑of‑the‑art but with a longer
run and a slower temperature schedule).  Finally a short simulated‑annealing
local search is performed to escape residual shallow minima.  The routine
respects the given time‑budget (default 1000 s) and returns the best
sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT of length fft_len.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the Adam surrogate gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact objective (the same as evaluate_sequence but using a fast FFT).
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    # sum is never too close to zero (we keep it normalized), but guard anyway
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:

    Adam on a smooth‑max surrogate (log‑sum‑exp) followed by an
    exponentiated‑gradient sub‑gradient phase.  The routine respects the
    global deadline and updates the best sequence/value seen so far.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04                         # slightly smaller than the previous code
    lam = 0.5                         # initial temperature for smooth‑max
    lam_factor = 1.005                # slower schedule → finer approximation

    best_local_seq = a0.copy()
    best_local_val = global_best_val

    last_improve = time.time()
    it = 0
    max_iters = 400_000                # more iterations than the previous run

    while time.time() < deadline and it < max_iters:
        # ----- simplex from logits (entropy mirror) -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- evaluate current surrogate objective -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_local_val:
            best_local_val = val
            best_local_seq = a.copy()
            global_best_seq = a.copy()
            global_best_val = val
            last_improve = time.time()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # ----- smooth‑max (log‑sum‑exp) weighting -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

    a = best_local_seq if best_local_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()

    for sub_it in range(8_000):
        if time.time() >= deadline:

            sub_last_improve = time.time()
            print(f"[SubGrad n={n} it={sub_it:04d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse to numerical zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 5.0:
            break

    return global_best_seq, global_best_val


# ----------------------------------------------------------------------
def _simulated_anneal(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    current_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: repeatedly move a tiny amount of mass between two
    random positions and accept with a temperature that cools linearly with the
    remaining time.  The routine stops when the global deadline is reached.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_val = current_best_val
    cur_val = best_val

    start_time = time.time()
    # initial temperature – we tune it empirically
    T0 = 0.08
    T_min = 1e-6
    max_iters = 150_000

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # linear cooling based on elapsed time
        elapsed = time.time() - start_time
        frac = max(0.0, 1.0 - elapsed / (deadline - start_time))
        T = max(T_min, T0 * frac)

        # propose a tiny transfer of mass
        i, j = np.random.randint(n, size=2)
        if i == j or a[i] <= 0.0:
            continue
        # move at most 5 % of the source bucket
        delta = a[i] * np.random.rand() * 0.05
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        conv = _full_convolution(a_new, conv_fft_len)
        val = 2.0 * n * conv.max() / (a_new.sum() ** 2)

        # Metropolis acceptance
        if val < cur_val or np.random.rand() < np.exp(-(val - cur_val) / (T + 1e-12)):
            a = a_new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[SA it={it:06d}] new best {val:.6f}")

        # occasional early‑exit if no progress for a while
        if it % 5000 == 0 and time.time() >= deadline:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point.  Returns a list of non‑negative heights (a step function)
    that attempts to minimise the evaluation function within the given time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------- problem size -------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------- warm start (if available) -------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] != n:
            if w.shape[0] > n:
                w = w[:n]
            else:
                w = np.pad(w, (0, n - w.shape[0]), mode="constant")
        w = np.maximum(w, 0.0)
        if w.sum() > 0.0:
            w = w / w.sum()
            val = _objective(w, n, conv_fft_len)
            best_seq = w.copy()
            best_val = val
            print(f"[warm] initial value {val:.6f}")

    # ------------------- baseline uniform -------------------
    uniform = np.full(n, 1.0 / n)
    val = _objective(uniform, n, conv_fft_len)
    if val < best_val:
        best_seq = uniform.copy()
        best_val = val
        print(f"[uniform] value {val:.6f}")

    # ------------------- parameter sweep (U‑shaped families) -------------------
    # 1) Power‑law on the product (edges heavy)
    idx = np.arange(1, n + 1, dtype=np.float64)                 # 1 … n
    prod = idx * (n + 1 - idx)                                 # i·(n−i+1)
    log_prod = np.log(prod)

    for alpha in np.linspace(-5.0, -0.05, 100):
        if time.time() >= deadline:
            break
        shape = np.exp(alpha * log_prod)                       # prod**alpha but stable
        if shape.sum() == 0:
            continue
        shape /= shape.sum()
        val = _objective(shape, n, conv_fft_len)
        if val < best_val:
            best_seq = shape.copy()
            best_val = val
            print(f"[alpha={alpha:5.3f}] new best {val:.6f}")

    # 2) Cosine‑power (|cos|^β) – also U‑shaped
    i = np.arange(n, dtype=np.float64)
    cos_vals = np.abs(np.cos(np.pi * i / (n - 1)))
    for beta in [0.5, 1, 2, 4, 8, 16]:
        if time.time() >= deadline:
            break
        shape = cos_vals ** beta
        shape /= shape.sum()
        val = _objective(shape, n, conv_fft_len)
        if val < best_val:
            best_seq = shape.copy()
            best_val = val
            print(f"[beta={beta}] new best {val:.6f}")

    # 3) Gaussian edges (sum of two symmetric Gaussians)
    for sigma_frac in [0.02, 0.05, 0.10, 0.20]:
        if time.time() >= deadline:
            break
        sigma = sigma_frac * n
        shape = np.exp(- (i ** 2) / (2.0 * sigma ** 2)) \
                + np.exp(- ((n - 1 - i) ** 2) / (2.0 * sigma ** 2))
        shape /= shape.sum()
        val = _objective(shape, n, conv_fft_len)
        if val < best_val:
            best_seq = shape.copy()
            best_val = val
            print(f"[sigma={sigma_frac:.2f}] new best {val:.6f}")

    # 4) Linear mixture of the best edge‑shape and the uniform vector
    if best_seq is not None:
        edge_shape = best_seq
        for w in np.linspace(0.0, 1.0, 11):
            if time.time() >= deadline:
                break
            mix = w * edge_shape + (1.0 - w) * uniform
            mix /= mix.sum()
            val = _objective(mix, n, conv_fft_len)
            if val < best_val:
                best_seq = mix.copy()
                best_val = val
                print(f"[mix w={w:.2f}] new best {val:.6f}")

    # ------------------- deep optimisation (Adam + sub‑grad) -------------------
    if best_seq is not None:
        best_seq, best_val = _deep_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
        )

    # ------------------- short simulated‑annealing tweak -------------------
    if best_seq is not None and time.time() + 5 < deadline:
        best_seq, best_val = _simulated_anneal(
            best_seq, n, conv_fft_len, deadline, best_val
        )

    # ------------------- optional upscale to 8192 (if time permits) -------------------
    if time.time() + 10 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation to the larger grid
            x_old = np.linspace(0.0, 1.0, n, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        n = n_big  # final length is now 8192

    # ------------------------------------------------------------------
    # fallback – if nothing succeeded, return a uniform vector
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Finished – best internal value: {best_val:.6f}")
    return best_seq.tolist()
