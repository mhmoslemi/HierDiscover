"""
Propose a non‑negative step‑function (probability vector) that minimises

    F(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works directly on the simplex using a **hard‑max sub‑gradient**
instead of the smooth‑max used before.  At each iteration we

1. compute the full convolution a*a via a cheap FFT,
2. locate the index k* where the convolution attains its maximum,
3. compute the exact gradient of conv[k*] w.r.t. a (which is just a shifted copy
   of a, scaled by 2),
4. back‑propagate through a soft‑max parametrisation (logits → probability
   vector) and apply Adam,
5. periodically project the logits back onto the simplex (via the soft‑max)
   and, when progress stalls, add a small random jitter.

The search starts from the best sequence found so far (`height_sequence_1`
if it exists) and then upgrades to larger lengths (1024 → 2048 → 4096)
by simple up‑sampling (repeating each entry and dividing the mass).  The
budget (default 1000 s) is split among the three lengths, giving the larger
instances the lion’s share of the time – they have the best theoretical
lower bound (≈1).  A cheap “mass‑shuffling” local refinement is performed
after each optimisation run.

The function returns the best sequence (as a plain Python list) discovered
before the deadline.
"""

import time
import numpy as np


def _fft_len_for_conv(n: int) -> int:
    """Smallest power‑of‑2 length that can hold a full convolution of length 2·n‑1."""
    return 1 << int(np.ceil(np.log2(2 * n - 1)))


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _hard_max_gradient(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Gradient of conv[k_max] w.r.t. a.
    For each i with 0 ≤ k_max‑i < n we have ∂/∂a_i = 2·a_{k_max‑i},
    otherwise 0.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)

    start = max(0, k_max - (n - 1))
    end = min(n - 1, k_max)          # inclusive
    i = np.arange(start, end + 1, dtype=np.int64)
    j = k_max - i
    grad[i] = 2.0 * a[j]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    # Implementation of the algorithm from
    # https://arxiv.org/abs/1101.6081 (Duchi et al., 2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

                    print(f"[n={n} restart={r}] start from height_sequence_1")
                else:
                    a0 = np.random.dirichlet(np.ones(n))
                    x = np.log(a0 + 1e-15)
                    print(f"[n={n} restart={r}] random start")

            # Adam state (operating on the logits)
            m = np.zeros_like(x)
            v = np.zeros_like(x)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.2                     # initial learning rate
            decay_every = 5000           # learning‑rate decay interval

            # Iteration budget – generous but always respects the global deadline.
            max_iters = 200000 if n == 1024 else 150000 if n == 2048 else 100000

            last_improve = time.time()

            for t in range(1, max_iters + 1):
                if time.time() > deadline:
                    break

                # ----- soft‑max to obtain a probability vector a (Σa = 1) -----
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()

                # ----- convolution and objective ----- 
                conv = _full_convolution(a, fft_len)
                k_max = int(np.argmax(conv))
                cur_val = 2.0 * n * conv[k_max]        # denominator = 1²

                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t:06d}] new best {cur_val:.6f}")

                # ----- gradient of the hard max (sub‑gradient) -----
                grad_a = _hard_max_gradient(a, k_max)    # ∂conv[k_max]/∂a
                grad_a *= 2.0 * n                       # ∂F/∂a = 2·n·∂conv/∂a

                # back‑propagation through the soft‑max
                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                # ----- Adam update on logits -----
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # learning‑rate decay
                if t % decay_every == 0:
                    lr *= 0.6

                # ----- stagnation handling -----
                if time.time() - last_improve > 30.0:
                    # small random jitter in log‑space to escape flat regions
                    x += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t}] jitter (stagnation)")

            # ------------------- cheap greedy mass shuffling -------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a_local = best_seq.copy()
                local_iters = 3000
                for it in range(local_iters):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    # move a small random fraction of a_local[i] to position j
                    eps_move = min(a_local[i], 0.006) * (np.random.rand() * 0.6 + 0.2)
                    if eps_move <= 0:
                        continue
                    a_new = a_local.copy()
                    a_new[i] -= eps_move
                    a_new[j] += eps_move
                    conv_new = _full_convolution(a_new, fft_len)
                    val_new = 2.0 * n * conv_new.max()
                    if val_new < best_val - 1e-12:
                        best_val = val_new
                        best_seq = a_new.copy()
                        a_local = a_new
                        print(f"[local refine n={n}] improved to {val_new:.6f}")

    # -------------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector (objective = 2)
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = 2.0

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    # Convert to plain Python list as required by the evaluation harness
    return best_seq.tolist()
