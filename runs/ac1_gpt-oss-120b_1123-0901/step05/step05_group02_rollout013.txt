def adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, best_val, best_seq):
    a = a0.copy()
    # ensure positivity
    a = np.maximum(a, 0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()
    logits = np.log(a + 1e-15)

    # Adam state
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05  # maybe adjust
    lam = 0.2
    lam_factor = 1.018

    # iteration count heuristics
    max_iters = int(20000 * 1024 / n) + 4000
    last_improve = time.time()
    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # softmax to get a
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()
        # conv
        conv = full_convolution(a, conv_fft_len)
        # compute weights
        w = softmax_weights(conv, lam)
        # gradient w.r.t a
        grad_a = 2.0 * cross_correlation(w, a, corr_fft_len)
        # gradient w.r.t logits (softmax backprop)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
        lam = min(lam * lam_factor, 1e6)

        # evaluation
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t}] new best {val:.6f}")

        # if stagnated
        if time.time() - last_improve > 8.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
    return best_seq, best_val

def local_refine(a, n, deadline, best_val, best_seq):

    a_curr = a.copy()
    # Precompute conv for current
    conv_curr = full_convolution(a_curr, next_pow2(2*n-1))
    max_c_curr = conv_curr.max()
    val_curr = 2.0 * n * max_c_curr / (a_curr.sum()**2)
    # maybe update best already but will check inside moves
    # move parameters
    move_iter = 2000
    for i in range(move_iter):
        if time.time() > deadline:
            break
        # choose donor and receiver
        donor = np.random.randint(n)
        receiver = np.random.randint(n)
        if donor == receiver:
            continue
        # define delta proportion of donor's mass
        delta = a_curr[donor] * (np.random.rand() * 0.1)  # up to 10% of donor mass
        if delta <= 0:
            continue
        a_new = a_curr.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        # ensure non-negative (donor may become zero)
        if a_new[donor] < 0:
            continue
        # evaluate exact
        conv = full_convolution(a_new, next_pow2(2*n-1))
        max_c = conv.max()
        val = 2.0 * n * max_c / (a_new.sum()**2)
        if val < best_val:
            best_val = val
            best_seq = a_new.copy()
            a_curr = a_new
            # optionally accept even if not best to explore
        # optionally accept also if val better than current
        if val < val_curr:
            a_curr = a_new
            val_curr = val
    return best_seq, best_val

"""
Propose a step‑function (non‑negative list) that approximately minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

where “a*a” is the linear convolution (numpy.convolve).

Algorithm
---------
* Work on a probability vector (∑a = 1); the objective then reduces to
  2·n·max(conv) and is scale‑invariant.
* For a given length n we keep a pool of diverse **initial guesses**
  (uniform, tent, power‑law, double‑exponential, random Dirichlet with
  random concentration, sparse “spike” vectors, etc.).
* Each candidate is optimised with three phases
    1. **Adam** on a smooth‑max surrogate (soft‑max of the convolution).
       The temperature λ is increased slowly so the surrogate
       converges to the true max.
    2. **Exponentiated‑gradient** (sub‑gradient) on the exact max.
    3. **Local random‑mass‑transfer** hill‑climbing.
* The best sequence found so far is used as a warm‑start for later
  lengths and for a final fine‑tuning pass.
* A small set of deterministic “smoothing” post‑processes (moving‑average
  filters) is tried on the global best; the best of all variants is
  returned.
* The whole search respects the supplied time budget (default 1000 s)
  and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Smooth‑max weights w_i = exp(lam·conv_i) / Σ_j exp(lam·conv_j)."""
    mx = conv.max()
    w = np.exp(lam * (conv - mx))
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / s


def _objective(a: np.ndarray) -> float:
    """Exact evaluation of 2·n·max(conv) / (sum a)²."""
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)

def _adam_optimize(

    conv_fft: int,
    corr_fft: int,
    deadline: float,
    best_val: float,
    best_seq: np.ndarray,
) -> tuple[np.ndarray, float]:
    """Adam optimisation on a smooth‑max surrogate."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.3
    lam_factor = 1.015

    # iteration budget scales inversely with n (longer vectors need fewer steps)
    max_iters = int(8000 * 1024 / n) + 4000
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max parameterisation (simple simplex)
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        conv = _full_convolution(a, conv_fft)
        w = _softmax_weights(conv, lam)

        # gradient of the smooth‑max w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft)

        # back‑prop through the soft‑max (log‑softmax Jacobian)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e6)

        # current exact objective (used for book‑keeping only)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t:05d}] new best {val:.6f}")

        # simple stagnation handling – inject a tiny random shake
        if time.time() - last_improve > 8.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    # return the latest logits‑derived a (not the raw logits)
    mx = logits.max()
    a = np.exp(logits - mx)
    a = a / a.sum()
    return a, best_val, best_seq


def _eg_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft: int,
    corr_fft: int,
    deadline: float,
    best_val: float,
    best_seq: np.ndarray,
) -> tuple[np.ndarray, float, np.ndarray]:
    """Exponentiated‑gradient on the exact max (sub‑gradient method)."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    eta = 0.55
    last_improve = time.time()
    max_iters = 5000

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, conv_fft)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.full_like(conv, 1.0 / conv.size)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} EG iter={it:04d}] new best {val:.6f}")

        if time.time() - last_improve > 8.0:
            break

    return a, best_val, best_seq


def _local_refine(
    a0: np.ndarray,
    n: int,
    deadline: float,
    best_val: float,
    best_seq: np.ndarray,
) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Very cheap random‑mass‑transfer hill climbing.
    A small amount of probability mass is moved from a random donor
    to a random receiver; the exact objective is recomputed.
    """
    a = a0.copy()
    conv_fft = _next_pow2(2 * n - 1)

    # current score
    val_curr = _objective(a)

    for _ in range(2000):
        if time.time() > deadline:
            break
        donor = np.random.randint(n)
        receiver = np.random.randint(n)
        if donor == receiver or a[donor] <= 1e-12:
            continue
        # transfer up to 10 % of donor's mass
        delta = a[donor] * (np.random.rand() * 0.1)
        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta

        val = _objective(a_new)
        if val < best_val:
            best_val = val
            best_seq = a_new.copy()
            print(f"[n={n} local refine] new best {val:.6f}")

        # accept the move if it improves the current candidate
        if val < val_curr:
            a = a_new
            val_curr = val

    return a, best_val, best_seq


def _deterministic_initials(n: int) -> list[np.ndarray]:
    """A modest collection of hand‑crafted shapes."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # symmetric tent
    left = np.linspace(0, 1, n // 2, endpoint=False)
    right = left[::-1]
    tent = np.concatenate([left, right])
    if tent.sum() > 0:
        cand.append(tent / tent.sum())

    # edge‑concentrated power‑law (negative exponent)
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 7):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # double‑sided exponential
    for gamma in [0.0005, 0.001, 0.003, 0.006]:
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        cand.append(shape / shape.sum())

    # a simple Gaussian‑like bump centred in the middle
    mu = (n - 1) / 2.0
    sigma = n / 8.0
    x = np.arange(n, dtype=np.float64)
    shape = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    cand.append(shape / shape.sum())

    return cand


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None          # type: np.ndarray | None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Lengths to explore – larger lengths give more freedom but are slower.
    lengths = [2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        # ------------------------------------------------------------------
        # Build initial candidates
        candidates = _deterministic_initials(n)

        # warm‑start from the current global best (if lengths match)
        if best_seq is not None and len(best_seq) == n:
            candidates.append(best_seq.copy())

        # historic solution, if present
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            cand = np.asarray(hseq, dtype=np.float64)
            if cand.shape[0] > n:
                cand = cand[:n]
            else:
                cand = np.pad(cand, (0, n - cand.shape[0]))
            cand = np.maximum(cand, 0.0)
            if cand.sum() < 0.01:
                cand = np.random.dirichlet(np.ones(n))
            else:
                cand = cand / cand.sum()
            candidates.append(cand)

        # ------------------------------------------------------------------
        # Optimise each deterministic candidate
        for cid, a0 in enumerate(candidates):
            if time.time() > deadline:
                break
            # normalise to a probability vector
            a0 = np.maximum(a0, 0.0)
            if a0.sum() < 1e-12:
                a0 = np.random.dirichlet(np.ones(n))
            a0 = a0 / a0.sum()

            # Adam phase
            a, best_val, best_seq = _adam_optimize(
                a0, n, conv_fft_len, corr_fft_len, deadline, best_val, best_seq
            )
            # Exponentiated‑gradient phase
            a, best_val, best_seq = _eg_optimize(
                a, n, conv_fft_len, corr_fft_len, deadline, best_val, best_seq
            )
            # Local random refinement
            a, best_val, best_seq = _local_refine(
                a, n, deadline, best_val, best_seq
            )
            print(f"[n={n} cand={cid}] best so far {best_val:.6f}")

        # ------------------------------------------------------------------
        # Random Dirichlet restarts until the time budget for this length
        # expires (or a reasonable number of restarts is reached).
        rand_trials = 0
        while time.time() < deadline - 2.0 and rand_trials < 80:
            conc = 10 ** np.random.uniform(-2.0, 0.0)   # between 0.01 and 1
            a0 = np.random.dirichlet(np.full(n, conc))
            a, best_val, best_seq = _adam_optimize(
                a0, n, conv_fft_len, corr_fft_len, deadline, best_val, best_seq
            )
            a, best_val, best_seq = _eg_optimize(
                a, n, conv_fft_len, corr_fft_len, deadline, best_val, best_seq
            )
            a, best_val, best_seq = _local_refine(
                a, n, deadline, best_val, best_seq
            )
            rand_trials += 1

        # ------------------------------------------------------------------
        # Final fine‑tuning for the *global* best of this length
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] final fine‑tune (current best {best_val:.6f})")
            a, best_val, best_seq = _adam_optimize(
                best_seq, n, conv_fft_len, corr_fft_len, deadline, best_val, best_seq
            )
            a, best_val, best_seq = _eg_optimize(
                a, n, conv_fft_len, corr_fft_len, deadline, best_val, best_seq
            )
            a, best_val, best_seq = _local_refine(
                a, n, deadline, best_val, best_seq
            )

    # ------------------------------------------------------------------
    # Post‑processing: try a few light smoothing filters on the best vector.
    if best_seq is not None:
        for width in (3, 5, 7):
            if time.time() > deadline:
                break
            kernel = np.ones(width) / width
            smoothed = np.convolve(best_seq, kernel, mode="same")
            smoothed = np.maximum(smoothed, 0.0)
            if smoothed.sum() < 1e-12:
                continue
            smoothed = smoothed / smoothed.sum()
            val = _objective(smoothed)
            if val < best_val:
                best_val = val
                best_seq = smoothed
                print(f"[smoothing w={width}] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (should never happen)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return plain Python list as required by the external evaluator
    return best_seq.tolist()
