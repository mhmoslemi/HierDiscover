def full_convolution(a, fft_len=None):
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2*n-1)))
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa*fa, n=fft_len)[:2*n-1]
    return conv

def cross_correlation(weights, a, fft_len):
    n = a.shape[0]
    # weights size = 2*n-1 same as conv
    rev_a = a[::-1]
    fw = np.fft.rfft(weights, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw*fr, n=fft_len)
    start = n-1
    return corr[start:start+n]
```

                shifted = (conv - c_max) / tau
                exp_shifted = np.exp(shifted)
                sum_exp = exp_shifted.sum()
                weights = exp_shifted / sum_exp  # derivative weights

                # gradient wrt a
                grad_a = 2.0 * cross_correlation(weights, a, fft_len=fft_len_corr)

                # gradient through softmax
                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                # Adam update
                t += 1
                m = beta1 * m + (1 - beta1) * grad_x
                v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
                m_hat = m / (1 - beta1 ** t)
                v_hat = v / (1 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # learning rate decay
                if t % 40000 == 0:
                    lr *= 0.7

                # stagnation detection
                if time.time() - last_improve > 45.0:
                    # add jitter to logits
                    jitter = (np.random.rand(n) - 0.5) * 8e-3
                    x += jitter
                    last_improve = time.time()
                    print(f'[n={n} restart={r} iter={t}] jitter applied')

            # after main loop: fine tuning with hard max (tau very low)
            # reinitialize Adam for fine tuning
            if best_seq is not None and best_seq.shape[0] == n:
                a_fine = best_seq.copy()
                x = np.log(a_fine + 1e-15)
                m = np.zeros(n)
                v = np.zeros(n)
                lr_fine = 0.02
                fine_iters = 20000
                for ft in range(fine_iters):
                    if time.time() > deadline:
                        break
                    # compute a, conv
                    max_x = np.max(x)
                    exp_x = np.exp(x - max_x)
                    a = exp_x / exp_x.sum()
                    conv = full_convolution(a, fft_len=fft_len_conv)
                    cur_val = 2.0 * n * conv.max()
                    if cur_val < best_val - 1e-12:
                        best_val = cur_val
                        best_seq = a.copy()
                        print(f'[fine {n}] improved {cur_val:.6f}')
                    # gradient using true max (weights only at argmax)
                    k_star = np.argmax(conv)
                    # weight vector w: zero except at k_star where 1
                    w = np.zeros_like(conv)
                    w[k_star] = 1.0
                    grad_a = 2.0 * cross_correlation(w, a, fft_len=fft_len_corr)
                    inner = np.dot(a, grad_a)
                    grad_x = a * (grad_a - inner)
                    # Adam
                    m = beta1 * m + (1 - beta1) * grad_x
                    v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
                    m_hat = m / (1 - beta1 ** (ft + 1))
                    v_hat = v / (1 - beta2 ** (ft + 1))
                    x = x - lr_fine * m_hat / (np.sqrt(v_hat) + eps)

            # cheap local refinement: random mass transfer
            if best_seq is not None and best_seq.shape[0] == n:
                a_local = best_seq.copy()
                local_iters = 5000
                for it in range(local_iters):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    # compute move amount proportionally to a_local[i]
                    move_frac = np.random.rand() * 0.5 + 0.1  # between 0.1 and 0.6 of small amount
                    eps_move = min(a_local[i] * 0.02, move_frac * a_local[i])
                    if eps_move <= 0:

                    a_new[i] -= eps_move
                    a_new[j] += eps_move
                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)

                        best_val = val_new
                        best_seq = a_new.copy()
                        a_local = a_new

        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback)

    print(f'\nFinal best value: {best_val:.6f} (len={len(best_seq)})')
    return best_seq.tolist()

propose_candidate
=================

Search for a non‑negative step‑function (a probability vector) that minimizes

    F(a) = 2·n·max(a*a) / (sum(a)²)

where ``a*a`` is the linear convolution of the vector with itself.
The algorithm is a multi‑stage optimisation:

1. **Adam on a soft‑max (logits) parametrisation** – a smooth‑max
   (log‑sum‑exp) is used while a temperature is annealed from a large
   value (≈0.2) down to ≈5e‑7.  This gives a usable gradient of the
   max‑convolution and quickly finds a good region of the simplex.

2. **Fine‑tuning with the true max** – once the temperature is very
   low the weight vector is concentrated on the current maximal
   convolution entry, i.e. the exact sub‑gradient of the max is used.

3. **Local greedy mass‑shuffling** – a cheap hill‑climbing phase moves
   a small amount of probability mass between random coordinates and
   accepts the move only if the objective improves.

The search is time‑bounded (default 1000 s).  Several problem sizes
(``n = 1024, 2048, 4096, 8192``) are tried, each with a few restarts.
If a previous best sequence is available (``height_sequence_1``) it is
used for warm‑starting.  The best sequence found before the deadline is
returned as a plain Python list.

The implementation relies only on ``numpy`` and the provided
``evaluate_sequence`` function; no external files or network access are
used.
"""

# Helper FFT‑based convolution utilities
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)‑1) using FFT.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]
