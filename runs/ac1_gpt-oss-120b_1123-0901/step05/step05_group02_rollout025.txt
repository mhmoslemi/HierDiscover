    n = len(values)

    left = 0

"""
Propose a step‑function (non‑negative list) that approximately minimises

        F(a) = 2·n·max(conv(a,a)) / (sum(a)²),

where conv is the linear convolution (numpy.convolve).

Algorithm
---------
* Build a diverse pool of initial vectors for each length n.
  – uniform, Dirichlet (various concentrations), exponential/linear decays,
    “zig‑zag” permutations of sorted samples, up‑scaled previous best, etc.
* For every candidate run three optimisation phases
  1. Adam on a smooth‑max surrogate of the convolution peak.
  2. Exponentiated‑gradient (sub‑gradient) descent on the exact max.
  3. Light permutation refinement – sort the entries descending and place them
     on the line in a zig‑zag order (large values far apart); keep any
     improvement.
* The best vector seen across all lengths (respecting the time budget) is
  returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the
    smooth‑max. Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)],   n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _obj_val(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(conv)/sum² (sum is assumed >0)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max() / (a.sum() ** 2)


def _softmax(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    mx = logits.max()
    ex = np.exp(logits - mx)
    return ex / ex.sum()


def _zigzag_permutation(sorted_desc: np.ndarray) -> np.ndarray:
    """Place the sorted values (largest→smallest) in a zig‑zag order:
       0, n‑1, 1, n‑2, …  This tends to keep large entries far apart."""
    n = sorted_desc.size
    out = np.empty_like(sorted_desc)
    left, right, pos = 0, n - 1, 0
    while left <= right:
        out[pos] = sorted_desc[left]
        left += 1
        pos += 1
        if left <= right:
            out[pos] = sorted_desc[right]
            right -= 1
            pos += 1
    return out


def _upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a shorter sequence to a longer length."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.arange(old_n, dtype=np.float64)
    x_new = np.linspace(0, old_n - 1, new_n)
    up = np.interp(x_new, x_old, seq)
    # ensure non‑negativity after interpolation
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s < 1e-12:
        up = np.random.dirichlet(np.ones(new_n))
    else:
        up /= s
    return up


def _make_initial_candidates(n: int, best_seq):
    """
    Produce a list of diverse starting vectors for a given length.
    All vectors are normalised to sum‑one.
    """
    cand = []

    # 1) warm‑start with previous best of same length
    if best_seq is not None and len(best_seq) == n:
        cand.append(np.asarray(best_seq, dtype=np.float64))

    # 2) up‑scaled previous best (if shorter)
    if best_seq is not None and len(best_seq) < n:
        cand.append(_upscale(np.asarray(best_seq, dtype=np.float64), n))

    # 3) uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 4) Dirichlet samples – concentrations 1 (standard) and <1 (spiky)
    for conc in [1.0, 0.2, 0.5, 0.8]:
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 5) exponential decays (unsymmetric)
    for gamma in [0.0004, 0.0008, 0.0015, 0.003]:
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i)
        shape /= shape.sum()
        cand.append(shape)

    # 6) linear decreasing shape
    lin = np.arange(n, 0, -1, dtype=np.float64)
    lin /= lin.sum()
    cand.append(lin)

    # 7) zig‑zag permutation of a sorted Dirichlet sample
    z = np.random.dirichlet(np.ones(n))
    z_sorted = np.sort(z)[::-1]               # descending
    cand.append(_zigzag_permutation(z_sorted))

    # 8) random permutation of a Dirichlet sample (adds extra stochasticity)
    p = np.random.dirichlet(np.ones(n))
    np.random.shuffle(p)
    cand.append(p)

    # 9) power‑law edge‑concentrated shapes (as in the reference)
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-1.8, -0.2, 9):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # 10) a “two‑sided” exponential (symmetric) – kept for completeness
    for gamma in [0.001, 0.002, 0.004]:
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        cand.append(shape)

    # 11) a smoothed “tent” (rise‑then‑fall) – also from the reference
    left = np.linspace(0, 1, n // 2, endpoint=False)
    right = left[::-1]
    tent = np.concatenate([left, right])
    tent = np.maximum(tent, 0)
    if tent.sum() > 0:
        tent /= tent.sum()
        cand.append(tent)

    # Guarantee all candidates are proper probability vectors
    cleaned = []
    for c in cand:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            c = np.random.dirichlet(np.ones(n))
            s = c.sum()
        cleaned.append(c / s)
    return cleaned


def _adam_optimize(a0: np.ndarray, n: int, conv_fft: int, corr_fft: int,
                   deadline: float, best_val_ref) -> tuple:
    """
    Adam optimisation on the smooth‑max surrogate.
    Returns (best_seq, best_val) observed during the run.
    """
    a = a0.copy()
    logits = np.log(a + 1e-15)

    # Adam hyper‑parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.3
    lam_factor = 1.015

    # iteration budget – larger n ⇒ fewer iterations (keeps total time low)
    max_iters = int(20000 * 1024 / n) + 4000
    last_improve = time.time()
    best_seq = a.copy()
    best_val = best_val_ref[0]

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # a = soft‑max(logits)  (always stays on the simplex)
        a = _softmax(logits)

        # convolution and smooth‑max weights
        conv = _full_convolution(a, conv_fft)
        max_c = conv.max()

        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # gradient of the surrogate (linear in conv)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e6)

        # exact objective (since sum(a) == 1)
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            best_val_ref[0] = val        # propagate to outer scope
            last_improve = time.time()
            print(f"[n={n} Adam iter={t:05d}] new best {val:.6f}")

        # simple stagnation handling – random shake if idle too long
        if time.time() - last_improve > 7.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_seq, best_val


def _subgradient_optimize(a0: np.ndarray, n: int, conv_fft: int, corr_fft: int,
                          deadline: float, best_val_ref) -> tuple:
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    Returns (best_seq, best_val) observed.
    """
    a = a0.copy()
    eta = 0.55
    max_iters = 5000
    last_improve = time.time()
    best_seq = a.copy()
    best_val = best_val_ref[0]

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, conv_fft)
        max_c = conv.max()
        # active set = all indices attaining the maximum (within tolerance)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)

        if not active.any():
            w = np.zeros_like(conv)
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft)
        a = a * np.exp(-eta * grad_a)

        # re‑normalise
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            best_val_ref[0] = val
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it}] new best {val:.6f}")

        if time.time() - last_improve > 7.0:
            break

    return best_seq, best_val


def _permutation_refine(a: np.ndarray, n: int, deadline: float,
                        best_val_ref) -> tuple:
    """
    Light permutation improvement: sort descending and place values in a
    zig‑zag order (large entries far apart). Keep the improvement if any.
    """
    sorted_desc = np.sort(a)[::-1]
    new_seq = _zigzag_permutation(sorted_desc)
    val = _obj_val(new_seq, n)

    if val < best_val_ref[0]:
        print(f"[n={n} perm‑refine] improvement {val:.6f}")
        best_val_ref[0] = val
        return new_seq, val
    return a, best_val_ref[0]


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin

    best_seq = None
    best_val = float("inf")

    # Lengths to explore – larger n gives more freedom but each iteration costs more.
    lengths = [2048, 4096, 8192, 16384]

    # Pre‑compute FFT lengths (next power of two) for each n.
    fft_len_conv = {n: _next_pow2(2 * n - 1) for n in lengths}
    fft_len_corr = {
        n: _next_pow2(2 * n - 1 + n - 1) for n in lengths
    }   # 3·n‑2 upper bound, rounded up

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising n = {len_seq} ===".replace("len_seq", str(n)))
        conv_len = fft_len_conv[n]
        corr_len = fft_len_corr[n]

        # ------------------------------------------------------------------
        # Build a pool of initial candidates for this length.
        candidate_pool = _make_initial_candidates(n, best_seq)

        # ------------------------------------------------------------------
        # Optimise each candidate independently.
        for cand_id, cand in enumerate(candidate_pool):
            if time.time() > deadline:
                break

            # Normalise – guarantees sum = 1 so the objective does not depend on scale.
            a = cand / cand.sum()
            # shared mutable holder for the best value (so inner phases can update it)
            cur_best_val = [best_val]

            # --------- Phase 1 – Adam on smooth‑max ---------
            a, val = _adam_optimize(a, n, conv_len, corr_len, deadline, cur_best_val)

            # --------- Phase 2 – exponentiated‑gradient sub‑gradient ---------
            a, val = _subgradient_optimize(a, n, conv_len, corr_len, deadline, cur_best_val)

            # --------- Phase 3 – cheap permutation refinement ---------
            a, val = _permutation_refine(a, n, deadline, cur_best_val)

            # Update global best if we have improved.
            if cur_best_val[0] < best_val:
                best_val = cur_best_val[0]
                best_seq = a.copy()
                print(f"[GLOBAL] new overall best {best_val:.6f} (n={n})")

        # ------------------------------------------------------------------
        # One final fine‑tune of the global best for this length (optional).
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] fine‑tuning global best (val={best_val:.6f})")
            a = best_seq.copy()
            # reuse the Adam code but with a smaller learning rate and slower temperature growth.
            logits = np.log(a + 1e-15)
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.04
            lam = 0.5
            lam_factor = 1.01
            last_improve = time.time()
            fine_iters = int(40000 * 1024 / n) + 5000

            for t in range(1, fine_iters + 1):
                if time.time() > deadline:
                    break

                a = _softmax(logits)
                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

                grad_a = 2.0 * _cross_correlation(w, a, corr_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                lam = min(lam * lam_factor, 1e6)

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} fine‑tune iter={t:05d}] new best {val:.6f}")

                if time.time() - last_improve > 10.0:
                    # small random shake before convergence
                    logits += (np.random.rand(n) - 0.5) * 5e-3
                    last_improve = time.time()

    # ------------------------------------------------------------------
    # Safety fallback – uniform vector if nothing was discovered.
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    return best_seq.tolist()
