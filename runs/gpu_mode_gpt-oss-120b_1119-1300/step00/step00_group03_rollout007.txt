@triton.jit
def layernorm_gate_kernel(
    out_ptr, out_gate_ptr, gamma_ptr, beta_ptr,
    B, N, H,
    stride_ob, stride_oi, stride_oj, stride_oh,
    eps,
    BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    total = B * N * N
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    out_base = out_ptr + b * stride_ob + i * stride_oi + j * stride_oj
    out_gate_base = out_gate_ptr + b * stride_ob + i * stride_oi + j * stride_oj

    sum = tl.float32(0.0)
    sum_sq = tl.float32(0.0)
    # compute sum & sum_sq
    for off in range(0, H, BLOCK_H):
        cur_block = tl.minimum(BLOCK_H, H - off) # but compile-time BLOCK_H constant cannot vary; we handle mask
        offs = tl.arange(0, BLOCK_H)
        mask = offs < (H - off)
        val = tl.load(out_base + (off + offs) * stride_oh, mask=mask, other=0.0).to(tl.float32)
        sum += tl.sum(val, axis=0)
        sum_sq += tl.sum(val * val, axis=0)
    mean = sum / H
    var = sum_sq / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # load gate, gamma, beta, apply
    for off in range(0, H, BLOCK_H):
        offs = tl.arange(0, BLOCK_H)
        mask = offs < (H - off)
        idx = off + offs
        out_val = tl.load(out_base + idx * stride_oh, mask=mask, other=0.0).to(tl.float32)
        gate = tl.load(out_gate_base + idx * stride_oh, mask=mask, other=0.0).to(tl.float32)
        gamma = tl.load(gamma_ptr + idx, mask=mask, other=0.0).to(tl.float32)
        beta = tl.load(beta_ptr + idx, mask=mask, other=0.0).to(tl.float32)

        norm = (out_val - mean) * inv_std * gamma + beta
        res = norm * gate
        tl.store(out_base + idx * stride_oh, res, mask=mask)

    right_gate_weight = weights["right_gate.weight"]

    left = left * left_gate
    right = right * right_gate

    # 5. Cast to fp16 for efficient GEMM
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # 6. Compute out via batched GEMM per hidden_dim using torch.bmm
    # reshape: (bs*hidden_dim, N, N) each
    left_reshape = left.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)
    right_reshape = right.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)
    out_batch = torch.bmm(left_reshape, right_reshape.transpose(1, 2))  # (bs*hidden_dim, N, N)
    out = out_batch.reshape(bs, hidden_dim, N, N).permute(0, 2, 3, 1)  # (bs,N,N,hidden_dim)

        eps=1e-5,
        BLOCK_H=32
    )
    # after kernel, out now holds normalized and gated values

    out = torch.nn.functional.linear(out, to_out_weight)  # bias is zero per spec

    return out

@triton.jit
def layernorm_gate_kernel(
    out_ptr, out_gate_ptr, gamma_ptr, beta_ptr,
    B, N,
    stride_ob, stride_oi, stride_oj, stride_oh,
    eps,
    H: tl.constexpr,
    BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    total = B * N * N
    # compute indices
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base pointer offsets
    out_base = out_ptr + b * stride_ob + i * stride_oi + j * stride_oj
    gate_base = out_gate_ptr + b * stride_ob + i * stride_oi + j * stride_oj

    # compute mean and variance
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)

    for off in range(0, H, BLOCK_H):
        cur_off = tl.arange(0, BLOCK_H)
        mask = (off + cur_off) < H
        idx = off + cur_off
        ptr = out_base + idx * stride_oh
        val = tl.load(ptr, mask=mask, other=0.0)
        val_f = val.to(tl.float32)
        sum_val += tl.sum(val_f, axis=0)
        sum_sq += tl.sum(val_f * val_f, axis=0)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # normalize, apply gamma/beta and gate
    for off in range(0, H, BLOCK_H):
        cur_off = tl.arange(0, BLOCK_H)
        mask = (off + cur_off) < H
        idx = off + cur_off
        out_val = tl.load(out_base + idx * stride_oh, mask=mask, other=0.0).to(tl.float32)
        gate_val = tl.load(gate_base + idx * stride_oh, mask=mask, other=0.0).to(tl.float32)

        beta_val = tl.load(beta_ptr + idx, mask=mask, other=0.0).to(tl.float32)

        norm = (out_val - mean) * inv_std * gamma_val + beta_val
        res = norm * gate_val

import torch
import triton
import triton.language as tl

@triton.jit
def layernorm_gate_kernel(out_ptr, out_gate_ptr, gamma_ptr, beta_ptr,
                          B, N,
                          stride_ob, stride_oi, stride_oj, stride_oh,
                          eps,
                          H: tl.constexpr,
                          BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    total = B * N * N
    # assign to guarantee pid < total
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    out_base = out_ptr + b * stride_ob + i * stride_oi + j * stride_oj
    gate_base = out_gate_ptr + b * stride_ob + i * stride_oi + j * stride_oj

    # Compute mean and variance over hidden dimension H
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)

    for off in range(0, H, BLOCK_H):
        idx = off + tl.arange(0, BLOCK_H)
        mask = idx < H
        val = tl.load(out_base + idx * stride_oh, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(val, axis=0)
        sum_sq += tl.sum(val * val, axis=0)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Apply layernorm, gamma/beta, and gating
    for off in range(0, H, BLOCK_H):
        idx = off + tl.arange(0, BLOCK_H)
        mask = idx < H
        x = tl.load(out_base + idx * stride_oh, mask=mask, other=0.0).to(tl.float32)
        gate = tl.load(gate_base + idx * stride_oh, mask=mask, other=0.0).to(tl.float32)
        gamma = tl.load(gamma_ptr + idx, mask=mask, other=0.0).to(tl.float32)
        beta = tl.load(beta_ptr + idx, mask=mask, other=0.0).to(tl.float32)

        norm = (x - mean) * inv_std * gamma + beta
        out_val = norm * gate
        tl.store(out_base + idx * stride_oh, out_val, mask=mask)

def custom_kernel(data):
    """
    Triton-accelerated outgoing Triangle Multiplicative Update (TriMul) forward pass.
    Implements the operations from the AlphaFold3 reference:
    - LayerNorm on the input.
    - Linear projections (left/right) and gates.
    - Optional input mask.
    - A batched GEMM that computes the triangular multiplicative update.
    - Triton kernel that fuses the second LayerNorm, the out_gate multiplication,
      and the final linear projection's weight+bias are applied with PyTorch (the
      heavy O(N^3) work is delegated to torch.bmm which uses cuBLAS/Tensor Cores).
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on the device).
    """
    import torch

    # Unpack input tuple
    input_tensor, mask, weights, config = data

    # Extract configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----------------------------------------------------------------------
    # 1) First LayerNorm over the channel (dim) dimension
    # ----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # LayerNorm expects normalized_shape = (dim,)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape [B, N, N, dim], dtype float32

    # ----------------------------------------------------------------------
    # 2) Linear projections (left/right) and gates
    # ----------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)

    # projection in fp16
    left = torch.nn.functional.linear(x, left_proj_weight)  # [B,N,N,hidden_dim], fp16
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ----------------------------------------------------------------------
    # 3) Apply optional mask (broadcast on hidden dimension)
    # ----------------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask_f = mask.to(left.dtype).unsqueeze(-1)  # shape [B,N,N,1], fp16
        left = left * mask_f
        right = right * mask_f

    # ----------------------------------------------------------------------
    # 4) Compute gates (in fp32) and cast to fp16 for multiplication
    # ----------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))   # fp32
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Cast gates to fp16 for element‑wise multiply
    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    left = left * left_gate_h
    right = right * right_gate_h

    # ----------------------------------------------------------------------
    # 5) Triangular multiplicative update via batched GEMM (out = left * right)
    #    Compute: out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
    # ----------------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape is [B,N,N,hidden_dim]

    # reshape for batched matrix multiplication
    left_reshape = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)  # (B*H,N,N)
    right_reshape = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # GEMM: left @ right^T  (using Tensor Cores on H100)
    out = torch.bmm(left_reshape, right_reshape.transpose(1, 2))
    # reshape back to [B, N, N, hidden_dim] and convert to fp32 for the following LayerNorm
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)  # [B,N,N,H]

    # ----------------------------------------------------------------------
    # 6) Second LayerNorm + out_gate multiplication – fused in Triton
    # ----------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]

    # Triton kernel fusing LayerNorm (mean/var over hidden_dim) and gating.
    # Each program handles a single (b, i, j) location.
    total_programs = B * N * N
    grid = (total_programs,)

    # Kernel launch (choose a reasonable block size)
    layernorm_gate_kernel[grid](
        out_ptr=out,
        out_gate_ptr=out_gate,
        gamma_ptr=to_out_norm_weight,
        beta_ptr=to_out_norm_bias,
        B=B,
        N=N,
        stride_ob=out.stride(0),
        stride_oi=out.stride(1),
        stride_oj=out.stride(2),
        stride_oh=out.stride(3),
        eps=eps,
        H=hidden_dim,
        BLOCK_H=32,            # 32‑wide vector loads; hidden_dim ≤ 384
        num_warps=4,
    )

    # ----------------------------------------------------------------------
    # 7) Final linear projection to original dimensionality
    # ----------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"]
    output = torch.nn.functional.linear(out, to_out_weight)  # bias is zero

    return output
```

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse the 2nd LayerNorm (over the hidden dimension) with
# the out‑gate multiplication.  One program instance works on a single
# (batch, i, j) coordinate and loops over the hidden channels.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    out_ptr,                     # float32*   [B, N, N, H]
    out_gate_ptr,                # float32*   [B, N, N, H]
    gamma_ptr,                   # float32*   [H]   (to_out_norm.weight)
    beta_ptr,                    # float32*   [H]   (to_out_norm.bias)
    B, N,                       # runtime scalars
    stride_ob, stride_oi, stride_oj, stride_oh,   # strides for out (in elements)
    eps,                         # scalar epsilon
    H: tl.constexpr,             # hidden dimension (compile‑time)
    BLOCK_H: tl.constexpr = 32  # vector load width
):
    pid = tl.program_id(0)                     # one program per (b,i,j)
    total = B * N * N
    # decode linear pid -> (b,i,j)
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base pointers for this (b,i,j) slice
    out_base = out_ptr + b * stride_ob + i * stride_oi + j * stride_oj
    gate_base = out_gate_ptr + b * stride_ob + i * stride_oi + j * stride_oj

    # --------------------------------------------------------------
    # 1) Compute mean & variance of out over the hidden dimension H
    # --------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq  = tl.float32(0.0)

    for off in range(0, H, BLOCK_H):
        idx = off + tl.arange(0, BLOCK_H)               # [BLOCK_H] vector of indices
        mask = idx < H                                   # handle tail
        val = tl.load(out_base + idx * stride_oh,
                      mask=mask, other=0.0).to(tl.float32)   # fp32 load
        sum_val += tl.sum(val)                           # scalar reduction
        sum_sq  += tl.sum(val * val)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # --------------------------------------------------------------
    # 2) Apply LayerNorm (gamma/beta) and multiply by out_gate
    # --------------------------------------------------------------
    for off in range(0, H, BLOCK_H):
        idx = off + tl.arange(0, BLOCK_H)
        mask = idx < H
        x   = tl.load(out_base + idx * stride_oh,
                      mask=mask, other=0.0).to(tl.float32)
        g   = tl.load(gate_base + idx * stride_oh,
                      mask=mask, other=0.0).to(tl.float32)
        gamma = tl.load(gamma_ptr + idx,
                        mask=mask, other=0.0).to(tl.float32)
        beta  = tl.load(beta_ptr + idx,
                        mask=mask, other=0.0).to(tl.float32)

        # LayerNorm + gate
        norm = (x - mean) * inv_std * gamma + beta
        out  = norm * g

        tl.store(out_base + idx * stride_oh, out, mask=mask)


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul) forward.
    Steps:
      1. LayerNorm over the last channel (dim).
      2. Linear projections (left/right) + optional masking.
      3. Gating (sigmoid(linear(...))) and element‑wise multiplication.
      4. Batched GEMM that realises the einsum
         out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:].
      5. Triton kernel fusing the second LayerNorm, the out_gate scaling,
         and the “to_out_norm” affine transformation.
      6. Final linear projection back to `dim`.
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on the GPU).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) First LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2) Linear projections (left/right) – use fp16 for speed
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)

    left  = torch.nn.functional.linear(x, left_proj_weight)   # fp16
    right = torch.nn.functional.linear(x, right_proj_weight)  # fp16

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask_f = mask.to(left.dtype).unsqueeze(-1)      # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gates (computed in fp32, then cast)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))   # fp32
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))  # fp32
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))    # fp32

    # cast gates to fp16 for the element‑wise multiplications
    left_gate_h  = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    left  = left * left_gate_h
    right = right * right_gate_h

    # ------------------------------------------------------------------
    # 5) Triangular multiplicative update via batched GEMM (Tensor Cores)
    #    out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*H, N, N) where H = hidden_dim
    left_reshape  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_reshape = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batched matrix multiply (uses cuBLAS + Tensor Cores on H100)
    out = torch.bmm(left_reshape, right_reshape.transpose(1, 2))

    # reshape back to [B, N, N, H] and promote to fp32 for the next LayerNorm
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6) Second LayerNorm + out_gate multiplication (fused in Triton)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]

    total_programs = B * N * N
    grid = (total_programs,)

    layernorm_gate_kernel[grid](
        out_ptr=out,
        out_gate_ptr=out_gate,
        gamma_ptr=to_out_norm_weight,
        beta_ptr=to_out_norm_bias,
        B=B,
        N=N,
        stride_ob=out.stride(0),
        stride_oi=out.stride(1),
        stride_oj=out.stride(2),
        stride_oh=out.stride(3),
        eps=eps,
        H=hidden_dim,
        BLOCK_H=32,               # 32‑wide vector loads (hidden_dim ≤ 384)
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7) Final linear projection back to the original ``dim``
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"]   # shape [dim, hidden_dim]
    output = torch.nn.functional.linear(out, to_out_weight)  # bias is zero

    return output
