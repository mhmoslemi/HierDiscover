```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul (outgoing) operation from AlphaFold3.
    The heavy part – a batched matrix multiplication that corresponds to the einsum
    '... i k d, ... j k d -> ... i j d' – is executed in a custom Triton kernel.
    All other ops (LayerNorm, linear projections, gating and final projection) are
    performed with PyTorch's fused kernels. The kernel works on FP16 data with
    a float32 accumulator and supports arbitrary batch size, sequence length and
    hidden dimension.
    """
    # unpack
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    # Config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------------------------------
    # LayerNorm on input (dim)
    # -------------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -------------------------------------------------------------------------
    # Linear projections (dim -> hidden_dim)
    # -------------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # -------------------------------------------------------------------------
    # Optional mask (broadcasted on hidden dimension)
    # -------------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape [B, N, N] -> [B, N, N, 1]
        mask = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # -------------------------------------------------------------------------
    # Gating (sigmoid of linear projection of the normalized input)
    # -------------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------------
    # Core TriMul: out_{i j d} = sum_k left_{i k d} * right_{j k d}
    # -------------------------------------------------------------------------
    B, N, _, _ = left.shape  # N = seq_len

    # Permute to (B, hidden, N, N) and make contiguous for easy view
    left_perm = left.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()    # (B, H, N, N)  (swap i/j -> k/j)

    # Convert to FP16 for the kernel (keep a copy in FP32 for later ops)
    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    # Collapse batch*hidden into a single batch dimension for the kernel
    batch_hidden = left_fp16.shape[0] * left_fp16.shape[1]   # B * H
    left_mat = left_fp16.view(batch_hidden, N, N)
    right_mat = right_fp16.view(batch_hidden, N, N)

    # Allocate output buffer (FP16)
    out_mat = torch.empty((batch_hidden, N, N), dtype=torch.float16, device=device)

    # -------------------------------------------------------------------------
    # Triton kernel launch
    # -------------------------------------------------------------------------
    # Block sizes – tuned for H100 Tensor Core usage
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # should divide nicely into N at typical sizes

    @triton.jit
    def batched_matmul_kernel(
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
    ):
        pid_b = tl.program_id(2)   # batch*hidden index
        pid_m = tl.program_id(1)   # tile row
        pid_n = tl.program_id(0)   # tile column

        # Compute start offsets for this tile
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

        # Accumulator in FP32
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # Loop over K dimension
        for k in range(0, tl.cdiv(K, BLOCK_K)):
            cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

            # Load A tile (M x K)
            a_ptrs = A + pid_b * stride_a_batch \
                     + offs_m[:, None] * stride_a_m \
                     + cur_k[None, :] * stride_a_k
            a_mask = (offs_m[:, None] < M) & (cur_k[None, :] < K)
            a = tl.load(a_ptrs, mask=a_mask, other=0.0)

            # Load B tile (K x N)
            b_ptrs = B + pid_b * stride_b_batch \
                     + cur_k[:, None] * stride_b_k \
                     + offs_n[None, :] * stride_b_n
            b_mask = (cur_k[:, None] < K) & (offs_n[None, :] < N)
            b = tl.load(b_ptrs, mask=b_mask, other=0.0)

            # Accumulate
            acc += tl.dot(a, b)

        # Write back C tile (M x N) as FP16
        c_ptrs = C + pid_b * stride_c_batch \
                 + offs_m[:, None] * stride_c_m \
                 + offs_n[None, :] * stride_c_n
        c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
        tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)

    # Compute strides (in elements)
    stride_a_batch = left_mat.stride(0)
    stride_a_m = left_mat.stride(1)
    stride_a_k = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_k = right_mat.stride(1)
    stride_b_n = right_mat.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_m = out_mat.stride(1)
    stride_c_n = out_mat.stride(2)

    grid = (triton.cdiv(N, BLOCK_N), triton.cdiv(N, BLOCK_M), batch_hidden)
    batched_matmul_kernel[grid](
        left_mat, right_mat, out_mat,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # -------------------------------------------------------------------------
    # Reshape kernel output back to [B, N, N, hidden_dim] and convert to FP32
    # -------------------------------------------------------------------------
    out_fp32 = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # -------------------------------------------------------------------------
    # Post‑processing: LayerNorm, gating, final projection
    # -------------------------------------------------------------------------
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    out = out_norm * out_gate

    final = F.linear(out, weights["to_out.weight"])
    return final
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module (AlphaFold3).

    Steps performed:
    1️⃣  Layer‑norm on the input tensor `x` (dim = last dimension).
    2️⃣  Linear projections `left` and `right` from `dim` → `hidden_dim`.
    3️⃣  Optional masking of the two projected tensors.
    4️⃣  Gating via sigmoid(linear(...)) and element‑wise multiplication.
    5️⃣  Core operation – a batched matrix multiplication that implements
        the einsum  `... i k d, ... j k d -> ... i j d` – is executed in a
        custom Triton kernel (FP16 arithmetic with a FP32 accumulator).
        The kernel works on a flattened batch dimension `B*hidden_dim`.
    6️⃣  Reshape the result back to `[B, N, N, hidden_dim]` and cast to FP32.
    7️⃣  Post‑processing: LayerNorm over `hidden_dim`, output gating and
        final linear projection back to `dim`.

    All heavy arithmetic (the N³ contraction) runs on the GPU with a fused
    kernel; all other ops use PyTorch’s highly‑optimised kernels.
    """
    # ----------------------------------------------------------------------
    # unpack inputs
    # ----------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------------------------------
    # 1️⃣  Input LayerNorm  (normalizes over the last dimension)
    # ----------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ----------------------------------------------------------------------
    # 2️⃣  Linear projections  (dim → hidden_dim) – no bias
    # ----------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ----------------------------------------------------------------------
    # 3️⃣  Apply optional mask (mask shape [B, N, N])
    # ----------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask may be bool or int; cast to the same dtype as `left/right`
        mask = mask.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ----------------------------------------------------------------------
    # 4️⃣  Gating (sigmoid of a linear projection of the *normalized* input)
    # ----------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 5️⃣  Core TriMul: out_{i j d} = Σ_k left_{i k d} * right_{j k d}
    # ----------------------------------------------------------------------
    B, N, _, _ = left.shape  # N == seq_len

    # Permute to bring the hidden dimension next to the batch dimension
    # left: (B, H, N, N)   where the last two axes are (i, k)
    # right: (B, H, N, N)  where the last two axes are (k, j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    # Cast to FP16 for the kernel (still keep a FP32 copy later)
    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    # Collapse batch*hidden into a single batch dimension for the kernel
    batch_hidden = left_fp16.shape[0] * left_fp16.shape[1]          # B * H
    left_mat = left_fp16.view(batch_hidden, N, N)   # (B*H, M=N, K=N)
    right_mat = right_fp16.view(batch_hidden, N, N) # (B*H, K=N, N=N)

    # Output buffer (FP16)
    out_mat = torch.empty((batch_hidden, N, N), dtype=torch.float16, device=device)

    # ----------------------------------------------------------------------
    # Triton kernel: batched matmul (M=N, K=N, N=N)
    # ----------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    @triton.jit
    def batched_matmul_kernel(
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
    ):
        pid_b = tl.program_id(2)   # batch*hidden index
        pid_m = tl.program_id(1)   # row tile
        pid_n = tl.program_id(0)   # col tile

        # Tile start indices
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

        # Accumulator in FP32 for precision
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # Loop over K dimension in blocks
        for k in range(0, tl.cdiv(K, BLOCK_K)):
            cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

            # Load A tile (M x K)
            a_ptrs = A + pid_b * stride_a_batch \
                     + offs_m[:, None] * stride_a_m \
                     + cur_k[None, :] * stride_a_k
            a_mask = (offs_m[:, None] < M) & (cur_k[None, :] < K)
            a = tl.load(a_ptrs, mask=a_mask, other=0.0)

            # Load B tile (K x N)
            b_ptrs = B + pid_b * stride_b_batch \
                     + cur_k[:, None] * stride_b_k \
                     + offs_n[None, :] * stride_b_n
            b_mask = (cur_k[:, None] < K) & (offs_n[None, :] < N)
            b = tl.load(b_ptrs, mask=b_mask, other=0.0)

            # Accumulate product
            acc += tl.dot(a, b)

        # Write out C tile (M x N) as FP16
        c_ptrs = C + pid_b * stride_c_batch \
                 + offs_m[:, None] * stride_c_m \
                 + offs_n[None, :] * stride_c_n
        c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
        tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)

    # Compute strides (in elements, not bytes)
    stride_a_batch = left_mat.stride(0)
    stride_a_m = left_mat.stride(1)
    stride_a_k = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_k = right_mat.stride(1)
    stride_b_n = right_mat.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_m = out_mat.stride(1)
    stride_c_n = out_mat.stride(2)

    # Launch grid = (tiles_N, tiles_M, batches)
    grid = (
        triton.cdiv(N, BLOCK_N),
        triton.cdiv(N, BLOCK_M),
        batch_hidden,
    )
    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----------------------------------------------------------------------
    # 6️⃣  Reshape kernel output back to [B, N, N, hidden_dim] (FP32)
    # ----------------------------------------------------------------------
    out_fp32 = out_mat.view(B, hidden_dim, N, N) \
               .permute(0, 2, 3, 1).to(torch.float32)

    # ----------------------------------------------------------------------
    # 7️⃣  Post‑processing: LayerNorm over hidden_dim, gating and final proj
    # ----------------------------------------------------------------------
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    out = out_norm * out_gate               # apply output gate
    final = F.linear(out, weights["to_out.weight"])   # (B, N, N, dim)

    return final
