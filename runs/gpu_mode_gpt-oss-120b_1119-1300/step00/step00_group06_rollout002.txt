def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]; hidden = config["hidden_dim"]; bs = input_tensor.shape[0]; N = input_tensor.shape[1]
    
    # Perform LayerNorm: x = (x - mean)/sqrt(var+eps) * weight + bias
    # use built-in torch.nn.functional.layer_norm? Actually we have weight and bias. Use torch.nn.functional.layer_norm with normalized_shape = dim.

    eps = 1e-5 maybe; in PyTorch default eps=1e-5.

    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Compute linear projections
    # shape (bs,N,N,dim) -> flatten (bs*N*N, dim)
    x_flat = x.view(-1, dim)

    left = torch.nn.functional.linear(x_flat, left_proj_weight) # weight shape (hidden_dim, dim)
    left = left.view(bs, N, N, hidden)

    right = torch.nn.functional.linear(x_flat, right_proj_weight)
    right = right.view(bs, N, N, hidden)

    # Compute gates
    left_gate = torch.nn.functional.linear(x_flat, left_gate_weight).view(bs,N,N,hidden).sigmoid()
    right_gate = torch.nn.functional.linear(x_flat, right_gate_weight).view(bs,N,N,hidden).sigmoid()
    out_gate = torch.nn.functional.linear(x_flat, out_gate_weight).view(bs,N,N,hidden).sigmoid()

    # Apply mask if not nominal; mask shape [bs,N,N]; unsqueeze dim
    if mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1).type_as(left)  # broadcast to hidden
        left = left * mask
        right = right * mask
    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Call Triton kernel
    # allocate out tensor
    out = torch.empty((bs, N, N, hidden), dtype=torch.float32, device=input_tensor.device)

    # compute strides
    # left_stride: For contiguous tensors, last dim stride = 1, then N stride = hidden, ... Actually using .stride()
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left.stride()
    # same for right
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h = right.stride()
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h = out.stride()

    # Define BLOCK sizes based on N; choose e.g., BLOCK_M = BLOCK_N = 64, BLOCK_K = 32 (tunable)
    # For performance we may adapt based on N: typical 128x128.

    # Launch kernel
    grid = ( (N + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N, bs * hidden)
    batched_matmul_kernel[grid](
        left, right, out,
        bs, N, hidden,
        stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
        stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
        stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # out torch tensor

    # Apply to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Multiply by out_gate
    out = out * out_gate

    # final linear
    out_flat = out.view(-1, hidden)
    out = torch.nn.functional.linear(out_flat, to_out_weight).view(bs, N, N, dim)

    return out

@triton.jit
def batched_gemm_out_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)
    batch = pid_bh // H
    hidden = pid_bh % H
    # Compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for batch and hidden
    a_batch_ptr = a_ptr + batch * stride_a_batch + hidden * stride_a_h
    b_batch_ptr = b_ptr + batch * stride_b_batch + hidden * stride_b_h
    c_batch_ptr = c_ptr + batch * stride_c_batch + hidden * stride_c_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask for k
        mask_k = cur_k < N

        # Pointer to A tile: (i,k)
        a_ptrs = a_batch_ptr + offs_i[:, None] * stride_a_i + cur_k[None, :] * stride_a_k
        # Pointer to B_T tile: (k,j). B original dimensions (j,k) so we index with k first, j second.
        b_ptrs = b_batch_ptr + cur_k[:, None] * stride_b_j + offs_j[None, :] * stride_b_k

        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float16)

        # dot product accumulates in fp32
        acc += tl.dot(a, b).to(tl.float32)

    # Store result
    c_ptrs = c_batch_ptr + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor,
                                      normalized_shape=(dim,),
                                      weight=weights["norm.weight"],
                                      bias=weights["norm.bias"])
    # Flatten for linear layers
    x_flat = x.reshape(-1, dim)

    # Linear projections
    left = torch.nn.functional.linear(x_flat, weights["left_proj.weight"]).view(bs, N, N, hidden)
    right = torch.nn.functional.linear(x_flat, weights["right_proj.weight"]).view(bs, N, N, hidden)

    # Gates
    left_gate = torch.nn.functional.linear(x_flat, weights["left_gate.weight"]).view(bs, N, N, hidden).sigmoid()
    right_gate = torch.nn.functional.linear(x_flat, weights["right_gate.weight"]).view(bs, N, N, hidden).sigmoid()
    out_gate = torch.nn.functional.linear(x_flat, weights["out_gate.weight"]).view(bs, N, N, hidden).sigmoid()

    # Apply mask if provided and not nomask
    if not config.get("nomask", False):
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [bs,N,N,1]
        left = left * mask
        right = right * mask

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Convert to half for kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # Prepare output tensor
    out = torch.empty((bs, N, N, hidden), dtype=torch.float32, device=device)

    # Compute strides
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left_fp16.stride()
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h = right_fp16.stride()
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h = out.stride()

    # Choose block sizes
    if N <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64
    else:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64

    grid = ( (N + BLOCK_M - 1)//BLOCK_M,
             (N + BLOCK_N - 1)//BLOCK_N,
             bs * hidden )
    batched_gemm_out_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        bs,
        N,
        hidden,
        stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
        stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
        stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # LayerNorm on hidden dim
    out = torch.nn.functional.layer_norm(out,
                                         normalized_shape=(hidden,),
                                         weight=weights["to_out_norm.weight"],
                                         bias=weights["to_out_norm.bias"])
    # Apply out gate
    out = out * out_gate

    # Final projection
    out_flat = out.view(-1, hidden)
    out = torch.nn.functional.linear(out_flat, weights["to_out.weight"]).view(bs, N, N, dim)

    return out

"""
Triton implementation of the outgoing TriMul (Triangular Multiplicative Update) layer.

The operation consists of:
1. Layer‑norm on the input tensor.
2. Linear projections to a hidden dimension, optional mask application and
   element‑wise gating (sigmoid).
3. A heavy pair‑wise multiplication:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to a batched GEMM:
        out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].T
   This reduction is executed with a custom Triton kernel.
4. Layer‑norm on the hidden dimension, gating, and a final linear projection
   back to the original dimension.

All steps except the batched GEMM are performed with PyTorch for simplicity,
while the O(N²·hidden) reduction is fused in Triton for maximum performance.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for out = left @ right^T
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_out_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute, for each batch `b` and hidden channel `h`:
           C[b, :, :, h] = A[b, :, :, h] @ B[b, :, :, h].T
       where A and B have shape (B, N, N, H).
    """
    pid_i = tl.program_id(0)          # block row index
    pid_j = tl.program_id(1)          # block col index
    pid_bh = tl.program_id(2)         # combined batch‑hidden index

    batch = pid_bh // H               # batch id
    hidden = pid_bh % H               # hidden‑channel id

    #--- offsets for the i and j dimensions ---------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    #--- base pointers for this (batch, hidden) slice -----------------------
    a_base = a_ptr + batch * stride_a_batch + hidden * stride_a_h
    b_base = b_ptr + batch * stride_b_batch + hidden * stride_b_h
    c_base = c_ptr + batch * stride_c_batch + hidden * stride_c_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    #--- iterate over the reduction (k) dimension ---------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        cur_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)   # absolute k indices
        mask_k = cur_k < N

        # A tile: (i, k)
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + cur_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)  # dtype matches a_ptr (fp16)

        # B_T tile: (k, j)  -> we reinterpret B as transposed on‑the‑fly
        b_ptrs = b_base + cur_k[:, None] * stride_b_j + offs_j[None, :] * stride_b_k
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)  # dtype matches b_ptr (fp16)

        # dot → fp32 accumulator
        acc += tl.dot(a, b).to(tl.float32)

    #--- write the result ----------------------------------------------------
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]   (float32)
        - mask_tensor  : [B, N, N]       (bool or float, may be omitted)
        - weights_dict : mapping of the layer's parameters
        - config_dict  : contains "dim", "hidden_dim", "nomask", ...

    Returns
    -------
    output : torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Linear projections (float32)
    # --------------------------------------------------------------
    x_flat = x.reshape(-1, dim)                     # (B*N*N, dim)

    left = torch.nn.functional.linear(
        x_flat, weights["left_proj.weight"]
    ).view(B, N, N, hidden)

    right = torch.nn.functional.linear(
        x_flat, weights["right_proj.weight"]
    ).view(B, N, N, hidden)

    # --------------------------------------------------------------
    # 3. Gating values (float32)
    # --------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x_flat, weights["left_gate.weight"]
    ).view(B, N, N, hidden).sigmoid()

    right_gate = torch.nn.functional.linear(
        x_flat, weights["right_gate.weight"]
    ).view(B, N, N, hidden).sigmoid()

    out_gate = torch.nn.functional.linear(
        x_flat, weights["out_gate.weight"]
    ).view(B, N, N, hidden).sigmoid()

    # --------------------------------------------------------------
    # 4. Optional mask (broadcasted over the hidden dimension)
    # --------------------------------------------------------------
    if not config.get("nomask", False):
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 5. Apply element‑wise gates
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6. Convert to fp16 for the reduction kernel (half the memory traffic)
    # --------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # --------------------------------------------------------------
    # 7. Batched GEMM: out = left @ right^T   →   shape [B, N, N, hidden]
    # --------------------------------------------------------------
    out = torch.empty((B, N, N, hidden), dtype=torch.float32, device=device)

    # Strides for each tensor (in elements, not bytes)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left_fp16.stride()
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h = right_fp16.stride()
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h = out.stride()

    # Heuristic block sizes (tuned for typical N's)
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64
    else:  # up to 1024
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden,
    )

    batched_gemm_out_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        hidden,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_a_h,
        stride_b_batch,
        stride_b_j,
        stride_b_k,
        stride_b_h,
        stride_c_batch,
        stride_c_i,
        stride_c_j,
        stride_c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 8. Hidden‑dim LayerNorm
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------
    # 9. Apply the output gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 10. Final projection back to `dim`
    # --------------------------------------------------------------
    out = torch.nn.functional.linear(
        out.view(-1, hidden), weights["to_out.weight"]
    ).view(B, N, N, dim)

    return out
