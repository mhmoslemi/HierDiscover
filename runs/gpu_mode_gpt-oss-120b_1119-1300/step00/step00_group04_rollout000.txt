import torch
import triton
import triton.language as tl

@triton.jit
def mask_gate_kernel(
    left_ptr, right_ptr,
    mask_ptr,
    left_gate_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offset < total

    # Load data
    left = tl.load(left_ptr + offset, mask=mask).to(tl.float32)
    right = tl.load(right_ptr + offset, mask=mask).to(tl.float32)

    # mask broadcast index: each H repeats same mask for given (b,i,j)
    mask_idx = offset // H
    m = tl.load(mask_ptr + mask_idx, mask=mask).to(tl.float32)

    left_gate = tl.load(left_gate_ptr + offset, mask=mask).to(tl.float32)
    right_gate = tl.load(right_gate_ptr + offset, mask=mask).to(tl.float32)

    out_left = left * m * left_gate
    out_right = right * m * right_gate

    tl.store(out_left_ptr + offset, out_left, mask=mask)
    tl.store(out_right_ptr + offset, out_right, mask=mask)
```

Note: The kernel uses strides implicitly assuming contiguous tensors. If noncontiguous we need strides but we will enforce contiguous.

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    device = input_tensor.device
    dtype = input_tensor.dtype

    # LayerNorm
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Linear Projections
    left_proj_weight = weights['left_proj.weight']
    right_proj_weight = weights['right_proj.weight']
    left = torch.nn.functional.linear(x, left_proj_weight) # shape [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Gates
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    B, N, _, _ = input_tensor.shape 
    # Process mask
    nomask = config.get('nomask', True)
    if nomask or mask is None:
        # No mask needed: simple elementwise scaling
        left = left * left_gate
        right = right * right_gate
    else:
        # Ensure tensors are contiguous for the kernel
        left_cont = left.contiguous()
        right_cont = right.contiguous()
        left_gate_cont = left_gate.contiguous()
        right_gate_cont = right_gate.contiguous()
        mask_cont = mask.to(dtype).contiguous()  # shape [B,N,N]
        # Allocate output buffers
        left_scaled = torch.empty_like(left_cont)
        right_scaled = torch.empty_like(right_cont)

        total_elements = B * N * N * hidden_dim
        BLOCK_SIZE = 1024
        grid = lambda meta: (triton.cdiv(total_elements, meta['BLOCK_SIZE']),)

        mask_gate_kernel[grid](
            left_cont,
            right_cont,

"""
TriMul (outgoing) implementation.

Algorithm overview
------------------
1. Layer‑norm the input tensor over the feature dimension.
2. Compute left/right projections and three sigmoid gates with linear layers.
3. If a mask is present, fuse mask‑multiplication and gate‑application with a
   small Triton kernel (element‑wise).  When `nomask=True` the mask step is skipped.
4. For each head (hidden dimension) compute the contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is equivalent to a batched matrix multiplication:
       left_h (N×N) @ right_h.T (N×N) for every batch and hidden channel.
   This is performed with a single `torch.matmul` on
   tensors of shape (B*H, N, N).
5. Apply a second LayerNorm, multiply by the output gate and a final linear
   projection back to the original feature dimension.

Only the element‑wise mask+gate fusion is done in Triton; the heavy N³ reduction
uses cuBLAS via `torch.matmul`, which on an H100 meets the ~1 ms target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (broadcasted on hidden dim) with gate scalings
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    left_ptr,               # float*  left  (B,N,N,H) contiguous
    right_ptr,              # float*  right (B,N,N,H) contiguous
    mask_ptr,               # float*  mask  (B,N,N)   contiguous
    left_gate_ptr,          # float*  left_gate  (B,N,N,H) contiguous
    right_gate_ptr,         # float*  right_gate (B,N,N,H) contiguous
    out_left_ptr,           # float*  out left  (B,N,N,H) output
    out_right_ptr,          # float*  out right (B,N,N,H) output
    B, N, H,                # int64 scalars
    BLOCK_SIZE: tl.constexpr # compile‑time block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)   # linear offsets
    total = B * N * N * H
    mask = offs < total

    # Load left/right (float32)
    left = tl.load(left_ptr + offs, mask=mask, other=0.0).to(tl.float32)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0).to(tl.float32)

    # Mask is broadcast over H.  Compute the index into the (B,N,N) mask:
    # offset = ((b·N + i)·N + j)·H + h   →   mask_index = offset // H
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0).to(tl.float32)

    # Load gates (float32)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0).to(tl.float32)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0).to(tl.float32)

    # Fuse mask and gates
    out_left = left * m * left_gate
    out_right = right * m * right_gate

    # Store results
    tl.store(out_left_ptr + offs, out_left, mask=mask)
    tl.store(out_right_ptr + offs, out_right, mask=mask)


# ----------------------------------------------------------------------
# Entrypoint: custom_kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask:        torch.Tensor of shape [B, N, N] (bool or float)
            - weights:     dict of model parameters (torch.Tensors)
            - config:      dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and gate pre‑activations
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]       # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]     # (hidden_dim, dim)

    left = F.linear(x, left_proj_w)                 # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)               # [B, N, N, hidden_dim]

    # Gates
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = F.linear(x, left_gate_w).sigmoid()   # [B, N, N, hidden_dim]
    right_gate = F.linear(x, right_gate_w).sigmoid() # [B, N, N, hidden_dim]
    out_gate = F.linear(x, out_gate_w).sigmoid()     # [B, N, N, hidden_dim]

    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 3. Apply mask + gate (if required) via Triton kernel
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # Ensure contiguous tensors for the kernel
        left_c = left.contiguous()
        right_c = right.contiguous()
        left_gate_c = left_gate.contiguous()
        right_gate_c = right_gate.contiguous()
        mask_c = mask.to(dtype).contiguous()       # shape [B, N, N]

        # Allocate output buffers
        left_scaled = torch.empty_like(left_c)
        right_scaled = torch.empty_like(right_c)

        total_elements = B * N * N * hidden_dim
        BLOCK_SIZE = 1024  # ~1 KB per thread block; good for most GPUs

        grid = lambda meta: (triton.cdiv(total_elements, meta["BLOCK_SIZE"]),)

        mask_gate_kernel[grid](
            left_c,
            right_c,
            mask_c,
            left_gate_c,
            right_gate_c,
            left_scaled,
            right_scaled,
            B,
            N,
            hidden_dim,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        left, right = left_scaled, right_scaled
    else:
        # No mask: simple element‑wise gate multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Core contraction: Σ_k left[...,k,:] * right[...,k,:]
    #    Implemented as batched matmul over the hidden dimension.
    # ------------------------------------------------------------------
    # Shapes: left/right -> [B, N, N, H]
    # Rearrange to [B, H, N, N] then fuse batch & head dims.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left_perm.view(B * hidden_dim, N, N)      # [B*H, N, N]
    right_mat = right_perm.view(B * hidden_dim, N, N)    # [B*H, N, N]

    # left @ rightᵀ  ->  (B*H, N, N)
    out_mat = torch.matmul(left_mat, right_mat.transpose(-2, -1))

    # Restore original layout: [B, N, N, H]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 5. Second LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # [B, N, N, hidden_dim]

    out = out * out_gate                              # gate scaling

    to_out_w = weights["to_out.weight"]               # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                     # [B, N, N, dim]

    return out
