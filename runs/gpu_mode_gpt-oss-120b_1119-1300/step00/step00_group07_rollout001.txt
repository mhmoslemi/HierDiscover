@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(axis=0)   # i tile
    pid_n = tl.program_id(axis=1) # j tile
    pid_bh = tl.program_id(axis=2) # batch*hidden tile

    # compute b and h indices
    b = pid_bh // H
    h = pid_bh % H

    # compute i and j offsets
    offs_i = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # pointers base
    left_ptrs = left_ptr + b * stride_left_b + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k + h * stride_left_h
    right_ptrs = right_ptr + b * stride_right_b + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j + h * stride_right_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for start_k in range(0, N, BLOCK_K):
        cur_k = start_k + offs_k
        k_mask = cur_k < N
        # load left tile: shape (BLOCK_M, BLOCK_K)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & k_mask[None, :], other=0.0)

"""
TriMul (outgoing) – fast Triton implementation

The heavy part of the TriMul forward pass is the three‑dimensional
contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where ``left`` and ``right`` are the projected and gated pair tensors
of shape [B, N, N, H] (H = hidden_dim).  By flattening the batch and
hidden‑dim dimensions we obtain B·H independent matrix‑multiplications
of size N×N.  The kernel below computes one (i,j) tile for a given
(b, h) pair using a classic GEMM tiling strategy (BLOCK_M×BLOCK_N
output tile, BLOCK_K reduction tile).  The rest of the TriMul pipeline
(LayerNorms, linear projections, sigmoids, and final projection) is
performed with PyTorch for simplicity.

The kernel works for any batch size B∈{1,2}, sequence length N up to
1024 and hidden dimension H up to 384.  It supports optional masking
(and skips the mask multiplication when ``config["nomask"]`` is True).
The final output is float‑32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _trimul_einsum_kernel(
    left_ptr,            # const float*   left  [B,N,N,H]
    right_ptr,           # const float*   right [B,N,N,H]
    out_ptr,             # float*         out   [B,N,N,H]
    B, N, H,             # int32 scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute a tile of the outgoing TriMul contraction."""
    pid_i = tl.program_id(0)        # i‑tile index
    pid_j = tl.program_id(1)        # j‑tile index
    pid_bh = tl.program_id(2)       # combined batch * hidden index

    # De‑compose pid_bh into batch b and hidden dimension h
    b = pid_bh // H
    h = pid_bh % H

    # 1‑D offsets inside the tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K + offs_k
        mask_k = cur_k < N

        # Pointers for the current k‑tile
        # left[b, i, cur_k, h]  -> shape (BLOCK_M, BLOCK_K)
        left_tile_ptr = (
            left_ptr
            + b * stride_left_b
            + offs_i[:, None] * stride_left_i
            + cur_k[None, :] * stride_left_k
            + h * stride_left_h
        )
        # right[b, cur_k, j, h] -> we need (k, j) layout for dot
        right_tile_ptr = (
            right_ptr
            + b * stride_right_b
            + cur_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j
            + h * stride_right_h
        )

        left = tl.load(left_tile_ptr,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)
        right = tl.load(right_tile_ptr,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # Accumulate the matrix product
        acc += tl.dot(left, right)    # (BLOCK_M, BLOCK_K) • (BLOCK_K, BLOCK_N)

    # Write the result tile
    out_tile_ptr = (
        out_ptr
        + b * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(out_tile_ptr, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel for the
    O(N³) contraction.  All other operations (LayerNorm, linear projections,
    sigmoids) are performed with PyTorch.

    Args:
        data: (input, mask, weights, config)
            input   – torch.Tensor of shape [B, N, N, dim]
            mask    – torch.Tensor of shape [B, N, N] (may be all‑ones)
            weights – dict with all model parameters
            config  – dict containing "dim", "hidden_dim" and optional "nomask"

    Returns:
        Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper – LayerNorm (single‑dim normalization over the last axis)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias) → [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # Optional masking
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # mask is float (0/1) – broadcast over hidden dimension
        mask_f = mask.to(dtype=x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton kernel: compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)   # allocate output [B,N,N,hidden_dim]

    # Tile sizes – chosen to respect H100 shared‑memory limits
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * hidden_dim)

    # Strides (int64) for pointer arithmetic
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    _trimul_einsum_kernel[
        grid
    ](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=stride_l_b,
        stride_left_i=stride_l_i,
        stride_left_k=stride_l_k,
        stride_left_h=stride_l_h,
        stride_right_b=stride_r_b,
        stride_right_j=stride_r_i,   # j‑stride is the same as the second dimension stride
        stride_right_k=stride_r_k,
        stride_right_h=stride_r_h,
        stride_out_b=stride_o_b,
        stride_out_i=stride_o_i,
        stride_out_j=stride_o_j,
        stride_out_h=stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                 # H100 prefers higher warp count
    )

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b
    )   # [B,N,N,hidden_dim]

    out = out * out_gate                      # apply output gate

    to_out_w = weights["to_out.weight"]
    output = torch.nn.functional.linear(out, to_out_w)   # → [B,N,N,dim]

    return output
