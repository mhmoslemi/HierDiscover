"""
TriMul (outgoing) custom kernel for AlphaFold3 style pairwise updates.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor over its last dimension (`dim`).
2. Project the normalized tensor to a hidden dimension (`hidden_dim`) with
   two linear layers: `left_proj` and `right_proj`.
3. (Optional) apply a pairwise mask of shape [B, N, N] to both projections.
4. Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`) from the
   normalized input and multiply the projected tensors with their respective
   gates.
5. The core operation is a batched matrix multiplication of the form
   `out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]`.
   This is equivalent to a series of GEMMs:
       out_h = left_h @ right_h.T   (for every batch and hidden head h)
   The kernel `tri_mul_kernel` implements this GEMM using Triton, loading
   tiles of size (BLOCK_M, BLOCK_N) and accumulating with `tl.dot`.  The
   computation is performed in FP16 (to exploit Tensor Cores) with FP32
   accumulation for numerical stability.
6. Reshape the result back to [B, N, N, hidden_dim] and apply a second
   Layer‑norm (`to_out_norm`).  Multiply with `out_gate`.
7. Final linear projection (`to_out`) brings the tensor back to `dim`.

Only the heavy O(N³·hidden_dim) matrix multiplication is executed on
Triton; all other operations are done with PyTorch (which also runs on the GPU).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------------------
# Triton kernel: batched GEMM computing left @ right.T for each hidden head
# ---------------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    batch, M, N, K,                    # integer sizes
    stride_ab, stride_am, stride_ak,   # strides for A (batch, M, K)
    stride_bb, stride_bk, stride_bn,   # strides for B (batch, K, N) – B is accessed transposed
    stride_cb, stride_cm, stride_cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ B.T.
    A:  [batch, M, K]   (float16)
    B:  [batch, N, K]   (float16) – accessed as (K,N) to achieve the transpose
    C:  [batch, M, N]   (float32)
    """
    pid_batch = tl.program_id(0)   # batch * hidden head index
    pid_m = tl.program_id(1)       # tile row
    pid_n = tl.program_id(2)       # tile column

    # Compute the start offsets of the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize the accumulation buffer
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k_tile * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Pointers for the A tile (M x K)
        a_ptrs = A_ptr + pid_batch * stride_ab \
                      + offs_m[:, None] * stride_am \
                      + offs_k[None, :] * stride_ak

        # Pointers for the B tile (K x N) – we read B transposed
        b_ptrs = B_ptr + pid_batch * stride_bb \
                      + offs_k[:, None] * stride_bk \
                      + offs_n[None, :] * stride_bn

        # Masks to avoid out‑of‑bounds loads
        a_mask = (offs_m < M)[:, None] & (offs_k < K)[None, :]
        b_mask = (offs_k < K)[:, None] & (offs_n < N)[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # FP16
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # FP16

        # dot returns FP32 accumulation when out_dtype is set
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Write‑back C tile
    c_ptrs = C_ptr + pid_batch * stride_cb \
                    + offs_m[:, None] * stride_cm \
                    + offs_n[None, :] * stride_cn
    c_mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc, mask=c_mask)


# ---------------------------------------------------------------------------
# Public entry point
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator using a custom Triton kernel.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32, on CUDA)
            - mask_tensor:  torch.Tensor of shape [B, N, N] (bool or float, on CUDA)
            - weights_dict: mapping from string names to torch.Tensor weights
            - config_dict: must contain "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # -----------------------------------------------------------------------
    # unpack arguments
    # -----------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------------
    # Helper to fetch weight tensors
    # -----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # -----------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, dim]

    # -----------------------------------------------------------------------
    # 2) Linear projections to hidden_dim (no bias)
    # -----------------------------------------------------------------------
    # shape: [B, N, N, hidden_dim]
    left = F.linear(x, left_proj_weight)   # (hidden_dim, dim) weight
    right = F.linear(x, right_proj_weight)

    # -----------------------------------------------------------------------
    # 3) Optional mask handling (broadcast to the hidden dimension)
    # -----------------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.to(dtype=x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------------
    # 4) Compute sigmoid gates and apply them
    # -----------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate          # still float32
    right = right * right_gate

    # -----------------------------------------------------------------------
    # 5) Core tri‑multiplicative update (batched GEMM)
    #    left @ right.T for each head
    # -----------------------------------------------------------------------
    # Move hidden dimension to the batch axis to obtain a huge batch of GEMMs
    # shape after view: [B * hidden_dim, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(B * hidden_dim, N, N).to(torch.float16)
    right_flat = right_perm.view(B * hidden_dim, N, N).to(torch.float16)

    # Output buffer (fp32 accumulation)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # -----------------------------------------------------------------------
    # Launch Triton kernel
    # -----------------------------------------------------------------------
    BLOCK_M = 64   # tune these values if needed
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                       # batch axis (including hidden heads)
        (N + BLOCK_M - 1) // BLOCK_M,         # tiles in M dimension
        (N + BLOCK_N - 1) // BLOCK_N,         # tiles in N dimension
    )

    # Strides are expressed in element units, not bytes
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * hidden_dim,
        N,
        N,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------------
    # 6) Second LayerNorm over hidden_dim, apply out_gate, final projection
    # -----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate                     # broadcast over hidden_dim
    out = F.linear(out, to_out_weight)       # (dim, hidden_dim) weight

    return out
