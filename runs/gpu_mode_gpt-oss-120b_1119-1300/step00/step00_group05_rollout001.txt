def custom_kernel(data):
    (x, mask, weights, config) = data

    # x shape [B,N,N,dim]
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    # Convert mask if needed.
    if mask is None:
        mask = torch.ones((B,N,N), device=x.device, dtype=x.dtype)
    else:
        mask = mask.to(x.dtype)

    # parameters:
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]; etc.

    eps = 1e-5

    # Norm (LayerNorm) across dim
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    # using torch.nn.functional.linear
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # apply mask and gating using Triton if mask not all ones
    # note: left_proj/right_proj are shape [B,N,N,hidden_dim]
    if config.get("nomask", False):
        # mask is not used; simply multiply by gates
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # allocate destination tensors
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)
        # gather strides for tensors
        stride_b, stride_i, stride_j, stride_d = left_proj.stride()
        mask_stride_b, mask_stride_i, mask_stride_j = mask.stride()
        # launch kernel
        BLOCK_I = 64
        BLOCK_J = 64
        BLOCK_D = 32
        grid = (B, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
        fuse_mask_gate_kernel[grid](
            left_proj, right_proj,
            left_gate, right_gate,
            mask,
            left, right,
            B, N, hidden_dim,
            stride_b, stride_i, stride_j, stride_d,
            mask_stride_b, mask_stride_i, mask_stride_j,
            BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_D=BLOCK_D,
        )
    # Compute batched matrix multiplication across hidden_dim channels
    # left and right shape [B,N,N,hidden_dim]
    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2)  # B, D, N, N
    right_perm = right.permute(0, 3, 1, 2)  # B, D, N, N

    # Perform batch matmul: left_perm @ right_perm.transpose(-2,-1)

    out = torch.matmul(left_perm, right_perm.transpose(-2, -1))

    )
    # multiply with out_gate
    out = out * out_gate

    # final linear projection to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

def fuse_mask_gate_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    B, N, D,
    stride_b, stride_i, stride_j, stride_d,
    mask_stride_b, mask_stride_i, mask_stride_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_D: tl.constexpr
):
    # program ids
    b = tl.program_id(0)
    i_base = tl.program_id(1) * BLOCK_I
    j_base = tl.program_id(2) * BLOCK_J

    # range vectors
    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)
    d = tl.arange(0, BLOCK_D)

    # mask for valid indices
    i_valid = i < N
    j_valid = j < N
    d_valid = d < D

    # broadcast mask for (i,j,d)
    ij_valid = i_valid[:, None] & j_valid[None, :]
    ijd_valid = ij_valid[:, :, None] & d_valid[None, None, :]

    # compute offsets
    # compute base offsets for each dimension
    # offset = b*stride_b + i*stride_i + j*stride_j + d*stride_d
    offset = (b * stride_b
              + i[:, None, None] * stride_i
              + j[None, :, None] * stride_j
              + d[None, None, :] * stride_d)

    # load tensors
    left_proj = tl.load(left_proj_ptr + offset, mask=ijd_valid, other=0.0)
    right_proj = tl.load(right_proj_ptr + offset, mask=ijd_valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offset, mask=ijd_valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + offset, mask=ijd_valid, other=0.0)

    mask = mask[:, :, None]

    tl.store(left_out_ptr + offset, left_out, mask=ijd_valid)
    tl.store(right_out_ptr + offset, right_out, mask=ijd_valid)

"""
custom_kernel
==============

Highly‑optimized forward implementation of the *outgoing* Triangle Multiplicative
Update (TriMul) used in AlphaFold‑3 and related protein‑structure models.

Algorithm
---------
1️⃣  Layer‑Norm the input tensor `x` over the last dimension (`dim`).
2️⃣  Linear projections (`left_proj`, `right_proj`) and gating logits
    (`left_gate`, `right_gate`, `out_gate`) are computed with bias‑free
    linear layers.  The gating logits are passed through a sigmoid.
3️⃣  **Triton kernel** – a fused kernel that, for every element
    `(b,i,j,d)` (batch, row, column, hidden), computes

        left_out  = left_proj  * left_gate  * mask
        right_out = right_proj * right_gate * mask

    where `mask` has shape `[B,N,N]` and is broadcast over the hidden
    dimension.  The kernel works on 3‑D tiles (`BLOCK_I × BLOCK_J × BLOCK_D`)
    and uses only element‑wise arithmetic, guaranteeing very low launch
    overhead.
4️⃣  Batched matrix multiplication is performed independently for each
    hidden channel:

        out_d = left_out_d @ right_out_dᵀ   for d = 0 … hidden_dim‑1

    This is expressed with a single `torch.matmul` on tensors of shape
    `[B, hidden_dim, N, N]`.
5️⃣  A second Layer‑Norm (`to_out_norm`) is applied over the hidden
    dimension, followed by element‑wise multiplication with `out_gate`.
6️⃣  A final bias‑free linear projection maps the hidden dimension back
    to the original channel size (`dim`).

All operations except the fused mask‑gate step are performed with PyTorch
high‑performance kernels; the Triton kernel provides the only custom
GPU code, satisfying the requirement to “use Triton for at least part of the
operations”.  The implementation works for all shapes listed in the
specification and runs entirely in `float32` (the final output type).

"""

import torch
import triton
import triton.language as tl


@triton.jit
def fuse_mask_gate_kernel(
    left_proj_ptr,            # [B,N,N,D]  (float32)
    right_proj_ptr,           # [B,N,N,D]  (float32)
    left_gate_ptr,            # [B,N,N,D]  (float32)
    right_gate_ptr,           # [B,N,N,D]  (float32)
    mask_ptr,                 # [B,N,N]    (float32, values 0/1)
    left_out_ptr,             # [B,N,N,D]  (float32) – result
    right_out_ptr,            # [B,N,N,D]  (float32) – result
    B, N, D,                  # runtime scalars
    stride_b, stride_i, stride_j, stride_d,   # strides for the 4‑D tensors
    mask_stride_b, mask_stride_i, mask_stride_j, # strides for mask
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Fused element‑wise kernel:
        left_out  = left_proj  * left_gate  * mask
        right_out = right_proj * right_gate * mask
    The mask is broadcast to the hidden dimension.
    """
    # ------------------------------------------------------------------------
    # Program IDs
    # ------------------------------------------------------------------------
    b = tl.program_id(0)                     # batch index
    pid_i = tl.program_id(1)                 # tile over rows   (i)
    pid_j = tl.program_id(2)                 # tile over cols   (j)

    # ------------------------------------------------------------------------
    # Compute absolute coordinates inside the tile
    # ------------------------------------------------------------------------
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)   # (BLOCK_I,)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)   # (BLOCK_J,)
    d = tl.arange(0, BLOCK_D)                    # (BLOCK_D,)

    # ------------------------------------------------------------------------
    # Bounds checking
    # ------------------------------------------------------------------------
    i_valid = i < N
    j_valid = j < N
    d_valid = d < D

    # (BLOCK_I, BLOCK_J, BLOCK_D) – useful for loading/storing
    ij_valid   = i_valid[:, None] & j_valid[None, :]                # (I,J)
    ijd_valid  = ij_valid[:, :, None] & d_valid[None, None, :]     # (I,J,D)

    # ------------------------------------------------------------------------
    # Offsets for the 4‑D tensors (same stride layout for all four tensors)
    # ------------------------------------------------------------------------
    # offset = b*stride_b + i*stride_i + j*stride_j + d*stride_d
    offs = (b * stride_b
            + i[:, None, None] * stride_i
            + j[None, :, None] * stride_j
            + d[None, None, :] * stride_d)        # (I,J,D)

    # ------------------------------------------------------------------------
    # Load tensors (masked loads return 0.0 for out‑of‑bounds elements)
    # ------------------------------------------------------------------------
    left_proj   = tl.load(left_proj_ptr   + offs, mask=ijd_valid, other=0.0)
    right_proj  = tl.load(right_proj_ptr  + offs, mask=ijd_valid, other=0.0)
    left_gate   = tl.load(left_gate_ptr   + offs, mask=ijd_valid, other=0.0)
    right_gate  = tl.load(right_gate_ptr  + offs, mask=ijd_valid, other=0.0)

    # ------------------------------------------------------------------------
    # Load mask (no hidden dimension) and broadcast
    # ------------------------------------------------------------------------
    mask_offs = (b * mask_stride_b
                 + i[:, None] * mask_stride_i
                 + j[None, :] * mask_stride_j)     # (I,J)
    mask = tl.load(mask_ptr + mask_offs, mask=ij_valid, other=0.0)  # (I,J)
    mask = mask[:, :, None]                                   # (I,J,1)

    # ------------------------------------------------------------------------
    # Compute and store results
    # ------------------------------------------------------------------------
    left_out  = left_proj  * left_gate  * mask
    right_out = right_proj * right_gate * mask

    tl.store(left_out_ptr  + offs, left_out,  mask=ijd_valid)
    tl.store(right_out_ptr + offs, right_out, mask=ijd_valid)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output of shape [batch, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------------
    x, mask, weights, config = data
    # x : [B, N, N, dim]  (float32)
    # mask : [B, N, N]    (bool/float, 0/1)

    device = x.device
    dtype = x.dtype

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    # ------------------------------------------------------------------------
    # Helper to fetch weight tensors (they are already on the correct device)
    # ------------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]   # [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]  # [hidden_dim, dim]

    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]     # [dim, hidden_dim]

    eps = 1e-5

    # ------------------------------------------------------------------------
    # 1️⃣ Layer‑Norm over the last dimension (dim)
    # ------------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------------
    # 2️⃣ Linear projections & gating logits (bias‑free)
    # ------------------------------------------------------------------------
    # Linear: out = x @ weight.T   (torch.nn.functional.linear does this)
    left_proj  = torch.nn.functional.linear(x_norm, left_proj_weight)   # [B,N,N,hidden]
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)  # [B,N,N,hidden]

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # ------------------------------------------------------------------------
    # 3️⃣ Apply mask & gates (fused Triton kernel if a mask is present)
    # ------------------------------------------------------------------------
    if config.get("nomask", False):
        # No mask – just element‑wise gate multiplication
        left  = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # Ensure mask is float (0.0 / 1.0)
        mask_f = mask.to(dtype=dtype)
        # Allocate output containers
        left  = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # Strides for the 4‑D tensors (all share the same layout)
        stride_b, stride_i, stride_j, stride_d = left_proj.stride()
        mask_stride_b, mask_stride_i, mask_stride_j = mask_f.stride()

        # Kernel launch configuration
        BLOCK_I = 64          # rows per tile
        BLOCK_J = 64          # cols per tile
        BLOCK_D = 32          # hidden‑channel tile
        grid = (B,
                (N + BLOCK_I - 1) // BLOCK_I,
                (N + BLOCK_J - 1) // BLOCK_J)

        fuse_mask_gate_kernel[grid](
            left_proj, right_proj,
            left_gate, right_gate,
            mask_f,
            left, right,
            B, N, hidden_dim,
            stride_b, stride_i, stride_j, stride_d,
            mask_stride_b, mask_stride_i, mask_stride_j,
            BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_D=BLOCK_D,
        )
        # Synchronize – Triton kernels are async, but the following torch ops
        # will implicitly wait for completion.

    # ------------------------------------------------------------------------
    # 4️⃣ Batched matmul per hidden channel:
    #     out_d = left_d @ right_dᵀ   (for each d)
    # ------------------------------------------------------------------------
    # Rearrange to [B, D, N, N] to expose the hidden dimension as batch
    left_perm  = left.permute(0, 3, 1, 2)   # B, D, N, N
    right_perm = right.permute(0, 3, 1, 2)  # B, D, N, N

    # Batched matmul (torch uses highly‑optimized CUTLASS kernels)
    out = torch.matmul(left_perm, right_perm.transpose(-2, -1))   # B, D, N, N
    out = out.permute(0, 2, 3, 1)                               # B, N, N, D

    # ------------------------------------------------------------------------
    # 5️⃣ Second Layer‑Norm over the hidden dimension
    # ------------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight,
        bias=to_out_norm_bias, eps=eps
    )

    # ------------------------------------------------------------------------
    # 6️⃣ Multiply by out_gate (broadcasted over hidden dimension)
    # ------------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------------
    # 7️⃣ Final linear projection back to the original channel size
    # ------------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
