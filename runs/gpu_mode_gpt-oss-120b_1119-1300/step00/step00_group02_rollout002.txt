"""
Custom Triton kernel for the outgoing TriMul operator.

Algorithm:
1. Apply LayerNorm over the last `dim` dimension of the input.
2. Compute left/right projections and their gates using linear layers and sigmoid.
3. Apply mask (if present) and gating to obtain tensors L, R of shape
   [B, N, N, H] (H = hidden_dim). Cast L and R to float16 for memory bandwidth.
4. Triton kernel computes the O(N^3 * H) ternary multiplication:
   out[b,i,j,h] = sum_k L[b,i,k,h] * R[b,j,k,h]
   The kernel performs a batched matrix multiplication for each hidden channel
   using tiling (BLOCK_M, BLOCK_N, BLOCK_K). Accumulation is in float32.
5. The result undergoes a second LayerNorm (with learned weight/bias), gated by
   `out_gate`, and a final linear projection back to `dim`.
Only the O(N^3) contraction is executed on the GPU via Triton; all remaining
operations use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # strides (in elements)
    stride_bl, stride_li, stride_lk, stride_lh,
    stride_br, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    # compile‑time constants
    B, N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]"""
    # 3‑D grid: (M‑tiles, N‑tiles, batch*hidden)
    pid_bh = tl.program_id(2)
    batch_idx = pid_bh // H
    hidden_idx = pid_bh % H

    pid_m = tl.program_id(0)   # tile along i dimension
    pid_n = tl.program_id(1)   # tile along j dimension

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this batch & hidden channel
    base_left = left_ptr + batch_idx * stride_bl + hidden_idx * stride_lh
    base_right = right_ptr + batch_idx * stride_br + hidden_idx * stride_rh
    base_out = out_ptr + batch_idx * stride_ob + hidden_idx * stride_oh

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k = (N + BLOCK_K - 1) // BLOCK_K
    for k_tile in range(num_k):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load left tile (M × K) ----
        a_ptrs = base_left + offs_m[:, None] * stride_li + offs_k[None, :] * stride_lk
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # ---- load right tile transposed (K × N) ----
        # right is stored as (j, k); we need (k, j) for the dot product.
        b_ptrs = base_right + offs_k[:, None] * stride_rk + offs_n[None, :] * stride_rj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product accumulates into FP32
        acc += tl.dot(a, b)

    # write out the tile
    out_ptrs = base_out + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using Triton for the
    O(N³·H) contraction.
    Input `data` is a tuple:
        (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # unpack arguments
    input_tensor, mask_tensor, weights, config = data
    # configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the inner `dim` dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast and apply)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask shape [B,N,N] -> [B,N,N,1]
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating values
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = F.linear(x, left_gate_weight).sigmoid()   # [B,N,N,hidden]
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    # Apply gates (cast gates to half for the elementwise mult)
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to half precision for the O(N³) kernel
    # ------------------------------------------------------------------
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    B, N_, N2, H_ = left.shape
    assert N_ == N2, "Non‑square pairwise dimension not supported"
    assert H_ == hidden_dim, "Hidden dimension mismatch"

    # Output buffer (float32)
    out = torch.empty_like(left, dtype=torch.float32)

    # Compute strides (in elements)
    stride_bl = left.stride(0)
    stride_li = left.stride(1)
    stride_lk = left.stride(2)
    stride_lh = left.stride(3)

    stride_br = right.stride(0)
    stride_rj = right.stride(1)
    stride_rk = right.stride(2)
    stride_rh = right.stride(3)

    stride_ob = out.stride(0)
    stride_oi = out.stride(1)
    stride_oj = out.stride(2)
    stride_oh = out.stride(3)

    # Tiling parameters – chosen to fit H100 shared memory comfortably.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Launch grid
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    # Launch kernel (pointers are passed as integers)
    trimul_outgoing_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        stride_bl, stride_li, stride_lk, stride_lh,
        stride_br, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        B, N, hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Second LayerNorm over hidden dimension, apply out_gate, final proj
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # Apply output gate (already float32)
    out = out * out_gate

    # Final linear projection back to `dim`
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)  # no bias

    return out

"""
Outgoing TriMul Triton implementation.

Algorithm
---------
1. Layer‑norm the input over its last dimension.
2. Linear projections (left/right) and three gating linear layers (no bias).
3. Apply an optional mask and the per‑position gates.
4. Cast the gated tensors to fp16 and run a Triton kernel that computes
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   The kernel treats each hidden channel as an independent batch dimension
   and performs a tiled batched mat‑mul (BLOCK_M×BLOCK_N tiles, BLOCK_K
   reduction).  Accumulation uses fp32 for numerical stability.
5. Layer‑norm over the hidden dimension, multiply by the output gate and
   project back to the original feature dimension.
Only the O(N³·H) contraction is executed on the GPU via Triton; the rest
uses PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # strides (in elements)
    stride_bl, stride_li, stride_lk, stride_lh,
    stride_br, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    # compile‑time constants
    B, N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where tensors have shape [B, N, N, H] and H is the hidden dimension.
    The kernel tiles the i/j dimensions (M/N) and reduces over k.
    """
    pid_bh = tl.program_id(2)
    batch_idx = pid_bh // H
    hidden_idx = pid_bh % H

    pid_m = tl.program_id(0)   # i‑tile
    pid_n = tl.program_id(1)   # j‑tile

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this (batch, hidden) pair
    base_left = left_ptr + batch_idx * stride_bl + hidden_idx * stride_lh
    base_right = right_ptr + batch_idx * stride_br + hidden_idx * stride_rh
    base_out = out_ptr + batch_idx * stride_ob + hidden_idx * stride_oh

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑tiles
    num_k = (N + BLOCK_K - 1) // BLOCK_K
    for k_tile in range(num_k):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M × K)  – fp16
        a_ptrs = base_left + offs_m[:, None] * stride_li + offs_k[None, :] * stride_lk
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load right tile transposed (K × N) – we need (k,j) order
        b_ptrs = base_right + offs_k[:, None] * stride_rk + offs_n[None, :] * stride_rj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Matrix‑multiply and accumulate (FP32 result)
        acc += tl.dot(a, b)

    # Store the tile
    out_ptrs = base_out + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[:, None])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    `data` is a tuple:
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns a tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm over the feature dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣  Optional mask (broadcasted over the hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣  Gating (sigmoid) + element‑wise application
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣  Cast to fp16 for the heavy O(N³) contraction
    # ------------------------------------------------------------------
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    B, N, N2, H = left.shape
    assert N == N2, "Pairwise dimensions must be square"
    assert H == hidden_dim, "Hidden dimension mismatch"

    # Output buffer (float32 for the subsequent LayerNorm)
    out = torch.empty_like(left, dtype=torch.float32)

    # Strides in number of elements (not bytes)
    stride_bl = left.stride(0)
    stride_li = left.stride(1)
    stride_lk = left.stride(2)
    stride_lh = left.stride(3)

    stride_br = right.stride(0)
    stride_rj = right.stride(1)
    stride_rk = right.stride(2)
    stride_rh = right.stride(3)

    stride_ob = out.stride(0)
    stride_oi = out.stride(1)
    stride_oj = out.stride(2)
    stride_oh = out.stride(3)

    # Tiling parameters – chosen for H100 shared‑memory limits
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (tiles over i, tiles over j, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    # Launch Triton kernel
    trimul_outgoing_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        stride_bl, stride_li, stride_lk, stride_lh,
        stride_br, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        B, N, hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6️⃣  Second LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # final projection to `dim`

    return out
