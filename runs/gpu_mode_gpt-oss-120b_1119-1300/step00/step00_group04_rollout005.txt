def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    # Norm parameters
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Use torch.nn.functional.layer_norm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )
    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Apply mask if not nomask
    if not config.get("nomask", False):
        # mask shape [B,N,N]
        mask_ = mask.unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_
        right = right * mask_
    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Convert to half for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left.shape
    # Prepare output tensor
    out_fp32 = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides
    stride_b = left_fp16.stride(0)
    stride_i = left_fp16.stride(1)
    stride_k = left_fp16.stride(2)
    stride_h = left_fp16.stride(3)

    # out stride (same layout)
    stride_out_b = out_fp32.stride(0)
    stride_out_i = out_fp32.stride(1)
    stride_out_j = out_fp32.stride(2)
    stride_out_h = out_fp32.stride(3)

    # Choose block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_D = 16  # tile size for hidden dim

    # grid size
    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N
    num_h = (hidden_dim + BLOCK_D - 1) // BLOCK_D
    total_tiles = B * num_i * num_j * num_h

    # Launch kernel
    grid = (total_tiles,)
    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        B, N, hidden_dim,
        stride_b, stride_i, stride_k, stride_h,
        stride_b, stride_i, stride_k, stride_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D,
    )

    # Apply to_out_norm and to_out linear
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Apply layer norm over hidden_dim dimension
    out = torch.nn.functional.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    # Apply out_gate (broadcast)
    out = out * out_gate
    # Final linear to map back to dim
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    num_h = tl.cdiv(H, BLOCK_D)

    b = pid // (num_i * num_j * num_h)
    rem = pid % (num_i * num_j * num_h)
    i_tile = rem // (num_j * num_h)
    rem2 = rem % (num_j * num_h)
    j_tile = rem2 // num_h
    h_tile = rem2 % num_h

    # base indices
    i_off = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    d_off = h_tile * BLOCK_D + tl.arange(0, BLOCK_D)

    i_mask = i_off < N
    j_mask = j_off < N
    d_mask = d_off < H

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # loop over K dimension
    for k_start in range(0, N, BLOCK_K):
        k_off = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # left pointer: (M, K, D)
        left_ptrs = left_ptr + b * stride_left_b \
                    + i_off[:, None, None] * stride_left_i \
                    + k_off[None, :, None] * stride_left_k \
                    + d_off[None, None, :] * stride_left_h
        left = tl.load(left_ptrs, mask = i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :], other=0.0)
        left = left.to(tl.float32)

        # right transposed pointer: (K, N, D)
        right_ptrs = right_ptr + b * stride_right_b \
                     + k_off[:, None, None] * stride_right_k \
                     + j_off[None, :, None] * stride_right_i \
                     + d_off[None, None, :] * stride_right_h
        right = tl.load(right_ptrs, mask = k_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :], other=0.0)
        right = right.to(tl.float32)

        # compute dot for each d
        for db in tl.static_range(0, BLOCK_D):
            acc[:, :, db] += tl.dot(left[:, :, db], right[:, :, db])
    # end for k

    # store result
    out_ptrs = out_ptr + b * stride_out_b \
               + i_off[:, None, None] * stride_out_i \
               + j_off[None, :, None] * stride_out_j \
               + d_off[None, None, :] * stride_out_h
    tl.store(out_ptrs, acc, mask=i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :])
```

"""
TriMul (outgoing) implementation.

This code follows the forward pass of the reference PyTorch `TriMul` module
and fuses the costly O(N³·hidden_dim) contraction
```
out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
```
using a custom Triton kernel.  The surrounding LayerNorms, Linear
projections and gating are performed with PyTorch (cuBLAS) – only the
inner triple‑multiplication is handled by Triton.  The kernel works for
arbitrary batch size (≤2), sequence length N (≤1024) and hidden
dimension hidden_dim (≤384).  Computations are performed in float16
inside the kernel and accumulated in float32 to keep numerical stability.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,                     # *float16
    right_ptr,                    # *float16
    out_ptr,                      # *float32
    B, N, H,                      # int32 scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)

    # ----- grid decomposition -------------------------------------------------
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    num_h = tl.cdiv(H, BLOCK_D)

    b = pid // (num_i * num_j * num_h)
    rem = pid % (num_i * num_j * num_h)
    i_tile = rem // (num_j * num_h)
    rem2 = rem % (num_j * num_h)
    j_tile = rem2 // num_h
    h_tile = rem2 % num_h

    # ----- tile offsets -------------------------------------------------------
    i_off = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    d_off = h_tile * BLOCK_D + tl.arange(0, BLOCK_D)

    i_mask = i_off < N
    j_mask = j_off < N
    d_mask = d_off < H

    # accumulator for the (M,N,D) tile
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ----- k‑loop (the reduction dimension) ----------------------------------
    for k_start in range(0, N, BLOCK_K):
        k_off = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # left[b, i, k, d]  -> shape (M, K, D)
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + i_off[:, None, None] * stride_left_i
            + k_off[None, :, None] * stride_left_k
            + d_off[None, None, :] * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :],
            other=0.0,
        )
        left = left.to(tl.float32)

        # right transposed: treat as right[b, k, j, d]  -> shape (K, N, D)
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + k_off[:, None, None] * stride_right_k
            + j_off[None, :, None] * stride_right_i
            + d_off[None, None, :] * stride_right_h
        )
        right = tl.load(
            right_ptrs,
            mask=k_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :],
            other=0.0,
        )
        right = right.to(tl.float32)

        # ----- per‑channel GEMM (M×K)·(K×N) ------------------------------------
        for d_idx in tl.static_range(0, BLOCK_D):
            # (M, K) · (K, N) -> (M, N)
            acc[:, :, d_idx] += tl.dot(left[:, :, d_idx], right[:, :, d_idx])

    # ----- write back ---------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + i_off[:, None, None] * stride_out_i
        + j_off[None, :, None] * stride_out_j
        + d_off[None, None, :] * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask         : torch.Tensor [B, N, N]   (bool/float) – may be ignored.
            weights      : dict of parameter tensors.
            config       : dict with keys:
                "dim", "hidden_dim", "nomask" (bool) … (others ignored here).
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ---------- LayerNorm on the input ---------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ---------- Linear projections -------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"]) # [B,N,N,hidden]

    # ---------- Optional mask -------------------------------------------------
    if not config.get("nomask", False):
        # mask: [B, N, N] -> broadcast to [B,N,N,1]
        mask_f = mask.to(dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ---------- Gating --------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to half and allocate output buffer -----------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    B, N, _, _ = left_fp16.shape

    out_fp32 = torch.empty(
        (B, N, N, hidden_dim), dtype=torch.float32, device=device
    )

    # ---------- Strides (in elements) ----------------------------------------
    stride_left_b = left_fp16.stride(0)
    stride_left_i = left_fp16.stride(1)
    stride_left_k = left_fp16.stride(2)
    stride_left_h = left_fp16.stride(3)

    # right tensor has identical layout
    stride_right_b = right_fp16.stride(0)
    stride_right_i = right_fp16.stride(1)  # used as column stride (j)
    stride_right_k = right_fp16.stride(2)
    stride_right_h = right_fp16.stride(3)

    # output strides (float32)
    stride_out_b = out_fp32.stride(0)
    stride_out_i = out_fp32.stride(1)
    stride_out_j = out_fp32.stride(2)
    stride_out_h = out_fp32.stride(3)

    # ---------- Triton launch parameters --------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_D = 16  # tile size for the hidden dimension

    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N
    num_h = (hidden_dim + BLOCK_D - 1) // BLOCK_D
    total_tiles = B * num_i * num_j * num_h

    grid = (total_tiles,)

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ---------- Second LayerNorm (hidden) ------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # ---------- Apply out gate ------------------------------------------------
    out = out * out_gate

    # ---------- Final linear projection back to dim ---------------------------
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)  # bias is None

    return out
