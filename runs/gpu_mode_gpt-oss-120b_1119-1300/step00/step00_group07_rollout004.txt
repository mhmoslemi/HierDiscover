    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # shape B,N,N,H
    # Get strides
    stride_b = left.stride(0)
    stride_i = left.stride(1)
    stride_k = left.stride(2)

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Input shape: B, N, N, dim
    B, N, _, D = input_tensor.shape
    dtype = input_tensor.dtype
    device = input_tensor.device

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Projections
    left_proj_weight = weights["left_proj.weight"] # shape (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply mask if needed:
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)  # shape (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Compute multiplicative update via Triton kernel
    out = tri_mul_triton(left, right, BLOCK_M=64, BLOCK_N=64, BLOCK_K=32, BLOCK_H=32)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # Apply out_gate
    out = out * out_gate

    # Final linear projection
    to_out_weight = weights["to_out.weight"]  # shape (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)  # shape (B,N,N,dim)

"""
TriMul (outgoing) implementation with a custom Triton kernel.

The heavy part of TriMul is the tensor contraction
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
which is equivalent to a batched matrix multiplication
    out[b, :, :, h] = left[b, :, :, h] @ right[b, :, :, h].transpose(1, 2).

We fuse this contraction across the hidden dimension `h` using a Triton
kernel.  The kernel tiles the (i, j) output plane (BLOCK_M×BLOCK_N) and
the reduction dimension `k` (BLOCK_K) while processing a small chunk of
the hidden channels (BLOCK_H) at a time.  Accumulation is performed in
float32 for numerical stability.  All other operations (LayerNorm,
Linear projections, gates, mask) are carried out with PyTorch.

The kernel works for any batch size (≤2), sequence length up to 1024 and
hidden dimension up to 384.  Boundary handling is done via masks, so
non‑divisible tile sizes are supported.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel for the core N^2·H reduction (left @ right^T)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_k, stride_h,
    out_stride_b, out_stride_i, out_stride_j, out_stride_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile
    pid_bh = tl.program_id(2)         # combined batch & hidden‑block index

    # --------------------------------------------------------------
    # Derive batch index and hidden‑block index from pid_bh
    # --------------------------------------------------------------
    num_h_blocks = tl.cdiv(H, BLOCK_H)
    b = pid_bh // num_h_blocks
    h_block = pid_bh % num_h_blocks
    h_start = h_block * BLOCK_H + tl.arange(0, BLOCK_H)

    # --------------------------------------------------------------
    # Offsets for the i, j and hidden dimensions
    # --------------------------------------------------------------
    i_start = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_start = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_start < N
    j_mask = j_start < N
    h_mask = h_start < H

    # accumulator of shape (BLOCK_M, BLOCK_N, BLOCK_H)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in tiles
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_start = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_start < N

        # ----- load left block: shape (M, K, H) -----
        left_offsets = (
            b * stride_b
            + i_start[:, None, None] * stride_i
            + k_start[None, :, None] * stride_k
            + h_start[None, None, :] * stride_h
        )
        left = tl.load(
            left_ptr + left_offsets,
            mask=i_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :],
            other=0.0,
        ).to(tl.float32)          # (M, K, H)

        # ----- load right block with k as leading dimension: shape (K, N, H) -----
        right_offsets = (
            b * stride_b
            + k_start[:, None, None] * stride_k
            + j_start[None, :, None] * stride_i
            + h_start[None, None, :] * stride_h
        )
        right = tl.load(
            right_ptr + right_offsets,
            mask=k_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],
            other=0.0,
        ).to(tl.float32)          # (K, N, H)

        # ----- inner reduction over hidden channels -----
        for h_idx in range(0, BLOCK_H):
            # left[:, :, h] is (M, K), right[:, :, h] is (K, N)
            acc[:, :, h_idx] += tl.dot(left[:, :, h_idx], right[:, :, h_idx])
    # --------------------------------------------------------------
    # Store the result
    # --------------------------------------------------------------
    out_offsets = (
        b * out_stride_b
        + i_start[:, None, None] * out_stride_i
        + j_start[None, :, None] * out_stride_j
        + h_start[None, None, :] * out_stride_h
    )
    tl.store(
        out_ptr + out_offsets,
        acc,
        mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def tri_mul_triton(left: torch.Tensor, right: torch.Tensor,
                   BLOCK_M: int = 64, BLOCK_N: int = 64,
                   BLOCK_K: int = 32, BLOCK_H: int = 32) -> torch.Tensor:
    """
    Batched TriMul using Triton.
    left, right : tensors of shape [B, N, N, H] (contiguous, float32/float16)
    Returns out of same shape and dtype (float32 accumulator -> float32 output).
    """
    assert left.shape == right.shape, "left and right must have identical shapes"
    B, N, _, H = left.shape
    device = left.device
    # Ensure contiguous layout for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # Strides in elements (not bytes)
    stride_b = left.stride(0)
    stride_i = left.stride(1)   # stride over the first N dimension
    stride_k = left.stride(2)   # stride over the second N dimension (the reduction dim)
    stride_h = left.stride(3)

    out_stride_b = out.stride(0)
    out_stride_i = out.stride(1)
    out_stride_j = out.stride(2)
    out_stride_h = out.stride(3)

    # Grid dimensions
    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    num_h_blocks = (H + BLOCK_H - 1) // BLOCK_H
    grid = (grid_i, grid_j, B * num_h_blocks)

    # Launch kernel
    tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, H,
        stride_b, stride_i, stride_k, stride_h,
        out_stride_b, out_stride_i, out_stride_j, out_stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,   # reasonable default for H100
    )
    return out


# ----------------------------------------------------------------------
# Public entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    data = (input_tensor, mask, weights, config)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension `dim`
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and gates
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 3. Apply mask (if present) and gates
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        # mask: [B, N, N] -> broadcast over hidden_dim
        mask = mask.to(dtype=dtype)
        mask_exp = mask.unsqueeze(-1)      # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Core TriMul contraction (batched matmul per hidden slice)
    # ------------------------------------------------------------------
    # The Triton kernel works in float32; if inputs are half, they are up‑cast.
    out = tri_mul_triton(left, right,
                         BLOCK_M=64, BLOCK_N=64, BLOCK_K=32, BLOCK_H=32)
    # The kernel returns float32; cast back to original dtype for following layers
    out = out.to(dtype)

    # ------------------------------------------------------------------
    # 5. Output layer‑norm + out_gate
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 6. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
