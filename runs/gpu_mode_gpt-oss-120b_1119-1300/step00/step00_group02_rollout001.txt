import triton
import triton.language as tl

@triton.jit
def mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j,
    stride_right_b, stride_right_i, stride_right_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j,
    stride_rgate_b, stride_rgate_i, stride_rgate_j,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_SIZE: tl.constexpr
):
    # compute offsets
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < B * N * N * H

    # compute linear indices
    # offsets: index into flatten of (B,N,N,H) where H is innermost.
    # Compute: index = ((b * N + i) * N + j) * H + d
    # So:
    # idx_h = offs % H
    # tmp = offs // H
    # idx_j = tmp % N
    # tmp2 = tmp // N
    # idx_i = tmp2 % N
    # idx_b = tmp2 // N
    idx_d = offs % H
    tmp = offs // H
    idx_j = tmp % N
    tmp2 = tmp // N
    idx_i = tmp2 % N
    idx_b = tmp2 // N

    # compute mask offset
    mask_offset = idx_b * stride_mask_b + idx_i * stride_mask_i + idx_j * stride_mask_j

    mask_val = tl.load(mask_ptr + mask_offset, mask=mask, other=0.0)

    # left offsets:
    left_offset = idx_b * stride_left_b + idx_i * stride_left_i + idx_j * stride_left_j + idx_d
    left_val = tl.load(left_ptr + left_offset, mask=mask, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + left_offset, mask=mask, other=0.0)
    left_gate_sig = tl.sigmoid(left_gate_val)

    left_out = left_val * mask_val * left_gate_sig
    tl.store(left_ptr + left_offset, left_out, mask=mask)  # in-place

    # same for right (use same offsets as left since shape same)
    right_val = tl.load(right_ptr + left_offset, mask=mask, other=0.0)  # using same offset
    right_gate_val = tl.load(right_gate_ptr + left_offset, mask=mask, other=0.0)
    right_gate_sig = tl.sigmoid(right_gate_val)

    right_out = right_val * mask_val * right_gate_sig
    tl.store(right_ptr + left_offset, right_out, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False) # Provided? Could be config includes "nomask"? In test maybe not; but config might include "nomask"? In test config include "nomask": boolean. So we handle.

    # Get weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Move all to same dtype (float32)
    # input_tensor shape: [B, N, N, dim]

    # LayerNorm over last dim
    eps = 1e-5
    # compute normalized x: (x - mean)/sqrt(var+eps)*weight + bias
    # using torch.nn.functional.layer_norm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Compute projections
    # Linear: out = x @ weight.T ; since weights stored as (out_features, in_features)
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Compute gates (pre-sigmoid)
    left_gate = torch.nn.functional.linear(x, left_gate_weight)
    right_gate = torch.nn.functional.linear(x, right_gate_weight)
    out_gate = torch.nn.functional.linear(x, out_gate_weight)

    # Apply mask and sigmoid gating using Triton
    if not nomask:
        # mask may be bool/float; ensure float
        mask_f = mask.to(dtype=x.dtype)
    else:
        # all ones mask
        mask_f = torch.ones_like(x[..., 0])  # shape [B,N,N]

    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_f = mask_f.contiguous()

    # Apply mask and gates via kernel
    left, right = apply_mask_gate(left, right, left_gate, right_gate, mask_f)

    # out_gate after sigmoid
    out_gate = torch.sigmoid(out_gate)

    # Compute pairwise product out = left @ right^T for each hidden dim and batch
    B, N, N2, H = left.shape
    assert N == N2
    # reshape to batch*H
    left_reshape = left.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H, N, N)
    right_reshape = right.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H, N, N)
    # out_reshape = left @ right^T
    out_reshape = torch.bmm(left_reshape, right_reshape.transpose(1, 2))  # (B*H, N, N)
    # reshape back
    out = out_reshape.reshape(B, H, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply out_gate
    out = out * out_gate

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
TriMul (outgoing) – Fast forward pass.

The pipeline follows the reference PyTorch implementation but fuses the
element‑wise mask, left‑gate and right‑gate operations into a single Triton
kernel.  The heavy N³ pairwise multiplication is carried out as a batched
matrix multiplication (`torch.bmm`) on reshaped tensors, which uses the
high‑performance cuBLAS kernels on H100.  All LayerNorms and linear
projections are performed with `torch.nn.functional` for simplicity.

Key steps
---------
1. Layer‑norm over the last channel (`dim`).
2. Linear projections → `left`, `right`.
3. Compute pre‑sigmoid gates (`left_gate`, `right_gate`, `out_gate`).
4. **Triton kernel** multiplies `left` and `right` by the broadcast mask
   and by `sigmoid(left_gate)` / `sigmoid(right_gate)` in‑place.
5. Compute the pairwise product:
   `out_d = left_d @ right_d.T` for each hidden channel `d` using a
   reshaped batched‑gemm.
6. Layer‑norm over `hidden_dim`, apply `out_gate` and final linear
   projection back to `dim`.

The kernel works for any batch size (1‑2), sequence length up to 1024,
`dim` ∈ {128,384} and `hidden_dim` = 128.  It handles both masked and
unmasked inputs – when `config["nomask"]` is True the mask is replaced by
all‑ones.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused mask × left‑gate / right‑gate update
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    left_ptr, right_ptr,                     # [B, N, N, H] (in‑place)
    left_gate_ptr, right_gate_ptr,           # pre‑sigmoid gates, same shape
    mask_ptr,                                # [B, N, N] (float32)
    B, N, H,                                 # static sizes
    stride_left_b, stride_left_i, stride_left_j, stride_left_d,
    stride_right_b, stride_right_i, stride_right_j, stride_right_d,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_d,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_d,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place mask & gate application for `left` and `right`.

    For every element (b,i,j,d):
        left[b,i,j,d]  = left[b,i,j,d]  * mask[b,i,j] * sigmoid(left_gate[b,i,j,d])
        right[b,i,j,d] = right[b,i,j,d] * mask[b,i,j] * sigmoid(right_gate[b,i,j,d])
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < B * N * N * H

    # -------- decode linear index into (b,i,j,d) --------------------
    d = offs % H
    tmp = offs // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # -------- load mask (no hidden dim) ----------------------------
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    mask_val = tl.load(mask_ptr + mask_off, mask=active, other=0.0)

    # -------- left side -------------------------------------------
    left_off = (b * stride_left_b + i * stride_left_i + j * stride_left_j
                + d * stride_left_d)
    left_val = tl.load(left_ptr + left_off, mask=active, other=0.0)

    left_gate_off = (b * stride_lgate_b + i * stride_lgate_i + j * stride_lgate_j
                     + d * stride_lgate_d)
    left_gate_val = tl.load(left_gate_ptr + left_gate_off, mask=active, other=0.0)

    left_out = left_val * mask_val * tl.sigmoid(left_gate_val)
    tl.store(left_ptr + left_off, left_out, mask=active)

    # -------- right side ------------------------------------------
    right_off = (b * stride_right_b + i * stride_right_i + j * stride_right_j
                 + d * stride_right_d)
    right_val = tl.load(right_ptr + right_off, mask=active, other=0.0)

    right_gate_off = (b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j
                      + d * stride_rgate_d)
    right_gate_val = tl.load(right_gate_ptr + right_gate_off, mask=active, other=0.0)

    right_out = right_val * mask_val * tl.sigmoid(right_gate_val)
    tl.store(right_ptr + right_off, right_out, mask=active)


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def apply_mask_and_gates(left, right, left_gate, right_gate, mask):
    """
    In‑place updates `left` and `right` using mask and the corresponding gates.
    All tensors must be contiguous and reside on the same CUDA device.
    """
    B, N, N2, H = left.shape
    assert N == N2, "left/right must be square in the last two dims"
    total = B * N * N * H
    BLOCK_SIZE = 256                     # works well for all test sizes
    grid = (triton.cdiv(total, BLOCK_SIZE),)

    # Strides (in elements, not bytes) – kept explicit for clarity
    sL_b, sL_i, sL_j, sL_d = left.stride()
    sR_b, sR_i, sR_j, sR_d = right.stride()
    sLG_b, sLG_i, sLG_j, sLG_d = left_gate.stride()
    sRG_b, sRG_i, sRG_j, sRG_d = right_gate.stride()
    sM_b, sM_i, sM_j = mask.stride()

    mask_gate_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask,
        B, N, H,
        sL_b, sL_i, sL_j, sL_d,
        sR_b, sR_i, sR_j, sR_d,
        sLG_b, sLG_i, sLG_j, sLG_d,
        sRG_b, sRG_i, sRG_j, sRG_d,
        sM_b, sM_i, sM_j,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4  # enough parallelism on H100
    )
    return left, right


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim]  (float32, CUDA)
            - mask_tensor : [B, N, N]       (float32 or bool, CUDA)
            - weights_dict: contains all model parameters.
            - config_dict: must contain "dim" and "hidden_dim"
                           and may contain "nomask" (bool).

    Returns:
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Load weights (all tensors are already on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    eps = 1e-5  # same as nn.LayerNorm default

    # ------------------------------------------------------------------
    # 1. Input layer‑norm over the channel dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Compute (pre‑sigmoid) gate tensors
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_weight)   # [B, N, N, H]
    right_gate = F.linear(x, right_gate_weight)  # [B, N, N, H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4. Optional mask handling
    # ------------------------------------------------------------------
    if nomask:
        # all‑ones mask (float32) – shape [B, N, N]
        mask = torch.ones_like(x[..., 0], dtype=x.dtype, device=x.device)
    else:
        mask = mask_tensor.to(dtype=x.dtype)

    # Ensure contiguous layout for the kernel
    left  = left.contiguous()
    right = right.contiguous()
    left_gate  = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask = mask.contiguous()

    # ------------------------------------------------------------------
    # 5. Fuse mask & gate using Triton
    # ------------------------------------------------------------------
    left, right = apply_mask_and_gates(left, right, left_gate, right_gate, mask)

    # ------------------------------------------------------------------
    # 6. Pairwise multiplication: einsum("... i k d, ... j k d -> ... i j d")
    #    Implemented as a batched matrix multiplication over the hidden channel.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_reshaped  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # out_d = left_d @ right_d.T   (sum over k)
    out_reshaped = torch.bmm(left_reshaped, right_reshaped.transpose(1, 2))

    # restore original shape -> [B, N, N, H]
    out = out_reshaped.view(B, H, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 7. Output layer‑norm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 8. Apply out‑gate (element‑wise) and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # bias is False in the reference implementation

    return out
