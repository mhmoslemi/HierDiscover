"""
custom_kernel : TriMul “outgoing” forward implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (B, N, N, C) with the given weight/bias.
2. Linear projections to a hidden dimension H (no bias):
   left  = x_norm @ left_proj_weight.T
   right = x_norm @ right_proj_weight.T
3. Optional elementwise mask (broadcast on the hidden dim).
4. Compute three gate tensors from the normalized input (sigmoid): 
   left_gate, right_gate, out_gate.
5. Apply gates: left  = left  * left_gate,  right = right * right_gate.
6. Compute the “tri‑multiplicative” update:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   → for each (b,h) this is a matrix multiplication
        (N×N) × (N×N)^T .
   We reshape the tensors so that each (b,h) pair becomes one batch
   entry and launch a fused Triton GEMM kernel (batched A @ B).
7. Layer‑norm the result (hidden dimension), multiply by out_gate,
   and project back to the original channel dimension C.

The heavy N³ reduction is performed by the Triton kernel, while
the cheap linear / gating / layer‑norm ops stay in PyTorch for
readability and to reuse highly‑optimised kernels.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM kernel (A @ B, A: (M×K), B: (K×N))
# ----------------------------------------------------------------------
@triton.jit
def _triton_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    # tensor dimensions
    BATCH, M, N, K,
    # strides (in elements)
    stride_batch_a, stride_m, stride_k,
    stride_batch_b, stride_kb, stride_n,
    stride_batch_c, stride_mc, stride_nc,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)

    # Number of blocks in each dimension
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    # Decode batch index and block coordinates from the 1‑D pid
    batch_idx = pid // (num_m * num_n)
    block_id  = pid % (num_m * num_n)
    block_m   = block_id // num_n
    block_n   = block_id % num_n

    # Offsets for the output tile
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to the current A and B tiles
        a_ptrs = (A_ptr + batch_idx * stride_batch_a
                  + offs_m[:, None] * stride_m
                  + offs_k[None, :] * stride_k)
        b_ptrs = (B_ptr + batch_idx * stride_batch_b
                  + offs_k[:, None] * stride_kb
                  + offs_n[None, :] * stride_n)

        # Load with boundary checks
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate the dot product
        acc += tl.dot(a, b)

    # Write back the output tile
    c_ptrs = (C_ptr + batch_idx * stride_batch_c
              + offs_m[:, None] * stride_mc
              + offs_n[None, :] * stride_nc)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B using Triton.
    A : (batch, M, K)   row‑major, contiguous
    B : (batch, K, N)   row‑major, contiguous
    Returns C : (batch, M, N) same dtype/device as A.
    """
    assert A.is_contiguous() and B.is_contiguous()
    BATCH, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    # Output allocation
    C = torch.empty((BATCH, M, N), dtype=A.dtype, device=A.device)

    # Strides in number of elements (not bytes)
    stride_batch_a = A.stride(0)
    stride_m = A.stride(1)          # = K
    stride_k = A.stride(2)          # = 1

    stride_batch_b = B.stride(0)
    stride_kb = B.stride(1)         # = N
    stride_n = B.stride(2)          # = 1

    stride_batch_c = C.stride(0)
    stride_mc = C.stride(1)         # = N
    stride_nc = C.stride(2)         # = 1

    # Heuristic block sizes (tuned for H100 FP32 matmul)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid size: one program per (batch, tile_m, tile_n)
    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = BATCH * num_blocks_m * num_blocks_n
    grid = (total_programs,)

    _triton_gemm_kernel[grid](
        A_ptr=A,
        B_ptr=B,
        C_ptr=C,
        BATCH=BATCH,
        M=M,
        N=N,
        K=K,
        stride_batch_a=stride_batch_a,
        stride_m=stride_m,
        stride_k=stride_k,
        stride_batch_b=stride_batch_b,
        stride_kb=stride_kb,
        stride_n=stride_n,
        stride_batch_c=stride_batch_c,
        stride_mc=stride_mc,
        stride_nc=stride_nc,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        # force 32‑bit indexing for large N (up to 1024 → safe)
        num_warps=4,
    )
    return C


# ----------------------------------------------------------------------
# Entry point required by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on the same CUDA device).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    # Shapes
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]   # c_z in the statement

    # ------------------------------------------------------------------
    # LayerNorm on the last channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])    # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Apply mask if supplied (mask shape: [B,N,N])
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Compute gating tensors from the normalized input (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"])
    )

    # Apply the gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton GEMM: for each (batch, hidden) compute left @ rightᵀ
    # ------------------------------------------------------------------
    # Reshape to (B*hidden, N, N) where the second tensor is already transposed
    left_f = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, seq_len, seq_len)
    right_f = right.permute(0, 3, 2, 1).contiguous().view(bs * hidden_dim, seq_len, seq_len)
    # (batch, N, N) @ (batch, N, N)  -> (batch, N, N)
    out_f = _batched_gemm_triton(left_f, right_f)

    # Restore original 4‑D layout
    out = out_f.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Output LayerNorm + final gate + projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                          # elementwise gating
    out = F.linear(out, weights["to_out.weight"])  # final projection to dim

    return out
