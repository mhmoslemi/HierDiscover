import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_i, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """Compute:
    out[b, h, i, j] = sum_k left[b, h, i, k] * right[b, h, k, j]
    using a block‑wise GEMM.
    """
    pid_i = tl.program_id(0)  # block row i
    pid_j = tl.program_id(1)  # block col j
    pid_bh = tl.program_id(2)  # flattened batch * hidden

    # decode batch and hidden index
    h = pid_bh % H
    b = pid_bh // H

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k-blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            b * stride_l_b + h * stride_l_h + \
            offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile transposed: we need right[k, j] so shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + \
            b * stride_r_b + h * stride_r_h + \
            offs_k[:, None] * stride_r_i + offs_j[None, :] * stride_r_k
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # accumulate the GEMM product
        acc += tl.dot(left, right)   # (M,K) @ (K,N) -> (M,N)

    # store result
    out_ptrs = out_ptr + \
        b * stride_o_b + h * stride_o_h + \
        offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

TriMul outgoing operator (AlphaFold3) – Triton‑accelerated core contraction.

The heavy computation is the three‑dimensional einsum
    out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]
which can be expressed as a batched matrix multiplication
    out[b, d] = left[b, d] @ right[b, d]ᵀ
for each batch b and hidden channel d.  The kernel tiles the (i, j) output
plane and accumulates over k using `tl.dot`, yielding a fast GEMM per
(b, d) pair.  All surrounding LayerNorm, Linear and gating steps are
performed in PyTorch; the contraction itself is executed on the GPU via
the Triton kernel below.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_i, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute a block‑wise matrix multiply:
        out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, k, j]

    Arguments
    ---------
    left_ptr, right_ptr, out_ptr : pointers to the 4‑D tensors
    B, H, N                     : batch size, hidden dimension, sequence length
    stride_*                     : strides (in elements) for each tensor dimension
    BLOCK_M, BLOCK_N, BLOCK_K    : compile‑time tile sizes
    """
    pid_i = tl.program_id(0)          # block row index   (i dimension)
    pid_j = tl.program_id(1)          # block col index   (j dimension)
    pid_bh = tl.program_id(2)         # flattened batch × hidden index

    # decode batch and hidden indices
    h = pid_bh % H
    b = pid_bh // H

    # tile offsets for i / j
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k in blocks of size BLOCK_K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N
