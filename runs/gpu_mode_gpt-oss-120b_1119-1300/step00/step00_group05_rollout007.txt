    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_a_batch, stride_a_h, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_h, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_h, stride_c_i, stride_c_j,

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch and hidden indices
    batch_idx = pid // H
    hidden_idx = pid % H

def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    
    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    
    # Projections
    left = torch.nn.functional.linear(x, left_proj_weight)      # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_weight)
    
    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))
    
    # Apply mask (maybe all ones) via Triton kernel
    B, N, _, H = left.shape
    if mask_tensor is None:
        mask = torch.ones((B,N,N), dtype=left.dtype, device=left.device)
    else:
        mask = mask_tensor.to(left.dtype)
    
    # Prepare output tensors for masked left/right
    left_masked = torch.empty_like(left)
    right_masked = torch.empty_like(right)
    
    # Kernel constants
    BLOCK_N = 64
    BLOCK_H = 32
    
    # Define Triton kernel for mask + gates (mask already applied, gates later)
    @triton.jit
    def apply_mask_kernel(
        left_in_ptr, right_in_ptr, mask_ptr,
        left_out_ptr, right_out_ptr,
        B, N, H,
        stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
        stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
        stride_m_batch, stride_m_i, stride_m_k,
        BLOCK_N: tl.constexpr, BLOCK_H: tl.constexpr):
        pid_b = tl.program_id(0)
        pid_i = tl.program_id(1)
        pid_j = tl.program_id(2)

        # i/j offsets
        offs_i = pid_i * BLOCK_N + tl.arange(0, BLOCK_N)
        offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

        # mask tile
        mask_tile_ptr = mask_ptr + pid_b * stride_m_batch + offs_i[:, None] * stride_m_i + offs_j[None, :] * stride_m_k
        mask_tile = tl.load(mask_tile_ptr, mask=(offs_i[:, None] < N) & (offs_j[None, :] < N), other=0.0)
        mask_tile = mask_tile[:, :, None]   # broadcast to hidden dim

        # loop over hidden tiles
        num_h_tiles = tl.cdiv(H, BLOCK_H)
        for h_tile in range(num_h_tiles):
            hid_start = h_tile * BLOCK_H
            offs_h = hid_start + tl.arange(0, BLOCK_H)
            mask_h = offs_h < H

            # left
            left_ptr = left_in_ptr + pid_b * stride_l_batch + offs_i[:, None, None] * stride_l_i + offs_j[None, :, None] * stride_l_k + offs_h[None, None, :] * stride_l_h
            left_val = tl.load(left_ptr, mask=(offs_i[:, None, None] < N) & (offs_j[None, :, None] < N) & mask_h[None, None, :], other=0.0)

            # right
            right_ptr = right_in_ptr + pid_b * stride_r_batch + offs_i[:, None, None] * stride_r_i + offs_j[None, :, None] * stride_r_k + offs_h[None, None, :] * stride_r_h
            right_val = tl.load(right_ptr, mask=(offs_i[:, None, None] < N) & (offs_j[None, :, None] < N) & mask_h[None, None, :], other=0.0)

            # store masked values (mask multiplication only)
            left_out_ptr = left_out_ptr + pid_b * stride_l_batch + offs_i[:, None, None] * stride_l_i + offs_j[None, :, None] * stride_l_k + offs_h[None, None, :] * stride_l_h
            right_out_ptr = right_out_ptr + pid_b * stride_r_batch + offs_i[:, None, None] * stride_r_i + offs_j[None, :, None] * stride_r_k + offs_h[None, None, :] * stride_r_h

            tl.store(left_out_ptr, left_val * mask_tile, mask=(offs_i[:, None, None] < N) & (offs_j[None, :, None] < N) & mask_h[None, None, :])
            tl.store(right_out_ptr, right_val * mask_tile, mask=(offs_i[:, None, None] < N) & (offs_j[None, :, None] < N) & mask_h[None, None, :])
    
    # compute strides
    l_str = left.stride()
    r_str = right.stride()
    m_str = mask.stride()  # (B,N,N) shape

    grid = (B, (N + BLOCK_N - 1) // BLOCK_N, (N + BLOCK_N - 1) // BLOCK_N)
    apply_mask_kernel[grid](
        left, right, mask,
        left_masked, right_masked,
        B, N, H,
        l_str[0], l_str[1], l_str[2], l_str[3],
        r_str[0], r_str[1], r_str[2], r_str[3],
        m_str[0], m_str[1], m_str[2],
        BLOCK_N=BLOCK_N,
        BLOCK_H=BLOCK_H
    )
    # Now left_masked and right_masked are left*mask and right*mask
    # Multiply by gates using PyTorch broadcasting
    left = left_masked * left_gate
    right = right_masked * right_gate
    
    # Compute outer product (triangular multiplicative update) using a batched GEMM kernel
    # Permute to [B,H,N,N] and transpose right for matmul
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,2,1).contiguous()
    
    out_perm = torch.empty_like(left_perm)  # same shape
    
    # Get strides
    a_str = left_perm.stride()
    b_str = right_perm.stride()
    c_str = out_perm.stride()
    
    # Matmul kernel constants
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    
    grid_mat = (B * H, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)
    
    @triton.jit
    def tri_mul_gemm_kernel(
        a_ptr, b_ptr, c_ptr,
        B, N, H,
        stride_a_batch, stride_a_h, stride_a_i, stride_a_k,
        stride_b_batch, stride_b_h, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_h, stride_c_i, stride_c_j,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
        pid = tl.program_id(0)
        pid_i = tl.program_id(1)
        pid_j = tl.program_id(2)
        batch_idx = pid // H
        hidden_idx = pid % H

        offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_i = offs_i < N
        mask_j = offs_j < N

        # base pointers for this (batch, hidden)
        a_base = a_ptr + batch_idx * stride_a_batch + hidden_idx * stride_a_h
        b_base = b_ptr + batch_idx * stride_b_batch + hidden_idx * stride_b_h
        c_base = c_ptr + batch_idx * stride_c_batch + hidden_idx * stride_c_h

        # pointers for each k-tile
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
        b_ptrs = b_base + tl.arange(0, BLOCK_K)[:, None] * stride_b_k + offs_j[None, :] * stride_b_n
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        for k in range(0, tl.cdiv(N, BLOCK_K)):
            a = tl.load(a_ptrs, mask=mask_i[:, None] & (tl.arange(0, BLOCK_K) + k*BLOCK_K < N)[None, :], other=0.0)
            b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K) + k*BLOCK_K)[:, None] < N & mask_j[None, :], other=0.0)
            # Cast to float32 for accumulation
            a = a.to(tl.float32)
            b = b.to(tl.float32)
            acc += tl.dot(a, b)
            # advance pointers
            a_ptrs += BLOCK_K * stride_a_k
            b_ptrs += BLOCK_K * stride_b_k

        c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
        tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
    
    tri_mul_gemm_kernel[grid_mat](
        left_perm, right_perm, out_perm,
        B, N, H,
        a_str[0], a_str[1], a_str[2], a_str[3],
        b_str[0], b_str[1], b_str[2], b_str[3],
        c_str[0], c_str[1], c_str[2], c_str[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    
    # permute back to original shape
    out = out_perm.permute(0,2,3,1).contiguous()  # [B,N,N,H]
    
    # Output norm
    out = torch.nn.functional.layer_norm(out, (H,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # shape [B,N,N,dim]
    
    return out

@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_a_batch, stride_a_h, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_h, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_h, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)   # combined batch*hidden index
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    batch_idx = pid // H
    hidden_idx = pid % H

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (batch, hidden) slice
    a_base = a_ptr + batch_idx * stride_a_batch + hidden_idx * stride_a_h
    b_base = b_ptr + batch_idx * stride_b_batch + hidden_idx * stride_b_h
    c_base = c_ptr + batch_idx * stride_c_batch + hidden_idx * stride_c_h

    # pointers for each iteration of k
    a_ptrs = a_base + offs_i[:, None] * stride_a_i + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    b_ptrs = b_base + tl.arange(0, BLOCK_K)[:, None] * stride_b_k + offs_j[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):

        k_offset = k * BLOCK_K
        cur_k = k_offset + tl.arange(0, BLOCK_K)

"""
TriMul “outgoing” forward implementation with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension.
2. Linear projections → left, right (shape [B, N, N, H]).
3. Compute three gates (left, right, out) with a sigmoid.
4. Apply the optional mask and the left/right gates (pure PyTorch, cheap).
5. Compute the triangular multiplicative update:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for every hidden dimension.
   We fuse it into a Triton kernel that treats each pair (batch, hidden)
   as an independent GEMM (A @ Bᵀ) with shape (N × N).
6. Layer‑norm over the hidden dimension, apply the output gate and a final
   linear projection back to the original channel size.
All heavy compute (step 5) runs on the GPU via Triton; the rest uses
high‑performance PyTorch ops.
"""

import torch
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: batched GEMM for the triangular multiplication
# -------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_a_batch, stride_a_h, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_h, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_h, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program_id(0) → combined (batch, hidden) index
    pid = tl.program_id(0)
    pid_i = tl.program_id(1)   # tile row
    pid_j = tl.program_id(2)   # tile column

    # decode batch and hidden indices
    batch_idx = pid // H
    hidden_idx = pid % H

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (batch, hidden) slice
    a_base = a_ptr + batch_idx * stride_a_batch + hidden_idx * stride_a_h
    b_base = b_ptr + batch_idx * stride_b_batch + hidden_idx * stride_b_h
    c_base = c_ptr + batch_idx * stride_c_batch + hidden_idx * stride_c_h

    # pointers that will be stepped in the k‑loop
    a_ptrs = a_base + offs_i[:, None] * stride_a_i + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    b_ptrs = b_base + tl.arange(0, BLOCK_K)[:, None] * stride_b_k + offs_j[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K
        cur_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # promote to float32 for stable accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)

        # advance pointers to the next k‑block
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # store the result
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# -------------------------------------------------------------
# Main entry point used by the evaluation harness
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights, config)

        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask_tensor  : torch.Tensor [B, N, N] (may be None)
        - weights      : dict of learned parameters
        - config       : dict containing 'dim' and 'hidden_dim'

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack and prepare
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # -------------------------------------------------------------
    # extract model weights (already on device)
    # -------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -------------------------------------------------------------
    # 1. layer‑norm over the last dimension
    # -------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )  # [B, N, N, dim]

    # -------------------------------------------------------------
    # 2. linear projections (no bias)
    # -------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B, N, N, H]

    # -------------------------------------------------------------
    # 3. compute gates (sigmoid after a linear projection)
    # -------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # -------------------------------------------------------------
    # 4. optional mask (if provided) and gate application
    # -------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask_tensor is None or config.get("nomask", False):
        mask = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask = mask_tensor.to(dtype)
    # broadcast mask over the hidden dimension
    mask = mask.unsqueeze(-1)          # [B, N, N, 1]
    left = left * mask * left_gate    # element‑wise
    right = right * mask * right_gate

    # -------------------------------------------------------------
    # 5. triangular multiplicative update via a batched GEMM kernel
    #    left : [B, N, N, H] → [B, H, N, N] (i‑k matrix)
    #    right: [B, N, N, H] → [B, H, N, N] (j‑k matrix, transposed to k‑j)
    # -------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] (k‑j)

    out_perm = torch.empty_like(left_perm)  # will hold [B, H, N, N]

    # strides for the three tensors (batch, hidden, i, k/j)
    a_str = left_perm.stride()
    b_str = right_perm.stride()
    c_str = out_perm.stride()

    # block sizes – tuned for H100, can be adjusted if needed
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_gemm_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B,
        N,
        hidden_dim,
        a_str[0],
        a_str[1],
        a_str[2],
        a_str[3],
        b_str[0],
        b_str[1],
        b_str[2],
        b_str[3],
        c_str[0],
        c_str[1],
        c_str[2],
        c_str[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # permute back to the original layout: [B, N, N, H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------------------
    # 6. final normalisation, output gate and projection back to dim
    # -------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
