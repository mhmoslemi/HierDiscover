@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_z = tl.program_id(0) # combined batch and channel
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # unpack batch and channel
    b = pid_z // D
    d = pid_z % D

    # compute start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask OOB
    mask_m = offs_m < N
    mask_n = offs_n < N

    # compute base pointer offsets for left (i,k) block
    # pointer: left_ptr + b*stride_lb + d*stride_ld
    # Build k loop
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # k loop
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offsets (shared across i and j)
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block (M,K)
        # left_ptr + b*stride_lb + offs_m[:,None]*stride_li + offs_k[None,:]*stride_lk + d*stride_ld
        left_ptrs = left_ptr + b*stride_lb + (offs_m[:, None] * stride_li) + (offs_k[None, :] * stride_lk) + d*stride_ld
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right block transposed (K,N) as needed for matmul: right[b, j, k, d] -> we need right_T[k, j]
        # right_ptr + b*stride_rb + offs_k[:,None]*stride_rk + offs_n[None,:]*stride_rj + d*stride_rd
        right_ptrs = right_ptr + b*stride_rb + (offs_k[:, None] * stride_rk) + (offs_n[None, :] * stride_rj) + d*stride_rd
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Compute dot product
        acc += tl.dot(left, right)   # left (M,K), right (K,N) => (M,N)

    # Apply mask for i,j OOB
    acc = tl.where(mask_m[:, None] & mask_n[None, :], acc, 0.0)

    # Store result
    out_ptrs = out_ptr + b*stride_ob + (offs_m[:, None] * stride_oi) + (offs_n[None, :] * stride_oj) + d*stride_od
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None,:])

"""
TriMul “outgoing” forward pass implemented with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (over the last dimension).
2. Linear projections -> left/right (shape [B,N,N,H]) where H = hidden_dim.
3. Optional pairwise mask multiplication (broadcast on the hidden dimension).
4. Gating: sigmoid(linear(x)) for left_gate, right_gate and out_gate.
   Apply left_gate/right_gate to the projected tensors.
5. Compute the expensive N³ reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply where each hidden‑dim slice performs
   L @ Rᵀ.  The reduction is done in a custom Triton kernel that works on
   FP16 data but accumulates in FP32.
6. Layer‑norm over the hidden dimension, multiply by out_gate and a final
   linear projection back to the original feature dimension.

The kernel is tiled in the (i,j) plane (BLOCK_M×BLOCK_N) and iterates over
the reduction axis k in blocks of size BLOCK_K.  The batch dimension B and
the hidden dimension H are fused into the first grid axis, so each
program instance processes a single (b, h) pair.

The implementation falls back to PyTorch for all lightweight operations,
ensuring correctness while the N³ part is heavily accelerated.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: one (batch, hidden) pair → GEMM of (N×N) @ (N×N)ᵀ
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_ld,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rd,   # right strides
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using tiling in (i,j) and reduction over k.
    """
    # --------------------------------------------------------------
    # Program IDs
    # --------------------------------------------------------------
    pid_z = tl.program_id(0)          # combined batch & hidden axis
    pid_m = tl.program_id(1)          # tile row (i)
    pid_n = tl.program_id(2)          # tile col (j)

    # Decode batch and hidden index from pid_z
    b = pid_z // H
    h = pid_z % H

    # Offsets for the M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bounds check for i and j
    mask_m = offs_m < N
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Accumulator (FP32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k_tile in range(num_k):
        # K‑axis offsets for this tile
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load left block: shape (M, K)
        #   pointer = b*lb + i*li + k*lk + h*ld
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + (offs_m[:, None] * stride_li)
            + (offs_k[None, :] * stride_lk)
            + h * stride_ld
        )
        left = tl.load(
            left_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # promote to FP32 for accumulation

        # ------------------------------------------------------------------
        # Load right block transposed: shape (K, N)
        #   we need right[b, j, k, h] → treat as (K, N) by swapping axes.
        #   pointer = b*rb + k*rk + j*rj + h*rd
        # ------------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + (offs_k[:, None] * stride_rk)
            + (offs_n[None, :] * stride_rj)
            + h * stride_rd
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # ------------------------------------------------------------------
        # GEMM for this K‑tile
        # ------------------------------------------------------------------
        acc += tl.dot(left, right)   # (M, K) @ (K, N) → (M, N)

    # Zero out-of‑bounds results (optional but safe)
    acc = tl.where(mask_m[:, None] & mask_n[None, :], acc, 0.0)

    # ------------------------------------------------------------------
    # Store the result (convert back to FP16)
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + (offs_m[:, None] * stride_oi)
        + (offs_n[None, :] * stride_oj)
        + h * stride_od
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Entry point called by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]            # input / output feature dimension
    hidden_dim = config["hidden_dim"]  # projection dimension
    eps = 1e-5                     # LayerNorm epsilon

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # shape: [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # → [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape: [B, N, N] → [B, N, N, 1] → broadcast
        mask_bc = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_bc
        right = right * mask_bc

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear projections of the normalized input)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy N³ reduction
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    B, N, _, H = left_fp16.shape   # H == hidden_dim

    # Allocate output tensor (FP16) for the kernel
    out_fp16 = torch.empty((B, N, N, H), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # Strides (in elements, not bytes) for pointer arithmetic in Triton
    # ------------------------------------------------------------------
    stride_lb, stride_li, stride_lk, stride_ld = left_fp16.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right_fp16.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out_fp16.stride()

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64   # rows per tile  (tune for H100 – 64/128 work well)
    BLOCK_N = 64   # cols per tile
    BLOCK_K = 64   # reduction tile size
    grid = (
        B * H,                              # combined batch & hidden dimension
        (N + BLOCK_M - 1) // BLOCK_M,       # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,       # j‑tiles
    )

    # Launch the kernel
    trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_ld,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rd,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to FP32 for the remaining layers
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # Post‑processing: hidden‑dim LayerNorm, out_gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
