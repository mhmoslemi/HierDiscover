```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    B, N,
    stride_x_batch, stride_x_i, stride_x_j, stride_x_d,
    stride_y_batch, stride_y_i, stride_y_j, stride_y_d,
    eps,
    BLOCK_D: tl.constexpr,
    D: tl.constexpr,
):
    """LayerNorm over the last dimension (size D) of a [B,N,N,D] tensor.
    Each program (pid) processes a single (b,i,j) element.
    """
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode linear index into b,i,j
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # base pointers for the row (b,i,j)
    x_base = x_ptr + b * stride_x_batch + i * stride_x_i + j * stride_x_j
    y_base = y_ptr + b * stride_y_batch + i * stride_y_i + j * stride_y_j

    # -------- first pass: compute mean and variance ----------
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)
    offset = 0
    while offset < D:
        cur_len = tl.minimum(BLOCK_D, D - offset)
        # indices within this block
        d_idx = tl.arange(0, cur_len) + offset
        # load
        x = tl.load(x_base + d_idx * stride_x_d)
        sum_val += tl.sum(x)
        sum_sq += tl.sum(x * x)
        offset += BLOCK_D

    mean = sum_val / D
    var = sum_sq / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # -------- second pass: write normalized values ----------
    offset = 0
    while offset < D:
        cur_len = tl.minimum(BLOCK_D, D - offset)
        d_idx = tl.arange(0, cur_len) + offset
        x = tl.load(x_base + d_idx * stride_x_d)
        # normalize
        norm = (x - mean) * inv_std
        # weight and bias (both are 1‑D tensors)
        w = tl.load(weight_ptr + d_idx)
        b_val = tl.load(bias_ptr + d_idx)
        y = norm * w + b_val
        tl.store(y_base + d_idx * stride_y_d, y)
        offset += BLOCK_D
```

def custom_kernel(data):
    """
    Triton-backed forward pass of the TriMul “outgoing” operator.
    The kernel only implements the first LayerNorm; the remaining
    linear / gating / cubic contraction is performed with highly
    optimized PyTorch primitives (linear + torch.matmul).
    """
    import torch
    import torch.nn.functional as F
    input_tensor, mask, weights, config = data
    # shapes
    B, N, N2, dim = input_tensor.shape
    assert N == N2, "pairwise tensor must be square"
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. LayerNorm over the channel dimension (dim) – executed in Triton
    # -----------------------------------------------------------------
    # allocate output buffer
    x_norm = torch.empty_like(input_tensor)

    # strides for input and output (contiguous)
    stride_x = input_tensor.stride()
    stride_y = x_norm.stride()
    # launch kernel, each program handles one (b,i,j) entry
    total = B * N * N
    grid = (total,)
    layernorm_kernel[grid](
        input_tensor,                # x_ptr
        x_norm,                      # y_ptr
        weights["norm.weight"],      # weight_ptr
        weights["norm.bias"],        # bias_ptr
        B, N,
        stride_x[0], stride_x[1], stride_x[2], stride_x[3],
        stride_y[0], stride_y[1], stride_y[2], stride_y[3],
        eps,
        BLOCK_D=128,                 # dim ≤ 384, 128 works for all
        D=dim,                       # compile‑time constant for the kernel
    )

    # -----------------------------------------------------------------
    # 2. Project to hidden dimension and compute gates (pure PyTorch)
    # -----------------------------------------------------------------
    left_proj   = F.linear(x_norm, weights["left_proj.weight"])
    right_proj  = F.linear(x_norm, weights["right_proj.weight"])
    left_gate   = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate  = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate    = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 3. Apply optional mask and element‑wise gates
    # -----------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        # mask shape [B,N,N] -> broadcast over hidden dimension
        mask = mask.to(dtype=left_proj.dtype).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    left  = left_proj * left_gate
    right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 4. Cubic contraction: sum over the pairwise dimension k
    #    (equivalent to a batched matrix multiply per hidden channel)
    # -----------------------------------------------------------------
    # reshape to (B, hidden_dim, N, N) and multiply L @ R^T
    L = left.permute(0, 3, 1, 2)            # [B, H, N, N]
    R = right.permute(0, 3, 2, 1)           # [B, H, N, N]
    out = torch.matmul(L, R)                # [B, H, N, N]
    out = out.permute(0, 2, 3, 1)           # [B, N, N, H]

    # -----------------------------------------------------------------
    # 5. Output layer‑norm, gating and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass – Triton‑accelerated LayerNorm + PyTorch kernels.

The kernel `layernorm_kernel` normalises the last dimension of a
[B, N, N, C] tensor (C = dim) using the provided γ (weight) and β (bias).
All other heavy operations (linear projections, gating, the cubic
contraction and the final post‑processing) are performed with highly
optimised PyTorch primitives (F.linear, torch.matmul and F.layer_norm).
The cubic contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched matrix multiplication per hidden channel:
L = left.permute(0,3,1,2) (B, H, N, N)
R = right.permute(0,3,2,1) (B, H, N, N)  →  out = L @ R
The result is cast back to FP32, normalised, gated and projected to the
original feature dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    x_ptr,                 # *const float32    input  [B,N,N,C]
    y_ptr,                 # *mut   float32    output [B,N,N,C]
    weight_ptr,            # *const float32    γ (C,)
    bias_ptr,              # *const float32    β (C,)
    B, N,                  # int32 scalar batch size and sequence length
    stride_x_batch, stride_x_i, stride_x_j, stride_x_c,
    stride_y_batch, stride_y_i, stride_y_j, stride_y_c,
    eps,                   # float32 scalar epsilon
    BLOCK_C: tl.constexpr,  # block size for the channel dimension (C)
    C: tl.constexpr,        # channel dimension (known at compile time)
):
    """
    LayerNorm over the last dimension C of a [B,N,N,C] tensor.
    One program instance processes a single (b,i,j) element.
    """
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode pid → (b,i,j)
    bn = N * N
    b = pid // bn
    rem = pid % bn
    i = rem // N
    j = rem % N

    # base pointers for the (b,i,j) vector of length C
    x_base = x_ptr + b * stride_x_batch + i * stride_x_i + j * stride_x_j
    y_base = y_ptr + b * stride_y_batch + i * stride_y_i + j * stride_y_j

    # -------- first pass: compute mean and variance ----------
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)

    # loop over C in chunks of BLOCK_C (C is a compile‑time constant)
    for start in range(0, C, BLOCK_C):
        cur = tl.minimum(BLOCK_C, C - start)
        c_idx = tl.arange(0, cur) + start               # indices within this chunk
        x = tl.load(x_base + c_idx * stride_x_c)        # (cur,)
        sum_val += tl.sum(x)                            # scalar
        sum_sq  += tl.sum(x * x)                        # scalar

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # -------- second pass: write normalised values ----------
    for start in range(0, C, BLOCK_C):
        cur = tl.minimum(BLOCK_C, C - start)
        c_idx = tl.arange(0, cur) + start
        x = tl.load(x_base + c_idx * stride_x_c)
        norm = (x - mean) * inv_std                      # (cur,)
        w = tl.load(weight_ptr + c_idx)                  # γ
        b_ = tl.load(bias_ptr + c_idx)                   # β
        y = norm * w + b_
        tl.store(y_base + c_idx * stride_y_c, y)


def custom_kernel(data):
    """
    Triton‑backed forward pass of the TriMul “outgoing” operator.
    Input `data` = (input_tensor, mask, weights, config).
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    # -----------------------------------------------------------------
    # unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, N2, dim = input_tensor.shape
    assert N == N2, "pairwise tensor must be square"
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. LayerNorm over the channel dimension (dim) – run on Triton
    # -----------------------------------------------------------------
    x_norm = torch.empty_like(input_tensor)                # [B,N,N,dim]

    # strides for the contiguous tensors
    sx = input_tensor.stride()    # (N*N*dim, N*dim, dim, 1)
    sy = x_norm.stride()

    total_positions = B * N * N
    grid = (total_positions,)

    # launch the Triton kernel – BLOCK_C = 128 works for all supported dim (≤384)
    layernorm_kernel[grid](
        input_tensor,                # x_ptr
        x_norm,                      # y_ptr
        weights["norm.weight"],      # γ
        weights["norm.bias"],        # β
        B, N,
        sx[0], sx[1], sx[2], sx[3],
        sy[0], sy[1], sy[2], sy[3],
        eps,
        BLOCK_C=128,
        C=dim,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections to hidden_dim and gating (pure PyTorch)
    # -----------------------------------------------------------------
    left_proj  = F.linear(x_norm, weights["left_proj.weight"])    # [B,N,N,H]
    right_proj = F.linear(x_norm, weights["right_proj.weight"])   # [B,N,N,H]

    left_gate  = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 3. Optional mask – broadcast over hidden_dim
    # -----------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        # mask: [B,N,N] → [B,N,N,1]
        mask = mask.to(dtype=left_proj.dtype).unsqueeze(-1)
        left_proj  = left_proj * mask
        right_proj = right_proj * mask

    # -----------------------------------------------------------------
    # 4. Apply element‑wise gates
    # -----------------------------------------------------------------
    left  = left_proj * left_gate       # [B,N,N,H]
    right = right_proj * right_gate     # [B,N,N,H]

    # -----------------------------------------------------------------
    # 5. Cubic contraction (batched matmul per hidden channel)
    #    Using FP16 for the heavy matmul to improve throughput.
    # -----------------------------------------------------------------
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape to (B, H, N, N) and compute L @ Rᵀ
    L = left_h.permute(0, 3, 1, 2)        # [B, H, N, N]
    R = right_h.permute(0, 3, 2, 1)       # [B, H, N, N] (transpose of the second N)
    out_h = torch.matmul(L, R)           # [B, H, N, N] (FP16 matmul)

    out = out_h.to(torch.float32).permute(0, 2, 3, 1)   # back to [B,N,N,H]

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                                 # gate
    out = F.linear(out, weights["to_out.weight"])        # final projection → [B,N,N,dim]

    return out
