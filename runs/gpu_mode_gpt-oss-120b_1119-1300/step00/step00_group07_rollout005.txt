@triton.jit
def layernorm_kernel(

   x_ptr, y_ptr, weight_ptr, bias_ptr,

   B, N, C, eps,

   stride_x0, stride_x1, stride_x2, stride_c,

   stride_y0, stride_y1, stride_y2, stride_yc,

   BLOCK_C: tl.constexpr

):

   pid = tl.program_id(0)

   total = B * N * N

   if pid >= total: return

   b = pid // (N * N)

   rem = pid % (N * N)

   i = rem // N

   j = rem % N

   # compute base pointer
   x_base = x_ptr + b*stride_x0 + i*stride_x1 + j*stride_x2

   y_base = y_ptr + b*stride_y0 + i*stride_y1 + j*stride_y2

   # First loop: compute sum, sum_sq
   sum_val = tl.zeros((1,), dtype=tl.float32)
   sum_sq = tl.zeros((1,), dtype=tl.float32)

   for off in range(0, C, BLOCK_C):
       cur = off + tl.arange(0, BLOCK_C)
       mask = cur < C
       x = tl.load(x_base + cur * stride_c, mask=mask, other=0.0)
       sum_val += tl.sum(x, axis=0)
       sum_sq += tl.sum(x * x, axis=0)

   mean = sum_val / C
   var = sum_sq / C - mean * mean
   inv_std = tl.rsqrt(var + eps)

   # Second pass: compute normalized and store
   for off in range(0, C, BLOCK_C):
       cur = off + tl.arange(0, BLOCK_C)
       mask = cur < C
       x = tl.load(x_base + cur * stride_c, mask=mask, other=0.0)
       # x_norm = (x - mean) * inv_std
       x_norm = (x - mean) * inv_std
       # apply weight and bias if provided, else just output normalized
       if weight_ptr != 0:
           w = tl.load(weight_ptr + cur, mask=mask, other=0.0)
           b_val = tl.load(bias_ptr + cur, mask=mask, other=0.0)
           y = x_norm * w + b_val
       else:
           y = x_norm
       # store
       tl.store(y_base + cur * stride_yc, y, mask=mask)

"""
TriMul (outgoing) forward pass – optimized with Triton.

Algorithm
---------
1) Layer‑norm the input tensor over the channel dimension C.
   A custom Triton kernel computes mean/variance per (batch, i, j) slot,
   applies the affine weight/bias and writes back a normalized tensor.

2) Linear projections and gating:
   * left  = Linear(x_norm, left_proj_weight)   (C → H)
   * right = Linear(x_norm, right_proj_weight) (C → H)
   * left_gate  = sigmoid( Linear(x_norm, left_gate_weight) )
   * right_gate = sigmoid( Linear(x_norm, right_gate_weight) )
   * out_gate   = sigmoid( Linear(x_norm, out_gate_weight)  )
   * Apply the optional pair‑wise mask (broadcast on H) and the gates:
       left  = left  * mask * left_gate
       right = right * mask * right_gate

3) Pair‑wise multiplicative update (einsum):
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply:
       out = (left.reshape(B,H,N,N)  @
              right.reshape(B,H,N,N).transpose(-2,-1))

4) Layer‑norm over the hidden dimension H (again a Triton kernel).

5) Multiply by out_gate and apply the final linear projection
   (H → C) to obtain the output tensor of shape [B,N,N,C].

Only the two LayerNorms are hand‑written in Triton; all dense
matrix multiplications use the highly‑optimised cuBLAS kernels
exposed through PyTorch.  The implementation works for any
B∈{1,2}, N∈{128,256,512,1024}, C∈{128,384}, H=128 and
handles both masked and un‑masked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel: layer‑norm over the last dimension (C)
# -------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    X_ptr,                 # const float*   input
    Y_ptr,                 # float*   output
    weight_ptr,            # const float*   gamma
    bias_ptr,              # const float*   beta
    eps,                   # float
    B: tl.constexpr,       # int
    N: tl.constexpr,       # int
    C: tl.constexpr,       # int
    stride_x0, stride_x1, stride_x2, stride_xc,   # int strides for X
    stride_y0, stride_y1, stride_y2, stride_yc,   # int strides for Y
    BLOCK_C: tl.constexpr  # tunable, compile‑time block size for channel dim
):
    pid = tl.program_id(0)
    total = B * N * N                       # total number of vectors to normalise
    if pid >= total:
        return

    # -----------------------------------------------------------------
    # decode (b,i,j) from pid  (pid = b * N * N + i * N + j)
    # -----------------------------------------------------------------
    N2 = N * N
    b = pid // N2
    rem = pid % N2
    i = rem // N
    j = rem % N

    # base pointers for this (b,i,j) vector
    x_ptr = X_ptr + b * stride_x0 + i * stride_x1 + j * stride_x2
    y_ptr = Y_ptr + b * stride_y0 + i * stride_y1 + j * stride_y2

    # -----------------------------------------------------------------
    # First pass: compute mean and variance (two scalars)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([1], dtype=tl.float32)
    sum_sq  = tl.zeros([1], dtype=tl.float32)

    for start in range(0, C, BLOCK_C):
        offs = start + tl.arange(0, BLOCK_C)
        mask = offs < C
        x = tl.load(x_ptr + offs * stride_xc, mask=mask, other=0.0)
        sum_val += tl.sum(x, axis=0)
        sum_sq  += tl.sum(x * x, axis=0)

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    inv_std = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Second pass: normalise, apply affine gamma/beta and store
    # -----------------------------------------------------------------
    for start in range(0, C, BLOCK_C):
        offs = start + tl.arange(0, BLOCK_C)
        mask = offs < C
        x = tl.load(x_ptr + offs * stride_xc, mask=mask, other=0.0)

        x_hat = (x - mean) * inv_std                # (x - μ) / σ
        w = tl.load(weight_ptr + offs, mask=mask, other=0.0)
        b_val = tl.load(bias_ptr + offs, mask=mask, other=0.0)
        y = x_hat * w + b_val
        tl.store(y_ptr + offs * stride_yc, y, mask=mask)


# -------------------------------------------------------------
# Helper: thin wrapper around the Triton kernel
# -------------------------------------------------------------
def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5) -> torch.Tensor:
    """
    Layer‑norm over the last dimension using the Triton kernel above.
    x, weight, bias are expected to be on the same device and of dtype float32.
    """
    B, N, _, C = x.shape
    out = torch.empty_like(x)

    # Strides for input and output (both have the same layout)
    stride_b, stride_i, stride_j, stride_c = x.stride()

    total = B * N * N
    grid = (total,)

    BLOCK_C = 128                     # works for C up to 384 (3 blocks max)
    layernorm_kernel[grid](
        x, out,
        weight, bias,
        eps,
        B, N, C,
        stride_b, stride_i, stride_j, stride_c,
        stride_b, stride_i, stride_j, stride_c,
        BLOCK_C=BLOCK_C,
    )
    return out

    mask = mask.unsqueeze(-1)   # shape [B, N, N, 1]

    # -----------------------------------------------------------------
    # 3) Linear projections (C → hidden_dim) and gating
    # -----------------------------------------------------------------
    # projections
    left  = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x_norm, weights["right_proj.weight"])  # [B,N,N,H]

    # mask before gating (as in reference implementation)
    left  = left  * mask
    right = right * mask

    # gates
    left_gate  = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))  # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))    # [B,N,N,H]

    # apply gates
    left  = left  * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4) Pair‑wise multiplicative update (einsum):
    #        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched mat‑mul left @ rightᵀ for each hidden channel.
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # reshape to (B*H, N, N) for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_mat = right.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)

    left_flat  = left_mat.view(B * H, N, N)
    right_flat = right_mat.view(B * H, N, N)

    # batched matrix multiplication with transpose on the second operand
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H,N,N)

    # reshape back to [B,N,N,H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # -----------------------------------------------------------------
    # 5) Layer‑norm over the hidden dimension (H)
    # -----------------------------------------------------------------
    out_norm = _layernorm_triton(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
    )  # shape [B,N,N,H]

    # -----------------------------------------------------------------
    # 6) Apply out‑gate and final linear projection back to dim (C)
    # -----------------------------------------------------------------
    out_norm = out_norm * out_gate                          # broadcasting over H
    output = F.linear(out_norm, weights["to_out.weight"])    # [B,N,N,C]

    return output
```
