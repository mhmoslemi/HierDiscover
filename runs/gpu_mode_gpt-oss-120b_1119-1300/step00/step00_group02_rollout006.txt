    BLOCK_N = 64
    BLOCK_K = 32

        a, b, c,
        M, M, K,
        a_s0, a_s1, a_s2,

        c_s0, c_s1, c_s2,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K

    out = out_norm * out_gate
    
    # Final linear projection (hidden_dim -> dim)
    output = torch.nn.functional.linear(out, to_out_weight)
    
    return output

"""
custom_kernel
==============

Implements the forward pass of the TriMul module (the “outgoing” version) used in
AlphaFold‑3 and related protein‑structure models.

The heavy O(N³) interaction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is performed as a batched matrix multiplication
    left[b,:,k,d] @ right[b,:,k,d]ᵀ
using a custom Triton kernel.  All other operations (LayerNorm,
linear projections, sigmoid gates and masking) are carried out with
standard PyTorch kernels for simplicity.

The kernel works for batch sizes 1‑2, sequence lengths up to 1024,
and hidden dimensions up to 384 (the hidden dimension after projection).
Mixed‑precision is used: the three‑dimensional multiplication is executed in
float16 with FP32 accumulation; the rest of the pipeline stays in FP32.
The final tensor has shape [B, N, N, dim] and dtype float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batch_matmul_kernel(
    a_ptr,                # pointer to A (BH, M, K) – FP16
    b_ptr,                # pointer to B (BH, K, N) – FP16
    c_ptr,                # pointer to C (BH, M, N) – FP32 (output)
    M, N, K,              # dimensions (compile‑time scalars are allowed)
    stride_ah, stride_am, stride_ak,   # strides of A
    stride_bh, stride_bk, stride_bn,   # strides of B
    stride_ch, stride_cm, stride_cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication C = A @ B where A, B, C have an extra
    leading batch dimension (BH = batch * hidden).  The kernel processes
    a tile of size BLOCK_M × BLOCK_N and iterates over K with blocks of
    size BLOCK_K.
    """
    pid_m = tl.program_id(0)                # tile row
    pid_n = tl.program_id(1)                # tile col
    pid_z = tl.program_id(2)                # batch * hidden index

    # ----- tile start offsets -----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- K‑loop -----
    for k_iter in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # pointers for the current tile
        a_ptrs = (
            a_ptr
            + pid_z * stride_ah
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_z * stride_bh
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        # mask out‑of‑bounds elements (needed for the last partial tile)
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # FP16
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # FP16

        # multiply‑accumulate (FP16×FP16 → FP32)
        acc += tl.dot(a, b)

    # ----- write back -----
    c_ptrs = (
        c_ptr
        + pid_z * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    store_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=store_mask)


def custom_kernel(data):
    """
    Forward pass of the TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (see key list below)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # unpack model weights
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1. Layer‑norm on the input
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] – float32

    # --------------------------------------------------------------
    # 2. Linear projections and gates (all in FP32)
    # --------------------------------------------------------------
    left_proj = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right_proj = F.linear(x, right_proj_weight)

    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()  # used after to_out_norm

    # --------------------------------------------------------------
    # 3. Apply mask (if present) and gating
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        mask = mask.to(x.dtype).unsqueeze(-1)            # [B, N, N, 1]
        left = left_proj * mask
        right = right_proj * mask
    else:
        left = left_proj
        right = right_proj

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 4. Cast to FP16 for the cubic‑time multiplication
    # --------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # --------------------------------------------------------------
    # 5. Prepare tensors for batched matmul:
    #    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    -> treat (b,d) as a single batch dimension and compute
    #       (i,k) @ (k,j)ᵀ with a custom Triton kernel.
    # --------------------------------------------------------------
    # left  : [B, N, N, H] -> [B, H, N, N] (i, k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()       # [B, H, N, N]
    # right : need (k, j) order -> permute to [B, H, N, N] (k, j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()    # [B, H, N, N]

    B, H, M, K = left_perm.shape          # M == K == N (seq_len)
    BH = B * H

    a = left_perm.view(BH, M, K)          # [BH, M, K]   FP16
    b = right_perm.view(BH, K, M)         # [BH, K, N]   FP16 (note N == M)

    # output in FP32
    c = torch.empty((BH, M, M), dtype=torch.float32, device=input_tensor.device)

    # strides (in number of elements)
    a_s0, a_s1, a_s2 = a.stride()
    b_s0, b_s1, b_s2 = b.stride()
    c_s0, c_s1, c_s2 = c.stride()

    # launch parameters – tuned for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (M + BLOCK_N - 1) // BLOCK_N,
        BH,
    )

    _batch_matmul_kernel[grid](
        a, b, c,
        M, M, K,
        a_s0, a_s1, a_s2,
        b_s0, b_s1, b_s2,
        c_s0, c_s1, c_s2,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = c.view(B, H, M, M).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # --------------------------------------------------------------
    # 6. Post‑matmul LayerNorm, out‑gate and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )  # [B, N, N, H] – float32

    out = out * out_gate                       # gating after norm
    output = F.linear(out, to_out_weight)      # [B, N, N, dim] – float32

    return output
```
