def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype  # likely torch.float32

    # LayerNorm over last dimension
    eps = 1e-5
    # compute mean and var per element (i.e., axis=-1)
    # We'll use torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    # Linear projections
    # left_proj = x_norm @ left_proj_weight.T
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # Apply mask
    if not config.get("nomask", True):
        mask_ = mask[..., None]  # shape [B,N,N,1]
        left = left * mask_
        right = right * mask_
    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))
    left = left * left_gate
    right = right * right_gate

    # Cast to half for Triton kernel
    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)

    B, N, _, H = left_half.shape

    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # Compute strides for contiguous tensors
    stride_b = left_half.stride(0)  # B stride
    stride_i = left_half.stride(1)  # N stride (for i)
    stride_j = left_half.stride(2)  # N stride (for k? Actually left dimension for k)

    right_reshaped = right_half.permute(0,3,1,2).contiguous()
    out_reshaped = out.permute(0,3,1,2).contiguous()  # shape (B,H,N,N)

    stride_bh = left_reshaped.stride(0)   # B*H stride

def trmul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    N,
    stride_abh, stride_ai, stride_ak,
    stride_bbh, stride_bj, stride_bk,
    stride_cht, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_z = tl.program_id(2)
    pid_x = tl.program_id(0)  # i block
    pid_y = tl.program_id(1)  # j block

    # offsets
    offs_i = pid_x * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_y * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers base for this slice
    a_ptr = A_ptr + pid_z * stride_abh
    b_ptr = B_ptr + pid_z * stride_bbh
    c_ptr = C_ptr + pid_z * stride_cht

    # load row indices for C
    # compute final offsets for C
    c_offs = (offs_i[:, None] * stride_ci) + (offs_j[None, :] * stride_cj)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, N, BLOCK_K):
        cur_k = k + offs_k
        mask_k = cur_k < N

        # load A tile
        a_offs = (offs_i[:, None] * stride_ai) + (cur_k[None, :] * stride_ak)
        a = tl.load(a_ptr + a_offs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # dot product: A (M,K) dot B (K,N) -> (M,N)
        acc += tl.dot(a, b)

    # store
    tl.store(c_ptr + c_offs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    eps=1e-5

    # 1. Input LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # 2. Linear Projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # 3. Mask (if present)
    if not config.get("nomask", True):
        mask_ = mask.unsqueeze(-1).to(dtype=left.dtype)  # broadcast
        left = left * mask_
        right = right * mask_
    # if mask is None or nomask True, no mask multiplication

    # 4. Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # 5. Triton kernel for bilinear product across seq dimension
    # Rearrange to (B, hidden_dim, N, N) for convenience.
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    out = torch.empty_like(left_perm)  # same shape and dtype

    # Kernel launch
    B, H, N, _ = left_perm.shape
    # Set block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # compute strides
    stride_bh = left_perm.stride(0)
    stride_h = left_perm.stride(1)  # not used
    stride_i = left_perm.stride(2)
    stride_k = left_perm.stride(3)

    # we also need stride for B: basically same as left
    # Right uses same stride but reversed for dot
    trimul_matmul_kernel[( (N + BLOCK_M - 1)//BLOCK_M,
                            (N + BLOCK_N - 1)//BLOCK_N,
                            B * H )](
        left_perm,
        right_perm,
        out,
        N,
        stride_abh=stride_bh,
        stride_ai=stride_i,
        stride_ak=stride_k,
        stride_bbh=stride_bh,
        stride_bj=stride_i,
        stride_bk=stride_k,
        stride_cht=stride_bh,
        stride_ci=stride_i,
        stride_cj=stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # after kernel, out shape (B, H, N, N)
    out = out.permute(0,2,3,1).contiguous() # back to (B,N,N,H)

    # 6. To_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # 7. Apply out_gate
    out = out * out_gate

    # 8. Final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul forward kernel (AlphaFold3 “outgoing” version)

Algorithm
---------
1. Layer‐norm the input tensor (over the channel dimension `dim`).
2. Linear projections → `left` and `right` (shape [B, N, N, H]).
3. (optional) elementwise mask multiplication.
4. Gating:  left_gate, right_gate, out_gate = sigmoid(linear(x_norm)).
   Apply gates to `left` and `right`.
5. Core O(N³·H) operation:
   For every batch `b` and hidden channel `h` compute
       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   i.e. a batched matrix multiply `C = A @ Bᵀ` where each slice
   `A = left[b, h, :, :]` and `B = right[b, h, :, :]`.
   This is implemented with a custom Triton kernel that tiles the
   `i`‑, `j`‑ and `k`‑dimensions (BLOCK_M×BLOCK_N×BLOCK_K) and
   accumulates in FP32.
6. Layer‑norm over the hidden dimension, apply `out_gate`
   and a final linear projection back to `dim`.

The kernel fuses the three‑nested loops of the original `einsum`
into a fast tiled GEMM and guarantees the output is in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the N³ “matrix‑multiply‑over‑k”
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                     # pointers
    N,                                        # sequence length
    stride_abh, stride_ai, stride_ak,        # strides for A  (B*H, i, k)
    stride_bbh, stride_bj, stride_bk,        # strides for B  (B*H, j, k) – we treat B as Bᵀ
    stride_cht, stride_ci, stride_cj,        # strides for C  (B*H, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)                     # index over (batch * hidden)
    pid_x = tl.program_id(0)                     # tile row (i)
    pid_y = tl.program_id(1)                     # tile col (j)

    # -------------------- offsets & masks --------------------
    offs_i = pid_x * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_y * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointer for this (b,h) slice
    A_base = A_ptr + pid_z * stride_abh
    B_base = B_ptr + pid_z * stride_bbh
    C_base = C_ptr + pid_z * stride_cht

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- tiled k‑loop --------------------
    for k in range(0, N, BLOCK_K):
        cur_k = k + offs_k
        mask_k = cur_k < N

        # A_tile : (BLOCK_M, BLOCK_K)   ← left[b,h,i,k]
        a = tl.load(
            A_base + (offs_i[:, None] * stride_ai) + (cur_k[None, :] * stride_ak),
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # B_tile : (BLOCK_K, BLOCK_N)   ← right[b,h,k,j]  (interpreted as Bᵀ)
        b = tl.load(
            B_base + (cur_k[:, None] * stride_bk) + (offs_j[None, :] * stride_bj),
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # accumulate   (FP32)
        acc += tl.dot(a, b)   # a: (M,K), b: (K,N) → (M,N)

    # -------------------- store --------------------
    c_ptrs = C_base + (offs_i[:, None] * stride_ci) + (offs_j[None, :] * stride_cj)
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” update.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    eps = 1e-5

    # --------------------------------------------------------------
    # 1. LayerNorm over the channel dimension (`dim`)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (mask shape: [B, N, N])
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        mask = mask_tensor.to(dtype=left.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core N³ computation – Triton GEMM over the sequence dimension
    #    We work on tensors of shape (B, H, N, N) to have a single
    #    batch dimension for the kernel.
    # --------------------------------------------------------------
    # reorder to (B, H, N, N) for a contiguous layout
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]
    out_perm = torch.empty_like(left_perm)               # allocate result

    B, H, N, _ = left_perm.shape

    # Tiling parameters – chosen for H100 fp32 tensor cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # strides (in number of elements, not bytes)
    stride_abh = left_perm.stride(0)   # step between (b,h) slices
    stride_ai = left_perm.stride(2)    # i (row) stride
    stride_ak = left_perm.stride(3)    # k (col) stride

    # for right we treat it as Bᵀ: k‑row stride = stride of innermost dim,
    # j‑col stride = stride of the second‑last dim
    stride_bbh = right_perm.stride(0)
    stride_bk = right_perm.stride(3)   # k (inner) stride → rows of Bᵀ
    stride_bj = right_perm.stride(2)   # j (outer) stride → columns of Bᵀ

    stride_cht = out_perm.stride(0)
    stride_ci = out_perm.stride(2)
    stride_cj = out_perm.stride(3)

    # launch grid: (tiles_i, tiles_j, B*H)
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )
    trimul_matmul_kernel[
        grid
    ](
        left_perm,
        right_perm,
        out_perm,
        N,
        stride_abh,
        stride_ai,
        stride_ak,
        stride_bbh,
        stride_bj,
        stride_bk,
        stride_cht,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # back to original layout [B, N, N, H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6. Output layer‑norm, gating and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
