```python
"""
TriMul 'outgoing' kernel.

The heavy computational part is the einsum:
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]

We implement it as a Triton kernel where each program (thread) computes a single
(i,j,d) element and accumulates over k in blocks of size BLOCK_K. The kernel
loads contiguous slices of length BLOCK_K from the left and right tensors,
performs a dot product, and accumulates in FP32. Input tensors are expected
to be contiguous in memory with layout (i,k,d). The rest of the forward pass
(LayerNorm, Linear, gating, etc.) is performed in PyTorch.
"""

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_elem_kernel(
    left_ptr,          # *float32
    right_ptr,         # *float32
    out_ptr,           # *float32
    N: tl.constexpr,
    stride_l_i: tl.constexpr, stride_l_k: tl.constexpr, stride_l_h: tl.constexpr,
    stride_r_j: tl.constexpr, stride_r_k: tl.constexpr, stride_r_h: tl.constexpr,
    stride_o_i: tl.constexpr, stride_o_j: tl.constexpr, stride_o_h: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    i = tl.program_id(0)   # output row   (i)
    j = tl.program_id(1)   # output col   (j)
    d = tl.program_id(2)   # hidden dim   (d)

    # Bounds – the launch grid may be larger than N/H, guard against overflow
    if i >= N or j >= N:
        return

    # accumulation in FP32
    acc = tl.float32(0.0)

    # iterate over k dimension in tiles of BLOCK_K
    for k_base in range(0, N, BLOCK_K):
        cur_k = tl.arange(0, BLOCK_K) + k_base               # (BLOCK_K,)
        k_mask = cur_k < N

        # compute pointers for this tile (vector of BLOCK_K elements)
        left_off = i * stride_l_i + cur_k * stride_l_k + d * stride_l_h
        right_off = j * stride_r_j + cur_k * stride_r_k + d * stride_r_h

        left_vec = tl.load(left_ptr + left_off,
                           mask=k_mask,
                           other=0.0)                         # (BLOCK_K,)
        right_vec = tl.load(right_ptr + right_off,
                            mask=k_mask,
                            other=0.0)                        # (BLOCK_K,)

        # dot product over the tile
        acc += tl.dot(left_vec, right_vec)   # scalar FP32

    # write the result
    out_off = i * stride_o_i + j * stride_o_j + d * stride_o_h
    tl.store(out_ptr + out_off, acc)


def custom_kernel(data):
    """
    Triton implementation of the “outgoing” TriMul operation.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack inputs
    input_tensor, mask_tensor, weights, config = data

    # ------------------------------------------------------------------
    # Retrieve configuration
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm on the input
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    # left_proj: Linear(dim, hidden_dim)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    # right_proj: Linear(dim, hidden_dim)
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gates (sigmoid of linear projections of the original normalized input)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # TriMul core (einsum) – performed with a Triton kernel
    # ------------------------------------------------------------------
    # Prepare output tensor
    batch_size, seq_len, _, _ = left.shape
    out = torch.empty(batch_size, seq_len, seq_len, hidden_dim,
                      dtype=torch.float32, device=left.device)

    # Ensure contiguous layout for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # Strides in element units (not bytes)
    # left layout: (B, i, k, h)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    # right layout: (B, j, k, h)
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    # out layout: (B, i, j, h)
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Choose block size for the k dimension. 32 is a good default for
    # the H100 architecture (full‑warp friendly).
    BLOCK_K = 32

    # Launch a separate kernel for each batch element to keep the kernel
    # signature simple.
    for b in range(batch_size):
        left_ptr = left.data_ptr() + b * stride_l_b * left.element_size()
        right_ptr = right.data_ptr() + b * stride_r_b * right.element_size()
        out_ptr = out.data_ptr() + b * stride_o_b * out.element_size()

        # Convert byte offsets to element offsets expected by Triton
        left_ptr = left_ptr // left.element_size()
        right_ptr = right_ptr // right.element_size()
        out_ptr = out_ptr // out.element_size()

        grid = (seq_len, seq_len, hidden_dim)  # (i, j, d)
        tri_mul_elem_kernel[grid](
            left_ptr,
            right_ptr,
            out_ptr,
            seq_len,
            stride_l_i, stride_l_k, stride_l_h,
            stride_r_j, stride_r_k, stride_r_h,
            stride_o_i, stride_o_j, stride_o_h,
            BLOCK_K=BLOCK_K,
        )

    # ------------------------------------------------------------------
    # Output projection (LayerNorm + gating + Linear)
    # ------------------------------------------------------------------
    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # multiply by out_gate
    out = out * out_gate

    # final linear projection to original dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) – Triton‑accelerated core.

Given tensors
    left  : [B, N, N, H]   (H = hidden_dim)
    right : [B, N, N, H]

the heavy operation is
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

We implement this einsum with a Triton kernel where each program (thread)
computes a single output element (i, j, h) and accumulates over k in
tiles of size BLOCK_K (default 32).  All other layers (LayerNorm,
Linear, gating) are performed in PyTorch.  The final output is
float‑32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_elem_kernel(
    left_ptr,          # *float32  (contiguous B×N×N×H slice for one batch)
    right_ptr,         # *float32
    out_ptr,           # *float32
    N: tl.constexpr,   # sequence length
    stride_l_i: tl.constexpr, stride_l_k: tl.constexpr, stride_l_h: tl.constexpr,
    stride_r_j: tl.constexpr, stride_r_k: tl.constexpr, stride_r_h: tl.constexpr,
    stride_o_i: tl.constexpr, stride_o_j: tl.constexpr, stride_o_h: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids: i – row, j – column, d – hidden channel
    i = tl.program_id(0)      # 0 ≤ i < N
    j = tl.program_id(1)      # 0 ≤ j < N
    d = tl.program_id(2)      # 0 ≤ d < H  (grid size guarantees this)

    # guard against over‑launch (only needed for safety)
    if i >= N or j >= N:
        return

    acc = tl.float32(0.0)     # accumulator in FP32

    # iterate over k in blocks of BLOCK_K
    k = 0
    while k < N:
        cur_k = tl.arange(0, BLOCK_K) + k                # (BLOCK_K,)
        k_mask = cur_k < N                               # mask for tail

        # left[i, cur_k, d]   -> pointer = i*stride_i + cur_k*stride_k + d*stride_h
        left_off = i * stride_l_i + cur_k * stride_l_k + d * stride_l_h
        # right[j, cur_k, d]  -> pointer = j*stride_j + cur_k*stride_k + d*stride_h
        right_off = j * stride_r_j + cur_k * stride_r_k + d * stride_r_h

        left_vec  = tl.load(left_ptr + left_off,  mask=k_mask, other=0.0)   # (BLOCK_K,)
        right_vec = tl.load(right_ptr + right_off, mask=k_mask, other=0.0)   # (BLOCK_K,)

        # dot over the BLOCK_K dimension
        acc += tl.dot(left_vec, right_vec)   # scalar FP32

        k += BLOCK_K

    # write the result
    out_off = i * stride_o_i + j * stride_o_j + d * stride_o_h
    tl.store(out_ptr + out_off, acc)


def custom_kernel(data):
    """
    Triton‑based forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    nomask       = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Normalisation of the input (LayerNorm over the last dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"]) # [B, N, N, H]

    # ------------------------------------------------------------------
    # Optional mask (mask is broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        m = mask_tensor.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left  = left * m
        right = right * m

    # ------------------------------------------------------------------
    # Gating (sigmoid of a linear projection of the normalised input)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # TriMul core – compute the einsum via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty(B, N, N, hidden_dim,
                      dtype=torch.float32,
                      device=left.device)

    # Ensure contiguous layout for safe stride arithmetic
    left  = left.contiguous()
    right = right.contiguous()
    out   = out.contiguous()

    # Strides are expressed in *elements* (not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tile size for the k dimension – 32 works well on H100
    BLOCK_K = 32

    # Launch one kernel per batch element (B ≤ 2, negligible overhead)
    for b in range(B):
        # base element offsets for this batch
        base_l = left.data_ptr() // left.element_size() + b * stride_l_b
        base_r = right.data_ptr() // right.element_size() + b * stride_r_b
        base_o = out.data_ptr() // out.element_size() + b * stride_o_b

        grid = (N, N, hidden_dim)   # (i, j, hidden channel)

        tri_mul_elem_kernel[grid](
            base_l,
            base_r,
            base_o,
            N,
            stride_l_i, stride_l_k, stride_l_h,
            stride_r_j, stride_r_k, stride_r_h,
            stride_o_i, stride_o_j, stride_o_h,
            BLOCK_K=BLOCK_K,
        )

    # ------------------------------------------------------------------
    # Output projection: LayerNorm → gating → Linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
