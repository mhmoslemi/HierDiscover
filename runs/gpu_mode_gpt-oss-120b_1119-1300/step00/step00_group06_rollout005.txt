@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_i, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # pid = program id
    pid = tl.program_id(0)
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    total_bd = B * D
    pid_bd = pid // (num_pid_m * num_pid_n)
    pid_rem = pid % (num_pid_m * num_pid_n)
    pid_m = pid_rem // num_pid_n
    pid_n = pid_rem % num_pid_n

    # compute batch and hidden dim indices
    b = pid_bd // D
    d = pid_bd % D

    # compute block offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers for left and right base for this batch+dim
    left_base = left_ptr + b * stride_l_b + d * stride_l_d
    right_base = right_ptr + b * stride_r_b + d * stride_r_d
    out_base = out_ptr + b * stride_o_b + d * stride_o_d

    # zero accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile
        left_ptrs = left_base + (offs_i[:, None] * stride_l_i) + (offs_k[None, :] * stride_l_k)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right tile (transposed)
        # we need right[b, k, j, d] -> pointer offsets: k*stride_r_i + j*stride_r_k
        right_ptrs = right_base + (offs_k[:, None] * stride_r_i) + (offs_j[None, :] * stride_r_k)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot product
        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_base + (offs_i[:, None] * stride_o_i) + (offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs, acc.to(tl.float16) if left_ptr.dtype==tl.float16 else acc, mask=mask_i[:, None] & mask_j[None, :])

@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_i, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    total_bd = B * D
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_bd = pid // (num_pid_m * num_pid_n)
    pid_rem = pid % (num_pid_m * num_pid_n)
    pid_m = pid_rem // num_pid_n
    pid_n = pid_rem % num_pid_n

    b = pid_bd // D
    d = pid_bd % D

    # Compute offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers base for this (b,d)
    left_base = left_ptr + b * stride_l_b + d * stride_l_d
    right_base = right_ptr + b * stride_r_b + d * stride_r_d
    out_base = out_ptr + b * stride_o_b + d * stride_o_d

        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(left, right)

    # Store result
    out_ptrs = out_base + (offs_i[:, None] * stride_o_i) + (offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul forward (outgoing) implementation with a fused Triton kernel.

The heavy part of the TriMul forward is the contraction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched matrix‑multiply per hidden dimension d.
A custom Triton kernel performs this N×N matrix multiplication for every
(batch, hidden_dim) pair, handling arbitrary batch size B, sequence length N
and hidden dimension D (called hidden_dim in the original code).  The rest of
the operations (LayerNorms, linear projections, gating, final projection) are
performed with standard PyTorch calls.

The kernel is launched with a 1‑D grid of size  B·D·⌈N/BLOCK_M⌉·⌈N/BLOCK_N⌉.
Each program instance computes a tile of size (BLOCK_M×BLOCK_N) of the
output matrix for a given (batch, hidden) pair, iterating over the K‑dimension
in BLOCK_K steps.  Bounds are handled with masks so the kernel works for any
N.  The accumulator is FP32 for numerical stability, while the inputs can be
FP16 or FP32 (the kernel loads them in their native type).

The final output is a float‑32 tensor of shape (B, N, N, dim) exactly as the
reference PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for the TriMul contraction
# --------------------------------------------------------------------------- #
@triton.jit
def _trimul_gemm_kernel(
    left_ptr,        # *float32 or *float16, shape [B, N, N, D]
    right_ptr,       # *float32 or *float16, shape [B, N, N, D]
    out_ptr,         # *float32, shape [B, N, N, D]
    B, N, D,         # runtime shape values
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_i, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out = Σ_k left[..., i,k,d] * right[..., j,k,d] for all i,j,d."""
    pid = tl.program_id(0)

    # ----- grid decomposition ------------------------------------------------
    total_bd = B * D                          # combined batch*hidden dimension
    num_pid_m = tl.cdiv(N, BLOCK_M)           # number of tiles along M (=i)
    num_pid_n = tl.cdiv(N, BLOCK_N)           # number of tiles along N (=j)

    pid_bd = pid // (num_pid_m * num_pid_n)   # index of (batch, hidden)
    pid_rem = pid % (num_pid_m * num_pid_n)
    pid_m = pid_rem // num_pid_n               # tile row index
    pid_n = pid_rem % num_pid_n                # tile col index

    b = pid_bd // D
    d = pid_bd % D

    # ----- tile coordinates --------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (batch, hidden) pair
    left_base  = left_ptr  + b * stride_l_b + d * stride_l_d
    right_base = right_ptr + b * stride_r_b + d * stride_r_d
    out_base   = out_ptr   + b * stride_o_b + d * stride_o_d

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- K‑dimension loop --------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + (offs_i[:, None] * stride_l_i) + (offs_k[None, :] * stride_l_k)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile (transposed): we need right[b, k, j, d]
        # original layout is (b, i, j, d); we treat i as K for the transposed view.
        right_ptrs = right_base + (offs_k[:, None] * stride_r_i) + (offs_j[None, :] * stride_r_k)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        acc += tl.dot(left, right)

    # ----- write back --------------------------------------------------------
    out_ptrs = out_base + (offs_i[:, None] * stride_o_i) + (offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------------------- #
# Helper that launches the Triton kernel
# --------------------------------------------------------------------------- #
def _trimul_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the TriMul contraction.
    left/right: (B, N, N, D) contiguous tensors (float16 or float32).
    Returns out of same dtype as left/right (accumulated in FP32, stored in FP32).
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, N2, D = left.shape
    assert N == N2, "left/right must be square on the last two dims"
    device = left.device

    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # --------------------------------------------------------------
    # Triton launch configuration
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = B * D * grid_m * grid_n

    # strides (int64)
    s_l_b, s_l_i, s_l_k, s_l_d = left.stride()
    s_r_b, s_r_i, s_r_k, s_r_d = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_d = out.stride()

    _trimul_gemm_kernel[(total_programs,)](
        left,
        right,
        out,
        B,
        N,
        D,
        s_l_b,
        s_l_i,
        s_l_k,
        s_l_d,
        s_r_b,
        s_r_i,
        s_r_k,
        s_r_d,
        s_o_b,
        s_o_i,
        s_o_j,
        s_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_stages=3,
        num_warps=8,
    )
    return out


# --------------------------------------------------------------------------- #
# Entry point expected by the evaluator
# --------------------------------------------------------------------------- #
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator using a fused Triton kernel
    for the cubic contraction.

    Arguments
    ---------
    data : tuple containing
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of model weight tensors
        - config       : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    torch._dynamo.disable()   # ensure torch.compile does not interfere
    torch.cuda._lazy_init()    # make sure CUDA context is ready

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Load model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # Normalisation and linear projections
    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # Linear projections (bias=False in the reference)
    left  = F.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)  # (B,N,N,hidden_dim)

    # Optional mask (mask is broadcast over hidden dimension)
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # TriMul contraction via Triton
    # --------------------------------------------------------------
    # Use FP32 for the heavy compute; we keep tensors contiguous.
    left  = left.contiguous()
    right = right.contiguous()
    # Triton kernel returns FP32 output
    out = _trimul_gemm(left, right)               # shape (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # Post‑contraction LayerNorm, gating, and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    out = out * out_gate

    # Final projection back to `dim` (bias=False in reference)
    out = F.linear(out, to_out_weight)   # (B,N,N,dim)

    return out
