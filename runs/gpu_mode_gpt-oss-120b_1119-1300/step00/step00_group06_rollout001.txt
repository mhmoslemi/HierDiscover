"""
custom_kernel: Highly‑optimized forward implementation of the “outgoing” Triangle
Multiplicative Update (TriMul) used in AlphaFold‑3 and related models.

Algorithm
---------
1. Layer‑normalize the input pair representation (last dimension = dim).
2. Linear projections → left, right (shape [..., hidden_dim]).
3. Apply an optional pair‑mask and sigmoid‑gates (left_gate, right_gate,
   out_gate) – all element‑wise.
4. Core operation:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each (b,h) pair:
        out_{bh} = left_{bh} @ right_{bh}ᵀ
   Implemented in a custom Triton kernel that tiles the i, j and k axes
   (BLOCK_M·BLOCK_N·BLOCK_K) and accumulates in FP32 while loading FP16.
5. Layer‑normalize the hidden‑dim output, apply the out‑gate, and project back
   to the original dimension with a final linear layer.

The kernel processes all (batch, hidden_dim) matrices in a single launch:
grid = (B*H, ceil(N/BLOCK_M), ceil(N/BLOCK_N)).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the TriMul core (left @ rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # shape / config
    B, N, H,
    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # program IDs
    # --------------------------------------------------------------
    pid_batch = tl.program_id(0)      # index = b * H + h
    pid_m = tl.program_id(1)          # i‑tile
    pid_n = tl.program_id(2)          # j‑tile

    # decode batch index and hidden‑dim index
    b = pid_batch // H
    h = pid_batch % H

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # base pointers for the current (b, h) matrix
    # left[b, i, k, h]  -> shape (BLOCK_M, BLOCK_K)
    # right_T[b, k, j, h] (the transposed view of right) -> shape (BLOCK_K, BLOCK_N)
    # --------------------------------------------------------------
    a_base = left_ptr + b * stride_left_b + h * stride_left_h
    b_base = right_ptr + b * stride_right_b + h * stride_right_h

    # pointer for the first tile (k = 0)
    a_ptrs = a_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
    b_ptrs = b_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over k dimension in tiles of size BLOCK_K
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tile in range(0, num_k_tiles):
        cur_k = tile * BLOCK_K
        cur_mask_k = (cur_k + offs_k) < N

        # load A tile (left)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & cur_mask_k[None, :],
                    other=0.0).to(tl.float16)

        # load B_T tile (right transposed)
        b = tl.load(b_ptrs,
                    mask=cur_mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float16)

        # dot product – accumulate in FP32
        acc += tl.dot(a, b, out_dtype=tl.float32)

        # advance pointers to next k‑tile
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # --------------------------------------------------------------
    # write back the result
    # --------------------------------------------------------------
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h \
               + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

    """
    Compute out = left @ rightᵀ for each (batch, hidden) pair.
    left, right : [B, N, N, H] (FP16)
    out         : [B, N, N, H] (FP32, pre‑allocated)
    """
    assert left_fp16.shape == right_fp16.shape == out_fp32.shape
    B, N, _, H = left_fp16.shape

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        B, N, H,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        out_fp32.stride(0), out_fp32.stride(1), out_fp32.stride(2), out_fp32.stride(3),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out_fp32

# ----------------------------------------------------------------------
# Entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]
        mask         : torch.Tensor [B, N, N]  or None
        weights      : dict of torch.Tensors containing all linear / LN weights
        config       : dict with keys "dim", "hidden_dim" and optional "eps"

    Returns
    -------
    torch.Tensor
        Updated pair representation of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # --------------------------------------------------------------
    # 1) LayerNorm over the last dimension
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)                     # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Optional mask (broadcast over the hidden dimension)
    # --------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_b = mask.to(left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_b
        right = right * mask_b

    # --------------------------------------------------------------
    # 4) Gating networks (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Core TriMul – batched matmul left @ rightᵀ per hidden channel
    #    Cast to fp16 for the heavy arithmetic, accumulate in fp32.
    # --------------------------------------------------------------
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_hidden = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # launch Triton kernel (tile sizes tuned for H100)
    trimul_batch_matmul(left_fp16, right_fp16, out_hidden,
                        BLOCK_M=64, BLOCK_N=64, BLOCK_K=64)

    # --------------------------------------------------------------
    # 6) Post‑processing: hidden‑dim LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(out_hidden, (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=eps)                         # [B, N, N, hidden_dim]

    out = out * out_gate                               # gating
    output = F.linear(out, weights["to_out.weight"])   # back to [B, N, N, dim]

    return output
