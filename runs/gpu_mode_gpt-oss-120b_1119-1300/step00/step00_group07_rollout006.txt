def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-6

    # weights
    # layer norm weight and bias
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # proj weights
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # 1) layernorm on input
    x_norm = triton_layernorm(input_tensor, norm_w, norm_b, eps) # shape [B,N,N,dim]

    # 2) linear transforms
    # Use torch.linear for each weight to produce [B,N,N,hidden_dim]

    left = torch.nn.functional.linear(x_norm, left_proj_w)  # weight shape [hidden_dim, dim]
    right = torch.nn.functional.linear(x_norm, right_proj_w)

    left_gate = torch.nn.functional.linear(x_norm, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_w).sigmoid()

    # apply mask if not nomask
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # fuse gating
    left = left * left_gate
    right = right * right_gate

    # 3) Compute out via batched matmul per hidden_dim:
    # Reshape left and right to (B*hidden_dim, N, N)
    B,N,_,_ = left.shape
    left_reshape = left.permute(0,3,1,2).reshape(B*hidden_dim, N, N)
    right_reshape = right.permute(0,3,1,2).reshape(B*hidden_dim, N, N)
    # Compute left @ right^T
    out = torch.bmm(left_reshape, right_reshape.transpose(1,2)) # shape (B*hidden_dim, N,N)
    out = out.reshape(B, hidden_dim, N, N).permute(0,2,3,1)  # shape (B,N,N,hidden_dim)

    # 4) to_out_norm
    out_norm = triton_layernorm(out, to_out_norm_w, to_out_norm_b, eps) # shape [B,N,N,hidden_dim]

    out_norm = out_norm * out_gate  # gating

    # 5) final linear
    out_final = torch.nn.functional.linear(out_norm, to_out_w)  # weight shape [dim, hidden_dim]

    return out_final
```

@triton.jit
def layernorm_kernel(
    x_ptr, out_ptr,
    weight_ptr, bias_ptr,
    M, C,
    stride_xm, stride_xc,
    stride_outm, stride_outc,
    eps,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)  # row index
    if pid >= M:
        return
    offs_c = tl.arange(0, BLOCK_SIZE)
    mask = offs_c < C

    x_ptrs = x_ptr + pid * stride_xm + offs_c * stride_xc
    x = tl.load(x_ptrs, mask=mask, other=0.0)

    # sum & sumsq
    sum_val = tl.sum(x, axis=0)
    sum_sq = tl.sum(x * x, axis=0)

    mean = sum_val / C
    var = sum_sq / C - mean * mean
    rsig = 1.0 / tl.sqrt(var + eps)

    # Normalize
    norm = (x - mean) * rsig

    # Apply weight and bias
    w = tl.load(weight_ptr + offs_c, mask=mask, other=0.0)
    b = tl.load(bias_ptr + offs_c, mask=mask, other=0.0)
    out = norm * w + b

    out_ptrs = out_ptr + pid * stride_outm + offs_c * stride_outc
    tl.store(out_ptrs, out, mask=mask)
```

def fuse_masks_gates_kernel(
    left_proj_ptr, right_proj_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return
    b = pid // (N * N)
    tmp = pid % (N * N)
    i = tmp // N
    j = tmp % N

    mask_ptr_base = mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    mask_val = tl.load(mask_ptr_base).to(tl.float32)

    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    left_ptr = left_proj_ptr + b * stride_left_b + i * stride_left_i + j * stride_left_j + offs_h * stride_left_h

    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # layernorm on input
    x_norm = torch.nn.functional.layer_norm(input_tensor,
                                            (dim,),
                                            weight=norm_weight,
                                            bias=norm_bias,
                                            eps=eps)

    # Linear projections (no bias)
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight)  # [B,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Gates (sigmoid)
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    if not config.get("nomask", False):
        # mask may be boolean; cast to float
        mask_f = mask.to(x_norm.dtype)
        # Launch Triton kernel to fuse mask and gates
        B, N, _, _ = left_proj.shape
        # compute strides
        l_stride_b, l_stride_i, l_stride_j, l_stride_h = left_proj.stride()
        r_stride_b, r_stride_i, r_stride_j, r_stride_h = right_proj.stride()
        lg_stride_b, lg_stride_i, lg_stride_j, lg_stride_h = left_gate.stride()
        rg_stride_b, rg_stride_i, rg_stride_j, rg_stride_h = right_gate.stride()
        mask_stride_b, mask_stride_i, mask_stride_j = mask_f.stride()
        total = B * N * N
        grid = (total,)
        fuse_masks_gates_kernel[grid](
            left_proj, right_proj,
            left_gate, right_gate,
            mask_f,
            B, N, hidden_dim,
            l_stride_b, l_stride_i, l_stride_j, l_stride_h,
            r_stride_b, r_stride_i, r_stride_j, r_stride_h,
            lg_stride_b, lg_stride_i, lg_stride_j, lg_stride_h,
            rg_stride_b, rg_stride_i, rg_stride_j, rg_stride_h,
            mask_stride_b, mask_stride_i, mask_stride_j,
            BLOCK_H=hidden_dim
        )
        left = left_proj
        right = right_proj
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Batched GEMM: left @ right^T per hidden dim
    B, N, _, _ = left.shape
    # reshape for batched matmul
    left_reshape = left.permute(0,3,1,2).reshape(B * hidden_dim, N, N)
    right_reshape = right.permute(0,3,1,2).reshape(B * hidden_dim, N, N)
    # GEMM
    out = torch.bmm(left_reshape, right_reshape.transpose(1,2))
    out = out.view(B, hidden_dim, N, N).permute(0,2,3,1)  # back to [B,N,N,hidden_dim]

    # Output norm
    out_norm = torch.nn.functional.layer_norm(out,
                                              (hidden_dim,),
                                              weight=to_out_norm_weight,
                                              bias=to_out_norm_bias,
                                              eps=eps)
    # Apply out gate
    out_gated = out_norm * out_gate

    # Final linear
    out_final = torch.nn.functional.linear(out_gated, to_out_weight)

    return out_final

"""
TriMul (outgoing) forward pass implemented with a hybrid Torch‑Triton strategy.

Algorithm
---------
1. Layer‑normalize the input tensor `x` using the provided weight/bias.
2. Compute four linear projections of the normalized tensor:
   * left_proj  = x @ W_left      (→ hidden_dim)
   * right_proj = x @ W_right
   * left_gate  = sigmoid(x @ W_left_gate)
   * right_gate = sigmoid(x @ W_right_gate)
   * out_gate   = sigmoid(x @ W_out_gate)   (used later)
3. If a mask is present, fuse the mask multiplication and the per‑position
   gating (`left = left_proj * mask * left_gate`,
    `right = right_proj * mask * right_gate`) with a small Triton kernel.
   When `nomask` is True the mask is omitted and the gating is performed
   directly in PyTorch.
4. Batched matrix‑multiply across the sequence dimension:
   For each hidden channel `h`,
        out[..., h] = left[..., :, :, h] @ right[..., :, :, h].T
   This is realized by reshaping to (B·hidden, N, N) and using
   `torch.bmm`, which invokes cuBLAS for maximal throughput.
5. Layer‑normalize the result (`out_norm`) with its own weight/bias.
6. Apply the output gate (`out_norm * out_gate`).
7. Project back to the original channel dimension with a final linear
   (`to_out_weight`).

Only the mask‑+‑gate fusion (step 3) is written in Triton, satisfying the
requirement that *part* of the computation runs on a custom kernel while the
cubic O(N³) multiplication leverages highly‑optimized cuBLAS.

The kernel works for any batch size `B∈{1,2}`, sequence length `N∈{128,…,1024}`,
input dimension `dim∈{128,384}` and hidden dimension `hidden_dim` (≤ 128 in
the test suite).  All tensors are assumed to be on the GPU.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion_kernel(
    # tensors
    left_proj_ptr, right_proj_ptr,          # [B,N,N,H] – will be overwritten in‑place
    left_gate_ptr, right_gate_ptr,          # [B,N,N,H]
    mask_ptr,                               # [B,N,N]
    # sizes
    B, N, H,
    # strides (contiguous layout, but taken from the tensors for safety)
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_h,
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    # compile‑time tile size (must be ≥ hidden_dim)
    BLOCK_H: tl.constexpr,
):
    """
    In‑place fuse mask and gate into left/right projections:
        left  = left_proj * mask * left_gate
        right = right_proj * mask * right_gate
    One program instance processes a single (b,i,j) position and vectorizes
    over the hidden dimension H.
    """
    pid = tl.program_id(0)                       # linear index over B·N·N
    total = B * N * N
    if pid >= total:
        return

    # decode 3‑D indices
    b = pid // (N * N)
    tmp = pid % (N * N)
    i = tmp // N
    j = tmp % N

    # ------------------------------------------------------------------
    # Load scalar mask value and broadcast to hidden dimension
    # ------------------------------------------------------------------
    mask_off = mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    mask_val = tl.load(mask_off).to(tl.float32)

    # ------------------------------------------------------------------
    # Vector offsets for the hidden dimension
    # ------------------------------------------------------------------
    off_h = tl.arange(0, BLOCK_H)
    mask_h = off_h < H                                 # guard for H < BLOCK_H

    # ------------------------------------------------------------------
    # LEFT projection + gate
    # ------------------------------------------------------------------
    left_off = (left_proj_ptr
                + b * stride_lp_b
                + i * stride_lp_i
                + j * stride_lp_j
                + off_h * stride_lp_h)
    left_val = tl.load(left_off, mask=mask_h, other=tl.float32(0.0))

    left_gate_off = (left_gate_ptr
                     + b * stride_lg_b
                     + i * stride_lg_i
                     + j * stride_lg_j
                     + off_h * stride_lg_h)
    left_gate_val = tl.load(left_gate_off, mask=mask_h, other=tl.float32(0.0))

    left_out = left_val * mask_val * left_gate_val
    tl.store(left_off, left_out, mask=mask_h)          # write back in‑place

    # ------------------------------------------------------------------
    # RIGHT projection + gate
    # ------------------------------------------------------------------
    right_off = (right_proj_ptr
                 + b * stride_rp_b
                 + i * stride_rp_i
                 + j * stride_rp_j
                 + off_h * stride_rp_h)
    right_val = tl.load(right_off, mask=mask_h, other=tl.float32(0.0))

    right_gate_off = (right_gate_ptr
                      + b * stride_rg_b
                      + i * stride_rg_i
                      + j * stride_rg_j
                      + off_h * stride_rg_h)
    right_gate_val = tl.load(right_gate_off, mask=mask_h, other=tl.float32(0.0))

    right_out = right_val * mask_val * right_gate_val
    tl.store(right_off, right_out, mask=mask_h)        # write back in‑place


def custom_kernel(data):
    """
    Entry‑point for the TriMul forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Load model weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) and gate logits
    # ------------------------------------------------------------------
    # shape after linear: [B,N,N,hidden_dim]
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight)
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)

    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()  # [B,N,N,hidden_dim]

    B, N, _, _ = left_proj.shape

    # ------------------------------------------------------------------
    # 3) Fuse mask and per‑position gates (if a mask is present)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # Ensure mask is float and on same device
        mask = mask.to(dtype=input_tensor.dtype, device=input_tensor.device)

        # Strides for the tensors (contiguous layout guarantees correctness)
        lp_b, lp_i, lp_j, lp_h = left_proj.stride()
        rp_b, rp_i, rp_j, rp_h = right_proj.stride()
        lg_b, lg_i, lg_j, lg_h = left_gate.stride()
        rg_b, rg_i, rg_j, rg_h = right_gate.stride()
        mask_b, mask_i, mask_j = mask.stride()

        total = B * N * N
        grid = (total,)

        # Launch the Triton kernel – it overwrites left_proj/right_proj in‑place
        _mask_gate_fusion_kernel[grid](
            left_proj, right_proj,
            left_gate, right_gate,
            mask,
            B, N, hidden_dim,
            lp_b, lp_i, lp_j, lp_h,
            rp_b, rp_i, rp_j, rp_h,
            lg_b, lg_i, lg_j, lg_h,
            rg_b, rg_i, rg_j, rg_h,
            mask_b, mask_i, mask_j,
            BLOCK_H=hidden_dim,
        )
        # After the kernel, left_proj/right_proj already contain the gated + masked values
        left = left_proj
        right = right_proj
    else:
        # No mask: just apply gates (element‑wise)
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4) Batched matrix multiplication over the sequence dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) → matmul → reshape back
    left_reshape = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_reshape = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Compute `left @ right.T` for each hidden slice using cuBLAS batched GEMM
    out = torch.bmm(left_reshape, right_reshape.transpose(1, 2))
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 5) Output layer‑norm
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6) Apply output gate
    # ------------------------------------------------------------------
    out_gated = out_norm * out_gate

    # ------------------------------------------------------------------
    # 7) Final projection back to `dim`
    # ------------------------------------------------------------------
    # weight shape: [dim, hidden_dim]
    final_out = torch.nn.functional.linear(out_gated, to_out_weight)  # [B,N,N,dim]

    return final_out
