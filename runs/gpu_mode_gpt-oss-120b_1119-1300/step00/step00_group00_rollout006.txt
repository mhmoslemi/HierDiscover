    pid_j = tl.program_id(1)

    h_mask = h < H

"""
TriMul (outgoing) forward implementation.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension (C).
2. Half‑precision linear projections:
      left  = x @ W_left      (B,N,N,hidden)
      right = x @ W_right
3. Optional pair‑wise mask (broadcasted on the last dim) applied to left/right.
4. Gating:
      left_gate  = sigmoid(x @ W_left_gate)
      right_gate = sigmoid(x @ W_right_gate)
      out_gate   = sigmoid(x @ W_out_gate)
   Apply the first two gates element‑wise to left/right.
5. Core TriMul contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This 4‑D “outer‑product‑sum” is realised with a custom Triton kernel
   that performs a blocked GEMM for each hidden slice, accumulating in FP32
   and storing the result in FP16.
6. Hidden‑dim LayerNorm + out‑gate.
7. Final linear projection back to the original dimension (FP32 output).

The heavy O(N³·hidden) contraction is executed on‑GPU with a fused Triton
kernel, while the remaining cheap operations are performed with PyTorch
(either FP16 or FP32) for simplicity.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# ------------------------------------------------------------
# Triton kernel:   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # problem sizes
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """Batched (batch, hidden) matrix multiplication.
    Each program instance handles a tile (i, j) of the output matrix for a
    specific batch and a block of hidden dimensions."""
    pid_i = tl.program_id(0)    # tile over output rows   (i)
    pid_j = tl.program_id(1)    # tile over output cols   (j)
    pid_z = tl.program_id(2)    # combined (batch, hidden‑block)

    # -----------------------------------------------------------------
    # decode pid_z -> batch index + hidden‑block index
    # -----------------------------------------------------------------
    num_h_blocks = tl.cdiv(H, BLOCK_H)
    pid_b  = pid_z // num_h_blocks                # batch
    pid_hb = pid_z %  num_h_blocks                # hidden‑block within batch

    # -----------------------------------------------------------------
    # tile offsets (may be out of bounds --> masked later)
    # -----------------------------------------------------------------
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    h = pid_hb * BLOCK_H + tl.arange(0, BLOCK_H)

    i_mask = i < N
    j_mask = j < N
    h_mask = h < H

    # accumulator for each hidden slice (FP32 for accuracy)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # -----------------------------------------------------------------
    # loop over k dimension
    # -----------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k < N

        # -----------------------------------------------------------------
        # inner loop over hidden slices inside the current block
        # -----------------------------------------------------------------
        for hh in range(0, BLOCK_H):
            # ----- early‑out if hidden index is OOB
            valid_h = h_mask[hh]

            # ----- compute absolute hidden index (scalar)
            hid = pid_hb * BLOCK_H + hh   # scalar int

            # ----- pointers for the current hidden slice
            left_off = (pid_b * stride_lb
                        + i[:, None] * stride_li
                        + k[None, :] * stride_lk
                        + hid * stride_lh)
            right_off = (pid_b * stride_rb
                         + j[:, None] * stride_rj
                         + k[None, :] * stride_rk
                         + hid * stride_rh)

            # ----- load tiles (M×K) and (N×K) – original tensors are FP16
            left_tile = tl.load(left_ptr + left_off,
                                mask=i_mask[:, None] & k_mask[None, :],
                                other=0.0).to(tl.float32)
            right_tile = tl.load(right_ptr + right_off,
                                 mask=j_mask[:, None] & k_mask[None, :],
                                 other=0.0).to(tl.float32)

            # ----- GEMM over K -> (M,N)
            prod = tl.dot(left_tile, right_tile)   # (BLOCK_M, BLOCK_N)

            # ----- accumulate
            acc[:, :, hh] = acc[:, :, hh] + prod * tl.where(valid_h, 1.0, 0.0)

    # -----------------------------------------------------------------
    # write results back (store as FP16)
    # -----------------------------------------------------------------
    for hh in range(0, BLOCK_H):
        hid = pid_hb * BLOCK_H + hh
        valid_h = hid < H

        out_off = (pid_b * stride_ob
                   + i[:, None] * stride_oi
                   + j[None, :] * stride_oj
                   + hid * stride_oh)

        mask_store = i_mask[:, None] & j_mask[None, :] & valid_h
        tl.store(out_ptr + out_off,
                 acc[:, :, hh].to(tl.float16),
                 mask=mask_store)


# ------------------------------------------------------------
# Helper: batched TriMul using the Triton kernel
# ------------------------------------------------------------
def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right: [B, N, N, H] (FP16, contiguous)
    returns:     [B, N, N, H] (FP16)
    """
    B, N, N2, H = left.shape
    assert N == N2, "Input must be square on the two middle axes."
    out = torch.empty_like(left)

    # strides as Python ints (bytes/element)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # ------------------- blocking parameters (tuned for H100) -------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_H = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,                     # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,                     # j‑tiles
        B * ((H + BLOCK_H - 1) // BLOCK_H)               # batch × hidden‑blocks
    )

    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
        num_warps=8,  # H100 works well with 8 warps per block
    )
    return out

# ------------------------------------------------------------
def custom_kernel(data):
    """
    data = (input_tensor, mask, weights, config)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # -----------------------------------------------------------------
    # unpack weights (ensure correct dtype/device)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)

    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    to_out_weight = weights["to_out.weight"].to(device)

    # -----------------------------------------------------------------
    # 1) Input LayerNorm over channel dim (C)
    # -----------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # -----------------------------------------------------------------
    # Cast to FP16 for the bulk of the work
    # -----------------------------------------------------------------
    x_h = x.half()

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x_h, left_proj_weight.half())   # [B,N,N,hidden]
    right = F.linear(x_h, right_proj_weight.half())

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcast on the last dim)
    # -----------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        # mask is [B, N, N]; expand to [B, N, N, 1]
        mask_f = mask.to(x_h.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4) Gating (sigmoid) – also in FP16
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_weight.half()))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_weight.half()))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight.half()))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Core TriMul contraction via Triton (still FP16)
    # -----------------------------------------------------------------
    out = tri_mul_triton(left, right)               # [B,N,N,hidden] (FP16)

    # -----------------------------------------------------------------
    # 6) Hidden‑dim LayerNorm + out‑gate
    # -----------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight.half(),
                       bias=to_out_norm_bias.half(),
                       eps=eps)
    out = out * out_gate

    # -----------------------------------------------------------------
    # 7) Final projection back to original dim (float32 output)
    # -----------------------------------------------------------------
    # Cast to FP32 before the final linear to match reference implementation
    out = F.linear(out.float(), to_out_weight.to(device))   # [B,N,N,dim]

    return out
