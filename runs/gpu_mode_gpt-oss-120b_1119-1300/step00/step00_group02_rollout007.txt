Implementation details for kernel:

```
@triton.jit
def tri_mul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_i, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_h = tl.program_id(2)

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J
    h_base = pid_h * BLOCK_H

    # offsets
    offs_i = i_base + tl.arange(0, BLOCK_I)
    offs_j = j_base + tl.arange(0, BLOCK_J)
    offs_h = h_base + tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # Loop over k dimension
    for k_base in range(0, N, BLOCK_K):
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute mask for load
        mask_left = mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]
        mask_right = mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]

        # compute pointers for left tile
        left_tile_ptr = left_ptr + (offs_i[:, None, None] * stride_left_i) \
                                    + (offs_k[None, :, None] * stride_left_k) \
                                    + (offs_h[None, None, :] * stride_left_h)

        # compute pointers for right tile
        right_tile_ptr = right_ptr + (offs_j[:, None, None] * stride_right_i) \
                                      + (offs_k[None, :, None] * stride_right_k) \
                                      + (offs_h[None, None, :] * stride_right_h)

        # load
        left_tile = tl.load(left_tile_ptr, mask=mask_left, other=0.0).to(tl.float32)
        right_tile = tl.load(right_tile_ptr, mask=mask_right, other=0.0).to(tl.float32)

        # iterate over k within tile
        for kk in range(BLOCK_K):
            k_valid = mask_k[kk]
            if not k_valid:
                continue

            # left_vec shape (BLOCK_I, BLOCK_H)
            left_vec = left_tile[:, kk, :]

            # right_vec shape (BLOCK_J, BLOCK_H)
            right_vec = right_tile[:, kk, :]

            # broadcast to compute outer product
            # left_vec[:, None, :] shape (BLOCK_I, 1, BLOCK_H)
            # right_vec[None, :, :] shape (1, BLOCK_J, BLOCK_H)
            acc += left_vec[:, None, :] * right_vec[None, :, :]
    # Store result
    out_ptrs = out_ptr + (offs_i[:, None, None] * stride_out_i) \
                       + (offs_j[None, :, None] * stride_out_j) \
                       + (offs_h[None, None, :] * stride_out_h)
    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])
```

def tri_mul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_i, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_h = tl.program_id(2)

    i_offset = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    h_offset = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_i = i_offset < N
    mask_j = j_offset < N
    mask_h = h_offset < H

    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # Loop over k dimension
    for k_base in range(0, N, BLOCK_K):
        k_offset = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load left tile: shape (BLOCK_I, BLOCK_K, BLOCK_H)
        left_ptrs = left_ptr \
            + i_offset[:, None, None] * stride_left_i \
            + k_offset[None, :, None] * stride_left_k \
            + h_offset[None, None, :] * stride_left_h

        right_ptrs = right_ptr \
            + j_offset[:, None, None] * stride_right_i \
            + k_offset[None, :, None] * stride_right_k \
            + h_offset[None, None, :] * stride_right_h

        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0).to(tl.float32)
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0).to(tl.float32)

        # For each k within tile, compute outer product across i,j per hidden dim
        for kk in range(BLOCK_K):
            k_valid = mask_k[kk]

            if not k_valid:

    # Store result

        + i_offset[:, None, None] * stride_out_i \
        + j_offset[None, :, None] * stride_out_j \
        + h_offset[None, None, :] * stride_out_h

"""
TriMul (outgoing) implementation for AlphaFold3.

The heavy part of the operator is the Tensor Einsum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where left and right are the projected and gated pairwise representations.
The einsum is a batched “outer‑product‑accumulate” over the sequence
dimension and is implemented in a custom Triton kernel.  The kernel tiles the
(i, j, d) output space and iterates over the k‑dimension in small blocks,
loading a (BLOCK_K, BLOCK_H) slice of left and right per thread and updating
the accumulator with an outer product.  This reduces memory traffic compared
to a naïve per‑element implementation while keeping the whole reduction on the
GPU.

All other operations (LayerNorm, Linear projections, gating, final
normalisation and output projection) are performed with PyTorch primitives.
The kernel works for arbitrary batch sizes (B≤2), sequence lengths (N up to
1024) and hidden dimensions (hidden_dim up to 384) and respects optional
masking.  The final output is a float‑32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_einsum_kernel(
    left_ptr,            # float16 *
    right_ptr,           # float16 *
    out_ptr,             # float32 *
    N, H,                # int32: sequence length, hidden dimension
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_i, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Compute: out[i, j, h] = sum_k left[i, k, h] * right[j, k, h]
    for a single batch (batch offset already added by the caller).
    """
    pid_i = tl.program_id(0)   # index of i‑tile
    pid_j = tl.program_id(1)   # index of j‑tile
    pid_h = tl.program_id(2)   # index of h‑tile

    # Global offsets for this tile
    i_off = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_off = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    h_off = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_i = i_off < N
    mask_j = j_off < N
    mask_h = h_off < H

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # Loop over k in blocks
    for k_base in range(0, N, BLOCK_K):
        k_off = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # Pointers for the current k‑block
        left_ptrs = left_ptr \
            + i_off[:, None, None] * stride_left_i \
            + k_off[None, :, None] * stride_left_k \
            + h_off[None, None, :] * stride_left_h

        right_ptrs = right_ptr \
            + j_off[:, None, None] * stride_right_i \
            + k_off[None, :, None] * stride_right_k \
            + h_off[None, None, :] * stride_right_h

        # Load tiles (masked loads return 0 outside the valid region)
        left_tile = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0
        ).to(tl.float32)   # (BLOCK_I, BLOCK_K, BLOCK_H)

        right_tile = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0
        ).to(tl.float32)   # (BLOCK_J, BLOCK_K, BLOCK_H)

        # Outer‑product‑accumulate over the inner k dimension
        for kk in range(BLOCK_K):
            # left_slice: (BLOCK_I, BLOCK_H)
            left_slice = left_tile[:, kk, :]
            # right_slice: (BLOCK_J, BLOCK_H)
            right_slice = right_tile[:, kk, :]
            # Broadcast to (BLOCK_I, BLOCK_J, BLOCK_H) and accumulate
            acc += left_slice[:, None, :] * right_slice[None, :, :]

    # Store the result with masking for the tail of each dimension
    out_ptrs = out_ptr \
        + i_off[:, None, None] * stride_out_i \
        + j_off[None, :, None] * stride_out_j \
        + h_off[None, None, :] * stride_out_h

    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor: [B, N, N, dim]   (float32)
        - mask_tensor : [B, N, N]       (bool / float, 0/1)
        - weights_dict: model parameters (torch tensors on the same device)
        - config_dict : contains "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack arguments
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]    # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]

    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim] (float32)
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted over hidden dimension)
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1).to(x.dtype)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating networks
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # float32
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # used later (float32)

    # Cast gates to fp16 to keep the same dtype as the projected tensors
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 5) Apply gates
    # ------------------------------------------------------------------
    left = left.to(torch.float16) * left_gate
    right = right.to(torch.float16) * right_gate

    # ------------------------------------------------------------------
    # 6) Triton kernel for the einsum (outgoing TriMul)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape   # N is the sequence length
    H = hidden_dim

    # Ensure tensors are contiguous (required for correct stride handling)
    left = left.contiguous()
    right = right.contiguous()

    # Allocate output buffer (float32 as required by downstream ops)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # Strides in element units (batch stride is omitted – we offset per‑batch manually)
    _, stride_left_i, stride_left_k, stride_left_h = left.stride()
    _, stride_right_i, stride_right_k, stride_right_h = right.stride()
    _, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Choose block sizes – tuned for H100 and the supported dimensions
    BLOCK_I = 16          # tile size along the first sequence axis
    BLOCK_J = 16          # tile size along the second sequence axis
    BLOCK_K = 4           # inner‑k tile (balances register pressure)
    BLOCK_H = 32          # hidden‑dim tile (fits into warp registers)

    # Grid dimensions (i‑tile, j‑tile, h‑tile)
    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
        (H + BLOCK_H - 1) // BLOCK_H,
    )

    # Launch the kernel once per batch (B ≤ 2, so the loop overhead is negligible)
    left_ptr = left.data_ptr()
    right_ptr = right.data_ptr()
    out_ptr = out.data_ptr()
    batch_stride = left.stride(0)  # elements between batches

    for b in range(B):
        # Offsets for the current batch (pointer arithmetic in element units)
        left_batch_ptr = left_ptr + b * batch_stride
        right_batch_ptr = right_ptr + b * batch_stride
        out_batch_ptr = out_ptr + b * batch_stride   # out has the same layout as left/right

        tri_mul_einsum_kernel[grid](
            left_batch_ptr,
            right_batch_ptr,
            out_batch_ptr,
            N,
            H,
            stride_left_i,
            stride_left_k,
            stride_left_h,
            stride_right_i,
            stride_right_k,
            stride_right_h,
            stride_out_i,
            stride_out_j,
            stride_out_h,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_H=BLOCK_H,
            num_warps=4,            # 4 warps give good occupancy on H100
        )

    # ------------------------------------------------------------------
    # 7) Output normalisation, gating and final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Normalise over the hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # Apply the output gate (still float32)
    out = out * out_gate

    # Final linear projection back to `dim` (no bias)
    output = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return output
