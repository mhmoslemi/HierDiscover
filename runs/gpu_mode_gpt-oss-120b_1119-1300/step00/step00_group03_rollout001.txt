def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, M, N, K, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    # compute batch and hidden from pid_z
    batch = pid_z // H
    hidden = pid_z % H

    # offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers with batch and hidden offsets
    a_base = left_ptr + batch * stride_left_b + hidden * stride_left_h
    b_base = right_ptr + batch * stride_right_b + hidden * stride_right_h

    # initialize accumulator

"""
TriMul “outgoing” implementation for AlphaFold‑3 style pairwise updates.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (shape [B,N,N,D]).
2. Compute four linear projections (left/right values and three gates) on the
   last dimension D → hidden dimension H.
3. Optionally mask the projected values.
4. Apply sigmoid gates to the projected values.
5. Core operation – compute, for every batch and hidden channel,
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
   This is a batched matrix multiplication `left @ rightᵀ` performed by a custom
   Triton kernel that tiles the N×N reduction (M,N) and accumulates over K.
6. Layer‑norm the result on the hidden dimension, multiply by the output gate,
   and finally a linear projection back to the original dimension D.

Only the heavy N³ reduction (step 5) is written in Triton; the surrounding
operations are performed with PyTorch (highly‑optimised CUDA kernels).
"""

import math
from typing import Tuple, Dict

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the `einsum`
# out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    # Pointers
    left_ptr, right_ptr, out_ptr,
    # Compile‑time shape constants
    B, M, N, K, H,
    # Strides (in elements) for the three tensors
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    # Tile sizes (must be constexpr)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)          # tile row   (i)
    pid_j = tl.program_id(1)          # tile col   (j)
    pid_z = tl.program_id(2)          # combined batch & hidden index

    # ------------------------------------------------------------------
    # Decode batch index and hidden channel from pid_z
    # ------------------------------------------------------------------
    batch = pid_z // H
    hidden = pid_z % H

    # ------------------------------------------------------------------
    # Offsets for the i‑ and j‑axes of the output tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # ------------------------------------------------------------------
    # Base pointers that already include the batch and hidden offset
    # ------------------------------------------------------------------
    a_base = left_ptr + batch * stride_l_b + hidden * stride_l_h
    b_base = right_ptr + batch * stride_r_b + hidden * stride_r_h
    c_base = out_ptr + batch * stride_o_b + hidden * stride_o_h

    # Accumulator – use float32 regardless of the input dtype for numerical safety
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction axis k in BLOCK_K sized chunks
    # ------------------------------------------------------------------
    for k_start in range(0, K, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # Pointers to the current K‑slice of A and B
        a_ptrs = a_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        b_ptrs = b_base + offs_j[:, None] * stride_r_i + offs_k[None, :] * stride_r_k

        # Masks for out‑of‑bounds loads (needed only on the last K‑tile)
        mask_a = (offs_i[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_j[:, None] < N) & (offs_k[None, :] < K)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0).to(tl.float32)

        # tl.dot performs Σ_k a[i,k] * b[j,k] → shape (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write the tile back to C, respecting the output matrix bounds
    # ------------------------------------------------------------------
    c_ptrs = c_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    mask_c = (offs_i[:, None] < M) & (offs_j[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right: [B, N, N, H] tensors (float32, contiguous)
    returns: out = left @ rightᵀ  (same shape)
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    # Ensure contiguous layout for simple stride calculations
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # Strides in elements (PyTorch strides are already in element units)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tile sizes – tuned for H100 (block‑wise shared‑memory fits)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid dimensions
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # j‑tiles
        B * H                                   # batch × hidden
    )

    # Launch the Triton kernel
    tri_mul_outgoing_kernel[grid](
        left, right, out,
        B, N, N, N, H,                     # compile‑time shape constants
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul operator.
    data = (input_tensor, mask_tensor, weights, config)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    use_mask = not config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Layer‑norm on the input (normalized over the last dim)
    # --------------------------------------------------------------
    eps = 1e-5
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias) → [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast on the hidden dimension)
    # --------------------------------------------------------------
    if use_mask:
        mask = mask_tensor.unsqueeze(-1).to(x.dtype)  # [B, N, N, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # --------------------------------------------------------------
    # 4. Gates (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates to the projected values
    left = left_proj * left_gate          # [B, N, N, hidden_dim]
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5. Core N³ reduction using the Triton kernel
    # --------------------------------------------------------------
    out = tri_mul_triton(left, right)    # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 6. Output layer‑norm on the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------
    # 7. Apply output gate and final linear projection back to dim
    # --------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
