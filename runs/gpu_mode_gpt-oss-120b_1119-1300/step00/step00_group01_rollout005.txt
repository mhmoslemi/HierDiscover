                     stride_xm, stride_xc, stride_outm, stride_outc,
                     BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    if pid >= M: return (maybe exit)

    # offsets for this row
    offs_d = tl.arange(0, BLOCK_D)
    mask = offs_d < D

    # compute pointer for x and out
    x_ptrs = x_ptr + pid * stride_xm + offs_d * stride_xc
    # load
    x = tl.load(x_ptrs, mask=mask, other=0.0).to(tl.float32) # convert to float

    # compute mean
    # sum across channels
    # Use reduction
    # We can compute using tl.sum directly
    mean = tl.sum(x, axis=0) / D

    # compute variance
    var = tl.sum((x - mean) ** 2, axis=0) / D

    # normalize
    inv_std = 1.0 / tl.sqrt(var + epsilon)

    # apply weight and bias if provided

def layernorm_kernel(x_ptr, out_ptr, weight_ptr, bias_ptr, epsilon, M, D,
                     stride_xm, stride_xc, stride_outm, stride_outc,
                     BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    if pid >= M:
        return

    offs_d = tl.arange(0, BLOCK_D)
    mask = offs_d < D

    # Load x row
    x_ptrs = x_ptr + pid * stride_xm + offs_d * stride_xc
    x = tl.load(x_ptrs, mask=mask, other=0.0).to(tl.float32)

    # Compute mean
    sum_val = tl.sum(x, axis=0)
    mean = sum_val / D

    # Compute variance
    diff = x - mean
    # zero out invalid channels
    diff = tl.where(mask, diff, 0.0)
    var = tl.sum(diff * diff, axis=0) / D

    inv_std = 1.0 / tl.sqrt(var + epsilon)

    y = diff * inv_std

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Extract weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    B, N, _, C = input_tensor.shape
    M = B * N * N

    # Flatten input
    x_flat = input_tensor.reshape(M, C).contiguous()

    # Normalize x using Triton
    x_norm_flat = triton_layernorm(x_flat, norm_weight, norm_bias, eps=eps)

    # Linear projections
    left = torch.nn.functional.linear(x_norm_flat, left_proj_weight)
    right = torch.nn.functional.linear(x_norm_flat, right_proj_weight)

    # Gates
    left_gate = torch.nn.functional.linear(x_norm_flat, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm_flat, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm_flat, out_gate_weight).sigmoid()

    # Apply mask if provided (if shape matches)
    if mask is not None:
        mask_flat = mask.reshape(M, 1).to(left.dtype)
        left = left * mask_flat
        right = right * mask_flat

    # Apply gate
    left = left * left_gate
    right = right * right_gate

    # Reshape for GEMM
    left = left.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()

    # Compute bilinear product via batched GEMM
    left_flat = left.view(B * hidden_dim, N, N)
    right_flat = right.view(B * hidden_dim, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Apply to_out_norm (layernorm) using Triton
    out_flat_norm_input = out.reshape(M, hidden_dim).contiguous()
    out_norm_flat = triton_layernorm(out_flat_norm_input, to_out_norm_weight, to_out_norm_bias, eps=eps)

    # Reshape back
    out_norm = out_norm_flat.view(B, N, N, hidden_dim)

    # Apply out_gate
    out_norm = out_norm * out_gate.view(B, N, N, hidden_dim)

    # Final linear projection to dim
    out_final = torch.nn.functional.linear(out_norm.view(M, hidden_dim), to_out_weight)

    out = out_final.view(B, N, N, dim)

    return out

import torch
import triton
import triton.language as tl

def triton_layernorm(x, weight, bias, eps=1e-5):
    """
    Apply LayerNorm on the last dimension of a 2‑D tensor ``[M, D]`` using a
    Triton kernel. ``weight`` and ``bias`` are 1‑D tensors of size ``D``.
    The kernel computes per‑row mean and variance, normalises the row and
    finally applies the affine transformation ``y = (x - μ) / sqrt(σ² + ε) *
    weight + bias``.
    """
    assert x.dim() == 2
    M, D = x.shape
    # Ensure contiguous layout
    x = x.contiguous()
    out = torch.empty_like(x)
    # Move parameters to same dtype/device as x
    weight = weight.to(x.dtype)
    bias = bias.to(x.dtype)

    # Pointer and stride information
    x_ptr = x.data_ptr()
    out_ptr = out.data_ptr()
    w_ptr = weight.data_ptr()
    b_ptr = bias.data_ptr()

    # For a contiguous [M, D] the stride between rows is D, between columns 1
    stride_xm = D
    stride_xc = 1
    stride_outm = D
    stride_outc = 1

    # Triton launch configuration
    BLOCK_D = 384  # maximum dimension we expect (C or hidden_dim <= 384)
    grid = (M,)

    layernorm_kernel[grid](x_ptr, out_ptr, w_ptr, b_ptr,
                           eps, M, D,
                           stride_xm, stride_xc,
                           stride_outm, stride_outc,
                           BLOCK_D=BLOCK_D,
                           num_warps=4)

    return out

@triton.jit
def layernorm_kernel(x_ptr, out_ptr, weight_ptr, bias_ptr,
                     epsilon, M, D,
                     stride_xm, stride_xc,
                     stride_outm, stride_outc,
                     BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    if pid >= M:
        return

    offs_d = tl.arange(0, BLOCK_D)

    # ------------------------------------------------------------------

    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # [M, hidden_dim]

    right = right * right_gate

    right_bmm = right.view(B * hidden_dim, N, N)

    # ------------------------------------------------------------------

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The reference implementation consists of two LayerNorms, four linear
projections, three sigmoid‑gates and a large O(N³) bilinear term

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d] .

All linear / gating operations are delegated to PyTorch (which uses
cuBLAS/TensorCore).  The two LayerNorms are the only parts rewritten in
Triton – they are applied on a flattened *[M, D]* tensor where *M = B·N·N*.
The kernel computes per‑row mean/variance, normalises the row and finally
applies the affine transformation ``γ·x + β``.

The overall algorithm follows the exact order of the PyTorch reference:

1.  LayerNorm on the input tensor.
2.  Linear projections → ``left`` / ``right``.
3.  Three gate‑projections → sigmoid.
4.  Apply optional mask and element‑wise gates.
5.  Bilinear term via a batched GEMM (torch.bmm) over the *k* dimension.
6.  LayerNorm on the hidden‑dim output.
7.  Apply the ``out_gate`` and a final linear projection back to ``dim``.

The implementation works for the test configurations listed in the
statement (B∈{1,2}, N up to 1024, dim∈{128,384}, hidden_dim∈{128,384})
and runs on Triton 3.3.1 / NVIDIA H100.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    x_ptr, out_ptr, weight_ptr, bias_ptr,
    epsilon, M, D,
    stride_xm, stride_xc, stride_outm, stride_outc,
    BLOCK_D: tl.constexpr,
):
    """LayerNorm kernel for a 2‑D tensor [M, D].

    Each program (pid) processes one row (size D).  The kernel loads the
    row, computes mean/variance, normalises, applies the affine
    transformation and writes the result.
    """
    pid = tl.program_id(0)
    if pid >= M:
        return

    offs_d = tl.arange(0, BLOCK_D)
    mask = offs_d < D

    # ------------------------------------------------------------------
    # Load a single row
    # ------------------------------------------------------------------
    x = tl.load(x_ptr + pid * stride_xm + offs_d * stride_xc,
                mask=mask, other=0.0).to(tl.float32)

    # ------------------------------------------------------------------
    # Mean and variance (masked for D < BLOCK_D)
    # ------------------------------------------------------------------
    row_sum = tl.sum(x, axis=0)
    mean = row_sum / D

    diff = x - mean
    diff = tl.where(mask, diff, 0.0)          # ignore padded elements
    var = tl.sum(diff * diff, axis=0) / D

    inv_std = 1.0 / tl.sqrt(var + epsilon)

    # ------------------------------------------------------------------
    # Normalise and apply affine parameters
    # ------------------------------------------------------------------
    y = diff * inv_std

    w = tl.load(weight_ptr + offs_d, mask=mask, other=0.0)
    b = tl.load(bias_ptr + offs_d, mask=mask, other=0.0)
    y = y * w + b

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    tl.store(out_ptr + pid * stride_outm + offs_d * stride_outc,
             y, mask=mask)


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Apply LayerNorm on the last dimension of ``x`` (shape [M, D]) using
    the Triton kernel defined above.  ``weight`` and ``bias`` must be 1‑D
    tensors of length D.  The function returns a tensor of the same shape
    and dtype as ``x``.
    """
    assert x.dim() == 2, "LayerNorm kernel expects a 2‑D tensor"
    M, D = x.shape

    # Ensure contiguous layout
    x = x.contiguous()
    out = torch.empty_like(x)

    # Cast parameters to the same dtype as the input
    weight = weight.to(x.dtype)
    bias = bias.to(x.dtype)

    # Pointer arithmetic (contiguous [M, D] → row stride = D, col stride = 1)
    x_ptr = x.data_ptr()
    out_ptr = out.data_ptr()
    w_ptr = weight.data_ptr()
    b_ptr = bias.data_ptr()

    stride_xm = D
    stride_xc = 1
    stride_outm = D
    stride_outc = 1

    # Upper bound on the channel dimension used in the test suite
    BLOCK_D = 384

    grid = (M,)
    layernorm_kernel[grid](
        x_ptr, out_ptr, w_ptr, b_ptr,
        eps, M, D,
        stride_xm, stride_xc,
        stride_outm, stride_outc,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul outgoing operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] torch.Tensor
            - mask: [B, N, N] torch.Tensor (or None)
            - weights: dict of model parameters
            - config: dict containing ``dim`` and ``hidden_dim``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    device = input_tensor.device

    # ------------------------------------------------------------------
    # Load and cast all weights to float32 (the Triton kernel works in fp32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device).to(torch.float32)
    norm_bias = weights["norm.bias"].to(device).to(torch.float32)

    left_proj_weight = weights["left_proj.weight"].to(device).to(torch.float32)
    right_proj_weight = weights["right_proj.weight"].to(device).to(torch.float32)

    left_gate_weight = weights["left_gate.weight"].to(device).to(torch.float32)
    right_gate_weight = weights["right_gate.weight"].to(device).to(torch.float32)
    out_gate_weight = weights["out_gate.weight"].to(device).to(torch.float32)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(torch.float32)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device).to(torch.float32)

    to_out_weight = weights["to_out.weight"].to(device).to(torch.float32)

    # ------------------------------------------------------------------
    # Dimensions
    # ------------------------------------------------------------------
    B, N, _, C = input_tensor.shape
    M = B * N * N  # total number of (b, i, j) positions

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton)
    # ------------------------------------------------------------------
    x_flat = input_tensor.reshape(M, C).contiguous().to(torch.float32)
    x_norm = triton_layernorm(x_flat, norm_weight, norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections and gates (pure PyTorch – fast cuBLAS/TensorCore)
    # ------------------------------------------------------------------
    # Projections to hidden dimension
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # [M, hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Gate projections + sigmoid
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    # Apply optional mask (broadcast over the hidden dimension)
    if mask is not None:
        mask_flat = mask.reshape(M, 1).to(x_norm.dtype)
        left = left * mask_flat
        right = right * mask_flat

    # Apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3️⃣ Bilinear term via batched GEMM
    #    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Reshape to [B, hidden_dim, N, N] for the batched matmul
    left = left.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()

    # Merge batch and channel dims → shape [B*hidden_dim, N, N]
    left_bmm = left.view(B * hidden_dim, N, N)
    right_bmm = right.view(B * hidden_dim, N, N)

    # Perform the O(N³) contraction with a batched matrix multiplication
    # left_bmm @ right_bmmᵀ  →  [B*hidden_dim, N, N]
    out_bmm = torch.bmm(left_bmm, right_bmm.transpose(1, 2))

    # Restore original ordering → [B, N, N, hidden_dim]
    out = out_bmm.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Output LayerNorm (second Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.reshape(M, hidden_dim).contiguous()
    out_norm = triton_layernorm(out_flat, to_out_norm_weight, to_out_norm_bias, eps=eps)
    out_norm = out_norm.view(B, N, N, hidden_dim)

    # Apply the final gating
    out_norm = out_norm * out_gate.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    out_final = torch.nn.functional.linear(
        out_norm.reshape(M, hidden_dim), to_out_weight
    )
    out = out_final.view(B, N, N, dim)

    return out
