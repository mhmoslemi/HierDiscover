    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,

    pid_n = tl.program_id(1)
    pid_bh = tl.program_id(2)

    h = pid_bh % H

    j0 = pid_n * BLOCK_N

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Cast to half
    x = input_tensor.to(torch.float16)
    mask_fp16 = mask.to(torch.float16) if mask is not None else torch.ones_like(x[...,0], dtype=torch.float16)
    # Normalization
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)
    x = torch.nn.functional.layer_norm(x, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)
    # Projections
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape B,N,N,H
    right = torch.nn.functional.linear(x, right_proj_weight)
    # Gates
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply mask if available
    if mask is not None:
        mask_exp = mask_fp16.unsqueeze(-1)  # shape B,N,N,1
        left = left * mask_exp
        right = right * mask_exp

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Contract with Triton
    out = tri_mul_einsum(left, right)  # shape B,N,N,hidden_dim, dtype float16

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # Apply out_gate
    out = out * out_gate

    # Final linear
    to_out_weight = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
TriMul (outgoing) implementation.
---------------------------------
The core operation is a batched contraction:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

We fuse this contraction in a custom Triton kernel that tiles the
(i, j) output plane (BLOCK_M × BLOCK_N) and iterates over k in
BLOCK_K chunks, accumulating in float32 and storing as float16.
All surrounding operations (LayerNorm, linear projections, gating)
are performed with PyTorch in half‑precision for speed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batch × hidden_dim independent GEMMs
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile over i
    pid_n = tl.program_id(1)          # tile over j
    pid_bh = tl.program_id(2)         # batch * hidden index

    # ------------------------------------------------------------------
    # Decode batch and hidden indices
    # ------------------------------------------------------------------
    b = pid_bh // H
    h = pid_bh % H

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Accumulator (float32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k dimension in blocks
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_tile * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + h * stride_l_h
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
        )

        # right tile: shape (BLOCK_K, BLOCK_N) – note the transpose on k/j
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + h * stride_r_h
            + offs_k[:, None] * stride_r_k
            + offs_j[None, :] * stride_r_i
        )

        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Accumulate the matrix product for this k‑tile
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + h * stride_o_h
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul_contract(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the einsum contraction using the Triton kernel.
    left/right: [B, N, N, H] (half‑precision)
    Returns out of the same shape and dtype.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Strides in element units (not bytes)
    l_s_b, l_s_i, l_s_k, l_s_h = left.stride()
    r_s_b, r_s_i, r_s_k, r_s_h = right.stride()
    o_s_b, o_s_i, o_s_j, o_s_h = out.stride()

    # Tile sizes – chosen to fit in registers on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,        # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,        # j‑tiles
        B * H                                 # batch * hidden
    )

    tri_mul_einsum_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        l_s_b,
        l_s_i,
        l_s_k,
        l_s_h,
        r_s_b,
        r_s_i,
        r_s_k,
        r_s_h,
        o_s_b,
        o_s_i,
        o_s_j,
        o_s_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # a modest number of warps works well on H100
    )
    return out


# ----------------------------------------------------------------------
# Public entry point – matches the required signature
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        output tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # Cast to half precision for the heavy path
    # --------------------------------------------------------------
    x = input_tensor.to(torch.float16)

    # --------------------------------------------------------------
    # First LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b, eps=1e-5)

    # --------------------------------------------------------------
    # Linear projections (no bias)
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x, left_proj_w)    # [B, N, N, H]
    right = F.linear(x, right_proj_w)  # [B, N, N, H]

    # --------------------------------------------------------------
    # Gating networks (sigmoid)
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------
    # Optional mask (broadcast over the hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        # mask is [B, N, N]; broadcast to [B, N, N, 1]
        mask_fp = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp
        right = right * mask_fp

    # --------------------------------------------------------------
    # Apply gates (element‑wise)
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # Core contraction via Triton
    # --------------------------------------------------------------
    out = _tri_mul_contract(left, right)                # [B, N, N, H]

    # --------------------------------------------------------------
    # to_out_norm (LayerNorm over hidden_dim)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w,
                       bias=to_out_norm_b, eps=1e-5)

    # --------------------------------------------------------------
    # Apply output gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # Final linear projection back to original dim
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)            # [B, N, N, dim]

    # Return in float32 for downstream compatibility
    return out.to(torch.float32)
