def tri_mul_kernel_launcher(left, right):
   B,N,N,H = left.shape
   # ensure contiguous
   left = left.contiguous()
   right = right.contiguous()
   out = torch.empty_like(left)
   # ensure dtype is fp16 for left/right
   if left.dtype != torch.float16:
       left_fp16 = left.to(torch.float16)
   else:
       left_fp16 = left
   # ensure same dtype for right
   if right.dtype != torch.float16:
       right_fp16 = right.to(torch.float16)
   else:
       right_fp16 = right
   # out must be fp32?
   out_fp32 = torch.empty((B,N,N,H), device=left.device, dtype=torch.float32)

   # compute strides
   stride_b_left, stride_i_left, stride_k_left, stride_h_left = left_fp16.stride()
   stride_b_right, stride_i_right, stride_k_right, stride_h_right = right_fp16.stride()
   stride_b_out, stride_i_out, stride_j_out, stride_h_out = out_fp32.stride()

   grid = (B*H, (N + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N)

    hidden_dim = config["hidden_dim"]

    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])

        left_proj = left_proj * mask
        right_proj = right_proj * mask

    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()

```python
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

def tri_mul_kernel_launcher(left_fp16: torch.Tensor, right_fp16: torch.Tensor) -> torch.Tensor:
    # left/right shape [B,N,N,hidden] contiguous fp16.
    B, N, _, H = left_fp16.shape
    out_fp16 = torch.empty_like(left_fp16)  # fp16
    # strides
    stride_bl, stride_il, stride_kl, stride_hl = left_fp16.stride()
    stride_br, stride_ir, stride_kr, stride_hr = right_fp16.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out_fp16.stride()
    # Block sizes (tuned for H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    # Compute grid: (batch*hidden, num_i_tiles, num_j_tiles)
    grid = (B * H, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)
    # Launch kernel
    tri_mul_kernel[grid](
        left_fp16, right_fp16, out_fp16,
        B, N, H,
        stride_bl, stride_il, stride_kl, stride_hl,
        stride_br, stride_ir, stride_kr, stride_hr,
        stride_bo, stride_io, stride_jo, stride_ho,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4
    )
    return out_fp16

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_ir, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    using a blocked matrix multiplication where the right operand is
    loaded in transposed order (k,j) to turn the outer product into a
    standard GEMM. The kernel processes one (batch, hidden) pair per
    program and tiles the i and j dimensions.
    """
    pid = tl.program_id(0)  # combines batch and hidden
    batch_idx = pid // H
    hid_idx   = pid % H

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Offsets for the i and j tile
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # Load left tile L[i,k,h]
        left_off = batch_idx * stride_bl \
                 + i[:, None] * stride_il \
                 + k_idx[None, :] * stride_kl \
                 + hid_idx * stride_hl
        left = tl.load(left_ptr + left_off, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile transposed R_T[k,j,h] = right[b,j,k,h]
        right_off = batch_idx * stride_br \
                  + k_idx[:, None] * stride_ir \
                  + j[None, :] * stride_kr \
                  + hid_idx * stride_hr
        right = tl.load(right_ptr + right_off, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Accumulate GEMM
        acc += tl.dot(left, right)

    # Write result
    out_off = batch_idx * stride_bo \
            + i[:, None] * stride_io \
            + j[None, :] * stride_jo \
            + hid_idx * stride_ho
    tl.store(out_ptr + out_off, acc, mask=mask_i[:, None] & mask_j[None, :])
```

"""
Optimized Triton implementation of the “outgoing” TriMul operator from AlphaFold‑3.

Algorithm
---------

*   Apply a LayerNorm to the input tensor X ∈ ℝ^{B×N×N×C}.
*   Compute four linear projections of X (left/right and three gating
    vectors).  The gating vectors are passed through a sigmoid.
*   Multiply the left/right projections by their respective gates and
    (optionally) by a binary mask.
*   The core O(N³) operation is
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]          (1)
    which can be rewritten as a batched matrix multiplication
        out_h = L_h @ R_hᵀ  with   L_h[i,k] = left[i,k,h],
                               R_h[j,k] = right[j,k,h].
    We implement (1) with a Triton kernel that processes one (batch,
    hidden) pair per program and tiles the i‑ and j‑dimensions.  The
    right operand is loaded in transposed order (k,j) so that a standard
    GEMM (`tl.dot`) can be used.
*   The result (still in fp16) is cast back to fp32, Layer‑normalized,
    gated by the output gate, and finally projected back to the original
    C‑dimensional space.

The kernel works for any batch size B∈{1,2}, sequence length N up to
1024 and hidden dimension H up to 384, and gracefully handles the
optional mask.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_ir, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using a blocked GEMM where the right matrix is accessed in
    transposed order (k,j).
    """
    pid = tl.program_id(0)                 # combines batch and hidden id
    batch_idx = pid // H
    hid_idx   = pid % H

    pid_i = tl.program_id(1)               # tile along i
    pid_j = tl.program_id(2)               # tile along j

    # i‑ and j‑coordinates of the current tile
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # accumulator (FP32 for numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # left tile L[i,k,h]
        left_off = batch_idx * stride_bl \
                 + i[:, None] * stride_il \
                 + k_idx[None, :] * stride_kl \
                 + hid_idx * stride_hl
        left = tl.load(left_ptr + left_off,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)          # fp16 load

        # right tile transposed R_T[k,j,h] = right[b,j,k,h]
        right_off = batch_idx * stride_br \
                  + k_idx[:, None] * stride_ir \
                  + j[None, :] * stride_kr \
                  + hid_idx * stride_hr
        right = tl.load(right_ptr + right_off,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)         # fp16 load

        # GEMM accumulation (fp16 × fp16 → fp32)
        acc += tl.dot(left, right)

    # store result (cast back to fp16)
    out_off = batch_idx * stride_bo \
            + i[:, None] * stride_io \
            + j[None, :] * stride_jo \
            + hid_idx * stride_ho
    tl.store(out_ptr + out_off, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def tri_mul_kernel_launcher(left_fp16: torch.Tensor,
                            right_fp16: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches ``tri_mul_kernel``.
    left_fp16 / right_fp16  : [B, N, N, H]  (contiguous, fp16)
    returns out_fp16         : [B, N, N, H]  (fp16)
    """
    B, N, _, H = left_fp16.shape
    out_fp16 = torch.empty_like(left_fp16)    # fp16 output

    # extract strides (in elements, not bytes)
    s_bl, s_il, s_kl, s_hl = left_fp16.stride()
    s_br, s_ir, s_kr, s_hr = right_fp16.stride()
    s_bo, s_io, s_jo, s_ho = out_fp16.stride()

    # Tunable block sizes (work well on H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left_fp16, right_fp16, out_fp16,
        B, N, H,
        s_bl, s_il, s_kl, s_hl,
        s_br, s_ir, s_kr, s_hr,
        s_bo, s_io, s_jo, s_ho,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # enough parallelism for the target shapes
    )
    return out_fp16


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module using a Triton kernel for the
    O(N³) bilinear update.

    Args:
        data: (input, mask, weights, config)
            - input  : torch.Tensor of shape [B, N, N, C]
            - mask   : torch.Tensor of shape [B, N, N] (may be None)
            - weights: dict with the model parameters
            - config : dict containing at least “dim”, “hidden_dim” and
                       optional “nomask”.

    Returns:
        torch.Tensor of shape [B, N, N, C]
    """
    # ------------------------------------------------------------------
    # 1️⃣  Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                       # input channel dimension C
    hidden_dim = config["hidden_dim"]         # projection dimension H
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 2️⃣  Layer‐norm over the last dimension (C)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------
    # 3️⃣  Linear projections (no bias) and optional mask
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if not nomask and mask_tensor is not None:
        # mask shape: [B, N, N] → broadcast over hidden dim
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4️⃣  Gating vectors (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"]
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"]
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"]
    ).sigmoid()

    # ------------------------------------------------------------------
    # 5️⃣  Apply gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate       # [B, N, N, H]
    right = right_proj * right_gate    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣  Triton kernel for the ∑ₖ bilinear product (Eq. 1)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16 = tri_mul_kernel_launcher(left_fp16, right_fp16)   # [B, N, N, H]

    # Cast back to FP32 for the remaining FP32‑only operations
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 7️⃣  Post‑processing: normalize, gate, final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
