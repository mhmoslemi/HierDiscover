def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # layernorm
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    # apply layernorm: (x - mean)/sqrt(var+eps) * weight + bias
    # Compute mean and var across dim axis (last dim). Use torch.float32.
    eps=1e-5 # default for LayerNorm. Possibly weights contain epsilon? Not needed.

    # We can use torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # linear proj (no bias)
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # mask
    if mask is not None and mask.numel()>0:
        mask_exp = mask.unsqueeze(-1).to(left.dtype) # shape [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Permute now for kernel
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()

    # compute out_perm via triton kernel
    out_perm = torch.empty_like(left_perm) # shape (B, H, N, N)
    trimul_outgoing_kernel[grid](left_perm, right_perm, out_perm,
                                 bs, hidden_dim, N,
                                 left_perm.stride(0), left_perm.stride(1), left_perm.stride(2), left_perm.stride(3),
                                 right_perm.stride(0), right_perm.stride(1), right_perm.stride(2), right_perm.stride(3),
                                 out_perm.stride(0), out_perm.stride(1), out_perm.stride(2), out_perm.stride(3),
                                 BLOCK_M=64, BLOCK_N=64, BLOCK_K=64)
    # Permute back
    out = out_perm.permute(0,2,3,1) # shape [B,N,N,H]

    # to_out_norm: layernorm on hidden_dim
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=1e-5)
    # multiply by out_gate
    out = out * out_gate
    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_i, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program id
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_bh = tl.program_id(2)  # batch*head

    # compute indices
    b = pid_bh // H
    h = pid_bh % H

    # compute offsets for rows and cols
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask to ensure not out of bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointers for left
        # left: shape (B, H, N (i), N (k))
        left_ptrs = left_ptr + b*stride_left_b + h*stride_left_h + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left_tile = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # compute pointers for right transposed: we need right[b, h, k, j]
        right_ptrs = right_ptr + b*stride_right_b + h*stride_right_h + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_i

"""
TriMul “outgoing” forward pass implemented with a fused Triton kernel.

The kernel computes, for each batch b and hidden head h (the projected feature
dimension),

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where `left` and `right` are the projected and gated tensors of shape
[B, N, N, H] (H = hidden_dim).  This is the heavy O(N³·H) operation in the
AlphaFold3 triangle multiplicative update.  The surrounding LayerNorms,
Linear projections, and gates are performed in PyTorch, while the
matrix‑multiply‑accumulate over the K dimension is carried out by a custom
Triton kernel that works in FP16 for the inputs and accumulates in FP32 for
numerical stability.

The implementation supports:
* batch size 1‑2
* sequence length up to 1024
* feature dimension up to 384 and hidden dimension up to 384
* optional binary mask (applied before gating)
* mixed‑precision execution (fp16 kernel, fp32 output)

Author: Triton‑enabled TriMul kernel
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # problem size
    B, H, N,
    # strides for left (B,H,i,k)
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    # strides for right (B,H,j,k)  – note j uses the same stride as i in left
    stride_right_b, stride_right_h, stride_right_i, stride_right_k,
    # strides for out (B,H,i,j)
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication over the K dimension.
    Computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
    """

    pid_m = tl.program_id(0)        # block row   (i)
    pid_n = tl.program_id(1)        # block col   (j)
    pid_bh = tl.program_id(2)       # batch*head

    # recover batch and head indices
    b = pid_bh // H
    h = pid_bh % H

    # offsets within the M/N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # out‑of‑bounds masks for i/j
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        # absolute K offsets for this tile
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---------- load A = left[b,h,i,k] ----------
        ptrs_a = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + offs_m[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        a = tl.load(
            ptrs_a,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # ---------- load B = right[b,h,j,k]  (will be used as Bᵀ) ----------
        # we need right[b,h,k,j] => swap the last two axes via pointer arithmetic
        ptrs_b = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + offs_k[:, None] * stride_right_k
            + offs_n[None, :] * stride_right_i
        )
        b_tile = tl.load(
            ptrs_b,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # accumulate the matrix product for this K tile
        acc += tl.dot(a, b_tile)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # ---------- write back ----------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_m[:, None] * stride_out_i
        + offs_n[None, :] * stride_out_j
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple of (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask        : torch.Tensor of shape [B, N, N] (float or bool)
            - weights     : dict with all Linear/LayerNorm weights (float32)
            - config      : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ----------------------------------------------------------------------
    # 1) Input LayerNorm
    # ----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ----------------------------------------------------------------------
    # 2) Linear projections (no bias) and optional masking
    # ----------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if mask is not None:
        mask = mask.to(left.dtype).unsqueeze(-1)      # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ----------------------------------------------------------------------
    # 3) Gating (sigmoid) and apply
    # ----------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )
    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 4) Fuse dimensions for Triton kernel: (B, H, N, N)
    # ----------------------------------------------------------------------
    # bring hidden dimension to the second axis to obtain a batch of matrices
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Cast to FP16 for the kernel (weights are kept in FP32)
    left_fp16 = left_perm.half()
    right_fp16 = right_perm.half()

    # Output tensor (FP32 accumulator)
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32)  # [B, H, N, N]

    # ----------------------------------------------------------------------
    # 5) Launch Triton kernel
    # ----------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,
        (seq_len + BLOCK_N - 1) // BLOCK_N,
        batch_size * hidden_dim,
    )

    trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        batch_size,
        hidden_dim,
        seq_len,
        # strides for left (B,H,i,k)
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        left_fp16.stride(3),
        # strides for right (B,H,j,k) – i‑stride equals left's i‑stride
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        right_fp16.stride(3),
        # strides for output (B,H,i,j)
        out_fp32.stride(0),
        out_fp32.stride(1),
        out_fp32.stride(2),
        out_fp32.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----------------------------------------------------------------------
    # 6) Restore original layout (B, N, N, H)
    # ----------------------------------------------------------------------
    out = out_fp32.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ----------------------------------------------------------------------
    # 7) Output LayerNorm, gating and final projection
    # ----------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                         # element‑wise gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
