"""
TriMul (outgoing) forward kernel.

The module performs the following steps on a 4‑D tensor `x` with shape
[batch, N, N, dim]:

1. Layer‑norm over the last dimension (dim).
2. Two linear projections (`left_proj`, `right_proj`) that map dim → hidden_dim.
3. Optional pair‑wise mask multiplication.
4. Gated element‑wise scaling (three sigmoidal gates).
5. A massive N×N matrix multiplication per hidden channel:
       out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   This is implemented as a batched GEMM on the GPU using a custom Triton
   kernel (one GEMM for each hidden channel, flattened into the batch axis).
6. Layer‑norm over hidden_dim, gated by `out_gate`, and final linear
   projection back to the original dimensionality.

The kernel operates on FP16 data (if the inputs are FP32 they are cast to
FP16 for the heavy matmul) while accumulating in FP32 to preserve
numerical quality. The final result is returned as FP32.

The implementation keeps the heavy N³ work inside Triton and leaves the
lighter linear / activation / norm work to PyTorch, which together meet
the target runtime on an H100.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ B) where
#   A : (batch, M, K)
#   B : (batch, K, N)
#   C : (batch, M, N)
# M == N == K == seq_len.  Hidden dimensions are fused into the batch axis.
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    BATCH, M, N, K,                     # problem sizes
    stride_Ab, stride_Ai, stride_Ak,    # strides of A
    stride_Bb, stride_Bk, stride_Bn,    # strides of B
    stride_Cb, stride_Ci, stride_Cn,    # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel (float16 input, float32 accumulator)."""
    pid_batch = tl.program_id(0)   # which (batch * hidden) matrix
    pid_m = tl.program_id(1)       # block index over rows  (M)
    pid_n = tl.program_id(2)       # block index over cols  (N)

    # Offsets for the current block of rows / columns
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for the current tile
        a_ptrs = (
            A_ptr
            + pid_batch * stride_Ab
            + offs_m[:, None] * stride_Ai
            + offs_k[None, :] * stride_Ak
        )
        b_ptrs = (
            B_ptr
            + pid_batch * stride_Bb
            + offs_k[:, None] * stride_Bk
            + offs_n[None, :] * stride_Bn
        )

        # Load tiles (FP16) and convert to FP32 for accumulation
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back the result
    c_ptrs = (
        C_ptr
        + pid_batch * stride_Cb
        + offs_m[:, None] * stride_Ci
        + offs_n[None, :] * stride_Cn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor_or_None, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # typically 128
    # ------------------------------------------------------------------
    # Load model weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left_gate_weight  = weights["left_gate.weight"]   # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]  # (hidden_dim, dim)
    out_gate_weight   = weights["out_gate.weight"]    # (hidden_dim, dim)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (hidden_dim,)
    to_out_norm_bias   = weights["to_out_norm.bias"]    # (hidden_dim,)

    to_out_weight = weights["to_out.weight"]            # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1. LayerNorm over dim
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (dim → hidden_dim)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask: [B, N, N] -> [B, N, N, 1] -> broadcast
        mask_f = mask.to(dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoids)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Prepare tensors for the batched GEMM
    #    left  : (B, hidden, N, N)   (rows = i, cols = k)
    #    right : (B, hidden, N, N)   (rows = j, cols = k)
    #    The wanted product is left @ rightᵀ, i.e. sum over k.
    #    We achieve this by transposing the two N axes of `right`
    #    and then performing a standard batched matmul.
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape  # batch, seq_len, seq_len, dim

    # Permute to bring the hidden channel to the second axis.
    left_perm = left.permute(0, 3, 1, 2).contiguous()            # (B, H, N, N)
    # Transpose the two N dimensions of right before flattening.
    right_perm = right.permute(0, 3, 2, 1).contiguous()          # (B, H, N, N) with swapped axes

    # Collapse (B * H) into a single batch dimension for the kernel.
    total_B = B * hidden_dim
    left_flat = left_perm.view(total_B, N, N)    # (B*H, N, N)
    right_flat = right_perm.view(total_B, N, N)  # (B*H, N, N)

    # Cast to FP16 for the heavy GEMM (memory‑bandwidth friendly); accumulator stays FP32.
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32)  # final accumulator

    # ------------------------------------------------------------------
    # 6. Launch the Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        total_B,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_matmul_kernel[grid](
        # Pointers
        left_fp16,
        right_fp16,
        out_fp32,
        # Problem sizes
        total_B,
        N,
        N,
        N,
        # Strides (in elements, not bytes)
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        out_fp32.stride(0),
        out_fp32.stride(1),
        out_fp32.stride(2),
        # Compile‑time tile sizes
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, hidden_dim, N, N] and then to [B, N, N, hidden_dim]
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 7. Post‑processing: layer‑norm, out‑gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # (dim, hidden_dim) → back to [B, N, N, dim]

    return out
