@triton.jit
def trilmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_i = tl.program_id(0)  # i block
    pid_j = tl.program_id(1)  # j block
    pid_z = tl.program_id(2)  # combined batch*head

    # compute batch and head indices
    b = pid_z // H
    h = pid_z % H

    # compute offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute within batch
    # mask out-of-range
    mask_i = i_offset < N
    mask_j = j_offset < N

    # offsets for i and j in 2D
    # Prepare accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # left: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            i_offset[:, None] * stride_left_i + \
            k_offset[None, :] * stride_left_k + \
            h * stride_left_h

        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right: shape (BLOCK_K, BLOCK_N) because we want k as first dim
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            k_offset[:, None] * stride_right_k + \
            j_offset[None, :] * stride_right_i + \
            h * stride_right_h

        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot product accumulation
        acc += tl.dot(left, right)  # left (M,K), right (K,N) => (M,N)

    # store result
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        i_offset[:, None] * stride_out_i + \
        j_offset[None, :] * stride_out_j + \
        h * stride_out_h

    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) implementation with a custom Triton kernel.

Algorithm
---------
1️⃣ Layer‑norm the input tensor `x` over its last dimension (dim).
2️⃣ Linear projections:
   * left  = x @ W_left   (shape [B, N, N, H])
   * right = x @ W_right  (shape [B, N, N, H])
3️⃣ Optional binary mask (shape [B, N, N]) is broadcast and applied
   to *left* and *right*.
4️⃣ Gating:
   left_gate  = sigmoid(x @ W_left_gate)
   right_gate = sigmoid(x @ W_right_gate)
   out_gate   = sigmoid(x @ W_out_gate)
   left  *= left_gate
   right *= right_gate
5️⃣ **Tri‑mul** – the cubic core:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batch of H independent matrix multiplications
   (N×N) @ (N×N)ᵀ.  The heavy part is executed by `trilmul_kernel`,
   a fused Triton GEMM that tiles the i‑, j‑ and k‑dimensions.
   The kernel works on fp16 data (left/right) and accumulates in fp32.
6️⃣ Layer‑norm over the hidden dimension H, multiply by `out_gate`,
   and a final linear projection back to `dim`.

The Triton kernel is deliberately simple:
* program_id(0/1) tiles the output i/j plane,
* program_id(2) enumerates the (batch, hidden‑channel) pair,
* a loop over k‑tiles performs `tl.dot` accumulation.
All out‑of‑bounds loads/stores are masked.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------
# Triton kernel: out = Σ_k left[..., i, k, h] * right[..., j, k, h]
# ---------------------------------------------------------
@triton.jit
def trilmul_kernel(
    left_ptr,         # *fp16   pointer to left  [B,N,N,H]
    right_ptr,        # *fp16   pointer to right [B,N,N,H]
    out_ptr,          # *fp32   pointer to out   [B,N,N,H]

    B, N, H,          # tensor dimensions

    # strides for left
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    # strides for out
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # -----------------------------------------------------------------
    # Identify the work‑item (i‑tile, j‑tile, (batch,head) pair)
    # -----------------------------------------------------------------
    pid_i = tl.program_id(0)          # i‑tile index
    pid_j = tl.program_id(1)          # j‑tile index
    pid_z = tl.program_id(2)          # combined batch * hidden‑channel

    # decode batch index b and hidden channel h from pid_z
    b = pid_z // H
    h = pid_z % H

    # offsets inside the i / j dimensions
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for boundary handling
    mask_i = i_offset < N
    mask_j = j_offset < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over k‑tiles
    # -----------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # -----------------------------------------------------------------
        # Load a block of left  : shape (BLOCK_M, BLOCK_K)
        # -----------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + i_offset[:, None] * stride_left_i
            + k_offset[None, :] * stride_left_k
            + h * stride_left_h
        )
        left_block = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # -----------------------------------------------------------------
        # Load a block of right : shape (BLOCK_K, BLOCK_N)
        # (k dimension is most‑significant so we can dot directly)
        # -----------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + k_offset[:, None] * stride_right_k
            + j_offset[None, :] * stride_right_i
            + h * stride_right_h
        )
        right_block = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # -----------------------------------------------------------------
        # Accumulate dot product: (M,K) · (K,N) -> (M,N)
        # -----------------------------------------------------------------
        acc += tl.dot(left_block, right_block)   # fp16·fp16 → fp32

    # -----------------------------------------------------------------
    # Write the result tile
    # -----------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + i_offset[:, None] * stride_out_i
        + j_offset[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ---------------------------------------------------------
# Python entry‑point used by the evaluation harness
# ---------------------------------------------------------

    """
    Forward pass of the outgoing TriMul operation.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]
        - mask         : torch.Tensor[B, N, N] or None
        - weights_dict : mapping from the layer’s parameter names to tensors
        - config_dict  : contains "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor[B, N, N, dim]  – the processed tensor.
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # H in the notation above

    # -----------------------------------------------------------------
    # Retrieve weights (all are stored as torch tensors on the correct device)
    # -----------------------------------------------------------------
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1️⃣ Layer‑norm over the last dimension (dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # shape [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections (dim → hidden_dim)
    # -----------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    # -----------------------------------------------------------------
    # 3️⃣ Optional binary mask (broadcasted on the hidden dimension)
    # -----------------------------------------------------------------
    if mask is not None:
        # ensure mask is float (0/1) and broadcastable
        mask_f = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of a linear projection of x)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Triton‑accelerated cubic core
    # -----------------------------------------------------------------
    # Cast to fp16 for reduced memory traffic – the kernel will
    # accumulate in fp32 for numerical stability.
    left_f16  = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()

    B, N, _, _ = input_tensor.shape

    # Allocate output buffer (fp32 because later layers expect fp32)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=input_tensor.device)

    # Strides (in elements, not bytes) – required by the Triton kernel
    s_left_b, s_left_i, s_left_k, s_left_h = left_f16.stride()
    s_right_b, s_right_i, s_right_k, s_right_h = right_f16.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = out.stride()

    # Blocking parameters – chosen to give good occupancy on H100.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (i‑blocks, j‑blocks, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    trilmul_kernel[grid](
        left_f16,
        right_f16,
        out,
        B,
        N,
        hidden_dim,
        s_left_b, s_left_i, s_left_k, s_left_h,
        s_right_b, s_right_i, s_right_k, s_right_h,
        s_out_b, s_out_i, s_out_j, s_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Post‑processing: hidden‑dim layer‑norm, gating and final linear
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate                     # broadcast over H
    out = F.linear(out, to_out_weight)       # back to `dim`

    return out
