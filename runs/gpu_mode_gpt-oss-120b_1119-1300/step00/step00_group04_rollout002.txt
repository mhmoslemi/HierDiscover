---------
1. **Layer‑Norm** on the input tensor  ``x``  over the last dimension ``dim``.
2. **Linear projections** (no bias) into a hidden space of size ``hidden_dim``:
   * left_proj, right_proj
   * left_gate, right_gate, out_gate (later passed through a sigmoid)
3. **Fused gating** – a tiny Triton kernel computes
   ``left = left_proj * sigmoid(left_gate)`` and
   ``right = right_proj * sigmoid(right_gate)`` in FP16.
   If a mask is supplied it is applied afterwards (mask is 0/1, so order does
   not matter).
4. **Batched matrix multiplication** (the expensive “EinSum”):
   ``out[i,j] = Σ_k left[i,k]·right[j,k]``.
   By reshaping we turn the 4‑D contraction into a batch of regular
   matrix‑multiplications ``(B*hidden_dim, N, N)`` that run in half‑precision
   on the H100.
5. **Output Layer‑Norm** over the hidden dimension, multiplied by the
   sigmoid of ``out_gate``.
6. **Final linear projection** back to the original ``dim``.

Only the gating step is written in Triton (to satisfy the “use Triton” rule);
all heavy linear algebra uses the highly‑optimised cuBLAS kernels via
``torch.nn.functional.linear`` / ``torch.bmm``.  The whole pipeline works in
FP16 for the inner product while the surrounding normalisations remain in
FP32, giving a fast yet numerically stable implementation.

The function signature matches the problem statement:
``custom_kernel(data) → Tensor`` where ``data`` is a tuple
(input, mask, weights, config).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    total_elements,
    BLOCK_SIZE: tl.constexpr
):
    """Fuse linear projections with sigmoid gates (FP16).

    out_left  = left  * sigmoid(left_gate)
    out_right = right * sigmoid(right_gate)
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # mask out-of‑range elements
    mask = offs < total_elements

    # load values (dtype inferred from the pointers – here FP16)
    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    # sigmoid + elementwise multiplication
    out_left  = left  * tl.sigmoid(left_gate)
    out_right = right * tl.sigmoid(right_gate)

    # store results
    tl.store(out_left_ptr  + offs, out_left,  mask=mask)
    tl.store(out_right_ptr + offs, out_right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # Unpack inputs -----------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    # Config ---------------------------------------------------------------
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps        = 1e-5
    nomask     = config.get("nomask", True)

    # ----------------------------------------------------------------------
    # 1. Input LayerNorm ----------------------------------------------------
    #   Equivalent to nn.LayerNorm(dim, eps=eps, elementwise_affine=True)
    # ----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                     # shape [B, N, N, dim] – float32

    # ----------------------------------------------------------------------
    # 2. Half‑precision linear projections (no bias) ------------------------
    # ----------------------------------------------------------------------
    # Cast everything to FP16 for the heavy linear algebra
    x_half = x.half()

    # Helper to cast weights once
    def _to_fp16(t):
        return t.to(torch.float16)

    left_proj_weight   = _to_fp16(weights["left_proj.weight"])
    right_proj_weight  = _to_fp16(weights["right_proj.weight"])
    left_gate_weight   = _to_fp16(weights["left_gate.weight"])
    right_gate_weight  = _to_fp16(weights["right_gate.weight"])
    out_gate_weight    = _to_fp16(weights["out_gate.weight"])
    to_out_norm_weight = weights["to_out_norm.weight"]   # stays FP32
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # Linear maps – all produce tensors of shape [B, N, N, hidden_dim] (FP16)
    left_proj   = F.linear(x_half, left_proj_weight)    # (out_features, in_features)
    right_proj  = F.linear(x_half, right_proj_weight)
    left_gate   = F.linear(x_half, left_gate_weight)
    right_gate  = F.linear(x_half, right_gate_weight)
    out_gate_pre= F.linear(x_half, out_gate_weight)    # will be used later (FP16)

    # ----------------------------------------------------------------------
    # 3. Fuse projection + sigmoid gate via Triton -------------------------
    # ----------------------------------------------------------------------
    total_elements = left_proj.numel()
    BLOCK_SIZE = 32768  # ~32k elements per program; fits comfortably in registers
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    # Allocate output buffers
    left_fused  = torch.empty_like(left_proj)   # FP16
    right_fused = torch.empty_like(right_proj)  # FP16

    _gate_fuse_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        left_fused,
        right_fused,
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # ----------------------------------------------------------------------
    # 4. Optional mask (mask is 0/1, broadcast over hidden_dim) ------------
    # ----------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_half = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_fused  = left_fused  * mask_half
        right_fused = right_fused * mask_half
    # (if nomask==True we simply skip this step – the tensor stays unchanged)

    # ----------------------------------------------------------------------
    # 5. Batch matrix multiplication – the core “einsum” --------------------
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ----------------------------------------------------------------------
    B, N, _, _ = left_fused.shape
    # reshape to (B*hidden_dim, N, N) for a batched GEMM
    left_mat  = left_fused.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right_fused.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # half‑precision batched matrix multiply (uses TensorCores on H100)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*HD, N, N]

    # reshape back to [B, N, N, hidden_dim] (still FP16)
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ----------------------------------------------------------------------
    # 6. Output LayerNorm over hidden_dim (float32) ------------------------
    # ----------------------------------------------------------------------
    out_fp32 = out_hidden.to(torch.float32)   # cast back to FP32 for stable LN
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )   # shape [B, N, N, hidden_dim]

    # ----------------------------------------------------------------------
    # 7. Apply out_gate (sigmoid) -------------------------------------------
    # ----------------------------------------------------------------------
    # out_gate_pre was computed in FP16; convert to float32 and apply sigmoid
    out_gate = out_gate_pre.to(torch.float32).sigmoid()   # [B,N,N,hidden_dim]
    out_norm = out_norm * out_gate

    # ----------------------------------------------------------------------
    # 8. Final linear projection back to original dim -----------------------
    # ----------------------------------------------------------------------
    # to_out_weight shape (dim, hidden_dim)
    output = F.linear(out_norm, to_out_weight)  # result: [B, N, N, dim] (FP32)

    return output
