```python
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, D,
    stride_left_i, stride_left_k, stride_left_d,
    stride_right_j, stride_right_k, stride_right_d,
    stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_d = tl.program_id(2)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # accumulators
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left: shape (M, K, D)
        left_ptrs = left_ptr + (offs_i[:, None, None] * stride_left_i) + \
                   (offs_k[None, :, None] * stride_left_k) + \
                   (offs_d[None, None, :] * stride_left_d)
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0).to(tl.float32)

        # right: we want shape (K, N, D)
        right_ptrs = right_ptr + (offs_k[:, None, None] * stride_right_k) + \
                    (offs_j[None, :, None] * stride_right_j) + \
                    (offs_d[None, None, :] * stride_right_d)
        right = tl.load(right_ptrs, mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :], other=0.0).to(tl.float32)

        # compute dot for each d slice
        for di in range(0, BLOCK_D):
            # left_slice: (M, K)
            l = left[:, :, di]
            # right_slice: (K, N)
            r = right[:, :, di]
            acc[:, :, di] += tl.dot(l, r)  # MxN accumulation
    # store result
    out_ptrs = out_ptr + (offs_i[:, None, None] * stride_out_i) + \
               (offs_j[None, :, None] * stride_out_j) + \
               (offs_d[None, None, :] * stride_out_d)
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])
```

"""
TriMul (outgoing) implementation with a fused Triton kernel.

The heavy part of the TriMul forward pass is the 4‑D tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication for every feature
dimension `d`.  The kernel tiles the `i`, `j` and `d` axes (BLOCK_M,
BLOCK_N, BLOCK_D) and reduces over the `k` dimension in blocks (BLOCK_K).
All arithmetic inside the reduction is performed in float16 and the
partial sums are accumulated in float32 to retain numeric stability while
exploiting H100 tensor‑cores.  The rest of the TriMul pipeline (LayerNorm,
Linear projections, sigmoid gates, final LayerNorm and output projection)
remains in PyTorch for simplicity.

The kernel is launched once per batch (B∈{1,2}) with a 3‑D grid:
  grid_x = ceil(N / BLOCK_M)
  grid_y = ceil(N / BLOCK_N)
  grid_z = ceil(hidden_dim / BLOCK_D)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the core contraction: out = Σ_k left[...,k,:] * right[...,k,:]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,            # *f16 left  [i, k, d]
    right_ptr,           # *f16 right [k, j, d]
    out_ptr,             # *f32 out   [i, j, d]
    N, D,                # problem sizes
    stride_left_i, stride_left_k, stride_left_d,
    stride_right_j, stride_right_k, stride_right_d,
    stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    # --------------------------------------------------------------
    # program IDs map to a block of (i, j, d)
    # --------------------------------------------------------------
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_d = tl.program_id(2)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # accumulator for the sum over k
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # --------------------------------------------------------------
    # k‑loop: tile the reduction dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load left[i, k, d]   -> shape (M, K, D)
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + offs_i[:, None, None] * stride_left_i
            + offs_k[None, :, None] * stride_left_k
            + offs_d[None, None, :] * stride_left_d
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
            dtype=tl.float16,
        )  # (M, K, D) in fp16

        # ------------------------------------------------------------------
        # Load right[k, j, d]  -> shape (K, N, D) (note k is first dim)
        # ------------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + offs_k[:, None, None] * stride_right_k
            + offs_j[None, :, None] * stride_right_j
            + offs_d[None, None, :] * stride_right_d
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
            other=0.0,
            dtype=tl.float16,
        )  # (K, N, D) in fp16

        # ------------------------------------------------------------------
        # Accumulate: for each feature d do a small GEMM (M×K)·(K×N)
        # ------------------------------------------------------------------
        for di in range(0, BLOCK_D):
            # left_slice: (M, K), right_slice: (K, N)
            l = left[:, :, di]      # fp16
            r = right[:, :, di]     # fp16
            # tl.dot on fp16 uses tensor‑cores; accumulate in float32
            acc[:, :, di] += tl.dot(l, r, out_dtype=tl.float32)

    # --------------------------------------------------------------
    # Store the final result
    # --------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + offs_d[None, None, :] * stride_out_d
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )


# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [bs, N, N, dim]
        - mask         : torch.Tensor [bs, N, N]  (or None when config["nomask"] = True)
        - weights      : dict of torch.Tensors containing all model parameters
        - config       : dict with keys:
            * dim         : input feature dimension
            * hidden_dim : projection dimension
            * nomask      : bool, whether to ignore the mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Shortcut to the required weight tensors (all are stored without bias)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # (1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [bs, N, N, dim]

    # ------------------------------------------------------------------
    # (2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # [bs, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)  # [bs, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # (3) Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [bs, N, N]; expand to broadcast over hidden dim
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # (4) Gating values (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [bs, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # (5) Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # (6) Core contraction via Triton (fp16 for speed)
    # ------------------------------------------------------------------
    # Cast projected tensors to fp16 before the kernel – this lets the
    # dot products run on Tensor‑Cores while we still accumulate in fp32.
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # output tensor (float32) – will hold the result of the contraction
    out = torch.empty(bs, N, N, hidden_dim, dtype=torch.float32, device=input_tensor.device)

    # block sizes – chosen to fit comfortably in registers on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_D = 8   # must divide hidden_dim efficiently; remainder handled by masks

    # launch one kernel per batch to keep the kernel signature simple
    for b in range(bs):
        left_b = left[b]          # shape [N, N, hidden_dim]
        right_b = right[b]        # shape [N, N, hidden_dim]
        out_b = out[b]            # shape [N, N, hidden_dim]

        # strides (in elements) for each dimension
        stride_l_i, stride_l_k, stride_l_d = left_b.stride()
        stride_r_j, stride_r_k, stride_r_d = right_b.stride()
        stride_o_i, stride_o_j, stride_o_d = out_b.stride()

        grid = (
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
            (hidden_dim + BLOCK_D - 1) // BLOCK_D,
        )

        tri_mul_outgoing_kernel[grid](
            left_b,
            right_b,
            out_b,
            N,
            hidden_dim,
            stride_l_i,
            stride_l_k,
            stride_l_d,
            stride_r_j,
            stride_r_k,
            stride_r_d,
            stride_o_i,
            stride_o_j,
            stride_o_d,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )
    # ------------------------------------------------------------------
    # (7) Post‑contraction processing: LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate   # element‑wise gating after the second LayerNorm
    out = F.linear(out, to_out_w)  # project back to [bs, N, N, dim]

    return out
