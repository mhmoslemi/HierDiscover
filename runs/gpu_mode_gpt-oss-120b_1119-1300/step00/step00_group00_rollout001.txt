```python
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # compute batch and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # Compute the ranges
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Mask to avoid out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # Pointers for left: shape (M, K)
    left_ptrs = left_ptr + b * stride_left_b + h * stride_left_h + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k

    # Pointers for right: shape (K, N)
    # Note we need right[j,k] -> we want matrix multiplication left[i,k] * right_T[k,j]; we will load right as (K,N) where N dimension is j
    # However right's dimensions: (B, N, N, H) where second dim is j index, third is k.
    # To get right_T (K,N) we need to access right[b, j, k, h] but we want right as (K,N): k as rows, j as columns.
    # So we compute pointer using stride_right_k for rows, stride_right_j for columns.
    right_ptrs = right_ptr + b * stride_right_b + h * stride_right_h + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # Load left block
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # Load right block
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Compute dot product
        acc += tl.dot(left, right)  # left: (M,K), right: (K,N)

        # Advance pointers
        left_ptrs += BLOCK_K * stride_left_k
        right_ptrs += BLOCK_K * stride_right_k
    # Write out result
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])

"""
high‑performance TriMul (outgoing) implementation

The kernel computes the core “einsum” of the TriMul module:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
where `left` and `right` are the linearly projected and gated pairwise
representations of shape [B, N, N, H] (H = hidden_dim).  The computation is
a batched matrix multiplication for each hidden channel (B·H independent
matmuls of size N×N).  It is executed in half‑precision with Tensor‑Core
dot‑products (tl.dot) while accumulating in FP32 for numerical stability.
All other cheap operations (LayerNorm, linear projections, sigmoid gates,
masking, final LayerNorm and output linear) are performed with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_matmul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # problem size
    B, N, H,
    # strides for left  (B, i, k, h)
    stride_la_b, stride_la_i, stride_la_k, stride_la_h,
    # strides for right (B, j, k, h)
    stride_ra_b, stride_ra_j, stride_ra_k, stride_ra_h,
    # strides for out   (B, i, j, h)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute a batch of independent N×N matmuls (one per hidden channel)."""
    pid_bh = tl.program_id(2)        # combined batch × hidden dim index
    pid_i  = tl.program_id(0)        # block row  (i)
    pid_j  = tl.program_id(1)        # block col  (j)

    # decode batch and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # bounds masks
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # Pointers to the current tile of left (M×K) and right (K×N)
    left_tile_ptr  = (left_ptr  + b * stride_la_b
                                   + h * stride_la_h
                                   + offs_i[:, None] * stride_la_i
                                   + offs_k[None, :] * stride_la_k)
    right_tile_ptr = (right_ptr + b * stride_ra_b
                                   + h * stride_ra_h
                                   + offs_k[:, None] * stride_ra_k
                                   + offs_j[None, :] * stride_ra_j)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the K dimension
    for _ in range(0, tl.cdiv(N, BLOCK_K)):
        # load tiles, padding with zeros outside the matrix
        left_tile = tl.load(left_tile_ptr,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)
        right_tile = tl.load(right_tile_ptr,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)

        # FP16 dot → FP32 accumulator (uses Tensor‑Core WMMA)
        acc += tl.dot(left_tile, right_tile)

        # advance K‑pointers
        left_tile_ptr  += BLOCK_K * stride_la_k
        right_tile_ptr += BLOCK_K * stride_ra_k

    # Write the result tile back to global memory (float32)
    out_tile_ptr = (out_ptr + b * stride_out_b
                                 + h * stride_out_h
                                 + offs_i[:, None] * stride_out_i
                                 + offs_j[None, :] * stride_out_j)
    tl.store(out_tile_ptr,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3 style).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor[B, N, N, dim] (float32)
            - mask: torch.Tensor[B, N, N]   (bool/float) or None
            - weights: dict of model parameters (LayerNorm, Linear, etc.)
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor[B, N, N, dim] (float32) – the transformed pairwise features.
    """
    # ------------------------------------------------------------------ #
    # unpack arguments
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # Helper shortcuts for model weights
    # ------------------------------------------------------------------ #
    # LayerNorm on the input
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # Linear projections (no bias)
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    # Gating projections (no bias)
    left_gate_weight  = weights["left_gate.weight"]   # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]  # (hidden_dim, dim)
    out_gate_weight   = weights["out_gate.weight"]    # (hidden_dim, dim)

    # Output norm and final linear
    to_out_norm_weight = weights["to_out_norm.weight"]  # (hidden_dim,)
    to_out_norm_bias   = weights["to_out_norm.bias"]    # (hidden_dim,)
    to_out_weight      = weights["to_out.weight"]       # (dim, hidden_dim)

    # ------------------------------------------------------------------ #
    # 1️⃣  Input LayerNorm (over the last dimension)
    # ------------------------------------------------------------------ #
    eps = 1e-5
    # torch.nn.functional.layer_norm expects normalized_shape = (dim,)
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------ #
    # 2️⃣  Linear projections + gating
    # ------------------------------------------------------------------ #
    # shape after projection: [B, N, N, hidden_dim]
    left  = F.linear(x, left_proj_weight)   # (B,N,N,H)
    right = F.linear(x, right_proj_weight)  # (B,N,N,H)

    # gates (sigmoid) – same shape as left/right
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------ #
    # 3️⃣  Optional mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------ #
    if not nomask and mask is not None:
        # mask shape: [B, N, N] → [B, N, N, 1] for broadcasting
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------ #
    # 4️⃣  Apply gating
    # ------------------------------------------------------------------ #
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5️⃣  Compute the core N³ contraction with a Triton kernel
    # ------------------------------------------------------------------ #
    # Cast to fp16 for Tensor‑Core efficiency
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    B, N, _, H = left_fp16.shape
    assert H == hidden_dim, "hidden dimension mismatch"

    # Output buffer (FP32 accumulator)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left_fp16.device)

    # Strides (in elements) for each tensor – needed by Triton
    s_la_b, s_la_i, s_la_k, s_la_h = left_fp16.stride()
    s_ra_b, s_ra_j, s_ra_k, s_ra_h = right_fp16.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = out.stride()

    # Tile sizes – tuned for H100 Tensor‑Core (fp16) matmul
    BLOCK_M = 64   # rows of the output tile
    BLOCK_N = 64   # cols of the output tile
    BLOCK_K = 32   # reduction tile size (must be a multiple of 8 for WMMA)

    grid = (
        triton.cdiv(N, BLOCK_M),          # number of row tiles
        triton.cdiv(N, BLOCK_N),          # number of column tiles
        B * H                             # batch × hidden‑dim dimension
    )

    _trimul_matmul_kernel[grid](
        left_fp16, right_fp16, out,
        B, N, H,
        s_la_b, s_la_i, s_la_k, s_la_h,
        s_ra_b, s_ra_j, s_ra_k, s_ra_h,
        s_out_b, s_out_i, s_out_j, s_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8, num_stages=3
    )

    # ------------------------------------------------------------------ #
    # 6️⃣  Output LayerNorm (over hidden_dim) and final gating
    # ------------------------------------------------------------------ #
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    out = out * out_gate  # element‑wise gating (both are FP32)

    # ------------------------------------------------------------------ #
    # 7️⃣  Final linear projection back to `dim`
    # ------------------------------------------------------------------ #
    out = F.linear(out, to_out_weight)   # (B, N, N, dim)

    return out
```
