def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # forward pass
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # Projections
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])
    # Mask
    if config.get("nomask", False) or mask is None:
        mask_tensor = torch.ones(bs, seq_len, seq_len, 1, device=input_tensor.device, dtype=x.dtype)
    else:
        mask_tensor = mask.unsqueeze(-1).type_as(x)
    left_proj = left_proj * mask_tensor
    right_proj = right_proj * mask_tensor
    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))
    left_proj = left_proj * left_gate
    right_proj = right_proj * right_gate

    # TriMul: batched matmul over last two dims with sum over k
    # Permute to (bs, hidden_dim, N, N) for good memory layout
    left = left_proj.permute(0, 3, 1, 2).contiguous()
    right = right_proj.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    M = B * H
    left_flat = left.view(M, N, N)
    right_flat = right.view(M, N, N)

    out_flat = torch.empty_like(left_flat)

    # Compute strides
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bi = right_flat.stride(1)  # original row stride (i)
    stride_bj = right_flat.stride(2)  # original col stride (k)

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # Block sizes (tuned for H100 FP16 matmul)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for batched matrix multiplication (A @ B^T) where
# A and B have shape (M, N, K) and the result has shape (M, N, N).
# In the TriMul operator the inner dimension K == N.
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bi, stride_bj,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # 3‑D grid: (row block, col block, batch index)
    batch_idx = tl.program_id(2)
    pid_m = tl.program_id(0)   # row block
    pid_n = tl.program_id(1)   # col block

    # tile offsets inside the N‑by‑N output matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K (= N)
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + batch_idx * stride_am + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # B tile (transposed on‑the‑fly): (BLOCK_K, BLOCK_N)
        # original B has shape (M, N, K). By swapping the 2nd and 3rd axes we get B^T.
        b_ptrs = b_ptr + batch_idx * stride_bm + offs_k[:, None] * stride_bi + offs_n[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # dot‑product accumulation (fp16 * fp16 -> fp32)
        acc += tl.dot(a, b)

    # write the result back
    c_ptrs = c_ptr + batch_idx * stride_cm + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.
    The heavy O(N³·C_z) contraction ``out[i,j] = Σₖ left[i,k]·right[j,k]`` is
    performed by a custom Triton kernel (batched GEMM with a transposed operand).
    All remaining linear / layer‑norm / gating operations are executed with
    regular PyTorch kernels.
    """
    input_tensor, mask, weights, config = data

    # ------------------------------------------------------------------
    # unpack configuration and weights
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, _ = input_tensor.shape

    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if config.get("nomask", False) or mask is None:
        mask_broadcast = torch.ones(bs, N, N, 1, dtype=x.dtype, device=x.device)
    else:
        mask_broadcast = mask.unsqueeze(-1).to(x.dtype)   # [B,N,N,1]

    left = left * mask_broadcast
    right = right * mask_broadcast

    # 4️⃣ Gating (sigmoid of a linear projection of the *normalized* input)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core TriMul contraction – batched matmul A @ Bᵀ
    # ------------------------------------------------------------------
    # reshape to (B, hidden_dim, N, N) for better memory layout
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    M = B * H                                 # total number of independent GEMMs

    # flatten the leading two axes → (M, N, N)
    left_flat  = left.view(M, N, N)
    right_flat = right.view(M, N, N)

    # output buffer
    out_flat = torch.empty_like(left_flat)

    # stride information (in elements, not bytes)
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bi = right_flat.stride(1)   # original row stride (i) – becomes “k” stride after transpose
    stride_bj = right_flat.stride(2)   # original col stride (k) – becomes “j” stride after transpose

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        M                          # independent GEMM instances
    )

        left_flat,
        right_flat,
        out_flat,
        M, N, N,

        stride_bm, stride_bi, stride_bj,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

                       bias=to_out_norm_bias)

    out = out * out_gate                     # broadcast over hidden dimension

    out = F.linear(out, to_out_weight)       # final projection back to dim

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) where A and B are (M, N, K)
# The TriMul contraction is exactly this operation with K == N.
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes (runtime)
    stride_am, stride_ai, stride_ak,   # A strides: (batch, row, col)
    stride_bm, stride_bi, stride_bj,   # B strides: (batch, row, col) – we read B transposed
    stride_cm, stride_ci, stride_cj,   # C strides: (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matmul kernel computing C[b] = A[b] @ B[b].T.

    A, B : (M, N, K)   (float32 on memory)
    C    : (M, N, N)   (float32 on memory)

    The kernel works on FP16 tiles and accumulates into FP32.
    """
    batch_idx = tl.program_id(2)      # which (batch * hidden) slice
    pid_m = tl.program_id(0)          # tile row index
    pid_n = tl.program_id(1)          # tile col index

    # Tile offsets inside the N×N output matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K (which equals N)
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ---- A tile (M×K) -------------------------------------------------
        a_ptrs = a_ptr + batch_idx * stride_am \
                 + offs_m[:, None] * stride_ai \
                 + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # ---- B tile (transposed on‑the‑fly) (K×N) -------------------------
        # Original B has shape (M, N, K). Swapping axes 1/2 gives Bᵀ.
        b_ptrs = b_ptr + batch_idx * stride_bm \
                 + offs_k[:, None] * stride_bi \
                 + offs_n[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # ---- dot product accumulation --------------------------------------
        acc += tl.dot(a, b)   # FP16⋅FP16 → FP32

    # ---- Store result ----------------------------------------------------
    c_ptrs = c_ptr + batch_idx * stride_cm \
             + offs_m[:, None] * stride_ci \
             + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
