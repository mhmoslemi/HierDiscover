"""
TriMul (outgoing) implementation with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B,N,N,C]).
2. Compute left/right projections and three gating projections (all linear layers).
3. Apply mask (if present) and the sigmoid gates.
4. The core operation is:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication `left @ right^T` for every
   batch * hidden_dim slice.  This is performed by a fused Triton kernel
   `batched_matmul_kernel` that computes many GEMMs in parallel.
5. Apply a second LayerNorm, multiply by the output gate, and project back
   to the original feature dimension.
All heavy work (the O(N³) contraction) lives in the Triton kernel; the
remaining linear / layer‑norm / sigmoid steps are done with PyTorch for
simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B^T  (A,B : (batch, M, K))
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr,                # *Pointer* to matrix A   [batch, M, K]
    B_ptr,                # *Pointer* to matrix B   [batch, N, K]
    C_ptr,                # *Pointer* to output C   [batch, M, N]
    BATCH, M, N, K,       # dimensions
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # --------------------------------------------------------------
    # Program IDs (grid is (blocks_M, blocks_N, batch))
    # --------------------------------------------------------------
    pid_z = tl.program_id(2)               # batch index
    pid_m = tl.program_id(0)               # block row index
    pid_n = tl.program_id(1)               # block column index

    # --------------------------------------------------------------
    # Offsets for the tiles this program works on
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base offsets for this batch
    a_batch_offset = pid_z * stride_a_batch
    b_batch_offset = pid_z * stride_b_batch
    c_batch_offset = pid_z * stride_c_batch

    # ------------------------------------------------------------------
    # Accumulator
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_offset = k_tile * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # --------------------------------------------------------------
        # Load A tile (shape BLOCK_M x BLOCK_K)
        # --------------------------------------------------------------
        a_ptrs = (
            A_ptr
            + a_batch_offset
            + (offs_m[:, None] * stride_a_m)
            + (offs_k[None, :] * stride_a_k)
        )
        a_mask = mask_m[:, None] & mask_k[None, :]
        a = tl.load(a_ptrs, mask=a_mask, other=0.0, dtype=tl.float16)

        # --------------------------------------------------------------
        # Load B tile transposed (shape BLOCK_K x BLOCK_N)
        # We need B^T, i.e. B[n, k] -> B_T[k, n]
        # --------------------------------------------------------------
        b_ptrs = (
            B_ptr
            + b_batch_offset
            + (offs_k[:, None] * stride_b_k)   # k dimension (now rows)
            + (offs_n[None, :] * stride_b_n)   # n dimension (now columns)
        )
        b_mask = mask_k[:, None] & mask_n[None, :]
        b = tl.load(b_ptrs, mask=b_mask, other=0.0, dtype=tl.float16)

        # --------------------------------------------------------------
        # Dot product and accumulate
        # --------------------------------------------------------------
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # --------------------------------------------------------------
    # Store result
    # --------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + c_batch_offset
        + (offs_m[:, None] * stride_c_m)
        + (offs_n[None, :] * stride_c_n)
    )
    c_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=c_mask)


def batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ B^T for each batch using Triton.
    A, B : torch.Tensor of shape (batch, M, K) and (batch, N, K) (dtype=torch.float16)
    Returns C : torch.Tensor of shape (batch, M, N)  (dtype=torch.float32)
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    batch2, N, K2 = B.shape
    assert batch == batch2 and K == K2, "Incompatible shapes for batched_matmul"

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # strides
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_n, stride_b_k = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Block size configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid layout (blocks_M, blocks_N, batch)
    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    batched_matmul_kernel[grid](
        A,
        B,
        C,
        batch,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_n,
        stride_b_k,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )
    return C


# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensor containing all linear / norm params
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    output : torch.Tensor [B, N, N, C] (float32)
    """
    # Unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # LayerNorm over the feature dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # Linear projections
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # Apply mask if required
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating projections
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the batched GEMM
    #   left/right shape: [B, N, N, hidden_dim]
    #   We need shape (batch = B*hidden_dim, N, N) with half precision
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # (B, hidden_dim, N, N) -> (B*hidden_dim, N, N)
    left_t = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()
    # Free the original tensors early to keep memory pressure low
    del left, right, x

    left_flat = left_t.view(-1, N, N).to(torch.float16)
    right_flat = right_t.view(-1, N, N).to(torch.float16)

    # ------------------------------------------------------------------
    # Core contraction: out_flat = left_flat @ right_flat^T per batch item
    # ------------------------------------------------------------------
    out_flat = batched_matmul(left_flat, right_flat)  # float32, shape (B*hidden_dim, N, N)

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # Release temporary half‑precision buffers
    del left_flat, right_flat, left_t, right_t, out_flat

    # ------------------------------------------------------------------
    # Second LayerNorm, output gate and final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # Multiply by output gate (same shape)
    out = out * out_gate

    # Final linear projection back to original dim
    to_out_weight = weights["to_out.weight"]
    output = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return output
