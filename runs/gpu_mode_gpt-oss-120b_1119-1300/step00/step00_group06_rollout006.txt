def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on input: mirror nn.LayerNorm default epsilon and weight+bias shape.
    # Since we have weight and bias supplied.
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Projections:
    left = torch.nn.functional.linear(x, weights["left_proj.weight"]) # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask
    if mask_tensor is not None and mask_tensor.ndim > 0:  # mask may be absent
        mask = mask_tensor[..., None]  # broadcast to hidden_dim
        left = left * mask
        right = right * mask

    # Gates (sigmoid after linear)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Prepare for batched matmul (out = left @ right^T for each hidden dim)
    # Permute and reshape:
    B, N, _, _ = left.shape
    left_mat = left.permute(0, 3, 1, 2).contiguous()  # B, H, N, N
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # B, H, N, N (right transposed)

    batch_total = B * hidden_dim

    left_mat = left_mat.view(batch_total, N, N)
    right_mat = right_mat.view(batch_total, N, N)

    # Output placeholder
    out_mat = torch.empty_like(left_mat)  # same dtype and device

    # Triton kernel launch
    matmul_kernel(left_mat, right_mat, out_mat, left_mat.stride(0), left_mat.stride(1), left_mat.stride(2), 
                  right_mat.stride(0), right_mat.stride(1), right_mat.stride(2), 
                  out_mat.stride(0), out_mat.stride(1), out_mat.stride(2), 
                  M=N, N=N, K=N,
                  BLOCK_M=128, BLOCK_N=128, BLOCK_K=32, num_warps=4)

    # Reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # to_out_norm: layernorm over hidden_dim
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # out_gate scaling
    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

Now define the matmul kernel:

```
@triton.jit
def matmul_kernel(A, B, C,
                  stride_az, stride_am, stride_ak,
                  stride_bz, stride_bk, stride_bn,
                  stride_cz, stride_cm, stride_cn,
                  M, N, K,
                  BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_z = tl.program_id(0)  # batch dimension
    pid_m = tl.program_id(1)  # block row
    pid_n = tl.program_id(2)  # block col

    # compute offsets
    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    off_k = tl.arange(0, BLOCK_K)

    # masks
    mask_m = off_m < M
    mask_n = off_n < N

    # pointers start
    a_ptr = A + pid_z * stride_az + off_m[:, None] * stride_am + off_k[None, :] * stride_ak
    b_ptr = B + pid_z * stride_bz + off_k[:, None] * stride_bk + off_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute current k indices for this tile
        k_offset = k * BLOCK_K
        k_ind = k_offset + off_k
        cur_mask_k = k_ind < K

        # load tiles
        a = tl.load(a_ptr, mask=mask_m[:, None] & cur_mask_k[None, :], other=0.0)
        b = tl.load(b_ptr, mask=cur_mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

        # increment pointers
        a_ptr += BLOCK_K * stride_ak
        b_ptr += BLOCK_K * stride_bk

    # store
    c_ptr = C + pid_z * stride_cz + off_m[:, None] * stride_cm + off_n[None, :] * stride_cn
    tl.store(c_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

```
@triton.jit
def batched_matmul_kernel(A, B, C,
                          stride_az, stride_am, stride_ak,
                          stride_bz, stride_bk, stride_bn,
                          stride_cz, stride_cm, stride_cn,
                          M, N, K,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
                          NUM_WARPS: tl.constexpr):
    pid = tl.program_id(0)
    batch = tl.program_id(1)

    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # compute actual block start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for this batch
    a_ptrs = A + batch * stride_az + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = B + batch * stride_bz + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k offset
        k_offset = k * BLOCK_K

def custom_kernel(data):
   """
   Triton-accelerated TriMul (outgoing) forward pass.
   Performs the heavy N^3 hidden_dim reduction via a custom batched GEMM kernel.
   Steps:
   1. LayerNorm over input dim.
   2. Linear projections and gates.
   3. Apply mask (if present) and gating.
   4. Compute out = left @ right^T for each hidden channel via batched matmul.
   5. LayerNorm on hidden dimension, multiply by out_gate, final linear projection.
   Returns tensor [B, N, N, dim] (float32).
   """
   import torch
   import torch.nn.functional as F
   import triton
   import triton.language as tl

   # unpack
   input_tensor, mask_tensor, weights, config = data
   dim = config["dim"]
   hidden_dim = config["hidden_dim"]
   eps = 1e-5

   # 1. Input layernorm
   x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

   # 2. Projections
   left = F.linear(x, weights["left_proj.weight"])
   right = F.linear(x, weights["right_proj.weight"])

   # 3. Optional mask
   nomask = config.get("nomask", False)
   if not nomask and mask_tensor is not None:
       mask_exp = mask_tensor[..., None]  # shape [B,N,N,1]
       left = left * mask_exp
       right = right * mask_exp

   # 4. Gates
   left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
   right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
   out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

   left = left * left_gate
   right = right * right_gate

   # 5. Batched matmul (left @ right^T for each hidden channel)
   B, N, _, _ = left.shape

   # reshape to [B, H, N, N] then flatten batch*H
   left_perm = left.permute(0, 3, 1, 2).contiguous()   # B, H, N, N
   right_perm = right.permute(0, 3, 2, 1).contiguous() # B, H, N, N (k dimension transposed)

   batch_total = B * hidden_dim
   left_flat = left_perm.view(batch_total, N, N)
   right_flat = right_perm.view(batch_total, N, N)

   out_flat = torch.empty_like(left_flat)

   # Triton kernel
   @triton.jit
   def batched_matmul_kernel(
       A, B, C,
       stride_az, stride_am, stride_ak,
       stride_bz, stride_bk, stride_bn,
       stride_cz, stride_cm, stride_cn,
       M, N, K,
       BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
   ):
       pid = tl.program_id(0)          # index of block (i,j)
       batch = tl.program_id(1)        # which matrix in the batch

       num_pid_n = tl.cdiv(N, BLOCK_N)

       pid_m = pid // num_pid_n
       pid_n = pid % num_pid_n

       # offsets within the matrix
       offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
       offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

       mask_m = offs_m < M
       mask_n = offs_n < N

       # base pointers for this batch
       a_ptr_base = A + batch * stride_az + offs_m[:, None] * stride_am
       b_ptr_base = B + batch * stride_bz + offs_n[None, :] * stride_bn

       acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

       # iterate over k dimension
       for k in range(0, tl.cdiv(K, BLOCK_K)):
           cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
           mask_k = cur_k < K

           a_ptr = a_ptr_base + cur_k[None, :] * stride_ak
           b_ptr = b_ptr_base + cur_k[:, None] * stride_bk

           a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
           b = tl.load(b_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

           acc += tl.dot(a, b)

       c_ptr = C + batch * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
       tl.store(c_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

   # launch configuration
   BLOCK_M = 128
   BLOCK_N = 128
   BLOCK_K = 32

   num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
   num_pid_n = (N + BLOCK_N - 1) // BLOCK_N

   grid = (num_pid_m * num_pid_n, batch_total)

   batched_matmul_kernel[grid](
       left_flat, right_flat, out_flat,
       left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
       right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
       out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
       N, N, N,
       BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
       num_warps=4
   )

   # reshape back to [B, N, N, H]
   out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

   # 6. to_out_norm + out_gate
   out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)
   out = out * out_gate

   # 7. final projection
   out = F.linear(out, weights["to_out.weight"])

   return out

"""
custom_kernel – Triton‑accelerated “outgoing” TriMul (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input tensor (dim‑last).
2. Linear projections (left/right) and three sigmoid gates.
3. Apply optional mask and the left/right gates.
4. The core O(N³·H) operation is
       out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
   which can be expressed as a batched matrix multiplication
       out_d = left_d @ right_dᵀ
   for each hidden channel d.
   A custom Triton kernel performs the batched GEMM for all
   B·H matrices (B ≤ 2, H = hidden_dim ≤ 128) in a single launch.
5. Layer‑norm over the hidden dimension, scale by out_gate, and a final
   linear projection back to the original feature dimension.

Only the heavy N³ reduction (step 4) is implemented in Triton; the
surrounding linear/LayerNorm/gate work uses PyTorch (highly optimized
Torch‑CUDA kernels). The result is a float32 tensor of shape
[batch, seq_len, seq_len, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A, B, C,
    stride_az, stride_am, stride_ak,      # A: [Z, M, K]
    stride_bz, stride_bk, stride_bn,      # B: [Z, K, N]
    stride_cz, stride_cm, stride_cn,      # C: [Z, M, N]
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: for each batch `z` compute C[z] = A[z] @ B[z].
    A  – [Z, M, K]  (float32)
    B  – [Z, K, N]  (float32)
    C  – [Z, M, N]  (float32)
    """
    pid = tl.program_id(0)          # linear id over M‑blocks × N‑blocks
    batch = tl.program_id(1)        # which matrix in the batch

    # number of blocks along the N dimension
    num_pid_n = tl.cdiv(N, BLOCK_N)

    # de‑compose pid into (block row, block col)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch (M‑axis for A, N‑axis for B)
    a_ptr_base = A + batch * stride_az + offs_m[:, None] * stride_am
    b_ptr_base = B + batch * stride_bz + offs_n[None, :] * stride_bn

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # pointers to the current K‑tile
        a_ptr = a_ptr_base + cur_k[None, :] * stride_ak
        b_ptr = b_ptr_base + cur_k[:, None] * stride_bk

        # load tiles (masked to avoid out‑of‑bounds)
        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # matrix multiply‑accumulate
        acc += tl.dot(a, b)

    # write back the result
    c_ptr = C + batch * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ----------------------------------------------------------------------
    # unpack inputs
    # ----------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5                       # same default as nn.LayerNorm

    # ----------------------------------------------------------------------
    # 1️⃣ Input layer‑norm
    # ----------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ----------------------------------------------------------------------
    # 2️⃣ Linear projections
    # ----------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # ----------------------------------------------------------------------
    # 3️⃣ Optional mask (mask shape: [B, N, N])
    # ----------------------------------------------------------------------
    if not config.get("nomask", False):
        # mask may be bool or float; ensure broadcastable
        mask = mask_tensor[..., None].to(left.dtype)    # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ----------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid after linear)
    # ----------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 5️⃣ Core N³ reduction via batched GEMM (left @ rightᵀ per hidden channel)
    # ----------------------------------------------------------------------
    B, N, _, _ = left.shape

    # bring hidden dim to the batch axis and make the matrices contiguous
    # left: [B, N, N, H] → [B, H, N, N] → view as [B*H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # transpose K‑axis

    batch_total = B * hidden_dim
    left_flat = left_perm.view(batch_total, N, N)
    right_flat = right_perm.view(batch_total, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # launch Triton kernel --------------------------------------------------
    BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_pid_m * num_pid_n, batch_total)   # (blocks over M×N, batch)

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        # strides for A (left)
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # strides for B (right) – note shape is [Z, K, N]
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        # strides for C (out)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        N, N, N,                                   # M, N, K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # ----------------------------------------------------------------------
    # reshape back to [B, N, N, H]
    # ----------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------------------
    # 6️⃣ Hidden‑dim LayerNorm + out_gate scaling
    # ----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ----------------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # ----------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
