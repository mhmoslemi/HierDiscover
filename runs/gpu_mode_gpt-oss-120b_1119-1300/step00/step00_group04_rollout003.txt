    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_bn, stride_left_bh, stride_left_n, stride_left_h,
    stride_right_bn, stride_right_bh, stride_right_n, stride_right_h,
    stride_out_bn, stride_out_bh, stride_out_n, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # b*h index = program id
    pid = tl.program_id(0)
    total = B * H
    b = pid // H
    h = pid % H

    # offsets for block start
    offs_m = tl.arange(0, BLOCK_M)
    offs_n = tl.arange(0, BLOCK_N)
    # compute starting idx in left and right.
    # left matrix dimension: rows: N (i), cols: N (k)
    # left_ptr offset for element (i,k): b*stride_left_bn + h*stride_left_bh + i*stride_left_n + k*stride_left_h

    left_base = left_ptr + b*stride_left_bn + h*stride_left_bh

    # Use accumulator

def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_l_bs, stride_l_h, stride_l_i, stride_l_k,
    stride_r_bs, stride_r_h, stride_r_i, stride_r_k,
    stride_o_bs, stride_o_h, stride_o_i, stride_o_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    pid_batch_channel = tl.program_id(0) # combined b_h
    pid_m = tl.program_id(1) # block row i
    pid_n = tl.program_id(2) # block col j

    # decode b and h
    b = pid_batch_channel // H
    h = pid_batch_channel % H

    # offsets
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # compute ranges
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # Mask for within bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers
    left_base = left_ptr + b * stride_l_bs + h * stride_l_h
    right_base = right_ptr + b * stride_r_bs + h * stride_r_h
    out_base = out_ptr + b * stride_o_bs + h * stride_o_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left block A [M, K]
        a_ptrs = left_base + (offs_i[:, None] * stride_l_i) + (offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right block B^T [K, N] where B^T[k, j] = right[j, k]
        # original right indices: (j, k)
        b_ptrs = right_base + (offs_j[None, :] * stride_r_i) + (offs_k[:, None] * stride_r_k)

        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)  # a: MxK, b: KxN -> MxN

    # write back
    out_ptrs = out_base + (offs_i[:, None] * stride_o_i) + (offs_j[None, :] * stride_o_k)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_i, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bc = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    b = pid_bc // H
    h = pid_bc % H

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # Offsets
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    # In-bounds mask
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointer increments
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A: left[i,k]
        a_ptrs = left_base + (offs_i[:, None] * stride_l_i) + (offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B: right_T[k,j] = right[j,k]
        b_ptrs = right_base + (offs_k[:, None] * stride_r_k) + (offs_j[None, :] * stride_r_i)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Store result
    out_ptrs = out_base + (offs_i[:, None] * stride_o_i) + (offs_j[None, :] * stride_o_k)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) implementation with a custom Triton kernel.

The heavy O(N³) computation
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
is equivalent to a batched matrix multiplication
    left[b, :, :, h] @ right[b, :, :, h].T
for each batch b and hidden channel h.
A Triton kernel fuses the load‑compute‑store of this matmul and
provides a large speed‑up over a pure‑Python implementation.
All other operations (LayerNorms, Linear projections and gating)
are performed with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_i, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out = left @ rightᵀ per (batch, hidden) pair."""
    # -------------------------------------------------------------
    # program IDs
    pid_bc = tl.program_id(0)          # combined batch * hidden index
    pid_m  = tl.program_id(1)          # tile row   (i)
    pid_n  = tl.program_id(2)          # tile col   (j)

    # decode batch and hidden dim
    b = pid_bc // H
    h = pid_bc % H

    # starting offsets for the current tile
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # offsets within the tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    # bounds checking
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # accumulator in FP32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction (k) dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- left block (M x K) -------------------------------------------------
        a_ptrs = left_base + (offs_i[:, None] * stride_l_i) + (offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)               # FP32 because left is FP32

        # ---- rightᵀ block (K x N) -----------------------------------------------
        # rightᵀ[k, j] = right[j, k]
        b_ptrs = right_base + (offs_k[:, None] * stride_r_k) + (offs_j[None, :] * stride_r_i)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # dot-product accumulate
        acc += tl.dot(a, b)               # a: (M,K), b: (K,N) → (M,N)

    # write back the result
    out_ptrs = out_base + (offs_i[:, None] * stride_o_i) + (offs_j[None, :] * stride_o_k)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[:, None])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, D_in]
        - mask         : torch.Tensor of shape [B, N, N] (or None)
        - weights      : dict of model parameters
        - config       : dict containing at least ``dim`` and ``hidden_dim`` and
                         ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D_in] (float32, on the same device as inputs).
    """
    # -------------------------------------------------------------------------
    # unpack arguments
    # -------------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # shapes / config
    B, N, _, D_in = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # usually 128
    dim = config["dim"]                        # D_in (should match)

    # -------------------------------------------------------------------------
    # Helper: LayerNorm (no bias in the reference is handled by weight/bias)
    # -------------------------------------------------------------------------
    def _layer_norm(x, weight, bias, normalized_shape):
        return torch.nn.functional.layer_norm(
            x, normalized_shape=normalized_shape,
            weight=weight, bias=bias, eps=1e-5
        )

    # -------------------------------------------------------------------------
    # 1. Input LayerNorm
    # -------------------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        normalized_shape=(dim,)
    )   # [B, N, N, dim]

    # -------------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------------------------------
    # 3. Apply mask if present
    # -------------------------------------------------------------------------
    use_mask = (mask is not None) and (not config.get("nomask", False))
    if use_mask:
        # mask broadcasted on feature dim
        mask_exp = mask.to(dtype=dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left  * mask_exp
        right = right * mask_exp

    # -------------------------------------------------------------------------
    # 4. Gating (sigmoid on linear projections of the normalized input)
    # -------------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left  * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------------
    # 5. Core TriMul matmul using Triton
    #    Compute out[h] = left[h] @ right[h].T for each batch / hidden channel.
    # -------------------------------------------------------------------------
    # Rearrange to (B, H, N, N) for convenient striding
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    out_perm = torch.empty_like(left_perm, dtype=left_perm.dtype)   # same shape

    # Extract strides (they are in element units, acceptable for Triton)
    stride_l_b, stride_l_h, stride_l_i, stride_l_k = left_perm.stride()
    stride_r_b, stride_r_h, stride_r_i, stride_r_k = right_perm.stride()
    stride_o_b, stride_o_h, stride_o_i, stride_o_k = out_perm.stride()

    # Triton block sizes – tuned for H100 (FP32 path)
    BLOCK_M = 64   # rows of the output tile
    BLOCK_N = 64   # cols of the output tile
    BLOCK_K = 64   # reduction tile size

    # Grid: (batch*hidden, num_tiles_m, num_tiles_n)
    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_matmul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B, hidden_dim, N,
        stride_l_b, stride_l_h, stride_l_i, stride_l_k,
        stride_r_b, stride_r_h, stride_r_i, stride_r_k,
        stride_o_b, stride_o_h, stride_o_i, stride_o_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Back to the original layout [B, N, N, H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------------------------------
    # 6. Output LayerNorm + gating
    # -------------------------------------------------------------------------
    out = _layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        normalized_shape=(hidden_dim,)
    )
    out = out * out_gate

    # -------------------------------------------------------------------------
    # 7. Final linear projection back to dim
    # -------------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
