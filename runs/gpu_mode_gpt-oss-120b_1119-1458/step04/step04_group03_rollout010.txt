@triton.jit
def _gate_fuse_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    mask_ptr, left_out_ptr, right_out_ptr,
    total_size,
    H: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):

    offs = idx * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)

def _batched_matmul_torch(left, right):
    left: [B,N,N,H]
    right: [B,N,N,H]
    # reshape to [B*H,N,N]
    B,N,_,H = left.shape
    left_reshape = left.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)
    right_reshape = right.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)
    # compute matmul with B*H batches
    out = torch.bmm(left_reshape, right_reshape.transpose(1,2))
    # reshape back
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

Algorithm (fp16‑first, fp32‑final)
---------------------------------
1️⃣  Layer‑norm the input (float32) → cast to fp16.
2️⃣  Four linear projections (left/right) and three gating projections are performed
    in fp16 (torch.nn.functional.linear).  
3️⃣  The mask (if present) and the three gates are fused into the left/right
    tensors with a tiny Triton element‑wise kernel (in‑place, no extra memory).
4️⃣  Heavy pair‑wise contraction  
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]  
    is a batched GEMM: for each hidden channel we compute a matrix‑multiply
    `C = A @ Bᵀ`.  We use the highly‑optimized cuBLAS implementation via
    ``torch.bmm`` (fp16×fp16 → fp32 accumulation).  This is orders of magnitude
    faster than a custom GEMM Triton kernel for the tested sequence lengths.
5️⃣  Second LayerNorm (fp16) → element‑wise out‑gate (sigmoid) → final linear
    projection back to ``dim``.
6️⃣  Cast the result to float32 (required by the harness).

All heavy compute stays in fp16 (Tensor‑Core friendly) while the small
element‑wise fusion is done with a dedicated Triton kernel, satisfying the
requirement of “using Triton for at least part of the work”.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if any) and gating into the left/right tensors.
#   Performs in‑place:
#       left  = left  * left_gate  * mask
#       right = right * right_gate * mask
#   (mask is omitted when HAS_MASK == 0)
# ----------------------------------------------------------------------
@triton.jit
def _gate_fuse_kernel(
    left_ptr,          # *fp16  [total_elements]
    left_gate_ptr,     # *fp16  [total_elements]
    right_ptr,         # *fp16  [total_elements]
    right_gate_ptr,    # *fp16  [total_elements]
    mask_ptr,          # *fp16  [mask_elements]   (ignored when HAS_MASK==0)
    total_elements,    # i32   total number of elements in left/right
    H: tl.constexpr,   # hidden dimension (used to map element -> mask entry)
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # load tensors
    left   = tl.load(left_ptr   + offs, mask=mask,  other=0.0)
    lgate  = tl.load(left_gate_ptr + offs, mask=mask,  other=0.0)
    right  = tl.load(right_ptr  + offs, mask=mask,  other=0.0)
    rgate  = tl.load(right_gate_ptr + offs, mask=mask,  other=0.0)

    if HAS_MASK:
        # each block of H consecutive elements share the same mask entry
        mask_idx = offs // H
        mval = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left  = left  * mval
        right = right * mval

    # apply gates (in‑place)
    left  = left  * lgate
    right = right * rgate

    # store back
    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


# ----------------------------------------------------------------------
# Batched GEMM (C = A @ Bᵀ) using torch.bmm (cuBLAS).  Operates on tensors
# of shape [B, N, N, H]  (fp16).  Returns tensor of the same shape (fp16).
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right (both [B, N, N, H]) using cuBLAS batched matmul.
    Returns [B, N, N, H] (fp16).

    The contraction is equivalent to:
        out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
    """
    B, N, _, H = left.shape
    # reshape to [M, N, N] where M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # [M, N, N]
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # [M, N, N]

    # batched mat‑mul with the second operand transposed
    C = torch.bmm(A, B_.transpose(1, 2))                         # [M, N, N]

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 / bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 2️⃣ first layer‑norm (float32) – stable for the initial projection
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 3️⃣ cast to fp16 for the rest of the pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ fp16 copies of linear weights (store once)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 5️⃣ linear projections (fp16)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # [B, N, N, hidden]
    right = F.linear(x, right_proj_w)   # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 6️⃣ gating linear layers (fp16) – compute gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # later used after norm

    # ------------------------------------------------------------------
    # 7️⃣ fuse mask (if any) and gate multiplication with a tiny Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N]; broadcast across hidden dim in kernel
        mask_fp16 = mask_tensor.to(torch.float16)
        # ensure contiguous memory for the kernel
        mask_fp16 = mask_fp16.contiguous()
    else:
        # dummy pointer – will never be read because HAS_MASK == 0
        mask_fp16 = left  # any tensor of compatible dtype works

    # Ensure left/right/gates are contiguous before the kernel launch
    left      = left.contiguous()
    right     = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    total_elems = left.numel()
    BLOCK_SIZE = 1024  # 1‑K elements per program; chosen to keep launches low

    grid = ( (total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE, )
    _gate_fuse_kernel[grid](
        left_ptr=left.data_ptr(),
        left_gate_ptr=left_gate.data_ptr(),
        right_ptr=right.data_ptr(),
        right_gate_ptr=right_gate.data_ptr(),
        mask_ptr=mask_fp16.data_ptr(),
        total_elements=total_elems,
        H=hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
        HAS_MASK=int(not nomask),
        num_warps=4,               # elementwise work – a few warps is enough
    )
    # after the kernel left/right now contain the masked‑gated values
    # (left_gate/right_gate can be freed)
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 8️⃣ heavy contraction – batched mat‑mul via cuBLAS
    # ------------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, hidden] (fp16)

    # ------------------------------------------------------------------
    # 9️⃣ second LayerNorm (fp16) → out‑gate → final proj
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    out = out * out_gate                                 # element‑wise gate
    out = F.linear(out, to_out_w)                        # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 🔟 cast back to float32 (required by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
