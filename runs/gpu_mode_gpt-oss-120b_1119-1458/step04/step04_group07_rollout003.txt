"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.
------------------------------------------------
The heavy tensor contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is realised as a batched GEMM C = A @ Bᵀ with a custom Triton kernel.
All surrounding ops (layer‑norms, linear projections, gating and masking)
are performed in FP16 with in‑place / fused operations to minimise memory
traffic.  The implementation follows the reference PyTorch module but
significantly reduces the number of full‑tensor passes (projections and
gates are concatenated) and keeps the O(N³) term entirely on the GPU.
The final result is returned as float‑32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : [M, N, K]   (M = batch * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
#   All tensors are fp16, accumulator is fp32.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # scalar sizes
    stride_am, stride_ai, stride_ak,   # A strides for (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides for (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides for (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # program IDs (tile indices)
    # ------------------------------------------------------------------
    pid_i = tl.program_id(0)   # i‑tile index
    pid_j = tl.program_id(1)   # j‑tile index
    pid_z = tl.program_id(2)   # batch*hidden index (M dimension)

    # ------------------------------------------------------------------
    # offsets within the tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # accumulator (fp32 for numerical stability)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # iterate over K dimension in blocks
    # ------------------------------------------------------------------
    for k_start in range(0, K, BLOCK_K):
        cur_k = k_start
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ----- load A tile, shape (BLOCK_M, BLOCK_K) -----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # ----- load B tile (transposed), shape (BLOCK_K, BLOCK_N) -----
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # ----- mixed‑precision matmul (fp16 × fp16 → fp32) -----
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # write back the tile (store as fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(
    left: torch.Tensor, right: torch.Tensor
) -> torch.Tensor:
    """
    Compute ``out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]``.
    ``left`` and ``right`` are expected to be FP16 tensors of shape
    [B, N, N, H] (contiguous).  The function returns a tensor of the same
    shape and dtype (FP16) containing the contracted result.
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] – each hidden channel becomes an independent batch
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A, dtype=torch.float16)  # output buffer

    # element‑wise strides (in element counts, not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # tile size selection
    if N >= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden
    )

    _tri_mul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps give good occupancy on H100
        num_stages=3,         # triple‑buffered pipeline
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operator (no backward).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (float32/bool) – may be ignored if ``nomask`` is True
        - weights_dict : mapping from weight name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default ``True``).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm (float32 for numerical stability)
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # -----------------------------------------------------------------
    # 2️⃣ Switch to fp16 for the rest of the pipeline
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Fuse linear projections (left/right) and gates (left/right)
    # -----------------------------------------------------------------
    # --- concatenate projection weights ---
    left_proj_w  = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    proj_w_cat   = torch.cat([left_proj_w, right_proj_w], dim=0)          # (2*hidden, dim)

    # --- concatenate gate weights (pre‑sigmoid) ---
    left_gate_w  = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    gate_w_cat   = torch.cat([left_gate_w, right_gate_w], dim=0)          # (2*hidden, dim)

    # --- linear projection for both sides in one forward pass ---
    proj_concat = torch.nn.functional.linear(x, proj_w_cat)  # [B, N, N, 2*hidden]
    left, right = proj_concat.chunk(2, dim=-1)               # each [B, N, N, hidden]

    # --- gate computation (sigmoid) ---
    gate_concat = torch.nn.functional.linear(x, gate_w_cat)  # [B, N, N, 2*hidden]
    gate_concat = torch.sigmoid(gate_concat)
    left_gate, right_gate = gate_concat.chunk(2, dim=-1)     # each [B, N, N, hidden]

    # -----------------------------------------------------------------
    # 4️⃣ Apply element‑wise gates (in‑place to reduce temporaries)
    # -----------------------------------------------------------------
    torch.mul(left, left_gate, out=left)   # left = left * left_gate
    torch.mul(right, right_gate, out=right)

    # -----------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden dimension)
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -----------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton (still FP16)
    # -----------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    contracted = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden] fp16

    # -----------------------------------------------------------------
    # 7️⃣ out‑gate (computed from the same normalised input x)
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(
            x,
            weights["out_gate.weight"].to(device=device, dtype=torch.float16),
        )
    )  # [B, N, N, hidden] fp16

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (still fp16)
    # -----------------------------------------------------------------
    contracted = torch.nn.functional.layer_norm(
        contracted,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device=device, dtype=torch.float16),
        bias=weights["to_out_norm.bias"].to(device=device, dtype=torch.float16),
    )

    # -----------------------------------------------------------------
    # 9️⃣ Final gating + projection back to original dimension
    # -----------------------------------------------------------------
    contracted = contracted * out_gate
    out = torch.nn.functional.linear(
        contracted,
        weights["to_out.weight"].to(device=device, dtype=torch.float16),
    )  # [B, N, N, dim] fp16

    # -----------------------------------------------------------------
    # 10️⃣ Cast back to float32 (the harness expects float32 output)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```
