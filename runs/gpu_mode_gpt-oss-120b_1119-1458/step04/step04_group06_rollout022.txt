def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0) # tile on first N dimension (i)
    pid_n = tl.program_id(1) # tile on second N dimension (j)
    pid_z = tl.program_id(2) # batch * hidden index (M)

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj

import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _pointwise_mul_kernel(x_ptr, y_ptr, N, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    y = tl.load(y_ptr + offs, mask=mask, other=0.0)
    tl.store(x_ptr + offs, x * y, mask=mask)  # inplace

def _pointwise_mul_inplace(x: torch.Tensor, y: torch.Tensor):
    # both x and y must be contiguous fp16 tensors
    assert x.is_contiguous() and y.is_contiguous()
    N = x.numel()
    BLOCK = 1024
    grid = ((N + BLOCK - 1) // BLOCK,)
    _pointwise_mul_kernel[grid](
        x, y, N,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,hidden]

Outgoing TriMul (AlphaFold‑3) forward pass – optimized for H100.

Algorithm
---------
1️⃣  LayerNorm the 4‑D input in FP32 for numerical stability, then cast to FP16.
2️⃣  Four linear projections (left/right) and three gating projections are computed

4️⃣  The cubic contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    handle the heavy work (FP16×FP16 → FP32 accumulation) – up to ~250 GFLOP.
5️⃣  Second LayerNorm (FP16) → out‑gate (sigmoid) → final linear projection
    back to the original `dim`.
6️⃣  Cast the result back to FP32 as required by the harness.

multiplications use cuBLAS, which on an H100 is far faster than a naïve

import triton

# Triton kernel: fused element‑wise multiplication (in‑place)
#    out ← out * mul
# ----------------------------------------------------------------------
@triton.jit
def _pointwise_mul_kernel(
    out_ptr,          # *fp16   input & output
    mul_ptr,          # *fp16   multiplier
    N,                # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    a = tl.load(out_ptr + offs, mask=mask, other=0.0)
    b = tl.load(mul_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _pointwise_mul_inplace(x: torch.Tensor, y: torch.Tensor) -> None:

    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)
    _pointwise_mul_kernel[grid](
        x, y, total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
