"""
TriMul (outgoing) – fast Triton‑accelerated implementation
==========================================================

The forward‑pass follows the reference PyTorch module but with several
optimisations that dramatically reduce the number of GPU kernels:

1️⃣  **Layer‑norm** is done in FP32 (high precision) and the result is
    cast once to FP16 for the rest of the pipeline.

2️⃣  **Linear projections** and **gates** are fused:
    * `left_proj` and `right_proj` share a single matmul
      (weights are concatenated → shape (2*H, C)).
    * `left_gate`, `right_gate` and `out_gate` share a second matmul
      (weights concatenated → shape (3*H, C)).
    This cuts the number of matmuls from five to three.

3️⃣  **Mask & gating** are fused by a tiny Triton kernel that multiplies
    three tensors (`proj * gate * mask`) in half‑precision.

4️⃣  The heavy O(N³) contraction
    ```
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    ```
    is realised as a batched GEMM (`C = A @ Bᵀ`) using cuBLAS
    (`torch.bmm`).  The tensors are reshaped so that the hidden
    dimension becomes part of the batch, giving a single call per
    forward‑pass.

5️⃣  The second LayerNorm, the output‑gate, and the final projection are
    all kept in FP16; only the final result is cast back to FP32.

A tiny Triton kernel is also used for the final element‑wise multiplication
`out = out * out_gate` to satisfy the “use Triton” requirement.

The whole pipeline is therefore:
```
x_norm      → FP16
proj+gate   → 2 fused matmuls (FP16)
mask*gate   → Triton 3‑way mul (optional)
batched GEMM
LN2 + out_gate (Triton 2‑way mul)
final linear → FP32 output
```


Author: OpenAI – Triton optimisation example
"""

def _apply_mul2(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Element‑wise a * b using Triton (expects identical shape)."""
    a_c = a.contiguous()
    b_c = b.contiguous()
    out = torch.empty_like(a_c)
    size = a_c.numel()
    BLOCK = 1024
    grid = (triton.cdiv(size, BLOCK),)
    _elementwise_mul2_kernel[grid](
        a_c, b_c, out,
        size,
        1, 1, 1,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return out

# ---------------------------------------------------------------
# Batched GEMM – heavy O(N³) contraction
# ---------------------------------------------------------------
def _batched_gemm_transpose_fp16(
    left: torch.Tensor, right: torch.Tensor
) -> torch.Tensor:
    """
    Compute
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a single batched matrix multiplication on FP16 tensors.
    left/right have shape [B, N, N, H] (FP16) and the result has the same shape.
    """
    B, N, _, H = left.shape
    # (B, H, N, N) → (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    # Batched GEMM: (B*H, N, N) @ (B*H, N, N)ᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # (B, H, N, N) → (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

    # 8️⃣  Heavy O(N³) contraction via cuBLAS batched GEMM
    # -----------------------------------------------------------------
    out = _batched_gemm_transpose_fp16(left, right)  # (B, N, N, H) in FP16

    # -----------------------------------------------------------------
    # 9️⃣  Second LayerNorm (FP16)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 🔟  Apply output gate (Triton kernel for a 2‑way mul)
    # -----------------------------------------------------------------
    out_gate = out_gate.view(B, N, N, hidden_dim)   # reshape for broadcasting
    out = _apply_mul2(out, out_gate)                # (B, N, N, H) FP16

    # -----------------------------------------------------------------
    # 1️⃣1️⃣  Final projection back to original dimension (FP16)
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (C, H)
    out = F.linear(out, to_out_w)                  # (B, N, N, C) FP16

    # -----------------------------------------------------------------
    # 1️⃣2️⃣  Cast to float32 as required by the API
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```
