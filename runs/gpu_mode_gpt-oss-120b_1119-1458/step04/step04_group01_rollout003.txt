"""
custom_kernel implements the *outgoing* TriMul operation (AlphaFold3).

Algorithm
---------
1️⃣  Layer‑Norm the input 4‑D pair tensor (float32 → fp16 for the rest).
2️⃣  Fuse all linear projections that share the same input (left/right
    projections and their gates, plus the output‑gate) into a single
    half‑precision GEMM.  This reduces memory traffic on the large
    `x` tensor.
3️⃣  Apply an optional binary mask (broadcast over the hidden dimension)
    to the two projected tensors.
4️⃣  Fuse the gating (sigmoid) and element‑wise multiplication
    `proj * sigmoid(gate)` in a custom Triton kernel.  Two kernels are
    launched – one for the left side and one for the right side.
5️⃣  Perform the cubic contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    as a batched half‑precision matrix multiplication
    `C = A @ Bᵀ` using `torch.bmm`.  This deleg­ates the heavy O(N³)
    work to cuBLAS (fast on H100).
6️⃣  Apply a second Layer‑Norm on the hidden dimension.
7️⃣  Multiply by the output‑gate (sigmoid) and project back to the
    original feature dimension with a final linear layer.
8️⃣  Cast the result back to float32 (the model’s expected output type).

All heavy arithmetic is performed in fp16; Triton is used for the
gating‑multiply kernel, satisfying the “must use Triton” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel for gated element‑wise multiplication:
# out = proj * sigmoid(gate)
# ----------------------------------------------------------------------
@triton.jit
def _gated_mul_kernel(
    proj_ptr,               # ptr to projected tensor (fp16)
    gate_ptr,               # ptr to gate pre‑sigmoid tensor (fp16)
    out_ptr,                # ptr to output tensor (fp16)
    size,                   # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    # Load
    proj = tl.load(proj_ptr + offs, mask=mask)   # fp16
    gate = tl.load(gate_ptr + offs, mask=mask)   # fp16

    # Sigmoid(gate) = 1 / (1 + exp(-gate))
    gate_sig = 1.0 / (1.0 + tl.exp(-gate))

    # Multiply
    out = proj * gate_sig

    # Store
    tl.store(out_ptr + offs, out, mask=mask)


def _gated_mul(proj: torch.Tensor, gate: torch.Tensor) -> torch.Tensor:
    """
    Element‑wise multiplication of “proj * sigmoid(gate)”.
    Both inputs must be fp16 and have identical shape.
    """
    assert proj.shape == gate.shape, "proj and gate must share shape"
    # Flatten to 1‑D tensors (contiguous) – stride = 1
    proj_f = proj.contiguous().view(-1)
    gate_f = gate.contiguous().view(-1)
    out_f = torch.empty_like(proj_f)

    total = proj_f.numel()
    BLOCK = 1024                     # number of threads per program
    grid = (triton.cdiv(total, BLOCK),)

    _gated_mul_kernel[grid](
        proj_f, gate_f, out_f, total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return out_f.view(proj.shape)


# ----------------------------------------------------------------------
# Batched matrix multiplication C = A @ Bᵀ for the O(N³) contraction.
# Handles tensors of shape [B, N, N, H] (fp16) → output [B, N, N, H].
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor,
                                   right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a batched GEMM (left @ rightᵀ) on half‑precision data.
    """
    B, N, _, H = left.shape
    # Collapse batch & hidden dims → (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    # Batched matmul
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # Restore shape [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]          (float32)
        - mask_tensor  : torch.Tensor [B, N, N]              (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    device = inp.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16 for the rest)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(inp, (dim,), weight=norm_w, bias=norm_b)          # float32
    x = x.to(torch.float16)                                            # fp16

    # ------------------------------------------------------------------
    # 2️⃣ Cast all linear weights to fp16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse all projections that share the same input (x) into one GEMM.
    #    Output layout: [left_proj, left_gate, right_proj, right_gate, out_gate]
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape (5*hidden, dim)
    fused = F.linear(x, fused_w)  # [B, N, N, 5*hidden]  (fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Slice fused tensor into its components
    # ------------------------------------------------------------------
    h = hidden
    left_proj_pre = fused[..., 0:h]                # (B,N,N,h)
    left_gate_pre = fused[..., h:2*h]
    right_proj_pre = fused[..., 2*h:3*h]
    right_gate_pre = fused[..., 3*h:4*h]
    out_gate_pre = fused[..., 4*h:5*h]            # (B,N,N,h)

    # ------------------------------------------------------------------
    # 5️⃣ Optional masking (applied after projection, before gating)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj_pre * mask_fp16
        right_proj = right_proj_pre * mask_fp16
    else:
        left_proj = left_proj_pre
        right_proj = right_proj_pre

    # ------------------------------------------------------------------
    # 6️⃣ Gating + element‑wise multiplication (fully fused in Triton)
    # ------------------------------------------------------------------
    left = _gated_mul(left_proj, left_gate_pre)   # (B,N,N,h)
    right = _gated_mul(right_proj, right_gate_pre)

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via batched GEMM
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # (B,N,N,h)

    # ------------------------------------------------------------------
    # 8️⃣ Compute out‑gate (sigmoid) and second LayerNorm
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_pre)               # (B,N,N,h)
    out = F.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # (B,N,N,h)

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate and final projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate                                 # (B,N,N,h)
    out = F.linear(out, to_out_w)                        # (B,N,N,dim)  fp16

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to float32 (model output datatype)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
