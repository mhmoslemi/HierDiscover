"""
Outgoing TriMul (AlphaFold‑3) forward pass – Triton‑accelerated contraction.

Algorithm
---------
1️⃣  Layer‑norm the input tensor (float32) across the last dimension.
2️⃣  Cast to fp16 – the heavy compute runs in half‑precision.
3️⃣  Four linear projections (left/right) and three gating projections
    (left_gate, right_gate, out_gate).  All weights are converted to fp16.
4️⃣  Optional pairwise mask is broadcast on the hidden dimension and applied
    to the left/right projections *before* gating.
5️⃣  Gating values (sigmoid) are fused in‑place into the projections.
6️⃣  O(N³) update:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    This is a batched GEMM C = A @ Bᵀ where
    A = left   → [B·H, N, N]   (M, I, K)
    B = right  → [B·H, N, N]   (M, J, K)   (transposed on‑the‑fly)
    The contraction is performed by a Triton kernel that tiles the
    (i, j) output plane (BLOCK_M × BLOCK_N) and iterates over K in
    BLOCK_K‑sized chunks, using fp16·fp16 → fp32 accumulation.
    For very short sequences (N ≤ 128) we fall back to `torch.einsum` to
    avoid kernel launch overhead.
7️⃣  Second LayerNorm (fp16) over the hidden dimension.
8️⃣  Multiply by the out‑gate (sigmoid) and apply the final linear
    projection back to the original `dim`.
9️⃣  Cast the result back to float32 (the harness expects FP32).

The implementation mixes torch primitives (layer‑norm, linear, sigmoid) with
a custom Triton kernel for the dominant O(N³) contraction, achieving
high throughput on H100 GPUs.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : [M, N, K]   (M = B*hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
#   A, B, C are fp16; accumulator is fp32.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K] (will be read transposed)
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # program‑size constants
    stride_am, stride_ai, stride_ak,   # A strides
    stride_bm, stride_bj, stride_bk,   # B strides
    stride_cm, stride_ci, stride_cj,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for each batch‑hidden slice.
    A: [M, N, K]  (i dimension = first N)
    B: [M, N, K]  (j dimension = second N, to be transposed)
    C: [M, N, N]
    """
    pid_m = tl.program_id(0)   # tile on i
    pid_n = tl.program_id(1)   # tile on j
    pid_z = tl.program_id(2)   # batch * hidden index (M)

    # ------------------------------------------------------------------
    # Tile offsets within the N×N output plane
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles needed
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # -----------------------------
        # Load A tile  (M, i, k)
        # -----------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # -----------------------------
        # Load B tile  (M, j, k)  – transposed on‑the‑fly → (k, j)
        # -----------------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # Mixed‑precision dot product (fp16 × fp16 → fp32 accumulation)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store result tile (cast back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = left @ rightᵀ where left/right are
    [B, N, N, H] (fp16).  Returns C in the same layout.
    """
    B, N, _, H = left.shape

    # Rearrange to [M, N, K] where M = B*H, I = N, K = N
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in element counts (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()   # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()  # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()   # (N*N, N, 1)

    # Tile size selection – larger tiles for larger N
    if N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    # Grid: (i‑tiles, j‑tiles, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Restore original layout [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32, bool or None)
        - weights_dict : mapping string → torch.Tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )  # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the remaining pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast model weights to fp16 (on‑device)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask – broadcast on hidden dim before gating
    # ------------------------------------------------------------------
    if not nomask:
        # Ensure mask is fp16 and has trailing dim for broadcasting
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Fuse gates in‑place to avoid extra temporaries
    left.mul_(left_gate)
    right.mul_(right_gate)

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction (batched GEMM)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    if N <= 128:
        # For very short sequences the Triton launch overhead dominates.
        out = torch.einsum("b i k h, b j k h -> b i j h", left, right)  # fp16 output
    else:
        out = _batched_matmul_transpose_fp16(left, right)               # fp16 output

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Output gating + final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 🔟 Cast back to float32 (expected by the evaluation harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
