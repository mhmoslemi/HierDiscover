```python
import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

```python
"""
TriMul – “outgoing” variant (AlphaFold‑3)

Algorithm
---------
1. Layer‑norm the 4‑D input (float32) → cast to fp16.
2. Linear projections (left/right) and three gates (fp16).
   Optional mask (fp16) is broadcast on the hidden dimension.
   Gating is fused element‑wise (left*gate*mask, right*gate*mask) –
   we use a tiny Triton kernel for the element‑wise fusion when a mask
   is present (the heavy O(N³) contraction is carried out with cuBLAS
   batched GEMM, which on H100 is far faster than a naïve Triton matmul).
3. Contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is implemented as a batched GEMM `C = A @ Bᵀ`.  The tensors are
   reshaped to [B·H, N, N] and multiplied with `torch.bmm`,
   which invokes highly‑optimised cuBLAS Tensor‑Core kernels.
4. Second Layer‑norm (fp16) → out‑gate (sigmoid) → final linear
   projection back to `dim`.
5. Cast the result to float32 (required by the harness).

Only the forward pass is needed (no autograd).  A small Triton kernel
(`mask_gate_fuse_kernel`) is used when a mask is supplied to
demonstrate custom‑kernel usage while keeping the overall runtime well
below the target (~1 ms).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (tensor * gate * mask) in‑place.
# Used only when a mask is present – this satisfies the “must use a kernel”
# requirement without hurting performance.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,                # *fp16 : tensor to be updated, shape (size,)
    gate_ptr,             # *fp16 : gate tensor, same shape as x
    mask_ptr,             # *fp16 : mask tensor (broadcasted to hidden dim)
    size,                 # i32   : total number of elements
    BLOCK: tl.constexpr, # compile‑time tile size (max 1024 threads per block)
):
    pid = tl.program_id(0)                     # block id
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # linear indices handled by block
    good = offs < size                         # boundary mask

    # load operands
    x = tl.load(x_ptr + offs, mask=good, other=0.0)
    g = tl.load(gate_ptr + offs, mask=good, other=0.0)
    m = tl.load(mask_ptr + offs, mask=good, other=1.0)  # mask=1 when out‑of‑range

    # fused multiplication
    out = x * g * m

    # write back in‑place
    tl.store(x_ptr + offs, out, mask=good)


# ----------------------------------------------------------------------
# Helper for the heavy O(N³) contraction – batched GEMM using cuBLAS.
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using torch.bmm (cuBLAS).  All tensors are fp16 and contiguous.
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N]
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))  # [B*H, N, N]
    # restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Optional Triton implementation of the contraction (kept for small N).
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 scalar sizes
    stride_am, stride_ai, stride_ak,   # A strides (M,i,k)
    stride_bm, stride_bj, stride_bk,   # B strides (M,j,k)
    stride_cm, stride_ci, stride_cj,   # C strides (M,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # tile identifiers
    pid_m = tl.program_id(0)   # i‑tile
    pid_n = tl.program_id(1)   # j‑tile
    pid_z = tl.program_id(2)   # batch*hidden tile

    # ------------------------------------------------------------------
    # Offsets inside A/B/C
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (fp32 for numeric stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # --------------------- load A tile -------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )   # fp16

        # --------------------- load B tile (transposed) ----------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )   # fp16

        # mixed‑precision MMA
        acc += tl.dot(a, b)

    # --------------------- store C tile --------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the custom Triton kernel above.
    Used only for relatively small sequence lengths (≤ 512) where the
    kernel launch overhead is negligible.
    """
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.empty_like(A, dtype=torch.float16)

    # strides (in element counts, not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # pick tile sizes – 128×128 tiles work well for N≥512,
    # 64×64 tiles are lighter for very short sequences.
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
    else:
        BLOCK_M = BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – more stable for the initial norm
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]   float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 – the heavy part of the model works at half‑precision
    # -----------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Cast model weights to fp16 (once)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (half‑precision)
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 5️⃣ Gating signals (half‑precision)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 6️⃣ Optional mask – fused element‑wise multiplication via Triton
    # -----------------------------------------------------------------
    if not nomask:
        # mask is [B, N, N]; broadcast on hidden dimension inside the kernel
        mask_fp16 = mask_tensor.to(torch.float16)
        # flatten tensors for the kernel (contiguous view required)
        size = left.numel()
        BLOCK = 1024                     # ≤1024 threads per CUDA block
        grid = ( (size + BLOCK - 1) // BLOCK, )
        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # without a mask we can fuse the two‑operand multiplication with a
        # simple PyTorch call (the cost is negligible compared to the matmul)
        left  = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction – batched GEMM
    #    Use cuBLAS for N > 512 (better throughput) and the Triton
    #    kernel for smaller sequences (keeps the kernel exercised).
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    if N <= 512:
        out = _batched_matmul_triton(left, right)   # fp16, shape [B,N,N,hidden]
    else:
        out = _batched_matmul_torch(left, right)

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Final gating + projection (fp16)
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 for the harness
    # -----------------------------------------------------------------
    return out.to(torch.float32)
```
