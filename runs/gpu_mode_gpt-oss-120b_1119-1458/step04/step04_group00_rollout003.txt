@triton.jit
def fused_mask_gate_kernel(
    proj_l_ptr, gate_l_ptr, proj_r_ptr, gate_r_ptr, mask_ptr,
    out_l_ptr, out_r_ptr,
    M, H,
    stride_proj_l_m, stride_proj_l_h,
    stride_gate_l_m, stride_gate_l_h,
    stride_proj_r_m, stride_proj_r_h,
    stride_gate_r_m, stride_gate_r_h,
    stride_out_l_m, stride_out_l_h,
    stride_out_r_m, stride_out_r_h,
    stride_mask,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr
):
   pid = tl.program_id(0)
   block_start = pid * BLOCK_M
   offs_m = block_start + tl.arange(0, BLOCK_M)
   offs_h = tl.arange(0, BLOCK_H)

   mask_m = tl.load(mask_ptr + offs_m * stride_mask, mask=offs_m < M, other=0.0).to(tl.float16)  # shape [BLOCK_M] 
   mask_h = mask_m[:, None]  # [BLOCK_M,1]

   # load tiles
   proj_l = tl.load(proj_l_ptr + offs_m[:, None] * stride_proj_l_m + offs_h[None, :] * stride_proj_l_h,
                    mask=(offs_m[:, None] < M) & (offs_h[None, :] < H), other=0.0)
   gate_l = tl.load(gate_l_ptr + offs_m[:, None] * stride_gate_l_m + offs_h[None, :] * stride_gate_l_h,
                    mask=(offs_m[:, None] < M) & (offs_h[None, :] < H), other=0.0)
   proj_r = tl.load(proj_r_ptr + offs_m[:, None] * stride_proj_r_m + offs_h[None, :] * stride_proj_r_h,
                    mask=(offs_m[:, None] < M) & (offs_h[None, :] < H), other=0.0)
   gate_r = tl.load(gate_r_ptr + offs_m[:, None] * stride_gate_r_m + offs_h[None, :] * stride_gate_r_h,
                    mask=(offs_m[:, None] < M) & (offs_h[None, :] < H), other=0.0)

   # fused operations
   out_l = proj_l * gate_l * mask_h
   out_r = proj_r * gate_r * mask_h

   # store
   tl.store(out_l_ptr + offs_m[:, None] * stride_out_l_m + offs_h[None, :] * stride_out_l_h,
            out_l,
            mask=(offs_m[:, None] < M) & (offs_h[None, :] < H))

   tl.store(out_r_ptr + offs_m[:, None] * stride_out_r_m + offs_h[None, :] * stride_out_r_h,
            out_r,
            mask=(offs_m[:, None] < M) & (offs_h[None, :] < H))

def custom_kernel(data):
    import torch, triton, triton.language as tl
    # Unpack
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # Cast weights to FP16 (some may need to be FP32)
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # Cast to FP16
    norm_w = norm_w.to(torch.float16)
    norm_b = norm_b.to(torch.float16)

    # Process input via layer norm (FP32->FP16)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    B, N, _, _ = x.shape
    M = B * N * N

    # Fused linear weights:
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    # Cast to FP16
    left_proj_w = left_proj_w.to(torch.float16)
    left_gate_w = left_gate_w.to(torch.float16)
    right_proj_w = right_proj_w.to(torch.float16)
    right_gate_w = right_gate_w.to(torch.float16)
    out_gate_w = out_gate_w.to(torch.float16)

    fused_weight = torch.cat([left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w], dim=0)
    # shape (5*hidden_dim, dim)

    # Flatten x and apply fused linear
    x_flat = x.reshape(-1, dim)  # (M, dim)
    fused_out = torch.nn.functional.linear(x_flat, fused_weight)  # (M, 5*hidden_dim)
    # if bias? None.

    # Split fused_out
    h = hidden_dim
    left_proj_raw = fused_out[:, 0:h]
    left_gate_raw = fused_out[:, h:2*h]
    right_proj_raw = fused_out[:, 2*h:3*h]
    right_gate_raw = fused_out[:, 3*h:4*h]
    out_gate_raw = fused_out[:, 4*h:5*h]

    # Apply sigmoid to gates
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw).view(B, N, N, hidden_dim)

    # At this point we have left_proj_raw and right_proj_raw raw values.

    # If mask is needed (nomask=False)
    if not nomask:
        # Flatten mask
        mask_fp16 = mask_tensor.to(torch.float16).reshape(-1)  # (M,)
        # Prepare output tensors
        left = torch.empty_like(left_proj_raw)
        right = torch.empty_like(right_proj_raw)
        # Determine strides
        stride_lp_m, stride_lp_h = left_proj_raw.stride()
        stride_lg_m, stride_lg_h = left_gate.stride()
        stride_rp_m, stride_rp_h = right_proj_raw.stride()
        stride_rg_m, stride_rg_h = right_gate.stride()
        stride_out_m, stride_out_h = left.stride()  # left same shape as output and same stride for right.

        stride_mask = mask_fp16.stride()[0]
        # But note that later we also need to pass right output stride (same as left). In our combined kernel, we need separate output strides for left and right.

        # Create kernel for fused mask gating for left & right.

        BLOCK_M = 64 # or tune
        BLOCK_H = hidden_dim
        grid = ( (M + BLOCK_M - 1) // BLOCK_M, )

        # Run kernel
        _fused_mask_gate_kernel[grid](
            left_proj_raw, left_gate,
            right_proj_raw, right_gate,
            mask_fp16,
            left, right,
            M, hidden_dim,
            stride_lp_m, stride_lp_h,
            stride_lg_m, stride_lg_h,
            stride_rp_m, stride_rp_h,
            stride_rg_m, stride_rg_h,
            stride_out_m, stride_out_h,
            stride_out_m, stride_out_h,  # same for right output
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
        # reshape
        left = left.view(B, N, N, hidden_dim)
        right = right.view(B, N, N, hidden_dim)
    else:
        # no mask: compute left and right via elementwise multiplication then reshape
        left = (left_proj_raw * left_gate).view(B, N, N, hidden_dim)
        right = (right_proj_raw * right_gate).view(B, N, N, hidden_dim)

    # Now compute contraction via batched GEMM
    # Reshape left and right to (B*hidden_dim, N, N) and multiply left * right^T.
    # Use torch.bmm.
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Second layernorm
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # Apply out_gate
    out = out * out_gate

    # Final linear
    to_out_weight = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_weight)  # shape (B,N,N,dim)

    # Cast to FP32
    out = out.to(torch.float32)
    return out

@triton.jit
def _fused_mask_gate_kernel(
    proj_l_ptr, gate_l_ptr, proj_r_ptr, gate_r_ptr, mask_ptr,
    out_l_ptr, out_r_ptr,
    M, H,
    stride_pl_m, stride_pl_h,
    stride_gl_m, stride_gl_h,
    stride_pr_m, stride_pr_h,
    stride_gr_m, stride_gr_h,
    stride_out_l_m, stride_out_l_h,
    stride_out_r_m, stride_out_r_h,
    stride_mask,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_M
    offs_m = block_start + tl.arange(0, BLOCK_M)
    offs_h = tl.arange(0, BLOCK_H)

    mask_val = tl.load(mask_ptr + offs_m * stride_mask,
                       mask=offs_m < M,
                       other=0.0).to(tl.float16)
    mask_h = mask_val[:, None] # shape [BLOCK_M,1]

    # Load proj_l
    proj_l = tl.load(
        proj_l_ptr + offs_m[:, None] * stride_pl_m + offs_h[None, :] * stride_pl_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    # Load gate_l
    gate_l = tl.load(
        gate_l_ptr + offs_m[:, None] * stride_gl_m + offs_h[None, :] * stride_gl_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    # Load proj_r
    proj_r = tl.load(
        proj_r_ptr + offs_m[:, None] * stride_pr_m + offs_h[None, :] * stride_pr_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    # Load gate_r
    gate_r = tl.load(
        gate_r_ptr + offs_m[:, None] * stride_gr_m + offs_h[None, :] * stride_gr_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    # Compute fused
    out_l = proj_l * gate_l * mask_h
    out_r = proj_r * gate_r * mask_h

    # Store
    tl.store(
        out_l_ptr + offs_m[:, None] * stride_out_l_m + offs_h[None, :] * stride_out_l_h,
        out_l,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
    )
    tl.store(
        out_r_ptr + offs_m[:, None] * stride_out_r_m + offs_h[None, :] * stride_out_r_h,
        out_r,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
    )

"""
Outgoing TriMul (AlphaFold3) – Fast forward pass

The implementation follows the reference PyTorch module but moves the
dominant O(N³) contraction to a cuBLAS batched GEMM and fuses the
five linear transformations (two projections, three gates) into a single
matrix‑multiply.  An optional mask is applied together with the
projection‑gate products in a tiny Triton kernel (single kernel for
both left and right branches).  All heavy work is done in FP16 for
Tensor‑Core throughput; the final result is cast back to FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left/right projection and gate (single launch)
# ----------------------------------------------------------------------
@triton.jit
def _fused_mask_gate_kernel(
    proj_l_ptr, gate_l_ptr, proj_r_ptr, gate_r_ptr, mask_ptr,
    out_l_ptr, out_r_ptr,
    M, H,
    stride_pl_m, stride_pl_h,    # left proj strides
    stride_gl_m, stride_gl_h,    # left gate strides
    stride_pr_m, stride_pr_h,    # right proj strides
    stride_gr_m, stride_gr_h,    # right gate strides
    stride_out_l_m, stride_out_l_h,
    stride_out_r_m, stride_out_r_h,
    stride_mask,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_M
    offs_m = block_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    offs_h = tl.arange(0, BLOCK_H)                         # [BLOCK_H]

    # Load mask (1‑D) and broadcast to H
    mask = tl.load(mask_ptr + offs_m * stride_mask,
                   mask=offs_m < M, other=0.0).to(tl.float16)   # [BLOCK_M]
    mask_h = mask[:, None]                                      # [BLOCK_M,1]

    # Load tiles of the four tensors
    proj_l = tl.load(
        proj_l_ptr + offs_m[:, None] * stride_pl_m + offs_h[None, :] * stride_pl_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    gate_l = tl.load(
        gate_l_ptr + offs_m[:, None] * stride_gl_m + offs_h[None, :] * stride_gl_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    proj_r = tl.load(
        proj_r_ptr + offs_m[:, None] * stride_pr_m + offs_h[None, :] * stride_pr_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    gate_r = tl.load(
        gate_r_ptr + offs_m[:, None] * stride_gr_m + offs_h[None, :] * stride_gr_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )

    # Fuse: proj * gate * mask
    out_l = proj_l * gate_l * mask_h
    out_r = proj_r * gate_r * mask_h

    # Store the fused results
    tl.store(
        out_l_ptr + offs_m[:, None] * stride_out_l_m + offs_h[None, :] * stride_out_l_h,
        out_l,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
    )
    tl.store(
        out_r_ptr + offs_m[:, None] * stride_out_r_m + offs_h[None, :] * stride_out_r_h,
        out_r,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
    )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float)
        - weights_dict : mapping string → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device as
        ``input_tensor``.
    """
    # --------------------------------------------------------------
    # 0. unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = int(config["dim"])
    hidden_dim = int(config["hidden_dim"])
    nomask = bool(config.get("nomask", True))

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1. LayerNorm (fp16)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)                      # [B, N, N, dim]

    B, N, _, _ = x.shape
    M = B * N * N                                 # flat pair count

    # --------------------------------------------------------------
    # 2. Fuse the five linear maps into a single matmul
    #    (left_proj, left_gate, right_proj, right_gate, out_gate)
    # --------------------------------------------------------------
    # cast weights once
    w_left_proj = weights["left_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    # concatenated weight: (5*hidden_dim, dim)
    fused_weight = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate], dim=0
    ).contiguous()

    # flatten input and apply fused linear
    x_flat = x.reshape(-1, dim)                  # (M, dim)
    fused_out = F.linear(x_flat, fused_weight)   # (M, 5*hidden_dim)

    # --------------------------------------------------------------
    # 3. Split and apply sigmoid where needed
    # --------------------------------------------------------------
    h = hidden_dim
    left_proj_raw = fused_out[:, 0:h]                       # (M, h)
    left_gate_raw = fused_out[:, h:2 * h]
    right_proj_raw = fused_out[:, 2 * h:3 * h]
    right_gate_raw = fused_out[:, 3 * h:4 * h]
    out_gate_raw = fused_out[:, 4 * h:5 * h]

    left_gate = torch.sigmoid(left_gate_raw)                # (M, h)
    right_gate = torch.sigmoid(right_gate_raw)              # (M, h)
    out_gate = torch.sigmoid(out_gate_raw).view(B, N, N, h) # (B, N, N, h)

    # --------------------------------------------------------------
    # 4. Optional mask + gate fusion (Triton)
    # --------------------------------------------------------------
    if not nomask:
        # flatten mask to (M,)
        mask_fp16 = mask_tensor.to(torch.float16).reshape(-1)

        # allocate output buffers
        left = torch.empty_like(left_proj_raw)
        right = torch.empty_like(right_proj_raw)

        # strides (in element units)
        stride_lp_m, stride_lp_h = left_proj_raw.stride()
        stride_lg_m, stride_lg_h = left_gate.stride()
        stride_rp_m, stride_rp_h = right_proj_raw.stride()
        stride_rg_m, stride_rg_h = right_gate.stride()
        stride_out_m, stride_out_h = left.stride()   # same for left/right outputs
        stride_mask = mask_fp16.stride()[0]

        BLOCK_M = 64                                 # empirical
        BLOCK_H = hidden_dim                         # hidden_dim ≤ 128 in all tests
        grid = ((M + BLOCK_M - 1) // BLOCK_M,)

        # fused mask·gate·proj for both sides in a single kernel launch
        _fused_mask_gate_kernel[grid](
            left_proj_raw, left_gate,
            right_proj_raw, right_gate,
            mask_fp16,
            left, right,
            M, hidden_dim,
            stride_lp_m, stride_lp_h,
            stride_lg_m, stride_lg_h,
            stride_rp_m, stride_rp_h,
            stride_rg_m, stride_rg_h,
            stride_out_m, stride_out_h,
            stride_out_m, stride_out_h,
            stride_mask,
            BLOCK_M=BLOCK_M,
            BLOCK_H=BLOCK_H,
            num_warps=4,
            num_stages=2,
        )
        # reshape to the 4‑D layout expected by the contraction
        left = left.view(B, N, N, hidden_dim)
        right = right.view(B, N, N, hidden_dim)
    else:
        # no mask – simple element‑wise product
        left = (left_proj_raw * left_gate).view(B, N, N, hidden_dim)
        right = (right_proj_raw * right_gate).view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5. O(N³) contraction via batched GEMM (Tensor‑Core)
    #    out_ijkh = Σₖ left_ik h * right_jk h
    # --------------------------------------------------------------
    # reshape to (B·h, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batched matrix multiplication: C = A @ Bᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )                                            # (B, N, N, h)

    # --------------------------------------------------------------
    # 6. Second LayerNorm (fp16)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b
    )

    # --------------------------------------------------------------
    # 7. Output gate and final linear projection (fp16)
    # --------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)               # (B, N, N, dim) in FP16

    # --------------------------------------------------------------
    # 8. Cast back to FP32 for the caller
    # --------------------------------------------------------------
    return out.to(torch.float32)
```
