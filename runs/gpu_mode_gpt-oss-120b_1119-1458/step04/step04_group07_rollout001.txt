"""
Outgoing TriMul – Triton‑accelerated fusion of the cubic contraction.

The forward pass consists of:
1️⃣ Layer‑norm the input (float32 for numerical stability).
2️⃣ Cast to half‑precision (fp16) and apply three linear projections.
3️⃣ (Optional) mask the projected tensors.
4️⃣ Apply three sigmoid gates (in‑place mul to avoid temporaries).
5️⃣ Compute the O(N³·H) contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a batched GEMM C = A @ Bᵀ with a custom Triton kernel.
   – A and B are stored as [M,N,N] where M = B·H.
   – Mixed‑precision dot (fp16×fp16 → fp32) for high throughput.
   – Tile sizes are chosen dynamically (64, 128 or 256) to keep the
     number of thread‑blocks low on H100 while respecting shared‑memory limits.
   – The kernel uses 8 warps and 3 pipeline stages (num_stages=3).
6️⃣ Layer‑norm the result (still fp16), apply the final gating,
   and a last linear projection back to the original dimension.
7️⃣ Cast the answer back to float32 (the harness expects fp32).

All heavy memory‑traffic (the cubic contraction) is performed inside
the Triton kernel; everything else uses high‑performance PyTorch kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : [M, N, N]   (M = batch * hidden_dim, N = seq_len)
#   C    : [M, N, N]
#   Uses fp16 inputs, fp32 accumulation.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    A_ptr, B_ptr, C_ptr,          # *fp16 pointers
    M, N, K,                      # scalar sizes (M = B*H, N = seq_len, K = seq_len)
    stride_am, stride_ai, stride_ak,   # strides for A (M,i,k)
    stride_bm, stride_bj, stride_bk,   # strides for B (M,j,k)
    stride_cm, stride_ci, stride_cj,   # strides for C (M,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Program IDs for the i‑tile, j‑tile and the batch‑hidden index.
    # ------------------------------------------------------------------
    pid_i = tl.program_id(0)   # tile along first N (i)
    pid_j = tl.program_id(1)   # tile along second N (j)
    pid_z = tl.program_id(2)   # batch * hidden dimension

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension in blocks
    # ------------------------------------------------------------------
    for k_start in range(0, K, BLOCK_K):
        cur_k = k_start
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # --------------------------------------------------------------
        # Load a tile of A (shape BLOCK_M × BLOCK_K)
        # --------------------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # --------------------------------------------------------------
        # Load a tile of B (transposed, shape BLOCK_K × BLOCK_N)
        # --------------------------------------------------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Mixed‑precision dot product (fp16 × fp16 → fp32)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result tile (convert back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_gemm_fp16(
    left: torch.Tensor,   # [B, N, N, H]  fp16 (contiguous)
    right: torch.Tensor,  # same shape     fp16 (contiguous)
) -> torch.Tensor:
    """
    Computes the cubic contraction using the Triton kernel above.
    Returns a tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape

    # --------------------------------------------------------------
    # Rearrange to [M, N, N] where M = B*H.
    #   (B, N, N, H) -> (B, H, N, N) -> view as (B*H, N, N)
    # --------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    A = left_perm.view(B * H, N, N)
    B_ = right_perm.view(B * H, N, N)

    # --------------------------------------------------------------
    # Output buffer
    # --------------------------------------------------------------
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in *elements* (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # --------------------------------------------------------------
    # Choose tile sizes based on N.
    #   Larger N → larger tiles to reduce grid size.
    # --------------------------------------------------------------
    if N >= 768:
        BLOCK_M = BLOCK_N = 256
    elif N >= 256:
        BLOCK_M = BLOCK_N = 128
    else:
        BLOCK_M = BLOCK_N = 64
    BLOCK_K = 64  # kept constant – works well on H100

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden
    )

    _trimul_gemm_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # --------------------------------------------------------------
    # Restore original layout: [B, N, N, H]
    # --------------------------------------------------------------
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operator (no backward).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """

    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    # We will run the heavy contraction in fp16 (or bfloat16 on H100) for bandwidth.
    # The initial LayerNorm is kept in float32 for stability.
    # -----------------------------------------------------------------
    # 1️⃣  First LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to half‑precision (fp16) – the rest of the pipeline stays in fp16.
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Cast all linear weights once to fp16 (avoid per‑call casts).
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast on hidden dimension)
    # -----------------------------------------------------------------
    if not nomask:
        # mask is [B,N,N] → broadcast on hidden dim
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -----------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – keep everything fp16.
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Fuse the gates in‑place to reduce temporaries.
    torch.mul(left,  left_gate,  out=left)
    torch.mul(right, right_gate, out=right)

    # -----------------------------------------------------------------
    # 7️⃣ Heavy cubic contraction via Triton.
    #     Both tensors are made contiguous to satisfy the kernel.
    # -----------------------------------------------------------------
    left  = left.contiguous()
    right = right.contiguous()
    out = _batched_gemm_fp16(left, right)   # [B,N,N,hidden] fp16

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (still fp16) – statistics are cheap now.
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Final gating + projection (still fp16)
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 for the harness.
    # -----------------------------------------------------------------
    return out.to(torch.float32)
