@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Determine indices
    batch_hidden = tl.program_id(2)
    b = batch_hidden // H
    h = batch_hidden % H
    # offsets for the start of this batch and hidden slice
    offset_b = b * stride_left_b
    offset_out_b = b * stride_out_b
    offset_h_left = h * stride_left_h
    offset_h_out = h * stride_out_h

    # compute tile start for i and j
    pid_m = tl.program_id(0)  # row blocks
    pid_n = tl.program_id(1)  # col blocks
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # define masks for boundaries
    mask_i = offs_i < N
    mask_j = offs_j < N

    # flatten for addr: left: b, i, k, h
    # tile actual k loop

    input_tensor, mask_tensor, weights, config = data

    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Norm weight & bias
    [...]
    # Cast to fp16
    ...

    # Step1: layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias).to(torch.float16)

    B,N,_,_ = x.shape
    M = B * N * N

    x_flat = x.view(-1, dim)  # shape (M, dim)

    # Combine left proj + gate
    left_proj_gate_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16)
    right_proj_gate_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)

    # Compute left proj+gate
    left_comb = torch.nn.functional.linear(x_flat, left_proj_gate_w)  # (M, 2*hidden)
    left_proj, left_gate_raw = left_comb.split([hidden, hidden], dim=1)
    left_gate = torch.sigmoid(left_gate_raw)

    # Right
    right_comb = torch.nn.functional.linear(x_flat, right_proj_gate_w)
    right_proj, right_gate_raw = right_comb.split([hidden, hidden], dim=1)
    right_gate = torch.sigmoid(right_gate_raw)

    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate_raw = torch.nn.functional.linear(x_flat, out_gate_w)  # (M, hidden)
    out_gate = torch.sigmoid(out_gate_raw).view(B, N, N, hidden)

    if not nomask:

        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16

        right = right_proj * right_gate

    right = right.view(B, N, N, hidden)

    out = torch.empty_like(left)  # also (B,N,N,hidden)

    left_stride = left.stride()
    right_stride = right.stride()
    out_stride = out.stride()

"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation

Algorithm
---------
1. LayerNorm on the input (FP16) – same as the official module.
2. Fuse each linear projection + gating pair:
   * left_proj + left_gate  →  weight matrix of shape (2·hidden, dim)
   * right_proj + right_gate → same layout.
   The linear layer is executed once per side, then the result is split
   and a sigmoid is applied to the gate part.
3. (Optional) apply the pairwise mask after gating (element‑wise, FP16).
4. Compute the dominant O(N³) contraction  
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]  
   with a single Triton GEMM kernel.  The batch dimension and the hidden
   channel are fused into a third program‑id dimension, so the kernel
   launches once (grid: tiles over i, j, and B·hidden).
5. LayerNorm over the hidden dimension, multiply by the out‑gate,
   and a final linear projection back to `dim`.
6. Cast the result to float32 (the expected output type).

The kernel uses FP16 (Tensor‑Core) arithmetic and fully exploits the
H100 Tensor‑Core pipelines, while the linear layers and LayerNorms are
performed with PyTorch’s highly‑optimized kernels.  The mask‑gate‑fusion
for the linear layers reduces the number of GEMM calls from four to three,
and the custom Triton contraction removes the overhead of many cuBLAS
batched launches.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: compute out = left @ rightᵀ per (batch, hidden) slice
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)                # tile over i (rows)
    pid_n = tl.program_id(1)                # tile over j (cols)
    pid_z = tl.program_id(2)                # batch*hidden index

    # ------------------------------------------------------------------
    # 0️⃣ decode batch and hidden slice
    # ------------------------------------------------------------------
    b = pid_z // H
    h = pid_z % H

    # ------------------------------------------------------------------
    # 1️⃣ block offsets for i, j, k
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # 2️⃣ base pointers for the current (b, h) slice
    # ------------------------------------------------------------------
    off_b_left  = b * stride_left_b
    off_b_right = b * stride_right_b
    off_b_out   = b * stride_out_b

    off_h_left  = h * stride_left_h
    off_h_out   = h * stride_out_h

    # ------------------------------------------------------------------
    # 3️⃣ accumulator
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # ------------------------------------------------------------------
    # 4️⃣ reduction over k
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)      # [BLOCK_K]
        mask_k = offs_k < N

            left_ptr

            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k

            mask=mask_i[:, None] & mask_k[None, :],

            other=0.0,

        # accumulate the block matrix product
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N)

    # 5️⃣ write out the tile
    # ------------------------------------------------------------------
    tl.store(

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul module (no gradients).

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
            - weights_dict : mapping from module parameter names to tensors (float32)
            - config_dict  : must contain ``dim`` and ``hidden_dim``.
                             Optional ``nomask`` bool (default True).

    Returns:
        Tensor of shape [B, N, N, dim] (float32) on the same device as input.
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP16)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16).to(device)
    norm_bias   = weights["norm.bias"].to(torch.float16).to(device)

    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    ).to(torch.float16)            # [B, N, N, dim] FP16

    B, N, _, _ = x.shape
    M = B * N * N                    # flatten batch‑pair count

    # --------------------------------------------------------------
    # 2️⃣ Concatenate projection & gate weights (FP16)
    # --------------------------------------------------------------
    left_w  = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16).to(device)      # (2*hidden, dim)

    right_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16).to(device)      # (2*hidden, dim)

    out_gate_w = weights["out_gate.weight"].to(torch.float16).to(device)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16).to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16).to(device)
    to_out_w = weights["to_out.weight"].to(torch.float16).to(device)   # (dim, hidden)

    # --------------------------------------------------------------
    # 3️⃣ Linear projection + gate (single‑pass per side)
    # --------------------------------------------------------------
    x_flat = x.view(-1, dim)                     # (M, dim)

    # left side: proj + gate
    left_pair = torch.nn.functional.linear(x_flat, left_w)  # (M, 2*hidden)
    left_proj, left_gate_raw = left_pair.split([hidden_dim, hidden_dim], dim=1)
    left_gate = torch.sigmoid(left_gate_raw)

    # right side: proj + gate
    right_pair = torch.nn.functional.linear(x_flat, right_w)  # (M, 2*hidden)
    right_proj, right_gate_raw = right_pair.split([hidden_dim, hidden_dim], dim=1)
    right_gate = torch.sigmoid(right_gate_raw)

    # out_gate (separate linear)
    out_gate_raw = torch.nn.functional.linear(x_flat, out_gate_w)  # (M, hidden)
    out_gate = torch.sigmoid(out_gate_raw).view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 4️⃣ Apply mask (if any) and combine projection + gate
    # --------------------------------------------------------------
    if not nomask:
        # mask: (B,N,N,1) broadcast over hidden
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left  = (left_proj * left_gate) * mask_fp16
        right = (right_proj * right_gate) * mask_fp16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # reshape to the 4‑D tensor expected by the contraction kernel
    left  = left.view(B, N, N, hidden_dim).contiguous()
    right = right.view(B, N, N, hidden_dim).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ O(N³) contraction – Triton fused batched GEMM
    # --------------------------------------------------------------
    out = torch.empty_like(left)  # (B, N, N, hidden) FP16

    # strides (in element counts)
    sL_b, sL_i, sL_k, sL_h = left.stride()
    sR_b, sR_i, sR_k, sR_h = right.stride()
    sO_b, sO_i, sO_j, sO_h = out.stride()

    # tiling parameters – tuned for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid_z = B * hidden_dim

    _tri_mul_outgoing_kernel[(grid_m, grid_n, grid_z)](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        sL_b,
        sL_i,
        sL_k,
        sL_h,
        sR_b,
        sR_i,
        sR_k,
        sR_h,
        sO_b,
        sO_i,
        sO_j,
        sO_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 7️⃣ Out‑gate + final linear projection
    # --------------------------------------------------------------
    out = out * out_gate
    out_flat = out.view(-1, hidden_dim)
    out = torch.nn.functional.linear(out_flat, to_out_w)   # (M, dim)
    out = out.view(B, N, N, dim)

    # --------------------------------------------------------------
    # 8️⃣ Cast back to FP32
    # --------------------------------------------------------------
    return out.to(torch.float32)
