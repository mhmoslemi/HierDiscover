"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass on H100.

Algorithm
---------
1️⃣  Layer‑norm the input in FP32 (numerical stability).
2️⃣  Cast to FP16 – the heavy part runs in FP16 (tensor‑core friendly).
3️⃣  **Fuse all five linear projections** (left_proj, right_proj,
    left_gate, right_gate, out_gate) into a single matmul:
       fused = x @ W_fused.T
    where W_fused is the concatenation of the five weight matrices.
    This reduces memory traffic because the input `x` is read only once.
4️⃣  Split the fused result, apply `sigmoid` to the three gate tensors,
    broadcast the optional mask (unsqueezed on the hidden dim) and
    fuse the gates into the projections.
5️⃣  Heavy O(N³) contraction:
       out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    which is a batched GEMM C = A @ Bᵀ with A,B ∈ FP16,
    FP32 accumulation and final FP16 store.
    The custom Triton kernel tiles the i‑ and j‑dimensions (128×128)
    and the k‑dimension (64).  For very short sequences (N ≤ 128) we
    fall back to `torch.einsum` to avoid kernel launch overhead.
6️⃣  Second LayerNorm (FP32 for statistics, stored back as FP16).
7️⃣  Gating with the previously computed `out_gate` and final linear
    projection back to the original `dim`.
8️⃣  Cast the result back to FP32 (the harness expects FP32).

All steps are pure‑CUDA, the only host‑side work are a few tensor
re‑shapes and the final cast.  The implementation conforms to the
required `custom_kernel` signature.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B,C are FP16, accumulator FP32)
#   A : [M, N, K]   (M = B * hidden_dim, N = seq_len, K = seq_len)
#   B : [M, N, K]
#   C : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # program‑size constants
    stride_am, stride_ai, stride_ak,   # A strides   (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides   (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides   (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # i‑tile
    pid_n = tl.program_id(1)   # j‑tile
    pid_z = tl.program_id(2)   # batch * hidden index (M)

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # --------------------------------------------------------------
        # Load A tile : (BLOCK_M, BLOCK_K) → A[pid_z, i, k]
        # --------------------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # --------------------------------------------------------------
        # Load B tile : (BLOCK_K, BLOCK_N) → B[pid_z, k, j]
        # --------------------------------------------------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Mixed‑precision dot (fp16×fp16 → fp32)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write C tile (fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )

# ----------------------------------------------------------------------
# Helper: batched GEMM with transposition (A @ Bᵀ) for the TriMul contraction.
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using Triton.
    left/right : torch.Tensor [B, N, N, H] (fp16, contiguous)
    Returns    : torch.Tensor [B, N, N, H] (fp16, contiguous)
    """
    B, N, _, H = left.shape

    # pack batch+head into a single leading dimension: [B·H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A, dtype=torch.float16)

    # stride values in elements (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()   # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()  # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()   # (N*N, N, 1)

    # tiling heuristics
    if N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    # restore layout [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

# ----------------------------------------------------------------------
# Triton‑accelerated outgoing TriMul forward pass (no gradients)
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]     (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the rest of the pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear projections (no bias) into a single matmul
    # ------------------------------------------------------------------
    # Gather the five weight matrices (all shape [hidden_dim, dim])
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    # Concatenate on the output‑feature dimension: (5*hidden_dim, dim)
    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    ).contiguous()

    # Single fused linear (FP16 matmul)
    fused = torch.nn.functional.linear(x, fused_weight)  # [B, N, N, 5*hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ Split, sigmoids, mask, and gate‑fusion
    # ------------------------------------------------------------------
    left_proj, left_gate_pre, right_proj, right_gate_pre, out_gate_pre = torch.split(
        fused, hidden_dim, dim=-1
    )

    # Sigmoid gates (still FP16)
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)   # shape [B,N,N,hidden_dim]

    # Optional mask (broadcast on hidden dim)
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # Fuse gating into the projections (in‑place, reduces temporaries)
    left_proj.mul_(left_gate)   # now `left_proj` holds the gated tensor
    right_proj.mul_(right_gate)

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via Triton (fp16 in, fp16 out)
    # ------------------------------------------------------------------
    B, N, _, _ = left_proj.shape
    if N <= 128:
        # Very short sequences – avoid kernel launch overhead.
        out = torch.einsum("b i k h, b j k h -> b i j h", left_proj, right_proj)
    else:
        out = _batched_matmul_transpose_fp16(left_proj, right_proj)   # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm.  Use FP32 statistics for better accuracy.
    # ------------------------------------------------------------------
    # Cast to FP32, apply LayerNorm, cast back to FP16
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float32)

    out = out.to(torch.float32)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out.to(torch.float16)

    # ------------------------------------------------------------------
    # 7️⃣ Gating with `out_gate` and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to float32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
