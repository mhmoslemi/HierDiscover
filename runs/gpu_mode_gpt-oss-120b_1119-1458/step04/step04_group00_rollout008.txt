def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', True)
    device = input_tensor.device

    # first layernorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias).to(torch.float16)  # (B,N,N,dim)

    B,N,_,_ = x.shape # assuming shape B,N,N,dim

    # flatten per pair
    M = B * N * N
    x_flat = x.view(M, dim)   # (M, dim)

    # Cast all required weights to fp16
    left_proj_weight = weights["left_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_proj_weight = weights["right_proj.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # combine left/right proj and gate weights for single matmul
    combined_weight = torch.cat([
        left_proj_weight,
        left_gate_weight,
        right_proj_weight,
        right_gate_weight,
    ], dim=0).to(torch.float16)

    # compute combined projection in FP16
    combined_out = torch.nn.functional.linear(x_flat, combined_weight)  # (M, 4 * hidden_dim)

    # split
    hidden = hidden_dim
    h_proj_start = 0
    left_proj_raw = combined_out[:, h_proj_start:h_proj_start+hidden]  # (M, hidden)
    left_gate_raw = combined_out[:, hidden:2*hidden]
    right_proj_raw = combined_out[:, 2*hidden:3*hidden]
    right_gate_raw = combined_out[:, 3*hidden:4*hidden]

    # apply sigmoid to gates
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # out gate linear
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, out_gate_weight.to(torch.float16)))  # (M, hidden)
    out_gate = out_gate.view(B, N, N, hidden)   # (B,N,N,hidden)

    # optional mask fuse
    if not nomask:
        # mask flatten to (M,)
        mask_flat = mask_tensor.to(torch.float16).view(M)  # (M,)
        # allocate outputs
        left = torch.empty_like(left_proj_raw)
        right = torch.empty_like(right_proj_raw)

        # Strides
        # left_proj_raw stride
        stride_lp_m, stride_lp_h = left_proj_raw.stride()
        stride_lg_m, stride_lg_h = left_gate.stride()
        stride_rp_m, stride_rp_h = right_proj_raw.stride()
        stride_rg_m, stride_rg_h = right_gate.stride()
        stride_mask = mask_flat.stride()[0]
        stride_o_m, stride_o_h = left.stride()   # same as left_proj_raw

        BLOCK_M = 64
        BLOCK_H = hidden  # tile along hidden dimension

        grid = ( (M + BLOCK_M - 1) // BLOCK_M, )

        # left = left_proj_raw * left_gate * mask
        _mask_gate_fuse_kernel[grid](
            left_proj_raw, left_gate, mask_flat, left,
            M, hidden,
            stride_lp_m, stride_lp_h,
            stride_lg_m, stride_lg_h,
            stride_o_m, stride_o_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
        # right = right_proj_raw * right_gate * mask
        _mask_gate_fuse_kernel[grid](
            right_proj_raw, right_gate, mask_flat, right,
            M, hidden,
            stride_rp_m, stride_rp_h,
            stride_rg_m, stride_rg_h,
            stride_o_m, stride_o_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
        # reshape
        left = left.view(B, N, N, hidden)
        right = right.view(B, N, N, hidden)
    else:
        # no mask: simple elementwise multiplication
        left = (left_proj_raw * left_gate).view(B, N, N, hidden)
        right = (right_proj_raw * right_gate).view(B, N, N, hidden)

    # O(N^3) contraction: out = left @ right^T across N dimension per batch and channel
    # reshape for batched matmul
    # left shape (B,N,N,hidden) -> (B*hidden, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    # batched gemm: (B*hidden, N, N) x (B*hidden, N, N)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*hidden, N, N)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # second LayerNorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_weight.to(torch.float16),
        bias=to_out_norm_bias.to(torch.float16),
    )
    # output gate multiplication
    out = out * out_gate

    # final linear projection to dim
    out_flat = out.view(M, hidden)  # (M, hidden)
    out_proj = torch.nn.functional.linear(out_flat, to_out_weight.to(torch.float16)) # (M, dim)
    out_final = out_proj.view(B, N, N, dim)

    # cast to fp32
    return out_final.to(torch.float32)

@triton.jit
def _mask_gate_fuse_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    M, H,
    stride_proj_m, stride_proj_h,
    stride_gate_m, stride_gate_h,
    stride_out_m, stride_out_h,
    stride_mask,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = tl.arange(0, BLOCK_H)

    mask = tl.load(mask_ptr + offs_m * stride_mask,
                   mask=offs_m < M, other=0.0).to(tl.float16)
    mask = mask[:, None]  # broadcast

    proj = tl.load(
        proj_ptr + offs_m[:, None] * stride_proj_m + offs_h[None, :] * stride_proj_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0)
    gate = tl.load(
        gate_ptr + offs_m[:, None] * stride_gate_m + offs_h[None, :] * stride_gate_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0)

    out = proj * gate * mask
    tl.store(out_ptr + offs_m[:, None] * stride_out_m + offs_h[None, :] * stride_out_h,
             out,
             mask=(offs_m[:, None] < M) & (offs_h[None, :] < H))

"""
Outgoing TriMul (AlphaFold3) – fast forward implementation

The implementation follows the reference PyTorch module but replaces the
five separate Linear layers (proj + gate + out‑gate) by a single fused
matrix multiplication whose weight matrix stacks all five linear maps.
Sigmoid gates are applied in PyTorch, while the optional mask multiplication
is fused with the projection & gate using a tiny Triton kernel (required
by the task statement).  The dominant O(N³) contraction is performed with
a batched GEMM (`torch.bmm`), which on an H100 exploits Tensor‑Core
pipelines.  All heavy arithmetic is carried out in FP16, the final
result is cast back to FP32.

The kernel `_mask_gate_fuse_kernel` multiplies a projection tensor, a
gate tensor (already passed through `sigmoid`), and a per‑pair mask in a
single pass:
    out = proj * gate * mask[...,None]
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _mask_gate_fuse_kernel(
    proj_ptr,      # *[M, H] projection
    gate_ptr,      # *[M, H] (sigmoid‑ed) gate
    mask_ptr,      # *[M]   mask (0/1)
    out_ptr,       # *[M, H] output
    M, H,          # problem size
    stride_proj_m, stride_proj_h,   # strides for proj
    stride_gate_m, stride_gate_h,   # strides for gate
    stride_out_m,  stride_out_h,    # strides for out
    stride_mask,                     # stride for mask (1‑D)
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_h = tl.arange(0, BLOCK_H)                  # [BLOCK_H]

    # Load mask (broadcast to H)
    mask = tl.load(mask_ptr + offs_m * stride_mask,
                   mask=offs_m < M, other=0.0).to(tl.float16)   # [BLOCK_M]
    mask = mask[:, None]                                          # [BLOCK_M,1]

    # Load projection and gate tiles
    proj = tl.load(
        proj_ptr + offs_m[:, None] * stride_proj_m
                  + offs_h[None, :] * stride_proj_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    gate = tl.load(
        gate_ptr + offs_m[:, None] * stride_gate_m
                  + offs_h[None, :] * stride_gate_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    # Element‑wise fusion
    out = proj * gate * mask

    # Store result
    tl.store(
        out_ptr + offs_m[:, None] * stride_out_m
                 + offs_h[None, :] * stride_out_h,
        out,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
    )


def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input [B, N, N, dim] (float32)
        torch.Tensor,                     # mask  [B, N, N]   (bool/float)
        Dict[str, torch.Tensor],          # weights dict
        Dict[str, Any]                    # config dict
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Returns a tensor of shape [B, N, N, dim] in float32.
    """
    # -------------------------------------------------------------
    # 0️⃣ unpack arguments
    # -------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------------------
    # 1️⃣ first LayerNorm (float32 → FP16)
    # -------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # LayerNorm in FP16 – weight/bias are kept in FP16 for speed
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    ).to(torch.float16)               # (B, N, N, dim) FP16

    B, N, _, _ = x.shape
    M = B * N * N                       # flattened pair count

    # -------------------------------------------------------------
    # 2️⃣ Cast linear weights to FP16 once
    # -------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # Stack all five linear maps (proj, gate, proj, gate, out_gate)
    stacked_weight = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(torch.float16)                 # (5*hidden, dim)

    # -------------------------------------------------------------
    # 3️⃣ Combined linear projection (FP16)
    # -------------------------------------------------------------
    x_flat = x.view(M, dim)             # (M, dim)
    combined_out = torch.nn.functional.linear(x_flat, stacked_weight)  # (M, 5*hidden)

    # -------------------------------------------------------------
    # 4️⃣ Slice & apply sigmoid gates
    # -------------------------------------------------------------
    h = hidden
    left_proj_raw = combined_out[:, 0 * h : 1 * h]          # (M, h)
    left_gate_raw = combined_out[:, 1 * h : 2 * h]          # (M, h)
    right_proj_raw = combined_out[:, 2 * h : 3 * h]         # (M, h)
    right_gate_raw = combined_out[:, 3 * h : 4 * h]         # (M, h)
    out_gate_raw = combined_out[:, 4 * h : 5 * h]           # (M, h)

    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw).view(B, N, N, h)

    # -------------------------------------------------------------
    # 5️⃣ Optional mask‑fused elementwise multiply (Triton)
    # -------------------------------------------------------------
    if not nomask:
        # mask → FP16 and flatten to (M,)
        mask_flat = mask_tensor.to(torch.float16).view(M)

        # allocate destination buffers
        left  = torch.empty_like(left_proj_raw)
        right = torch.empty_like(right_proj_raw)

        # strides (in element counts)
        stride_lp_m, stride_lp_h = left_proj_raw.stride()
        stride_lg_m, stride_lg_h = left_gate.stride()
        stride_rp_m, stride_rp_h = right_proj_raw.stride()
        stride_rg_m, stride_rg_h = right_gate.stride()
        stride_out_m, stride_out_h = left.stride()          # same for left/right
        stride_mask = mask_flat.stride()[0]

        BLOCK_M = 64                                 # rows per block
        BLOCK_H = h                                   # tile hidden dim

        grid = ((M + BLOCK_M - 1) // BLOCK_M,)

        # left = left_proj * left_gate * mask
        _mask_gate_fuse_kernel[grid](
            left_proj_raw, left_gate, mask_flat, left,
            M, h,
            stride_lp_m, stride_lp_h,
            stride_lg_m, stride_lg_h,
            stride_out_m, stride_out_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
        # right = right_proj * right_gate * mask
        _mask_gate_fuse_kernel[grid](
            right_proj_raw, right_gate, mask_flat, right,
            M, h,
            stride_rp_m, stride_rp_h,
            stride_rg_m, stride_rg_h,
            stride_out_m, stride_out_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
        # reshape to (B, N, N, h)
        left  = left.view(B, N, N, h)
        right = right.view(B, N, N, h)
    else:
        # No mask – simple element‑wise fuse in PyTorch
        left  = (left_proj_raw * left_gate).view(B, N, N, h)
        right = (right_proj_raw * right_gate).view(B, N, N, h)

    # -------------------------------------------------------------
    # 6️⃣ O(N³) contraction via batched GEMM (left @ rightᵀ)
    # -------------------------------------------------------------
    # reshape to (B*hidden, N, N) for batched matmul
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * h, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * h, N, N)

    # batch matrix multiplication: (B*h, N, N) x (B*h, N, N)ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*h, N, N)
    out = out_mat.view(B, h, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,h)

    # -------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # -------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (h,),
        weight=to_out_norm_w.to(torch.float16),
        bias=to_out_norm_b.to(torch.float16),
    )

    # -------------------------------------------------------------
    # 8️⃣ Output gate & final linear projection (FP16)
    # -------------------------------------------------------------
    out = out * out_gate                                            # (B,N,N,h)
    out_flat = out.view(M, h)                                       # (M, h)
    out_proj = torch.nn.functional.linear(
        out_flat,
        to_out_w.to(torch.float16),                                 # (dim, h)
    )                                                               # (M, dim)
    out_fp16 = out_proj.view(B, N, N, dim)                         # (B,N,N,dim)

    # -------------------------------------------------------------
    # 9️⃣ Cast back to FP32 (expected by caller)
    # -------------------------------------------------------------
    return out_fp16.to(torch.float32)
