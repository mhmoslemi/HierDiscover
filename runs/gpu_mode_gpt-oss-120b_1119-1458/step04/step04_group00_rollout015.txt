```python
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH_H, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # Compute batch index and tile i/j
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_blocks_m * num_blocks_n)
    tile = pid % (num_blocks_m * num_blocks_n)
    pid_m = tile // num_blocks_n
    pid_n = tile % num_blocks_n

    # Pointers offsets
    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    off_k = tl.arange(0, BLOCK_K)
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # Loop over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # load A block: shape [BLOCK_M, BLOCK_K]
        a_ptrs = A_ptr + batch*stride_a_batch + off_m[:, None] * stride_a_m + (k_offset + off_k)[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=(off_m[:, None] < M) & (k_offset + off_k)[None, :] < K, other=0.0).to(tl.float16)
        # load B block (transposed for B^T)
        b_ptrs = B_ptr + batch*stride_b_batch + (k_offset + off_k)[:, None] * stride_b_k + off_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=(k_offset + off_k)[:, None] < K & (off_n[None, :] < N), other=0.0).to(tl.float16)
        # compute
        acc += tl.dot(a, b)  # returns float32 accumulation? default returns FP32 in Triton?
        # Advancing pointers automatically by next loop iteration
    # cast to fp16 and store
    c_ptrs = C_ptr + batch*stride_c_batch + off_m[:, None] * stride_c_m + off_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=(off_m[:, None] < M) & (off_n[None, :] < N))

@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # Determine tile indices
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch_idx = pid // (num_blocks_m * num_blocks_n)
    tile_idx = pid % (num_blocks_m * num_blocks_n)
    pid_m = tile_idx // num_blocks_n
    pid_n = tile_idx % num_blocks_n

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # Load A tile (M x K)
        a_ptrs = A_ptr + batch_idx * stride_a_batch + \
                 offs_m[:, None] * stride_a_m + (k_offset + offs_k)[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & ((k_offset + offs_k)[None, :] < K),
                    other=0.0).to(tl.float16)

        # Load B tile (K x N) i.e., transposed B to get (K,N)
        b_ptrs = B_ptr + batch_idx * stride_b_batch + \
                 (k_offset + offs_k)[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=((k_offset + offs_k)[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0).to(tl.float16)

        # dot product
        acc += tl.dot(a, b)

    # Store
    c_ptrs = C_ptr + batch_idx * stride_c_batch + \
             offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))
```

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                           batch, M, N, K,
                           stride_a_batch, stride_a_m, stride_a_k,
                           stride_b_batch, stride_b_k, stride_b_n,
                           stride_c_batch, stride_c_m, stride_c_n,
                           BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # Compute program ID and block coordinates
    pid = tl.program_id(0)
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch_idx = pid // (num_blocks_m * num_blocks_n)
    tile_idx = pid % (num_blocks_m * num_blocks_n)
    pid_m = tile_idx // num_blocks_n
    pid_n = tile_idx % num_blocks_n

    # Compute offsets within the matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]
    offs_k = tl.arange(0, BLOCK_K)                    # [BLOCK_K]

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # Load A tile (M x K)
        a_ptrs = A_ptr + batch_idx * stride_a_batch \
                + offs_m[:, None] * stride_a_m \
                + (k_offset + offs_k)[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & ((k_offset + offs_k)[None, :] < K),
                    other=0.0).to(tl.float16)

        # Load B tile for B^T (K x N). B is stored as (batch, N, K) – we treat
        # rows = k (original col) and columns = n (original row).
        b_ptrs = B_ptr + batch_idx * stride_b_batch \
                + (k_offset + offs_k)[:, None] * stride_b_k \
                + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=((k_offset + offs_k)[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0).to(tl.float16)

        # Compute dot product
        acc += tl.dot(a, b)     # FP32 accumulation

    # Write back to C
    c_ptrs = C_ptr + batch_idx * stride_c_batch \
            + offs_m[:, None] * stride_c_m \
            + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Outgoing TriMul forward pass.
    The O(N³) contraction is performed with a custom Triton batched GEMM kernel.
    All heavy arithmetic runs in FP16; the final result is returned in FP32.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ---------- 0. Load weights (FP16) ----------
    norm_weight = weights["norm.weight"].to(device).to(torch.float16)
    norm_bias = weights["norm.bias"].to(device).to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)

    # ---------- 1. First LayerNorm ----------
    # torch.layer_norm works in the dtype of the input; we compute in FP32 then cast
    x_norm = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    ).to(torch.float16)   # shape [B,N,N,dim]

    B, N, _, _ = x_norm.shape
    M = B * N * N           # number of position pairs

    # ---------- 2. Linear projection + gating ----------
    # flatten spatial dimensions
    x_flat = x_norm.view(-1, dim)   # (M, dim)

    # Combined linear for left/right projections and gates
    # Concatenate the four weight matrices to reduce the number of matmuls
    proj_gate_weight = torch.cat([left_proj_w,
                                  left_gate_w,
                                  right_proj_w,
                                  right_gate_w], dim=0)   # (4*hidden, dim)

    proj_gate = torch.nn.functional.linear(x_flat, proj_gate_weight)  # (M, 4*hidden)

    # Slice results
    h = hidden_dim
    left_proj = proj_gate[:, :h]
    left_gate_raw = proj_gate[:, h:2*h]
    right_proj = proj_gate[:, 2*h:3*h]
    right_gate_raw = proj_gate[:, 3*h:4*h]

    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # Apply mask if present
    if not nomask:
        # mask shape (B,N,N) -> flatten to (M,)
        mask_flat = mask_tensor.to(torch.float16).view(-1)   # (M,)
        # broadcast mask to hidden dimension
        mask_exp = mask_flat[:, None]    # (M,1)
        left = left_proj * left_gate * mask_exp
        right = right_proj * right_gate * mask_exp
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Reshape back to 4-D tensors
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # ---------- 3. Out Gate ----------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, out_gate_w)).view(B, N, N, hidden_dim)

    # ---------- 4. O(N³) contraction (batched GEMM) ----------
    # Permute to (B, hidden, N, N) and merge batch*hidden for gemm
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # (BH, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N) # (BH, N, N)

    # Allocate output
    out_mat = torch.empty_like(left_mat)   # same dtype (fp16)

    # Gather strides for kernel
    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_m, stride_b_k = right_mat.stride()   # m= row (N), k=col (N)
    # For B we need stride for dimension k (original col) and n (row after transpose)
    # right_mat has stride: batch, row (N), col (1)
    stride_b_batch = right_mat.stride(0)
    stride_b_k = right_mat.stride(2)   # column stride (1)
    stride_b_n = right_mat.stride(1)   # row stride (N)

    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()

    batch_hidden = B * hidden_dim
    # Choose block sizes (tuned for H100, FP16)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute grid size
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch_hidden * num_blocks_m * num_blocks_n
    grid = (total_blocks,)

    # Launch Triton kernel
    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch_hidden,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # Reshape back to original layout: (B,N,N,hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- 5. Second LayerNorm ----------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ---------- 6. Apply out gate ----------
    out = out * out_gate

    # ---------- 7. Final projection ----------
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim) in FP16

    # ---------- 8. Cast back to FP32 ----------
    return out.to(torch.float32)

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³) contraction.
#   For each (batch * hidden) pair we compute
#       C = A @ Bᵀ
#   where A and B are (N, N) FP16 matrices.  The kernel uses
#   FP16 loads + TensorCore dot (accumulated in FP32) and writes
#   the result back as FP16.
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                     # pointers
    batch, M, N, K,                          # dimensions
    stride_a_batch, stride_a_m, stride_a_k,  # strides for A
    stride_b_batch, stride_b_k, stride_b_n,  # strides for B (used as Bᵀ)
    stride_c_batch, stride_c_m, stride_c_n,  # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C[batch, i, j] = Σₖ A[batch, i, k] * B[batch, j, k]
    (i.e. A @ Bᵀ) for all batches in one launch.
    """
    pid = tl.program_id(0)

    # 1️⃣ 2‑D tiling within each batch
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch_idx = pid // (num_blocks_m * num_blocks_n)               # which GEMM
    tile_idx  = pid %  (num_blocks_m * num_blocks_n)               # which tile
    pid_m = tile_idx // num_blocks_n
    pid_n = tile_idx %  num_blocks_n

    # 2️⃣ Offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # rows of A / C
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # cols of Bᵀ / C
    offs_k = tl.arange(0, BLOCK_K)                    # reduction axis

    # 3️⃣ Accumulator (FP32 for TensorCore)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # 4️⃣ Loop over K dimension in BLOCK_K steps
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # Load A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + batch_idx * stride_a_batch \
                 + offs_m[:, None] * stride_a_m \
                 + (k_offset + offs_k)[None, :] * stride_a_k
        a = tl.load(
            a_ptrs,
            mask=(offs_m[:, None] < M) & ((k_offset + offs_k)[None, :] < K),
            other=0.0,
        ).to(tl.float16)

        # Load B tile transposed: we need Bᵀ, i.e. rows = k, cols = j.
        # B is stored as (batch, N, K) → stride_b_k = 1, stride_b_n = N.
        b_ptrs = B_ptr + batch_idx * stride_b_batch \
                 + (k_offset + offs_k)[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n
        b = tl.load(
            b_ptrs,
            mask=((k_offset + offs_k)[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        ).to(tl.float16)

        # Tensor‑core dot (accumulates in FP32)
        acc += tl.dot(a, b)

    # 5️⃣ Write back C tile (FP16)
    c_ptrs = C_ptr + batch_idx * stride_c_batch \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module.

    - All heavy arithmetic (projections, gating, O(N³) contraction)
      is performed in FP16 to exploit Tensor‑Core throughput.
    - The O(N³) contraction is realized as a custom Triton batched
      GEMM kernel (`_batched_matmul_kernel`) that computes
      ``out = left @ rightᵀ`` for every hidden channel in a single launch.
    - Optional masking is applied element‑wise with a cheap broadcast.
    - The final output is returned in FP32 as required by the caller.
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ Load model weights and cast to FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device).to(torch.float16)
    norm_bias   = weights["norm.bias"].to(device).to(torch.float16)

    left_proj_w   = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device).to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device).to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(device).to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ First LayerNorm (shape: [B, N, N, dim])
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    ).to(torch.float16)                     # FP16 for the rest

    B, N, _, _ = x.shape
    M = B * N * N                            # flattened pair count

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections + gating (single fused matmul for four linear ops)
    # ------------------------------------------------------------------
    x_flat = x.view(-1, dim)                 # (M, dim)

    # Concatenate the four weight matrices to perform one matmul instead of four.
    # Order: left_proj, left_gate, right_proj, right_gate
    proj_gate_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )                                        # (4*hidden, dim)

    proj_gate = torch.nn.functional.linear(x_flat, proj_gate_w)   # (M, 4*hidden)

    h = hidden_dim
    left_proj      = proj_gate[:, :h]
    left_gate_raw  = proj_gate[:, h:2 * h]
    right_proj     = proj_gate[:, 2 * h:3 * h]
    right_gate_raw = proj_gate[:, 3 * h:4 * h]

    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    if not nomask:
        # mask shape: (B, N, N) → (M,)
        mask_flat = mask_tensor.to(torch.float16).view(-1)          # (M,)
        mask_exp  = mask_flat[:, None]                             # (M,1) broadcast
        left  = left_proj * left_gate * mask_exp
        right = right_proj * right_gate * mask_exp
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # Reshape back to 4‑D tensors
    left  = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Out‑gate (independent linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, out_gate_w)
    ).view(B, N, N, hidden_dim)   # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 5️⃣ O(N³) contraction using the Triton batched GEMM kernel
    # ------------------------------------------------------------------
    # Shape required for the kernel: (batch*hidden, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.empty_like(left_mat)   # FP16 output buffer

    # Strides (in element counts)
    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()    # (BH, N, 1)
    stride_b_batch = right_mat.stride(0)                         # (BH, N, N)
    stride_b_k = right_mat.stride(2)   # col‑stride = 1
    stride_b_n = right_mat.stride(1)   # row‑stride = N
    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()

    batch_hidden = B * hidden_dim

    # Tile sizes – tuned for H100 FP16 Tensor Core execution
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    n_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    n_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch_hidden * n_blocks_m * n_blocks_n
    grid = (total_blocks,)

    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch_hidden,
        N,                      # M
        N,                      # N
        N,                      # K
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # Reshape back to (B, N, N, hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (over hidden dimension)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Apply out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, dim) FP16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 (expected by the caller)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
