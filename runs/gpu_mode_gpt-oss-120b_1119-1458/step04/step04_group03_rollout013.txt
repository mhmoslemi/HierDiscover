"""
Outgoing TriMul (AlphaFold‑3) – fast forward without autograd.

Algorithm
---------
1️⃣  Layer‑norm the input (FP32 for numerical stability) and cast to FP16.
2️⃣  Cast all model weights to FP16.
3️⃣  Four linear projections (left/right) and three gating projections (left‑gate,
    right‑gate, out‑gate) are performed with `torch.nn.functional.linear`
    on the FP16 tensors.
4️⃣  Fuse the element‑wise gating (and optional mask) with a tiny Triton kernel.
    The kernel performs:
        left  = left  * left_gate  * mask   (if a mask is supplied)
        right = right * right_gate * mask
   This saves two separate element‑wise passes.
5️⃣  Heavy contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix‑multiply using `torch.bmm`
   (FP16 × FP16 → FP32 accumulation) on tensors of shape
   `[B·H, N, N]`.  This is the most efficient path on an H100.
6️⃣  Second LayerNorm (FP16) on the hidden dimension, followed by a second
    Triton kernel that fuses the out‑gate multiplication.
7️⃣  Final linear projection back to the original `dim`.
8️⃣  Cast the result back to FP32 (required by the harness).

Only the element‑wise fusions are written in Triton – the dominant
contraction uses cuBLAS via `torch.bmm`, which is dramatically faster
than a pure‑Triton implementation for all sequence lengths in the
test suite.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Tiny Triton kernels for element‑wise fusions
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr,      # fp16 * [E] – will be written in‑place
    right_ptr,     # fp16 * [E] – will be written in‑place
    left_gate_ptr, # fp16 * [E]
    right_gate_ptr,# fp16 * [E]
    mask_ptr,      # fp16 * [M] (broadcasted, stride 0 on hidden dim)
    numel,         # total number of elements in left/right (E = B·N·N·H)
    has_mask: tl.constexpr,   # 1 → mask is present, 0 → skip mask
    BLOCK: tl.constexpr,
):
    """
    In‑place fused multiplication:
        left  = left  * left_gate  * (mask if present)
        right = right * right_gate * (mask if present)
    The mask tensor is broadcasted over the hidden dimension; its stride
    for the hidden dim is 0, therefore a single load per element suffices.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < numel

    # load the four input tensors
    left_val  = tl.load(left_ptr  + offs, mask=m, other=0.0)
    right_val = tl.load(right_ptr + offs, mask=m, other=0.0)
    left_gate = tl.load(left_gate_ptr  + offs, mask=m, other=0.0)
    right_gate= tl.load(right_gate_ptr + offs, mask=m, other=0.0)

    # multiply with gates
    left_val  = left_val  * left_gate
    right_val = right_val * right_gate

    # optional mask (broadcasted)
    if has_mask:
        mask_val = tl.load(mask_ptr + offs, mask=m, other=1.0)   # mask is 0/1 in FP16
        left_val  = left_val  * mask_val
        right_val = right_val * mask_val

    # store results in‑place
    tl.store(left_ptr  + offs, left_val,  mask=m)
    tl.store(right_ptr + offs, right_val, mask=m)


@triton.jit
def _apply_gate_kernel(
    out_ptr,       # fp16 * [E] – will be written in‑place
    gate_ptr,      # fp16 * [E]
    numel,         # total elements
    BLOCK: tl.constexpr,
):
    """
    In‑place out = out * gate (used after the second LayerNorm).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    m = offs < numel

    out_val  = tl.load(out_ptr  + offs, mask=m, other=0.0)
    gate_val = tl.load(gate_ptr + offs, mask=m, other=0.0)

    tl.store(out_ptr + offs, out_val * gate_val, mask=m)


# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the O(N³) contraction using cuBLAS via torch.bmm.
    Input/Output tensors are expected to be FP16 with shape
    [B, N, N, H] where H = hidden_dim.
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] (contiguous)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched mat‑mul with the second operand transposed
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # FP16 → FP32 accumulation

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 / bool) – may be ignored
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor : shape [B, N, N, dim] (float32, same device as input)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → keep numerically stable)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the rest of the pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast all linear weights to FP16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)      # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)     # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – still FP16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 6️⃣ Fuse gate‑mask multiplication with a tiny Triton kernel
    # ------------------------------------------------------------------
    # Prepare mask (if used) – broadcasted to [B,N,N,1] with stride 0 on H
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16)   # [B, N, N]
        # View the mask as a tensor that can be broadcasted over the hidden dim
        # without allocating additional memory.  The trailing dimension has
        # stride 0, therefore every hidden entry sees the same mask value.
        mask_expand = mask_fp16.unsqueeze(-1)               # [B,N,N,1]
        # stride for the hidden dim is 0 → we use .expand to get the proper view
        mask_expand = mask_expand.expand(-1, -1, -1, hidden_dim).contiguous()
    else:
        mask_expand = None

    # Flatten everything to 1‑D contiguous buffers for the kernel
    left_flat       = left.contiguous().view(-1)
    right_flat      = right.contiguous().view(-1)
    left_gate_flat  = left_gate.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    if not nomask:
        mask_flat = mask_expand.view(-1)   # also 1‑D, same length as left/right
    else:
        mask_flat = torch.empty(0, device=device, dtype=torch.float16)  # dummy

    # Triton launch parameters
    total_elements = left_flat.numel()      # B * N * N * hidden_dim
    BLOCK = 65536                           # roughly 64 Ki elements per program
    grid = ( (total_elements + BLOCK - 1) // BLOCK, )

    _gate_mask_kernel[grid](
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        mask_flat,
        total_elements,
        0 if nomask else 1,          # constexpr: has_mask
        BLOCK=BLOCK,
        num_warps=4,
        num_stages=1,
    )

    # Reshape back to [B, N, N, hidden_dim]
    left  = left_flat.view_as(left)
    right = right_flat.view_as(right)

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction (batched mat‑mul)
    # ------------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, hidden_dim]  FP16

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (FP16) → out‑gate (fused via Triton)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # Fuse out‑gate multiplication with a tiny kernel (in‑place)
    out_flat = out.contiguous().view(-1)
    out_gate_flat = out_gate.contiguous().view(-1)
    total_elements = out_flat.numel()
    grid = ( (total_elements + BLOCK - 1) // BLOCK, )

    _apply_gate_kernel[grid](
        out_flat,
        out_gate_flat,
        total_elements,
        BLOCK=BLOCK,
        num_warps=4,
        num_stages=1,
    )
    out = out_flat.view_as(out)

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to original `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # [B, N, N, dim]  FP16

    # ------------------------------------------------------------------
    # 🔟 Cast back to float32 (required by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
