"""
TriMul (outgoing) – fast forward implementation
------------------------------------------------
The operation is defined on a 4‑D tensor X ∈ ℝ^{B×N×N×C}.

1️⃣ Layer‑norm the input in fp32 → cast to fp16.
2️⃣ Two linear projections (left/right) and three gating
   linear layers (left_gate, right_gate, out_gate) are computed
   in fp16.  Gating uses a sigmoid.
3️⃣ When a mask is present we fuse the element‑wise multiplication
   x←x·gate·mask with a small Triton kernel (mask is broadcast over the
   hidden dimension without materialising a large copy).
   When there is no mask we fuse just x←x·gate with an in‑place PyTorch
   multiplication.
4️⃣ The heavy O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched half‑precision GEMM (C = A @ Bᵀ) using
   `torch.bmm`, which invokes highly‑optimized Tensor‑Core kernels on
   H100.  For very short sequences an alternative Triton GEMM kernel is
   provided, but the cuBLAS path is the default for N > 512.
5️⃣ A second Layer‑norm over the hidden dimension, an output‑gate
   (sigmoid) and a final linear projection back to C are applied in fp16.
6️⃣ The result is cast back to fp32.

Only the forward pass is required – no autograd bookkeeping.
A Triton kernel is used for the mask‑gate fusion (or pure gate fusion
when a mask is supplied), satisfying the “must use a kernel” requirement.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse (tensor * gate * mask) in‑place.
# Supports broadcasting of mask over the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def fuse_gate_mask_kernel(
    x_ptr,          # *fp16   – tensor to be updated, shape (size,)
    gate_ptr,       # *fp16   – gate tensor, same shape as x
    mask_ptr,       # *fp16   – mask tensor, shape (B,N,N) – broadcasted
    size,           # i32     – total number of elements in x/gate
    hidden_dim,     # i32     – size of the last (hidden) dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    # Load the two tensors
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Compute the index for the mask (broadcasted over hidden_dim)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # default mask = 1

    # Fused multiply
    out = x * g * m

    # Store result back in-place
    tl.store(x_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched half‑precision GEMM using cuBLAS (torch.bmm)
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """Perform out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using torch.bmm.

    left/right: fp16 tensors of shape [B, N, N, H]
    Returns: fp16 tensor of shape [B, N, N, H]
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] (A) and [B*H, N, N] (B)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # batched matmul: (B*H, N, N) @ (B*H, N, N)^T
    c = torch.bmm(a, b.transpose(1, 2))          # [B*H, N, N]
    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Optional Triton GEMM for very small N (≤ 512). Kept for completeness.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # i‑tile
    pid_n = tl.program_id(1)   # j‑tile
    pid_z = tl.program_id(2)   # batch*head tile

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A[i,k]
        a_ptrs = (A_ptr
                  + pid_z * stride_am
                  + offs_i[:, None] * stride_ai
                  + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B[j,k] (transposed)
        b_ptrs = (B_ptr
                  + pid_z * stride_bm
                  + offs_k[:, None] * stride_bk
                  + offs_j[None, :] * stride_bj)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)

    c_ptrs = (C_ptr
              + pid_z * stride_cm
              + offs_i[:, None] * stride_ci
              + offs_j[None, :] * stride_cj)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """Batched GEMM via the custom Triton kernel (used only for N ≤ 512)."""
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.empty_like(A, dtype=torch.float16)

    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    BLOCK_M = BLOCK_N = 64 if N <= 256 else 128
    BLOCK_K = 64

    grid = ((N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
            B * H)

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ---------------------------------------------------------------
    # 3️⃣ Linear projections (half‑precision)
    # ---------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)    # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,hidden]

    # ---------------------------------------------------------------
    # 4️⃣ Gating signals (half‑precision)
    # ---------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ---------------------------------------------------------------
    # 5️⃣ Mask & gate fusion (Triton kernel) – required when mask present
    # ---------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → fp16, keep original layout
        mask_fp16 = mask_tensor.to(torch.float16)
        size = left.numel()
        BLOCK = 1024  # ≤ 1024 threads per block
        grid = ((size + BLOCK - 1) // BLOCK,)

        # In‑place fusion: left ← left * left_gate * mask
        fuse_gate_mask_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            size,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # In‑place fusion: right ← right * right_gate * mask
        fuse_gate_mask_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            size,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # free temporary tensors to keep memory pressure low
        del left_gate, right_gate, mask_fp16
    else:
        # No mask – simple in‑place multiplication (gate only)
        left = left.mul_(left_gate)
        right = right.mul_(right_gate)
        del left_gate, right_gate

    # ---------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction – batched GEMM (cuBLAS for N>512)
    # ---------------------------------------------------------------
    B, N, _, _ = left.shape
    if N <= 512:
        # For short sequences the Triton matmul has lower launch overhead
        out = _batched_matmul_triton(left, right)          # fp16, shape [B,N,N,hidden]
    else:
        out = _batched_matmul_torch(left, right)           # fp16, shape [B,N,N,hidden]

    # ---------------------------------------------------------------
    # 7️⃣ Second LayerNorm (hidden dimension)
    # ---------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ---------------------------------------------------------------
    # 8️⃣ Final gating + projection back to dim
    # ---------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)        # fp16 → [B,N,N,dim]

    # ---------------------------------------------------------------
    # 9️⃣ Cast back to float32 for the harness
    # ---------------------------------------------------------------
    return out.to(torch.float32)
```
