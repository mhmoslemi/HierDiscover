def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # Cast weights to half
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Input to fp16
    x = input_tensor.to(torch.float16)
    # First layernorm in fp16
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )
    # Projections
    left = torch.nn.functional.linear(x, left_proj_w)  # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)

    # Masked if needed
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # Gating weights
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))

    # Apply gates (elementwise)
    left = left * left_gate
    right = right * right_gate

    # contraction O(N^3)
    B, N, _, _ = left.shape
    if N <= 256:
        # Use Triton kernel implementation
        out = _batched_matmul_transpose_fp16(left, right)
    else:
        # Use torch.bmm for speed
        # Rearrange to [B*H,N,N]
        left_ = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
        right_ = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

        out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),

    return out.to(torch.float32)

```python
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel for batched GEMM C = A @ B^T
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)  # i tile
    pid_j = tl.program_id(1)  # j tile
    pid_z = tl.program_id(2)  # batch*hidden

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_start in range(0, K, BLOCK_K):
        cur_k = k_start
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )

        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk   # row = k
            + offs_j[None, :] * stride_bj   # col = j
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )

        # fp16 * fp16 -> fp32 accumulation
        acc += tl.dot(a, b)

    # store result
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )

def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a custom Triton kernel. left/right must be fp16 and contiguous.
    Returns fp16 tensor of shape [B,N,N,hidden].
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A, dtype=torch.float16)

    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # choose tile sizes
    if N >= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    # reshape back to original layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator.
    Implements the heavy pairwise contraction with a fast batched GEMM;
    for moderate sequence lengths a Triton kernel is used instead.
    All heavy linear work runs in fp16; the final result is returned as fp32.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----- 1. First LayerNorm (float32) -----
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # torch.layer_norm works with float32 tensors by default
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ----- 2. Cast to fp16 for all subsequent ops -----
    x = x.to(torch.float16)

    # ----- 3. Cast weights -----
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ----- 4. Linear projections (fp16) -----
    left = torch.nn.functional.linear(x, left_proj_w)      # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)    # [B,N,N,hidden]

    # ----- 5. Optional mask (broadcast over hidden) -----
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ----- 6. Gating (sigmoid) -----
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ----- 7. Heavy O(N³) contraction -----
    B, N, _, H = left.shape
    if N <= 256:
        # for small sequences use the Triton kernel we wrote above
        out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hidden] fp16
    else:
        # for larger sequences we rely on cuBLAS batched GEMM (torch.bmm)
        # reshape to [B*hidden, N, N]
        left_t = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
        right_t = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
        out = torch.bmm(left_t, right_t.transpose(1, 2))
        out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----- 8. Second LayerNorm over hidden_dim -----
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ----- 9. Out gating (sigmoid) -----
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B,N,N,hidden]

    out = out * out_gate

    # ----- 10. Final linear projection back to original dim -----
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # ----- 11. Cast back to float32 for the harness -----
    return out.to(torch.float32)

"""
Outgoing TriMul (AlphaFold‑3) – fast forward only.

*   The heavy O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is implemented as a batched GEMM (C = A @ Bᵀ).  For small
    sequence lengths (N ≤ 256) a custom Triton kernel is used,
    otherwise we fall back to cuBLAS via ``torch.bmm`` (the
    latter is far‑faster on an H100).  All linear projections,
    gating, and the second LayerNorm run in fp16; the first
    LayerNorm and the final cast back to fp32 are kept in fp32
    for numerical stability.  The implementation respects the
    optional mask, works for any batch size ≤ 2, any hidden
    dimension, and any of the test shapes listed in the problem
    statement.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : [M, N, K]   (M = batch*hidden, N = seq_len, K = seq_len)
#   C    : [M, N, N]
#   All tensors are fp16, accumulator is fp32.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # scalar sizes
    stride_am, stride_ai, stride_ak,   # A strides (M,i,k)
    stride_bm, stride_bj, stride_bk,   # B strides (M,j,k)
    stride_cm, stride_ci, stride_cj,   # C strides (M,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # Program IDs – (i‑tile, j‑tile, batch‑hidden index)
    # --------------------------------------------------------------
    pid_i = tl.program_id(0)   # tile along first N dimension (i)
    pid_j = tl.program_id(1)   # tile along second N dimension (j)
    pid_z = tl.program_id(2)   # batch*hidden index

    # --------------------------------------------------------------
    # Offsets of the current tile
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # FP32 accumulator
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K in tiles
    # --------------------------------------------------------------
    for start_k in range(0, K, BLOCK_K):
        cur_k = start_k
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # B tile, but transposed: shape (BLOCK_K, BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk   # row   = k
            + offs_j[None, :] * stride_bj   # column= j
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # dot → fp32 accumulation
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Store the tile (convert back to fp16)
    # --------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(left: torch.Tensor,
                                   right: torch.Tensor) -> torch.Tensor:
    """
    Compute ``out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`` using the
    Triton kernel above.  ``left`` and ``right`` must be fp16 and contiguous.

    Parameters
    ----------
    left, right : torch.Tensor
        Shape ``[B, N, N, H]`` (fp16).

    Returns
    -------
    out : torch.Tensor
        Shape ``[B, N, N, H]`` (fp16).
    """
    B, N, _, H = left.shape

    # Rearrange to [B*H, N, N] – each hidden head becomes its own batch
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A, dtype=torch.float16)

    # Strides (in element counts, not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tile size selection
    if N >= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64

    # Grid: (i‑tiles, j‑tiles, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # Reshape back to the original 4‑D layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor],
                Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operator (no backward).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool / float)
        - weights_dict  : mapping name → torch.Tensor (float32)
        - config_dict   : must contain ``dim`` and ``hidden_dim``.
                          Optional ``nomask`` (bool, default ``True``).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm – stay in float32 for best stats
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )                        # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the rest of the pipeline
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Cast all linear weights once to fp16
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)    # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – fp16
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    if N <= 256:
        # Small sequences → custom Triton kernel
        out = _batched_matmul_transpose_fp16(left, right)   # fp16
    else:
        # Large sequences → cuBLAS batched GEMM (torch.bmm)
        # Bring hidden dim into the batch axis
        left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim,
                                                               N, N)
        right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim,
                                                                 N, N)
        out = torch.bmm(left_flat, right_flat.transpose(1, 2))
        out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                        # fp16

    # -----------------------------------------------------------------
    # 9️⃣ Out‑gate (sigmoid) and gating
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))
    out = out * out_gate

    # -----------------------------------------------------------------
    # 🔟 Final projection back to the original dimensionality
    # -----------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)      # fp16

    # -----------------------------------------------------------------
    # Final cast back to float32 for the external harness
    # -----------------------------------------------------------------
    return out.to(torch.float32)
