def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward of the outgoing TriMul operator (gradient‑free).

    1️⃣ LayerNorm on the input (FP32 → FP16)
    2️⃣ Fuse five linear projections (left_proj, left_gate, right_proj,
        right_gate, out_gate) into a single matrix‑multiply to reduce
        memory traffic.
    3️⃣ Triton kernel fuses element‑wise masking and gating, producing the
        gated tensors ``left`` and ``right``.
    4️⃣ Batched matrix‑multiply ``left @ rightᵀ`` via torch.bmm (cuBLAS TF32/
        Tensor‑Core fast path).  This implements the O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
    5️⃣ Apply the second LayerNorm, output‑gate and final projection.
    6️⃣ Return FP32 tensor.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # ------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)  # half‑precision for the heavy work

    B, N, _, _ = x.shape
    # ------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (weight concat)
    # ------------------------------------------------------------
    # cast all needed weights to fp16 once
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # shape (5*hidden, dim)
    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w], dim=0
    )  # (5*hidden, dim)

    # Flatten input to (B·N·N, dim) and run the single linear
    x_flat = x.reshape(-1, dim)  # (B*N*N, dim)
    fused_out = torch.nn.functional.linear(x_flat, fused_weight)  # (B*N*N, 5*hidden)

    # split back into the five tensors
    splits = torch.split(fused_out, hidden_dim, dim=1)
    left_proj, left_gate, right_proj, right_gate, out_gate = splits
    # reshape to [B, N, N, hidden]
    left_proj = left_proj.view(B, N, N, hidden_dim).contiguous()
    left_gate = left_gate.view(B, N, N, hidden_dim).contiguous()
    right_proj = right_proj.view(B, N, N, hidden_dim).contiguous()
    right_gate = right_gate.view(B, N, N, hidden_dim).contiguous()
    out_gate = out_gate.view(B, N, N, hidden_dim).contiguous()

    # ------------------------------------------------------------
    # 3️⃣ Triton kernel – fuse mask (if any) and gating
    # ------------------------------------------------------------
    total_elements = left_proj.numel()  # B·N·N·hidden
    # Prepare mask – a 1‑D tensor of length B·N·N (broadcast over hidden)
    if not nomask and mask_tensor is not None:
        mask_flat = mask_tensor.to(torch.float16).reshape(-1)  # (B*N*N,)
    else:
        mask_flat = torch.ones(B * N * N, dtype=torch.float16, device=device)

    # Allocate output containers
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # Flatten everything for the kernel (contiguous views)
    lp_flat = left_proj.view(-1)
    lg_flat = left_gate.view(-1)
    rp_flat = right_proj.view(-1)
    rg_flat = right_gate.view(-1)
    left_out_flat = left.view(-1)
    right_out_flat = right.view(-1)

    BLOCK_SIZE = 8192  # 8 K elements per program
    grid = ( (total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE, )
    _gate_and_mask_kernel[grid](
        lp_flat,
        lg_flat,
        rp_flat,
        rg_flat,
        mask_flat,
        left_out_flat,
        right_out_flat,
        total_elements,
        hidden_dim,
        B * N * N,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=8,
        num_stages=2,
    )
    # ------------------------------------------------------------
    # 4️⃣ Heavy O(N³) contraction – batched GEMM (torch.bmm)
    # ------------------------------------------------------------
    # reshape to (B*hidden, N, N) with the proper ordering
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*hidden, N, N)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, hidden)

    # ------------------------------------------------------------
    # 5️⃣ Second LayerNorm, output gating, final projection
    # ------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out_gated = out_norm * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, hidden)
    out_final = torch.nn.functional.linear(out_gated, to_out_w)

    # ------------------------------------------------------------
    # 6️⃣ Cast back to FP32 (expected by caller)
    # ------------------------------------------------------------
    return out_final.to(torch.float32)

"""
Outgoing TriMul (AlphaFold‑3) – fast forward implementation (gradient‑free).

Algorithm
---------
1. Layer‑norm the 4‑D input (FP32) and cast to FP16.
2. Fuse the five linear projections (left_proj, left_gate, right_proj,
   right_gate, out_gate) into a *single* mat‑mul by concatenating the
   five weight matrices.  This reduces the number of passes over the
   massive [B·N·N, dim] tensor from five to one, cutting memory traffic
   dramatically.
3. Apply element‑wise mask (if any) and the three sigmoid gates with a
   custom Triton kernel that also broadcasts the 2‑D mask over the hidden
   dimension, producing the gated tensors `left` and `right`.
4. Perform the heavy O(N³) contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a batched matrix multiplication using cuBLAS (`torch.bmm`), which
   fully exploits Tensor‑Cores on H100.
5. Second LayerNorm, output‑gate multiplication and final linear
   projection back to `dim`.
6. Cast the result to FP32 (the caller’s expected dtype).

Only the mask‑plus‑gate step is executed in Triton; all heavy linear
algebra uses highly‑optimized cuBLAS kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused mask + gate (no bias, fp16 arithmetic)
# ----------------------------------------------------------------------
@triton.jit
def _gate_and_mask_kernel(
    lp_ptr,  # left_proj  (flattened)
    lg_ptr,  # left_gate  (flattened, already sigmoid‑ed)
    rp_ptr,  # right_proj (flattened)
    rg_ptr,  # right_gate (flattened, already sigmoid‑ed)
    mask_ptr,  # [B·N·N] mask (fp16, values 0/1), broadcast over hidden
    left_out_ptr,  # output left  (flattened)
    right_out_ptr,  # output right (flattened)
    numel,            # total number of elements = B·N·N·H
    hidden_dim,       # H
    mask_len,         # B·N·N
    BLOCK_SIZE: tl.constexpr,
):
    """
    left  = left_proj  * left_gate  * mask
    right = right_proj * right_gate * mask
    All tensors are flattened 1‑D; `mask` is broadcast over the hidden
    dimension using integer division.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    active = offs < numel                     # bounds guard for the big tensor
    # broadcasting mask: each hidden entry shares the same mask entry
    mask_idx = offs // hidden_dim
    mask_ok = mask_idx < mask_len

    lp = tl.load(lp_ptr + offs, mask=active, other=0.0)
    lg = tl.load(lg_ptr + offs, mask=active, other=0.0)
    rp = tl.load(rp_ptr + offs, mask=active, other=0.0)
    rg = tl.load(rg_ptr + offs, mask=active, other=0.0)
    m  = tl.load(mask_ptr + mask_idx, mask=mask_ok, other=1.0)

    left  = lp * lg * m
    right = rp * rg * m

    tl.store(left_out_ptr + offs,  left,  mask=active)
    tl.store(right_out_ptr + offs, right, mask=active)
