"""
TriMul (outgoing) – fused, mixed‑precision implementation.

Algorithm
---------
1.  Layer‑norm the 4‑D input in FP32 (numerical stability).
2.  Cast to FP16 for the heavy part of the pipeline.
3.  Fuse the five linear projections (left, right, three gates) into one
    `torch.nn.functional.linear` by concatenating the weight matrices.
    This reduces the number of passes over the huge tensor `x`.
4.  Split the fused projection, apply the optional mask and the sigmoid gates,
    and fuse the gates in‑place (`left *= left_gate`, `right *= right_gate`).
5.  Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is a batched GEMM (A @ Bᵀ) with shapes
        A,B : [M, N, K]   where M = B·hidden, N = seq_len, K = seq_len.
    A fused Triton kernel performs the GEMM with:
        - fp16 inputs,
        - fp32 accumulation (Tensor‑core friendly),
        - autotuned tile sizes (64/128 × 64/128 × 64) and warps.
    For very short sequences (N ≤ 128) the launch overhead dominates,
    therefore a simple `torch.einsum` fallback is used.
6.  Second LayerNorm (FP16) → multiply by the out‑gate → final linear
    projection back to the original `dim`.
7.  Cast the result back to FP32 – the harness expects this dtype.

The kernel only handles the O(N³) contraction; all other operations stay in
high‑level PyTorch to keep the implementation clear and portable.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (fp16 → fp32 accumulation → fp16)
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        # Small‑sequence configuration (more warps = better latency hiding)
        triton.Config({'BLOCK_M': 64,  'BLOCK_N': 64,  'BLOCK_K': 64},  num_warps=4,  num_stages=2),
        # Large‑sequence configuration (maximises data reuse)
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 128, 'BLOCK_K': 64},  num_warps=8,  num_stages=2),
        # Mixed configuration – useful when N is not a multiple of 128
        triton.Config({'BLOCK_M': 64,  'BLOCK_N': 128, 'BLOCK_K': 32},  num_warps=8,  num_stages=2),
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 64,  'BLOCK_K': 32},  num_warps=8,  num_stages=2),
    ],
    key=['N'],  # autotune based on the sequence length
)
@triton.jit
def _trimul_gemm_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # program‑size constants
    stride_am, stride_ai, stride_ak,   # strides for A (M, i, k)
    stride_bm, stride_bj, stride_bk,   # strides for B (M, j, k)
    stride_cm, stride_ci, stride_cj,   # strides for C (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for TriMul: C = A @ Bᵀ (M = B·hidden)."""
    pid_m = tl.program_id(0)   # i‑tile
    pid_n = tl.program_id(1)   # j‑tile
    pid_z = tl.program_id(2)   # batch‑hidden tile

    # ------------------------------------------------------------------
    # Tile offsets in the output matrix
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # --------------------------------------------------------------
        # Load a tile of A : [BLOCK_M, BLOCK_K]  (i,k)
        # --------------------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # FP16

        # --------------------------------------------------------------
        # Load a tile of B transposed : [BLOCK_K, BLOCK_N]  (k,j)
        # --------------------------------------------------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # FP16

        # Mixed‑precision dot product (FP16 × FP16 → FP32)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result tile back to global memory (convert to FP16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = left @ rightᵀ  with fp16 inputs, fp32 accumulation.
    left/right have shape [B, N, N, H] (H = hidden_dim).
    Returns C of the same shape and dtype (fp16, contiguous).
    """
    B, N, _, H = left.shape
    # Collapse batch and hidden dimensions:  [B·H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in element counts (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()      # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()    # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()     # (N*N, N, 1)

    # Choose tile size based on sequence length (simple heuristic)
    if N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch‑hidden dimension
    )

    _trimul_gemm_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8 if BLOCK_M == 128 else 4,
        num_stages=2,
    )

    # Reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32, bool or None)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # -------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )                          # [B, N, N, dim]  FP32

    # -------------------------------------------------
    # 2️⃣ FP16 conversion for the heavy part
    # -------------------------------------------------
    x = x.to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Fuse the five linear projections into one matmul
    #    (weights are stored as FP32 → cast to FP16 on the fly)
    # -------------------------------------------------
    weight_all = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)   # (5*hidden, dim)

    # Shape after fused linear: [B, N, N, 5*hidden]
    fused = torch.nn.functional.linear(x, weight_all)   # FP16

    # -------------------------------------------------
    # 4️⃣ Slice the fused tensor
    # -------------------------------------------------
    H = hidden_dim
    left      = fused[..., 0:H]                 # [B,N,N,H]
    right     = fused[..., H:2 * H]
    left_gate = fused[..., 2 * H:3 * H]
    right_gate = fused[..., 3 * H:4 * H]
    out_gate  = fused[..., 4 * H:5 * H]

    # -------------------------------------------------
    # 5️⃣ Optional mask (broadcast on hidden dimension)
    # -------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → [B, N, N, 1]
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 6️⃣ Apply sigmoid gates (FP16) and fuse them in‑place
    # -------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left.mul_(left_gate)   # in‑place fusion
    right.mul_(right_gate)

    # -------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction (batched GEMM)
    # -------------------------------------------------
    B, N, _, H = left.shape
    if N <= 128:
        # Fallback for very short sequences – launch overhead dominates.
        out = torch.einsum('b i k h, b j k h -> b i j h', left, right)
    else:
        out = _batched_gemm_fp16(left, right)   # fp16 Tensor‑core GEMM

    # -------------------------------------------------
    # 8️⃣ Second LayerNorm (FP16)
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 9️⃣ Final gating and linear projection back to original dim
    # -------------------------------------------------
    out = out * out_gate                                   # gate (FP16)
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)       # [B,N,N,dim] FP16

    # -------------------------------------------------
    # 🔟 Cast back to FP32 (required by the harness)
    # -------------------------------------------------
    return out.to(torch.float32)
```
