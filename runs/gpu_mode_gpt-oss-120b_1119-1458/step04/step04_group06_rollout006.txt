"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (float32 → numerical stability).
2️⃣ Cast to fp16; all subsequent heavy work runs in fp16 with fp32 accumulation.
3️⃣ Linear projections (left/right) and three gating projections (left_gate,
   right_gate, out_gate).  The linear layers have no bias – we use torch.nn.functional.linear.
4️⃣ If a mask is provided we broadcast it on the hidden dimension and multiply it
   with the left/right projections **before** the gates.
5️⃣ Apply sigmoid gates and fuse them into the projections in‑place.
6️⃣ Heavy O(N³) contraction:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batch of matrix multiplications C = A @ Bᵀ where
   A,B ∈ ℝ^{(B·H) × N × N}.  For small N (≤128) we use a custom Triton
   kernel (mixed‑precision MMA); for larger N we fall back to cuBLAS via
   torch.matmul which is faster on H100 for big matrices.
7️⃣ Second layer‑norm on the hidden dimension (fp16), multiply by out_gate,
   and linearly project back to the original `dim`.
8️⃣ Cast the result back to float32 (the harness expects FP32).

The Triton kernel implements a tiled GEMM with 128×128×64 tiles (or 64×64×64
for very small N).  It uses fp16 inputs, fp32 accumulation, and 8 warps /
2 pipeline stages – a sweet spot for H100 tensor‑cores.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (fp16 in, fp32 acc, fp16 out)
#   A, B, C : [M, N, K]   (M = B·H, N = seq_len, K = seq_len)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # program‑size constants
    stride_am, stride_ai, stride_ak,   # A strides (M,i,k)
    stride_bm, stride_bj, stride_bk,   # B strides (M,j,k)
    stride_cm, stride_ci, stride_cj,   # C strides (M,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # i‑tile
    pid_n = tl.program_id(1)   # j‑tile
    pid_z = tl.program_id(2)   # batch·hidden tile

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (mixed‑precision MMA)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # --------------------------------------------------------------
        # Load A tile   (BLOCK_M × BLOCK_K)
        # --------------------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # --------------------------------------------------------------
        # Load B tile   (BLOCK_K × BLOCK_N) – transposed on‑the‑fly
        # --------------------------------------------------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # Mixed‑precision dot: fp16 × fp16 → fp32
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store C tile (fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(left: torch.Tensor,
                                  right: torch.Tensor) -> torch.Tensor:
    """
    Contracts left and right via out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].
    Uses a Triton kernel for N ≤ 128, otherwise delegates to cuBLAS
    (torch.matmul) which is faster for large sequence lengths.
    All tensors must be fp16 and contiguous.

    Parameters
    ----------
    left, right : torch.Tensor
        Shape [B, N, N, H] (half‑precision).

    Returns
    -------
    out : torch.Tensor
        Shape [B, N, N, H] (half‑precision, contiguous).
    """
    B, N, _, H = left.shape
    # choose dispatch based on N
    if N <= 128:
        # ---- Triton kernel path ------------------------------------------------
        # Pack batch & hidden into a single leading dimension: [B·H, N, N]
        A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
        B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

        # Output placeholder
        C = torch.empty_like(A, dtype=torch.float16)

        # Strides in element counts (not bytes)
        stride_am, stride_ai, stride_ak = A.stride()
        stride_bm, stride_bj, stride_bk = B_.stride()
        stride_cm, stride_ci, stride_cj = C.stride()

        # Tile size heuristic
        if N <= 64:
            BLOCK_M = BLOCK_N = 64
        else:
            BLOCK_M = BLOCK_N = 64   # small N path keeps tiles modest
        BLOCK_K = 64

        grid = (
            (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
            (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
            B * H,                          # batch·hidden
        )

        _trimul_matmul_kernel[grid](
            A, B_, C,
            B * H, N, N,
            stride_am, stride_ai, stride_ak,
            stride_bm, stride_bj, stride_bk,
            stride_cm, stride_ci, stride_cj,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=8,
            num_stages=2,
        )
        # Restore original layout
        out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
        return out
    else:
        # ---- cuBLAS (torch.matmul) path ---------------------------------------
        # Shape to [B·H, N, N]
        A = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
        B_ = right.permute(0, 3, 1, 2).reshape(B * H, N, N)
        # Matrix multiply A @ Bᵀ  (cuBLAS, fp16 inputs, fp32 accum.)
        C = torch.matmul(A, B_.transpose(-2, -1))
        # Reshape back
        out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
        return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------
    # 2️⃣ Cast to fp16 for the remainder of the pipeline
    # -------------------------------------------------
    x = x.to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Cast all model weights to fp16 on the target device
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # -------------------------------------------------
    # 4️⃣ Linear projections (half‑precision)
    # -------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,hidden_dim]

    # -------------------------------------------------
    # 5️⃣ Optional mask (broadcast on hidden dim)
    # -------------------------------------------------
    if not nomask:
        # Ensure mask is fp16 and broadcastable
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 6️⃣ Gating (sigmoid) – half‑precision
    # -------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Fuse gates in‑place to reduce memory traffic
    left.mul_(left_gate)
    right.mul_(right_gate)

    # -------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction
    # -------------------------------------------------
    # left/right: [B, N, N, hidden_dim] (fp16)
    # Result: out [B, N, N, hidden_dim] (fp16)
    out = _batched_matmul_transpose_fp16(left, right)

    # -------------------------------------------------
    # 8️⃣ Second LayerNorm (half‑precision)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 9️⃣ Output gating & final linear projection back to `dim`
    # -------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # [B, N, N, dim] (fp16)

    # -------------------------------------------------
    # 🔟 Cast back to float32 (expected by the harness)
    # -------------------------------------------------
    return out.to(torch.float32)
