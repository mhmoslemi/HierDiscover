"""
Outgoing TriMul (AlphaFold‑3) – high‑performance forward pass.

Algorithm
---------
1️⃣ Layer‑norm on the input (float32) – the only normalization required.
2️⃣ Cast to fp16 for the rest of the pipeline.
3️⃣ Fuse the linear projection and its gate:
   * left  = Linear(x,   [W_left_proj ; W_left_gate])  → split →  proj / gate
   * right = Linear(x,   [W_right_proj; W_right_gate]) → split →  proj / gate
   (Both projections and gates are computed with a single matmul each.)
4️⃣ Optional mask multiplication (broadcast on the hidden dim).
5️⃣ Apply the sigmoid gates and obtain the gated tensors *left* and *right*.
6️⃣ Heavy contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented with a custom Triton kernel that works directly on the
   original layout [B, N, N, H] – **no permute/contiguous copy**.
   The kernel treats (batch, hidden) as a flattened dimension and
   computes a batched GEMM C = A @ Bᵀ with fp16×fp16 → fp32 accumulation.
7️⃣ Second LayerNorm over the hidden dimension (fp16).
8️⃣ Compute the output gate (sigmoid) and fuse it element‑wise.
9️⃣ Final linear projection back to `dim`.
🔟 Cast the result to float32 (required by the harness).

The kernel avoids the costly `permute(...).contiguous().view()` that was
present in the reference implementation and reduces the number of
separate `torch.nn.functional.linear` calls from six to four.
Typical speed‑ups on an H100 for the test configurations are well
below the 1 ms target.
"""

import torch
import torch.nn.functional as F
import tr Triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (fp16→fp16, accumulation fp32)
# A, B, C: [B, N, N, H]   (row‑major, contiguous)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel_fused(
    A_ptr, B_ptr, C_ptr,                     # *fp16 tensors
    BATCH, SEQLEN, HIDDEN, KSIZE,            # scalar sizes (int32)
    stride_ab, stride_ai, stride_ak, stride_ah,   # strides for A  (B,N,N,H)
    stride_bb, stride_bj, stride_bk, stride_bh,   # strides for B  (B,N,N,H)
    stride_cb, stride_ci, stride_cj, stride_ch,   # strides for C  (B,N,N,H)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    H: tl.constexpr,                         # hidden dimension (compile‑time)
):
    """Batched GEMM for the TriMul contraction (no data copies)."""
    pid_m = tl.program_id(0)   # tile along i (rows)
    pid_n = tl.program_id(1)   # tile along j (cols)
    pid_z = tl.program_id(2)   # flattened (batch, hidden) index

    # ------------------------------------------------------------------
    # Compute batch and hidden indices from the flattened pid_z
    # ------------------------------------------------------------------
    b_idx = pid_z // H
    h_idx = pid_z % H

    # base offsets for the current (b, h) slice
    offset_A = b_idx * stride_ab + h_idx * stride_ah
    offset_B = b_idx * stride_bb + h_idx * stride_bh
    offset_C = b_idx * stride_cb + h_idx * stride_ch

    # ------------------------------------------------------------------
    # Tile offsets for i, j and masks
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < SEQLEN
    mask_j = offs_j < SEQLEN

    # accumulator in FP32 (tensor cores accumulate in FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(KSIZE, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # -------------------- load A tile (i, k) --------------------
        a_ptrs = (
            A_ptr
            + offset_A
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < KSIZE),
            other=0.0,
        )   # fp16

        # -------------------- load B tile (j, k) as transposed --------------------
        # B is indexed as (batch, j, k, h); we need Bᵀ so we load (k, j)
        b_ptrs = (
            B_ptr
            + offset_B
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < KSIZE) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # fp16 × fp16 → fp32 accumulation
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result (cast back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + offset_C
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton_fused(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Heavy contraction step for TriMul using the fused Triton kernel.
    Arguments:
        left, right – tensors of shape [B, N, N, H] (fp16)
    Returns:
        out – tensor of shape [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    C = torch.empty_like(left)  # fp16 output

    # strides are in *elements* (not bytes)
    stride_ab, stride_ai, stride_ak, stride_ah = left.stride()
    stride_bb, stride_bj, stride_bk, stride_bh = right.stride()
    stride_cb, stride_ci, stride_cj, stride_ch = C.stride()

    # All dimensions are square in the contraction: K = SEQLEN = N
    K = N

    # Tile size – empirically good on H100 for fp16 tensor‑core GEMM
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,        # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,        # j‑tiles
        B * H,                               # flattened (batch, hidden)
    )

    _trimul_matmul_kernel_fused[grid](
        left,
        right,
        C,
        B,
        N,
        H,
        K,
        stride_ab,
        stride_ai,
        stride_ak,
        stride_ah,
        stride_bb,
        stride_bj,
        stride_bk,
        stride_bh,
        stride_cb,
        stride_ci,
        stride_cj,
        stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        H=H,                # compile‑time constant for division/mod
        num_warps=8,
        num_stages=2,
    )
    return C

        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 / bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 2️⃣ first LayerNorm (float32) – preserve numerical stability
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 3️⃣ cast to fp16 for the remaining ops
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ fuse projection + gate linear maps (single matmuls per side)
    # ------------------------------------------------------------------
    #   left:  [W_left_proj ; W_left_gate]  → shape (2*hidden_dim, dim)
    #   right: [W_right_proj; W_right_gate] → shape (2*hidden_dim, dim)
    left_proj_gate_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                     # (2*H, dim)
    right_proj_gate_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                     # (2*H, dim)

    # fused linear → split later
    left_fused = F.linear(x, left_proj_gate_w)   # [B, N, N, 2*H] (fp16)
    right_fused = F.linear(x, right_proj_gate_w) # [B, N, N, 2*H] (fp16)

    # split projection and gate pre‑sigmoid
    left_proj = left_fused[..., :hidden_dim]                # [B, N, N, H]
    left_gate_pre = left_fused[..., hidden_dim:]            # [B, N, N, H]
    right_proj = right_fused[..., :hidden_dim]              # [B, N, N, H]
    right_gate_pre = right_fused[..., hidden_dim:]          # [B, N, N, H]

    # ------------------ optional mask ------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # apply sigmoid gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ out‑gate (independent of the contraction)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ heavy contraction – batched GEMM C = A @ Bᵀ
    # ------------------------------------------------------------------
    out = _batched_matmul_triton_fused(left, right)    # [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 7️⃣ second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # ------------------------------------------------------------------
    # 8️⃣ element‑wise out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)   # [dim, hidden_dim]
    out = F.linear(out, to_out_w)                         # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 9️⃣ cast back to float32 (the harness expects fp32)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
