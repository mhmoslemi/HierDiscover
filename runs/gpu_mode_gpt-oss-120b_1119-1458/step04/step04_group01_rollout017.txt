@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
    batch_head = tl.program_id(0)  # from 0 to B*H - 1
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute b and h
    b = batch_head // H
    h = batch_head % H

    # compute block offsets
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    # create offsets within tile
    offs_i = i_offset + tl.arange(0, BLOCK_M)
    offs_j = j_offset + tl.arange(0, BLOCK_N)
    # create offset for k blocks
    k_offset = 0

    # masks for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute base pointers for left and right and output for this batch-head
    # left base for this head: left_ptr + b*stride_left_b + h*stride_left_h
    # right base: right_ptr + b*stride_right_b + h*stride_right_h
    # out base: out_ptr + b*stride_out_b + h*stride_out_h

    base_left = left_ptr + b * stride_left_b + h * stride_left_h
    base_right = right_ptr + b * stride_right_b + h * stride_right_h
    base_out = out_ptr + b * stride_out_b + h * stride_out_h

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension blocks
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        # compute start of k block
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile: shape (BLOCK_M, BLOCK_K)
        # off: left[ b, i, k, h ]
        a_ptrs = base_left + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)  # fp16
        # load right tile transposed: shape (BLOCK_K, BLOCK_N)
        # right[ b, j, k, h ] => we want as K rows, N cols: pointer = base_right + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        b_ptrs = base_right + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate dot product
        # a shape (M,K) (fp16); b shape (K,N) (fp16)
        acc += tl.dot(a, b)  # returns fp32

    # store result into out
    # out shape (M,N)
    out_ptrs = base_out + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    # cast to fp16
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

```python
"""
TriMul (outgoing) forward pass optimized with Triton.

Algorithm:
1. Layer‑norm x (float32) -> fp16.
2. Six linear projections (proj, gate, out_gate) performed as
   half‑precision matmuls (one for each weight matrix) on the
   flattened 4‑D tensor [B·N·N, dim] → [B·N·N, hidden_dim].
3. Optional binary mask applied to the projected tensors.
4. Multiply projections with sigmoid‑gates (elementwise) producing
   tensors L,R of shape [B, N, N, hidden_dim].
5. Heavy O(N³) contraction L·Rᵀ is executed by a custom Triton kernel
   which treats (batch, head) as a fused batch dimension and performs
   a tiled matrix multiplication without any intermediate reshapes.
6. Second LayerNorm on the hidden dimension, followed by out‑gate,
   final linear projection back to the original dimension, and cast to
   float32. All heavy work runs on fp16 with fp32 accumulation.
"""

import math
from typing import Tuple, Dict, Any

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the core O(N³) contraction:
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# This kernel treats (b,h) as a combined batch dimension.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,               # tensor pointers (fp16)
    B, N, H,                                    # dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_batch_h = tl.program_id(0)   # 0 … B*H-1
    pid_i = tl.program_id(1)        # block row of the output matrix
    pid_j = tl.program_id(2)        # block column

    # decode batch and head indices
    b = pid_batch_h // H
    h = pid_batch_h % H

    # starting coordinates of the block
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # generate range vectors
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b, h)
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the K dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_iter * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right block transposed: we load as (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate; tl.dot returns fp32 when inputs are fp16
        acc += tl.dot(a, b)

    # write the result back to memory (fp16)
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right using the custom Triton kernel.
    left/right: [B, N, N, H] (fp16)
    Returns out: same shape (fp16)
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in element units
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # block sizes – tuned for FP16 matmul on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # grid dimensions
    total_batch_h = B * H
    grid = (
        total_batch_h,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,            # enough warps for the block sizes
        # an explicit num_stages can be omitted; Triton defaults work well
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator (gradient‑free).

    Arguments:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ----------------------- unpack -----------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ----------------------- layer‑norm -----------------------
    # norm parameters are float32 as in the reference implementation
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32
    x = x_norm.to(torch.float16)  # fp16 for the remainder

    # ----------------------- weight casting -----------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ----------------------- linear projections -----------------------
    # x has shape [B, N, N, dim] -> flatten to (B*N*N, dim) for each linear op
    # torch.nn.functional.linear works directly on the last dimension.
    left_proj = F.linear(x, left_proj_w)           # [B, N, N, hidden_dim]
    right_proj = F.linear(x, right_proj_w)         # [B, N, N, hidden_dim]

    # ----------------------- optional mask -----------------------
    if not nomask:
        # mask is [B, N, N]; broadcast over hidden dimension
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ----------------------- gating -----------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # [B,N,N,hidden_dim]

    # combine projection with gate
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ----------------------- O(N³) contraction (Triton) -----------------------
    # Result: out [B, N, N, hidden_dim] in fp16
    out = _batched_matmul_triton(left, right)

    # ----------------------- second LayerNorm (fp16) -----------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ----------------------- final gating + projection -----------------------
    out = out * out_gate                     # element‑wise gate
    out = F.linear(out, to_out_w)            # back to dim (fp16)

    # ----------------------- cast back to float32 -----------------------
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass – a highly‑optimised Triton implementation.

The algorithm follows the reference PyTorch module but runs the expensive
O(N³) contraction with a custom Triton kernel:

    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

All data after the first LayerNorm is kept in FP16 (accumulating in FP32)
to maximise throughput on an H100. Linear projections, gates and the final
LayerNorm / projection are performed with torch.nn.functional in half
precision, while the contraction is a fused batched GEMM that avoids any
intermediate reshapes or copies. The kernel treats (batch, head) as a
combined batch dimension and computes a tiled matrix multiplication for
each head in parallel.

The implementation is fully differentiable‑free (forward only) and works
for every configuration listed in the problem statement.
"""

import math
from typing import Tuple, Dict, Any

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the O(N³) contraction.
# out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers (fp16)
    B, N, H,                                    # dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # 1) Program IDs
    # ------------------------------------------------------------------
    pid_batch_h = tl.program_id(0)   # combined (batch, head) index
    pid_i = tl.program_id(1)        # output row block   (i)
    pid_j = tl.program_id(2)        # output col block   (j)

    # ------------------------------------------------------------------
    # 2) Decode batch and head from the flattened dimension
    # ------------------------------------------------------------------
    b = pid_batch_h // H
    h = pid_batch_h % H

    # ------------------------------------------------------------------
    # 3) Compute the start of the current output block
    # ------------------------------------------------------------------
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # 4) Base pointers for the (b, h) slice
    # ------------------------------------------------------------------
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # ------------------------------------------------------------------
    # 5) Accumulator (FP32) – we keep precision while the tensors are FP16
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 6) Main K‑loop – tiled matrix multiplication
    # ------------------------------------------------------------------
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_iter * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block [M, K]  (i dimension, k dimension)
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=tl.float16(0))

        # right block transposed → [K, N] (k dimension, j dimension)
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=tl.float16(0))

        # accumulate the dot product
        acc += tl.dot(a, b)   # FP32 accumulation

    # ------------------------------------------------------------------
    # 7) Write the result back (store as FP16)
    # ------------------------------------------------------------------
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract two tensors of shape [B, N, N, H] using the custom Triton kernel.
    Both inputs must be dtype torch.float16 and contiguous.
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)   # same dtype & shape

    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tuned tile sizes – work well for FP16 matmul on an H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid: (batch*head, row_blocks, col_blocks)
    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,            # enough warps for the chosen tile size
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward‑only implementation of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)
        - input_tensor : [B, N, N, dim]       (float32)
        - mask_tensor  : [B, N, N]            (float32 / bool)
        - weight_dict  : mapping name → tensor   (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, on the same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – same as reference
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                     # [B, N, N, dim]  float32
    x = x_norm.to(torch.float16)   # cast to half‑precision for the rest

    # ------------------------------------------------------------------
    # 2️⃣ Cast all linear weights to fp16 (once)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_w)      # [B, N, N, hidden_dim]
    right_proj = F.linear(x, right_proj_w)    # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – all in FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # [B,N,N,hidden_dim]

    # Apply hidden‑dim gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via the custom Triton kernel
    # ------------------------------------------------------------------
    # left/right are already contiguous FP16 tensors
    out = _batched_matmul_triton(left, right)   # [B, N, N, hidden_dim] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Final gating + projection back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate                         # element‑wise gate
    out = F.linear(out, to_out_w)                # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 (expected output dtype)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
