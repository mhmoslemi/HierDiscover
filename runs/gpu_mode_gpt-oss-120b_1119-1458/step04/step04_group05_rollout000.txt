"""
fast_tri_mul_outgoing.py

Outgoing TriMul implementation (AlphaFold‑3 style) using a mixed‑precision
batched GEMM kernel written in Triton.

The heavy O(N³) contraction

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

is performed as a batched matrix‑multiply:
    left  → A of shape (B·H, N, N)
    right → Bᵀ of shape (B·H, N, N)   (the k‑axis is the common reduction)

A custom Triton kernel (`_trimul_matmul_kernel`) computes this GEMM in FP16
with FP32 accumulation.  For modest sequence lengths (≤ 512) the kernel is
used; for larger N we fall back to cuBLAS‑backed `torch.bmm`, which is
already highly optimized on an H100.  All other operations (LayerNorm,
linear projections, gating, output gating and final projection) are
performed in FP16 with PyTorch kernels.  The final result is returned in
FP32 as required by the API.

The kernel tiles the i‑ and j‑axes with square blocks (`BLOCK_M`×`BLOCK_N`)
and iterates over the k‑axis in `BLOCK_K`‑size chunks.  Each block uses a
single warp‑level `tl.dot` to accumulate a (M×N) sub‑matrix in FP32, after
which the result is written back in FP16.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A : (M, N, K)   – left  tensor  (M = B·H, N = seq_len, K = seq_len)
#   B : (M, K, N)   – right tensor already transposed (k‑j)
#   C : (M, N, N)   – result for each head
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bk, stride_bj,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs: tile over i (pid_m), j (pid_n) and batch*head (pid_z)
    pid_m = tl.program_id(0)  # i‑tile index
    pid_n = tl.program_id(1)  # j‑tile index
    pid_z = tl.program_id(2)  # batch*head index

    # Compute absolute i/j coordinates for this tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols (only needed when N not a multiple of tile)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32 (higher precision for the sum over K)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles of size BLOCK_K
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # -------- load A[i, k] ---------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # FP16

        # -------- load Bᵀ[k, j] (i.e. B[k, j]) -----------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # FP16

        # -------- dot product (FP16×FP16 → FP32) and accumulate -----
        acc += tl.dot(a, b)

    # Write result back (cast to FP16)
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(
    left: torch.Tensor,
    right: torch.Tensor,
    tile: int = 128,
    k_tile: int = 64,
) -> torch.Tensor:
    """
    Perform the batched GEMM `out = left @ rightᵀ` using the Triton kernel
    above.  Input tensors have shape [B, N, N, H] and are expected to be in
    ``torch.float16``.  The function pads the sequence dimension to a multiple
    of ``tile`` to avoid bounds checks inside the kernel, runs the kernel,
    and finally removes the padding.

    Parameters
    ----------
    left, right : torch.Tensor
        Tensors of shape [B, N, N, H] (FP16).
    tile : int
        Tile size for the i‑ and j‑axes (BLOCK_M/ BLOCK_N).
    k_tile : int
        Tile size for the reduction (BLOCK_K).

    Returns
    -------
    torch.Tensor
        Result tensor of shape [B, N, N, H] (FP16, same device as inputs).
    """
    B, N, _, H = left.shape
    pad_len = (tile - (N % tile)) % tile
    if pad_len != 0:
        N_pad = N + pad_len
        left_pad = left.new_zeros((B, N_pad, N_pad, H), dtype=torch.float16)
        right_pad = right.new_zeros((B, N_pad, N_pad, H), dtype=torch.float16)
        left_pad[:, :N, :N, :] = left
        right_pad[:, :N, :N, :] = right
    else:
        left_pad = left
        right_pad = right
        N_pad = N

    # (B, N_pad, N_pad, H) -> (B, H, N_pad, N_pad) -> (B*H, N_pad, N_pad)
    left_perm = left_pad.permute(0, 3, 1, 2).contiguous()
    right_perm = right_pad.permute(0, 3, 2, 1).contiguous()   # transpose k‑j

    A = left_perm.view(B * H, N_pad, N_pad)
    B_ = right_perm.view(B * H, N_pad, N_pad)

    # Output buffer
    C = torch.empty_like(A)

    # Strides (in element units)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bk, stride_bj = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Launch grid: (i‑tiles, j‑tiles, batch*head)
    grid = (
        (N_pad + tile - 1) // tile,
        (N_pad + tile - 1) // tile,
        B * H,
    )
    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N_pad,
        N_pad,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bk,
        stride_bj,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=tile,
        BLOCK_N=tile,
        BLOCK_K=k_tile,
        num_warps=8,
        num_stages=3,
    )
    # Reshape back to [B, N, N, H] and strip any padding
    C = C.view(B, H, N_pad, N_pad).permute(0, 2, 3, 1).contiguous()
    if pad_len != 0:
        C = C[:, :N, :N, :]
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the **outgoing** TriMul module (gradient‑free).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 1️⃣ Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2️⃣ First LayerNorm (run directly in FP16 to avoid an FP32 copy)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)
    # LayerNorm expects normalized_shape = (dim,)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    ).to(torch.float16)  # [B, N, N, dim] in FP16

    # ------------------------------------------------------------------
    # 3️⃣ Cast all model weights needed for the heavy part to FP16 once
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)       # (dim, hidden)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections & gating (all in FP16)
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    flat = x.reshape(-1, dim)               # (B·N·N, dim)

    # Projections (no bias)
    left = torch.nn.functional.linear(flat, left_proj_w).view(B, N, N, hidden_dim)
    right = torch.nn.functional.linear(flat, right_proj_w).view(B, N, N, hidden_dim)

    # Optional mask (broadcast over the hidden dimension)
    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_fp16
        right = right * mask_fp16

    # Gating: sigmoid(linear(x))
    left_gate = torch.sigmoid(torch.nn.functional.linear(flat, left_gate_w)).view(
        B, N, N, hidden_dim
    )
    right_gate = torch.sigmoid(torch.nn.functional.linear(flat, right_gate_w)).view(
        B, N, N, hidden_dim
    )
    out_gate = torch.sigmoid(torch.nn.functional.linear(flat, out_gate_w)).view(
        B, N, N, hidden_dim
    )

    # Apply gates to the projected tensors
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction – batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    # Use Triton kernel for N ≤ 512; otherwise rely on cuBLAS via torch.bmm.
    if N <= 512:
        # tile sizes tuned for H100 FP16 performance
        tile = 128          # BLOCK_M / BLOCK_N
        k_tile = 64         # BLOCK_K
        out = _batched_matmul_triton(left, right, tile=tile, k_tile=k_tile)
    else:
        # reshape to (B*hidden, N, N) and use cuBLAS batched GEMM
        left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
        # rightᵀ: need (k, j) ordering => transpose last two dims
        right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)
        out_mat = torch.bmm(left_mat, right_mat)               # (B*hidden, N, N)
        out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm, output gating, final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, dim) still FP16

    # ------------------------------------------------------------------
    # 7️⃣ Cast back to float32 (the public API expects FP32)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
