"""
Outgoing TriMul (AlphaFold‑3) – fast Triton‐accelerated contraction.

The heavy computation is the 3‑D contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
which is a batched GEMM C = A @ Bᵀ for every hidden head h.
We fuse this contraction in a custom Triton kernel that

* works on fp16 inputs and accumulates in fp32;
* avoids any extra copy of the input tensor by using a permuted view
  (no .contiguous() on the 4‑D tensors);
* chooses 128×128 tiles (or 64×64 for tiny N) with 8 warps and 3
  pipeline stages – well suited to H100 tensor‑cores.

All surrounding LayerNorms, linear projections and gating are performed
with high‑performance PyTorch primitives in fp16.  The function returns
float‑32 tensors to match the reference implementation.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : [M, N, K]   (M = batch * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
#   All tensors are fp16, accumulator is fp32.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # scalar sizes
    stride_am, stride_ai, stride_ak,   # A strides for (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides for (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides for (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM with A @ Bᵀ, using fp16 inputs and fp32 accumulation."""
    pid_i = tl.program_id(0)  # tile along first N dimension (i)
    pid_j = tl.program_id(1)  # tile along second N dimension (j)
    pid_z = tl.program_id(2)  # batch*hidden index (M)

    # Offsets for the i‑ and j‑tiles
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # fp32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k_start in range(0, K, BLOCK_K):
        cur_k = k_start
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # A tile: shape (BLOCK_M, BLOCK_K)   = left[b,i,k,h]
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=tl.float16(0.0),
        )  # fp16

        # B tile (transposed) : shape (BLOCK_K, BLOCK_N) = right[b,j,k,h]ᵀ
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=tl.float16(0.0),
        )  # fp16

        # Mixed‑precision dot (fp16 × fp16 → fp32)
        acc += tl.dot(a, b)

    # Write out the tile (convert back to fp16)
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(left: torch.Tensor,
                                   right: torch.Tensor) -> torch.Tensor:
    """
    Compute ``out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`` using a
    Triton kernel.  The inputs must be FP16 tensors of shape
    ``[B, N, N, H]`` (contiguous).  No extra copy of the 4‑D tensors is
    performed – we merely view them as ``[B*H, N, N]`` with appropriate
    strides.
    """
    B, N, _, H = left.shape

    # View as (B*H, N, N) without materialising a copy.
    # (B, N, N, H) → (B, H, N, N) → (B*H, N, N)
    left_view = left.permute(0, 3, 1, 2).reshape(-1, N, N)
    right_view = right.permute(0, 3, 1, 2).reshape(-1, N, N)

    M = left_view.shape[0]               # = B*H
    # Output buffer (FP16, same layout as left_view/right_view)
    out = torch.empty_like(left_view, dtype=torch.float16)

    # Strides in element counts (not bytes)
    stride_am, stride_ai, stride_ak = left_view.stride()
    stride_bm, stride_bj, stride_bk = right_view.stride()
    stride_cm, stride_ci, stride_cj = out.stride()

    # Tile size selection
    if N >= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8
    else:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        M,                               # batch * hidden
    )

    _trimul_matmul_kernel[grid](
        left_view,
        right_view,
        out,
        M,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=3,
    )

    # Reshape back to original 4‑D layout: (B, N, N, H)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operator (no backward).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default ``True``).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (float32 for numerical stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the rest of the pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Cast linear weights once to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)    # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Fuse the gates in‑place to avoid extra temporaries
    torch.mul(left, left_gate, out=left)
    torch.mul(right, right_gate, out=right)

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via Triton (fp16 in/out)
    # ------------------------------------------------------------------
    # left/right are already contiguous after the linear ops, but we
    # deliberately avoid any copy before feeding them to the kernel.
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hidden] fp16

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Final gating + projection (fp16)
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 🔟 Cast back to float32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
