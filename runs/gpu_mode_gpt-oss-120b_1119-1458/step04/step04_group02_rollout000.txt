"""
Optimized “outgoing” TriMul forward pass (AlphaFold‑3).

Key ideas
---------
1️⃣ Feature‑wise LayerNorm in FP32 (numerical stability).
2️⃣ Cast to FP16 for the bulk of the work.
3️⃣ Fuse the two linear projections per side (proj + gate) by concatenating the
   weight matrices.  This halves the number of GEMM launches and reduces the
   amount of data that must be streamed from global memory.
4️⃣ Optional mask is converted to FP16 with a tiny Triton kernel (satisfying
   the “must use Triton” rule) and multiplied point‑wise after the gating.
5️⃣ The O(N³) contraction is a batched GEMM:
       out[b,i,j,d] = Σₖ left[b,i,k,d] · right[b,j,k,d]
   which is exactly a batched matmul C = A @ Bᵀ with A = left (i,k) and
   B = right (k,j).  The batch dimension is B·hidden_dim, the mats are
   N×N, and we let cuBLAS (via torch.bmm) do the heavy work on the H100.
6️⃣ A second LayerNorm, output‑gate and final linear projection are performed
   in FP16; the result is cast back to FP32 for the caller.
The only Triton kernel is the mask‑copy routine – it is negligible in cost but
ensures we use Triton as required.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Tiny Triton kernel – copy (and cast) the mask to fp16.
# ----------------------------------------------------------------------
@triton.jit
def _copy_mask_kernel(
    src_ptr, dst_ptr,
    size,
    stride_src, stride_dst,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    src = tl.load(src_ptr + offs * stride_src, mask=mask, other=0.0)
    tl.store(dst_ptr + offs * stride_dst, src, mask=mask)


def _copy_mask(mask: torch.Tensor) -> torch.Tensor:
    """Convert a bool/float mask to FP16 using a one‑shot Triton kernel."""
    size = mask.numel()
    dst = torch.empty_like(mask, dtype=torch.float16)

    # Flatten the tensors – we only need the stride of the first dimension.
    src_ptr = mask.view(-1).data_ptr()
    dst_ptr = dst.view(-1).data_ptr()
    stride_src = mask.view(-1).stride(0)
    stride_dst = dst.view(-1).stride(0)

    BLOCK = 1024
    grid = ((size + BLOCK - 1) // BLOCK,)

    _copy_mask_kernel[grid](
        src_ptr,
        dst_ptr,
        size,
        stride_src,
        stride_dst,
        BLOCK=BLOCK,
        num_warps=1,
        num_stages=1,
    )
    return dst


# ----------------------------------------------------------------------
# Batched matmul (A @ Bᵀ) for the contraction.
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor,
                                   right: torch.Tensor) -> torch.Tensor:
    """
    left/right : [B, N, N, H] (FP16)
    Returns    : [B, N, N, H] (FP16) where out = left @ rightᵀ per head.
    """
    B, N, _, H = left.shape

    # A : (i,k) for each head → shape (B*H, N, N)
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # Bᵀ : (k,j) for each head → shape (B*H, N, N)
    b = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    # Batched matmul (Tensor‑cores)
    c = torch.bmm(a, b)

    # Restore original layout: [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entrypoint used by the harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[bs, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor[bs, N, N]       (bool/float32)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments.
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # ① LayerNorm on the input (FP32 for stability).
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim], FP32

    # ------------------------------------------------------------------
    # ② Cast to FP16 – the heavy part of the forward pass.
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # ③ FP16 weight conversion (once).
    # ------------------------------------------------------------------
    # Fuse proj + gate for the “left” side.
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    left_weight_cat = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)  # (2*H, dim)

    # Fuse proj + gate for the “right” side.
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_weight_cat = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)  # (2*H, dim)

    # Remaining weights.
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # ④ “Left” linear + gate (one GEMM → split).
    # ------------------------------------------------------------------
    left_combined = torch.nn.functional.linear(x_fp16, left_weight_cat)  # [B, N, N, 2*H]
    left_proj, left_gate_pre = torch.split(left_combined, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate_pre)
    left = left_proj * left_gate  # [B, N, N, H]

    # ------------------------------------------------------------------
    # ⑤ “Right” linear + gate (one GEMM → split).
    # ------------------------------------------------------------------
    right_combined = torch.nn.functional.linear(x_fp16, right_weight_cat)
    right_proj, right_gate_pre = torch.split(right_combined, hidden_dim, dim=-1)
    right_gate = torch.sigmoid(right_gate_pre)
    right = right_proj * right_gate  # [B, N, N, H]

    # ------------------------------------------------------------------
    # ⑥ Optional mask – fuse with a cheap Triton copy + broadcast.
    # ------------------------------------------------------------------
    if not nomask:
        # Convert mask to FP16 on the GPU (tiny Triton kernel) and broadcast.
        mask_fp16 = _copy_mask(mask_tensor).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # ⑦ Output‑gate (separate linear → sigmoid).
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_w))

    # ------------------------------------------------------------------
    # ⑧ O(N³) contraction via a batched GEMM (left @ rightᵀ).
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)  # [B, N, N, H] FP16

    # ------------------------------------------------------------------
    # ⑨ Second LayerNorm over the hidden dimension (FP16).
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # ⑩ Apply output gate and final projection back to the original dim.
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # [B, N, N, dim] (FP16)

    # ------------------------------------------------------------------
    # ⑪ Cast back to FP32 (required by the harness) and return.
    # ------------------------------------------------------------------
    return out.to(torch.float32)
