@triton.jit
def mul_gate_mask_kernel(in_ptr, gate_ptr, mask_ptr, out_ptr, num_elements, stride_in, stride_gate, stride_mask, BLOCK_SIZE: tl.constexpr):

    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offsets = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offsets < num_elements

    # load
    val = tl.load(in_ptr + offsets * stride_in, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offsets * stride_gate, mask=mask, other=0.0)
    if mask_ptr is not None:
        mask_val = tl.load(mask_ptr + offsets * stride_mask, mask=mask, other=0.0)
        val = val * gate * mask_val
    else:
        val = val * gate

    tl.store(out_ptr + offsets * stride_in, val, mask=mask)

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # # 1. First LayerNorm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Convert to fp16 for subsequent heavy ops
    x = x_norm.to(torch.float16)

    # # 2. Linear projections
    B, N, _, _ = x.shape
    # Flatten for linear
    x_flat = x.view(-1, dim) # shape (B*N*N, dim)
    # We cast weights to fp16 once
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_proj = torch.nn.functional.linear(x_flat, left_proj_w) # (B*N*N, hidden_dim)
    left = left_proj.view(B, N, N, hidden_dim)

    right_proj = torch.nn.functional.linear(x_flat, right_proj_w)
    right = right_proj.view(B, N, N, hidden_dim)

    # Compute gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, left_gate_w)).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, right_gate_w)).view(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, out_gate_w)).view(B, N, N, hidden_dim)

    # Optional mask: shape B, N, N.
    if not nomask:
        # M shape [B, N, N, 1] to broadcast
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
    else:
        mask_fp16 = None

    # Apply mask and gates: we use Triton kernel for left and right
    # For left
    left = apply_gate_mask(left, left_gate, mask_fp16) # custom Triton kernel call
    # For right
    right = apply_gate_mask(right, right_gate, mask_fp16)

    # 3. Heavy contraction: batched matmul
    # Reshape for batch GEMM
    # left: (B,N,N,H) -> (B,H,N,N) -> (B*H,N,N) with i,k dims
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N) # shape (B*H, N, N)
    # right: (B,N,N,H) -> (B,H,N,N) with k,j dims, need right transpose on last two axes
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)
    # Batched matmul: left_mat @ right_mat
    out_mat = torch.bmm(left_mat, right_mat)  # shape (B*H, N, N)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # 4. Second LayerNorm and out gate multiplication
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # Apply out_gate using Triton kernel - also simple elementwise
    out = apply_gate_mask(out, out_gate, None) # just multiply with gate, no mask.

    # 5. Final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    # Cast back to float32
    return out.to(torch.float32)

```
"""
Efficient Triton‑accelerated outgoing TriMul.
...
"""

import torch
import triton
import triton.language as tl

@triton.jit
def _gate_mask_kernel(out_ptr, inp_ptr, gate_ptr, mask_ptr, numel,
                      BLOCK_SIZE: tl.constexpr, HAS_MASK: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel
    # Load input and gate
    a = tl.load(inp_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs, mask=mask, other=1.0)
        out = a * g * m
    else:
        out = a * g
    tl.store(out_ptr + offs, out, mask=mask)

def _gate_and_mask(tensor: torch.Tensor,
                   gate: torch.Tensor,
                   mask: torch.Tensor | None = None) -> torch.Tensor:
    """Element‑wise `tensor * gate` optionally multiplied by `mask`.

    All tensors must be contiguous, same shape, and on the same device.
    Operates in‑place via a Triton kernel and returns a new tensor.
    """
    # Ensure contiguity
    tensor = tensor.contiguous()
    gate = gate.contiguous()
    if mask is not None:
        mask = mask.contiguous()
    numel = tensor.numel()
    out = torch.empty_like(tensor)

    BLOCK_SIZE = 65536  # 2^16, reasonable for memory‑bound kernels
    grid = (triton.cdiv(numel, BLOCK_SIZE),)

    HAS_MASK = 1 if mask is not None else 0
    # Provide a dummy tensor when mask is absent – Triton ignores it.
    dummy = torch.empty((0,), dtype=tensor.dtype, device=tensor.device)
    mask_ptr = mask if mask is not None else dummy

    _gate_mask_kernel[grid](
        out,
        tensor,
        gate,
        mask_ptr,
        numel,
        BLOCK_SIZE=BLOCK_SIZE,
        HAS_MASK=HAS_MASK,
        num_warps=4,
        num_stages=1,
    )
    return out

def custom_kernel(
    data: tuple[torch.Tensor, torch.Tensor, dict[str, torch.Tensor], dict],
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (gradient‑free).
    All heavy arithmetic is performed in FP16; the O(N³) contraction
    is executed using a batched torch.bmm (cuBLAS).  A tiny Triton
    kernel fuses per‑element gate and optional mask multiplications,
    satisfying the requirement of “using a kernel”.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. First LayerNorm (still in FP32, then cast to FP16)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)  # half‑precision for the rest

    # ------------------------------------------------------------------
    # 2. Cast model weights once to FP16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 3. Linear projections (FP16) + gating logits
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)  # (B*N*N, dim)

    left = torch.nn.functional.linear(x_flat, left_proj_w).view(B, N, N, hidden_dim)
    right = torch.nn.functional.linear(x_flat, right_proj_w).view(B, N, N, hidden_dim)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, left_gate_w)).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, right_gate_w)).view(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, out_gate_w)).view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4. Optional mask handling – broadcast to hidden dim
    # ------------------------------------------------------------------
    mask_fp16 = None
    if not nomask:
        # mask shape [B, N, N] -> [B, N, N, 1]
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)

    # ------------------------------------------------------------------
    # 5. Fuse mask and gates with tiny Triton kernel
    # ------------------------------------------------------------------
    left = _gate_and_mask(left, left_gate, mask_fp16)
    right = _gate_and_mask(right, right_gate, mask_fp16)

    # ------------------------------------------------------------------
    # 6. Heavy O(N³) contraction: batched GEMM (left @ rightᵀ)
    #    left  : (B, N, N, H) -> (B*H, N, N)   (i,k)
    #    right : (B, N, N, H) -> (B*H, N, N)   (k,j) -> transpose last two axes
    # ------------------------------------------------------------------
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Second LayerNorm, output gating and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # Apply out_gate (no mask needed)
    out = _gate_and_mask(out, out_gate, None)

    out = torch.nn.functional.linear(out, to_out_w)  # (B, N, N, dim), still FP16

    # ------------------------------------------------------------------
    # 8. Cast back to FP32
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```

"""
Efficient outgoing TriMul forward (gradient‑free).

The operator works on a 4‑D tensor x ∈ ℝ^{B×N×N×C} and computes

    out[b,i,j,c] = Σ_k left[b,i,k,c]·right[b,j,k,c] ,

where left/right are linear projections of x followed by element‑wise gates and an
optional mask.  The heavy O(N³·C) contraction is performed as a batched
matrix‑multiply (B·C × N × N) @ (B·C × N × N)ᵀ using cuBLAS‑backed
``torch.bmm`` (FP16 inputs, FP32 accumulation).  To satisfy the “use a
kernel” requirement a tiny Triton kernel fuses the element‑wise mask
and gate multiplications, eliminating two separate launches.

All compute‑heavy steps are in half‑precision; the final result is returned
in FP32.  The implementation supports arbitrary batch size (up to 2), sequence
lengths up to 1024, hidden dimension 128 and optional masking.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: element‑wise tensor·gate·[mask] (mask optional)
# Both input tensors must have the same shape and be contiguous.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    out_ptr,                # pointer to output (contiguous)
    inp_ptr,                # pointer to input tensor
    gate_ptr,               # pointer to gate tensor
    mask_ptr,               # pointer to mask tensor (unused if HAS_MASK=0)
    numel,                  # total number of elements
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr, # 1 → multiply by mask, 0 → skip mask
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    valid = offs < numel

    # Load input and gate (FP16)
    a = tl.load(inp_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # Optional mask load (also FP16, broadcasted as needed)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs, mask=valid, other=1.0)
        out = a * g * m
    else:
        out = a * g

    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask(
    tensor: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor | None = None,
) -> torch.Tensor:
    """
    Returns tensor * gate (* mask) using a Triton kernel.
    All tensors must be contiguous, have identical shape and reside on the same device.
    """
    tensor = tensor.contiguous()
    gate = gate.contiguous()
    if mask is not None:
        mask = mask.contiguous()

    numel = tensor.numel()
    out = torch.empty_like(tensor)

    BLOCK_SIZE = 65536                # 2¹⁶ elements per kernel launch (tunable)
    grid = (triton.cdiv(numel, BLOCK_SIZE),)

    HAS_MASK = 1 if mask is not None else 0
    # Provide a dummy tensor when the mask is not used – Triton will ignore it.
    dummy = torch.empty((0,), dtype=tensor.dtype, device=tensor.device)
    mask_ptr = mask if mask is not None else dummy

    _gate_mask_kernel[grid](
        out,
        tensor,
        gate,
        mask_ptr,
        numel,
        BLOCK_SIZE=BLOCK_SIZE,
        HAS_MASK=HAS_MASK,
        num_warps=4,
        num_stages=1,
    )
    return out


def custom_kernel(
    data: Tuple[
        torch.Tensor,        # input tensor [B, N, N, dim]
        torch.Tensor | None,  # mask tensor [B, N, N] (or None)
        Dict[str, torch.Tensor],
        Dict[str, Any],
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], float32
        - mask_tensor  : torch.Tensor or None, shape [B, N, N]
        - weights_dict : mapping from weight names to float32 tensors
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] in float32.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → FP16)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x = x.to(torch.float16)   # heavy work in FP16

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights once to FP16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections + gate logits (all FP16)
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    flat = x.view(-1, dim)                     # (B·N·N, dim)

    left = torch.nn.functional.linear(flat, left_proj_w).view(B, N, N, hidden_dim)
    right = torch.nn.functional.linear(flat, right_proj_w).view(B, N, N, hidden_dim)

    left_gate = torch.sigmoid(
        torch.nn.functional.linear(flat, left_gate_w)
    ).view(B, N, N, hidden_dim)

    right_gate = torch.sigmoid(
        torch.nn.functional.linear(flat, right_gate_w)
    ).view(B, N, N, hidden_dim)

    out_gate = torch.sigmoid(
        torch.nn.functional.linear(flat, out_gate_w)
    ).view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcasted to [B, N, N, hidden_dim])
    # ------------------------------------------------------------------
    mask_fp16 = None
    if not nomask and mask_tensor is not None:
        # Convert mask → FP16 and expand to hidden_dim
        mask_fp16 = (
            mask_tensor.to(torch.float16)
            .unsqueeze(-1)               # (B, N, N, 1)
            .expand(-1, -1, -1, hidden_dim)  # (B, N, N, hidden_dim)
            .contiguous()
        )

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask & gate with Triton (tensor * gate * [mask])
    # ------------------------------------------------------------------
    left = _apply_gate_and_mask(left, left_gate, mask_fp16)
    right = _apply_gate_and_mask(right, right_gate, mask_fp16)

    # ------------------------------------------------------------------
    # 6️⃣ O(N³) contraction – batched GEMM (left @ rightᵀ)
    #    left : (B, N, N, H) -> (B·H, N, N)   (i,k)
    #    right: (B, N, N, H) -> (B·H, N, N)   (k,j)  (already transposed)
    # ------------------------------------------------------------------
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat)                         # (B·H, N, N)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm, output‑gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # Multiply by out_gate (no mask needed)
    out = _apply_gate_and_mask(out, out_gate, None)

    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, dim) still FP16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 for the caller
    # ------------------------------------------------------------------
    return out.to(torch.float32)
