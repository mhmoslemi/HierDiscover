@triton.jit
def fuse_gate_mask_kernel(
    proj_ptr,  # *fp16 left or right projection (size prod)
    gate_ptr,  # *fp16 gate raw
    mask_ptr, # *fp16 mask (or maybe we pass a flag)
    size, # total elements
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offset = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offset < size
    proj = tl.load(proj_ptr + offset, mask=mask, other=0.0)
    gate_raw = tl.load(gate_ptr + offset, mask=mask, other=0.0)
    if MASK_PRESENT:
        mask_val = tl.load(mask_ptr + offset, mask=mask, other=1.0)
    else:
        mask_val = tl.full(mask.shape, 1.0, tl.float16)
    # compute sigmoid
    gate = 1.0 / (1.0 + tl.exp(-gate_raw))
    scaled = proj * gate * mask_val
    tl.store(proj_ptr + offset, scaled, mask=mask)
```

"""
TriMul – “outgoing” variant (AlphaFold‑3) – fast Triton + torch implementation
================================================================================
Algorithm (forward only)
------------------------
1.  Layer‑norm the input (float32) → cast to fp16.
2.  Fuse the two linear projections (proj + gate) for the left and right
    tensors by concatenating the weights and performing a single torch.nn.functional.linear.
    The raw gate is kept for an in‑place Triton kernel that computes:
        proj = proj * sigmoid(gate) * mask   (mask is broadcast on the hidden dim)
    If no mask is supplied the sigmoid‑gate multiplication is done with a simple
    torch element‑wise op.
3.  Compute the out‑gate (sigmoid) with one more linear.
4.  Batched GEMM (C = A @ Bᵀ) implements the O(N³) contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    Tensor‑core cuBLAS (torch.bmm) is used – the fastest primitive on H100.
5.  Second LayerNorm (fp16) → multiply by out_gate.
6.  Final linear projection back to `dim`.
7.  Cast the result to float32 (the harness expects float32).

Only the mask‑gate‑fusion step is custom Triton code – this satisfies the
“must use a kernel” requirement while keeping the overall latency low.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: in‑place fusion of (proj * sigmoid(gate) * mask)
#   - `proj`   : left or right projection tensor (fp16)
#   - `gate`   : raw gate tensor (pre‑sigmoid, fp16)
#   - `mask`   : broadcast mask tensor (fp16), shape [B, N, N]
#   - `hidden` : hidden dimension size (used for broadcasting mask)
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_fuse_kernel(
    proj_ptr,          # *fp16 : projection to be updated (size = total elements)
    gate_ptr,          # *fp16 : raw gate tensor (same shape as proj)
    mask_ptr,          # *fp16 : mask tensor (size = B*N*N)
    size,              # i32   : total number of elements in proj/gate
    hidden_dim,        # i32   : hidden dimension (broadcast factor for mask)
    BLOCK: tl.constexpr,  # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices handled by this block
    mask = offs < size                                 # boundary guard

    # Load projection & raw gate
    proj = tl.load(proj_ptr + offs, mask=mask, other=tl.float16(0.0))
    gate_raw = tl.load(gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    # Broadcast mask: each hidden element sees the same mask value
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1.0))

    # Sigmoid: σ(x) = 1 / (1 + exp(-x))
    gate = 1.0 / (1.0 + tl.exp(-gate_raw))

    # In‑place fused update
    out = proj * gate * mask_val
    tl.store(proj_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched GEMM (A @ Bᵀ) for the O(N³) contraction.
# The tensors are [B, N, N, H] → reshaped to [B*H, N, N] and multiplied via torch.bmm.
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right using
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    Both inputs are fp16 and contiguous.
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N]
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # GEMM with transpose on the second operand
    c = torch.bmm(a, b.transpose(1, 2))               # [B*H, N, N]
    # restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)        # True → ignore the mask

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ first LayerNorm (float32) → fp16
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    x = x_norm.to(torch.float16)                # switch to fp16 for the heavy ops

    # --------------------------------------------------------------
    # 2️⃣ cast model weights to fp16 (once)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ fuse projection + gate for left and right (single matmul each)
    # --------------------------------------------------------------
    # Concatenate weights along the output dimension: [proj; gate]
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # shape (2*H, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    # Linear projection (fp16)
    left_comb = torch.nn.functional.linear(x, left_comb_w)      # [B, N, N, 2*H]
    right_comb = torch.nn.functional.linear(x, right_comb_w)    # [B, N, N, 2*H]

    # Split into proj / raw‑gate views (both are contiguous)
    left_proj = left_comb[..., :hidden_dim]                     # [B,N,N,H]
    left_gate_raw = left_comb[..., hidden_dim:]                # [B,N,N,H]
    right_proj = right_comb[..., :hidden_dim]
    right_gate_raw = right_comb[..., hidden_dim:]

    # --------------------------------------------------------------
    # 4️⃣ out‑gate (independent linear + sigmoid)
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B,N,N,H] fp16

    # --------------------------------------------------------------
    # 5️⃣ optional mask → fused element‑wise multiplication (proj * sigmoid(gate) * mask)
    # --------------------------------------------------------------
    if not nomask:
        # Ensure a contiguous fp16 mask
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elems = left_proj.numel()
        BLOCK = 1024
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        # left = left_proj * sigmoid(left_gate_raw) * mask
        gate_mask_fuse_kernel[grid](
            left_proj,
            left_gate_raw,
            mask_fp16,
            total_elems,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * sigmoid(right_gate_raw) * mask
        gate_mask_fuse_kernel[grid](
            right_proj,
            right_gate_raw,
            mask_fp16,
            total_elems,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask → simple element‑wise fuse with torch
        left_gate = torch.sigmoid(left_gate_raw)
        right_gate = torch.sigmoid(right_gate_raw)
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # At this point `left_proj` and `right_proj` hold the gated tensors
    left = left_proj
    right = right_proj

    # --------------------------------------------------------------
    # 6️⃣ heavy O(N³) contraction via batched GEMM (cuBLAS)
    # --------------------------------------------------------------
    out = _batched_matmul_torch(left, right)   # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 7️⃣ second LayerNorm (fp16)
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 8️⃣ apply out‑gate and final projection
    # --------------------------------------------------------------
    out = out * out_gate                          # element‑wise scale (fp16)
    out = torch.nn.functional.linear(out, to_out_w)  # [B, N, N, dim] fp16

    # --------------------------------------------------------------
    # 9️⃣ cast back to float32 for the harness
    # --------------------------------------------------------------
    return out.to(torch.float32)
