"""TriMul outgoing kernel – fast forward pass (no grads)

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input in FP32 (numerical stability).
2️⃣  Cast to FP16 – the heavy part runs completely in half‑precision.
3️⃣  Four linear projections (left / right) and three gates (left, right,
    out).  All weights are cast to FP16 and the linear + sigmoid are
    performed with torch primitives.
4️⃣  Optional mask (broadcast on the hidden dimension) is applied *before*
    gating (as in the reference implementation).
5️⃣  Fuse the gates into the projections ∈‑place.
6️⃣  O(N³) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM C = A @ Bᵀ where A and B are reshaped to
   [B·H, N, N] (H = hidden_dim).  A custom Triton kernel launches a
   2‑level tiled MMA (BLOCK_M×BLOCK_N×BLOCK_K).  The kernel uses
   fp16 × fp16 → fp32 accumulation (Tensor‑core friendly) and stores the
   result back in fp16.  For N ≤ 128 we fall back to torch.einsum to
   avoid launch overhead.
7️⃣  Second LayerNorm (fp16) → out‑gate (sigmoid) → final linear
   projection back to the original `dim`.
8️⃣  Cast the result back to FP32 (the harness expects float32).

Only the O(N³) contraction is written in Triton; all other ops use highly
optimized PyTorch primitives.  The kernel is tuned for the H100 (SM₉₀):
  * for N > 256 we use 256×256 tiles, BLOCK_K = 128 and 16 warps;
  * otherwise 128×128 tiles, BLOCK_K = 64 and 8 warps.
  * three pipeline stages (`num_stages=3`) hide memory latency.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : [M, N, K]   (M = B * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]   (result stored in fp16)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # program‑size constants
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile on i‑dimension
    pid_n = tl.program_id(1)   # tile on j‑dimension
    pid_z = tl.program_id(2)   # batch*hidden index (M)

    # ------------------------------------------------------------------
    # Tile start indices
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # FP32 accumulator (better numeric stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # --------------------------------------------------------------
        # Load A tile (BLOCK_M × BLOCK_K)
        # --------------------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # --------------------------------------------------------------
        # Load B tile (BLOCK_N × BLOCK_K) – stored in row‑major order
        # --------------------------------------------------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_j[:, None] * stride_bj
            + offs_k[None, :] * stride_bk
        )
        b = tl.load(
            b_ptrs,
            mask=mask_j[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # Mixed‑precision dot: a (M × K)  •  bᵀ (K × N)  →  (M × N)
        # `trans_b=True` tells Triton to treat `b` as transposed.
        acc += tl.dot(a, b, trans_b=True)

    # ------------------------------------------------------------------
    # Store the tile (fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )

# Helper that prepares the view, launches the kernel and restores layout.
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a fused Triton GEMM (A @ Bᵀ).  Input tensors must be
    contiguous fp16 with shape [B, N, N, H].

    Returns a tensor of the same shape (fp16).
    """
    B, N, _, H = left.shape

    # Merge batch and hidden dimensions → [B·H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in *elements*
    stride_am, stride_ai, stride_ak = A.stride()   # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()  # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()   # (N*N, N, 1)

    # Heuristic tiles
    if N <= 256:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8
    else:
        BLOCK_M = BLOCK_N = 256
        BLOCK_K = 128
        num_warps = 16

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=3,
    )

    # Restore original layout [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

# ----------------------------------------------------------------------
# Triton‑accelerated TriMul – outgoing version (no gradients)
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (inference only).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack and move everything to the target device
    # -----------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    device = x.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim]  fp32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part of the pipeline
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Cast model weights once (fp16, same device)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,H]

    # -----------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast on hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -----------------------------------------------------------------
    # 6️⃣ Gate computations (sigmoid) – fp16
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Fuse gates into the projections in‑place
    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    if N <= 128:
        # For tiny N the Triton launch overhead dominates.
        out_hidden = torch.einsum('b i k h, b j k h -> b i j h', left, right)
    else:
        out_hidden = _batched_matmul_transpose_fp16(left, right)   # fp16

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    out_hidden = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Final gating + projection back to `dim` (fp16)
    # -----------------------------------------------------------------
    out_hidden = out_hidden * out_gate
    out = torch.nn.functional.linear(out_hidden, to_out_w)   # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 (the harness expects FP32)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
