def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. layer norm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Cast to fp16
    x = x.to(torch.float16)

    B, N, _, _ = x.shape
    M = B * N * N

    # cast weights (less memory overhead)
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(torch.float16)

    # flatten input for linear
    x_flat = x.view(M, dim)

    # Compute projections and gates
    left_proj = torch.nn.functional.linear(x_flat, left_proj_weight)  # (M, hidden)
    right_proj = torch.nn.functional.linear(x_flat, right_proj_weight)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, out_gate_weight)).view(B, N, N, hidden_dim)

    # mask handling
    if not nomask:
        mask = mask_tensor.view(M).to(torch.float16)   # shape (M)
        # allocate fused tensors
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)
        # stride info
        stride_lp_m, stride_lp_h = left_proj.stride()
        stride_lg_m, stride_lg_h = left_gate.stride()
        stride_l_o_m, stride_l_o_h = left.stride()
        stride_rp_m, stride_rp_h = right_proj.stride()
        stride_rg_m, stride_rg_h = right_gate.stride()
        stride_r_o_m, stride_r_o_h = right.stride()
        stride_mask = mask.stride()[0]

        BLOCK_M = 64
        BLOCK_H = hidden_dim
        grid = ((M + BLOCK_M - 1)//BLOCK_M,)

        _mask_gate_fuse_kernel[grid](
            left_proj, left_gate, mask, left,
            M, hidden_dim,
            stride_lp_m, stride_lp_h,
            stride_lg_m, stride_lg_h,
            stride_l_o_m, stride_l_o_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
        _mask_gate_fuse_kernel[grid](
            right_proj, right_gate, mask, right,
            M, hidden_dim,
            stride_rp_m, stride_rp_h,
            stride_rg_m, stride_rg_h,
            stride_r_o_m, stride_r_o_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
        # reshape
        left = left.view(B, N, N, hidden_dim)
        right = right.view(B, N, N, hidden_dim)
    else:
        left = left_proj * left_gate
        left = left.view(B, N, N, hidden_dim)
        right = right_proj * right_gate
        right = right.view(B, N, N, hidden_dim)

    # contraction: batch GEMM
    # reshape to (B*hidden, N, N)
    left_mat = left.permute(0,3,1,2).reshape(B*hidden_dim, N, N)
    right_mat = right.permute(0,3,1,2).reshape(B*hidden_dim, N, N)

    # compute out_mat = left_mat @ right_mat.T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    # reshape back

"""
Optimised outgoing TriMul (Triangle Multiplicative Update) forward pass.

Algorithm
---------
1️⃣ Layer‑norm the input tensor (FP32 → FP16) – per‑position normalisation over
    the feature dimension `dim`.

2️⃣ Compute the four linear projections (left/right · proj/gate) and the output
    gate in half‑precision.  When a mask is present we fuse the element‑wise
    mask, the projection, and the sigmoid‑gate with a tiny Triton kernel
    (`_mask_gate_fuse_kernel`).  When `nomask=True` a simple PyTorch
    element‑wise multiplication is used.

3️⃣ The dominant O(N³) contraction:
       out_{i,j,d} = Σ_k left_{i,k,d} · right_{j,k,d}
   is performed as a batched matrix multiplication
   `left @ rightᵀ` using `torch.bmm`.  The tensors are reshaped to
   (B·hidden, N, N) so that cuBLAS/Tensor‑Core GEMMs on H100 are fully
   exploited.

4️⃣ Second layer‑norm (over the hidden dimension), multiplication by the
   output gate and a final linear projection back to `dim` are carried out
   in FP16.  The final tensor is cast back to FP32 for the caller.

All heavy arithmetic is carried out in half‑precision; only the mask‑gate
fusion kernel is written in Triton, satisfying the requirement to use
Triton while keeping the O(N³) step on the highly optimised cuBLAS path.
"""

import math
from typing import Tuple, Dict, Any

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask, projection and gate (optional)
#   out = proj * gate * mask[...,None]   # proj,gate: (M, H), mask: (M,)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,          # tensors
    M, H,                                           # problem size
    stride_proj_m, stride_proj_h,                   # strides of proj
    stride_gate_m, stride_gate_h,                   # strides of gate
    stride_out_m,  stride_out_h,                    # strides of out
    stride_mask,                                    # stride of mask (1‑D)
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,   # tile sizes
):
    pid = tl.program_id(0)                     # block index over rows
    block_start = pid * BLOCK_M
    offs_m = block_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    offs_h = tl.arange(0, BLOCK_H)                       # [BLOCK_H]

    # Load mask (1‑D) and broadcast to H
    mask = tl.load(mask_ptr + offs_m * stride_mask,
                   mask=offs_m < M, other=0.0).to(tl.float16)   # [BLOCK_M]
    mask_h = mask[:, None]                                   # [BLOCK_M,1]

    # Load projection and gate tiles
    proj = tl.load(
        proj_ptr + offs_m[:, None] * stride_proj_m
                  + offs_h[None, :] * stride_proj_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    gate = tl.load(
        gate_ptr + offs_m[:, None] * stride_gate_m
                  + offs_h[None, :] * stride_gate_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    # Fusion (gate already contains sigmoid)
    out = proj * gate * mask_h

    # Store result
    tl.store(
        out_ptr + offs_m[:, None] * stride_out_m
                 + offs_h[None, :] * stride_out_h,
        out,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
    )


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N]   (bool/float)
        - weights_dict : mapping from module parameter names to tensors
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device as input.
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # LayerNorm over last dimension ``dim``
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    ).to(torch.float16)   # [B, N, N, dim] in FP16

    B, N, _, _ = x.shape
    M = B * N * N                 # flattened pairwise count

    # ------------------------------------------------------------------
    # 2️⃣ Cast all linear weights to FP16 once
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left_gate_weight  = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight   = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(torch.float16)
    to_out_weight      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections & sigmoid gates (FP16)
    # ------------------------------------------------------------------
    x_flat = x.view(M, dim)   # (M, dim)

    # Separate linear calls – important to keep shapes simple
    left_proj = torch.nn.functional.linear(x_flat, left_proj_weight)      # (M, hidden)
    right_proj = torch.nn.functional.linear(x_flat, right_proj_weight)    # (M, hidden)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, left_gate_weight))   # (M, hidden)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, right_gate_weight)) # (M, hidden)

    # Output gate (used after the second LayerNorm)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, out_gate_weight))
    out_gate = out_gate.view(B, N, N, hidden_dim)   # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask + gate fusion (Triton)
    # ------------------------------------------------------------------
    if not nomask:
        # Flatten mask to (M,)
        mask_fp16 = mask_tensor.to(torch.float16).view(-1)

        # Allocate fused tensors
        left  = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # Strides (in element counts)
        stride_lp_m, stride_lp_h = left_proj.stride()
        stride_lg_m, stride_lg_h = left_gate.stride()
        stride_l_o_m, stride_l_o_h = left.stride()

        stride_rp_m, stride_rp_h = right_proj.stride()
        stride_rg_m, stride_rg_h = right_gate.stride()
        stride_r_o_m, stride_r_o_h = right.stride()

        stride_mask = mask_fp16.stride()[0]

        # Triton grid configuration (tuned for H100)
        BLOCK_M = 64                # rows per launch block
        BLOCK_H = hidden_dim        # tile the full hidden dim (≤128)
        grid = ((M + BLOCK_M - 1) // BLOCK_M,)

        # left = left_proj * left_gate * mask
        _mask_gate_fuse_kernel[grid](
            left_proj, left_gate, mask_fp16, left,
            M, hidden_dim,
            stride_lp_m, stride_lp_h,
            stride_lg_m, stride_lg_h,
            stride_l_o_m, stride_l_o_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
        # right = right_proj * right_gate * mask
        _mask_gate_fuse_kernel[grid](
            right_proj, right_gate, mask_fp16, right,
            M, hidden_dim,
            stride_rp_m, stride_rp_h,
            stride_rg_m, stride_rg_h,
            stride_r_o_m, stride_r_o_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
        # Reshape to 4‑D tensors
        left  = left.view(B, N, N, hidden_dim)
        right = right.view(B, N, N, hidden_dim)
    else:
        # No mask – simple element‑wise fuse (fast)
        left  = (left_proj * left_gate).view(B, N, N, hidden_dim)
        right = (right_proj * right_gate).view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ O(N³) contraction via batched GEMM (cuBLAS/TensorCore)
    #    Compute out = left @ rightᵀ for each (batch, hidden) pair.
    # ------------------------------------------------------------------
    # Permute to (B, hidden, N, N) → flatten batch*hidden dim
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Batched matrix multiplication (C = A @ Bᵀ)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Output gate + final linear projection (FP16)
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # (B, N, N, dim) FP16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 (expected by the caller)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
