"""
TriMul (outgoing) – fast forward pass implementation
===================================================

The following kernel implements the “outgoing” TriMul operator from
AlphaFold‑3.  The heavy O(N³) contraction is performed with a
batched GEMM (torch.bmm) on half‑precision tensors – this is the fastest
approach on H100 because cuBLAS/Tensor‑Core pipelines are heavily
optimised for dense matrix multiplication.

Key performance tricks
----------------------
* **Linear fusion** – the left/right projection and their corresponding
  gates share the same input.  We concatenate the two weight matrices
  (proj + gate) and perform a single half‑precision matmul, then split
  the result.  This reduces the number of GEMMs from 5 to 3.
* **Mask‑gate‑proj fusion** – when a mask is present we fuse the element‑wise
  mask multiplication together with the gate and projection using a tiny
  Triton kernel (`_mask_gate_fuse_kernel`).  This saves two extra
  PyTorch element‑wise passes.
* **Mixed‑precision** – all heavy arithmetic (LayerNorm, linear layers,
  GEMM, gating) is carried out in `float16`; only the final result is
  cast back to `float32` as required by the caller.
* **Batched GEMM** – the O(N³) contraction \
  `out[b,i,j] = Σₖ left[b,i,k] * right[b,j,k]` \
  is realised as a batched matrix multiplication
  `out = left @ rightᵀ` on the reshaped tensors of shape
  `(B*hidden, N, N)`.  This fully exploits Tensor‑Core throughput.

The implementation works for all shapes described in the problem
statement (B∈{1,2}, N up to 1024, dim={128,384},
hidden_dim=128) and handles both masked and un‑masked inputs.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, gate and projection (optional)
#   out = proj * gate * mask[...,None]   # proj,gate: (M, H), mask: (M,)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,        # tensors
    M, H,                                         # problem size
    stride_proj_m, stride_proj_h,                 # strides of proj
    stride_gate_m, stride_gate_h,                 # strides of gate
    stride_out_m, stride_out_h,                   # strides of out
    stride_mask,                                  # stride of mask (1‑D)
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr, # tile sizes
):
    pid = tl.program_id(0)  # block index over rows (M dimension)
    block_start = pid * BLOCK_M
    offs_m = block_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    offs_h = tl.arange(0, BLOCK_H)                              # [BLOCK_H]

    # Load mask (1‑D) and broadcast to H
    mask = tl.load(mask_ptr + offs_m * stride_mask,
                   mask=offs_m < M, other=0.0).to(tl.float16)  # [BLOCK_M]
    mask_h = mask[:, None]                                      # [BLOCK_M,1]

    # Load projection and gate tiles (M x H)
    proj = tl.load(
        proj_ptr + offs_m[:, None] * stride_proj_m
                + offs_h[None, :] * stride_proj_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    gate = tl.load(
        gate_ptr + offs_m[:, None] * stride_gate_m
                + offs_h[None, :] * stride_gate_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )

    # Fuse: proj * gate * mask
    out = proj * gate * mask_h

    # Store result
    tl.store(
        out_ptr + offs_m[:, None] * stride_out_m
                 + offs_h[None, :] * stride_out_h,
        out,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
    )

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32 or bool)
        - weights_dict : {name: torch.Tensor} (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device as
        ``input_tensor``.
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias   = weights["norm.bias"].to(torch.float16)

    # Apply LayerNorm in fp32 then cast – this keeps LN numerically stable
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight.to(torch.float32), bias=norm_bias.to(torch.float32)
    ).to(torch.float16)          # [B, N, N, dim]  (fp16)

    B, N, _, _ = x.shape
    M = B * N * N                               # flattened pair count

    # --------------------------------------------------------------
    # 2️⃣ Prepare FP16 linear weights (combine proj+gate for left/right)
    # --------------------------------------------------------------
    # left: (proj, gate) → (2*hidden, dim)
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)      # (2*hidden, dim)

    # right: (proj, gate) → (2*hidden, dim)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)      # (2*hidden, dim)

    # out_gate weight (hidden, dim)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # Output linear weights
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)   # (dim, hidden)

    # --------------------------------------------------------------
    # 3️⃣ Linear transforms (single matmul for proj+gate)
    # --------------------------------------------------------------
    x_flat = x.view(-1, dim)                      # (M, dim)

    # Left: [proj | gate] → (M, 2*hidden)
    left_comb = torch.nn.functional.linear(x_flat, left_comb_w)   # (M, 2*hidden)
    left_proj, left_gate = torch.split(left_comb, hidden_dim, dim=1)

    # Right: [proj | gate] → (M, 2*hidden)
    right_comb = torch.nn.functional.linear(x_flat, right_comb_w) # (M, 2*hidden)
    right_proj, right_gate = torch.split(right_comb, hidden_dim, dim=1)

    # Out‑gate (separate linear + sigmoid)
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, out_gate_w)
    ).view(B, N, N, hidden_dim)                # (B, N, N, hidden)

    # --------------------------------------------------------------
    # 4️⃣ Optional mask + elementwise fusion (Triton)
    # --------------------------------------------------------------
    if nomask:
        # No mask – simple elementwise multiply in PyTorch
        left  = (left_proj * left_gate).view(B, N, N, hidden_dim)   # (B,N,N,hidden)
        right = (right_proj * right_gate).view(B, N, N, hidden_dim)
    else:
        # Flatten mask and cast to fp16
        mask_fp16 = mask_tensor.to(torch.float16).view(-1)  # (M,)

        # Allocate fused output buffers
        left  = torch.empty_like(left_proj)   # will hold fused result
        right = torch.empty_like(right_proj)

        # Strides (in elements) for the kernel
        lp_m, lp_h = left_proj.stride()
        lg_m, lg_h = left_gate.stride()
        rp_m, rp_h = right_proj.stride()
        rg_m, rg_h = right_gate.stride()
        lo_m, lo_h = left.stride()
        ro_m, ro_h = right.stride()
        mask_stride = mask_fp16.stride()[0]

        # Kernel launch configuration
        BLOCK_M = 64                          # rows per block (tuned)
        BLOCK_H = hidden_dim                  # tile full hidden dim (≤384)

        grid = ( (M + BLOCK_M - 1) // BLOCK_M, )

        # left = left_proj * left_gate * mask
        _mask_gate_fuse_kernel[grid](
            left_proj, left_gate, mask_fp16, left,
            M, hidden_dim,
            lp_m, lp_h,
            lg_m, lg_h,
            lo_m, lo_h,
            mask_stride,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
        # right = right_proj * right_gate * mask
        _mask_gate_fuse_kernel[grid](
            right_proj, right_gate, mask_fp16, right,
            M, hidden_dim,
            rp_m, rp_h,
            rg_m, rg_h,
            ro_m, ro_h,
            mask_stride,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )

        # Reshape back to (B,N,N,hidden)
        left  = left.view(B, N, N, hidden_dim)
        right = right.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5️⃣ O(N³) contraction via batched GEMM
    #    out = left @ rightᵀ   (per‑hidden channel)
    # --------------------------------------------------------------
    # Reshape to (B*hidden, N, N) for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiplication (C = A @ Bᵀ)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # Restore original layout: (B, N, N, hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b
    )

    # --------------------------------------------------------------
    # 7️⃣ Output gating + final linear projection (FP16)
    # --------------------------------------------------------------
    out = out * out_gate                           # elementwise gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim)

    # --------------------------------------------------------------
    # 8️⃣ Cast back to FP32 (expected by the caller)
    # --------------------------------------------------------------
    return out.to(torch.float32)
