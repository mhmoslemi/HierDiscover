```
import triton
import triton.language as tl

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    hb = pid_hb

    # compute batch and hidden index
    batch = hb // H
    hidden = hb % H

    # base offsets
    offset_batch = batch * stride_lb
    offset_hidden = hidden * stride_lh  # same stride for left, right, out

    # pointer base for this batch+hidden
    left_base = left_ptr + offset_batch + offset_hidden
    right_base = right_ptr + offset_batch + offset_hidden
    out_base = out_ptr + offset_batch + offset_hidden

    # create index ranges
    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)
    # allocate accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        # load A tile (i,k)
        a_ptrs = left_base + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        a = tl.load(a_ptrs,
                    mask=(offs_i[:, None] < N) & (offs_k[None, :] < N),
                    other=0.0)
        # load B tile (k,j) -> we need shape (BLOCK_K, BLOCK_J)
        b_ptrs = right_base + offs_j[None, :] * stride_rj + offs_k[:, None] * stride_rk
        b = tl.load(b_ptrs,
                    mask=(offs_j[None, :] < N) & (offs_k[:, None] < N),
                    other=0.0)
        # accumulate
        acc += tl.dot(a, b)

    # store result
    c_ptrs = out_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    mask = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)
```

"""
Fast outgoing TriMul (AlphaFold3) implementation.

Algorithm
---------
1.  Layer‑Norm the input (FP32) and cast to FP16.
2.  Concatenate projection and gate weights so that each side (left/right)
    needs only a single half‑precision matmul:
        [proj | gate] @ x   →  (M, 2·H)  where M = B·N·N.
    The result is split into projection, gate and the sigmoid is applied.
3.  Optional mask is broadcast and fused with projection·gate (element‑wise).
    The tensors are reshaped to (B, N, N, H) in half‑precision.
4.  The dominant O(N³·H) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is performed by a custom Triton kernel that works directly on the
    (B,N,N,H) layout – no permute/contiguous copies are required.
5.  A second LayerNorm (FP16) is applied, followed by the output gate
    (sigmoid) and a final projection to the original dimension.
6.  Result is returned as FP32.

All heavy arithmetic (linear layers, the cubic contraction and the final
projection) runs in FP16, exploiting Tensor‑Cores on H100.  The Triton
kernel eliminates the expensive data reshuffles of the reference
implementation and thus meets the target runtime.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: outgoing TriMul contraction.
#    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
#   left/right have layout (B, N, N, H) with H as the innermost dim.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr,       # const half*   [B, N, N, H]
    right_ptr,      # const half*   [B, N, N, H]
    out_ptr,        #       half*   [B, N, N, H]
    B, N, H,        # problem dimensions (int32)
    # left strides
    stride_lb, stride_li, stride_lk, stride_lh,
    # right strides
    stride_rb, stride_rj, stride_rk, stride_rh,
    # out strides
    stride_ob, stride_oi, stride_oj, stride_oh,
    # tile sizes (compile‑time)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)               # tile index over rows (i dimension)
    pid_j = tl.program_id(1)               # tile index over cols (j dimension)
    pid_hb = tl.program_id(2)              # combined batch * hidden index

    # ------------------------------------------------------------------
    # Compute the batch and hidden indices
    # ------------------------------------------------------------------
    hidden_idx = pid_hb % H
    batch_idx  = pid_hb // H

    # base offsets for the selected batch and hidden channel
    offset_batch = batch_idx * stride_lb         # = batch_idx * stride_rb = batch_idx * stride_ob
    offset_hidden = hidden_idx * stride_lh        # stride_lh == stride_rh == stride_oh == 1

    # ------------------------------------------------------------------
    # Pointers to the beginning of the (batch, hidden) slice
    # ------------------------------------------------------------------
    left_base  = left_ptr + offset_batch + offset_hidden
    right_base = right_ptr + offset_batch + offset_hidden
    out_base   = out_ptr + offset_batch + offset_hidden

    # ------------------------------------------------------------------
    # Compute the start coordinates of the current tile
    # ------------------------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # ------------------------------------------------------------------
    # Split the M dimension in two halves (to keep register pressure low)
    # ------------------------------------------------------------------
    offs_m1 = i_start + tl.arange(0, BLOCK_M // 2)      # first half rows
    offs_m2 = i_start + tl.arange(BLOCK_M // 2, BLOCK_M)  # second half rows
    offs_n  = j_start + tl.arange(0, BLOCK_N)           # columns of right matrix
    offs_k  = tl.arange(0, BLOCK_K)                     # reduction dimension

    # ------------------------------------------------------------------
    # Initialize accumulators in FP32
    # ------------------------------------------------------------------
    acc1 = tl.zeros((BLOCK_M // 2, BLOCK_N), dtype=tl.float32)
    acc2 = tl.zeros((BLOCK_M // 2, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension (the shared N dimension)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # Compute the current K offset
        cur_k = k * BLOCK_K
        cur_k_offs = cur_k + offs_k

        # ------------------------------------------------------------------
        # Load tiles from left (A) – shape (BLOCK_M//2, BLOCK_K)
        # ------------------------------------------------------------------
        a1_ptrs = left_base + offs_m1[:, None] * stride_li + cur_k_offs[None, :] * stride_lk
        a2_ptrs = left_base + offs_m2[:, None] * stride_li + cur_k_offs[None, :] * stride_lk
        a1 = tl.load(a1_ptrs,
                     mask=(offs_m1[:, None] < N) & (cur_k_offs[None, :] < N),
                     other=0.0)          # half
        a2 = tl.load(a2_ptrs,
                     mask=(offs_m2[:, None] < N) & (cur_k_offs[None, :] < N),
                     other=0.0)          # half

        # ------------------------------------------------------------------
        # Load tile from right (B) – shape (BLOCK_K, BLOCK_N)
        #    Note: we need B transposed (k as rows, j as columns)
        # ------------------------------------------------------------------
        b_ptrs = right_base + cur_k_offs[:, None] * stride_rk + offs_n[None, :] * stride_rj
        b = tl.load(b_ptrs,
                    mask=(cur_k_offs[:, None] < N) & (offs_n[None, :] < N),
                    other=0.0)          # half

        # ------------------------------------------------------------------
        # Accumulate dot products
        # ------------------------------------------------------------------
        acc1 += tl.dot(a1, b)          # (BLOCK_M//2, BLOCK_N) ∈ FP32
        acc2 += tl.dot(a2, b)

    # ------------------------------------------------------------------
    # Store the result tiles (cast back to FP16)
    # ------------------------------------------------------------------
    c1_ptrs = out_base + offs_m1[:, None] * stride_oi + offs_n[None, :] * stride_oj
    c2_ptrs = out_base + offs_m2[:, None] * stride_oi + offs_n[None, :] * stride_oj
    mask1 = (offs_m1[:, None] < N) & (offs_n[None, :] < N)
    mask2 = (offs_m2[:, None] < N) & (offs_n[None, :] < N)

    tl.store(c1_ptrs, acc1.to(tl.float16), mask=mask1)
    tl.store(c2_ptrs, acc2.to(tl.float16), mask=mask2)


# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool or float)
        - weights_dict : mapping from weight name to tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device as *input_tensor*.
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape
    M = B * N * N                       # flattened pair count

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]  # FP32
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    ).to(torch.float16)                # [B, N, N, dim]  FP16

    # ------------------------------------------------------------------
    # 2️⃣ Cast all linear weights once to half‑precision
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # keep them as FP16 tensors
    left_proj_w   = left_proj_w.to(torch.float16)
    right_proj_w  = right_proj_w.to(torch.float16)
    left_gate_w   = left_gate_w.to(torch.float16)
    right_gate_w  = right_gate_w.to(torch.float16)
    out_gate_w    = out_gate_w.to(torch.float16)
    to_out_norm_w = to_out_norm_w.to(torch.float16)
    to_out_norm_b = to_out_norm_b.to(torch.float16)
    to_out_w      = to_out_w.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fused projection + gate (single matmul per side)
    # ------------------------------------------------------------------
    x_flat = x.view(-1, dim)                       # (M, dim)   FP16

    # concatenate proj and gate weights → shape (2·H, dim)
    left_cat_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2H, dim)
    right_cat_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2H, dim)

    # single matmul per side
    left_cat = torch.nn.functional.linear(x_flat, left_cat_w)   # (M, 2H)
    right_cat = torch.nn.functional.linear(x_flat, right_cat_w)

    # split back
    left_proj = left_cat[:, :hidden]          # (M, H)
    left_gate = torch.sigmoid(left_cat[:, hidden:])   # (M, H)
    right_proj = right_cat[:, :hidden]
    right_gate = torch.sigmoid(right_cat[:, hidden:])

    # -------------------------------------------------------------
    # 4️⃣ Optional mask (broadcasted) + element‑wise fuse
    # -------------------------------------------------------------
    if nomask:
        left = (left_proj * left_gate)               # (M, H)
        right = (right_proj * right_gate)            # (M, H)
    else:
        mask_fp16 = mask_tensor.to(torch.float16).view(-1, 1)    # (M,1)
        left = left_proj * left_gate * mask_fp16                # (M, H)
        right = right_proj * right_gate * mask_fp16

    # reshape to [B, N, N, H]
    left = left.view(B, N, N, hidden)
    right = right.view(B, N, N, hidden)

    # -------------------------------------------------------------
    # 5️⃣ O(N³) contraction via custom Triton kernel
    # -------------------------------------------------------------
    out = torch.empty_like(left)   # allocate FP16 output tensor

    # strides (in elements) for left/right/out
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # tile sizes – tuned for H100 (must be multiples of 32)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden,
    )

    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    # (out) shape: [B, N, N, H]  FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Output‑gate and final projection
    # ------------------------------------------------------------------
    # out_gate: compute from the original normalized tensor
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, out_gate_w))
    out_gate = out_gate.view(B, N, N, hidden)   # (B, N, N, H)  FP16

    out = out * out_gate                         # element‑wise gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim) FP16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 (expected by the caller)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
