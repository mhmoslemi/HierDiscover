custom_kernel – out‑going TriMul (AlphaFold3)

Implements the forward pass of the “TriMul (outgoing)” module used in AlphaFold3.
The operator works on a 4‑D tensor [B, N, N, C] (C == dim) and produces a tensor
of the same shape.  All heavy compute is performed in FP16:
  • First LayerNorm  (float16)
  • Two fused linear+gate matmuls (left & right) – each a single GEMM whose
    output is split into a projection and a sigmoid‑gate.
  • Optional elementwise mask‑gate fusion performed by a tiny Triton kernel.
  • O(N³) contraction is a batched matrix‑multiply
        out[b, d, i, j] = Σₖ left[b, d, i, k] * right[b, d, j, k]
    realised with `torch.matmul` on tensors of shape (B, hidden, N, N),

  • Second LayerNorm, output‑gate and final linear projection.
The final result is cast back to float32 as required by the caller.

# Triton kernel: elementwise fusion of projection, gate and mask.
#   out = proj * gate * mask[...,None]

    M, H,                                        # problem size
    stride_proj_m, stride_proj_h,                # strides for proj
    stride_gate_m, stride_gate_h,                # strides for gate
    stride_out_m,  stride_out_h,                 # strides for out
    stride_mask,                                 # stride for mask (==1)
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,  # tile sizes
):
    pid = tl.program_id(0)                     # one block per row tile
    block_start = pid * BLOCK_M
    offs_m = block_start + tl.arange(0, BLOCK_M)      # [BLOCK_M]
    offs_h = tl.arange(0, BLOCK_H)                    # [BLOCK_H]

    # Load mask (scalar per row) and broadcast to H
    mask_val = tl.load(
        mask_ptr + offs_m * stride_mask,
        mask=offs_m < M,
        other=0.0,
    ).to(tl.float16)                                 # [BLOCK_M]
    mask_h = mask_val[:, None]                        # [BLOCK_M,1]

    # Load projection and gate tiles
    proj = tl.load(
        proj_ptr
        + offs_m[:, None] * stride_proj_m
        + offs_h[None, :] * stride_proj_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    gate = tl.load(
        gate_ptr
        + offs_m[:, None] * stride_gate_m
        + offs_h[None, :] * stride_gate_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )

    # Fuse: projection * gate * mask
    out = proj * gate * mask_h

    # Store result
    tl.store(
        out_ptr
        + offs_m[:, None] * stride_out_m
        + offs_h[None, :] * stride_out_h,
        out,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
    )


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32, bool or None)
        - weights_dict : mapping from weight name to torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device as the input.
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → FP16)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_w, bias=norm_b
    ).to(torch.float16)               # [B, N, N, dim]  FP16

    B, N, _, _ = x.shape
    M = B * N * N                       # flattened pair count

    # ------------------------------------------------------------------
    # 2️⃣ Cast all model weights to FP16
    # ------------------------------------------------------------------
    # combine proj+gate weights for left / right side (2*hidden x dim)
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                # (2*hidden, dim)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                # (2*hidden, dim)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden)

    # ------------------------------------------------------------------
    # 3️⃣ Joint linear+gate for left and right (FP16)
    # ------------------------------------------------------------------
    x_flat = x.view(-1, dim)                     # (M, dim)

    # left side: one GEMM → (M, 2*hidden) → split
    left_comb = torch.nn.functional.linear(x_flat, left_comb_w)   # (M, 2*hidden)
    left_proj = left_comb[:, :hidden_dim]                        # (M, hidden)
    left_gate_pre = left_comb[:, hidden_dim:]                    # (M, hidden)
    left_gate = torch.sigmoid(left_gate_pre)                     # (M, hidden)

    # right side: same pattern
    right_comb = torch.nn.functional.linear(x_flat, right_comb_w)
    right_proj = right_comb[:, :hidden_dim]
    right_gate_pre = right_comb[:, hidden_dim:]
    right_gate = torch.sigmoid(right_gate_pre)

    # out_gate (used later)
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, out_gate_w)
    ).view(B, N, N, hidden_dim)                                 # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask * gate * projection fusion (Triton)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is 1‑D after flattening, convert to FP16
        mask_fp16 = mask_tensor.to(torch.float16).view(-1)      # (M,)

        # Allocate fused results (contiguous)
        left = torch.empty((M, hidden_dim), dtype=torch.float16, device=device)
        right = torch.empty((M, hidden_dim), dtype=torch.float16, device=device)

        # strides (in element counts)
        stride_lp_m, stride_lp_h = left_proj.stride()
        stride_lg_m, stride_lg_h = left_gate.stride()
        stride_rp_m, stride_rp_h = right_proj.stride()
        stride_rg_m, stride_rg_h = right_gate.stride()
        stride_out_m, stride_out_h = left.stride()    # same for right
        stride_mask = mask_fp16.stride()[0]

        # Choose a reasonable tile size (tuned for H100 and hidden≤128)
        BLOCK_M = 256               # rows per launch block
        BLOCK_H = hidden_dim        # tile the whole hidden dimension

        grid = ((M + BLOCK_M - 1) // BLOCK_M,)

        # left = left_proj * left_gate * mask
        _mask_gate_fuse_kernel[grid](
            left_proj, left_gate, mask_fp16, left,
            M, hidden_dim,
            stride_lp_m, stride_lp_h,
            stride_lg_m, stride_lg_h,
            stride_out_m, stride_out_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
        # right = right_proj * right_gate * mask
        _mask_gate_fuse_kernel[grid](
            right_proj, right_gate, mask_fp16, right,
            M, hidden_dim,
            stride_rp_m, stride_rp_h,
            stride_rg_m, stride_rg_h,
            stride_out_m, stride_out_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
    else:
        # No mask → simple element‑wise multiplication (fast)
        left = (left_proj * left_gate).view(B, N, N, hidden_dim)
        right = (right_proj * right_gate).view(B, N, N, hidden_dim)

    # If we used the Triton path we must reshape now
    if not nomask:
        left = left.view(B, N, N, hidden_dim)
        right = right.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Batched contraction via torch.matmul (FP16)
    #    out_{i j d} = Σₖ left_{i k d} · right_{j k d}
    # ------------------------------------------------------------------
    # bring hidden dimension to the batch axis (no extra copy)
    left_perm = left.permute(0, 3, 1, 2)          # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2)        # (B, hidden, N, N)

    # out_perm shape (B, hidden, N, N)
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))

    # reshape back to (B, N, N, hidden)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Output gate + final projection (FP16)
    # ------------------------------------------------------------------
    out = out * out_gate                         # broadcast over hidden
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim) FP16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 (expected by the caller)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
