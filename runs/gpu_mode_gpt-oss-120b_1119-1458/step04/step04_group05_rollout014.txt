import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel for elementwise gating with mask.
@triton.jit
def _mask_gate_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                      total_elements, hidden,
                      BLOCK: tl.constexpr):
    # Global index
    idx = tl.program_id(0) * BLOCK + tl.arange(0, BLOCK)
    mask = idx < total_elements
    # Load values
    proj = tl.load(proj_ptr + idx, mask=mask)
    gate = tl.load(gate_ptr + idx, mask=mask)
    # Compute mask index: each hidden entries share a mask entry
    mask_idx = idx // hidden
    m = tl.load(mask_ptr + mask_idx, mask=mask)
    # Apply mask and gate
    out = proj * gate * m
    tl.store(out_ptr + idx, out, mask=mask)

def _apply_mask_gate(projected: torch.Tensor,
                     gate: torch.Tensor,
                     mask: torch.Tensor,
                     hidden_dim: int,
                     block: int = 128) -> torch.Tensor:
    """
    Fuse element‑wise mask and gating using a Triton kernel.
    *projected* and *gate* are contiguous FP16 tensors of shape [B,N,N,H].
    *mask* is FP16 tensor of shape [B,N,N] and is broadcast across H.
    Returns a new FP16 tensor of the same shape as *projected*.
    """
    total = projected.numel()
    out = torch.empty_like(projected)
    # Launch Triton kernel
    grid = (triton.cdiv(total, block),)
    _mask_gate_kernel[grid](
        projected.view(-1), gate.view(-1), mask.view(-1), out.view(-1),
        total,
        hidden_dim,
        BLOCK=block,
        num_warps=4,
        num_stages=2,
    )
    return out

def custom_kernel(
    data: Tuple[torch.Tensor,
                torch.Tensor,
                Dict[str, torch.Tensor],
                Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (gradient‑free).
    The heavy O(N³) contraction ``out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]``
    is performed with a highly‑optimized cuBLAS batched GEMM (torch.bmm).
    Linear projections and gating are fused into a single matmul; the optional
    mask+gate multiplication is carried out by a tiny Triton kernel.
    All compute‑intensive stages use FP16; the final tensor is returned as FP32.
    """
    # --------------------------------------------------------------
    # 1️⃣ Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2️⃣ First LayerNorm (float32 -> FP16)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim], float32
    x = x.to(torch.float16)  # switch to half‑precision for heavy work

    # --------------------------------------------------------------
    # 3️⃣ Cast model weights to FP16 once
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, hidden)

    # --------------------------------------------------------------
    # 4️⃣ One‑shot linear projection + gate computation
    # --------------------------------------------------------------
    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)  # (B·N·N, dim)

    # Concatenate all five weight matrices (targets: left, right, left_gate,
    # right_gate, out_gate). Biases are all zero by construction.
    concat_w = torch.cat([
        left_proj_w,
        right_proj_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
    ], dim=0)  # shape (5·hidden_dim, dim)

    # A single GEMM produces all five intermediate tensors.
    # Result shape: (B·N·N, 5·hidden_dim)
    all_z = torch.nn.functional.linear(x_flat, concat_w)

    # Split into the original heads.
    z_left, z_right, z_lgate, z_rgate, z_outgate = torch.split(
        all_z, hidden_dim, dim=1
    )
    left_proj = z_left.view(B, N, N, hidden_dim)
    right_proj = z_right.view(B, N, N, hidden_dim)
    left_gate = torch.sigmoid(z_lgate).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(z_rgate).view(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(z_outgate).view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5️⃣ Fuse mask (if present) and gating for left/right tensors
    # --------------------------------------------------------------
    if nomask:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        mask_fp16 = mask_tensor.to(torch.float16)
        left = _apply_mask_gate(left_proj, left_gate, mask_fp16, hidden_dim)
        right = _apply_mask_gate(right_proj, right_gate, mask_fp16, hidden_dim)

    # --------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction using batched GEMM (torch.bmm)
    #    left   : (B, N, N, H) → (B·H, N, N)   (i,k)
    #    right  : (B, N, N, H) → (B·H, N, N)   (j,k)   → transpose to (k,j)
    # --------------------------------------------------------------
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat)               # (B·H, N, N)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm, output gating and final projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # (B, N, N, dim) in FP16

    # --------------------------------------------------------------
    # 8️⃣ Cast back to FP32 (the caller expects float32)
    # --------------------------------------------------------------
    return out.to(torch.float32)

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# -------------------------------------------------------------------------
# Triton kernel: fuses mask, projection‑gate multiplication.
#   proj : FP16 tensor   [M]         (flattened left/right projection)
#   gate : FP16 tensor   [M]         (flattened left/right gate)
#   mask : FP16 tensor   [M // hidden]   (flattened mask, broadcasted)
#   out  : FP16 tensor   [M]         (result = proj * gate * mask)
# -------------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elems, hidden_dim,
    BLOCK: tl.constexpr,
):
    idx = tl.program_id(0) * BLOCK + tl.arange(0, BLOCK)
    valid = idx < total_elems

    # load projection and gate values
    proj = tl.load(proj_ptr + idx, mask=valid)
    gate = tl.load(gate_ptr + idx, mask=valid)

    # mask is shared among `hidden_dim` consecutive values
    mask_idx = idx // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid)

    out = proj * gate * m
    tl.store(out_ptr + idx, out, mask=valid)


def _apply_mask_gate(
    proj: torch.Tensor,
    gate: torch.Tensor,
    mask: torch.Tensor,
    hidden_dim: int,
    block: int = 128,
) -> torch.Tensor:
    """
    Apply ``proj * gate * mask`` using the Triton kernel above.

    All tensors must be contiguous FP16.
    `proj`/`gate` have shape [B, N, N, hidden_dim],
    `mask` has shape [B, N, N] and is broadcasted over the hidden dimension.
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    grid = (triton.cdiv(total, block),)
    _mask_gate_kernel[grid](
        proj.view(-1), gate.view(-1), mask.view(-1), out.view(-1),
        total,
        hidden_dim,
        BLOCK=block,
        num_warps=4,
        num_stages=2,
    )
    return out

    The heavy O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    linear projections/gates are fused into a single matmul to reduce memory

    """
    # --------------------------------------------------------------
    # 1️⃣ unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data

    hidden_dim = config["hidden_dim"]

        input_tensor,

        weight=norm_w,
        bias=norm_b,

    x = x.to(torch.float16)                # switch to FP16 for the heavy work

    # --------------------------------------------------------------
    # 3️⃣ cast all model weights to FP16 once

    right_proj_w = weights["right_proj.weight"].to(torch.float16)  # (hidden, dim)

    right_gate_w = weights["right_gate.weight"].to(torch.float16)  # (hidden, dim)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)    # (hidden, dim)

    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)    # (hidden,)
    to_out_w = weights["to_out.weight"].to(torch.float16)            # (dim, hidden)

    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)                # (B·N·N, dim)

    # --------------------------------------------------------------

    )                                        # (5·hidden, dim)

    # single matmul → (B·N·N, 5·hidden)
    all_z = torch.nn.functional.linear(x_flat, concat_w)

    # split back into the original heads
    z_left, z_right, z_lgate, z_rgate, z_outgate = torch.split(
        all_z, hidden_dim, dim=1

    right_gate = torch.sigmoid(z_rgate).view(B, N, N, hidden_dim)

    out_mat = torch.bmm(left_mat, right_mat)                     # (B·H, N, N)

    # reshape back to [B, N, N, hidden]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )

    # --------------------------------------------------------------
    # 7️⃣ second LayerNorm, output gating, final projection
    # --------------------------------------------------------------

        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)  # (B, N, N, dim), FP16

    # --------------------------------------------------------------
    # 8️⃣ cast back to FP32 (the caller expects float32)
    # --------------------------------------------------------------
    return out.to(torch.float32)
