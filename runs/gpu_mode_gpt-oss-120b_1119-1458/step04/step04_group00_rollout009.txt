"""
Highly optimized outgoing TriMul forward implementation.

The algorithm follows the reference PyTorch implementation but moves the
dominant cubic contraction into a batched half‑precision GEMM (leveraging
cuBLAS/Tensor‑Core).  A small Triton kernel fuses the optional mask,
projection and gate tensors into a single element‑wise pass, reducing the
number of memory round‑trips.  All linear layers and LayerNorms run in
FP16; the final result is cast back to float32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel that fuses mask, projection and gating:
@triton.jit
def _mask_gate_fuse_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    M, H,
    stride_proj_m, stride_proj_h,
    stride_gate_m, stride_gate_h,
    stride_out_m, stride_out_h,
    stride_mask,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    # Each program processes a tile of size BLOCK_M × BLOCK_H.
    pid = tl.program_id(0)
    block_start = pid * BLOCK_M
    offs_m = block_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    offs_h = tl.arange(0, BLOCK_H)                         # [BLOCK_H]

    # Load mask and broadcast to hidden dimension.
    mask_val = tl.load(mask_ptr + offs_m * stride_mask,
                       mask=offs_m < M,
                       other=0.0).to(tl.float16)        # [BLOCK_M]
    mask_h = mask_val[:, None]                              # [BLOCK_M, 1]

    # Load projection and gate tiles.
    proj = tl.load(
        proj_ptr + offs_m[:, None] * stride_proj_m + offs_h[None, :] * stride_proj_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    gate = tl.load(
        gate_ptr + offs_m[:, None] * stride_gate_m + offs_h[None, :] * stride_gate_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    # Element‑wise fusion.
    out = proj * gate * mask_h

    # Store result.
    tl.store(
        out_ptr + offs_m[:, None] * stride_out_m + offs_h[None, :] * stride_out_h,
        out,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
    )

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float)
        - weights_dict : mapping from layer name to weight tensors (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         ``nomask`` (bool, default True) indicates whether the mask
                         should be applied.

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32), residing on the same
        device as the input.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last channel (fp16 for speed)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # Cast weights to fp16 once.
    norm_weight_fp16 = norm_weight.to(torch.float16)
    norm_bias_fp16   = norm_bias.to(torch.float16)

    # LayerNorm works on the last dimension (dim)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight_fp16,
        bias=norm_bias_fp16,
    ).to(torch.float16)  # [B, N, N, dim]

    B, N, _, _ = x.shape
    M = B * N * N  # flatten dimension for per‑pair operations

    # ------------------------------------------------------------------
    # 2️⃣ Cast all linear weights to fp16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (hidden, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)     # (dim, hidden)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections and sigmoid gates (all fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(-1, dim)                         # (M, dim)

    # Linear projections – fusing matmul for speed
    left_proj = torch.nn.functional.linear(x_flat, left_proj_w)   # (M, hidden)
    right_proj = torch.nn.functional.linear(x_flat, right_proj_w)

    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, left_gate_w)
    )                                                       # (M, hidden)
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, right_gate_w)
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, out_gate_w)
    )                                                       # (M, hidden)

    # Reshape out_gate early – it is needed after the second layer norm.
    out_gate = out_gate.view(B, N, N, hidden_dim)          # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask‑gate‑projection fusion (via Triton)
    # ------------------------------------------------------------------
    if not nomask:
        # Flatten mask to shape (M,) and cast to fp16.
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)

        # Allocate output buffers that will hold the fused tensors.
        left  = torch.empty_like(left_proj)    # (M, hidden)
        right = torch.empty_like(right_proj)

        # Strides in element counts.
        stride_lp_m, stride_lp_h = left_proj.stride()
        stride_lg_m, stride_lg_h = left_gate.stride()
        stride_rp_m, stride_rp_h = right_proj.stride()
        stride_rg_m, stride_rg_h = right_gate.stride()
        stride_left_m, stride_left_h = left.stride()
        stride_right_m, stride_right_h = right.stride()
        stride_mask = mask_fp16.stride()[0]

        # Tune tile size – we tile the whole hidden dimension (<=384) to
        # keep the kernel simple and avoid inner loops.
        BLOCK_M = 64  # rows per program
        BLOCK_H = hidden_dim  # we tile the full hidden dim

        # Number of programs needed (one per BLOCK_M rows)
        grid = ((M + BLOCK_M - 1) // BLOCK_M,)

        # Fuse left path.
        _mask_gate_fuse_kernel[grid](
            left_proj, left_gate, mask_fp16, left,
            M, hidden_dim,
            stride_lp_m, stride_lp_h,
            stride_lg_m, stride_lg_h,
            stride_left_m, stride_left_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
        # Fuse right path.
        _mask_gate_fuse_kernel[grid](
            right_proj, right_gate, mask_fp16, right,
            M, hidden_dim,
            stride_rp_m, stride_rp_h,
            stride_rg_m, stride_rg_h,
            stride_right_m, stride_right_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
        # Reshape back to (B, N, N, hidden_dim)
        left  = left.view(B, N, N, hidden_dim)
        right = right.view(B, N, N, hidden_dim)
    else:
        # No mask – element‑wise fusion in PyTorch (fast)
        left  = (left_proj * left_gate).view(B, N, N, hidden_dim)
        right = (right_proj * right_gate).view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Cubic contraction via batched GEMM (Tensor‑Core)
    # ------------------------------------------------------------------
    # Rearrange tensors to (B, hidden, N, N) and collapse batch×hidden.
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiply: out = left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # Restore original layout: (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (hidden dim) – still in fp16
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Output gating and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, dim) fp16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to float32 for the caller
    # ------------------------------------------------------------------
    return out.to(torch.float32)

"""
Outgoing TriMul forward pass – highly‑optimized Triton+PyTorch implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input (float16 for speed).
2. Compute the four linear projections (left, right) and three sigmoid gates
   (left, right, out) in half‑precision.
3. If a mask is required, a tiny Triton kernel fuses
   `proj * gate * mask` for both left and right tensors, removing two
   extra PyTorch element‑wise passes.
4. The dominant O(N³) contraction is performed as a batched GEMM:
   for every hidden channel we compute `left @ rightᵀ` using `torch.bmm`
   (cuBLAS/Tensor‑Core).  This reduces the cubic loop to an extremely
   fast matrix multiplication.
5. Apply a second LayerNorm on the hidden dimension, multiply by the
   output gate and finish with the output linear projection.
6. Cast the result back to float32 (the expected dtype for the caller).

All linear layers and LayerNorms run in FP16; only the final result is
returned in FP32.  The Triton kernel is mandatory (mask‑gate‑projection
fusion) as required by the problem statement.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _mask_gate_fuse_kernel(
    proj_ptr,          # *const half*   (M, H)
    gate_ptr,          # *const half*   (M, H)
    mask_ptr,          # *const half*   (M,)
    out_ptr,           # *half*         (M, H)
    M, H,              # problem size
    stride_proj_m, stride_proj_h,   # strides of proj (in elements)
    stride_gate_m, stride_gate_h,   # strides of gate
    stride_out_m,  stride_out_h,    # strides of out
    stride_mask,                     # stride of mask (1‑D)
    BLOCK_M: tl.constexpr,           # rows per program
    BLOCK_H: tl.constexpr,           # hidden dim per tile (≤384)
):
    """Fuse `proj * gate * mask` for a tile of size BLOCK_M×BLOCK_H."""
    pid = tl.program_id(0)                             # program ID = tile index
    block_start = pid * BLOCK_M
    offs_m = block_start + tl.arange(0, BLOCK_M)       # [BLOCK_M]
    offs_h = tl.arange(0, BLOCK_H)                     # [BLOCK_H]

    # ----- mask (scalar per row) -----
    mask_val = tl.load(
        mask_ptr + offs_m * stride_mask,
        mask=offs_m < M,
        other=0.0,
    ).to(tl.float16)                                   # [BLOCK_M]
    mask_h = mask_val[:, None]                         # [BLOCK_M,1] for broadcasting

    # ----- projection and gate -----
    proj = tl.load(
        proj_ptr + offs_m[:, None] * stride_proj_m
                     + offs_h[None, :] * stride_proj_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    gate = tl.load(
        gate_ptr + offs_m[:, None] * stride_gate_m
                     + offs_h[None, :] * stride_gate_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    # ----- element‑wise fusion -----
    out = proj * gate * mask_h

    # ----- write back -----
    tl.store(
        out_ptr + offs_m[:, None] * stride_out_m
                 + offs_h[None, :] * stride_out_h,
        out,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
    )


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be ignored.
        - weights_dict : dict of weight tensors (float32) as described in the
                         reference PyTorch module.
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device as the input.
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ first LayerNorm (FP16)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)

    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    ).to(torch.float16)               # [B, N, N, dim] in half

    B, N, _, _ = x.shape
    M = B * N * N                      # flattened pair count

    # --------------------------------------------------------------
    # 2️⃣ cast all linear weights to half once
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)    # (dim, hidden)

    # --------------------------------------------------------------
    # 3️⃣ linear projections + sigmoid gates (FP16)
    # --------------------------------------------------------------
    x_flat = x.view(-1, dim)                    # (M, dim)

    left_proj  = torch.nn.functional.linear(x_flat, left_proj_w)   # (M, hidden)
    right_proj = torch.nn.functional.linear(x_flat, right_proj_w)

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_flat, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_flat, out_gate_w))

    out_gate = out_gate.view(B, N, N, hidden_dim)    # (B, N, N, hidden)

    # --------------------------------------------------------------
    # 4️⃣ optional mask‑gate‑projection fusion (Triton)
    # --------------------------------------------------------------
    if not nomask:
        # Flatten mask to (M,) and cast to half
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)

        # Allocate fused output buffers
        left  = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # Strides (in element counts)
        stride_lp_m, stride_lp_h = left_proj.stride()
        stride_lg_m, stride_lg_h = left_gate.stride()
        stride_rp_m, stride_rp_h = right_proj.stride()
        stride_rg_m, stride_rg_h = right_gate.stride()
        stride_l_out_m, stride_l_out_h = left.stride()
        stride_r_out_m, stride_r_out_h = right.stride()
        stride_mask = mask_fp16.stride()[0]

        # Tile size – full hidden dimension (≤384) per tile
        BLOCK_M = 64                     # rows per launch block
        BLOCK_H = hidden_dim             # tile the entire hidden dim

        grid = ((M + BLOCK_M - 1) // BLOCK_M,)

        # left = left_proj * left_gate * mask
        _mask_gate_fuse_kernel[grid](
            left_proj, left_gate, mask_fp16, left,
            M, hidden_dim,
            stride_lp_m, stride_lp_h,
            stride_lg_m, stride_lg_h,
            stride_l_out_m, stride_l_out_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
        # right = right_proj * right_gate * mask
        _mask_gate_fuse_kernel[grid](
            right_proj, right_gate, mask_fp16, right,
            M, hidden_dim,
            stride_rp_m, stride_rp_h,
            stride_rg_m, stride_rg_h,
            stride_r_out_m, stride_r_out_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )

        # Reshape back to (B, N, N, hidden)
        left  = left.view(B, N, N, hidden_dim)
        right = right.view(B, N, N, hidden_dim)
    else:
        # No mask – simple element‑wise fuse in PyTorch
        left  = (left_proj * left_gate).view(B, N, N, hidden_dim)
        right = (right_proj * right_gate).view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5️⃣ O(N³) contraction via batched GEMM (Tensor‑Core)
    # --------------------------------------------------------------
    # Rearrange to (B, hidden, N, N) → (B*hidden, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiply C = A @ Bᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore (B, N, N, hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ second LayerNorm on hidden dimension (FP16)
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 7️⃣ output gate + final linear projection
    # --------------------------------------------------------------
    out = out * out_gate                                     # gating
    out = torch.nn.functional.linear(out, to_out_w)          # (B, N, N, dim) fp16

    # --------------------------------------------------------------
    # 8️⃣ cast back to FP32 (the caller expects float32)
    # --------------------------------------------------------------
    return out.to(torch.float32)
