"""
TriMul (outgoing) – fast forward pass.

The heavy O(N³) part of the layer is the batched matrix multiplication

    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]  ( = left @ rightᵀ per head)

which we implement with a custom Triton GEMM kernel.  All surrounding
operations (LayerNorm, Linear projections and gating) are performed with
high‑performance torch primitives.  The two projection+gate pairs are
fused into a single matmul with doubled output channels, reducing the
number of GEMM calls from six to four.  The kernel works on fp16 values,
accumulates in fp32 and returns fp16‑typed results that are finally cast
to float32 for the harness.

The implementation handles optional masks (nomask flag) and works for
all test configurations listed in the problem statement.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : [M, N, K]   (M = batch * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
#   All tensors are fp16, accumulator is fp32.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K] (the second operand, will be accessed as Bᵀ)
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # scalar sizes
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bk, stride_bj,   # B strides (M, k, j)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)   # tile along first N dimension (i)
    pid_j = tl.program_id(1)   # tile along second N dimension (j)
    pid_z = tl.program_id(2)   # combined batch*head index (M)

    # Offsets for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k_start in range(0, K, BLOCK_K):
        cur_k = k_start
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # Load B tile transposed (Bᵀ): shape (BLOCK_K, BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # Mixed‑precision dot product (fp16 × fp16 → fp32)
        acc += tl.dot(a, b)

    # Store the result tile (convert back to fp16)
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute ``out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`` using the
    Triton kernel above.  Tensors must be fp16 and contiguous.

    Parameters
    ----------
    left, right : torch.Tensor
        Shape ``[B, N, N, H]`` (fp16).

    Returns
    -------
    out : torch.Tensor
        Shape ``[B, N, N, H]`` (fp16).
    """
    B, N, _, H = left.shape

    # Rearrange to [B*H, N, N] – each head gets its own GEMM batch.
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in element counts (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bk, stride_bj = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tile size selection
    if N >= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8
    else:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch × head
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bk, stride_bj,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=3,
    )

    # Reshape back to the original 4‑D layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operator (no gradient).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32/bool) – may be a dummy tensor
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default ``True``).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------
    # 2️⃣ Cast to fp16 for the rest of the pipeline
    # -------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Fuse projection + gate linear layers
    # -------------------------------------------------
    # Left side (proj + gate)
    left_proj_w = weights["left_proj.weight"]          # (hidden, dim)
    left_gate_w = weights["left_gate.weight"]          # (hidden, dim)
    left_w_fused = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)
    left_comb = torch.nn.functional.linear(x_fp16, left_w_fused)   # [B,N,N,2*hidden]
    left_proj = left_comb[..., :hidden_dim]          # [B,N,N,hidden]
    left_gate = torch.sigmoid(left_comb[..., hidden_dim:])  # [B,N,N,hidden]

    # Right side (proj + gate)
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_w_fused = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)
    right_comb = torch.nn.functional.linear(x_fp16, right_w_fused)
    right_proj = right_comb[..., :hidden_dim]
    right_gate = torch.sigmoid(right_comb[..., hidden_dim:])

    # -------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dim)
    # -------------------------------------------------
    if not nomask:
        # mask may be bool or float; ensure fp16 and raise dim for broadcasting
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # -------------------------------------------------
    # 5️⃣ Apply gates (element‑wise)
    # -------------------------------------------------
    left = left_proj * left_gate   # [B,N,N,hidden]
    right = right_proj * right_gate

    # -------------------------------------------------
    # 6️⃣ Out‑gate (computed from x before the heavy contraction)
    # -------------------------------------------------
    out_gate_w = weights["out_gate.weight"]
    out_gate_raw = torch.nn.functional.linear(x_fp16, out_gate_w.to(torch.float16))
    out_gate = torch.sigmoid(out_gate_raw)   # [B,N,N,hidden]

    # -------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via Triton (fp16 in/out)
    # -------------------------------------------------
    # Ensure contiguous layout before permuting for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hidden] fp16

    # -------------------------------------------------
    # 8️⃣ Second LayerNorm on the hidden dimension
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # fp16

    # -------------------------------------------------
    # 9️⃣ Apply out‑gate and final projection
    # -------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden)
    out = torch.nn.functional.linear(out, to_out_w)        # [B,N,N,dim] fp16

    # -------------------------------------------------
    # 🔟 Cast back to float32 for the external interface
    # -------------------------------------------------
    return out.to(torch.float32)
