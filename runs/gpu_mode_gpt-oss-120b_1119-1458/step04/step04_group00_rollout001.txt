"""
TriMul (outgoing) – fast forward Triton‑+‑PyTorch implementation
================================================================

The forward pass consists of four logical stages:

1️⃣ **LayerNorm & flatten** – the 4‑D input `[B,N,N,C]` is normalized
   (FP16) and reshaped to `(M, C)` where `M = B·N·N`.

2️⃣ **Fused projection + gating** – the two linear layers
   (`*_proj` and `*_gate`) are concatenated into a single GEMM
   per side, halving the number of matrix multiplies.
   For a masked run the element‑wise mask, projection and gate
   are fused in a tiny Triton kernel (`_mask_gate_fuse_kernel`);
   otherwise a simple element‑wise product is used.

3️⃣ **Batched tensor‑core GEMM** – after gating the tensors have
   shape `[B,N,N,hidden]`.  By permuting to `(B, hidden, N, N)` we can
   call `torch.matmul` which internally launches a single cuBLAS
   batched GEMM (Tensor‑Core Friendly) for all `B·hidden` pairs.

4️⃣ **Post‑processing** – a second LayerNorm, an output gate
   (sigmoid), and a final linear projection back to `C`.
   All heavy arithmetic stays in FP16, the result is cast to FP32.

Only the mask‑gate fusion is Triton‑based (as required), the dominant
`O(N³·hidden)` contraction uses the highly‑optimised cuBLAS kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse 2‑D projection, gate and 1‑D mask.
# out = proj * gate * mask[..., None]
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,       # tensors
    M, H,                                        # problem size
    stride_proj_m, stride_proj_h,                # strides of proj
    stride_gate_m, stride_gate_h,                # strides of gate
    stride_out_m, stride_out_h,                  # strides of out
    stride_mask,                                 # stride of mask (1‑D)
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,   # tile sizes
):
    pid = tl.program_id(0)                     # block index over rows
    block_start = pid * BLOCK_M
    offs_m = block_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_h = tl.arange(0, BLOCK_H)                # [BLOCK_H]

    # Load mask (float16) and broadcast across H
    mask = tl.load(mask_ptr + offs_m * stride_mask,
                   mask=offs_m < M,
                   other=0.0).to(tl.float16)    # [BLOCK_M]
    mask_h = mask[:, None]                         # [BLOCK_M,1]

    # Load projection tile
    proj = tl.load(
        proj_ptr + offs_m[:, None] * stride_proj_m
                  + offs_h[None, :] * stride_proj_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    # Load gate tile
    gate = tl.load(
        gate_ptr + offs_m[:, None] * stride_gate_m
                  + offs_h[None, :] * stride_gate_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    # Fuse
    out = proj * gate * mask_h

    # Store
    tl.store(
        out_ptr + offs_m[:, None] * stride_out_m
                 + offs_h[None, :] * stride_out_h,
        out,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
    )


# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32/bool) or None
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32) on the same device as *input_tensor*.
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack & config
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 input → fp16 after LN)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    ).to(dtype_fp16)                                   # [B, N, N, dim]  fp16

    B, N, _, _ = x.shape
    M = B * N * N                                       # flattened pair count

    # --------------------------------------------------------------
    # 2️⃣ Cast all linear weights once to fp16 (avoid repeated casts)
    # --------------------------------------------------------------
    left_proj_w    = weights["left_proj.weight"].to(dtype_fp16)
    left_gate_w    = weights["left_gate.weight"].to(dtype_fp16)
    right_proj_w   = weights["right_proj.weight"].to(dtype_fp16)
    right_gate_w   = weights["right_gate.weight"].to(dtype_fp16)
    out_gate_w     = weights["out_gate.weight"].to(dtype_fp16)

    to_out_norm_w  = weights["to_out_norm.weight"].to(dtype_fp16)
    to_out_norm_b  = weights["to_out_norm.bias"].to(dtype_fp16)
    to_out_w       = weights["to_out.weight"].to(dtype_fp16)

    # --------------------------------------------------------------
    # 3️⃣ Flatten and perform *fused* projection+gate GEMMs
    # --------------------------------------------------------------
    x_flat = x.reshape(-1, dim)                         # (M, dim)

    # Concatenate projection and gate weights (out_features stacked)
    left_comb_w  = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*hidden, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0) # (2*hidden, dim)

    # One GEMM per side
    left_comb  = torch.nn.functional.linear(x_flat, left_comb_w)   # (M, 2*hidden)
    right_comb = torch.nn.functional.linear(x_flat, right_comb_w)  # (M, 2*hidden)

    # Split back
    left_proj = left_comb[:, :hidden_dim]               # (M, hidden)
    left_gate = torch.sigmoid(left_comb[:, hidden_dim:])# (M, hidden)

    right_proj = right_comb[:, :hidden_dim]             # (M, hidden)
    right_gate = torch.sigmoid(right_comb[:, hidden_dim:])# (M, hidden)

    # --------------------------------------------------------------
    # 4️⃣ Output gate (does NOT depend on mask)
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, out_gate_w))
    out_gate = out_gate.view(B, N, N, hidden_dim)       # (B, N, N, hidden)

    # --------------------------------------------------------------
    # 5️⃣ Optional mask + gate fusion (Triton) or plain element‑wise
    # --------------------------------------------------------------
    if not nomask:
        # flatten mask to (M,) and cast to fp16
        mask_fp16 = mask_tensor.to(dtype_fp16).reshape(-1)

        # Allocate fused tensors
        left  = torch.empty_like(left_proj)   # (M, hidden)
        right = torch.empty_like(right_proj)

        # Strides in *elements* (not bytes)
        stride_lp_m, stride_lp_h = left_proj.stride()
        stride_lg_m, stride_lg_h = left_gate.stride()
        stride_rp_m, stride_rp_h = right_proj.stride()
        stride_rg_m, stride_rg_h = right_gate.stride()
        stride_out_m, stride_out_h = left.stride()
        stride_mask = mask_fp16.stride()[0]

        # Tile size – hidden_dim ≤ 128 → we can load whole hidden dim per tile
        BLOCK_M = 64                     # rows per kernel launch
        BLOCK_H = hidden_dim             # tile whole hidden dimension

        grid = ((M + BLOCK_M - 1) // BLOCK_M,)

        # left = left_proj * left_gate * mask
        _mask_gate_fuse_kernel[grid](
            left_proj, left_gate, mask_fp16, left,
            M, hidden_dim,
            stride_lp_m, stride_lp_h,
            stride_lg_m, stride_lg_h,
            stride_out_m, stride_out_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
        # right = right_proj * right_gate * mask
        _mask_gate_fuse_kernel[grid](
            right_proj, right_gate, mask_fp16, right,
            M, hidden_dim,
            stride_rp_m, stride_rp_h,
            stride_rg_m, stride_rg_h,
            stride_out_m, stride_out_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
        # Reshape to 4‑D tensors
        left  = left.view(B, N, N, hidden_dim)
        right = right.view(B, N, N, hidden_dim)
    else:
        # No mask → simple elementwise fusion
        left  = (left_proj * left_gate).view(B, N, N, hidden_dim)
        right = (right_proj * right_gate).view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 6️⃣ Batched Tensor‑Core GEMM (left @ rightᵀ) via torch.matmul
    # --------------------------------------------------------------
    # Bring `hidden_dim` to the second axis: (B, hidden, N, N)
    left_t  = left.permute(0, 3, 1, 2)    # (B, hidden, N, N)
    right_t = right.permute(0, 3, 1, 2)   # (B, hidden, N, N)

    # matmul on the last two dims, batch = B*hidden
    out_t = torch.matmul(left_t, right_t.transpose(-1, -2))  # (B, hidden, N, N)

    # Return to original layout
    out = out_t.permute(0, 2, 3, 1).contiguous()  # (B, N, N, hidden)

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 8️⃣ Output gate + final linear projection (FP16)
    # --------------------------------------------------------------
    out = out * out_gate                                 # element‑wise gating
    out = torch.nn.functional.linear(out, to_out_w)      # (B,N,N,dim) fp16

    # --------------------------------------------------------------
    # 9️⃣ Cast back to FP32 (as expected by downstream code)
    # --------------------------------------------------------------
    return out.to(torch.float32)
