"""
Highly‑optimized forward pass of the “outgoing” TriMul module (AlphaFold‑3).

Algorithm
---------
1️⃣  Layer‑Norm (float32) of the input tensor.
2️⃣  Cast to fp16 for the rest of the pipeline.
3️⃣  Fuse the 5 linear projections (left, right, left‑gate, right‑gate,
    out‑gate) into a *single* GEMM by stacking their weight matrices.
    This reduces the number of full‑matrix‑multiply passes over the huge
    [B, N, N, dim] tensor from 5→1.
4️⃣  Apply the optional mask (broadcast on the hidden dimension) and the
    two “value” gates (sigmoid left‑gate/right‑gate) in‑place.
5️⃣  Heavy pair‑wise contraction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    i.e. a batched matrix‑multiply C = A @ Bᵀ with A/B shape
    (B·hidden, N, N).  For N ≤ 256 a handcrafted Triton kernel
    (64×64×64 tiles, fp16 multiplication, fp32 accumulation) is used.
    Larger N fall back to cuBLAS via ``torch.bmm`` which is
    heavily tuned on H100.
6️⃣  Second Layer‑Norm (fp16) on the contracted tensor, element‑wise
    multiplication with the out‑gate, and a final linear projection back
    to ``dim``.
7️⃣  Cast the result back to float32 (required by the harness).

All operations are pure forward‑only (no gradients) and are performed on
the GPU device of the input tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (fp16 → fp16, accumulator fp32)
#   A, B : [M, N, K]   (M = B * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # strides of A (M,i,k)
    stride_bm, stride_bj, stride_bk,   # strides of B (M,j,k)
    stride_cm, stride_ci, stride_cj,   # strides of C (M,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel for the TriMul contraction."""
    pid_m = tl.program_id(0)   # tile along i
    pid_n = tl.program_id(1)   # tile along j
    pid_z = tl.program_id(2)   # index over M (= batch * hidden)

    # Offsets for the output tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile (i, k)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # Load B tile (j, k) – note transposed view
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk   # k as first dim
            + offs_j[None, :] * stride_bj   # j as second dim
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Mixed‑precision dot: fp16 × fp16 → fp32
        acc += tl.dot(a, b)

    # Store the result
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract ``left`` and ``right`` (both [B, N, N, H]) using the Triton kernel.
    Returns a tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    # reshape to [M, N, N] where M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)          # fp16 output

    # strides (in elements, not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tile size – 64×64×64 works well on H100 for fp16 MMA
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract ``left`` and ``right`` using torch.bmm (cuBLAS).
    Input tensors are fp16; accumulation is fp32 automatically.
    """
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # [M, N, K]
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # [M, N, K]
    C = torch.bmm(A, B_.transpose(1, 2))                         # [M, N, N]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()     # [B, N, N, H]
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 / bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # 1️⃣ Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 2️⃣ First LayerNorm (float32) – stability
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 3️⃣ Cast to fp16 for the heavy part of the pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Stack the five linear weight matrices (left, right, gate_*),
    #    cast them once to fp16.
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # (5 * hidden_dim, dim)
    stacked_proj_w = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(torch.float16, non_blocking=True)

    # ------------------------------------------------------------------
    # 5️⃣ Fused linear projection -> shape [B, N, N, 5*hidden_dim]
    # ------------------------------------------------------------------
    combined = F.linear(x, stacked_proj_w)  # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Slice the combined tensor
    # ------------------------------------------------------------------
    H = hidden_dim
    left      = combined[..., 0:H]
    right     = combined[..., H:2 * H]
    left_gate = torch.sigmoid(combined[..., 2 * H:3 * H])
    right_gate = torch.sigmoid(combined[..., 3 * H:4 * H])
    out_gate   = torch.sigmoid(combined[..., 4 * H:5 * H])

    # ------------------------------------------------------------------
    # 7️⃣ (Optional) mask the value tensors
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 8️⃣ Apply value gates (in‑place to save memory)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 9️⃣ Heavy O(N³) contraction (batched matmul)
    # ------------------------------------------------------------------
    if N <= 256:
        # Small sequences – use the handcrafted Triton kernel.
        out = _batched_matmul_triton(left, right)   # [B, N, N, hidden] (fp16)
    else:
        # Larger sequences – fall back to cuBLAS via torch.bmm.
        out = _batched_matmul_torch(left, right)    # [B, N, N, hidden] (fp16)

    # ------------------------------------------------------------------
    # 🔟 Second LayerNorm (fp16) → out‑gate → final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # → [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Cast back to float32 (the harness expects float32)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
