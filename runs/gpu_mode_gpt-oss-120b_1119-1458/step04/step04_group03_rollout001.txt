"""
Outgoing TriMul (AlphaFold‑3) – optimized forward pass.

Algorithm (no gradients)
------------------------
1. Layer‑norm the input (float32) → fp16.
2. Four linear projections (left/right) and three gating linear layers
   (left_gate/right_gate/out_gate) in fp16.
3. Optional mask (broadcast on hidden_dim) is fused into left/right.
4. Apply sigmoid gates to left/right, then contract:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply of shape
        [B·H, N, N] @ [B·H, N, N]ᵀ
   The heavy O(N³) step is performed by a custom Triton kernel for
   small N (≤256) and by torch.bmm (cuBLAS) for larger N.
5. Second LayerNorm (fp16) → element‑wise out‑gate → final linear
   projection back to `dim`.
6. Cast the result to float32 (required by the harness).

The Triton kernel is a straightforward blocked GEMM (64×64×64 tiles,
8 warps, 2 stages) that accumulates in fp32 and stores fp16.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : [M, N, K]   (M = B*hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
#   All tensors are fp16; accumulator is fp32.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 scalar dimensions
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel for the TriMul contraction.

    Computes, for each batch‑hidden index z:
        C[z,i,j] = Σₖ A[z,i,k] * B[z,j,k]
    """
    pid_z = tl.program_id(2)   # batch*hidden index
    pid_i = tl.program_id(0)   # tile along i (row) dimension
    pid_j = tl.program_id(1)   # tile along j (col) dimension

    # Offsets within the N dimension
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # Load B tile transposed on‑the‑fly:
        # we need B[z, j, k] but we load as B_T[k, j]
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk   # k dimension first
            + offs_j[None, :] * stride_bj   # j dimension second
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # Mixed‑precision dot product → fp32 accumulation
        acc += tl.dot(a, b)

    # Store the result (cast back to fp16)
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract `left` and `right` (both [B, N, N, H]) using the custom Triton
    kernel. Returns a tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    # Reshape to [M, N, N] with M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)   # fp16 output buffer

    # Strides in element‑counts (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()   # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()  # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()   # (N*N, N, 1)

    # Tile sizes – 64 works well on H100 for fp16 mma.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden dimension
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,       # M (unused inside kernel, kept for API compatibility)
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract using cuBLAS via torch.bmm (fp16 × fp16 → fp32 accumulation).
    Returns [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)          # [M, N, K]
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)        # [M, N, K]
    C = torch.bmm(A, B_.transpose(1, 2))                                 # [M, N, N]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()              # [B, N, N, H]
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).
    The heavy O(N³) contraction is partially implemented in Triton.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # 1️⃣  Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 2️⃣  First LayerNorm (float32 for numerical stability)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim], float32

    # --------------------------------------------------------------
    # 3️⃣  Cast to fp16 for the remaining fast path
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4️⃣  Cast linear weights to fp16 once
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 5️⃣  Linear projections (fp16, bias=False)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_w)    # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)  # same shape

    # --------------------------------------------------------------
    # 6️⃣  Optional mask (broadcast on hidden channel dim)
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] → fp16
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 7️⃣  Gating (sigmoid) – fp16
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # Fuse gates (in‑place multiplication)
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 8️⃣  Heavy O(N³) contraction
    # --------------------------------------------------------------
    seq_len = input_tensor.shape[1]
    if seq_len <= 256:
        # Small N → custom Triton kernel
        out = _batched_matmul_triton(left, right)   # [B, N, N, hidden_dim], fp16
    else:
        # Large N → cuBLAS via torch.bmm (much faster)
        out = _batched_matmul_torch(left, right)    # [B, N, N, hidden_dim], fp16

    # --------------------------------------------------------------
    # 9️⃣  Second LayerNorm, out‑gate, and final linear projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # back to dim, still fp16

    # --------------------------------------------------------------
    # 🔟  Cast back to float32 (required by harness)
    # --------------------------------------------------------------
    return out.to(torch.float32)
