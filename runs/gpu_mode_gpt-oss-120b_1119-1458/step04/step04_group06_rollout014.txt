```
@triton.jit
def _trimul_gemm_kernel(A_ptr, B_ptr, C_ptr,
                        M, N, K,
                        stride_am, stride_ai, stride_ak,
                        stride_bm, stride_bj, stride_bk,
                        stride_cm, stride_ci, stride_cj,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
                        BLOCK_K: tl.constexpr):
    pid_m = tl.program_id(0)  # i tile
    pid_n = tl.program_id(1)  # j tile
    pid_z = tl.program_id(2)  # batch*hidden

    # compute offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        # pointers using offsets
        a_ptr = A_ptr + pid_z * stride_am + offs_i[:, None]*stride_ai + offs_k[None, :] * stride_ak
        b_ptr = B_ptr + pid_z * stride_bm + offs_k[:, None]*stride_bk + offs_j[None, :] * stride_bj
        a = tl.load(a_ptr, mask=(mask_i[:, None] & (offs_k[None,:] < K)), other=0.0)
        b = tl.load(b_ptr, mask=((offs_k[:,None] < K) & mask_j[None,:]), other=0.0)
        acc += tl.dot(a, b)

    # store
    c_ptr = C_ptr + pid_z * stride_cm + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptr, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.
-------------------------------------------------
The operator works on a 4‑D tensor x ∈ ℝ^{B×N×N×C} and consists of:

1️⃣  Layer‑norm over the last channel (float‑32 for accuracy).
2️⃣  Cast to fp16 – the heavy part runs in half‑precision.
3️⃣  Four linear projections (left/right) + three gating projections
    (left_gate, right_gate, out_gate).  All use the same half‑precision
    weights and a sigmoid non‑linearity.
4️⃣  Optional mask (broadcasted on the hidden dimension) applied before
    gating.
5️⃣  Fuse gating into the projections in‑place.
6️⃣  O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    This is a batched GEMM C = A @ Bᵀ where A and B are reshaped to
    [B·H, N, N] (H = hidden_dim).  The contraction is performed by a
    custom Triton kernel using 128×128×64 tiles, 8 warps and 2 pipeline
    stages.  Computation is fp16×fp16 → fp32 (Tensor‑core friendly),
    the result is stored back in fp16.
7️⃣  Second LayerNorm (fp16), element‑wise out‑gate, and a final linear
    projection back to the original channel dimension.
8️⃣  Cast the result back to float32 (the harness expects FP32).

The only operation written in Triton is the O(N³) batched GEMM.
All other steps use highly‑optimised PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A : [M, N, K]   (fp16)
#   B : [M, N, K]   (fp16) – accessed as B[m, j, k] and used as (K, N)
#   C : [M, N, N]   (fp16)
#   M = B * hidden_dim (batch * hidden)
#   N = seq_len
#   K = seq_len (summation dim)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M,
    N,
    K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    #  Program IDs
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)   # tile over i‑dimension
    pid_n = tl.program_id(1)   # tile over j‑dimension
    pid_z = tl.program_id(2)   # index over M = batch*hidden

    # ------------------------------------------------------------------
    #  Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 (mixed‑precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # --------------------------------------------------------------
        # Load A tile (BLOCK_M × BLOCK_K)   → shape (BLOCK_M, BLOCK_K)
        # --------------------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # --------------------------------------------------------------
        # Load B tile (BLOCK_K × BLOCK_N)   → shape (BLOCK_K, BLOCK_N)
        #   We read B as B[m, j, k] and interpret it as (k, j).
        # --------------------------------------------------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # Mixed‑precision dot product (accums in fp32)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store C tile (fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper: batched GEMM (A @ Bᵀ) used by TriMul
# ----------------------------------------------------------------------
def _batched_gemm_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Parameters
    ----------
    left, right : torch.Tensor
        Shape [B, N, N, H]  (fp16, contiguous)

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, H] (fp16, contiguous)
    """
    B, N, _, H = left.shape

    # Merge batch and hidden dims for a standard GEMM layout
    #   A[m,i,k] and B[m,i,k] where m = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in element counts
    stride_am, stride_ai, stride_ak = A.stride()   # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()  # (N*N, N, 1) – note b_j = stride on j‑dim
    stride_cm, stride_ci, stride_cj = C.stride()   # (N*N, N, 1)

    # Choose tile size based on N (larger N → larger tiles)
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # merged batch‑hidden dimension
    )

    _trimul_gemm_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,
    )

    # Restore original layout [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32, bool or None)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # -------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part
    # -------------------------------------------------
    x = x.to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Prepare fp16 weights (single conversion)
    # -------------------------------------------------
    def to_fp16(t):
        return t.to(device=device, dtype=torch.float16)

    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = to_fp16(weights["to_out_norm.bias"])
    to_out_w      = to_fp16(weights["to_out.weight"])

    # -------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # -------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,hidden]

    # -------------------------------------------------
    # 5️⃣ Apply optional mask (broadcast on hidden dim)
    # -------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] (fp16) for broadcasting
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 6️⃣ Gating (sigmoid) – fp16
    # -------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Fuse gates into the projections (in‑place)
    left.mul_(left_gate)
    right.mul_(right_gate)

    # -------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via custom Triton kernel
    # -------------------------------------------------
    B_sz, N, _, H = left.shape  # B_sz = batch size, H = hidden_dim
    if N <= 128:
        # For very small N the kernel launch overhead dominates.
        out = torch.einsum('b i k h, b j k h -> b i j h', left, right)
    else:
        out = _batched_gemm_transpose_fp16(left, right)   # [B,N,N,hidden] fp16

    # -------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 9️⃣ Gating + final linear projection back to original dim
    # -------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # -------------------------------------------------
    # 🔟 Cast back to float32 (required by the harness)
    # -------------------------------------------------
    return out.to(torch.float32)
