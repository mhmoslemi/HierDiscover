"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass

Algorithm
---------
1️⃣  Layer‑norm the full 4‑D input in FP32 (numerical stability).
2️⃣  Cast to FP16 – the heavy part works exclusively in FP16 (Tensor‑core friendly).
3️⃣  Apply the four linear projections (left/right) and the three gating projections
    (left_gate, right_gate, out_gate).  All weights are pre‑converted to FP16; the
    operations are performed with `torch.nn.functional.linear` and `torch.sigmoid`.
4️⃣  If a mask is present, broadcast it on the hidden dimension and multiply it
    with the left/right projections *before* gating (exactly as in the reference).
5️⃣  Fuse the gates into the projections in‑place.
6️⃣  Heavy O(N³) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM C = A @ Bᵀ where A and B are reshaped to
   [B·H, N, N] (H = hidden_dim).  The contraction is performed by a custom
   Triton kernel that tiles the output in 128×128 (or 64×64 for tiny N) and
   uses FP16 inputs with FP32 accumulation (mixed‑precision MMA).  For very
   short sequences (N ≤ 256) we fall back to `torch.einsum` to avoid launch
   overhead.
7️⃣  Second LayerNorm (FP16) → out‑gate (sigmoid) → final linear projection
   back to the original `dim`.
8️⃣  Cast the result back to FP32 (the harness expects FP32).

Only the O(N³) contraction is written in Triton; the remaining operations use
highly‑optimized PyTorch primitives.  The kernel works for the full range of
shapes used in the test matrix (B ∈ {1,2}, N ∈ {128,…,1024}, dim ∈ {128,384},
hidden_dim = 128) and respects optional masking.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : [M, N, K]   (M = B * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
#   All tensors are FP16, accumulator is FP32.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # program‑size constants
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ with A,B [M,N,K] and C [M,N,N].
    The kernel processes one block of the (i,j) output plane per program
    instance while iterating over K in tiles.
    """
    pid_i = tl.program_id(0)   # i‑tile index
    pid_j = tl.program_id(1)   # j‑tile index
    pid_m = tl.program_id(2)   # combined batch‑hidden index

    # ------------------------------------------------------------------
    # Tile offsets for the i and j dimensions
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Accumulator in FP32 (mixed‑precision)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # --------------------------------------------------------------
        # Load A tile : shape (BLOCK_M, BLOCK_K)
        # --------------------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_m * stride_am                # offset for the M dimension
            + offs_i[:, None] * stride_ai      # offset for i within the tile
            + offs_k[None, :] * stride_ak      # offset for k within the tile
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # --------------------------------------------------------------
        # Load B tile (transposed) : shape (BLOCK_K, BLOCK_N)
        # --------------------------------------------------------------
        b_ptrs = (
            B_ptr
            + pid_m * stride_bm
            + offs_k[:, None] * stride_bk      # k dimension (will be summed)
            + offs_j[None, :] * stride_bj      # j dimension (output column)
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # Mixed‑precision dot product (fp16×fp16 → fp32)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result tile (back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_m * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper: batched GEMM with transposition (A @ Bᵀ) for the TriMul contraction.
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using the custom Triton kernel above.
    Arguments
    ---------
    left, right : torch.Tensor [B, N, N, H]   (fp16, contiguous)
    Returns
    -------
    out : torch.Tensor [B, N, N, H] (fp16, contiguous)
    """
    B, N, _, H = left.shape

    # Merge batch and hidden dimensions → [B·H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (fp16)
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in element counts (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()   # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()  # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()   # (N*N, N, 1)

    # Heuristic tile size selection – larger tiles for larger N
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch × hidden
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,   # double‑buffered pipeline
    )

    # Restore original layout [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------
    # 2️⃣ Cast to fp16 for the rest of the pipeline
    # -------------------------------------------------
    x = x.to(dtype=torch.float16)

    # -------------------------------------------------
    # 3️⃣ Cast model weights to fp16 on the same device
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # -------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # -------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)    # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)   # [B, N, N, hidden_dim]

    # -------------------------------------------------
    # 5️⃣ Optional mask (broadcast on hidden dimension)
    # -------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → broadcast to hidden dim
        mask_fp16 = mask_tensor.to(dtype=torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 6️⃣ Gating (sigmoid) – fp16
    # -------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Fuse gate into projection in‑place (reduces memory traffic)
    left.mul_(left_gate)
    right.mul_(right_gate)

    # -------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via Triton (fp16 in, fp16 out)
    # -------------------------------------------------
    B, N, _, H = left.shape
    # For very short sequences the Triton launch overhead dominates.
    if N <= 256:
        # torch.einsum leverages cuBLAS‑based batched GEMM internally.
        out = torch.einsum('b i k h, b j k h -> b i j h', left, right)
    else:
        # Reshape & launch the custom batched GEMM kernel.
        out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, H] fp16

    # -------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 9️⃣ Final gating + projection back to dim (fp16)
    # -------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] fp16

    # -------------------------------------------------
    # 🔟 Cast back to float32 (required by the harness)
    # -------------------------------------------------
    return out.to(dtype=torch.float32)
