"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

Key ideas
---------
* The heavy N³ contraction is performed by a custom Triton GEMM kernel that
  computes, for each hidden channel `h`,
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
  (a batched GEMM `A @ Bᵀ`).  The kernel works on fp16 inputs and
  accumulates in fp32 for accuracy, using 128×128×64 tiles (8 warps,
  3 pipeline stages) for large sequence lengths.
* The four linear projections (`left_proj`, `right_proj`,
  `left_gate`, `right_gate`) and the `out_gate` projection are fused
  into **one** large linear layer (`5 * hidden_dim` outputs).  This cuts
  the number of cuBLAS kernel launches dramatically and improves memory
  locality.
* Mask handling is done once after the fused projection/gating step.
* The remaining operations (first/second LayerNorm, final gating and
  output projection) stay in PyTorch – they are comparatively cheap
  (O(N²·hidden_dim·dim)) and do not dominate runtime.

The function `custom_kernel` follows the exact API required by the
challenge and returns a float‑32 tensor of shape `[B, N, N, dim]`.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B,C are fp16, acc in fp32)
#   A : [M, N, K]   (M = batch*hidden, N = seq_len, K = seq_len)
#   B : [M, N, K]   (same layout as A)
#   C : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # scalar sizes
    stride_am, stride_ai, stride_ak,   # A strides for (M,i,k)
    stride_bm, stride_bj, stride_bk,   # B strides for (M,j,k)
    stride_cm, stride_ci, stride_cj,   # C strides for (M,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ----------------------------------------------------------
    # Program IDs (i‑tile, j‑tile, batch‑hidden index)
    # ----------------------------------------------------------
    pid_i = tl.program_id(0)   # tile along first N dimension (i)
    pid_j = tl.program_id(1)   # tile along second N dimension (j)
    pid_z = tl.program_id(2)   # batch*hidden (M) index

    # ----------------------------------------------------------
    # Offsets for this tile
    # ----------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ----------------------------------------------------------
    # Accumulator (fp32) – start from zero
    # ----------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------------------------------------------
    # Loop over K dimension in tiles
    # ----------------------------------------------------------
    for k_start in range(0, K, BLOCK_K):
        cur_k = k_start
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile : (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # Load B tile (transposed) : (BLOCK_K, BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Mixed‑precision dot (fp16×fp16 → fp32)
        acc += tl.dot(a, b)

    # ----------------------------------------------------------
    # Store tile (convert back to fp16)
    # ----------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(
    left: torch.Tensor, right: torch.Tensor
) -> torch.Tensor:
    """
    Compute ``out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`` using the
    Triton kernel above.  Arguments must be fp16 contiguous tensors of shape
    ``[B, N, N, H]``.  Returns an fp16 tensor of the same shape.
    """
    B, N, _, H = left.shape

    # Bring hidden dimension into the first axis for the kernel
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Allocate output
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in elements (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Choose tile sizes (larger tiles for long sequences)
    if N >= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,   # a little deeper pipeline for large K
    )

    # Restore original layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operator (no backward).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (bool/float, may be None)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default ``True``).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32, stable)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape [B, N, N, dim], dtype float32

    # -------------------------------------------------
    # 2️⃣ Cast to fp16 for the rest of the pipeline
    # -------------------------------------------------
    x = x.to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Fuse all linear projections (proj & gate) into one matmul
    # -------------------------------------------------
    #   order: left_proj, right_proj, left_gate, right_gate, out_gate
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # shape [(5*hidden_dim), dim]

    fused_out = torch.nn.functional.linear(x, fused_weight)  # [B,N,N,5*hidden]

    # Split the fused result
    split_tensors = torch.split(fused_out, hidden_dim, dim=-1)
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = split_tensors

    # Apply sigmoid to gate tensors
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)   # shape [B,N,N,hidden]

    # -------------------------------------------------
    # 4️⃣ Apply gating (in‑place) and optional mask
    # -------------------------------------------------
    left = left_raw * left_gate
    right = right_raw * right_gate

    if not nomask and mask_tensor is not None:
        # mask is broadcast over the hidden dimension
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via Triton (fp16 in/out)
    # -------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hidden] fp16

    # -------------------------------------------------
    # 6️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # still fp16

    # -------------------------------------------------
    # 7️⃣ Final gating + linear projection
    # -------------------------------------------------
    out = out * out_gate                     # element‑wise gating (in‑place)
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden)
    out = torch.nn.functional.linear(out, to_out_w)        # [B,N,N,dim] fp16

    # -------------------------------------------------
    # 8️⃣ Cast back to float32 for the harness / downstream code
    # -------------------------------------------------
    return out.to(torch.float32)
