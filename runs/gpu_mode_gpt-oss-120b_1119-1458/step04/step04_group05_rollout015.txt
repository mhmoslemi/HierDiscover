class TriMul(nn.Module):

        super().__init__()

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N, hidden_dim,
    stride_batch,
    stride_i,
    stride_k,
    stride_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # pids
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    # compute batch and head indices
    b = pid_z // hidden_dim
    h = pid_z - b * hidden_dim

    # compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # guard
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointer for this batch+head
    base_left = left_ptr + b * stride_batch + h
    base_right = right_ptr + b * stride_batch + h
    base_out = out_ptr + b * stride_batch + h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):

        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # mask for k dimension
        mask_k = offs_k < N

        # load left block

@triton.jit
def trimul_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,
    N: tl.constexpr, H: tl.constexpr,
    stride_batch, stride_i, stride_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)
    # compute batch and head indices
    b = pid_z // H
    h = pid_z - b * H
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N
    # Base pointer per batch+head
    base_A = A_ptr + b * stride_batch + h
    base_B = B_ptr + b * stride_batch + h
    base_C = C_ptr + b * stride_batch + h
    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    num_k = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N
        # load A[i,k]
        a_ptrs = base_A + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # load B[k,j] (transposed)
        b_ptrs = base_B + offs_k[:, None] * stride_i + offs_j[None,:] * stride_k
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        # dot
        acc += tl.dot(a, b)
    # store
    c_ptrs = base_C + offs_i[:, None] * stride_i + offs_j[None,:] * stride_k
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

TriMul (outgoing) – fast forward implementation with Triton.

The heavy O(N³) operation
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is realized as a batched matrix multiplication per head:
    left  → A (shape B·H, N, N)
    right → Bᵀ (shape B·H, N, N)  (k‑dimension is the first spatial axis)
The kernel works directly on the original layout [B, N, N, H] and therefore

The implementation supports both masked and un‑masked inputs and falls
back to cuBLAS‑based `torch.bmm` for very long sequences (N > 512),
though the Triton kernel is already efficient up to N=1024 on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A : [M, N, K]   (M = B·H, N = seq_len, K = seq_len)
#   B : same layout as A but we read it transposed: Bᵀ[k, j] = B[i, j] with i←k
#   C : [M, N, N]
#   Layout of A/B/C = [batch, i, j, head] (contiguous in head dimension)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    A_ptr,                     # left  tensor  (fp16)
    B_ptr,                     # right tensor  (fp16)
    C_ptr,                     # output tensor (fp16)
    N,                         # sequence length
    H,                         # hidden_dim (number of heads)
    stride_batch,              # stride to jump to next batch element (N*N*H)
    stride_i,                  # stride for the i‑axis       (N*H)
    stride_k,                  # stride for the k‑/j‑axis  (H)
    BLOCK_M: tl.constexpr,     # tile size in the i‑dimension
    BLOCK_N: tl.constexpr,     # tile size in the j‑dimension
    BLOCK_K: tl.constexpr,     # tile size in the reduction (k) dimension
):
    # program IDs
    pid_i = tl.program_id(0)          # tile along i (first spatial axis)
    pid_j = tl.program_id(1)          # tile along j (second spatial axis)
    pid_z = tl.program_id(2)          # combined batch·head index (M)

    # decode batch and head from pid_z
    b = pid_z // H                    # batch index
    h = pid_z - b * H                 # head index (0 ≤ h < H)

    # i / j offsets inside the tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds checks
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointer for this (batch, head) slice.
    # because the head dimension is contiguous (stride = 1) we only need
    # to add `b * stride_batch + h` to reach the element (b, 0, 0, h).
    base_A = A_ptr + b * stride_batch + h
    base_B = B_ptr + b * stride_batch + h
    base_C = C_ptr + b * stride_batch + h

    # accumulator in FP32 (higher precision for the sum over k)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k‑tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----------------------- load A[i,k] -----------------------
        a_ptrs = base_A + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )  # shape (BLOCK_M, BLOCK_K), fp16

        # ----------------------- load Bᵀ[k,j] -----------------------
        # Bᵀ is accessed as B[k, j] (i.e. we swap the original i and k axes)
        b_ptrs = base_B + offs_k[:, None] * stride_i + offs_j[None, :] * stride_k
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )  # shape (BLOCK_K, BLOCK_N), fp16

        # ------------------------ dot product -----------------------
        acc += tl.dot(a, b)   # fp16 * fp16 → fp32 accumulation

    # -------------------------- store --------------------------
    c_ptrs = base_C + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor,
                           right: torch.Tensor,
                           tile: int = 64,
                           k_tile: int = 64) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    Input tensors are assumed to be FP16, contiguous, and have shape
    [B, N, N, H].  Returns C with the same shape and dtype.
    """
    B, N, _, H = left.shape
    # ensure contiguous layout (last dimension is already contiguous)
    left = left.contiguous()
    right = right.contiguous()

    stride_batch = left.stride(0)   # B → N*N*H
    stride_i = left.stride(1)       # i → N*H
    stride_k = left.stride(2)       # k / j → H

    out = torch.empty_like(left)    # allocate FP16 output

    grid = (
        (N + tile - 1) // tile,    # tiles along i
        (N + tile - 1) // tile,    # tiles along j
        B * H,                     # batch * heads
    )
    _trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        N,
        H,
        stride_batch,
        stride_i,
        stride_k,
        BLOCK_M=tile,
        BLOCK_N=tile,
        BLOCK_K=k_tile,
        num_warps=8,
        num_stages=3,
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (gradient‑free).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32, bool or None)
        - weights_dict : dict of model parameters (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # 1️⃣ Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2️⃣ First LayerNorm (float32 → FP16)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias)                 # [B, N, N, dim] float32
    x = x.to(torch.float16)                           # heavy work in FP16

    # --------------------------------------------------------------
    # 3️⃣ Cast model weights to FP16 (except for final norm/linear)
    #    which are cheap enough to keep in FP32.
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (hidden, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"]   # keep FP32 for stability
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"].to(torch.float16)   # (dim, hidden)

    # --------------------------------------------------------------
    # 4️⃣ Linear projections & gating (all FP16)
    # --------------------------------------------------------------
    B, N, _, _ = x.shape
    flat = x.reshape(-1, dim)                         # (B·N·N, dim)

    # Projections
    left  = F.linear(flat, left_proj_w).view(B, N, N, hidden_dim)
    right = F.linear(flat, right_proj_w).view(B, N, N, hidden_dim)

    # Apply optional mask (broadcast over the hidden dimension)
    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # Gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(flat, left_gate_w)).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(flat, right_gate_w)).view(B, N, N, hidden_dim)
    out_gate   = torch.sigmoid(F.linear(flat, out_gate_w)).view(B, N, N, hidden_dim)

    # Fuse gates into the projected tensors (in‑place to save memory)
    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction (batched matmul)
    # --------------------------------------------------------------
    # Use Triton for N ≤ 512, otherwise cuBLAS-backed torch.bmm.
    if N <= 512:
        out = _batched_matmul_triton(left, right, tile=64, k_tile=64)
    else:
        # cuBLAS path: reshape to (B·H, N, N) and perform bmm
        left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
        right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)
        out_mat   = torch.bmm(left_mat, right_mat)               # (B·H, N, N)
        out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm, output gating, final linear projection
    # --------------------------------------------------------------
    # LayerNorm over hidden_dim (still FP16, we pass FP32 weight/bias)
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)                # FP16 → fp16
    out = out * out_gate                                      # gating (fp16)

    # Final linear projection back to dim
    out = F.linear(out, to_out_w)                             # FP16 result

    # --------------------------------------------------------------
    # 7️⃣ Cast back to FP32 for the caller
    # --------------------------------------------------------------
    return out.to(torch.float32)
```
