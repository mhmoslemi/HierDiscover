    left_ptr, right_ptr, out_ptr,
    B, N, hidden,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

):

    pid_m = tl.program_id(0) # row block index
    pid_n = tl.program_id(1) # col block index

    batch = batch_hidden // hidden

    # right base pointer:
    right_base = right_ptr + batch * stride_right_b + h * stride_right_h

"""
TriMul (outgoing) – Triton‑fused implementation
================================================

This implementation follows the reference PyTorch module but moves all
the heavy O(N³) work to a custom Triton kernel.  The steps are:

1️⃣  Layer‑norm on the input (FP16).
2️⃣  Fuse projection + gate for the left and right branches using a
    stacked linear weight (only two matmuls instead of four).
3️⃣  Optional mask‑gate fusion in a small element‑wise Triton kernel.
4️⃣  Batched GEMM (out[i,j] = Σₖ left[i,k]·right[j,k]) computed by a
    Triton kernel that works directly on the original (B,N,N,hidden)
    layout – no costly ``contiguous`` copies are needed.
5️⃣  Second Layer‑norm, output gate and final Linear projection back to
    ``dim`` (all in FP16, accumulator FP32 for the GEMM).
6️⃣  Cast the result back to float‑32.

The kernel for step 4 uses tiling (BLOCK_M×BLOCK_N×BLOCK_K) with
Tensor‑Core friendly FP16 arithmetic and accumulates in FP32 for
numerical stability.  The mask‑gate fusion kernel (step 3) saves two
extra PyTorch broadcasts.  Together this reduces data movement and
runtime compared with a naïve torch.bmm implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Kernel 1: mask * projection * gate  (M = B·N·N, H = hidden_dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,      # tensors
    M, H,                                       # problem size
    stride_proj_m, stride_proj_h,               # strides of proj
    stride_gate_m, stride_gate_h,               # strides of gate
    stride_out_m, stride_out_h,                 # strides of out
    stride_mask,                                # stride of mask (1‑D)
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)                     # block index over rows
    block_start = pid * BLOCK_M
    offs_m = block_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    offs_h = tl.arange(0, BLOCK_H)                       # [BLOCK_H]

    # Load mask (1‑D) and broadcast to H
    mask = tl.load(mask_ptr + offs_m * stride_mask,
                   mask=offs_m < M, other=0.0).to(tl.float16)   # [BLOCK_M]
    mask_h = mask[:, None]                                   # [BLOCK_M,1]

    # Load projection and gate tiles
    proj = tl.load(
        proj_ptr + offs_m[:, None] * stride_proj_m
                + offs_h[None, :] * stride_proj_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    gate = tl.load(
        gate_ptr + offs_m[:, None] * stride_gate_m
                + offs_h[None, :] * stride_gate_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    # Fusion
    out = proj * gate * mask_h

    # Store result
    tl.store(
        out_ptr + offs_m[:, None] * stride_out_m
                 + offs_h[None, :] * stride_out_h,
        out,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
    )


# ----------------------------------------------------------------------
# Kernel 2: batched matrix multiplication for the O(N³) contraction
#           out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile over rows (i)
    pid_n = tl.program_id(1)   # tile over cols (j)
    pid_bh = tl.program_id(2)  # batch * hidden index

    # --------------------------------------------------------------
    # Decode batch and hidden index from pid_bh
    # --------------------------------------------------------------
    h = pid_bh % H
    b = pid_bh // H

    # --------------------------------------------------------------
    # Compute base pointers for this (b,h) pair
    # --------------------------------------------------------------
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # --------------------------------------------------------------
    # Tile offsets in i/j dimensions
    # --------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # --------------------------------------------------------------
    # Accumulator (FP32 for Tensor‑Core accumulation)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tk in range(num_k_tiles):
        k_start = tk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # left[i,k] tile
        left = tl.load(
            left_base
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k,
            mask=(offs_i[:, None] < N) & (offs_k[None, :] < N),
            other=0.0,
        )

        # right[j,k] tile – note the transpose (k as rows, j as cols)
        right = tl.load(
            right_base
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_i,
            mask=(offs_k[:, None] < N) & (offs_j[None, :] < N),
            other=0.0,
        )
        # Accumulate:  (BLOCK_M,BLOCK_K) • (BLOCK_K,BLOCK_N) → (BLOCK_M,BLOCK_N)
        acc += tl.dot(left, right)

    # --------------------------------------------------------------
    # Store the result, casting back to FP16
    # --------------------------------------------------------------
    out_val = acc.to(tl.float16)
    tl.store(
        out_base
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j,
        out_val,
        mask=(offs_i[:, None] < N) & (offs_j[None, :] < N),
    )


# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Input tuple:
        - input tensor  : [B, N, N, dim]   (float32)
        - mask tensor   : [B, N, N]        (float32 / bool)  (might be unused)
        - weights dict  : model weights
        - config dict   : at least ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns:
        Tensor of shape [B, N, N, dim] (float32) on the same device as the input.
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    ).to(torch.float16)                     # [B, N, N, dim]  FP16

    B, N, _, _ = x.shape
    M = B * N * N                              # flattened pair count

    # ------------------------------------------------------------------
    # 2️⃣ Cast linear weights to FP16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections + sigmoid gates (combined proj+gate per side)
    # ------------------------------------------------------------------
    x_flat = x.view(-1, dim)                  # (M, dim)

    # left side: concat(proj, gate) → (M, 2*hidden)
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)      # (2*hidden, dim)
    left_comb = F.linear(x_flat, left_comb_w)                       # (M, 2*hidden)
    left_proj_flat = left_comb[:, :hidden]                          # (M, hidden)
    left_gate_logits = left_comb[:, hidden:]                        # (M, hidden)
    left_gate = torch.sigmoid(left_gate_logits)                     # (M, hidden)

    # right side: concat(proj, gate) → (M, 2*hidden)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)   # (2*hidden, dim)
    right_comb = F.linear(x_flat, right_comb_w)                     # (M, 2*hidden)
    right_proj_flat = right_comb[:, :hidden]                        # (M, hidden)
    right_gate_logits = right_comb[:, hidden:]                      # (M, hidden)
    right_gate = torch.sigmoid(right_gate_logits)                   # (M, hidden)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask + gate fusion (Triton)
    # ------------------------------------------------------------------
    if not nomask:
        # Convert mask to FP16 and flatten -> (M,)
        mask_fp16 = mask_tensor.to(torch.float16).view(-1)

        # Allocate fused buffers (still (M, hidden))
        left_flat = torch.empty_like(left_proj_flat)
        right_flat = torch.empty_like(right_proj_flat)

        # Strides for the Triton kernel (in element counts)
        stride_lp_m, stride_lp_h = left_proj_flat.stride()
        stride_lg_m, stride_lg_h = left_gate.stride()
        stride_rp_m, stride_rp_h = right_proj_flat.stride()
        stride_rg_m, stride_rg_h = right_gate.stride()
        stride_o_m,  stride_o_h  = left_flat.stride()   # output stride (same as proj)

        stride_mask = mask_fp16.stride()[0]

        # Tile size – we tile the whole hidden dimension (≤128) at once
        BLOCK_M = 64
        BLOCK_H = hidden

        grid = ((M + BLOCK_M - 1) // BLOCK_M,)

        # left = left_proj * left_gate * mask
        _mask_gate_fuse_kernel[grid](
            left_proj_flat, left_gate, mask_fp16, left_flat,
            M, hidden,
            stride_lp_m, stride_lp_h,
            stride_lg_m, stride_lg_h,
            stride_o_m, stride_o_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
        # right = right_proj * right_gate * mask
        _mask_gate_fuse_kernel[grid](
            right_proj_flat, right_gate, mask_fp16, right_flat,
            M, hidden,
            stride_rp_m, stride_rp_h,
            stride_rg_m, stride_rg_h,
            stride_o_m, stride_o_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
    else:
        # No mask – simple element‑wise multiplication in PyTorch
        left_flat  = left_proj_flat * left_gate
        right_flat = right_proj_flat * right_gate

    # Reshape to the 4‑D layout used by the contraction
    left  = left_flat.view(B, N, N, hidden)      # (B, N, N, hidden)
    right = right_flat.view(B, N, N, hidden)     # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 5️⃣ O(N³) contraction via custom Triton GEMM (avoids .contiguous())
    # ------------------------------------------------------------------
    out = torch.empty_like(left)                 # (B, N, N, hidden)

    # Strides in element counts for the three tensors
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_k, stride_o_h = out.stride()

    # GEMM tile sizes – tuned for H100 FP16 Tensor‑Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        ( (N + BLOCK_M - 1) // BLOCK_M ),
        ( (N + BLOCK_N - 1) // BLOCK_N ),
        B * hidden,
    )

    _tri_mul_outgoing_gemm_kernel[grid](
        left, right, out,
        B, N, hidden,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_k, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8, num_stages=2,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm on the hidden dimension (FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Output gate and final projection (FP16)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_w)).view(B, N, N, hidden)
    out = out * out_gate
    out = F.linear(out, to_out_w)                # (B, N, N, dim)  FP16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 (the caller expects float32)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
