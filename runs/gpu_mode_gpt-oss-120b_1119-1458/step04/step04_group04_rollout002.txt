@triton.jit
def mask_gate_fuse_kernel(
    x_ptr, gate_ptr, mask_ptr,
    hidden_dim: tl.constexpr,
    size: tl.int32,
    BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    # compute mask index
    mask_offs = offs // hidden_dim
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def mask_gate_fuse_kernel(x_ptr, gate_ptr, mask_ptr,
                          hidden_dim, size,
                          BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    mask_offs = offs // hidden_dim
    # Load
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.
    1. FP16 LayerNorm on the input.
    2. Linear projections and sigmoid gates (FP16).
    3. Optional element‑wise fusion with mask via a tiny Triton kernel.
    4. Batched GEMM (`torch.bmm`) implements the N³ contraction.
    5. FP16 LayerNorm, out‑gate multiplication and final projection.
    6. Cast back to FP32.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # ----- 1️⃣ FP16 layer‑norm on the input ----------------------------
    # Cast weights/bias to fp16 once
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    # Cast input to fp16
    x = input_tensor.to(torch.float16)
    # LayerNorm over the last dimension (dim)
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # ----- 2️⃣ Linear projections (no bias) and gates -----------------
    # Cast all linear weights to fp16
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # Projections
    # shape: [B, N, N, hidden_dim]
    left  = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,hidden)
    right = torch.nn.functional.linear(x, right_proj_w)  # (B,N,N,hidden)

    # Gates (sigmoid)
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ----- 3️⃣ Optional mask fusion -----------------------------------
    if not nomask:
        # mask shape [B,N,N] → fp16
        mask_fp16 = mask_tensor.to(torch.float16)
        # flatten tensors
        total_elements = left.numel()   # B*N*N*hidden_dim
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)
        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            hidden_dim,
            total_elements,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            hidden_dim,
            total_elements,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask: simple elementwise multiplication
        left = left * left_gate
        right = right * right_gate

    # ----- 4️⃣ Batched GEMM (outgoing contraction) -------------------
    B, N, _, _ = left.shape
    # reshape to [B*hidden_dim, N, N]
    # Permute such that hidden becomes batch dimension
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # GEMM: [B*H, N, N] @ [B*H, N, N].T  -> [B*H, N, N]
    # torch.bmm uses cuBLAS tensor cores for fp16
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # restore shape [B,N,N,hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----- 5️⃣ Second layer norm, out‑gate, final projection ----------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # out_gate was already computed from x (the normalized input)
    # Multiply elementwise
    out = out * out_gate
    # Linear projection back to dim
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim)

    # ----- 6️⃣ Cast back to float32 -----------------------------------
    return out.to(torch.float32)

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1. Cast the input to FP16 and apply the first LayerNorm (FP16 weights/bias).
2. Compute the two linear projections (`left_proj`, `right_proj`) and the three
   sigmoid gates (`left_gate`, `right_gate`, `out_gate`) in FP16.
3. If a mask is supplied, fuse the element‑wise multiplication
   `proj * gate * mask` with a tiny Triton kernel (the mask is broadcast over
   the hidden dimension).  When `nomask=True` a plain PyTorch multiplication
   is used.
4. Perform the O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a batched GEMM using `torch.bmm`, i.e. a single cuBLAS call on the
   reshaped tensors `[B·H, N, N]`.
5. Apply a second LayerNorm over the hidden dimension, multiply by the
   `out_gate` and finally project back to `dim` with a linear layer.
6. Cast the result back to float32 (the harness expects FP32 output).

All heavy element‑wise work that can benefit from Triton (mask × gate × proj)
is off‑loaded to a custom kernel; the expensive contraction leverages the
highly‑optimised cuBLAS Tensor‑Core kernels.  The implementation works for
any batch size ≤ 2, sequence lengths up to 1024 and hidden dimensions used by
AlphaFold‑3 / Chai / Protenix.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (proj * gate * mask) in‑place.
# mask has shape [B, N, N] and is broadcast over the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    proj_ptr,          # *fp16 : projection tensor (flattened)
    gate_ptr,          # *fp16 : gate tensor (same shape as projection)
    mask_ptr,          # *fp16 : mask tensor (flattened, size = B*N*N)
    hidden_dim,        # i32   : size of the hidden dimension
    total_elems,       # i32   : total number of elements in the projection tensor
    BLOCK: tl.constexpr,  # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)        # linear indices for this block
    mask = offs < total_elems                        # out‑of‑range guard

    # broadcasted mask index = element_index // hidden_dim
    mask_offs = offs // hidden_dim

    # load operands
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    msk  = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    # fused multiplication
    out = proj * gate * msk

    # write back (in‑place)
    tl.store(proj_ptr + offs, out, mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = int(config["dim"])
    hidden_dim = int(config["hidden_dim"])
    nomask = bool(config.get("nomask", True))

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ FP16 first LayerNorm (over channel dim)
    # -----------------------------------------------------------------
    # Cast everything to FP16 once
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    x = input_tensor.to(torch.float16)
    # LayerNorm on the last dimension (size = dim)
    x = torch.nn.functional.layer_norm(
        x, (dim,), weight=norm_w, bias=norm_b
    )  # [B, N, N, dim]  fp16

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections and gate computations (FP16)
    # -----------------------------------------------------------------
    # Linear weights (bias=False in the reference implementation)
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    # Projections → [B, N, N, hidden_dim]
    left  = torch.nn.functional.linear(x, left_proj_w)
    right = torch.nn.functional.linear(x, right_proj_w)

    # Gates (sigmoid) → same shape
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask fusion using Triton
    #    (broadcast mask over hidden_dim)
    # -----------------------------------------------------------------
    if not nomask:
        # mask → fp16, shape: [B, N, N]
        mask_fp16 = mask_tensor.to(torch.float16)

        # flatten tensors for the kernel
        total_elements = left.numel()   # B * N * N * hidden_dim
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            hidden_dim,
            total_elements,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            hidden_dim,
            total_elements,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask → simple point‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 4️⃣ Batched GEMM: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as B*hidden_dim independent GEMMs via torch.bmm
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    # Rearrange to [B*hidden_dim, N, N]
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # GEMM with cuBLAS (Tensor‑Core kernels for fp16)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout → [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 5️⃣ Second LayerNorm, out‑gate multiplication and final projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    # Apply the out‑gate (broadcast over the hidden dimension)
    out = out * out_gate
    # Final linear projection back to `dim`
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim]  fp16

    # -----------------------------------------------------------------
    # 6️⃣ Cast result back to FP32 for the harness
    # -----------------------------------------------------------------
    return out.to(torch.float32)
