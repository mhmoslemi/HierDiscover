"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (float32 → fp16 for the heavy part).
2️⃣ Fuse the four linear projections (left/right) and the three
   gating projections (left_gate, right_gate, out_gate) into two
   dense matrix multiplications:
      * [left_proj ; left_gate]        → 2·hidden
      * [right_proj ; right_gate ; out_gate] → 3·hidden
   This reduces the number of kernel launches and the memory traffic
   for the massive tensor `x`.
3️⃣ Apply the mask (if present) after the gating – this is equivalent
   to the reference implementation but avoids an extra memory pass.
4️⃣ Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM   C = A @ Bᵀ   with torch.bmm,
   which on H100 uses Tensor‑cores (fp16×fp16 → fp32 accumulation).
5️⃣ Second LayerNorm → out‑gate (sigmoid) → final linear projection.
6️⃣ Cast the result back to float32 (the harness expects FP32).

Only the O(N³) part is written in Triton (fallback for very short
sequences), the rest is fused with highly‑optimized PyTorch kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Tiny Triton kernel (used only for N ≤ 128 as a sanity fallback)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_fallback_kernel(
    left_ptr,          # *fp16  [B, N, N, H]
    right_ptr,         # *fp16  [B, N, N, H]
    out_ptr,           # *fp16  [B, N, N, H]
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_ri, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_ok, stride_oh,   # out strides
    BLOCK_N: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_n = tl.arange(0, BLOCK_N)
    mask_i = pid_i * BLOCK_N + offs_n < N
    mask_j = pid_j * BLOCK_N + offs_n < N

    out = tl.zeros((BLOCK_N, BLOCK_N, H), dtype=tl.float32)

    # iterate over k dimension
    for k in range(N):
        # load slices
        l_ptrs = left_ptr + pid_b * stride_lb \
                 + (pid_i * BLOCK_N + offs_n)[:, None, None] * stride_li \
                 + (k) * stride_lk \
                 + tl.arange(0, H)[None, None, :] * stride_lh
        r_ptrs = right_ptr + pid_b * stride_rb \
                 + (pid_j * BLOCK_N + offs_n)[:, None, None] * stride_ri \
                 + (k) * stride_rk \
                 + tl.arange(0, H)[None, None, :] * stride_rh
        l = tl.load(l_ptrs, mask=mask_i[:, None, None] & (tl.arange(0, H) < H), other=0.0)  # (i,1,H)
        r = tl.load(r_ptrs, mask=mask_j[:, None, None] & (tl.arange(0, H) < H), other=0.0)  # (j,1,H)

        # broadcast to (i,j,H) and accumulate
        out += tl.dot(l, r)   # uses tensor‑core friendly dot

    # store
    out_ptrs = out_ptr + pid_b * stride_ob \
               + (pid_i * BLOCK_N + offs_n)[:, None, None] * stride_oi \
               + (pid_j * BLOCK_N + offs_n)[None, :, None] * stride_ok \
               + tl.arange(0, H)[None, None, :] * stride_oh
    tl.store(out_ptrs, out.to(tl.float16), mask=mask_i[:, None, None] & mask_j[None, :, None])

# ----------------------------------------------------------------------
def _fallback_trimul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """Very small‑N fallback using the tiny Triton kernel above."""
    B, N, _, H = left.shape
    out = torch.empty_like(left)
    # strides in element counts
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_ok, stride_oh = out.stride()
    grid = (
        B,
        (N + 31) // 32,   # i‑tiles (BLOCK_N=32)
        (N + 31) // 32,   # j‑tiles
    )
    _trimul_fallback_kernel[grid](
        left, right, out,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_ri, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_ok, stride_oh,
        BLOCK_N=32,
        num_warps=4,
    )
    return out

# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor,
                                   right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = A @ Bᵀ for tensors of shape [B,N,N,H] (fp16).
    Uses torch.bmm which on H100 maps to Tensor‑core GEMM.
    """
    B, N, _, H = left.shape
    M = B * H
    # reshape to [M, N, N]
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(M, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(M, N, N)
    # A @ Bᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # restore original layout [B,N,N,H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                     # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part of the pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the four projection + gate linear layers into two GEMMs
    # ------------------------------------------------------------------
    # left side (proj + gate)
    left_concat_w = torch.cat(
        [
            weights["left_proj.weight"],    # (H, D)
            weights["left_gate.weight"],    # (H, D)
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)            # (2H, D)

    # right side (proj + gate + out_gate)
    right_concat_w = torch.cat(
        [
            weights["right_proj.weight"],   # (H, D)
            weights["right_gate.weight"],   # (H, D)
            weights["out_gate.weight"],    # (H, D)
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)            # (3H, D)

    # fused linear passes (no bias)
    y_left = torch.nn.functional.linear(x, left_concat_w)   # [B,N,N,2H]
    y_right = torch.nn.functional.linear(x, right_concat_w) # [B,N,N,3H]

    # split and apply sigmoid where needed
    left_proj = y_left[..., :hidden_dim]
    left_gate = torch.sigmoid(y_left[..., hidden_dim:])

    right_proj = y_right[..., :hidden_dim]
    right_gate = torch.sigmoid(y_right[..., hidden_dim: 2 * hidden_dim])
    out_gate = torch.sigmoid(y_right[..., 2 * hidden_dim:])   # (B,N,N,H)

    # fuse gates into the projected tensors
    left = left_proj * left_gate      # [B,N,N,H]
    right = right_proj * right_gate   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction (batched GEMM)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    if N <= 128:
        # Very small N: use the tiny Triton fallback (avoid launch overhead)
        out = _fallback_trimul(left, right)   # [B,N,N,H] fp16
    else:
        # Use torch.bmm – on H100 this maps to a highly‑optimized cuBLAS kernel
        out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (fp16) → out‑gate → final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # Apply the out‑gate (computed earlier)
    out = out * out_gate

    # Final projection back to the original dimension
    to_out_w = weights["to_out.weight"].to(device=device, dtype=torch.float16)  # (dim, H)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Cast back to float32 (expected by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
