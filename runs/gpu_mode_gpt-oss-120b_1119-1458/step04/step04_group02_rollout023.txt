    out_ptr, out_gate_ptr, out_norm_ptr,
    norm_weight_ptr, norm_bias_ptr,
    total_positions,
    epsilon,
    BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    if pid >= total_positions:
        return

    # lane indices
    lane = tl.arange(0, BLOCK_H)
    mask = lane < BLOCK_H  # always true but used if more generic.

    # compute offset in flattened layout
    offset = pid * BLOCK_H + lane

    # load values
    out_val = tl.load(out_ptr + offset, mask=mask, other=0.0)
    gate_val = tl.load(out_gate_ptr + offset, mask=mask, other=0.0)
    w = tl.load(norm_weight_ptr + lane, mask=mask, other=0.0)
    b = tl.load(norm_bias_ptr + lane, mask=mask, other=0.0)

    # convert to fp32 for accumulation
    out_f32 = out_val.to(tl.float32)
    gate_f32 = gate_val.to(tl.float32)
    w_f32 = w.to(tl.float32)
    b_f32 = b.to(tl.float32)

    # compute mean and variance across hidden dimension
    sum_val = tl.sum(out_f32, axis=0)
    sum_sq = tl.sum(out_f32 * out_f32, axis=0)
    mean = sum_val / BLOCK_H

    inv_std = tl.rsqrt(var + epsilon)

    # normalize and scale
    normed = (out_f32 - mean) * inv_std
    normed = normed * w_f32 + b_f32
    # apply gate

"""
custom_kernel – Optimised “outgoing” TriMul forward pass (AlphaFold‑3).

Key ideas
---------
1️⃣  First LayerNorm is performed in fp32 (numerical stability).
2️⃣  Linear projections for left/right sides and their gates are fused:
        [proj, gate] → a single GEMM of shape (dim → 2·hidden) and then split.
    The output‑gate is computed with another single GEMM.
3️⃣  Optional mask is applied only to the projected values (left/right).
4️⃣  The O(N³) contraction is a batched GEMM:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    Implemented with torch.bmm on the reshaped tensors (full Tensor‑core speed).
5️⃣  A tiny Triton kernel fuses the second LayerNorm (over the hidden dimension),
    the output‑gate multiplication and the element‑wise affine transform
    (γ,β).  The kernel processes one (b,i,j) position per program and reduces
    across the hidden axis with warp‑level reductions.
6️⃣  The final projection back to the model dimension is a regular linear layer
    (fp16, cuBLAS) followed by a cast back to fp32 for the harness.

The heavy work is done by cuBLAS (steps 4 & 6) while the custom Triton kernel
covers the remaining per‑position work, guaranteeing at least one Triton use
and reducing the number of memory passes compared with the naïve PyTorch
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm (hidden dim), out‑gate multiplication.
# ----------------------------------------------------------------------
@triton.jit
def _norm_gate_kernel(
    out_ptr,            # FP16 tensor: [B,N,N,hidden] – raw contraction result
    out_gate_ptr,       # FP16 tensor: [B,N,N,hidden] – sigmoid(out_gate)
    out_norm_ptr,       # FP16 tensor: output of this kernel (same shape)
    norm_weight_ptr,    # FP16 tensor: γ of to_out_norm, shape [hidden]
    norm_bias_ptr,      # FP16 tensor: β of to_out_norm, shape [hidden]
    total_positions,    # int – B*N*N
    epsilon,            # float scalar for LayerNorm epsilon
    BLOCK_H: tl.constexpr,   # hidden dimension (block size, multiple of 32)
):
    """
    Each program handles one (b,i,j) position.  BLOCK_H threads load the
    hidden‑dim vector, compute mean/variance, apply LayerNorm, multiply by the
    output gate and write the result back.
    """
    pid = tl.program_id(0)                     # index of the (b,i,j) entry
    if pid >= total_positions:
        return

    # lane index inside the block (0 … BLOCK_H‑1)
    lane = tl.arange(0, BLOCK_H)

    # pointer offset for the current hidden vector (contiguous layout)
    off = pid * BLOCK_H + lane                  # element offset (FP16)

    # masks (needed only when BLOCK_H > hidden, which never happens here)
    mask = lane < BLOCK_H

    # ---- load raw values -------------------------------------------------
    out_val = tl.load(out_ptr + off, mask=mask, other=0.0).to(tl.float32)
    gate_val = tl.load(out_gate_ptr + off, mask=mask, other=0.0).to(tl.float32)
    gamma = tl.load(norm_weight_ptr + lane, mask=mask, other=0.0).to(tl.float32)   # γ
    beta = tl.load(norm_bias_ptr + lane, mask=mask, other=0.0).to(tl.float32)     # β

    # ---- layer‑norm reduction over hidden dimension ----------------------
    sum_val = tl.sum(out_val, axis=0)                 # Σ_h out
    sum_sq  = tl.sum(out_val * out_val, axis=0)       # Σ_h out²
    inv_hidden = 1.0 / BLOCK_H
    mean = sum_val * inv_hidden
    var = sum_sq * inv_hidden - mean * mean
    inv_std = tl.rsqrt(var + epsilon)

    # ---- apply norm, affine transform and output gate --------------------
    normalized = (out_val - mean) * inv_std
    normalized = normalized * gamma + beta
    normalized = normalized * gate_val                 # out_gate multiplication

    # ---- store result ----------------------------------------------------
    tl.store(out_norm_ptr + off, normalized.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------
# Helper: batched matmul of the form   out = left @ rightᵀ    (hidden‑wise)
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor,
                                   right: torch.Tensor) -> torch.Tensor:
    """
    left/right: [B, N, N, H]  (fp16)
    Returns:    [B, N, N, H]  (fp16) where out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] – (i,k) for left, (k,j) for right
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (i,k)
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)  # (k,j)
    out_flat = torch.bmm(left_flat, right_flat)                         # (i,j) = Σ_k
    # restore original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor[B, N, N]       (float32, bool or all‑ones)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : {"dim":int, "hidden_dim":int, "nomask":bool (optional)}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5  # LayerNorm epsilon

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy part
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)                     # [B,N,N,dim] fp16
    B, N, _, _ = x_h.shape
    flat_x = x_h.view(-1, dim)                    # (B*N*N, dim)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gating linear layers (left & right)
    # ------------------------------------------------------------------
    # left side: [proj, gate] → weight shape (2*hidden, dim)
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                           # (2*hidden, dim)
    left_concat = F.linear(flat_x, left_concat_w)  # (B*N*N, 2*hidden)
    left_proj_raw, left_gate_raw = left_concat.split(hidden_dim, dim=1)
    left_gate = torch.sigmoid(left_gate_raw)       # (B*N*N, hidden)
    left = left_proj_raw

    # right side: [proj, gate]
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)
    right_concat = F.linear(flat_x, right_concat_w)
    right_proj_raw, right_gate_raw = right_concat.split(hidden_dim, dim=1)
    right_gate = torch.sigmoid(right_gate_raw)
    right = right_proj_raw

    # ------------------------------------------------------------------
    # 4️⃣ Output gate (single linear)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate_raw = F.linear(flat_x, out_gate_w)    # (B*N*N, hidden)
    out_gate = torch.sigmoid(out_gate_raw)         # (B*N*N, hidden)

    # ------------------------------------------------------------------
    # 5️⃣ Reshape to 4‑D tensors
    # ------------------------------------------------------------------
    left = left.view(B, N, N, hidden_dim)          # [B,N,N,hidden]
    left_gate = left_gate.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    right_gate = right_gate.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 6️⃣ Optional mask (apply to projections only, before gating)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(dtype=torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 7️⃣ Apply the per‑side gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 8️⃣ O(N³) contraction via batched GEMM (Tensor‑core bound)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hidden] fp16

    # ------------------------------------------------------------------
    # 9️⃣ Fuse second LayerNorm, its affine transform and the output gate
    # ------------------------------------------------------------------
    # allocate buffer for the normalized & gated tensor
    out_normed = torch.empty_like(out)                 # fp16

    total_positions = B * N * N
    grid = (total_positions,)

    # weights for the hidden‑dim LayerNorm (γ, β)
    norm_w_fp16 = weights["to_out_norm.weight"].to(torch.float16)
    norm_b_fp16 = weights["to_out_norm.bias"].to(torch.float16)

    # launch Triton kernel
    _norm_gate_kernel[grid](
        out.data_ptr(),
        out_gate.data_ptr(),
        out_normed.data_ptr(),
        norm_w_fp16.data_ptr(),
        norm_b_fp16.data_ptr(),
        total_positions,
        eps,
        BLOCK_H=hidden_dim,
        num_warps=max(1, hidden_dim // 32),   # at least one warp
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 10️⃣ Final projection back to original dimension (no bias)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden)
    out_flat = out_normed.view(-1, hidden_dim)              # (B*N*N, hidden)
    final = F.linear(out_flat, to_out_w)                    # (B*N*N, dim)
    final = final.view(B, N, N, dim)

    # ------------------------------------------------------------------
    # 11️⃣ Cast back to float32 as required by the harness
    # ------------------------------------------------------------------
    return final.to(torch.float32)
