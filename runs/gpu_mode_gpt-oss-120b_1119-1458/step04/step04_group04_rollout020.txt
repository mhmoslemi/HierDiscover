def fused_mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    mask_ptr, hidden_dim, size, BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    # load left, left_gate, right, right_gate
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    l_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    r_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # compute mask index
    # Compute parent index for mask: each hidden_dim elements share same mask value
    mask_idx = offs // hidden_dim
    # We need mask offset in pointer: mask_ptr + mask_idx
    # This requires integer division on int64; tl.int32 supports it.
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # default 1.0 if out-of-range

    # fused multiplication
    left_out = left * l_gate * mask_val
    right_out = right * r_gate * mask_val

    # store back
    tl.store(left_ptr + offs, left_out, mask=mask)
    tl.store(right_ptr + offs, right_out, mask=mask)

def fused_gate_mask_kernel(
    proj_ptr,      # *fp16 - left_proj flattened
    gate_pre_ptr,  # *fp16 - left_gate_pre flattened
    proj2_ptr,     # *fp16 - right_proj flattened
    gate_pre2_ptr, # *fp16 - right_gate_pre flattened
    mask_ptr,      # *fp16 - mask flattened, length size//hidden_dim
    hidden_dim: tl.constexpr,
    size: tl.int32,
    BLOCK: tl.constexpr,
):

    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    
    # Load
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate_pre = tl.load(gate_pre_ptr + offs, mask=mask, other=0.0)
    proj2 = tl.load(proj2_ptr + offs, mask=mask, other=0.0)

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # layernorm
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # cast to fp16
    x_fp16 = x_norm.to(torch.float16)
    B, N, _, _ = x_fp16.shape

    # flatten input for linear
    x_flat = x_fp16.view(-1, dim)

    # assemble combined weight
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    combined_w = torch.cat([left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0) # shape (4*hidden_dim, dim)

    combined_out = torch.nn.functional.linear(x_flat, combined_w)  # [M, 4*hidden_dim]

    left_gate_pre = combined_out[..., hidden_dim:2*hidden_dim]

    right_gate_pre = combined_out[..., 3*hidden_dim:]

    out_gate_w = weights["out_gate.weight"].to(torch.float16)

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def fused_gate_mask_kernel(
    proj_ptr,          # *fp16 : left projection (mutated)
    gate_pre_ptr,      # *fp16 : left gate pre-sigmoid (read-only)
    proj2_ptr,         # *fp16 : right projection (mutated)
    gate2_pre_ptr,     # *fp16 : right gate pre-sigmoid (read-only)
    mask_ptr,          # *fp16 : mask flattened (size = B*N*N)
    hidden_dim,        # i32   : hidden dimension (for broadcasting)
    size,              # i32   : total number of elements (B*N*N*hidden_dim)
    BLOCK: tl.constexpr,# i32   : tile size (num threads per block)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)  # linear indices
    mask = offs < size

    # load data
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)       # left projection
    gate_pre = tl.load(gate_pre_ptr + offs, mask=mask, other=0.0)  # left gate pre
    proj2 = tl.load(proj2_ptr + offs, mask=mask, other=0.0)    # right projection
    gate2_pre = tl.load(gate2_pre_ptr + offs, mask=mask, other=0.0)  # right gate pre

    # compute mask index (broadcast across hidden_dim)
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # default 1.0

    # sigmoid of gate values
    gate = 1.0 / (1.0 + tl.exp(-gate_pre))
    gate2 = 1.0 / (1.0 + tl.exp(-gate2_pre))

    # fused multiply and mask
    proj = proj * gate * mask_val
    proj2 = proj2 * gate2 * mask_val

    # write results back in-place
    tl.store(proj_ptr + offs, proj, mask=mask)
    tl.store(proj2_ptr + offs, proj2, mask=mask)

def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform batched GEMM: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using torch.bmm (cuBLAS). Tensors are assumed fp16, shape [B,N,N,hidden].
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N]
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))  # [B*H, N, N]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).
    Implements fused weight projection + gate + optional mask using a custom Triton kernel,
    and performs the heavy O(N³) contraction via a batched GEMM.
    All heavy computation is in FP16; only the first LayerNorm is performed in FP32.
    Returns a float32 tensor of shape [B, N, N, dim].
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable for the initial normalization
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the bulk of the computation
    # ------------------------------------------------------------------
    x = x_norm.to(torch.float16)    # [B,N,N,dim] fp16
    B, N, _, _ = x.shape
    M = B * N * N

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (left/right) *and* their gates in a single GEMM
    # ------------------------------------------------------------------
    # gather weight tensors and convert to fp16 once
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)

    # combine the four matrices into one big weight [4*hidden, dim]
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )  # shape (4*hidden_dim, dim)

    # flatten input for linear
    x_flat = x.view(-1, dim)  # [M, dim]

    # single linear for all four transforms
    combined = torch.nn.functional.linear(x_flat, combined_weight)  # [M, 4*hidden]
    combined = combined.view(B, N, N, 4 * hidden_dim).contiguous()

    # slice the four tensors
    left_proj   = combined[..., :hidden_dim]
    left_gate_pre = combined[..., hidden_dim:2 * hidden_dim]
    right_proj  = combined[..., 2 * hidden_dim:3 * hidden_dim]
    right_gate_pre = combined[..., 3 * hidden_dim:]

    # ------------------------------------------------------------------
    # 4️⃣ Out‑gate (separate linear) – kept separate because it
    #    is needed after the contraction.
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate_pre = torch.nn.functional.linear(x_flat, out_gate_w)  # [M, hidden]
    out_gate = torch.sigmoid(out_gate_pre).view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Apply gate + optional mask (fused Triton kernel)
    # ------------------------------------------------------------------
    if not nomask:
        # mask broadcasted on the hidden dimension
        mask_fp16 = mask_tensor.to(torch.float16)   # [B,N,N]
        mask_flat = mask_fp16.view(-1)              # [B*N*N]

        size = left_proj.numel()    # B*N*N*hidden_dim
        BLOCK = 1024                # threads per block, 1024 is PTX max
        grid = ( (size + BLOCK - 1) // BLOCK, )

        fused_gate_mask_kernel[grid](
            left_proj.data_ptr(),
            left_gate_pre.data_ptr(),
            right_proj.data_ptr(),
            right_gate_pre.data_ptr(),
            mask_flat.data_ptr(),
            hidden_dim,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # kernels wrote the final gated+masked values back into left_proj/right_proj
        left = left_proj
        right = right_proj
    else:
        # No mask – just apply sigmoid gate
        left = left_proj * torch.sigmoid(left_gate_pre)
        right = right_proj * torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction – batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    # ensure tensors are contiguous before permute
    left = left.contiguous()
    right = right.contiguous()
    out = _batched_matmul_torch(left, right)   # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16) → gating → final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float16)   # shape (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)        # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to float32 for downstream consumption
    # ------------------------------------------------------------------
    return out.to(torch.float32)

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse projected values, sigmoid‑gates and optional mask.
#   proj   := left or right linear projection (fp16)
#   gate   := pre‑sigmoid gate (fp16)
#   mask   := broadcast mask (fp16) of shape [B,N,N] -> flattened [B*N*N]
#   hidden := size of the last dimension (hidden_dim)
# The kernel overwrites proj with   proj * sigmoid(gate) * mask .
# ----------------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    proj_ptr,          # *fp16 : left projection (in‑place output)
    gate_pre_ptr,      # *fp16 : left gate   (read‑only)
    proj2_ptr,         # *fp16 : right projection (in‑place output)
    gate2_pre_ptr,     # *fp16 : right gate  (read‑only)
    mask_ptr,          # *fp16 : flattened mask (size = B*N*N)
    hidden_dim,        # i32   : hidden dimension (for broadcasting)
    size,              # i32   : total number of elements (B*N*N*hidden_dim)
    BLOCK: tl.constexpr,   # compile‑time tile size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)      # linear offsets
    valid = offs < size

    # load tensors
    proj = tl.load(proj_ptr + offs,   mask=valid, other=0.0)
    gate_pre = tl.load(gate_pre_ptr + offs, mask=valid, other=0.0)
    proj2 = tl.load(proj2_ptr + offs, mask=valid, other=0.0)
    gate2_pre = tl.load(gate2_pre_ptr + offs, mask=valid, other=0.0)

    # broadcast mask: each hidden dimension share the same mask entry
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    # sigmoid = 1/(1+exp(-x)) – works in fp16
    gate = 1.0 / (1.0 + tl.exp(-gate_pre))
    gate2 = 1.0 / (1.0 + tl.exp(-gate2_pre))

    # fused multiplication + mask
    proj = proj * gate * mask_val
    proj2 = proj2 * gate2 * mask_val

    # write results back in‑place
    tl.store(proj_ptr + offs, proj, mask=valid)
    tl.store(proj2_ptr + offs, proj2, mask=valid)

def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """Batched matrix multiply (cuBLAS) on fp16 tensors."""
    B, N, _, H = left.shape
    # reshape to [B*H, N, N]
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # restore original layout: [B,N,N,H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out
