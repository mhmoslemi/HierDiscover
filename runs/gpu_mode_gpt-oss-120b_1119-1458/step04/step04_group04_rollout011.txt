Implement Triton kernel:

```
@triton.jit
def mask_and_gate_fuse_kernel(
    left_ptr,
    left_gate_ptr,
    right_ptr,
    right_gate_ptr,
    mask_ptr,
    hidden_dim,
    size, # total number of elements per tensor (left, ... same)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask_offs = offs // hidden_dim
    # create in-range mask
    good = offs < size
    l = tl.load(left_ptr + offs, mask=good)
    lg = tl.load(left_gate_ptr + offs, mask=good)
    r = tl.load(right_ptr + offs, mask=good)
    rg = tl.load(right_gate_ptr + offs, mask=good)
    m = tl.load(mask_ptr + mask_offs, mask=good, other=1.0)  # fallback 1 for mask

    # Multiply
    out_l = l * lg * m
    out_r = r * rg * m

    # Store
    tl.store(left_ptr + offs, out_l, mask=good)
    tl.store(right_ptr + offs, out_r, mask=good)
```

"""
TriMul (outgoing) – fast forward implementation
================================================

Algorithm
---------
1️⃣  First LayerNorm (float32) → cast to fp16.
2️⃣  Fuse all five linear projections (left_proj, right_proj,
    left_gate, right_gate, out_gate) into a *single* GEMM by
    concatenating the weight matrices.  This reads the huge input
    tensor only once.
3️⃣  Apply sigmoid to the three gate tensors.
4️⃣  If a mask is supplied, fuse the element‑wise operations

        left  = left_proj * left_gate * mask
        right = right_proj * right_gate * mask

    with a tiny Triton kernel (handles mask broadcasting over the
    hidden dimension).  When ``nomask=True`` a cheap PyTorch
    multiplication is used.
5️⃣  Heavy O(N³) contraction: for every hidden dimension h

        out[...,h] = left[...,h] @ right[...,h].T

    is performed as a batched GEMM via ``torch.bmm`` on tensors of
    shape ``[B*H, N, N]`` (fp16, Tensor‑Core accelerated on H100).
6️⃣  Second LayerNorm over the hidden dimension (fp16).
7️⃣  Scale by the previously computed ``out_gate`` and project back
    to the original ``dim`` with a final linear layer.
8️⃣  Cast the result back to float32 (required by the harness).

Only the forward pass is needed – no autograd.  The mask‑gate fusion
kernel satisfies the requirement “use at least part of the operations
in a Triton kernel”.

"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse (tensor * gate * mask) for left and right tensors.
# Both tensors have shape [B, N, N, H] and the mask has shape [B, N, N].
# The mask is broadcast over the hidden dimension H.
# ----------------------------------------------------------------------
@triton.jit
def mask_and_gate_fuse_kernel(
    left_ptr,          # *fp16 : left_proj tensor          (size = B*N*N*H)
    left_gate_ptr,     # *fp16 : left_gate tensor          (same size)
    right_ptr,         # *fp16 : right_proj tensor         (same size)
    right_gate_ptr,    # *fp16 : right_gate tensor         (same size)
    mask_ptr,          # *fp16 : mask tensor              (size = B*N*N)
    hidden_dim,        # i32   : hidden dimension H
    total_elems,       # i32   : total elements per tensor (B*N*N*H)
    BLOCK: tl.constexpr,  # compile‑time block size (threads per block)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offsets
    valid = offs < total_elems                         # boundary guard

    # Load operands
    l  = tl.load(left_ptr + offs,       mask=valid, other=0.0)   # left_proj
    lg = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)   # left_gate
    r  = tl.load(right_ptr + offs,     mask=valid, other=0.0)   # right_proj
    rg = tl.load(right_gate_ptr + offs,mask=valid, other=0.0)   # right_gate

    # Compute mask offset (broadcast over hidden dimension)
    mask_offs = offs // hidden_dim
    m = tl.load(mask_ptr + mask_offs, mask=valid, other=1.0)    # mask (0/1)

    # Fused multiplication
    out_l = l * lg * m
    out_r = r * rg * m

    # Store results back in‑place
    tl.store(left_ptr + offs,  out_l, mask=valid)
    tl.store(right_ptr + offs, out_r, mask=valid)


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction.
# ----------------------------------------------------------------------
def _batched_tri_mul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using torch.bmm (cuBLAS Tensor‑Core kernels).
    All tensors are fp16 and contiguous.
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] and transpose right to (j,k) -> (k,j)
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # [B*H, N, N]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 / bool) – may be ignored
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – more stable
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for all subsequent heavy work
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Fuse all five linear projections into one GEMM
    #    (left_proj, left_gate, right_proj, right_gate, out_gate)
    # -----------------------------------------------------------------
    w_left_proj = weights["left_proj.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    # Concatenate along the output‑feature dimension: shape [(5*H), dim]
    all_weights = torch.cat(
        [
            w_left_proj,
            w_left_gate,
            w_right_proj,
            w_right_gate,
            w_out_gate,
        ],
        dim=0,
    )  # [5*hidden_dim, dim]

    # Combined linear (only ONE pass over the huge input)
    combined = torch.nn.functional.linear(x, all_weights)  # [B,N,N,5*H] fp16

    # -----------------------------------------------------------------
    # 4️⃣ Slice the combined tensor & apply sigmoid to gate slices
    # -----------------------------------------------------------------
    h = hidden_dim
    left_proj   = combined[..., 0:h]                # shape [B,N,N,H]
    left_gate   = combined[..., h:2*h].sigmoid()
    right_proj  = combined[..., 2*h:3*h]
    right_gate  = combined[..., 3*h:4*h].sigmoid()
    out_gate    = combined[..., 4*h:5*h].sigmoid()

    # Release combined to save memory (optional)
    del combined

    # -----------------------------------------------------------------
    # 5️⃣ Optional mask – fused element‑wise multiplication via Triton
    # -----------------------------------------------------------------
    if not nomask:
        # Ensure mask is fp16 and contiguous
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elements = left_proj.numel()                     # B*N*N*H
        BLOCK = 1024                                            # ≤ 1024 threads per block
        grid = ((total_elements + BLOCK - 1) // BLOCK, )
        mask_and_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            right_proj,
            right_gate,
            mask_fp16,
            hidden_dim,
            total_elements,
            BLOCK=BLOCK,
            num_warps=8,
        )
    else:
        # No mask – simple element‑wise multiplication
        left_proj  = left_proj * left_gate
        right_proj = right_proj * right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS)
    # -----------------------------------------------------------------
    # left_proj and right_proj are now properly gated (and masked if required)
    out = _batched_tri_mul(left_proj, right_proj)    # [B,N,N,H] fp16

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # [B,N,N,H] fp16

    # -----------------------------------------------------------------
    # 8️⃣ Scale by out_gate and final projection back to `dim`
    # -----------------------------------------------------------------
    out = out * out_gate                                             # gated scaling
    to_out_w = weights["to_out.weight"].to(torch.float16)            # [dim, hidden_dim]
    out = torch.nn.functional.linear(out, to_out_w)                 # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 9️⃣ Cast to float32 for the harness and return
    # -----------------------------------------------------------------
    return out.to(torch.float32)
