def mask_gate_mul_kernel(x_ptr, gate_ptr, mask_ptr, size, hidden, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    mask_idx = offs // hidden
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    tl.store(x_ptr + offs, x * g * m, mask=mask)

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no gradients).
    Implements:
        1) LayerNorm on input.
        2) Linear projections (left/right) and three gates (left_gate, right_gate, out_gate).
        3) Optional mask fused with gating via a tiny Triton kernel.
        4) Batched GEMM: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
        5) Second LayerNorm, out‑gate, final linear projection.
    All heavy numeric work is performed in fp16; final result returned as fp32.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # 1. First LayerNorm (float32 for numeric stability)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    # Cast to fp16 for subsequent heavy ops
    x = x.to(torch.float16)

    # 2. Cast model weights to fp16 once
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # 3. Linear projections (fp16)
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)

    # 4. Gates (sigmoid of linear)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # 5. Optional mask fused with gating (element‑wise product)
    if not nomask:
        # mask shape [B,N,N] -> fp16 1‑D view
        mask_fp16 = mask_tensor.to(torch.float16)
        # Flatten tensors
        size = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(size, BLOCK),)

        left_flat = left.view(-1)
        left_gate_flat = left_gate.view(-1)
        mask_flat = mask_fp16.view(-1)

        mask_gate_mul_kernel[grid](
            left_flat, left_gate_flat, mask_flat,
            size, hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )

        right_flat = right.view(-1)
        right_gate_flat = right_gate.view(-1)
        mask_gate_mul_kernel[grid](
            right_flat, right_gate_flat, mask_flat,
            size, hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask: simple element‑wise fusion with a tiny Triton kernel
        size = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(size, BLOCK),)

        left_flat = left.view(-1)
        left_gate_flat = left_gate.view(-1)
        elem_mul_kernel[grid](
            left_flat, left_gate_flat,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        right_flat = right.view(-1)
        right_gate_flat = right_gate.view(-1)
        elem_mul_kernel[grid](
            right_flat, right_gate_flat,
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )

    # 6. Heavy O(N³) contraction via batched GEMM (cuBLAS)
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to [B*H, N, N] for torch.bmm
    left_bmm = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_bmm = right.permute(0, 3, 1, 2).reshape(B * H, N, N)
    # C = A @ Bᵀ
    out_bmm = torch.bmm(left_bmm, right_bmm.transpose(1, 2))
    # restore original layout -> [B, N, N, hidden]
    out = out_bmm.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 7. Second LayerNorm over hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # 8. Apply out‑gate
    out = out * out_gate

    # 9. Final linear projection back to dim
    out = torch.nn.functional.linear(out, to_out_w)

    # 10. Cast back to float32 for downstream code
    return out.to(torch.float32)
```

"""
Outgoing TriMul (AlphaFold‑3) – fast Triton‑enabled forward pass.

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (float32) → cast to fp16.
2️⃣  Fuse the five *weight·input* linear layers (left‑proj, left‑gate,
    right‑proj, right‑gate, out‑gate) into a single GEMM:
        proj = X @ Wᵀ   with W = [left_proj; left_gate; right_proj;
                               right_gate; out_gate]    (shape 5·H × D)
   Split the result into the five tensors and apply sigmoid to the
   three gate tensors.
3️⃣  Optional mask: a tiny Triton kernel fuses mask·gate·tensor for
   `left` and `right` (mask is broadcast over the hidden dimension).
   If no mask is present a second tiny kernel just does `x *= gate`.
4️⃣  Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (`C = A @ Bᵀ`) using cuBLAS via
   `torch.bmm`.  The tensors are reshaped to [B·H, N, N] before the
   matmul and reshaped back afterwards.
5️⃣  Second layer‑norm on the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.
6️⃣  Cast the result back to float32 (the harness expects float32).

Only the forward pass is required; the Triton kernels are used for
the element‑wise fusions (mask + gate or plain gate) which guarantees
that the solution contains custom Triton code while keeping the
overall runtime dominated by highly‑optimised cuBLAS kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiply (x *= y)
# ----------------------------------------------------------------------
@triton.jit
def elem_mul_kernel(
    x_ptr,          # *fp16
    y_ptr,          # *fp16
    size,           # i32 – total number of elements
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    y = tl.load(y_ptr + offs, mask=mask, other=0.0)

    tl.store(x_ptr + offs, x * y, mask=mask)


# ----------------------------------------------------------------------
# Triton kernel: multiply three tensors (x *= gate * mask)
# mask is a 1‑D tensor of size (B·N·N); it is broadcast over the
# hidden dimension (`hidden`).  The kernel computes
#   out[i] = x[i] * gate[i] * mask[i // hidden]
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_mul_kernel(
    x_ptr,          # *fp16  (B·N·N·H)
    gate_ptr,       # *fp16  same shape as x
    mask_ptr,       # *fp16  (B·N·N)
    size,           # i32   total number of elements in x
    hidden,         # i32   hidden dimension (H)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    # load operands
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # compute the corresponding mask index (broadcast over hidden)
    mask_idx = offs // hidden
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # 1.0 = no‑mask

    # fused multiplication
    tl.store(x_ptr + offs, x * g * m, mask=mask)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` flag.

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack
    # -----------------------------------------------------------------
    inp, mask, weight, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣  First LayerNorm (float32 for stability)
    # -----------------------------------------------------------------
    norm_w = weight["norm.weight"]
    norm_b = weight["norm.bias"]
    x = torch.nn.functional.layer_norm(inp, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)                     # half‑precision for the rest

    # -----------------------------------------------------------------
    # 2️⃣  Fuse five linear projections into a single matmul
    # -----------------------------------------------------------------
    # Convert weights once
    left_proj_w   = weight["left_proj.weight"].to(torch.float16)
    right_proj_w  = weight["right_proj.weight"].to(torch.float16)
    left_gate_w   = weight["left_gate.weight"].to(torch.float16)
    right_gate_w  = weight["right_gate.weight"].to(torch.float16)
    out_gate_w    = weight["out_gate.weight"].to(torch.float16)

    # Concatenate: [left_proj; left_gate; right_proj; right_gate; out_gate]
    fused_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape (5*hidden, dim)

    # Flatten input for a single GEMM
    B, N, _, _ = x.shape
    x_flat = x.contiguous().view(-1, dim)           # (B·N·N, dim)
    proj_all = torch.nn.functional.linear(x_flat, fused_w)      # (B·N·N, 5*hidden)
    proj_all = proj_all.view(B, N, N, 5 * hidden)   # (B, N, N, 5*hidden)

    # Split the concatenated projection
    left      = proj_all[..., 0                : hidden]               # (B,N,N,H)
    left_gate = proj_all[..., hidden            : hidden * 2]          # (B,N,N,H)
    right     = proj_all[..., hidden * 2        : hidden * 3]          # (B,N,N,H)
    right_gate= proj_all[..., hidden * 3        : hidden * 4]          # (B,N,N,H)
    out_gate  = proj_all[..., hidden * 4        : hidden * 5]          # (B,N,N,H)

    # Apply sigmoid to gate tensors
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # -----------------------------------------------------------------
    # 3️⃣  Optional mask – fused with gating via Triton
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(torch.float16).contiguous()           # (B,N,N)
        # flatten views
        total_elems = left.numel()                              # B·N·N·H
        BLOCK = 1024
        grid = (triton.cdiv(total_elems, BLOCK),)

        left_flat_view      = left.contiguous().view(-1)
        left_gate_flat_view = left_gate.contiguous().view(-1)
        mask_flat_view      = mask_fp16.view(-1)

        mask_gate_mul_kernel[grid](
            left_flat_view,
            left_gate_flat_view,
            mask_flat_view,
            total_elems,
            hidden,
            BLOCK=BLOCK,
            num_warps=8,
        )

        right_flat_view      = right.contiguous().view(-1)
        right_gate_flat_view = right_gate.contiguous().view(-1)

        mask_gate_mul_kernel[grid](
            right_flat_view,
            right_gate_flat_view,
            mask_flat_view,
            total_elems,
            hidden,
            BLOCK=BLOCK,
            num_warps=8,
        )
    else:
        # No mask: plain element‑wise fuse using a tiny kernel
        total_elems = left.numel()
        BLOCK = 1024
        grid = (triton.cdiv(total_elems, BLOCK),)

        elem_mul_kernel[grid](
            left.contiguous().view(-1),
            left_gate.contiguous().view(-1),
            total_elems,
            BLOCK=BLOCK,
            num_warps=8,
        )
        elem_mul_kernel[grid](
            right.contiguous().view(-1),
            right_gate.contiguous().view(-1),
            total_elems,
            BLOCK=BLOCK,
            num_warps=8,
        )

    # -----------------------------------------------------------------
    # 4️⃣  Heavy O(N³) contraction (batched GEMM via cuBLAS)
    # -----------------------------------------------------------------
    # left/right now have shape (B, N, N, hidden)
    # reshape to [B·hidden, N, N] for torch.bmm
    left_bmm  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right_bmm = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N)

    # C = A @ Bᵀ   (each batch corresponds to a different hidden channel)
    out_bmm = torch.bmm(left_bmm, right_bmm.transpose(1, 2))
    # reshape back to [B, N, N, hidden]
    out = out_bmm.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 5️⃣  Second LayerNorm, out‑gate and final projection
    # -----------------------------------------------------------------
    to_out_norm_w = weight["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weight["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate                       # broadcasted element‑wise

    to_out_w = weight["to_out.weight"].to(torch.float16)   # (dim, hidden)
    out = torch.nn.functional.linear(out, to_out_w)       # -> (B,N,N,dim)

    # -----------------------------------------------------------------
    # 6️⃣  Cast back to float32 for the harness
    # -----------------------------------------------------------------
    return out.to(torch.float32)
