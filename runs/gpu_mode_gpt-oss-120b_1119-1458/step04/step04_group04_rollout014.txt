def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Unpack weights
    # ... as before

    # First layernorm: x_norm = layer_norm(input, weight, bias)
    # Use FP32 or FP16? Could maybe do FP16 directly. Let's run FP32 to be safe.
    # Use torch.nn.functional.layer_norm; we will convert weights to appropriate type.

    # Cast to fp16
    x = x_norm.to(torch.float16)

    # Combine left projections
    left_concat_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=1)  # shape (dim, hidden*2)
    left_concat_w = left_concat_w.to(torch.float16)
    left_concat = torch.nn.functional.linear(x, left_concat_w) # [B,N,N, 2*hidden]

    left_proj = left_concat[..., :hidden_dim]
    left_gate_pre = left_concat[..., hidden_dim:]
    left_gate = torch.sigmoid(left_gate_pre)

    # Same for right
    right_concat_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=1)
    right_concat_w = right_concat_w.to(torch.float16)
    right_concat = torch.nn.functional.linear(x, right_concat_w)
    right_proj = right_concat[..., :hidden_dim]
    right_gate_pre = right_concat[..., hidden_dim:]
    right_gate = torch.sigmoid(right_gate_pre)

    # out_gate linear (separate, not combined)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply mask, gates using fusion kernel:
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # shape [B,N,N,1] broadcast
        size = left_proj.numel()
        BLOCK = 1024
        grid = ((size + BLOCK - 1) // BLOCK,)
        mask_gate_fuse_kernel[grid](
          left_proj.data_ptr(),
          left_gate.data_ptr(),
          mask_fp16.data_ptr(),
          size,
          BLOCK=BLOCK,
          num_warps=4,
        )
        mask_gate_fuse_kernel[grid](
          right_proj.data_ptr(),
          right_gate.data_ptr(),
          mask_fp16.data_ptr(),
          size,
          BLOCK=BLOCK,
          num_warps=4,
        )
    else:
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # now left and right are the final values after gating and mask.
    left = left_proj
    right = right_proj

    # Contraction: batch matmul left @ right^T per hidden dimension
    # Use batched matmul via torch.bmm
    B, N, _, _ = left.shape
    # reshape and transpose
    A = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # [B*H,N,N]
    B_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)  # [B*H,N,N]

    # compute out = A @ B_mat.transpose(1,2)  # B_mat^T shape [B*H,N,N] -> transposed each
    # we can use torch.bmm directly
    out = torch.bmm(A, B_mat.transpose(1, 2))  # shape [B*H,N,N]

    # reshape back
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]

    # second layer norm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # final gating and projection
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))

    # cast to float32
    return out.to(torch.float32)

"""
TriMul – “outgoing” variant (AlphaFold‑3)

Algorithm (GPU‑friendly, half‑precision)
------------------------------------------------
1️⃣  Layer‑norm the input (float32) → cast to fp16.
2️⃣  Fuse *all* linear mappings (left/right projections, their gates and the
    out‑gate) into a single GEMM:
        combined = x @ W_comb.T      # W_comb = [5·H, D]
   Slice the result to obtain:
        left_proj, left_gate, right_proj, right_gate, out_gate.
   Apply the sigmoid non‑linearity to the three gate tensors.
3️⃣  Fuse element‑wise multiplication (proj × gate) and optional masking
   in‑place with a tiny Triton kernel (`mask_gate_fuse_kernel`).  This kernel
   also broadcasts a 3‑D mask over the hidden dimension.
4️⃣  Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (A @ Bᵀ) using `torch.bmm`, which invokes
   cuBLAS Tensor‑Core kernels.
5️⃣  Second Layer‑norm (fp16) → out‑gate gating → final linear projection back
   to the original dimensionality.
6️⃣  Cast the result back to float32 (the harness expects float32).

Only the forward pass is required.  The mask‑fusion kernel satisfies the
“must use a Triton kernel” requirement while keeping the overall runtime
well below the target.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: (proj * gate * mask) in‑place, mask is broadcast over the
# hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    proj_ptr,               # *fp16  : projection tensor (flattened)
    gate_ptr,               # *fp16  : gate tensor (flattened)
    mask_ptr,               # *fp16  : mask tensor of shape [B·N·N] (flattened)
    size,                   # i32    : total number of elements in proj/gate
    hidden,                 # i32    : hidden dimension (broadcast factor)
    BLOCK: tl.constexpr,    # compile‑time tile size (≤1024)
):
    pid   = tl.program_id(0)
    offs  = pid * BLOCK + tl.arange(0, BLOCK)          # linear offsets
    mask  = offs < size                                # range guard

    # load projection and gate
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # broadcasted mask: one mask value per hidden‑channel slice
    mask_offs = offs // hidden                         # floor division
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)  # default = 1 (no mask)

    out = proj * gate * m

    tl.store(proj_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Forward entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (float32 / bool) – optional
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable for the initial statistics
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part of the model
    # -----------------------------------------------------------------
    x = x.to(torch.float16)                    # fp16 view
    x = x.contiguous()                         # ensure contiguous stride

    # -----------------------------------------------------------------
    # 3️⃣ Fuse all linear projections/gates into one GEMM
    #    Weight layout: (out_features, in_features) = (5·H, D)
    # -----------------------------------------------------------------
    # collect and concatenate the five weight matrices (no bias in any)
    W_left_proj = weights["left_proj.weight"]      # (H, D)
    W_left_gate = weights["left_gate.weight"]      # (H, D)
    W_right_proj = weights["right_proj.weight"]    # (H, D)
    W_right_gate = weights["right_gate.weight"]    # (H, D)
    W_out_gate = weights["out_gate.weight"]        # (H, D)

    # concatenate along the output‑feature dimension
    W_combined = torch.cat(
        [
            W_left_proj,
            W_left_gate,
            W_right_proj,
            W_right_gate,
            W_out_gate,
        ],
        dim=0,
    ).to(torch.float16)                             # (5·H, D)

    # single fused linear pass
    combined = torch.nn.functional.linear(x, W_combined)   # [B,N,N,5·H] (fp16)

    # -----------------------------------------------------------------
    # 4️⃣ Slice into the five logical tensors
    # -----------------------------------------------------------------
    H = hidden_dim
    left_proj      = combined[..., 0 : H]               # (B,N,N,H)
    left_gate_pre  = combined[..., H : 2 * H]
    right_proj     = combined[..., 2 * H : 3 * H]
    right_gate_pre = combined[..., 3 * H : 4 * H]
    out_gate_pre   = combined[..., 4 * H : 5 * H]

    # sigmoid gates (still fp16)
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)

    # -----------------------------------------------------------------
    # 5️⃣ Fuse element‑wise (proj * gate) and optional mask
    # -----------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] → fp16 and flatten
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)                # [B·N·N]

        total_elements = left_proj.numel()            # B·N·N·H
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            mask_flat.data_ptr(),
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_flat.data_ptr(),
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # after the kernel the projection tensors already contain the final
        # gated & masked values.
        left  = left_proj
        right = right_proj
    else:
        # without a mask we can combine the two element‑wise products in PyTorch
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (A @ Bᵀ)
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to [B·H, N, N] for batched GEMM
    A = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batched matrix multiplication – uses cuBLAS Tensor‑Core kernels
    out = torch.bmm(A, B_.transpose(1, 2))  # [B·H, N, N]

    # restore original layout: [B, N, N, H]
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16) → out‑gate gating → final projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim]  fp16

    # -----------------------------------------------------------------
    # 8️⃣ Cast back to float32 for the harness
    # -----------------------------------------------------------------
    return out.to(torch.float32)
