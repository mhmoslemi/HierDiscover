Optimized Triton-based implementation of the outgoing TriMul operation.

* First LayerNorm (float32) then cast to fp16.
* Combine the five linear projections (left_proj, left_gate, right_proj, right_gate, out_gate)
  into a single GEMM to reduce kernel launch overhead.
* Apply sigmoid to the gate tensors.
* Fuse mask multiplication with the element‑wise products `left_proj * left_gate`
  and `right_proj * right_gate` using a small Triton kernel (mask is broadcast
  over the hidden dimension).
* Perform the heavy O(N³) contraction with a batched GEMM (`torch.matmul`);
  it operates on a view `[..., hidden, N, N]` without copying data.
* Second LayerNorm, out‑gate, and final linear projection are carried out in
  fp16 and the result is cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel – fused element‑wise multiplication with optional mask.
# Works on tensors flattened to 1‑D contiguous memory.
# ----------------------------------------------------------------------
@triton.jit
def _fused_gate_mask_kernel(
    a_ptr,          # left_proj  (float16)
    b_ptr,          # left_gate  (float16)
    c_ptr,          # right_proj (float16)
    d_ptr,          # right_gate (float16)
    mask_ptr,       # mask (float16) – length M = B*seq*seq, broadcast over hidden
    out_a_ptr,      # left_out   (float16)
    out_b_ptr,      # right_out  (float16)
    hidden_dim,     # int   – hidden dimension, used for broadcast
    total_size,     # int   – total number of elements (M * hidden_dim)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_size

    # Load the four inputs.
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    c = tl.load(c_ptr + offs, mask=mask)
    d = tl.load(d_ptr + offs, mask=mask)

    # Broadcast the binary mask over the hidden dimension.
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask)

    # Store the fused products.
    tl.store(out_a_ptr + offs, a * b * m, mask=mask)
    tl.store(out_b_ptr + offs, c * d * m, mask=mask)


def _fused_gate_mask(
    left_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_proj: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
    hidden_dim: int,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Apply ``left = left_proj * left_gate * mask`` and ``right = right_proj * right_gate * mask``.
    All tensors must be contiguous, half‑precision and flattened before calling.
    """
    assert left_proj.shape == left_gate.shape == right_proj.shape == right_gate.shape
    total_elems = left_proj.numel()
    block = 1024
    grid = (triton.cdiv(total_elems, block),)

    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    # launch kernel; the flatten view supplies a 1‑D pointer.
    _fused_gate_mask_kernel[grid](
        left_proj, left_gate,
        right_proj, right_gate,
        mask,
        left_out, right_out,
        hidden_dim,
        total_elems,
        BLOCK_SIZE=block,
        num_warps=4,
    )
    return left_out, right_out

    """
    Forward pass of the outgoing TriMul module (no gradient).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool) – may be unused.
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    device = x.device
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # The reference uses float32 LayerNorm; we keep that for exactness.
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # Cast to fp16 – all later ops run in half precision.
    x_fp16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣  Fuse the five linear projections (proj + gates) into one GEMM.
    # ------------------------------------------------------------------
    # Combine weight matrices: shape (5*hidden, dim)
    proj_weights = torch.cat(

            weights["left_gate.weight"],

            weights["out_gate.weight"],

    M = B * N * N  # product of batch and two sequence dimensions

    x_flat = x_fp16.view(M, dim)

    combined = F.linear(x_flat, proj_weights)  # (M, 5*hidden)

    # Split into the five components.
    splits = torch.split(combined, hidden_dim, dim=1)
    left_proj_raw = splits[0].view(B, N, N, hidden_dim)
    left_gate_raw = splits[1].view(B, N, N, hidden_dim)
    right_proj_raw = splits[2].view(B, N, N, hidden_dim)
    right_gate_raw = splits[3].view(B, N, N, hidden_dim)
    out_gate_raw = splits[4].view(B, N, N, hidden_dim)

    # Release the concatenated tensor early to free memory.
    del combined, proj_weights, x_flat, x_fp16, x_norm

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 4️⃣  Mask + element‑wise product for left/right.
    # ------------------------------------------------------------------
    if not nomask:

        mask_fp16 = mask.to(torch.float16).view(M)

        # Flatten the four tensors to 1‑D contiguous memory.
        left_proj_flat = left_proj_raw.contiguous().view(-1)
        left_gate_flat = left_gate.contiguous().view(-1)
        right_proj_flat = right_proj_raw.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)

        # Fuse element‑wise multiplication and mask broadcasting.
        left, right = _fused_gate_mask(
            left_proj_flat, left_gate_flat,
            right_proj_flat, right_gate_flat,
            mask_fp16,
            hidden_dim,
        )
        # Reshape back to [B, N, N, hidden]
        left = left.view(B, N, N, hidden_dim)
        right = right.view(B, N, N, hidden_dim)
    else:
        # Simple element‑wise product – torch is already highly optimized.
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate

    # Free tensors that are no longer needed.
    del left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw

    # ------------------------------------------------------------------
    # 5️⃣  Heavy O(N³) contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Transform to (B, hidden, N, N) view and use batched matmul.
    left_t = left.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2) # (B, H, N, N)

    # (B, H, N, N) @ (B, H, N, N)^T → (B, H, N, N)
    out = torch.matmul(left_t, right_t.transpose(-2, -1))

    # Return the intermediate tensors not needed any more.
    del left, right, left_t, right_t

    # ------------------------------------------------------------------
    # 6️⃣  Second LayerNorm over the hidden dimension.
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out.permute(0, 2, 3, 1),  # bring hidden to last dim → (B, N, N, H)
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )  # still fp16

    # ------------------------------------------------------------------
    # 7️⃣  Apply the output gate and final projection back to `dim`.
    # ------------------------------------------------------------------
    out = out * out_gate  # element‑wise gate (fp16)

    # Final linear projection (hidden → dim).
    to_out_weight = weights["to_out.weight"].to(torch.float16)
    B_, N_, _, _ = out.shape
    out_flat = out.view(B_ * N_ * N_, hidden_dim)
    out_proj = F.linear(out_flat, to_out_weight)  # (B*N*N, dim)

    # Reshape to original 4‑D layout and cast back to float32.
    out_final = out_proj.view(B, N, N, dim).to(torch.float32)

    return out_final

    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    device = x.device
    B, N, _, _ = x.shape
    M = B * N * N  # flattened batch × sequence^2

    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm (float32) → fp16 for the rest
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # float32
    x_fp16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣  Fuse five linear layers into a single GEMM
    # ------------------------------------------------------------------
    # Stack weight matrices: each has shape (hidden, dim)
    proj_weights = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # (5*hidden, dim)

    # Flatten x to (M, dim) and apply the large GEMM.
    x_flat = x_fp16.view(M, dim)                      # (M, dim)
    combined = F.linear(x_flat, proj_weights)         # (M, 5*hidden)

    # Split the concatenated output.
    splits = torch.split(combined, hidden_dim, dim=1)
    left_proj = splits[0].view(B, N, N, hidden_dim)
    left_gate_raw = splits[1].view(B, N, N, hidden_dim)
    right_proj = splits[2].view(B, N, N, hidden_dim)
    right_gate_raw = splits[3].view(B, N, N, hidden_dim)
    out_gate_raw = splits[4].view(B, N, N, hidden_dim)

    # Release unused temporary tensors early.
    del combined, proj_weights, x_flat, x_fp16, x_norm

    # ------------------------------------------------------------------
    # 3️⃣  Gates (sigmoid) – stay in fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 4️⃣  Apply mask (if present) and element‑wise products.
    # ------------------------------------------------------------------
    if not nomask:
        # mask → fp16 and flatten to (M,)
        mask_fp16 = mask.to(torch.float16).view(M)

        # Ensure contiguity before flattening.
        left_proj_f = left_proj.contiguous().view(-1)
        left_gate_f = left_gate.contiguous().view(-1)
        right_proj_f = right_proj.contiguous().view(-1)
        right_gate_f = right_gate.contiguous().view(-1)

        # Fuse multiplication + mask with a Triton kernel.
        left, right = _apply_fused_gate_mask(
            left_proj_f,
            left_gate_f,
            right_proj_f,
            right_gate_f,
            mask_fp16,
            hidden_dim,
        )
        left = left.view(B, N, N, hidden_dim)
        right = right.view(B, N, N, hidden_dim)
    else:
        # Simple element‑wise products (torch‑level kernel).
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Release tensors that are no longer needed.
    del left_proj, left_gate, right_proj, right_gate, left_gate_raw, right_gate_raw

    # ------------------------------------------------------------------
    # 5️⃣  Heavy O(N³) contraction via batched matmul.
    # ------------------------------------------------------------------
    # Bring hidden dimension to the batch axis: (B, H, N, N)
    left_t = left.permute(0, 3, 1, 2)   # (B, hidden, N, N)
    right_t = right.permute(0, 3, 1, 2) # (B, hidden, N, N)

    # out = left @ rightᵀ  → (B, hidden, N, N)
    out = torch.matmul(left_t, right_t.transpose(-2, -1))

    # Clean up.
    del left, right, left_t, right_t

    # ------------------------------------------------------------------
    # 6️⃣  Second LayerNorm over the hidden channel.
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    # Move hidden to the last dimension for LayerNorm.
    out = out.permute(0, 2, 3, 1)  # (B, N, N, hidden)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # ------------------------------------------------------------------
    # 7️⃣  Output gate and final projection back to `dim`.
    # ------------------------------------------------------------------
    out = out * out_gate  # element‑wise gating (fp16)

    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, hidden)

    out_flat = out.view(M, hidden_dim)    # (M, hidden)
    out_proj = F.linear(out_flat, to_out_w)  # (M, dim)

    # Reshape to original 4‑D layout and cast back to float32.
    final_out = out_proj.view(B, N, N, dim).to(torch.float32)

    return final_out
