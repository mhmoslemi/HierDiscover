def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # LayerNorm(x)
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias).to(torch.float16)

    B,N,_,_ = x.shape
    M = B * N * N

    # Cast weight matrices to fp16
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    # combine weights into single big weight
    w_cat = torch.cat([left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w], dim=0).to(torch.float16)

    # flatten x
    x_flat = x.view(-1, dim) # (M, dim)
    # Multi-linear pass
    multi = torch.nn.functional.linear(x_flat, w_cat) # shape (M,5*hidden)
    # split
    # Indices offsets
    o0 = hidden_dim
    o1 = hidden_dim*2
    o2 = hidden_dim*3
    o3 = hidden_dim*4
    o4 = hidden_dim*5 # overall size

    left_proj = multi[:, 0:o0]
    left_gate_pre = multi[:, o0:o1]
    right_proj = multi[:, o1:o2]
    right_gate_pre = multi[:, o2:o3]
    out_gate_pre = multi[:, o3:o4]

    # Apply sigmoid to gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

"""
custom_kernel – fast forward pass of the outgoing TriMul operator

Algorithm
---------
1️⃣  Layer‑Norm the input (FP16) – this is the only LN that cannot be fused.
2️⃣  Fuse **all** linear projections that share the same input (left_proj,
    left_gate, right_proj, right_gate, out_gate) into ONE half‑precision
    GEMM (`torch.nn.functional.linear`).  The result is sliced into the
    five tensors and the three gate tensors are passed through a sigmoid.
    This reduces the number of times we read the huge ``[B,N,N,dim]`` tensor
    from memory (only once instead of five) and eliminates four kernel launches.
3️⃣  Optional mask‑gate‑fusion: if a mask is supplied we launch a tiny Triton
    kernel that multiplies the projected tensor, its gate and the mask in a
    single pass (`proj * gate * mask`).  When ``nomask=True`` we simply
    multiply in PyTorch.
4️⃣  O(N³) contraction – the dominant work – is performed with a **batched
    GEMM** (`torch.bmm`).  By reshaping the tensors we turn the Einstein sum

        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    into a batch of standard matrix‑multiplications
    `C = A @ Bᵀ` with batch size `B*hidden_dim`.  This call uses cuBLAS/Tensor‑Core
    and is the main speed driver.
5️⃣  Second Layer‑Norm (FP16), element‑wise multiplication with the output‑gate
    and the final linear projection back to ``dim`` channels.
6️⃣  Cast the result back to ``float32`` (the expected user dtype).

The implementation keeps memory traffic low (only one big projection pass and
a single batched GEMM) while still using Triton for the mask‑gate fusion as
required.
"""

import math
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fused element‑wise multiplication of proj, gate and mask.
#   out = proj * gate * mask[...,None]   (proj,gate : (M,H), mask : (M,))
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,        # tensors
    M, H,                                         # problem sizes
    stride_proj_m, stride_proj_h,                 # proj strides (row, col)
    stride_gate_m, stride_gate_h,                 # gate strides
    stride_out_m,  stride_out_h,                  # out strides
    stride_mask,                                  # mask stride (1‑D)
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr, # tile sizes
):
    pid = tl.program_id(0)                     # block index over rows (M)
    block_start = pid * BLOCK_M
    offs_m = block_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    offs_h = tl.arange(0, BLOCK_H)                       # [BLOCK_H]

    # Load mask (scalar per row) and broadcast to H
    mask = tl.load(
        mask_ptr + offs_m * stride_mask,
        mask=offs_m < M,
        other=0.0,
    ).to(tl.float16)                # [BLOCK_M]
    mask_h = mask[:, None]                     # [BLOCK_M, 1]

    # Load projection tile
    proj = tl.load(
        proj_ptr + offs_m[:, None] * stride_proj_m
               + offs_h[None, :] * stride_proj_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    # Load gate tile (already sigmoid‑ed)
    gate = tl.load(
        gate_ptr + offs_m[:, None] * stride_gate_m
               + offs_h[None, :] * stride_gate_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    # Fuse: proj * gate * mask
    out = proj * gate * mask_h

    # Store result
    tl.store(
        out_ptr + offs_m[:, None] * stride_out_m
               + offs_h[None, :] * stride_out_h,
        out,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
    )


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradient).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float32)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device as
        ``input_tensor``.
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack & config
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 -> FP16)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias   = weights["norm.bias"].to(torch.float16)

    # Convert input to fp16 *before* LN to keep everything in half‑precision
    x = torch.nn.functional.layer_norm(
        input_tensor.to(torch.float16),
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, dim]  (half)

    B, N, _, _ = x.shape
    M = B * N * N                     # flattened pair count (B*N*N)

    # --------------------------------------------------------------
    # 2️⃣ Fuse all per‑position linear maps into one GEMM
    #    (left_proj, left_gate, right_proj, right_gate, out_gate)
    # --------------------------------------------------------------
    # gather and cast the five weight matrices (shape: hidden x dim)
    w_left_proj   = weights["left_proj.weight"]
    w_left_gate   = weights["left_gate.weight"]
    w_right_proj  = weights["right_proj.weight"]
    w_right_gate  = weights["right_gate.weight"]
    w_out_gate    = weights["out_gate.weight"]

    # Concatenate along the output dimension (5*hidden , dim)
    w_cat = torch.cat(
        [
            w_left_proj,
            w_left_gate,
            w_right_proj,
            w_right_gate,
            w_out_gate,
        ],
        dim=0,
    ).to(torch.float16)   # (5*hidden, dim)

    # Flatten the input to M x dim
    x_flat = x.view(-1, dim)                     # (M, dim) half

    # One huge linear projection – reads ``x`` only once
    all_proj = torch.nn.functional.linear(x_flat, w_cat)   # (M, 5*hidden)

    # --------------------------------------------------------------
    # 3️⃣ Slice the fused tensor and apply sigmoid to the three gates
    # --------------------------------------------------------------
    h = hidden_dim
    left_proj   = all_proj[:, 0 * h : 1 * h]       # (M, h)
    left_gate   = torch.sigmoid(all_proj[:, 1 * h : 2 * h])
    right_proj  = all_proj[:, 2 * h : 3 * h]
    right_gate  = torch.sigmoid(all_proj[:, 3 * h : 4 * h])
    out_gate    = torch.sigmoid(all_proj[:, 4 * h : 5 * h])   # (M, h)
    out_gate    = out_gate.view(B, N, N, hidden_dim)          # (B,N,N,h)

    # --------------------------------------------------------------
    # 4️⃣ Optional mask‑gate‑fusion (Triton) – produce ``left`` and ``right``
    # --------------------------------------------------------------
    if not nomask:
        # mask is [B,N,N] -> flatten to (M,)
        mask_fp16 = mask_tensor.to(torch.float16).view(-1)   # (M,)

        left = torch.empty_like(left_proj)      # allocate output buffer
        right = torch.empty_like(right_proj)

        # Strides – note that .stride() returns element‑wise strides
        stride_lp_m, stride_lp_h = left_proj.stride()
        stride_lg_m, stride_lg_h = left_gate.stride()
        stride_rp_m, stride_rp_h = right_proj.stride()
        stride_rg_m, stride_rg_h = right_gate.stride()
        stride_lo_m, stride_lo_h = left.stride()
        stride_ro_m, stride_ro_h = right.stride()
        stride_mask = mask_fp16.stride()[0]

        BLOCK_M = 64
        BLOCK_H = hidden_dim
        grid = ((M + BLOCK_M - 1) // BLOCK_M,)

        # left = left_proj * left_gate * mask
        _mask_gate_fuse_kernel[grid](
            left_proj, left_gate, mask_fp16, left,
            M, hidden_dim,
            stride_lp_m, stride_lp_h,
            stride_lg_m, stride_lg_h,
            stride_lo_m, stride_lo_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
        # right = right_proj * right_gate * mask
        _mask_gate_fuse_kernel[grid](
            right_proj, right_gate, mask_fp16, right,
            M, hidden_dim,
            stride_rp_m, stride_rp_h,
            stride_rg_m, stride_rg_h,
            stride_ro_m, stride_ro_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
        # reshape to [B,N,N,hidden]
        left = left.view(B, N, N, hidden_dim)
        right = right.view(B, N, N, hidden_dim)
    else:
        # No mask – simple elementwise fuse in PyTorch (fast)
        left = (left_proj * left_gate).view(B, N, N, hidden_dim)
        right = (right_proj * right_gate).view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5️⃣ O(N³) contraction via a batched GEMM (cuBLAS / Tensor‑Core)
    # --------------------------------------------------------------
    # reshape to (B*hidden, N, N) for batched matmul
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Compute C = A @ Bᵀ  (batch size = B*hidden)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original ordering: (B,N,N,hidden)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 7️⃣ Output gate & final projection back to ``dim``
    # --------------------------------------------------------------
    out = out * out_gate                      # element‑wise gate (half)
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden)

    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim) half

    # --------------------------------------------------------------
    # 8️⃣ Cast back to float32 for the caller
    # --------------------------------------------------------------
    return out.to(torch.float32)
