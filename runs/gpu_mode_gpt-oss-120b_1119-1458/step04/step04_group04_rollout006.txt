def batched_gemm(left, right):
    B, N, _, H = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous() # [B,H,N,N]
    A = left_perm.view(B * H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B_ = right_perm.view(B * H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2)) # shape [B*H,N,N]
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out
```

```
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16
    gate_ptr,       # *fp16
    mask_ptr,       # *fp16 (flattened mask)
    size,           # i32 total elements in x
    hidden_dim,     # i32 number of channels per mask entry
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    # load operands
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    # compute mask offset by dividing by hidden_dim (broadcast)
    m_offs = offs // hidden_dim
    m = tl.load(mask_ptr + m_offs, mask=mask, other=1.0)

    # fused multiply
    out = x * g * m

    tl.store(x_ptr + offs, out, mask=mask)
```

def _batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    # left,right shape [B,N,N,H]; fp16
    B, N, _, H = left.shape
    # reshape to [B*H, N, N]
    a = left.permute(0, 3, 1, 2).reshape(B * H, N, N)  # contiguous view
    b = right.permute(0, 3, 1, 2).reshape(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

"""
TriMul (outgoing) – optimized forward pass.

Algorithm
---------
1. Layer‑norm the input tensor (FP32) → cast to FP16.
2. Linear projections (left/right) and three gating streams are computed in FP16.
3. If a mask is present we fuse the three operations
   `proj * gate * mask` with a tiny Triton kernel (`mask_gate_fuse_kernel`).
   The mask is broadcast over the hidden dimension inside the kernel, so
   only a single launch per tensor is needed.
4. The heavy contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM: for each head `h` we compute
   `C = A @ Bᵀ` with `torch.bmm`, which dispatches to cuBLAS Tensor‑Core
   kernels (the fastest path on an H100).  No custom Triton matmul is used
   – the extra launch overhead of a Triton kernel would dominate for N≤1024.
5. Second Layer‑norm (FP16) → out‑gate (sigmoid) → final linear projection
   back to the original channel dimension.
6. Cast the result back to FP32 (the harness expects FP32).

Only the forward pass is required; gradients are not computed.
At least one custom Triton kernel is used (the mask‑gate fusion).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (tensor * gate * mask) – mask is broadcast over
# the hidden dimension `hidden_dim`.  The kernel works on a flat view of
# the 4‑D tensor `[B, N, N, hidden_dim]`.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,            # *fp16 : tensor to be updated, shape (size,)
    gate_ptr,         # *fp16 : gate tensor, same shape as x
    mask_ptr,         # *fp16 : mask tensor of shape (B*N*N,) – broadcasted
    size,             # i32   : total number of elements in x (B*N*N*hidden_dim)
    hidden_dim,       # i32   : width of the hidden dimension
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)                     # block index
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # linear offsets handled by this block
    mask = offs < size                         # boundary mask

    # Load data
    x   = tl.load(x_ptr   + offs, mask=mask, other=0.0)
    g   = tl.load(gate_ptr+ offs, mask=mask, other=0.0)

    # Broadcast mask: every hidden_dim consecutive elements share the same mask entry
    mask_offs = offs // hidden_dim
    m   = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)

    # Fused multiplication
    out = x * g * m

    # Store result back in‑place
    tl.store(x_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Batched GEMM helper (uses cuBLAS via torch.bmm)
# ----------------------------------------------------------------------
def _batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the O(N³) contraction out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a batched GEMM.  Both `left` and `right` are FP16 tensors of shape
    [B, N, N, H].  The result has the same shape.
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N]  –  A @ Bᵀ  for each head
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # GEMM (FP16 inputs → FP32 accumulation inside cuBLAS)
    c = torch.bmm(a, b.transpose(1, 2))
    # reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point called by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor      [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor      [B, N, N]    (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain `dim` and `hidden_dim`;
                         optional key `nomask` (True → no mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim        = config["dim"]          # original channel dimension (c_z)
    hidden_dim = config["hidden_dim"]   # projection width (c)
    nomask     = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → FP16
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                       # [B, N, N, dim]  float32
    x = x_norm.half()      # cast once to FP16

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16 (once)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w      = weights["to_out.weight"].half()

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4️⃣ Gating signals (sigmoid + FP16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask – fused element‑wise multiplication via Triton
    # ------------------------------------------------------------------
    if nomask:
        # In‑place multiplication (no extra memory allocation)
        left  = left.mul_(left_gate)
        right = right.mul_(right_gate)
    else:
        # mask shape: [B, N, N] → broadcast over hidden_dim inside the kernel
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)  # flatten to (B*N*N,)
        total_elements = left.numel()  # B * N * N * hidden_dim
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # left = left * left_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            left,
            left_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask   (in‑place)
        mask_gate_fuse_kernel[grid](
            right,
            right_gate,
            mask_fp16,
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # gating tensors are no longer needed
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy contraction (batched GEMM) – cuBLAS implementation
    # ------------------------------------------------------------------
    out = _batched_gemm_fp16(left, right)   # [B, N, N, hidden_dim]  FP16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Out‑gate multiplication + final projection
    # ------------------------------------------------------------------
    out = out * out_gate                      # element‑wise gating (FP16)
    out = torch.nn.functional.linear(out, to_out_w)   # back to `dim` (FP16)

    # ------------------------------------------------------------------
    # 9️⃣ Cast result back to float32 for the harness
    # ------------------------------------------------------------------
    return out.float()
