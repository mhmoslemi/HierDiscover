@triton.jit
def mask_gate_fuse_kernel(
    x_ptr, gate_ptr, mask_ptr,
    size, hidden,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    base = pid * BLOCK
    offs = base + tl.arange(0, BLOCK)
    mask = offs < size

    # load
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    # compute mask offset
    mask_offs = offs // hidden
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)
```

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    # config dims
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', True)
    # other config like if mask is None?

    # LayerNorm
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Cast to fp16
    x = x_norm.to(torch.float16)

    # Cast weights to fp16
    left_proj_w = weights['left_proj.weight'].to(torch.float16)
    left_gate_w = weights['left_gate.weight'].to(torch.float16)
    right_proj_w = weights['right_proj.weight'].to(torch.float16)
    right_gate_w = weights['right_gate.weight'].to(torch.float16)
    out_gate_w = weights['out_gate.weight'].to(torch.float16)

    # Concatenate left weights: [proj;gate]
    left_concat_w = torch.cat([left_proj_w, left_gate_w], dim=0) # shape (2*hidden, dim)
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    # Compute fused left and right
    left_concat = torch.nn.functional.linear(x, left_concat_w)  # shape [B,N,N,2*hidden]
    right_concat = torch.nn.functional.linear(x, right_concat_w)

    # Split
    left_proj = left_concat[..., :hidden_dim]  # shape [B,N,N,hidden]
    left_gate_raw = left_concat[..., hidden_dim:]  # shape [B,N,N,hidden]
    left_gate = torch.sigmoid(left_gate_raw)

    right_proj = right_concat[..., :hidden_dim]
    right_gate_raw = right_concat[..., hidden_dim:]
    right_gate = torch.sigmoid(right_gate_raw)

    # Compute out gate separately
    out_gate_raw = torch.nn.functional.linear(x, out_gate_w)  # shape [B,N,N,hidden]
    out_gate = torch.sigmoid(out_gate_raw)

    # Mask handling
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16)
        size = left_proj.numel()
        BLOCK = 1024
        grid = ((size + BLOCK - 1) // BLOCK,)
        # left = left_proj * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            size,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right similarly
        mask_gate_fuse_kernel[grid](
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            size,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # O(N^3) contraction
    # shape left_proj: [B,N,N,hidden]
    # Use torch.bmm: reshape to [B*hidden, N, N]
    B, N, _, H = left_proj.shape
    # Permute to [B, H, N, N] then view
    left_perm = left_proj.permute(0, 3, 1, 2).contiguous()
    right_perm = right_proj.permute(0, 3, 1, 2).contiguous()
    a = left_perm.view(B * H, N, N)
    b = right_perm.view(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Second layer norm
    to_out_norm_w = weights['to_out_norm.weight'].to(torch.float16)
    to_out_norm_b = weights['to_out_norm.bias'].to(torch.float16)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection
    to_out_w = weights['to_out.weight'].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)  # shape [B,N,N,dim]
    # cast back to float32
    return out.to(torch.float32)

"""
Efficient outgoing TriMul implementation.

Algorithm:
1. LayerNorm over dim (float32) → convert to fp16.
2. Fuse left/right linear projections and gating linear weights into concatenated
   weight matrices to reduce kernel launches.
3. Compute left/right projections and gating signals in fp16; apply sigmoid.
4. Optional mask: use a Triton kernel that multiplies (proj * gate * mask) in‑place,
   broadcasting mask over the hidden dimension.
5. O(N³) contraction performed as a batched GEMM: for each hidden channel we
   compute a matrix product A @ Bᵀ with torch.bmm (cuBLAS Tensor Cores).
6. Apply a second LayerNorm (fp16), out‑gate (sigmoid), and final linear projection.
7. Cast result back to float32.

All heavy ops stay in half‑precision; only the first LayerNorm is float32.
The Triton kernel is required for the mask‑gate fusion.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,           # *fp16: tensor to be updated, flatten view of [B, N, N, H]
    gate_ptr,        # *fp16: gate tensor, same shape as x
    mask_ptr,        # *fp16: mask tensor of shape [B, N, N] (flattened)
    size,            # i32: total number of elements in the flattened tensor
    hidden,          # i32: hidden dimension (broadcast factor)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    # Load operands
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # broadcast mask: the mask index is the element index divided by hidden
    idx_mask = offs // hidden
    m = tl.load(mask_ptr + idx_mask, mask=mask, other=1.0)

    # fused multiplication
    out = x * g * m

    tl.store(x_ptr + offs, out, mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # First LayerNorm (float32) for numerical stability
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] fp32

    # Cast to half‑precision for the heavy part
    x = x_norm.to(torch.float16)  # [B, N, N, dim] fp16

    # ------------------------------
    # Cast model weights to fp16 once
    # ------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (hidden, dim)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)        # (dim, hidden)

    # -----------------------------------------------------------------
    # 1) Fused linear projections for left/right (proj + gate)
    # -----------------------------------------------------------------
    left_concat_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*hidden, dim)
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    # Produce concatenated tensors of shape [B, N, N, 2*hidden]
    left_concat = torch.nn.functional.linear(x, left_concat_w)   # fp16
    right_concat = torch.nn.functional.linear(x, right_concat_w)

    # Split into projection and gate parts
    left_proj = left_concat[..., :hidden_dim]            # [B,N,N,hidden]
    left_gate_raw = left_concat[..., hidden_dim:]        # [B,N,N,hidden]
    right_proj = right_concat[..., :hidden_dim]
    right_gate_raw = right_concat[..., hidden_dim:]

    # Apply sigmoid to gating tensors
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # out_gate is computed from the original normalized input x
    out_gate_raw = torch.nn.functional.linear(x, out_gate_w)   # [B,N,N,hidden]
    out_gate = torch.sigmoid(out_gate_raw)

    # --------------------------------------------------------------
    # 2) Optional mask – fuse (proj * gate * mask) using Triton
    # --------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16)  # [B,N,N] fp16
        # Ensure mask is contiguous
        mask_fp16 = mask_fp16.contiguous()
        # Total number of elements for left/right tensors
        total_elems = left_proj.numel()
        BLOCK = 1024
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        mask_gate_fuse_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elems,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        mask_gate_fuse_kernel[grid](
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elems,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – simple element‑wise multiplication
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # --------------------------------------------------------------
    # 3) O(N³) contraction: batched GEMM (left @ rightᵀ) in half‑precision
    # --------------------------------------------------------------
    # Shapes: left_proj, right_proj => [B, N, N, hidden]
    B, N, _, H = left_proj.shape
    # Rearrange to [B*H, N, N] and perform bmm
    left_perm = left_proj.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_proj.permute(0, 3, 1, 2).contiguous()
    a = left_perm.view(B * H, N, N)
    b = right_perm.view(B * H, N, N)

    # bmm computes a @ bᵀ   -> shape [B*H, N, N]
    c = torch.bmm(a, b.transpose(1, 2))

    # Reshape back to the original layout [B, N, N, hidden]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 4) Second LayerNorm (fp16)
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 5) Final gating and projection
    # --------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # Cast back to float32 for the external harness
    return out.to(torch.float32)

"""
Outgoing TriMul (AlphaFold‑3) – fast half‑precision implementation.

Algorithm
---------
1.  First LayerNorm (float32) → cast to fp16.
2.  Fuse left/right linear projections and their gate‑projections into two

3.  Apply sigmoid to the gate tensors.
4.  Optional mask: a tiny Triton kernel multiplies (proj * gate * mask) in‑place,
    broadcasting the mask over the hidden dimension (no extra copies).
5.  Heavy O(N³) contraction is a batched GEMM:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    Implemented as torch.bmm on tensors reshaped to [B·H, N, N] (Tensor‑Core

6.  Second LayerNorm (fp16) → elementwise out‑gate (sigmoid) → final linear
    projection back to `dim`.
7.  Cast the result to float32 for the harness.

At least one operation (mask‑gate fusion) is performed with a custom Triton
kernel, while the compute‑bound contraction uses cuBLAS via torch.bmm for
maximal throughput on H100.

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

def mask_gate_fuse_kernel(
    x_ptr,        # *fp16 : tensor to be updated (flattened [B,N,N,H])
    gate_ptr,     # *fp16 : gate tensor (same shape as x)

    size,         # i32   : total number of elements in x/gate
    hidden,       # i32   : hidden dimension (broadcast factor)
    BLOCK: tl.constexpr,  # compile‑time tile size (≤1024)
):

        - mask_tensor  : torch.Tensor, shape [B, N, N] (any dtype, may be bool)
        - weights_dict : dict mapping weight names to tensors (float32)
        - config_dict  : must contain "dim" and "hidden_dim";
                         optional "nomask" (default True)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device as input.
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm in float32 for stability
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for all subsequent heavy work
    # -----------------------------------------------------------------
    x = x_norm.to(torch.float16)  # [B, N, N, dim] fp16

    # -----------------------------------------------------------------
    # 3️⃣ Cast model weights to fp16 once
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (hidden, dim)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (hidden, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)      # (dim, hidden)

    # -----------------------------------------------------------------
    # 4️⃣ Fused linear projections (proj + gate) for left and right
    # -----------------------------------------------------------------
    left_concat_w  = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*hidden, dim)
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    # Concatenated outputs: [B, N, N, 2*hidden]
    left_concat  = torch.nn.functional.linear(x, left_concat_w)    # fp16
    right_concat = torch.nn.functional.linear(x, right_concat_w)

    # Split projection and raw gate parts
    left_proj = left_concat[..., :hidden_dim]          # [B, N, N, hidden]
    left_gate_raw = left_concat[..., hidden_dim:]      # [B, N, N, hidden]

    right_proj = right_concat[..., :hidden_dim]
    right_gate_raw = right_concat[..., hidden_dim:]

    # Apply sigmoid to raw gate tensors
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # out_gate (computed from the original normalized input)
    out_gate_raw = torch.nn.functional.linear(x, out_gate_w)  # [B, N, N, hidden]
    out_gate = torch.sigmoid(out_gate_raw)

    # -----------------------------------------------------------------
    # 5️⃣ Optional mask – fuse (proj * gate * mask) via Triton
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()      # [B, N, N] fp16
        total_elements = left_proj.numel()                         # B·N·N·hidden
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elements,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # Release now‑unused tensors early (optional)
        del left_gate_raw, right_gate_raw, mask_fp16
    else:
        # No mask – simple element‑wise multiplication
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate
        del left_gate_raw, right_gate_raw

    # -----------------------------------------------------------------
    # 6️⃣ Batched GEMM (O(N³) contraction) – torch.bmm (cuBLAS Tensor‑Core)
    # -----------------------------------------------------------------
    B, N, _, H = left_proj.shape  # H == hidden_dim
    # Rearrange to [B·H, N, N] (contiguous layout required for cuBLAS)
    left_perm = left_proj.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_proj.permute(0, 3, 1, 2).contiguous()
    a = left_perm.view(B * H, N, N)    # [B*H, N, N]
    b = right_perm.view(B * H, N, N)

    # Compute A @ Bᵀ for each hidden slice
    c = torch.bmm(a, b.transpose(1, 2))  # [B*H, N, N]

    # Restore original layout [B, N, N, hidden]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Out‑gate multiplication and final projection
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] fp16

    # -----------------------------------------------------------------
    # 9️⃣ Cast back to float32 for the external harness
    # -----------------------------------------------------------------
    return out.to(torch.float32)
