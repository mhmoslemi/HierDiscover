"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

Algorithm
---------
1️⃣  Layer‑norm the input in fp32 → cast to fp16.
2️⃣  Fuse the five linear projections (left, right, left‑gate, right‑gate,
    out‑gate) into a single `torch.nn.functional.linear` call.
3️⃣  Slice the fused output, apply the sigmoid to the three gate‑segments
    and (optionally) the pair‑wise mask.
4️⃣  Fuse left/right with their gates (element‑wise) and contract:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    The contraction is a batched matrix‑multiply `C = A @ Bᵀ`.  For
    sequence lengths ≤256 we use a hand‑written Triton GEMM kernel;
    otherwise we fall back to cuBLAS via `torch.bmm`.
5️⃣  A second LayerNorm over the hidden dimension, an element‑wise
    out‑gate, and the final linear projection back to `dim`.
6️⃣  Cast the result to float32 (required by the harness).

All heavy linear work is done in fp16 (Tensor‑cores) with fp32
accumulation where needed.  The implementation reduces memory traffic
by fusing the five linear layers into one GEMM and therefore achieves
significant speed‑ups over the naïve PyTorch reference.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (fp16 → fp16, accumulator fp32)
#   A, B : [M, N, K] (M = B*hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel for the TriMul contraction."""
    pid_z = tl.program_id(0)   # batch*hidden index
    pid_m = tl.program_id(1)   # i‑tile
    pid_n = tl.program_id(2)   # j‑tile

    # ------------------- tile offsets ---------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # --------- load A tile (i, k) --------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )          # fp16

        # --------- load B tile (j, k) – transposed view ---------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk   # k axis first
            + offs_j[None, :] * stride_bj   # j axis second
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )          # fp16 (shape K×BLOCK_N)

        # mixed‑precision dot: fp16 × fp16 → fp32 accumulate
        acc += tl.dot(a, b)

    # ------------------- write result ---------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right (both [B, N, N, H]) using the Triton kernel.
    Returns a tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    # reshape to [M, N, N] where M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)          # fp16 output

    # strides in element counts (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # choose tile size – 64×64×64 works well on H100 for fp16 MMA
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                               # pid_z
        (N + BLOCK_M - 1) // BLOCK_M,        # pid_m
        (N + BLOCK_N - 1) // BLOCK_N,        # pid_n
    )

    _trimul_gemm_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right using torch.bmm (cuBLAS, fp16×fp16→fp32).
    """
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 2️⃣ first LayerNorm (float32) – stability
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 3️⃣ cast to fp16 for the rest of the pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ fuse five linear projections into one GEMM
    # ------------------------------------------------------------------
    # collect and cast weights to fp16
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    fused_proj_w = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(torch.float16)                     # shape [5*hidden_dim, dim]

    # one fused linear
    proj = F.linear(x, fused_proj_w)        # [B, N, N, 5*hidden_dim], fp16

    # ------------------------------------------------------------------
    # 5️⃣ slice the fused result
    # ------------------------------------------------------------------
    H = hidden_dim
    left      = proj[..., 0:H]                                   # [B, N, N, H]
    right     = proj[..., H:2 * H]                               # [B, N, N, H]
    left_gate = proj[..., 2 * H:3 * H]                           # [B, N, N, H]
    right_gate= proj[..., 3 * H:4 * H]                           # [B, N, N, H]
    out_gate  = proj[..., 4 * H:5 * H]                           # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 7️⃣ gating (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # fuse gates in‑place
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 8️⃣ heavy contraction: batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    seq_len = input_tensor.shape[1]
    # Use Triton for very small sequences; otherwise cuBLAS via torch.bmm
    if seq_len <= 256:
        out = _batched_matmul_triton(left, right)          # fp16
    else:
        out = _batched_matmul_torch(left, right)           # fp16

    # ------------------------------------------------------------------
    # 9️⃣ second LayerNorm (fp16) → out‑gate → final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)            # [B, N, N, dim], fp16

    # ------------------------------------------------------------------
    # 🔟 cast back to float32
    # ------------------------------------------------------------------
    return out.to(torch.float32)
