"""
TriMul (outgoing) – Triton‑accelerated implementation

This forward module follows the reference PyTorch code but:
  1️⃣  First LayerNorm is done in FP32 (torch) and the result is cast to FP16.
  2️⃣  Linear projections + gating are fused: the projection weight and its
      gate weight are concatenated and a single torch.nn.functional.linear is
      performed.  The mask (if present) is applied only to the projection part.
  3️⃣  The heavy O(N³) contraction
         out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
      is realised as a batched GEMM (A @ Bᵀ) with a custom Triton kernel.
      The batch dimension is the product (batch * hidden_dim); the kernel works
      with FP16 inputs and accumulates in FP32 for numerical stability.
  4️⃣  The second LayerNorm, final gating, and output projection are performed
      in FP16 (torch ops).  The final result is cast back to FP32.
The implementation is fully functional on an H100 (Triton 3.3.1) and satisfies
the “must use Triton” requirement while reducing the number of separate
linear calls and avoiding extra memory copies.
"""

import triton
import triton.language as tl
import torch
import torch.nn.functional as F
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton batched GEMM kernel (A @ Bᵀ) for the TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr,            # left tensor  (batch_hidden, M, K)   – FP16
    b_ptr,            # right tensor (batch_hidden, N, K)  – FP16 (will be accessed transposed)
    c_ptr,            # output       (batch_hidden, M, N) – FP16
    stride_ah, stride_ak, stride_ab,          # strides for A
    stride_bk, stride_bn, stride_bb,          # strides for Bᵀ (row = k, col = n)
    stride_ch, stride_cn, stride_cb,          # strides for C
    M, N, K,                                   # matrix dimensions
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ where
        A : (batch_h, M, K)
        B : (batch_h, N, K)   (the kernel reads it as (K, N) to achieve the transpose)
        C : (batch_h, M, N)
    All tensors are stored in row‑major (C) order.
    """
    pid_batch = tl.program_id(2)            # batch_hidden (batch * hidden_dim)
    pid_m = tl.program_id(0)                # tile along M (i)
    pid_n = tl.program_id(1)                # tile along N (j)

    # -------------------- tile offsets --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)                    # [0, BLOCK_K)

    # -------------------- predicates for out‑of‑bounds --------------------
    mask_m = offs_m < M
    mask_n = offs_n < N

    # -------- pointers for the current tile (will be advanced in the k‑loop) --------
    a_tile_ptr = a_ptr + pid_batch * stride_ab + offs_m[:, None] * stride_ah + offs_k[None, :] * stride_ak
    b_tile_ptr = b_ptr + pid_batch * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
    c_tile_ptr = c_ptr + pid_batch * stride_cb + offs_m[:, None] * stride_ch + offs_n[None, :] * stride_cn

    # accumulator in FP32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- main loop over K --------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # load tiles, respecting OOB with a mask (the other argument is ignored for OOB)
        a = tl.load(a_tile_ptr, mask=(mask_m[:, None] & (offs_k < K)), other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_tile_ptr, mask=(mask_n[None, :] & (offs_k < K)), other=0.0)   # (BLOCK_K, BLOCK_N)
        # accumulate product
        acc += tl.dot(a, b)    # (BLOCK_M, BLOCK_N)  (FP32)

        # advance to next K‑tile
        a_tile_ptr += BLOCK_K * stride_ak
        b_tile_ptr += BLOCK_K * stride_bk

    # write result (cast back to FP16)
    c = acc.to(tl.float16)
    tl.store(c_tile_ptr, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor,
                                   right: torch.Tensor,
                                   block_m: int = 128,
                                   block_n: int = 128,
                                   block_k: int = 32) -> torch.Tensor:
    """
    left, right : [B, N, N, H]  (FP16)
    Returns out : [B, N, N, H] (FP16) where
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    """
    B, N, _, H = left.shape
    batch_hidden = B * H

    # bring hidden dimension into the batch dimension
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(batch_hidden, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(batch_hidden, N, N)
    out_flat = torch.empty_like(left_flat)

    # strides for the three tensors (row‑major storage)
    a_batch_stride = left_flat.stride(0)
    a_row_stride = left_flat.stride(1)      # M‑stride
    a_col_stride = left_flat.stride(2)      # K‑stride

    # for Bᵀ we need row‑stride = original col‑stride (1), col‑stride = original row‑stride (N)
    b_batch_stride = right_flat.stride(0)
    b_row_stride = right_flat.stride(2)     # stride for k (original col)
    b_col_stride = right_flat.stride(1)     # stride for n (original row)

    c_batch_stride = out_flat.stride(0)
    c_row_stride = out_flat.stride(1)
    c_col_stride = out_flat.stride(2)

    grid = ( (N + block_m - 1) // block_m,
             (N + block_n - 1) // block_n,
             batch_hidden)

    _batched_matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        a_row_stride, a_col_stride, a_batch_stride,
        b_row_stride, b_col_stride, b_batch_stride,
        c_row_stride, c_col_stride, c_batch_stride,
        N, N, N,
        BLOCK_M=block_m, BLOCK_N=block_n, BLOCK_K=block_k,
        num_warps=8,
    )
    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point – forward pass of the outgoing TriMul module
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass (no gradients) of the TriMul outgoing module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype torch.float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float or bool). May be ignored.
        - weights_dict : mapping from layer name to weight Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional key ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → cast to FP16
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] – float32
    x = x_norm.to(torch.float16)   # subsequent ops in FP16

    # ------------------------------------------------------------------
    # 2️⃣ Prepare fused weights (concatenate proj & gate) and cast to FP16
    # ------------------------------------------------------------------
    # left: projection + left gate
    left_proj_w = weights["left_proj.weight"]          # (hid, dim)
    left_gate_w = weights["left_gate.weight"]          # (hid, dim)
    left_fused_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)  # (2*hid, dim)

    # right: projection + right gate
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_fused_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)  # (2*hid, dim)

    # out_gate weight
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # final projection & second LayerNorm weights
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear + gating (fused) for left and right side
    # ------------------------------------------------------------------
    # fused linear gives concat([proj, gate])
    left_fused = F.linear(x, left_fused_w)          # [B,N,N,2*hid] – FP16
    right_fused = F.linear(x, right_fused_w)       # same shape

    # split projection and gate parts
    left_proj = left_fused[..., :hidden_dim]        # [B,N,N,hid]
    left_gate_pre = left_fused[..., hidden_dim:]    # [B,N,N,hid]

    right_proj = right_fused[..., :hidden_dim]
    right_gate_pre = right_fused[..., hidden_dim:]

    # apply mask only to the projection part (if present)
    if not nomask:
        # mask shape → [B, N, N, 1] broadcast over hidden dimension
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # gating (sigmoid) – NOT masked
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # combine projection with its gate
    left = left_proj * left_gate          # [B,N,N,hid]
    right = right_proj * right_gate       # [B,N,N,hid]

    # ------------------------------------------------------------------
    # 4️⃣ Out‑gate (computed from the *original* normalized tensor)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # [B,N,N,hid] – FP16

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction – custom Triton batched GEMM
    # ------------------------------------------------------------------
    # out = Σ_k left[...,k,:] * right[...,k,:]  (batch‑wise over hidden)
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hid] – FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (over hidden dimension)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply out‑gate and final projection back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate                                   # element‑wise gating
    out = F.linear(out, to_out_w)                          # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to float32 for the final output
    # ------------------------------------------------------------------
    return out.to(torch.float32)
