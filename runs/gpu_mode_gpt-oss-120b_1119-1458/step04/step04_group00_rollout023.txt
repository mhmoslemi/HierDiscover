def fused_proj_gate_mask_kernel(
    proj_left_ptr, gate_left_ptr,
    proj_right_ptr, gate_right_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    M, H,
    stride_pl_m, stride_pl_h,
    stride_gl_m, stride_gl_h,
    stride_pr_m, stride_pr_h,
    stride_gr_m, stride_gr_h,
    stride_outl_m, stride_outl_h,
    stride_outr_m, stride_outr_h,
    stride_mask,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_M
    offs_m = block_start + tl.arange(0, BLOCK_M)
    offs_h = tl.arange(0, BLOCK_H)

    # mask load
    mask_val = tl.load(mask_ptr + offs_m * stride_mask,
                       mask=offs_m < M, other=0.0)
    mask_val = mask_val[:, None]  # broadcast to hidden

    # load proj/gate
    proj_l = tl.load(proj_left_ptr + offs_m[:, None] * stride_pl_m + offs_h[None, :] * stride_pl_h,
                     mask=(offs_m[:, None] < M) & (offs_h[None, :] < H), other=0.0)
    gate_l = tl.load(gate_left_ptr + offs_m[:, None] * stride_gl_m + offs_h[None, :] * stride_gl_h,
                     mask=(offs_m[:, None] < M) & (offs_h[None, :] < H), other=0.0)

    proj_r = tl.load(proj_right_ptr + offs_m[:, None] * stride_pr_m + offs_h[None, :] * stride_pr_h,
                     mask=(offs_m[:, None] < M) & (offs_h[None, :] < H), other=0.0)
    gate_r = tl.load(gate_right_ptr + offs_m[:, None] * stride_gr_m + offs_h[None, :] * stride_gr_h,
                     mask=(offs_m[:, None] < M) & (offs_h[None, :] < H), other=0.0)

    out = torch.nn.functional.linear(out, to_out_weight)  # (B,N,N,dim) in fp16

    return out.to(torch.float32)

"""
TriMul (outgoing) – fast forward implementation
================================================

This implementation reproduces the reference ``TriMul`` forward pass while
optimising the heavy work with three ideas:

1️⃣  A **single fused linear** maps the input to all five intermediate
    tensors (left‑proj, left‑gate, right‑proj, right‑gate, out‑gate).  This
    reduces the number of matmul passes over the large ``[B,N,N,dim]`` tensor
    from five to one and cuts memory traffic dramatically.

2️⃣  A **Triton kernel** fuses the sigmoid gates, the optional mask and the
    element‑wise products for *both* the left and the right branches.  The
    kernel loads ``proj`` and ``gate`` together, computes the sigmoid in‑place,
    applies the mask and stores the resulting ``left``/``right`` tensors in a
    single launch.

3️⃣  The ``O(N³)`` contraction ``out = left @ rightᵀ`` is performed with a
    batched GEMM using ``torch.bmm`` (cuBLAS/Tensor‑Core).  cuBLAS already
    provides an extremely well‑optimised implementation for the large matrix
    sizes required (N ≤ 1024) and the total batch size (B·hidden_dim ≤ 256).

All heavy arithmetic is carried out in FP16; the final result is cast back
to FP32 as expected by the caller.  The code works for any of the test
configurations described in the problem statement and respects the optional
mask flag.

"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, projection, gate for *both* left and right
#   out = proj * sigmoid(gate) * mask     (mask broadcast over hidden)
# ----------------------------------------------------------------------
@triton.jit
def _fused_proj_gate_mask_kernel(
    proj_l_ptr, gate_l_ptr,         # left projection / gate
    proj_r_ptr, gate_r_ptr,         # right projection / gate
    mask_ptr,                       # 1‑D mask (M,)
    out_l_ptr, out_r_ptr,           # output buffers (M, H)
    M, H,
    stride_proj_l_m, stride_proj_l_h,
    stride_gate_l_m, stride_gate_l_h,
    stride_proj_r_m, stride_proj_r_h,
    stride_gate_r_m, stride_gate_r_h,
    stride_out_l_m, stride_out_l_h,
    stride_out_r_m, stride_out_r_h,
    stride_mask,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    offs_h = tl.arange(0, BLOCK_H)                         # [BLOCK_H]

    # ---------- mask (broadcast to hidden) ----------
    mask_val = tl.load(mask_ptr + offs_m * stride_mask,
                       mask=offs_m < M,
                       other=0.0).to(tl.float16)        # [BLOCK_M]
    mask_val = mask_val[:, None]                           # [BLOCK_M, 1]

    # ---------- left side ----------
    proj_l = tl.load(
        proj_l_ptr + offs_m[:, None] * stride_proj_l_m
                     + offs_h[None, :] * stride_proj_l_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    gate_l = tl.load(
        gate_l_ptr + offs_m[:, None] * stride_gate_l_m
                     + offs_h[None, :] * stride_gate_l_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    # sigmoid in FP16
    gate_l_sig = 1.0 / (1.0 + tl.exp(-gate_l))

    out_l = proj_l * gate_l_sig * mask_val

    # ---------- right side ----------
    proj_r = tl.load(
        proj_r_ptr + offs_m[:, None] * stride_proj_r_m
                     + offs_h[None, :] * stride_proj_r_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    gate_r = tl.load(
        gate_r_ptr + offs_m[:, None] * stride_gate_r_m
                     + offs_h[None, :] * stride_gate_r_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    gate_r_sig = 1.0 / (1.0 + tl.exp(-gate_r))

    out_r = proj_r * gate_r_sig * mask_val

    # ---------- write ----------
    tl.store(
        out_l_ptr + offs_m[:, None] * stride_out_l_m
                     + offs_h[None, :] * stride_out_l_h,
        out_l,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
    )
    tl.store(
        out_r_ptr + offs_m[:, None] * stride_out_r_m
                     + offs_h[None, :] * stride_out_r_h,
        out_r,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
    )


# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Parameters
    ----------
    data :
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device as the input.
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    device = inp.device
    B, N, _, _ = inp.shape
    M = B * N * N                      # flattened pair‑count

    # --------------------------------------------------------------
    # 1️⃣ first LayerNorm (float32 → FP16)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    x = torch.nn.functional.layer_norm(inp, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)            # [B,N,N,dim] FP16

    # --------------------------------------------------------------
    # 2️⃣ single fused linear for all 5 branches
    # --------------------------------------------------------------
    # cast all linear weights once
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (hidden, dim)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (hidden, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # concatenate vertically: shape (5*hidden, dim)
    fused_W = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )  # (5*hidden, dim)

    # flatten input for the linear
    x_flat = x.view(-1, dim)                    # (M, dim)
    fused_out = torch.nn.functional.linear(x_flat, fused_W)  # (M, 5*hidden)

    # split into the five tensors
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = torch.split(
        fused_out, hidden, dim=1
    )                                            # each (M, hidden)

    # --------------------------------------------------------------
    # 3️⃣ optional mask + gate fusion (Triton)
    # --------------------------------------------------------------
    if nomask:
        # no mask – simple element‑wise fuse in PyTorch (fast)
        left_gate = torch.sigmoid(left_gate_raw)
        right_gate = torch.sigmoid(right_gate_raw)

        left  = left_proj_raw * left_gate
        right = right_proj_raw * right_gate
    else:
        # flatten mask to (M,) and cast to FP16
        mask_fp16 = mask.to(torch.float16).view(-1)  # (M,)

        # allocate the fused outputs
        left = torch.empty_like(left_proj_raw)
        right = torch.empty_like(right_proj_raw)

        # strides (in element counts)
        s_lp_m, s_lp_h = left_proj_raw.stride()
        s_lg_m, s_lg_h = left_gate_raw.stride()
        s_rp_m, s_rp_h = right_proj_raw.stride()
        s_rg_m, s_rg_h = right_gate_raw.stride()
        s_lo_m, s_lo_h = left.stride()
        s_ro_m, s_ro_h = right.stride()
        s_mask = mask_fp16.stride()[0]

        # launch parameters – tuned for H100
        BLOCK_M = 64                 # rows per program
        BLOCK_H = hidden             # whole hidden channel (≤384)

        grid = ((M + BLOCK_M - 1) // BLOCK_M,)

        _fused_proj_gate_mask_kernel[grid](
            left_proj_raw, left_gate_raw,
            right_proj_raw, right_gate_raw,
            mask_fp16,
            left, right,
            M, hidden,
            s_lp_m, s_lp_h,
            s_lg_m, s_lg_h,
            s_rp_m, s_rp_h,
            s_rg_m, s_rg_h,
            s_lo_m, s_lo_h,
            s_ro_m, s_ro_h,
            s_mask,
            BLOCK_M=BLOCK_M,
            BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )

    # reshape the left/right tensors back to (B,N,N,hidden)
    left = left.view(B, N, N, hidden)
    right = right.view(B, N, N, hidden)

    # --------------------------------------------------------------
    # 4️⃣ O(N³) contraction via batched GEMM (cuBLAS/TensorCore)
    # --------------------------------------------------------------
    #  (B,N,N,hidden) → (B,hidden,N,N) → (B*hidden,N,N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden, N, N)  # note the (k,j) swap

    # Batched matrix multiplication: C = A @ B
    out_mat = torch.bmm(left_mat, right_mat)                           # (B*hidden, N, N)

    # reshape back to (B,N,N,hidden)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ second LayerNorm (FP16)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out, (hidden,), weight=to_out_norm_w, bias=to_out_norm_b
    )

    # --------------------------------------------------------------
    # 6️⃣ out‑gate gating & final linear projection
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw).view(B, N, N, hidden)
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden)
    out = torch.nn.functional.linear(out, to_out_w)        # (B,N,N,dim) FP16

    # --------------------------------------------------------------
    # 7️⃣ cast back to FP32
    # --------------------------------------------------------------
    return out.to(torch.float32)
