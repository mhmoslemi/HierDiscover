@triton.jit
def _mask_gate_fuse_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    M, H,
    stride_proj_m, stride_proj_h,
    stride_gate_m, stride_gate_h,
    stride_out_m, stride_out_h,
    stride_mask,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask = tl.load(mask_ptr + offs_m * stride_mask, mask=offs_m < M, other=0.0).to(tl.float16)
    mask_h = mask[:, None]

    proj = tl.load(proj_ptr + offs_m[:, None] * stride_proj_m + offs_h[None, :] * stride_proj_h,
                   mask=(offs_m[:, None] < M) & (offs_h[None, :] < H), other=0.0)
    gate = tl.load(gate_ptr + offs_m[:, None] * stride_gate_m + offs_h[None, :] * stride_gate_h,
                   mask=(offs_m[:, None] < M) & (offs_h[None, :] < H), other=0.0)
    out = proj * gate * mask_h

    tl.store(out_ptr + offs_m[:, None] * stride_out_m + offs_h[None, :] * stride_out_h,
             out,
             mask=(offs_m[:, None] < M) & (offs_h[None, :] < H))
```

"""
TriMul (outgoing) – highly‑optimized forward pass

The implementation follows the reference PyTorch module but fuses several
operations to minimise memory traffic:

1. **LayerNorm** – performed once in fp16.
2. **5 linear transforms** (left/right projections, three gates) are
   merged into a single GEMM (input × concatenated weight matrix).  This
   eliminates repeated loads of the large input tensor.
3. **Mask + gate fusion** – a small Triton kernel computes
   `proj * gate * mask` for the left and right branches in a single pass.
   When there is no mask the element‑wise multiplication is performed with
   a fused PyTorch kernel.
4. **O(N³) contraction** – realised as a batched GEMM
   `out = left @ rightᵀ` where the batch dimension is `batch*hidden_dim`.
   The tensors are only permuted (viewed) – no extra copies are made.
5. **Second LayerNorm**, output‑gate scaling and final linear projection are
   carried out in fp16, the result is cast back to fp32 for the caller.

All heavy arithmetic stays in fp16 (Tensor Cores) while the two
element‑wise kernels are written in Triton.  The code works for the full
range of test configurations (different batch sizes, sequence lengths,
`dim` ∈ {128, 384}, `hidden_dim` = 128 and optional masks).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fused mask, projection and gate
#   out = proj * gate * mask[...,None]   # proj, gate: (M, H), mask: (M,)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,               # tensors
    M, H,                                                 # problem size
    stride_proj_m, stride_proj_h,                         # proj strides
    stride_gate_m, stride_gate_h,                         # gate strides
    stride_out_m,  stride_out_h,                          # out strides
    stride_mask,                                          # mask stride (1‑D)
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,        # tile sizes
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)     # [BLOCK_M]
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)     # [BLOCK_H]

    # Load mask (1‑D) and broadcast to hidden dim
    mask = tl.load(mask_ptr + offs_m * stride_mask,
                   mask=offs_m < M, other=0.0).to(tl.float16)   # [BLOCK_M]
    mask_h = mask[:, None]                                          # [BLOCK_M,1]

    # Load projection and gate tiles
    proj = tl.load(
        proj_ptr + offs_m[:, None] * stride_proj_m
                  + offs_h[None, :] * stride_proj_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    gate = tl.load(
        gate_ptr + offs_m[:, None] * stride_gate_m
                  + offs_h[None, :] * stride_gate_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    # Fusion
    out = proj * gate * mask_h

    # Store result
    tl.store(
        out_ptr + offs_m[:, None] * stride_out_m
                 + offs_h[None, :] * stride_out_h,
        out,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
    )


# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32, bool or None)
        - weights_dict : {name: torch.Tensor} (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device as
        ``input_tensor``.
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    ).to(torch.float16)  # shape [B, N, N, dim] in fp16

    B, N, _, _ = x.shape
    M = B * N * N                     # flattened pair‑wise positions

    # --------------------------------------------------------------
    # 2️⃣ Merge the five linear transforms into a single GEMM
    # --------------------------------------------------------------
    # linear weights have shape (out_features, in_features) = (hidden, dim)
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]
    # concatenate along the output‑feature dimension (0)
    combined_w = torch.cat([
        left_proj_w, right_proj_w,
        left_gate_w, right_gate_w,
        out_gate_w,
    ], dim=0).to(torch.float16)               # shape (5*hidden, dim)

    # flatten input to (M, dim)
    x_flat = x.view(-1, dim)                  # (M, dim)

    # single GEMM: (M, dim) @ (5*hidden, dim).T  → (M, 5*hidden)
    combined_out = torch.nn.functional.linear(x_flat, combined_w)  # fp16

    # --------------------------------------------------------------
    # 3️⃣ Slice the concatenated result
    # --------------------------------------------------------------
    h = hidden_dim
    left_proj     = combined_out[:, 0 * h : 1 * h]
    right_proj    = combined_out[:, 1 * h : 2 * h]
    left_gate_raw = combined_out[:, 2 * h : 3 * h]
    right_gate_raw= combined_out[:, 3 * h : 4 * h]
    out_gate_raw  = combined_out[:, 4 * h : 5 * h]

    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    # --------------------------------------------------------------
    # 4️⃣ Optional mask + gate fusion (Triton)
    # --------------------------------------------------------------
    if not nomask:
        # flatten mask to (M,)
        mask_fp16 = mask_tensor.to(torch.float16).view(-1)   # (M,)

        # allocate fused outputs
        left  = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # strides for the tensors (in element counts)
        stride_lp_m, stride_lp_h = left_proj.stride()
        stride_lg_m, stride_lg_h = left_gate.stride()
        stride_rp_m, stride_rp_h = right_proj.stride()
        stride_rg_m, stride_rg_h = right_gate.stride()
        stride_l_m,  stride_l_h  = left.stride()
        stride_r_m,  stride_r_h  = right.stride()
        stride_mask = mask_fp16.stride()[0]

        # kernel launch configuration
        BLOCK_M = 64                # rows per block (tunable)
        BLOCK_H = 64                # hidden tile size
        grid = (
            (M + BLOCK_M - 1) // BLOCK_M,
            (h + BLOCK_H - 1) // BLOCK_H,
        )

        # left = left_proj * left_gate * mask
        _mask_gate_fuse_kernel[grid](
            left_proj, left_gate, mask_fp16, left,
            M, h,
            stride_lp_m, stride_lp_h,
            stride_lg_m, stride_lg_h,
            stride_l_m,  stride_l_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
        # right = right_proj * right_gate * mask
        _mask_gate_fuse_kernel[grid](
            right_proj, right_gate, mask_fp16, right,
            M, h,
            stride_rp_m, stride_rp_h,
            stride_rg_m, stride_rg_h,
            stride_r_m,  stride_r_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
    else:
        # no mask – simple element‑wise fuse (fast, single kernel)
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # reshape back to (B, N, N, hidden)
    left  = left.view(B, N, N, h)
    right = right.view(B, N, N, h)

    # --------------------------------------------------------------
    # 5️⃣ O(N³) contraction via batched GEMM (cublas / TensorCore)
    #    Compute out = left @ rightᵀ for each (batch, hidden) pair.
    # --------------------------------------------------------------
    # Bring hidden to the batch dimension without copying:
    #   (B, N, N, h) → (B, h, N, N) → (B*h, N, N)
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * h, N, N)   # (B*h, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * h, N, N)

    # Batched matrix multiply: (B*h, N, N) @ (B*h, N, N)ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*h, N, N)

    # Restore original layout: (B, N, N, h)
    out = out_mat.view(B, h, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (fp16)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (h,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 7️⃣ Output gate and final linear projection (fp16)
    # --------------------------------------------------------------
    out = out * out_gate                     # element‑wise gating
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden)

    # flatten to (M, hidden) → linear → reshape back
    out = torch.nn.functional.linear(out.view(-1, h), to_out_w)  # (M, dim)
    out = out.view(B, N, N, dim)

    # --------------------------------------------------------------
    # 8️⃣ Cast back to fp32 (expected by the caller)
    # --------------------------------------------------------------
    return out.to(torch.float32)
