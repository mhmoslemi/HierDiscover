def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    B_dim, N, H, K,
    stride_ab, stride_ai, stride_ak,
    stride_bb, stride_bj, stride_bk,
    stride_cb, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0) # tile i
    pid_n = tl.program_id(1) # tile j
    pid_z = tl.program_id(2) # combined batch*hidden index [M]

    Bdim = B_dim # batch sizes
    Hdim = H

    # compute b and h
    b = pid_z // Hdim
    h = pid_z % Hdim

    # base pointers
    a_ptr = A_ptr + b * stride_ab + h * 1 # stride_h = 1
    b_ptr = B_ptr + b * stride_bb + h * 1
    c_ptr = C_ptr + b * stride_cb + h * 1

    # offsets for i and j
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k_offset = k_tile * BLOCK_K
        offs_k = cur_k_offset + tl.arange(0, BLOCK_K)

        # load A tile (i,k)
        a_ptrs = a_ptr + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak

"""
Outgoing TriMul (AlphaFold‑3) – Optimized forward pass.
=====================================================
Algorithm
---------
1️⃣  Layer‑norm the input (float32) → cast to fp16.
2️⃣  Fuse the four linear projections (left/right) and their gates:
       * concatenate the projection and gate weight matrices,
       * perform one matmul per side, then split + sigmoid.
3️⃣  Apply optional mask (broadcast on hidden dim) to the projections.
4️⃣  Compute the heavy O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   – For all sequence lengths we use a custom Triton GEMM that treats the
     hidden dimension as a batch dimension (M = B·H).  Tiles are
     128×128×128 (M‑tile: 128, N‑tile: 128, K‑tile: 128), accumulating in
     fp32 for numerical stability.
5️⃣  Second LayerNorm (fp16) → element‑wise out‑gate (sigmoid) → final linear
   projection back to `dim`.
6️⃣  Cast the result to float32 (required by the harness).

All linear weights are kept in fp16.  The Triton kernel works directly on
the original `[B, N, N, H]` layout thus no costly permute+contiguous copy is
performed for the contraction.  The implementation is fully self‑contained
and supports the `nomask` flag.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM   C = A @ Bᵀ
#   A, B : [B, N, N, H] (fp16)
#   C    : [B, N, N, H] (fp16)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [B, N, N, H]
    B_ptr,               # *fp16  [B, N, N, H]
    C_ptr,               # *fp16  [B, N, N, H]
    batch, seq_len, hidden, K,
    stride_am, stride_ai, stride_ak,   # A strides (batch, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (batch, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the TriMul contraction.

    The hidden dimension is folded into the batch dimension:
    program_id(2) == batch * hidden_index.
    """
    pid_m = tl.program_id(0)   # tile along i
    pid_n = tl.program_id(1)   # tile along j
    pid_z = tl.program_id(2)   # combined batch*hidden index

    # decode batch & hidden index from pid_z
    hidden_dim = hidden
    b = pid_z // hidden_dim          # 0 <= b < batch
    h = pid_z % hidden_dim           # 0 <= h < hidden

    # offsets within the i / j dimensions
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < seq_len
    mask_j = offs_j < seq_len

    # pointers to the beginning of the (b, h) slice
    a_base = A_ptr + b * stride_am + h        # hidden stride = 1
    b_base = B_ptr + b * stride_bm + h
    c_base = C_ptr + b * stride_cm + h

    # accumulator (fp32) for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ----- load A[i, k] ------------------------------------------------
        a_ptrs = a_base \
                 + offs_i[:, None] * stride_ai \
                 + offs_k[None, :] * stride_ak
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # ----- load B[j, k] (note the transpose) -------------------------
        b_ptrs = b_base \
                 + offs_k[:, None] * stride_bk \
                 + offs_j[None, :] * stride_bj
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # mixed‑precision dot product
        acc += tl.dot(a, b)

    # ----- store the result ------------------------------------------------
    c_ptrs = c_base \
             + offs_i[:, None] * stride_ci \
             + offs_j[None, :] * stride_cj
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract `left` and `right` (both [B, N, N, H]) with the Triton kernel.
    Returns tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    C = torch.empty_like(left, dtype=torch.float16)

    # strides in element units (not bytes)
    stride_am, stride_ai, stride_ak, _ = left.stride()
    stride_bm, stride_bj, stride_bk, _ = right.stride()
    stride_cm, stride_ci, stride_cj, _ = C.stride()

    # tile size – 128×128×128 works well on H100 for fp16 MMA
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 128

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden
    )

    _trimul_matmul_kernel[grid](
        left, right, C,
        B, N, H, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    return C


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract `left` and `right` using torch.bmm (cuBLAS).
    All tensors are fp16, accumulation is fp32 automatically.
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] (contiguous after a permute)
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 / bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ first LayerNorm (float32) – stability
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2️⃣ cast to fp16 for the bulk of the computation
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ cast all linear weights to fp16 once
    # ------------------------------------------------------------------
    # fuse projection + gate for left side
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2*H, dim)

    # fuse projection + gate for right side
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, dim)

    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, H)

    # ------------------------------------------------------------------
    # 4️⃣ fused linear projections + gates (fp16)
    # ------------------------------------------------------------------
    # left side
    left_comb = F.linear(x, left_comb_w)          # [B, N, N, 2*H]
    left_proj = left_comb[..., :hidden]           # [B, N, N, H]
    left_gate = torch.sigmoid(left_comb[..., hidden:])  # [B, N, N, H]

    # right side
    right_comb = F.linear(x, right_comb_w)        # [B, N, N, 2*H]
    right_proj = right_comb[..., :hidden]
    right_gate = torch.sigmoid(right_comb[..., hidden:])

    # ------------------------------------------------------------------
    # 5️⃣ optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ element‑wise gating
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ output gate (same for the whole tensor)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8️⃣ heavy O(N³) contraction (batched GEMM)
    # ------------------------------------------------------------------
    seq_len = input_tensor.shape[1]
    if seq_len <= 256:
        # Small sequences – custom Triton kernel (no extra copies)
        out = _batched_matmul_triton(left, right)       # [B, N, N, H] fp16
    else:
        # Larger sequences – use cuBLAS via torch.bmm (still efficient)
        out = _batched_matmul_torch(left, right)

    # ------------------------------------------------------------------
    # 9️⃣ second LayerNorm → out‑gate → final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)          # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 🔟 cast back to float32 (required by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
