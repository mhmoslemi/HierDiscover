"""
Fast “outgoing” TriMul (AlphaFold‑3) forward pass – Triton‑fused mask & gating.

Algorithm
---------
1️⃣ Layer‑norm the input (float‑32 for stability).
2️⃣ Cast to fp16 and perform **two** large linear projections:
   * one concatenating left/right projections   → shape [..., 2·H]
   * one concatenating three gate projections   → shape [..., 3·H]
3️⃣ Split the results → left, right, left_gate, right_gate, out_gate.
   Apply sigmoid to the three gate tensors.
4️⃣ If a mask is present, fuse mask‑multiplication and gate‑multiplication
   into a tiny Triton kernel that works **in‑place** on ``left`` and ``right``.
   (When ``nomask=True`` the same effect is achieved by a simple PyTorch
   element‑wise multiplication.)
5️⃣ Batched matrix‑multiply the transformed tensors:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a strided‑batched GEMM (B·H batches) performed with ``torch.bmm``
   which on an H100 uses Tensor‑cores.
6️⃣ Second LayerNorm over the hidden dimension (fp16), multiply by ``out_gate``.
7️⃣ Final linear projection back to the original dimension and cast to float‑32.

Only the mask‑&‑gate fusion kernel is written in Triton – the heavy O(N³)
contraction uses cuBLAS (torch.bmm) for maximal throughput.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate multiplications.
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_fused_kernel(
    left_ptr,               # fp16 * (B·N·N·H)
    right_ptr,              # fp16 * (B·N·N·H)
    left_gate_ptr,          # fp16 * (B·N·N·H)
    right_gate_ptr,         # fp16 * (B·N·N·H)
    mask_ptr,               # fp16 * (B·N·N)  (broadcasted over H)
    H: tl.constexpr,        # hidden dimension
    TOTAL_ELEMS: tl.constexpr,  # B·N·N·H
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < TOTAL_ELEMS

    # index into the mask (broadcast H times)
    mask_idx = offs // H

    # load values (fp16)
    l   = tl.load(left_ptr   + offs,          mask=mask, other=tl.float16(0.0))
    r   = tl.load(right_ptr  + offs,          mask=mask, other=tl.float16(0.0))
    lg  = tl.load(left_gate_ptr  + offs,     mask=mask, other=tl.float16(0.0))
    rg  = tl.load(right_gate_ptr + offs,     mask=mask, other=tl.float16(0.0))
    m   = tl.load(mask_ptr + mask_idx,       mask=mask, other=tl.float16(0.0))

    # fused element‑wise multiplication
    l = l * lg * m
    r = r * rg * m

    # write back in‑place
    tl.store(left_ptr  + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched matmul C = A @ Bᵀ   (A,B : [B,N,N,H] fp16) → out [B,N,N,H]
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right: [B, N, N, H]  (fp16)
    Returns    : [B, N, N, H] (fp16) where out = left @ rightᵀ per hidden slice.
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N]   (i,k) and (k,j)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (i,k)
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)  # (k,j)
    out_flat = torch.bmm(left_flat, right_flat)  # (i,j) = Σ_k left * right
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor[B, N, N]       (float32 / bool) – optional
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------
    # 2️⃣ Cast to fp16 for all heavy work
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Convert model weights to fp16 once
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4️⃣ Fuse the two projection linear layers
    # --------------------------------------------------------------
    proj_weight = torch.cat([left_proj_w, right_proj_w], dim=0)          # [2·H, dim]
    proj = F.linear(x_fp16, proj_weight)                               # [B, N, N, 2·H]
    left  = proj[..., :hidden_dim]                                     # [B, N, N, H]
    right = proj[..., hidden_dim:]                                     # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Fuse the three gate linear layers
    # --------------------------------------------------------------
    gate_weight = torch.cat([left_gate_w, right_gate_w, out_gate_w], dim=0)  # [3·H, dim]
    gates = F.linear(x_fp16, gate_weight)                                   # [B, N, N, 3·H]
    left_gate  = torch.sigmoid(gates[..., :hidden_dim])
    right_gate = torch.sigmoid(gates[..., hidden_dim:2 * hidden_dim])
    out_gate   = torch.sigmoid(gates[..., 2 * hidden_dim:])

    # --------------------------------------------------------------
    # 6️⃣ Optional mask + gate fusion (Triton)
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → fp16 and broadcast over H
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(-1)          # (B·N·N,)
        # Ensure tensors are contiguous before we hand over raw pointers
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total_elems = left.numel()                     # B·N·N·H
        BLOCK = 8192
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        _apply_mask_gate_fused_kernel[grid](
            left.data_ptr(),
            right.data_ptr(),
            left_gate.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            hidden_dim,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
            num_stages=2,
        )
    else:
        # No mask → just apply the gates (still fp16)
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 7️⃣ O(N³) contraction via a batched GEMM (Tensor‑core)
    # --------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, H] (fp16)

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # still fp16

    # --------------------------------------------------------------
    # 9️⃣ Output gating and final projection
    # --------------------------------------------------------------
    out = out * out_gate                       # element‑wise fp16
    out = F.linear(out, to_out_w)              # [B, N, N, dim] fp16

    # --------------------------------------------------------------
    # 10️⃣ Cast back to float32 (required by the harness)
    # --------------------------------------------------------------
    return out.to(torch.float32)
