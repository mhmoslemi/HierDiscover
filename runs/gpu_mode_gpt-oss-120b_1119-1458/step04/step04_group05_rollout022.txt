```python
def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # First layernorm
    norm_weight = weights["norm.weight"].to(dtype=torch.float32)
    norm_bias = weights["norm.bias"].to(dtype=torch.float32)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    x = x.to(torch.float16)

    # Prepare fused weights
    left_comb_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16)  # shape (2*hidden_dim, dim)
    right_comb_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)  # shape (hidden_dim, dim)
    # not forgetting to .contiguous()
    # Flatten input for linear.
    B, N, _, _ = x.shape
    flat = x.view(-1, dim)  # (B*N*N, dim)

    # Compute left_proj+gate and right_proj+gate via fused linear.
    # left_comb: (B*N*N, 2*hidden_dim)
    left_comb = torch.nn.functional.linear(flat, left_comb_weight)  # no bias
    # left combination split
    left_proj = left_comb[:, :hidden_dim]
    left_gate = left_comb[:, hidden_dim:]  # shape (B*N*N, hidden_dim)
    left_gate = torch.sigmoid(left_gate)

    # Similar right
    right_comb = torch.nn.functional.linear(flat, right_comb_weight)  # (B*N*N, 2*hidden_dim)
    right_proj = right_comb[:, :hidden_dim]
    right_gate = right_comb[:, hidden_dim:]
    right_gate = torch.sigmoid(right_gate)

    # Compute out_gate
    out_gate = torch.nn.functional.linear(flat, out_gate_weight)  # shape (B*N*N, hidden_dim)
    out_gate = torch.sigmoid(out_gate)

    # Reshape back to (B,N,N,hidden_dim)
    left_proj = left_proj.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)
```

```
            other=0.0,
        )
```

        out = inp * mask[..., None]
        # store
        tl.store(

TriMul (outgoing) – fast Triton‑augmented implementation.

The forward pass consists of:
1️⃣   LayerNorm of the input (float32→float16).
2️⃣   Fused linear projections for left/right features and their gates
     (one matmul per side, i.e. 2 large GEMMs instead of 4).
3️⃣   Sigmoid‑gating and optional mask‑application.  Masking is performed
     with a tiny Triton kernel that multiplies an (B,N,N,H) tensor by a
     (B,N,N) mask, broadcasting over the hidden dimension.
4️⃣   The O(N³) “triangle” contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
     Implemented as a batched GEMM (torch.bmm) on FP16 tensors.  This
     leverages the highly‑optimized cuBLAS kernels on H100 and is far
     faster than a custom Triton matmul for the sizes used in the tests.
5️⃣   Second LayerNorm, output‑gate (sigmoid) and final projection back to
     the original dimension.
All heavy arithmetic is in FP16; the final tensor is returned as float32.

        other=0.0,

    # Loop over the hidden dimension in tiles
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        cur_h = h_tile * BLOCK_H
        offs_h = cur_h + tl.arange(0, BLOCK_H)
        h_mask = offs_h < H

        # Load a tile of the input tensor
        inp = tl.load(
            inp_ptr
            + pid_b * stride_in_0
            + offs_i[:, None, None] * stride_in_1
            + offs_j[None, :, None] * stride_in_2
            + offs_h[None, None, :] * stride_in_3,
            mask=(
                (offs_i[:, None, None] < N)
                & (offs_j[None, :, None] < N)
                & (offs_h[None, None, :] < H)
            ),
            other=0.0,
        )  # fp16

        # Apply the mask (broadcast over hidden dim)
        out = inp * mask[..., None]

        # Store the result
        tl.store(
            out_ptr
            + pid_b * stride_in_0
            + offs_i[:, None, None] * stride_in_1
            + offs_j[None, :, None] * stride_in_2
            + offs_h[None, None, :] * stride_in_3,
            out,
            mask=(
                (offs_i[:, None, None] < N)
                & (offs_j[None, :, None] < N)
                & (offs_h[None, None, :] < H)
            ),
        )


def _apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Apply a (B,N,N) mask to a (B,N,N,H) tensor using the Triton kernel above.
    The mask is expected to contain 0.0/1.0 values (float16) and is broadcast
    over the hidden dimension.
    """
    B, N, _, H = tensor.shape
    # Ensure the mask is 3‑D and contiguous
    mask3d = mask.contiguous().view(B, N, N)

    out = torch.empty_like(tensor)

    # Tile sizes – tuned for H100 (fits into registers & shared memory)
    BLOCK_N = 64
    BLOCK_H = 32

    grid = (
        B,
        (N + BLOCK_N - 1) // BLOCK_N,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    stride_t = tensor.stride()
    stride_m = mask3d.stride()

    _mask_apply_kernel[grid](
        tensor,
        mask3d,
        out,
        B,
        N,
        H,
        stride_t[0],
        stride_t[1],
        stride_t[2],
        stride_t[3],
        stride_m[0],
        stride_m[1],
        stride_m[2],
        BLOCK_N=BLOCK_N,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return out


def custom_kernel(
    data: Tuple[
        torch.Tensor,                # input tensor (B, N, N, dim) – float32
        torch.Tensor,                # mask tensor  (B, N, N)   – float32 / bool / None
        Dict[str, torch.Tensor],     # model weights (float32)
        Dict[str, Any],              # config (contains dim, hidden_dim, nomask)
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (gradient‑free).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # 1️⃣ Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2️⃣ First LayerNorm (float32 → FP16)
    # --------------------------------------------------------------
    # LayerNorm params are stored as float32 – keep them in that dtype for
    # numerical stability, then cast the result to FP16 for the bulk of the
    # computation.
    norm_weight = weights["norm.weight"].to(device)    # (dim,)
    norm_bias   = weights["norm.bias"].to(device)      # (dim,)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32
    x = x.to(torch.float16)                            # switch to half‑precision

    # --------------------------------------------------------------
    # 3️⃣ Fuse linear projections for left/right + their gates
    # --------------------------------------------------------------
    # left_proj + left_gate  →  (2*hidden_dim, dim)
    # right_proj + right_gate → (2*hidden_dim, dim)
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                                   # (2*hidden_dim, dim)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                                   # (2*hidden_dim, dim)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)    # (hidden_dim, dim)

    B, N, _, _ = x.shape
    flat = x.view(-1, dim)                                # (B·N·N, dim)

    # Fused projection + gate (no bias)
    left_comb = torch.nn.functional.linear(flat, left_comb_w)   # (B·N·N, 2*hidden_dim)
    right_comb = torch.nn.functional.linear(flat, right_comb_w) # (B·N·N, 2*hidden_dim)

    # Split back to proj / gate and apply sigmoid on the gate part
    left_proj = left_comb[:, :hidden_dim]
    left_gate = left_comb[:, hidden_dim:].sigmoid()
    right_proj = right_comb[:, :hidden_dim]
    right_gate = right_comb[:, hidden_dim:].sigmoid()

    # Output‑gate (no split, just sigmoid)
    out_gate = torch.nn.functional.linear(flat, out_gate_w).sigmoid()

    # Reshape to 4‑D tensors
    left = left_proj.view(B, N, N, hidden_dim) * left_gate.view(B, N, N, hidden_dim)
    right = right_proj.view(B, N, N, hidden_dim) * right_gate.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (fused via a tiny Triton kernel)
    # --------------------------------------------------------------
    if not nomask:
        # mask may be bool or float – cast to FP16 and make it 3‑D
        mask_fp16 = mask_tensor.to(torch.float16)
        left = _apply_mask_triton(left, mask_fp16)      # (B,N,N,hidden_dim)
        right = _apply_mask_triton(right, mask_fp16)    # (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction – batched GEMM (torch.bmm)
    # --------------------------------------------------------------
    # Convert to (B*H, N, N) where H = hidden_dim.
    # left  : (i,k)   → rows=i, cols=k
    # right : (j,k)   → we need (k,j) → transpose before batched matmul.
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat)                # (B*H, N, N)

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # out shape: (B, N, N, hidden_dim) – still FP16

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm, output‑gate, and final linear projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate            # element‑wise gating (FP16)
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim) FP16

    # --------------------------------------------------------------
    # 7️⃣ Cast back to FP32 for the caller
    # --------------------------------------------------------------
    return out.to(torch.float32)
