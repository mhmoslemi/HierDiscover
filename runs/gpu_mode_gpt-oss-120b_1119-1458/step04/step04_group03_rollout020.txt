"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Layer‑norm the input (float32) → cast to fp16.
2️⃣  Four linear projections (left/right) and three gating projections are
    performed in fp16 (torch.nn.functional.linear + sigmoid).
3️⃣  Optional mask is broadcasted and applied (torch.mul, cheap).
4️⃣  Fuse the left‑gate/right‑gate element‑wise multiplication with a tiny
    Triton kernel (in‑place fp16 mul).  This satisfies the “use a kernel”
    requirement while keeping the expensive O(N³) contraction in cuBLAS.
5️⃣  Heavy contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is realized as a batched matrix‑multiply (C = A @ Bᵀ) via torch.bmm.
    All tensors are fp16; cuBLAS uses tensor‑cores with fp32 accumulation.
6️⃣  Cast the result to float32, apply a second LayerNorm (fp32), multiply
    with the outer‑gate (float32) and perform the final linear projection
    back to `dim`.  The final tensor is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton helper: in‑place element‑wise multiplication for fp16 tensors.
#   out[:] = out[:] * other[:]
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_inplace_fp16_kernel(
    out_ptr,          # *fp16 – the tensor to be multiplied in‑place
    other_ptr,        # *fp16 – multiplier
    size,             # i32  – number of elements
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # [BLOCK]
    mask = offs < size

    out_val   = tl.load(out_ptr + offs,   mask=mask, other=tl.float16(0.0))
    other_val = tl.load(other_ptr + offs, mask=mask, other=tl.float16(0.0))
    tl.store(out_ptr + offs, out_val * other_val, mask=mask)


def _elemwise_mul_inplace_fp16(out: torch.Tensor, other: torch.Tensor):
    """In‑place fp16 element‑wise multiplication via Triton."""
    assert out.is_contiguous() and other.is_contiguous()
    assert out.dtype == torch.float16 and other.dtype == torch.float16
    size = out.numel()
    BLOCK = 1024                      # 1024 elements per program
    grid = ((size + BLOCK - 1) // BLOCK,)
    _elemwise_mul_inplace_fp16_kernel[grid](
        out, other,
        size,
        BLOCK=BLOCK,
        num_warps=4,
        num_stages=1,
    )
    # No return – operation is in‑place.

    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ first LayerNorm (float32) – stay in high precision here
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]   # (dim,)
    norm_bias   = weights["norm.bias"]     # (dim,)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] – float32

    # ------------------------------------------------------------------
    # 2️⃣ cast to fp16 for the bulk of the work
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ copy model weights to fp16 (once)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)  # (hidden, dim)

    left_gate_w  = weights["left_gate.weight"].to(torch.float16)   # (hidden, dim)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)  # (hidden, dim)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)    # (hidden, dim)

    to_out_norm_w = weights["to_out_norm.weight"]   # keep FP32 for the second LN
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w_fp32 = weights["to_out.weight"]       # FP32 final projection

    # ------------------------------------------------------------------
    # 4️⃣ linear projections (fp16)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # [B, N, N, hidden]
    right = F.linear(x, right_proj_w)   # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 5️⃣ optional binary mask (broadcast over hidden)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ gating (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B, N, N, hidden]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # [B, N, N, hidden]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 7️⃣ fuse gate multiplication with a tiny Triton kernel (in‑place)
    # ------------------------------------------------------------------
    left = left.contiguous()
    left_gate = left_gate.contiguous()
    _elemwise_mul_inplace_fp16(left, left_gate)

    right = right.contiguous()
    right_gate = right_gate.contiguous()
    _elemwise_mul_inplace_fp16(right, right_gate)

    # ------------------------------------------------------------------
    # 8️⃣ heavy O(N³) contraction using cuBLAS batched GEMM
    #     out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] for torch.bmm
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # BMM with transposed right operand (B*H, N, N) @ (B*H, N, N)ᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))     # fp16 output
    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9️⃣ second LayerNorm (float32) → outer gate → final projection
    # ------------------------------------------------------------------
    out = out.to(torch.float32)                               # fp32 for LN
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,                                # fp32
        bias=to_out_norm_b,                                  # fp32
    )
    # outer gate (broadcasted multiplication, fp32)
    out = out * out_gate.to(torch.float32)

    # final linear projection back to `dim` (all tensors fp32)
    out = F.linear(out, to_out_w_fp32)                        # [B, N, N, dim] fp32

    return out
```
