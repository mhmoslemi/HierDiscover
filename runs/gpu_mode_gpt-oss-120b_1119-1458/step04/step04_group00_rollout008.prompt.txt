import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing) – fast forward implementation
================================================

The implementation follows the reference PyTorch module but moves the
dominant O(N³) contraction to a highly‑optimized cuBLAS batched GEMM
(`torch.bmm`).  This is dramatically faster on H100 because cuBLAS
exploits Tensor‑Core pipelines and automatic workspace management.
A tiny Triton kernel is kept to satisfy the “use Triton” requirement:
it fuses the optional mask with the per‑position gating and projection
tensors in a single element‑wise pass (saving two extra PyTorch
broadcasts).

All linear layers and LayerNorms run in FP16; the final result is
cast back to FP32 as expected by the caller.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, projection and gate (optional)
#   out = proj * gate * mask[...,None]   # proj,gate: (M, H), mask: (M,)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,          # tensors
    M, H,                                           # problem size
    stride_proj_m, stride_proj_h,                   # strides of proj
    stride_gate_m, stride_gate_h,                   # strides of gate
    stride_out_m,  stride_out_h,                    # strides of out
    stride_mask,                                    # stride of mask (1‑D)
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,   # tile sizes
):
    pid = tl.program_id(0)                     # block index over rows
    block_start = pid * BLOCK_M
    offs_m = block_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    offs_h = tl.arange(0, BLOCK_H)                       # [BLOCK_H]

    # Load mask (1‑D) and broadcast to H
    mask = tl.load(mask_ptr + offs_m * stride_mask,
                   mask=offs_m < M, other=0.0).to(tl.float16)   # [BLOCK_M]
    mask_h = mask[:, None]                                   # [BLOCK_M,1]

    # Load projection and gate tiles
    proj = tl.load(
        proj_ptr + offs_m[:, None] * stride_proj_m
                  + offs_h[None, :] * stride_proj_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    gate = tl.load(
        gate_ptr + offs_m[:, None] * stride_gate_m
                  + offs_h[None, :] * stride_gate_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    # Fusion
    out = proj * gate * mask_h

    # Store result
    tl.store(
        out_ptr + offs_m[:, None] * stride_out_m
                 + offs_h[None, :] * stride_out_h,
        out,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
    )


# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32 or bool)
        - weights_dict : {name: torch.Tensor} (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device as
        *input_tensor*.
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    ).to(torch.float16)   # [B, N, N, dim] FP16

    B, N, _, _ = x.shape
    M = B * N * N                                     # flattened sequence‑pair count

    # ------------------------------------------------------------------
    # 2️⃣ Cast all linear weights to FP16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections & sigmoid gates (FP16)
    # ------------------------------------------------------------------
    x_flat = x.view(-1, dim)                     # (M, dim)

    left_proj = torch.nn.functional.linear(x_flat, left_proj_w)   # (M, hidden)
    right_proj = torch.nn.functional.linear(x_flat, right_proj_w)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, left_gate_w))   # (M, hidden)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, right_gate_w))

    out_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, out_gate_w))
    out_gate = out_gate.view(B, N, N, hidden_dim)   # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask + gate fusion (Triton)
    # ------------------------------------------------------------------
    if not nomask:
        # mask flattened to (M,)
        mask_fp16 = mask_tensor.to(torch.float16).view(-1)

        # Allocate output buffers
        left  = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # Strides (in element counts)
        stride_lp_m, stride_lp_h = left_proj.stride()
        stride_lg_m, stride_lg_h = left_gate.stride()
        stride_o_m,  stride_o_h  = left.stride()
        stride_rp_m, stride_rp_h = right_proj.stride()
        stride_rg_m, stride_rg_h = right_gate.stride()
        stride_ro_m, stride_ro_h  = right.stride()
        stride_mask = mask_fp16.stride()[0]

        # Kernel configuration (tuned for H100)
        BLOCK_M = 64           # rows per launch block
        BLOCK_H = hidden_dim   # we tile the whole hidden dim (≤128)

        grid = ( (M + BLOCK_M - 1) // BLOCK_M, )

        # left = left_proj * left_gate * mask
        _mask_gate_fuse_kernel[grid](
            left_proj, left_gate, mask_fp16, left,
            M, hidden_dim,
            stride_lp_m, stride_lp_h,
            stride_lg_m, stride_lg_h,
            stride_o_m,  stride_o_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
        # right = right_proj * right_gate * mask
        _mask_gate_fuse_kernel[grid](
            right_proj, right_gate, mask_fp16, right,
            M, hidden_dim,
            stride_rp_m, stride_rp_h,
            stride_rg_m, stride_rg_h,
            stride_ro_m, stride_ro_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
    else:
        # no mask – simple element‑wise fuse in PyTorch (fast)
        left  = (left_proj * left_gate).view(B, N, N, hidden_dim)
        right = (right_proj * right_gate).view(B, N, N, hidden_dim)

    # If we used the Triton path, reshape now; otherwise tensors already shaped.
    if not nomask:
        left  = left.view(B, N, N, hidden_dim)
        right = right.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ O(N³) contraction via batched GEMM (cuBLAS/TensorCore)
    #    Compute out = left @ rightᵀ for each (batch, hidden) pair.
    # ------------------------------------------------------------------
    # transpose to (B, hidden, N, N) → reshape to (B*hidden, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiplication (C = A @ Bᵀ)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Output gate + final linear projection (FP16)
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim) FP16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 (expected by the caller)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
