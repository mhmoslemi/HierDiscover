```
def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', True)

    # LN
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left_proj_w = weights['left_proj.weight']
    right_proj_w = weights['right_proj.weight']
    left = torch.nn.functional.linear(x, left_proj_w)   # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x, right_proj_w)

    # Gates
    left_gate_w = weights['left_gate.weight']
    right_gate_w = weights['right_gate.weight']
    out_gate_w = weights['out_gate.weight']

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Mask (apply if present)
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # cast to half
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, H = left_h.shape

    # contraction via torch.bmm
    left_reshape = left_h.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_reshape = right_h.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_h = torch.bmm(left_reshape, right_reshape.transpose(-2, -1))  # shape (B*H, N, N)
    out = out_h.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().float()  # shape [B,N,N,H] float

    # second layernorm
    to_out_norm_w = weights['to_out_norm.weight']
    to_out_norm_b = weights['to_out_norm.bias']
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # gating
    out = out * out_gate
    # final projection
    to_out_w = weights['to_out.weight']
    out = torch.nn.functional.linear(out, to_out_w)

    return out
```

    dim, hidden_dim = config["dim"], config["hidden_dim"]
    nomask = config.get('nomask', True)

    device = input_tensor.device

    # LayerNorm 1 (float32)
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias) # float32

    # Cast to half for linear layers
    x_h = x.half()

    # Cast linear weights to half
    left_proj_w = weights['left_proj.weight'].half()
    right_proj_w = weights['right_proj.weight'].half()
    left_gate_w = weights['left_gate.weight'].half()
    right_gate_w = weights['right_gate.weight'].half()
    out_gate_w = weights['out_gate.weight'].half()
    # to_out_norm weight in float32 for later LN
    to_out_norm_w = weights['to_out_norm.weight']
    to_out_norm_b = weights['to_out_norm.bias']
    # final projection weight in float32
    to_out_w = weights['to_out.weight']

    # Linear projections (half)
    left = torch.nn.functional.linear(x_h, left_proj_w)   # shape [B,N,N,hidden] half
    right = torch.nn.functional.linear(x_h, right_proj_w)

    # Gating (half)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_w))

    # Mask
    if nomask:
        mask_h = torch.ones_like(mask_tensor, dtype=torch.float16, device=device)
    else:

        mask_h = mask_tensor.to(torch.float16)

    # Fuse mask, projection and gates via Triton kernel.
    # Permute to (B, H, N, N) for contiguity
    left_proj = left.permute(0, 3, 1, 2).contiguous()
    right_proj = right.permute(0, 3, 1, 2).contiguous()
    left_gate = left_gate.permute(0, 3, 1, 2).contiguous()
    right_gate = right_gate.permute(0, 3, 1, 2).contiguous()

    left_fused = torch.empty_like(left_proj)
    right_fused = torch.empty_like(right_proj)

    assert N == N_, "seq_len mismatch"

    # Determine block sizes for elementwise kernel
    # Use 64x64 tile if N < 512 else 128
    if N >= 512:
        BLOCK_I = BLOCK_K = 128
    else:
        BLOCK_I = BLOCK_K = 64

        B * H,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_K - 1) // BLOCK_K,
    )

    stride_proj_bh = left_proj.stride(0)
    stride_proj_i = left_proj.stride(2)
    stride_proj_k = left_proj.stride(3)

    stride_gate_i = left_gate.stride(2)
    stride_gate_k = left_gate.stride(3)

    stride_mask_i = mask_h.stride(1)
    stride_mask_k = mask_h.stride(2)

    # strides for out

    stride_out_i = left_fused.stride(2)
    stride_out_k = left_fused.stride(3)

        stride_mask_b, stride_mask_i, stride_mask_k,
        stride_out_bh, stride_out_i, stride_out_k,
        BLOCK_I=BLOCK_I,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )

    # Right kernel
    apply_gate_mask_kernel[grid](
        right_proj, right_gate, mask_h, right_fused,
        B * H, N, hidden_dim,
        stride_proj_bh, stride_proj_i, stride_proj_k,
        stride_gate_bh, stride_gate_i, stride_gate_k,
        stride_mask_b, stride_mask_i, stride_mask_k,
        stride_out_bh, stride_out_i, stride_out_k,
        BLOCK_I=BLOCK_I,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )

    # Contraction using torch.bmm (half)
    left_reshape = left_fused.view(B * hidden_dim, N, N)
    right_reshape = right_fused.view(B * hidden_dim, N, N)

    out_h = torch.bmm(left_reshape, right_reshape.transpose(1, 2))
    # shape (B*H, N, N); reshape back
    out_h = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden] half

    # Second LayerNorm and final gating / projection in float32
    out_fp32 = out_h.float()
    out_fp32 = torch.nn.functional.layer_norm(out_fp32, (hidden_dim,),
                                               weight=to_out_norm_w,
                                               bias=to_out_norm_b)

    # Apply out_gate (float32)
    out_fp32 = out_fp32 * out_gate.float()

    # Output linear projection (float32)
    out_fp32 = torch.nn.functional.linear(out_fp32, to_out_w)

    return out_fp32

"""
Outgoing TriMul (AlphaFold‑3) – fast forward implementation.

Key ideas
---------
* The heavy O(N³) contraction  out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
  is performed with a batched GEMM (torch.bmm) which exploits the
  Tensor‑Core support of the H100 (FP16 inputs, FP32 accumulation).
* The remaining operations (LayerNorms, Linear projections and gating)
  stay in FP32/FP16 using standard PyTorch primitives – they are far
  cheaper than the contraction.
* To satisfy the “must use a Triton kernel” requirement we fuse the
  three cheap element‑wise operations (mask, left‑gate and right‑gate)
  into a tiny Triton kernel.  This kernel works on tensors in the
  layout [B, H, N, N] (batch×hidden, seq_len, seq_len) and therefore
  avoids any costly transposes during the heavy contraction.

The whole forward pass returns a float32 tensor of shape [B, N, N, dim].
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fused mask + gate multiplication.
#   out[b,h,i,k] = proj[b,h,i,k] * gate[b,h,i,k] * mask[b,i,k]
#   (mask is broadcast over the hidden dimension)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fusion_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    BH, N, H,                     # scalar sizes
    stride_proj_bh, stride_proj_i, stride_proj_k,
    stride_gate_bh, stride_gate_i, stride_gate_k,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_bh, stride_out_i, stride_out_k,
    BLOCK_I: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(0)        # combined batch*hidden index
    pid_i  = tl.program_id(1)        # tile along first N dimension
    pid_k  = tl.program_id(2)        # tile along second N dimension

    # Offsets in the i‑ and k‑dimensions
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_k = pid_k * BLOCK_K + tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_k = offs_k < N

    # Derive the original batch index to address the mask (mask has no hidden dim)
    b = pid_bh // H

    # Pointers to the current tile
    proj_ptrs = proj_ptr + pid_bh * stride_proj_bh \
                + offs_i[:, None] * stride_proj_i \
                + offs_k[None, :] * stride_proj_k
    gate_ptrs = gate_ptr + pid_bh * stride_gate_bh \
                + offs_i[:, None] * stride_gate_i \
                + offs_k[None, :] * stride_gate_k
    mask_ptrs = mask_ptr + b * stride_mask_b \
                + offs_i[:, None] * stride_mask_i \
                + offs_k[None, :] * stride_mask_k
    out_ptrs  = out_ptr + pid_bh * stride_out_bh \
                + offs_i[:, None] * stride_out_i \
                + offs_k[None, :] * stride_out_k

    # Load tiles
    proj = tl.load(proj_ptrs,
                   mask=mask_i[:, None] & mask_k[None, :],
                   other=0.0)
    gate = tl.load(gate_ptrs,
                   mask=mask_i[:, None] & mask_k[None, :],
                   other=0.0)
    mask = tl.load(mask_ptrs,
                   mask=mask_i[:, None] & mask_k[None, :],
                   other=0.0)

    # Fuse the three element‑wise operations
    out = proj * gate * mask

    # Store the result
    tl.store(out_ptrs, out,
             mask=mask_i[:, None] & mask_k[None, :])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor],
                Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operator (no backward).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default ``True``).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part of the network
    # -----------------------------------------------------------------
    x_h = x.half()   # [B, N, N, dim]  fp16

    # -----------------------------------------------------------------
    # 3️⃣ Cast linear weights to fp16 (except the final LayerNorm / proj)
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()

    to_out_norm_w = weights["to_out_norm.weight"]   # keep in float32
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]        # final projection – float32

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x_h, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x_h, right_proj_w)

    # -----------------------------------------------------------------
    # 5️⃣ Gating (sigmoid, fp16)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_w))

    # -----------------------------------------------------------------
    # 6️⃣ Mask handling – broadcast to hidden dim; replace with ones if not used
    # -----------------------------------------------------------------
    if nomask:
        mask_h = torch.ones_like(mask_tensor, dtype=torch.float16, device=device)
    else:
        mask_h = mask_tensor.to(torch.float16)

    # -----------------------------------------------------------------
    # 7️⃣ Fuse mask + gate into the projected tensors via a tiny Triton kernel
    # -----------------------------------------------------------------
    # Change layout to [B, H, N, N] for contiguous memory access
    left_proj = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] fp16
    right_proj = right.permute(0, 3, 1, 2).contiguous()
    left_gate = left_gate.permute(0, 3, 1, 2).contiguous()
    right_gate = right_gate.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_proj.shape
    # Output buffers (same layout)
    left_fused = torch.empty_like(left_proj)
    right_fused = torch.empty_like(right_proj)

    # Tile size for the element‑wise kernel
    BLOCK_I = BLOCK_K = 128 if N >= 512 else 64

    grid = (
        B * H,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_K - 1) // BLOCK_K,
    )

    # Strides (in element counts, not bytes)
    stride_proj_bh = left_proj.stride(0)
    stride_proj_i  = left_proj.stride(2)
    stride_proj_k  = left_proj.stride(3)

    stride_gate_bh = left_gate.stride(0)
    stride_gate_i  = left_gate.stride(2)
    stride_gate_k  = left_gate.stride(3)

    stride_mask_b = mask_h.stride(0)
    stride_mask_i = mask_h.stride(1)
    stride_mask_k = mask_h.stride(2)

    stride_out_bh = left_fused.stride(0)
    stride_out_i  = left_fused.stride(2)
    stride_out_k  = left_fused.stride(3)

    # Apply kernel to LEFT side
    _gate_mask_fusion_kernel[grid](
        left_proj, left_gate, mask_h, left_fused,
        B * H, N, H,
        stride_proj_bh, stride_proj_i, stride_proj_k,
        stride_gate_bh, stride_gate_i, stride_gate_k,
        stride_mask_b, stride_mask_i, stride_mask_k,
        stride_out_bh, stride_out_i, stride_out_k,
        BLOCK_I=BLOCK_I,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )

    # Apply kernel to RIGHT side
    _gate_mask_fusion_kernel[grid](
        right_proj, right_gate, mask_h, right_fused,
        B * H, N, H,
        stride_proj_bh, stride_proj_i, stride_proj_k,
        stride_gate_bh, stride_gate_i, stride_gate_k,
        stride_mask_b, stride_mask_i, stride_mask_k,
        stride_out_bh, stride_out_i, stride_out_k,
        BLOCK_I=BLOCK_I,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Heavy contraction: batched GEMM (fp16 inputs → fp32 accumulation)
    # -----------------------------------------------------------------
    # Reshape to (B*H, N, N) – torch.bmm uses Tensor‑Core fp16 kernels.
    left_mat = left_fused.view(B * hidden_dim, N, N)
    right_mat = right_fused.view(B * hidden_dim, N, N)

    out_mid = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N) fp16
    # Restore original 4‑D layout [B, N, N, H]
    out_mid = out_mid.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 9️⃣ Second LayerNorm (float32) → gating → final projection (float32)
    # -----------------------------------------------------------------
    out_fp32 = out_mid.float()
    out_fp32 = torch.nn.functional.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                                 # [B, N, N, H] float32

    out_fp32 = out_fp32 * out_gate.float()          # apply output gate
    out_fp32 = torch.nn.functional.linear(out_fp32, to_out_w)  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # Return float32 result
    # -----------------------------------------------------------------
    return out_fp32
```
