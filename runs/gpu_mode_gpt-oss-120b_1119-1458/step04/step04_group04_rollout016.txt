@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # pid identifiers
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)  # combination of batch and hidden

    # compute batch and hidden index
    b = pid_z // H
    h = pid_z % H

    # offsets for i and j
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tk in range(num_k_tiles):
        cur_k = tk * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile
        # left[b,i,k,h]
        a_ptrs = left_ptr + b * stride_left_b + h * stride_left_h + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None,:], other=0.0)

        # Load right tile transposed: right[b,j,k,h] need (k,j) shape
        b_ptrs = right_ptr + b * stride_right_b + h * stride_right_h + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None,:], other=0.0)

        # multiply-accumulate using dot (M,K)*(K,N) -> (M,N)
        acc += tl.dot(a, b)

    # store out tile
    c_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
Optimized “outgoing” TriMul kernel (AlphaFold‑3).

Algorithm
---------
1.  Layer‑norm the 4‑D input (FP32) → cast to FP16.
2.  Fuse the 5 linear projections (left_proj, left_gate,
    right_proj, right_gate, out_gate) into a single GEMM:
       combined = x @ W_combinedᵀ
   where W_combined is the concatenation of the five weight
   matrices.  Split the result and apply sigmoid to the three gate
   slices.
3.  Fuse element‑wise gating (proj * gate) and the optional mask
   with a cheap broadcasted multiplication.
4.  Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton kernel that works directly on the
   original layout (no permutes, no extra copies).  For very short
   sequences (N ≤ 512) the Triton kernel is used; otherwise we fall
   back to a single cuBLAS batched GEMM (torch.bmm) which is
   optimal on H100.
5.  Second Layer‑norm (FP16) → multiply by out‑gate → final linear
   projection back to `dim`.
6.  Cast the result to FP32 (the harness expects float32).

All major compute‑heavy steps are performed with half‑precision
Tensor‑Core kernels; the whole forward pass uses only a handful of
CUDA launches and fits comfortably within the ~1 ms budget.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction.
#   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# INPUT layout for left/right/out: [B, N, N, H] (contiguous)
# H = hidden dimension, B = batch size.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,          # *fp16  [B, N, N, H]
    right_ptr,         # *fp16  [B, N, N, H]
    out_ptr,           # *fp16  [B, N, N, H]
    B, N, H,           # i32 scalar sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """TriMul contraction kernel (i‑tile, j‑tile, batch*hidden tile)."""
    pid_i = tl.program_id(0)   # i‑tile
    pid_j = tl.program_id(1)   # j‑tile
    pid_z = tl.program_id(2)   # combined batch × hidden index

    # Resolve batch (b) and hidden (h) from the flattened index
    b = pid_z // H
    h = pid_z - b * H   # equivalent to pid_z % H

    # Offsets for the i‑ and j‑dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32 for numeric stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        cur_k = kk * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b,i,k,h]   – shape (M,K) where M=BLOCK_M, K=BLOCK_K
        a_ptrs = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right[b,j,k,h] but we need its transpose (k,j)
        b_ptrs = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_i   # i‑stride = j‑stride
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Mixed‑precision MMA (FP16 * FP16 → FP32 accumulation)
        acc += tl.dot(a, b)

    # Store the result (cast back to FP16)
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def _trimul_contraction(
    left: torch.Tensor,
    right: torch.Tensor,
) -> torch.Tensor:
    """
    Compute the TriMul outer product.
    For N ≤ 512 we launch the custom Triton kernel, otherwise we
    fall back to torch.bmm (cuBLAS) which is faster on large N.
    Both tensors are FP16 of shape [B, N, N, H] and must be contiguous.
    """
    B, N, _, H = left.shape
    # Choose implementation based on sequence length.
    if N <= 512:
        # Allocate output tensor.
        out = torch.empty_like(left)

        # Strides in element units (not bytes).
        s_lb, s_li, s_lk, s_lh = left.stride()
        s_rb, s_ri, s_rk, s_rh = right.stride()
        s_ob, s_oi, s_oj, s_oh = out.stride()

        # Tile sizes – 64×64 works well for short sequences,
        # 128×128 for a bit larger ones.
        BLOCK_M = BLOCK_N = 64 if N <= 256 else 128
        BLOCK_K = 32

        grid = (
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
            B * H,
        )
        _trimul_outgoing_kernel[grid](
            left,
            right,
            out,
            B,
            N,
            H,
            s_lb,
            s_li,
            s_lk,
            s_lh,
            s_rb,
            s_ri,
            s_rk,
            s_rh,
            s_ob,
            s_oi,
            s_oj,
            s_oh,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=8,
            num_stages=2,
        )
        return out
    else:
        # Use an optimized cuBLAS batched GEMM.
        # reshape to [B*H, N, N] and compute A @ Bᵀ.
        a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
        b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
        c = torch.bmm(a, b.transpose(1, 2))
        out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
        return out

    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 / bool) – optional
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable for the initial norm.
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim], float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for all subsequent heavy work.
    # -----------------------------------------------------------------
    x = x.to(dtype_fp16)

    # -----------------------------------------------------------------
    # 3️⃣ Fuse the five linear transforms into a single GEMM.
    #    Weight shape: [5*hidden_dim, dim].
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    # Cast weights to fp16 once.
    left_proj_w = left_proj_w.to(dtype_fp16)
    left_gate_w = left_gate_w.to(dtype_fp16)
    right_proj_w = right_proj_w.to(dtype_fp16)
    right_gate_w = right_gate_w.to(dtype_fp16)
    out_gate_w = out_gate_w.to(dtype_fp16)

    # Concatenate in the exact order needed for later splitting.
    combined_weight = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    )  # [5*hidden_dim, dim]

    # No bias in any of the Linear layers.
    combined = torch.nn.functional.linear(x, combined_weight)  # [B,N,N,5*H], fp16

    # -----------------------------------------------------------------
    # 4️⃣ Split and apply the three sigmoids.
    # -----------------------------------------------------------------
    splits = torch.split(combined, hidden_dim, dim=-1)
    left_proj, left_gate_pre, right_proj, right_gate_pre, out_gate_pre = splits

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)  # [B,N,N,hidden_dim], fp16

    # -----------------------------------------------------------------
    # 5️⃣ Element‑wise fused gating (proj * gate) and optional mask.
    # -----------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    if not nomask:
        # mask shape: [B, N, N] → broadcast over hidden_dim
        mask_fp16 = mask_tensor.to(dtype_fp16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16
    # otherwise left/right are already the gated tensors

    # -----------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction (custom Triton kernel or cuBLAS).
    # -----------------------------------------------------------------
    out = _trimul_contraction(left, right)   # [B,N,N,hidden_dim], fp16

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype_fp16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # -----------------------------------------------------------------
    # 8️⃣ Gating by out_gate and final projection.
    # -----------------------------------------------------------------
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(dtype_fp16)  # shape [dim, hidden_dim]
    out = torch.nn.functional.linear(out, to_out_w)    # [B,N,N,dim], fp16

    # -----------------------------------------------------------------
    # 9️⃣ Cast back to float32 for the harness.
    # -----------------------------------------------------------------
    return out.to(torch.float32)
