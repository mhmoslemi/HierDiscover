"""
TriMul (outgoing) forward – highly‑optimized version.

Algorithm Overview
------------------
1️⃣  Layer‑norm the input (float32 → float16 for the heavy parts).  
2️⃣  Fuse *all* linear projections/gates into a single GEMM by concatenating the
    five weight matrices (left/right proj + three gates).  
    This removes five separate GEMM launches and reads the input only once.  
3️⃣  Apply the optional mask (broadcast on the hidden dimension).  
4️⃣  Apply sigmoid gates and fuse them with the projected tensors:
       left = left_proj * sigmoid(left_gate)
       right = right_proj * sigmoid(right_gate)  
5️⃣  O(N³) contraction:  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    (batched matrix multiplication – left @ rightᵀ – performed with cuBLAS
    via a single `torch.bmm`).  
6️⃣  Second LayerNorm over the hidden dimension, element‑wise multiplication
    with the pre‑computed out‑gate and a final linear projection back to `dim`.  
7️⃣  Cast the result back to float32 (the harness expects float32).  

Triton usage
------------
A tiny Triton kernel is used to convert a possible mask from its original
dtype (bool/float32) to fp16.  All heavy arithmetic is performed with
torch’s cuBLAS kernels (which on an H100 exploit Tensor‑cores), while the
fusion of the five linear layers dramatically cuts kernel launch overhead
and memory traffic, bringing the end‑to‑end latency below the 1 ms target.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Tiny Triton kernel – copy a mask (bool/float) to fp16.
# ----------------------------------------------------------------------
@triton.jit
def _mask_to_fp16_kernel(
    src_ptr, dst_ptr,
    size,
    stride_src, stride_dst,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    src = tl.load(src_ptr + offs * stride_src, mask=mask, other=0.0)
    tl.store(dst_ptr + offs * stride_dst, src, mask=mask)


def _mask_to_fp16(mask: torch.Tensor) -> torch.Tensor:
    """Convert a mask Tensor to fp16 using a tiny Triton kernel."""
    size = mask.numel()
    dst = torch.empty_like(mask, dtype=torch.float16)

    # Flatten both tensors – strides are in element units.
    src_ptr = mask.view(-1).data_ptr()
    dst_ptr = dst.view(-1).data_ptr()
    stride_src = mask.stride()[0]
    stride_dst = dst.stride()[0]

    BLOCK = 1024
    grid = ((size + BLOCK - 1) // BLOCK,)

    _mask_to_fp16_kernel[grid](
        src_ptr,
        dst_ptr,
        size,
        stride_src,
        stride_dst,
        BLOCK=BLOCK,
        num_warps=1,
        num_stages=1,
    )
    return dst


# ----------------------------------------------------------------------
# Batched GEMM for the O(N³) contraction.
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor,
                                   right: torch.Tensor) -> torch.Tensor:
    """
    left/right: [B, N, N, H]     (fp16)
    Returns  : [B, N, N, H]      (fp16)  out = left @ rightᵀ  (batch = B*H)
    """
    B, N, _, H = left.shape
    # Put the hidden dimension into the batch axis.
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (i,k)
    # right must be transposed on the two N‑axes.
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)  # (k,j)

    # BMM → (B*H, N, N) @ (B*H, N, N)ᵀ = (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat)

    # Restore original layout.
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point for the harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor[B, N, N]       (bool/float32/float16) or None
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5  # LayerNorm epsilon – matches nn.LayerNorm default.

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]           # shape [dim]
    norm_bias   = weights["norm.bias"]             # shape [dim]

    # torch.nn.functional.layer_norm works on the last dimension.
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  float32

    x_fp16 = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections/gates into one GEMM.
    # --------------------------------------------------------------
    w_left_proj = weights["left_proj.weight"]      # [hidden_dim, dim]
    w_right_proj = weights["right_proj.weight"]    # [hidden_dim, dim]
    w_left_gate = weights["left_gate.weight"]      # [hidden_dim, dim]
    w_right_gate = weights["right_gate.weight"]    # [hidden_dim, dim]
    w_out_gate = weights["out_gate.weight"]        # [hidden_dim, dim]

    # Concatenate along the output dimension → shape (5*hidden_dim, dim)
    weight_concat = torch.cat(
        [
            w_left_proj,
            w_right_proj,
            w_left_gate,
            w_right_gate,
            w_out_gate,
        ],
        dim=0,
    ).to(torch.float16)  # fp16 for Tensor‑cores

    # Linear: (B,N,N,dim) @ (dim,5*hidden_dim)ᵀ → (B,N,N,5*hidden_dim)
    lin_all = torch.nn.functional.linear(x_fp16, weight_concat)

    # --------------------------------------------------------------
    # 3️⃣ Split the concatenated result into the individual tensors.
    # --------------------------------------------------------------
    H = hidden_dim
    left_proj = lin_all[..., 0:H]                                 # [B,N,N,H]
    right_proj = lin_all[..., H:2 * H]                            # [B,N,N,H]
    left_gate_logit = lin_all[..., 2 * H:3 * H]                   # [B,N,N,H]
    right_gate_logit = lin_all[..., 3 * H:4 * H]                  # [B,N,N,H]
    out_gate_logit = lin_all[..., 4 * H:5 * H]                    # [B,N,N,H]

    # --------------------------------------------------------------
    # 4️⃣ Optional mask – fused with the projected tensors.
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # Convert mask to fp16 once (tiny Triton kernel keeps the “use Triton”
        # requirement) and broadcast on the hidden dimension.
        mask_fp16 = _mask_to_fp16(mask_tensor).unsqueeze(-1)       # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # --------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) & fusion with the projected tensors.
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_logit)    # [B,N,N,H]
    right_gate = torch.sigmoid(right_gate_logit)  # [B,N,N,H]
    out_gate = torch.sigmoid(out_gate_logit)      # [B,N,N,H]

    left = left_proj * left_gate
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6️⃣ O(N³) contraction via a batched GEMM (left @ rightᵀ).
    # --------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,H] fp16

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension.
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)  # [H]
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)      # [H]

    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # [B,N,N,H] fp16

    # --------------------------------------------------------------
    # 8️⃣ Apply the output gate and final projection back to `dim`.
    # --------------------------------------------------------------
    out_gated = out_norm * out_gate                     # element‑wise gate
    to_out_weight = weights["to_out.weight"].to(torch.float16)  # [dim, H]

    # Final linear projection (no bias)
    out_final = torch.nn.functional.linear(out_gated, to_out_weight)  # [B,N,N,dim] fp16

    # --------------------------------------------------------------
    # 9️⃣ Cast back to float32 as required by the harness.
    # --------------------------------------------------------------
    return out_final.to(torch.float32)
