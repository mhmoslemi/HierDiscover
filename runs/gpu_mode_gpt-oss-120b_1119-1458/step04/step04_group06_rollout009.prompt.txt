import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass on H100.

Algorithm
---------
1️⃣  Layer‑norm the full 4‑D input (float32 for numerical stability).
2️⃣  Cast the normalized tensor to fp16 – the heavy part of the kernel works
    exclusively in fp16 (Tensor‑core friendly).
3️⃣  Apply the four linear projections (left/right) and the three gating
    projections (left_gate, right_gate, out_gate).  All weights are
    pre‑converted to fp16; the operations are the PyTorch `linear` and a
    `sigmoid` (both fp16).
4️⃣  If a mask is present, broadcast it on the hidden dimension and multiply
    it with the left/right projections *before* gating (exactly as in the
    reference implementation).
5️⃣  Fuse the gates into the projections with an in‑place element‑wise
    multiplication (no extra temporaries).
6️⃣  Heavy O(N³) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM C = A @ Bᵀ where A and B are reshaped to
   [B·H, N, N] (H = hidden_dim).  The contraction is performed by a custom
   Triton kernel that uses 128×128×64 tiles, 8 warps and 2 pipeline stages.
   The kernel uses fp16 inputs and fp32 accumulation (mixed‑precision MMA) – the
   final result is stored back in fp16.
   For very short sequences (N ≤ 128) we fall back to `torch.einsum` to avoid
   launch overhead.
7️⃣  Second LayerNorm (fp16) → out‑gate (sigmoid) → final linear projection
   back to the original `dim`.
8️⃣  Cast the result back to float32 (the harness expects FP32).

Only the O(N³) contraction is written in Triton; the remaining ops use highly
optimized PyTorch primitives.  The kernel works for the full range of shapes
used in the test matrix (B ∈ {1,2}, N ∈ {128,…,1024}, dim ∈ {128,384},
hidden_dim = 128) and respects optional masking.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : [M, N, K]   (M = B * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
#   A, B, C are fp16; accumulator is fp32.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # program‑size constants
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile on the first N dimension (i)
    pid_n = tl.program_id(1)   # tile on the second N dimension (j)
    pid_z = tl.program_id(2)   # batch * hidden index (M)

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (to keep numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # --------------------------------------------------------------
        # Load A tile   (BLOCK_M × BLOCK_K)  – shape (M_tile, K_tile)
        # --------------------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # --------------------------------------------------------------
        # Load B tile   (BLOCK_K × BLOCK_N)  (transposed) → shape (K_tile, N_tile)
        # --------------------------------------------------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # Mixed‑precision dot: fp16 × fp16 → fp32 accumulation
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store C tile (fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )

# ----------------------------------------------------------------------
# Helper: batched GEMM with transposition (A @ Bᵀ) for the TriMul contraction.
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using the Triton kernel above.
    Arguments
    ---------
    left, right : torch.Tensor [B, N, N, H]   (fp16, contiguous)
    Returns
    -------
    out : torch.Tensor [B, N, N, H] (fp16, contiguous)
    """
    B, N, _, H = left.shape

    # Pack batch & hidden into a single leading dimension: [B·H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (fp16)
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in element counts (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()   # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()  # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()   # (N*N, N, 1)

    # Simple heuristic for tile size – larger tiles for larger N
    if N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Restore original layout [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------
    # 2️⃣ Cast to fp16 for the rest of the pipeline
    # -------------------------------------------------
    x = x.to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Cast all model weights to fp16 on the same device
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # -------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # -------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,hidden]

    # -------------------------------------------------
    # 5️⃣ Optional mask (broadcast on hidden dimension)
    # -------------------------------------------------
    if not nomask:
        # mask is of shape [B, N, N]; make it [B, N, N, 1] to broadcast
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 6️⃣ Gating (sigmoid) – fp16
    # -------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Fuse gate into projection in‑place (less memory traffic)
    left.mul_(left_gate)
    right.mul_(right_gate)

    # -------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via Triton (fp16 in, fp16 out)
    # -------------------------------------------------
    B, N, _, H = left.shape
    if N <= 128:
        # For very short sequences the Triton launch overhead dominates.
        # torch.einsum is a fast fallback that uses highly‑optimized cuBLAS.
        out = torch.einsum('b i k h, b j k h -> b i j h', left, right)
    else:
        # Reshape & launch the custom batched GEMM kernel.
        out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hidden] fp16

    # -------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 9️⃣ Final gating + projection back to dim (fp16)
    # -------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # -------------------------------------------------
    # 🔟 Cast back to float32 (required by the harness)
    # -------------------------------------------------
    return out.to(torch.float32)
