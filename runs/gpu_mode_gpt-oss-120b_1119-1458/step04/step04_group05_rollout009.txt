"""
TriMul (outgoing) – highly‑optimised forward implementation.

The heavy O(N³) contraction

    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]

is performed as a batched matrix‑multiply (A @ Bᵀ) with a custom Triton
kernel.  All linear layers are fused where possible (two calls instead of
five) and the whole pipeline runs in FP16 except the two LayerNorms, which
stay in FP32 for numerical stability.  The final result is returned as
float32.

Author: Triton‑expert rewrite.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A : (M, N, K)   – M = batch * hidden, N = seq_len (i), K = seq_len (k)
#   B : (M, K, N)   – already transposed (k, j)
#   C : (M, N, N)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,   # A strides
    stride_bm, stride_bk, stride_bj,   # B strides
    stride_cm, stride_ci, stride_cj,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_i = tl.program_id(0)   # tile along i (first seq axis)
    pid_j = tl.program_id(1)   # tile along j (second seq axis)
    pid_z = tl.program_id(2)   # batch * hidden index

    # offsets for the i / j tiles
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (to retain precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ----- load A[i,k] -----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # ----- load Bᵀ[k,j] (i.e. B[k,j]) -----
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # dot product (fp16 x fp16 -> fp32) and accumulate
        acc += tl.dot(a, b)

    # write back C[i,j]
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(
    left: torch.Tensor,
    right: torch.Tensor,
    tile: int = 128,
    k_tile: int = 64,
) -> torch.Tensor:
    """
    Batched matrix multiplication using the above Triton kernel.
    `left` and `right` are FP16 tensors of shape [B, N, N, H].
    Returns a FP16 tensor of the same shape (no padding needed – out‑of‑bounds
    accesses are masked inside the kernel).
    """
    B, N, _, H = left.shape
    # pack B*H into the first dimension and transpose `right` to Bᵀ
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)          # (M, N, K)
    B_ = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)       # (M, K, N)

    # empty output container
    C = torch.empty_like(A)

    # strides (in element units)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bk, stride_bj = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    grid = (
        (N + tile - 1) // tile,   # i‑tiles
        (N + tile - 1) // tile,   # j‑tiles
        B * H,                    # batch×hidden
    )
    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bk,
        stride_bj,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=tile,
        BLOCK_N=tile,
        BLOCK_K=k_tile,
        num_warps=8,
        num_stages=3,
    )

    # reshape back to [B, N, N, H]
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul block (gradient‑free).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32), same device as the inputs.
    """
    # ------------------------------------------------------------------
    # 1️⃣ Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2️⃣ First LayerNorm (float32 → FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # LayerNorm over the last dimension (dim)
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32
    x = x_norm.to(torch.float16)  # switch to half‑precision for the rest

    # ------------------------------------------------------------------
    # 3️⃣ Fuse linear layers (project & gates)
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    # flatten to (B·N·N, dim)
    flat = x.view(-1, dim)  # (M, dim)

    # ---- 5 → 2 linear calls -------------------------------------------------
    # left/right projections (2 * hidden_dim outputs)
    proj_weight = torch.cat(
        [
            weights["left_proj.weight"],   # (hidden, dim)
            weights["right_proj.weight"],  # (hidden, dim)
        ],
        dim=0,  # concatenate on the out‑feature axis
    ).to(torch.float16)  # shape (2*hidden, dim)

    proj_out = torch.nn.functional.linear(flat, proj_weight)  # (M, 2*hidden)
    left_proj, right_proj = torch.split(proj_out, hidden_dim, dim=1)
    left_proj = left_proj.view(B, N, N, hidden_dim)
    right_proj = right_proj.view(B, N, N, hidden_dim)

    # gates + out_gate (3 * hidden_dim outputs)
    gate_weight = torch.cat(
        [
            weights["left_gate.weight"],   # (hidden, dim)
            weights["right_gate.weight"],  # (hidden, dim)
            weights["out_gate.weight"],    # (hidden, dim)
        ],
        dim=0,
    ).to(torch.float16)  # shape (3*hidden, dim)

    gate_out = torch.nn.functional.linear(flat, gate_weight)  # (M, 3*hidden)
    left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        gate_out, hidden_dim, dim=1
    )
    left_gate = torch.sigmoid(left_gate_raw).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(right_gate_raw).view(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(out_gate_raw).view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden) – applied *before* gating
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left_proj * mask_fp16
        right = right_proj * mask_fp16
    else:
        left = left_proj
        right = right_proj

    # Apply gating (element‑wise)
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction using the Triton batched GEMM
    # ------------------------------------------------------------------
    out = _batched_matmul_triton(left, right, tile=128, k_tile=64)  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm, output gating and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, hidden)
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim)  FP16

    # ------------------------------------------------------------------
    # 7️⃣ Cast back to FP32 (user‑facing precision)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
