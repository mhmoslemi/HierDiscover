"""
Optimized outgoing TriMul kernel (AlphaFold‑3 “outgoing” variant).

Algorithm
---------
1️⃣  First LayerNorm (float32) on the 4‑D input.
2️⃣  Cast to FP16 – the heavy part of the model runs in half precision.
3️⃣  Fuse the two linear projections (proj & gate) for the left and right side:
    a single matmul with a weight of shape (2·hidden_dim, dim) is performed for
    each side, then the result is split into the projection and the raw gate.
    This reduces the number of dense matmul passes from five to three.
4️⃣  Apply sigmoid to each raw gate.
5️⃣  Optional mask – when a mask is present we fuse ``tensor * gate * mask`` in‑place
    with a tiny Triton kernel.  The kernel broadcasts the mask across the hidden
    dimension by computing the mask index as ``idx // hidden_dim``.
    When there is no mask we simply multiply the tensor and gate in‑place.
6️⃣  Heavy O(N³) contraction:
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is implemented as a batched GEMM (A @ Bᵀ) using torch.bmm – this calls
    cuBLAS Tensor‑Core kernels on H100 and is the fastest choice for all N.
7️⃣  Second LayerNorm (FP16) on the hidden dimension.
8️⃣  Multiply by the out‑gate (sigmoid) and project back to ``dim`` with a final
    Linear.
9️⃣  Cast the result back to float32 for the harness.

The kernel in step 5 guarantees that we use a custom Triton kernel while the
overall runtime is dominated by the fused matmuls and the cuBLAS batched GEMM.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (tensor * gate * mask) in‑place.
# The mask is broadcast over the hidden dimension; we compute the mask index
# by integer dividing the linear offset by hidden_dim.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,            # *fp16 : tensor to be updated, shape (size,)
    gate_ptr,         # *fp16 : gate tensor, same shape as x
    mask_ptr,         # *fp16 : mask tensor of shape (size // hidden_dim,)
    size,             # i32   : total number of elements (B·N·N·hidden)
    hidden_dim,       # i32   : hidden dimension (broadcast factor)
    BLOCK: tl.constexpr,  # compile‑time tile size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)            # linear element indices
    mask = offs < size                                   # range guard

    # compute corresponding mask index (broadcast over hidden_dim)
    mask_offs = offs // hidden_dim

    # load operands
    x = tl.load(x_ptr + offs,      mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs,  mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)  # mask=1 → no effect

    # fused multiplication
    out = x * g * m

    # write result back in‑place
    tl.store(x_ptr + offs, out, mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (float32 or bool)
        - weights_dict : mapping from weight names to torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) – keep the statistics in full precision
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (float32)

    # -----------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy work
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Cast model weights to FP16 once
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Fuse projection + gate for left and right side
    #    (2·hidden_dim outputs → split into proj & gate)
    # -----------------------------------------------------------------
    # left side
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0)          # (2*H, dim)
    left_concat = torch.nn.functional.linear(x, left_combined_w)           # [B,N,N,2*H]
    left_proj, left_gate_raw = torch.split(left_concat, hidden_dim, dim=-1)
    left_proj = left_proj.contiguous()
    left_gate = torch.sigmoid(left_gate_raw).contiguous()

    # right side
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0)      # (2*H, dim)
    right_concat = torch.nn.functional.linear(x, right_combined_w)         # [B,N,N,2*H]
    right_proj, right_gate_raw = torch.split(right_concat, hidden_dim, dim=-1)
    right_proj = right_proj.contiguous()
    right_gate = torch.sigmoid(right_gate_raw).contiguous()

    # out‑gate (no fusion required)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w)).contiguous()

    # -----------------------------------------------------------------
    # 5️⃣ Optional mask – fused element‑wise multiplication via Triton
    # -----------------------------------------------------------------
    if not nomask:
        # mask of shape [B, N, N]; broadcast over hidden_dim inside the kernel
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
        total_elems = left_proj.numel()                     # B·N·N·H
        BLOCK = 1024                                        # ≤1024 threads per block
        grid = ((total_elems + BLOCK - 1) // BLOCK,)

        # left = left_proj * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elems,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total_elems,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # mask absent – simple in‑place multiply with the gate
        left_proj.mul_(left_gate)
        right_proj.mul_(right_gate)

    # -----------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS)
    # -----------------------------------------------------------------
    B, N, _, _ = left_proj.shape
    # reshape to [B·H, N, N] for torch.bmm
    a = left_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    b = right_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # Batched matmul: A @ Bᵀ
    c = torch.bmm(a, b.transpose(1, 2))                     # [B·H, N, N]
    out = c.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Final gating + projection back to dim
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)      # [B,N,N,dim] (FP16)

    # -----------------------------------------------------------------
    # 9️⃣ Cast back to float32 for the external harness
    # -----------------------------------------------------------------
    return out.to(torch.float32)
