```python
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,       # *fp16
    gate_ptr,    # *fp16
    mask_ptr,    # *fp16 (broadcasted)
    hidden: tl.constexpr, # hidden dimension size for broadcasting
    size: tl.int32,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size

    # Load x and gate
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Compute mask index (broadcast across hidden dim)
    mask_idx = offs // hidden
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """Fast outgoing TriMul forward (no gradients).

    Implements the “outgoing” TriMul operator with a fused
    linear‑gate‑mask kernel and a single large projection to limit
    memory traffic. All heavy parts run in fp16; the final result is
    returned in float32.
    """
    # Unpack
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # ---------- 1. First LayerNorm (float32) ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # Cast to fp16 for all subsequent ops
    x = x.to(torch.float16)

    # ---------- 2. Combined projection for proj & gates ----------
    # Build combined weight matrix: [left_proj, left_gate, right_proj,
    # right_gate, out_gate]  – shape (5*hidden_dim, dim)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    B, N, _, _ = x.shape
    M = B * N * N  # number of pairwise positions

    # Linear projection – input flattened to (M, dim)
    x_flat = x.view(M, dim)
    proj = torch.nn.functional.linear(x_flat, combined_weight)  # (M, 5*hidden)

    # Reshape to [B, N, N, 5*hidden]
    proj = proj.view(B, N, N, 5 * hidden_dim)

    # ---------- 3. Slice and apply sigmoid for gates ----------
    # left projection
    left_proj = proj[..., :hidden_dim]
    left_gate = proj[..., hidden_dim : 2 * hidden_dim].sigmoid_()

    # right projection
    right_proj = proj[..., 2 * hidden_dim : 3 * hidden_dim]
    right_gate = proj[..., 3 * hidden_dim : 4 * hidden_dim].sigmoid_()

    # out‑gate
    out_gate = proj[..., 4 * hidden_dim : 5 * hidden_dim].sigmoid_()

    # ---------- 4. Fuse mask, gate and projection (Triton) ----------
    # Prepare mask (flattened). If nomask, use a tensor of ones.
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).flatten()
    else:
        mask_fp16 = torch.ones(M, dtype=torch.float16, device=device)

    # Triton kernel launch parameters
    BLOCK = 1024
    size = left_proj.numel()  # M * hidden_dim
    grid = ((size + BLOCK - 1) // BLOCK,)

    # left = left_proj * left_gate * mask
    mask_gate_fuse_kernel[grid](
        left_proj.data_ptr(),
        left_gate.data_ptr(),
        mask_fp16.data_ptr(),
        hidden_dim,
        size,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # right = right_proj * right_gate * mask
    mask_gate_fuse_kernel[grid](
        right_proj.data_ptr(),
        right_gate.data_ptr(),
        mask_fp16.data_ptr(),
        hidden_dim,
        size,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # left_proj and right_proj now hold the gated, masked tensors
    left = left_proj
    right = right_proj

    # ---------- 5. Heavy O(N³) batched GEMM  ----------
    # Reshape for batched matmul: [B*hidden, N, N]
    left_bmm = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_bmm = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # C = A @ Bᵀ   (out_i_j = Σ_k left_i_k * right_j_k)
    out = torch.bmm(left_bmm, right_bmm.transpose(1, 2))

    # Restore original layout: [B, N, N, hidden]
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- 6. Second LayerNorm ----------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ---------- 7. Gating and final projection ----------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim) fp16

    # ---------- 8. Cast back to float32 ----------
    return out.to(torch.float32)

# -*- coding: utf-8 -*-
"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

Key ideas
---------
1. **Fuse all linear projections** – left/right projections and all three
   gate projections are merged into a single `torch.nn.functional.linear`
   call (`[5·hidden, dim]` weight).  This cuts the number of times the huge
   input `[B, N, N, dim]` has to be read from memory from 5 → 1.
2. **Mask‑gate‑fusion kernel** – after the sigmoid‑gates we fuse
   `proj * gate * mask` (or just `proj * gate` when `nomask=True`)
   with a tiny Triton kernel.  The kernel broadcasts the 2‑D mask over the
   hidden dimension by integer‑division (`offs // hidden`).
3. **Batched GEMM for the O(N³) contraction** – the core contraction
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
   is performed with `torch.bmm` on tensors reshaped to `[B·H, N, N]`.  This
   invokes cuBLAS Tensor‑Core kernels on the H100, which are far faster than
   a naïve Triton matmul for the large sequence lengths used in the tests.
4. All heavy math is carried out in **fp16**; only the first LayerNorm is
   done in fp32 for numerical stability.  The final result is cast back to
   fp32 as required by the harness.

The function `custom_kernel` follows the signature requested by the
evaluation harness.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused multiplication of (tensor * gate * mask).
# Mask is a 1‑D tensor of length M = B·N·N and is broadcast over the
# hidden dimension H (passed as a compile‑time constant).  When no mask
# is required the caller supplies a tensor of ones, which makes the
# kernel perform a simple element‑wise product.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 – tensor to be overwritten (proj)
    gate_ptr,       # *fp16 – sigmoid‑gate tensor (same shape as x)
    mask_ptr,       # *fp16 – flattened mask of shape (M,)
    hidden: tl.constexpr,  # hidden dimension (broadcast factor)
    size: tl.int32, # total number of elements (M * hidden)
    BLOCK: tl.constexpr,   # tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    mask = offs < size                                 # out‑of‑range guard

    # Load the two operands
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Broadcast the mask: each block of `hidden` entries shares the same
    # mask element.  Integer division works because `hidden` is a
    # compile‑time constant.
    mask_idx = offs // hidden
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # 1.0 == "no mask"

    # Fused product
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Entry point used by the benchmark harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor   [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor   [B, N, N]   (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32, bias=False)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    device = inp.device
    B, N, _, _ = inp.shape
    M = B * N * N                     # number of pairwise positions

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – stable for the initial distribution
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(inp, (dim,), weight=norm_w, bias=norm_b)

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part of the model
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Combined projection for left/right and all three gates
    #               0 … hidden‑1 : left_proj
    #           hidden … 2·hidden‑1 : left_gate
    #       2·hidden … 3·hidden‑1 : right_proj
    #       3·hidden … 4·hidden‑1 : right_gate
    #       4·hidden … 5·hidden‑1 : out_gate
    # -----------------------------------------------------------------
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                       # (5·hidden, dim)

    # Flatten [B,N,N,dim] → (M, dim) and apply the big linear
    x_flat = x.view(M, dim)                    # (M, dim)
    proj = torch.nn.functional.linear(x_flat, combined_weight)   # (M,5·hidden)
    proj = proj.view(B, N, N, 5 * hidden)      # back to 4‑D

    # -----------------------------------------------------------------
    # 4️⃣ Slice and apply sigmoid to gate portions (in‑place)
    # -----------------------------------------------------------------
    left_proj = proj[..., :hidden]                           # (B,N,N,hidden)
    left_gate = proj[..., hidden:2 * hidden].sigmoid_()      # in‑place sigmoid

    right_proj = proj[..., 2 * hidden:3 * hidden]            # (B,N,N,hidden)
    right_gate = proj[..., 3 * hidden:4 * hidden].sigmoid_() # in‑place sigmoid

    out_gate = proj[..., 4 * hidden:5 * hidden].sigmoid_()  # (B,N,N,hidden)

    # -----------------------------------------------------------------
    # 5️⃣ Fuse mask, gate and projection (Triton)
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(torch.float16).flatten()   # (M,)
    else:
        mask_fp16 = torch.ones(M, dtype=torch.float16, device=device)  # dummy mask

    BLOCK = 1024
    total_elems = left_proj.numel()                     # M * hidden
    grid = ((total_elems + BLOCK - 1) // BLOCK,)

    # left = left_proj * left_gate * mask
    mask_gate_fuse_kernel[grid](
        left_proj.data_ptr(),
        left_gate.data_ptr(),
        mask_fp16.data_ptr(),
        hidden,
        total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # right = right_proj * right_gate * mask
    mask_gate_fuse_kernel[grid](
        right_proj.data_ptr(),
        right_gate.data_ptr(),
        mask_fp16.data_ptr(),
        hidden,
        total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )
    left = left_proj   # now contain the gated, masked tensor
    right = right_proj

    # -----------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction – batched GEMM (cuBLAS)
    # -----------------------------------------------------------------
    # reshape to [B·hidden, N, N] for torch.bmm
    left_bmm = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_bmm = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out = torch.bmm(left_bmm, right_bmm.transpose(1, 2))   # (B·hidden, N, N)

    # restore (B,N,N,hidden) layout
    out = out.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16) → out‑gate → final projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate                                 # gating
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)      # back to [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 8️⃣ Cast back to float32 for the harness
    # -----------------------------------------------------------------
    return out.to(torch.float32)
