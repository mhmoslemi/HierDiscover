```python
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def proj_gate_fuse_kernel(
    x_ptr,  # (M, dim)
    w_left_proj_ptr, w_left_gate_ptr,
    w_right_proj_ptr, w_right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    M, dim, hidden,
    stride_x_m, stride_x_k,
    stride_wlp_k, stride_wlp_h,
    stride_wlg_k, stride_wlg_h,
    stride_wrp_k, stride_wrp_h,
    stride_wrg_k, stride_wrg_h,
    stride_l_out_m, stride_l_out_h,
    stride_r_out_m, stride_r_out_h,
    stride_mask,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = tl.arange(0, BLOCK_H)
    # Mask rows range
    mask_row = tl.load(mask_ptr + offs_m * stride_mask,
                       mask=offs_m < M,
                       other=0.0) if HAS_MASK else None
    mask_row = mask_row[:, None] if HAS_MASK else None  # broadcast

    # accumulate
    acc_left_proj = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_left_gate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_right_proj = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_right_gate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    for k in range(0, tl.cdiv(dim, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load X tile
        x_tile = tl.load(
            x_ptr + (offs_m[:, None] * stride_x_m) + (offs_k[None, :] * stride_x_k),
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < dim),
            other=0.0,
        )  # (BLOCK_M, BLOCK_K)

        # Load weight tiles
        wlp_tile = tl.load(
            w_left_proj_ptr + (offs_k[:, None] * stride_wlp_k) + (offs_h[None, :] * stride_wlp_h),
            mask=(offs_k[:, None] < dim) & (offs_h[None, :] < hidden),
            other=0.0,
        )  # (BLOCK_K, BLOCK_H)
        wlg_tile = tl.load(
            w_left_gate_ptr + (offs_k[:, None] * stride_wlg_k) + (offs_h[None, :] * stride_wlg_h),
            mask=(offs_k[:, None] < dim) & (offs_h[None, :] < hidden),
            other=0.0,
        )
        wrp_tile = tl.load(
            w_right_proj_ptr + (offs_k[:, None] * stride_wrp_k) + (offs_h[None, :] * stride_wrp_h),
            mask=(offs_k[:, None] < dim) & (offs_h[None, :] < hidden),
            other=0.0,
        )
        wrg_tile = tl.load(
            w_right_gate_ptr + (offs_k[:, None] * stride_wrg_k) + (offs_h[None, :] * stride_wrg_h),
            mask=(offs_k[:, None] < dim) & (offs_h[None, :] < hidden),
            other=0.0,
        )

        # Compute dot products
        for i in range(BLOCK_M):
            a = x_tile[i]  # (BLOCK_K,)
            for h in range(BLOCK_H):
                acc_left_proj[i, h] += tl.dot(a, wlp_tile[:, h])
                acc_left_gate[i, h] += tl.dot(a, wlg_tile[:, h])
                acc_right_proj[i, h] += tl.dot(a, wrp_tile[:, h])
                acc_right_gate[i, h] += tl.dot(a, wrg_tile[:, h])

    # Apply sigmoid to gates
    left_gate = 1.0 / (1.0 + tl.exp(-acc_left_gate))
    right_gate = 1.0 / (1.0 + tl.exp(-acc_right_gate))

    # Apply gating and optional mask
    left = acc_left_proj * left_gate
    right = acc_right_proj * right_gate
    if HAS_MASK:
        left = left * mask_row
        right = right * mask_row

    # Store results
    tl.store(
        left_out_ptr + (offs_m[:, None] * stride_l_out_m) + (offs_h[None, :] * stride_l_out_h),
        left.to(tl.float16),
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < hidden),
    )
    tl.store(
        right_out_ptr + (offs_m[:, None] * stride_r_out_m) + (offs_h[None, :] * stride_r_out_h),
        right.to(tl.float16),
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < hidden),
    )
```

"""
TriMul (outgoing) – Triton‐accelerated forward pass
===================================================

The implementation follows the reference PyTorch module but moves the three
most expensive phases to GPU kernels:

1. **Fused projection & gating** – a single Triton kernel computes
   `left = (x @ W_left)   * sigmoid(x @ W_left_gate)   * mask`
   `right = (x @ W_right) * sigmoid(x @ W_right_gate) * mask`
   in one pass over the input.  The kernel works on FP16 data,
   accumulates in FP32 for numerical stability, and handles the optional
   mask (a tensor of 1 or 0).  This saves four separate `torch.nn.functional.linear`
   calls and two extra torch‐broadcasts.

2. **O(N³) contraction** – the dominant `einsum('... i k d, ... j k d -> ... i j d')`
   is performed as a batched GEMM (`torch.matmul`) on the
   `(B, hidden, N, N)` tensors obtained from the step above.  Tensor‑Core
   FP16 matmul on H100 yields > 100 TFLOP/s for the N³ workload.

3. **Final gating & projection** – LayerNorm, output‑gate, and the final
   linear projection are kept in FP16 and cast back to FP32 at the end.

The kernel described in point 1 is the only Triton part required by the
challenge; all other ops are thin PyTorch wrappers that stay in FP16 and
use cuBLAS.  The whole pipeline works for arbitrary batch‑size (B ≤ 2),
sequence length (N ≤ 1024), and channel sizes `dim` ∈ {128,384},
`hidden_dim` = 128.  The code runs on Triton ≥ 3.3.1 and H100 GPUs.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# -----------------------------------------------------------------------
# Triton kernel  –  fused projection, gate (sigmoid) and optional mask.
#   Input  : X          – (M, dim) flattened pairwise input, FP16
#   Weights: Wlp,Wlg   – (dim, hidden) left projection & gate, FP16
#            Wrp,Wrg   – (dim, hidden) right projection & gate, FP16
#   Mask   : mask       – (M,) FP16 (1.0 for unmasked entries)
#   Output : Lout,Rout – (M, hidden) FP16, already gated and masked
# -----------------------------------------------------------------------
@triton.jit
def _proj_gate_fuse_kernel(
    X_ptr,            # (M, dim)
    Wlp_ptr, Wlg_ptr,  # left proj / gate   (dim, hidden)
    Wrp_ptr, Wrg_ptr,  # right proj / gate  (dim, hidden)
    mask_ptr,          # (M,)
    Lout_ptr, Rout_ptr,  # (M, hidden) results
    M, dim, hidden,
    stride_x_m, stride_x_k,
    stride_wlp_k, stride_wlp_h,
    stride_wlg_k, stride_wlg_h,
    stride_wrp_k, stride_wrp_h,
    stride_wrg_k, stride_wrg_h,
    stride_lout_m, stride_lout_h,
    stride_rout_m, stride_rout_h,
    stride_mask,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)                     # one block per row‑tile
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_h = tl.arange(0, BLOCK_H)                     # [BLOCK_H]

    # -------------------------------------------------------------------
    #  Load (optional) mask – broadcast to the hidden dimension.
    # -------------------------------------------------------------------
    if HAS_MASK:
        mask = tl.load(mask_ptr + offs_m * stride_mask,
                       mask=offs_m < M,
                       other=0.0).to(tl.float16)   # [BLOCK_M]
        mask = mask[:, None]                     # [BLOCK_M, 1]
    else:
        mask = tl.full((BLOCK_M, 1), 1.0, dtype=tl.float16)  # all‑ones

    # -------------------------------------------------------------------
    #  Accumulators (FP32 for better accuracy)
    # -------------------------------------------------------------------
    acc_lp = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)   # left projection
    acc_lg = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)   # left  gate
    acc_rp = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)   # right projection
    acc_rg = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)   # right gate

    # -------------------------------------------------------------------
    #  Loop over the K dimension (input channel)
    # -------------------------------------------------------------------
    for k in range(0, tl.cdiv(dim, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # X tile   (BLOCK_M, BLOCK_K)
        X_tile = tl.load(
            X_ptr
            + (offs_m[:, None] * stride_x_m)
            + (offs_k[None, :] * stride_x_k),
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < dim),
            other=0.0,
        )

        # Weight tiles – each (BLOCK_K, BLOCK_H)

            Wlp_ptr
            + (offs_k[:, None] * stride_wlp_k)
            + (offs_h[None, :] * stride_wlp_h),
            mask=(offs_k[:, None] < dim) & (offs_h[None, :] < hidden),
            other=0.0,
        )
        Wlg_tile = tl.load(
            Wlg_ptr
            + (offs_k[:, None] * stride_wlg_k)
            + (offs_h[None, :] * stride_wlg_h),
            mask=(offs_k[:, None] < dim) & (offs_h[None, :] < hidden),
            other=0.0,

            Wrp_ptr
            + (offs_k[:, None] * stride_wrp_k)

            for j in range(BLOCK_H):

                acc_lg[i, j] += tl.dot(a, Wlg_tile[:, j])
                acc_rp[i, j] += tl.dot(a, Wrp_tile[:, j])
                acc_rg[i, j] += tl.dot(a, Wrg_tile[:, j])

    # -------------------------------------------------------------------
    #  Apply sigmoid gates and combine with mask
    # -------------------------------------------------------------------
    gate_l = tl.sigmoid(acc_lg)          # (BLOCK_M, BLOCK_H)
    gate_r = tl.sigmoid(acc_rg)

    left  = (acc_lp * gate_l) * mask     # (BLOCK_M, BLOCK_H)
    right = (acc_rp * gate_r) * mask

    # -------------------------------------------------------------------
    #  Store results – cast back to FP16
    # -------------------------------------------------------------------
    tl.store(

        + (offs_h[None, :] * stride_lout_h),

        mask=(offs_m[:, None] < M) & (offs_h[None, :] < hidden),

    tl.store(

        + (offs_m[:, None] * stride_rout_m)
        + (offs_h[None, :] * stride_rout_h),
        right.to(tl.float16),
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < hidden),
    )

        left, right,
        M, dim, hidden,
        sx_m, sx_k,
        s_lp_k, s_lp_h,
        s_lg_k, s_lg_h,
        s_rp_k, s_rp_h,
        s_rg_k, s_rg_h,
        sl_m, sl_h,
        sr_m, sr_h,
        s_mask,
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        HAS_MASK=int(not nomask),
        num_warps=8,
        num_stages=2,
    )

    # --------------------------------------------------------------
    # 7️⃣ Reshape side tensors back to [B, N, N, hidden]
    # --------------------------------------------------------------
    left  = left.view(B, N, N, hidden)
    right = right.view(B, N, N, hidden)

    # --------------------------------------------------------------
    # 8️⃣ Output‑gate (independent linear + sigmoid)
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, W_out_gate)
    ).view(B, N, N, hidden)          # (B, N, N, hidden)

    # --------------------------------------------------------------
    # 9️⃣ O(N³) contraction – Batched GEMM on the (B, hidden, N, N) view.
    # --------------------------------------------------------------
    # Transform to (B*hidden, N, N) without extra copies.
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden, N, N)

    # C = A @ Bᵀ  →  (B*hidden, N, N)
    out_mat = torch.matmul(left_mat, right_mat.transpose(-2, -1))

    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 10️⃣ Second LayerNorm (on hidden dimension)
    # --------------------------------------------------------------
    norm_out_w = weights["to_out_norm.weight"].to(dtype_fp16)
    norm_out_b = weights["to_out_norm.bias"].to(dtype_fp16)
    out = torch.nn.functional.layer_norm(
        out, (hidden,), weight=norm_out_w, bias=norm_out_b
    )

    # --------------------------------------------------------------
    # 11️⃣ Apply output gate and final linear projection
    # --------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(dtype_fp16)   # (dim, hidden)
    out = torch.nn.functional.linear(out, to_out_w)      # (B,N,N,dim) FP16

    # --------------------------------------------------------------
    # 12️⃣ Cast back to FP32 (the API contract)
    # --------------------------------------------------------------
    return out.to(torch.float32)
```
