"""
Outgoing TriMul (AlphaFold‑3) – highly‑optimized forward pass.

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (float32) → cast to fp16.
2️⃣  Four linear projections (left/right value & left/right gate) are fused
    into a single mat‑mul to reduce kernel launches.
3️⃣  Optional mask is applied (broadcast on the hidden dimension).
    Gating uses sigmoid and is fused with the value projections.
4️⃣  Heavy contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is performed as a batched GEMM (A @ Bᵀ) using `torch.bmm`,
    which on H100 exploits tensor‑cores (fp16×fp16 → fp32 accumulation).
    A small Triton kernel is provided for completeness (and can be used for
    very short sequences), but the default fast path is the cuBLAS batched GEMM.
5️⃣  Second LayerNorm (fp16) → element‑wise out‑gate (sigmoid) → final
    linear projection back to the original `dim`.
6️⃣  Cast the final result back to float32.

Only the forward pass is required – no autograd graph is built.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel (optional fast path for tiny N).  It computes
#   C = A @ Bᵀ   for A,B of shape [M, N, K] → C of shape [M, N, N].
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # i‑tile index
    pid_n = tl.program_id(1)   # j‑tile index
    pid_z = tl.program_id(2)   # batch‑*hidden index

    # Offsets for the i/j tiles
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator (fp32) for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # Main K‑loop
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A[i,k] for the current hidden slice `pid_z`
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )
        # Load B[j,k] (transposed view) for the same hidden slice
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )
        # Mixed‑precision dot (fp16×fp16 → fp32)
        acc += tl.dot(a, b)

    # Store C = acc (cast back to fp16)
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right (both [B, N, N, H]) via a batched GEMM
    `C = left @ rightᵀ`.  All tensors are fp16; accumulation is fp32.
    Returns a tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    # reshape to [M, N, N] where M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched mat‑mul with transposed second operand
    C = torch.bmm(A, B_.transpose(1, 2))
    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor   : torch.Tensor [B, N, N]       (float32 / bool)
        - weights_dict   : mapping name → torch.Tensor (float32)
        - config_dict    : must contain ``dim`` and ``hidden_dim``.
                          Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack & configuration
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # 2️⃣ first LayerNorm (float32) – keep numerical stability
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] (float32)

    # ------------------------------------------------------------------
    # 3️⃣ cast to fp16 for the rest of the pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ linear projections – fuse the four projections into a single
    #    mat‑mul to reduce kernel launch overhead.
    # ------------------------------------------------------------------
    # Gather the four weight matrices (all bias‑free) and cast them once.
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)

    # Concatenate along the output dimension → shape (4*H, dim)
    concat_proj_gate_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    )  # (4*hidden_dim, dim)

    # Single fused linear: [B, N, N, 4*H]
    proj_gate = F.linear(x, concat_proj_gate_w)

    # Slice the four components
    H = hidden_dim
    left_proj   = proj_gate[..., :H]          # shape [..., H]
    left_gate   = proj_gate[..., H:2 * H]
    right_proj  = proj_gate[..., 2 * H:3 * H]
    right_gate  = proj_gate[..., 3 * H:]

    # ------------------------------------------------------------------
    # 5️⃣ optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ gating (sigmoid) and combine with value projections
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate     # [B, N, N, H]
    right = right_proj * right_gate  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7️⃣ heavy O(N³) contraction – batched GEMM (A @ Bᵀ)
    # ------------------------------------------------------------------
    # The massive contraction dominates runtime; cuBLAS batched GEMM
    # (torch.bmm) is very fast on H100 and works for all sequence lengths.
    # For completeness we keep a Triton fallback for tiny N (≤128), but
    # in practice torch.bmm is the fastest path.
    seq_len = input_tensor.shape[1]
    if seq_len <= 128:
        # tiny sequences: use the handcrafted Triton kernel
        B, N, _, H = left.shape
        # reshape to [M, N, N] where M = B*H
        A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
        B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

        C = torch.empty_like(A)  # fp16 output placeholder

        # strides (in elements)
        stride_am, stride_ai, stride_ak = A.stride()
        stride_bm, stride_bj, stride_bk = B_.stride()
        stride_cm, stride_ci, stride_cj = C.stride()

        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64
        grid = (
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
            B * H,
        )
        _trimul_matmul_kernel[grid](
            A,
            B_,
            C,
            B * H,
            N,
            N,
            stride_am,
            stride_ai,
            stride_ak,
            stride_bm,
            stride_bj,
            stride_bk,
            stride_cm,
            stride_ci,
            stride_cj,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=8,
            num_stages=2,
        )
        out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    else:
        # fast path – cuBLAS batched GEMM
        out = _batched_matmul_torch(left, right)  # [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 8️⃣ out‑gate linear + second LayerNorm
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))  # [B,N,N,H] fp16

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ final projection back to original dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)          # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 🔟 cast back to float32 (required by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
