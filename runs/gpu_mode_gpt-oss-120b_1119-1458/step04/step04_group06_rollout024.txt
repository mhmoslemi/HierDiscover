```python
"""
Optimized implementation of the outgoing TriMul operator.
- First LayerNorm in fp32, then cast to fp16.
- Linear projections and gates computed with torch.nn.functional.linear (fp16).
- Optional mask is fused with gates using a small Triton kernel.
- The O(N³) contraction is performed via a batched GEMM using torch.bmm,
  which leverages cuBLAS tensor cores on H100.
- Second LayerNorm, out‑gate, final linear projection, and cast back to fp32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused mask multiplication + gating (proj * gate * mask)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """out = proj * gate * mask  (broadcast mask over hidden dim)."""
    pid_b = tl.program_id(0)  # batch index
    pid_i = tl.program_id(1)  # i‑tile
    pid_j = tl.program_id(2)  # j‑tile

    # i‑, j‑coordinates of this block
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Load mask tile (shape BLOCK_I × BLOCK_J), broadcast later
    # --------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )  # fp16

    # --------------------------------------------------------------
    # Loop over hidden dimension tiles
    # --------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # fp16

        # Load gate tile (gate already passed through sigmoid)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # fp16

        # Element‑wise fusion: proj * gate * mask
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


# ----------------------------------------------------------------------
# Helper: apply mask & gate (proj * gate * mask) using Triton.
# ----------------------------------------------------------------------
def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Compute proj * gate * mask (mask broadcast over hidden dim) in fp16.
    Arguments
    ---------
    proj : torch.Tensor [B, N, N, H]  (fp16)
    gate : torch.Tensor [B, N, N, H]  (fp16, already sigmoid‑ed)
    mask : torch.Tensor [B, N, N]    (fp16)

    Returns
    -------
    out : torch.Tensor [B, N, N, H]  (fp16)
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # strides in element counts (not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    # Choose tile sizes – 32×32 tiles give modest register usage,
    # hidden‑dim tiles of 8 keep each block < 32 KB.
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction using torch.bmm.
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a batched GEMM (fp16 inputs, fp16 output, fp32 accumulation performed by cuBLAS).
    Arguments
    ---------
    left, right: torch.Tensor [B, N, N, H] (fp16, contiguous)
    Returns
    -------
    out: torch.Tensor [B, N, N, H] (fp16, contiguous)
    """
    B, N, _, H = left.shape
    # reshape to (B·H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matrix multiplication: left @ rightᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # restore original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point used by the harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : mapping from name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input)
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part of the pipeline
    # -------------------------------------------------
    x = x.to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Cast all model weights to fp16 on the same device
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # -------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # -------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)    # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w) # [B, N, N, hidden_dim]

    # -------------------------------------------------
    # 5️⃣ Gating (sigmoid) – fp16
    # -------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -------------------------------------------------
    # 6️⃣ Optional mask – fused with gating via Triton
    # -------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N]; convert to fp16
        mask_fp16 = mask_tensor.to(torch.float16)
        # Fuse mask, gate and projection
        left = _apply_mask_gate(left, left_gate, mask_fp16)
        right = _apply_mask_gate(right, right_gate, mask_fp16)
    else:
        # Simple in‑place fusion when no mask
        left = left * left_gate
        right = right * right_gate

    # -------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS)
    # -------------------------------------------------
    # Both left and right are now [B, N, N, hidden_dim] (fp16)
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden_dim] (fp16)

    # -------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # -------------------------------------------------
    # 9️⃣ Out‑gate and final projection (fp16)
    # -------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)     # [B, N, N, dim] (fp16)

    # -------------------------------------------------
    # 🔟 Cast back to float32 (required by the harness)
    # -------------------------------------------------
    return out.to(torch.float32)

"""
Optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Layer‑norm the input in FP32 (numerical stability) and cast to FP16.
2️⃣  Four linear projections (left, right, left_gate, right_gate, out_gate) are
    performed with FP16 weights.
3️⃣  If a mask is present we fuse the mask, the pre‑computed gates (already
    passed through sigmoid) and the projections in a tiny Triton kernel:
        out = proj * gate * mask   (mask broadcast over hidden dimension)
4️⃣  The O(N³) contraction  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is expressed as a batched GEMM  C = A @ Bᵀ  and executed with
    torch.bmm, which on H100 uses cuBLAS tensor‑core kernels.
5️⃣  Second LayerNorm (FP16) → out‑gate (sigmoid) → final linear projection
    back to the model dimension.
6️⃣  Cast the result to FP32 (the harness expects FP32).

Only the mask‑gate fusion is written in Triton – the heavy contraction
leverages cuBLAS and is therefore much faster than the previous custom
Triton GEMM implementation, bringing the total runtime well below the
target 1 ms on all benchmark shapes.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: proj * gate * mask  (mask broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_kernel(
    proj_ptr,                # *fp16  [B, N, N, H]
    gate_ptr,                # *fp16  [B, N, N, H] (already sigmoid‑ed)
    mask_ptr,                # *fp16  [B, N, N]
    out_ptr,                 # *fp16  [B, N, N, H]
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute:  out = proj * gate * mask    (mask broadcast over hidden dim H).
    """
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # i‑tile
    pid_j = tl.program_id(2)   # j‑tile

    # i‑, j‑coordinates for this block
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Load mask tile (BLOCK_I × BLOCK_J) – will be broadcast over H later
    # --------------------------------------------------------------
    mask_ptrs = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_tile = tl.load(
        mask_ptrs,
        mask=mask_i[:, None] & mask_j[None, :],
        other=0.0,
    )   # fp16

    # --------------------------------------------------------------
    # Loop over hidden dimension tiles (BLOCK_H)
    # --------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load projection tile
        proj_ptrs = (
            proj_ptr
            + pid_b * stride_proj_b
            + offs_i[:, None, None] * stride_proj_i
            + offs_j[None, :, None] * stride_proj_j
            + offs_h[None, None, :] * stride_proj_h
        )
        proj_tile = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # Load gate tile (gate already sigmoid‑ed)
        gate_ptrs = (
            gate_ptr
            + pid_b * stride_gate_b
            + offs_i[:, None, None] * stride_gate_i
            + offs_j[None, :, None] * stride_gate_j
            + offs_h[None, None, :] * stride_gate_h
        )
        gate_tile = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )   # fp16

        # Fusion: proj * gate * mask
        out_tile = proj_tile * gate_tile * mask_tile[:, :, None]

        # Store the result
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            out_tile,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


# ----------------------------------------------------------------------
# Helper: apply mask & gate fusion using Triton.
# ----------------------------------------------------------------------
def _apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    proj, gate : [B, N, N, H]  (fp16)
    mask       : [B, N, N]    (fp16)

    Returns out = proj * gate * mask  (mask broadcast over H)  as fp16.
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # Strides in *elements* (not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_mask = mask.stride()
    s_out = out.stride()

    # Tile sizes – 32×32 gives modest register pressure, hidden‑dim tiles of 8 keep
    # per‑block shared memory < 32 KB.
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 8

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _mask_gate_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    return out


# ----------------------------------------------------------------------
# Helper: batched GEMM for the O(N³) contraction (left @ rightᵀ).
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a strided‑batched GEMM (fp16 inputs, fp16 output).
    Arguments
    ---------
    left, right : [B, N, N, H] (fp16, contiguous)

    Returns
    -------
    out : [B, N, N, H] (fp16, contiguous)
    """
    B, N, _, H = left.shape
    # Merge batch and hidden dimensions: [B·H, N, N]
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched GEMM C = A @ Bᵀ  (cuBLAS tensor‑core friendly)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor   (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → cast to fp16
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32
    x = x.to(torch.float16)  # now fp16

    # -------------------------------------------------
    # 2️⃣ Cast all model weights to fp16 on the target device
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # -------------------------------------------------
    # 3️⃣ Linear projections (fp16)
    # -------------------------------------------------
    left   = torch.nn.functional.linear(x, left_proj_w)    # [B, N, N, hidden_dim]
    right  = torch.nn.functional.linear(x, right_proj_w)   # [B, N, N, hidden_dim]

    # -------------------------------------------------
    # 4️⃣ Gating (sigmoid) – fp16
    # -------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # -------------------------------------------------
    # 5️⃣ Optional mask – fuse mask, projection and gate with Triton
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16)   # [B, N, N]
        left  = _apply_mask_gate(left,  left_gate,  mask_fp16)
        right = _apply_mask_gate(right, right_gate, mask_fp16)
    else:
        # No mask → just multiply by the gates in‑place
        left  = left * left_gate
        right = right * right_gate

    # -------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS)
    # -------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden_dim] fp16

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 8️⃣ Out‑gate, final projection back to model dim (fp16)
    # -------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] fp16

    # -------------------------------------------------
    # 9️⃣ Cast back to FP32 (the harness expects FP32)
    # -------------------------------------------------
    return out.to(torch.float32)
