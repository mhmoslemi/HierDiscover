```
@triton.jit
def mul_gate_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    left_out_ptr, right_out_ptr,
    n_elements,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements
    left_proj = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    left_out = left_proj * left_gate
    tl.store(left_out_ptr + offs, left_out, mask=mask)

    right_proj = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    right_out = right_proj * right_gate
    tl.store(right_out_ptr + offs, right_out, mask=mask)

"""
TriMul (outgoing) – fast forward pass.

This implementation follows the reference PyTorch module but
* uses FP16 for all heavy arithmetic,
* fuses each projection+gate pair into a single linear layer,
* performs the O(N³) contraction with a cuBLAS‑backed batched GEMM
  (torch.bmm) – the fastest path on an H100,
* applies the element‑wise projection‑gate fusion with a tiny Triton kernel
  (required by the problem statement).

The overall algorithm:
1️⃣ Layer‑norm the 4‑D input tensor.
2️⃣ Flatten to (B·N·N, dim) and compute for each position:
   – left_proj + left_gate   (concatenated weight → split & sigmoid)
   – right_proj + right_gate (concatenated weight → split & sigmoid)
   – out_gate                (single linear + sigmoid)
3️⃣ Fuse the element‑wise products left=proj*gate and right=proj*gate
   with a Triton kernel.
4️⃣ Optional mask (broadcast over the hidden dimension).
5️⃣ Perform the heavy contraction using a batched matrix‑multiply:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as torch.bmm on tensors of shape (B·H, N, N).
6️⃣ Apply a second Layer‑Norm on the hidden dimension, multiply by out_gate
   and finish with a linear projection back to `dim`.
7️⃣ Return the result in float32 (the original forward expects FP32).

All weight tensors are cast once to FP16; the only Triton code is the
element‑wise “gate” kernel.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication of projection and gate.
#   out_left = left_proj * left_gate
#   out_right = right_proj * right_gate
# ----------------------------------------------------------------------
@triton.jit
def _gate_fusion_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    left_out_ptr, right_out_ptr,
    n_elements,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    # left side
    lp = tl.load(left_proj_ptr + offs, mask=mask, other=tl.float16(0.0))
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    lo = lp * lg
    tl.store(left_out_ptr + offs, lo, mask=mask)

    # right side
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=tl.float16(0.0))
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    ro = rp * rg
    tl.store(right_out_ptr + offs, ro, mask=mask)


def _apply_gate_fusion(
    left_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_proj: torch.Tensor,
    right_gate: torch.Tensor,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Execute the `_gate_fusion_kernel` on a contiguous view of the tensors.
    All tensors are expected to be FP16 and of the same shape
    (B,N,N,hidden_dim). Returns the gated tensors in the same layout.
    """
    assert left_proj.shape == left_gate.shape == right_proj.shape == right_gate.shape
    total = left_proj.numel()
    # flatten to 1‑D (contiguous) tensors
    lp_flat = left_proj.view(-1)
    lg_flat = left_gate.view(-1)
    rp_flat = right_proj.view(-1)
    rg_flat = right_gate.view(-1)

    left_out = torch.empty_like(lp_flat)
    right_out = torch.empty_like(rp_flat)

    BLOCK = 8192  # enough threads per block for this simple kernel
    grid = ( (total + BLOCK - 1) // BLOCK, )
    _gate_fusion_kernel[grid](
        lp_flat,
        lg_flat,
        rp_flat,
        rg_flat,
        left_out,
        right_out,
        total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    # reshape back
    left = left_out.view_as(left_proj)
    right = right_out.view_as(right_proj)
    return left, right


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (gradient‑free).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # 1️⃣ Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2️⃣ First LayerNorm (still in FP32 for numerical stability)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    # switch to FP16 for the heavy parts
    x = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Cast all model weights to FP16 once
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (hidden_dim, dim)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (hidden_dim, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16) # (hidden_dim, dim)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16) # (hidden_dim, dim)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)   # (hidden_dim, dim)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)  # (hidden_dim,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)    # (hidden_dim,)
    to_out_w      = weights["to_out.weight"].to(torch.float16)       # (dim, hidden_dim)

    # --------------------------------------------------------------
    # 4️⃣ Fuse each projection+gate pair via a single linear call
    # --------------------------------------------------------------
    B, N, _, _ = x.shape
    flat = x.view(-1, dim)          # (B·N·N, dim)

    # left_proj + left_gate together
    left_concat_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2*hidden_dim, dim)
    left_concat = torch.nn.functional.linear(flat, left_concat_w)  # (B·N·N, 2*hidden_dim)
    left_proj, left_gate_pre = left_concat.split([hidden_dim, hidden_dim], dim=1)
    left_gate = torch.sigmoid(left_gate_pre)

    # right_proj + right_gate together
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*hidden_dim, dim)
    right_concat = torch.nn.functional.linear(flat, right_concat_w)  # (B·N·N, 2*hidden_dim)
    right_proj, right_gate_pre = right_concat.split([hidden_dim, hidden_dim], dim=1)
    right_gate = torch.sigmoid(right_gate_pre)

    # reshape back to 4‑D tensors (FP16)
    left_proj = left_proj.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)
    right_proj = right_proj.view(B, N, N, hidden_dim)
    right_gate = right_gate.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5️⃣ Fuse element‑wise multiplication with a tiny Triton kernel
    # --------------------------------------------------------------
    left, right = _apply_gate_fusion(
        left_proj, left_gate, right_proj, right_gate
    )

    # free memory for un‑needed tensors
    del left_proj, left_gate, right_proj, right_gate, left_concat, right_concat

    # --------------------------------------------------------------
    # 6️⃣ Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction – batched matrix multiplication
    # --------------------------------------------------------------
    # reshape for batched GEMM: (B·hidden_dim, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # (i,k)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N) # (k,j)
    out_mat = torch.bmm(left_mat, right_mat)  # (B·hidden_dim, N, N)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,hidden_dim)

    # --------------------------------------------------------------
    # 8️⃣ Output gate
    # --------------------------------------------------------------
    out_gate_pre = torch.nn.functional.linear(flat, out_gate_w)  # (B·N·N, hidden_dim)
    out_gate = torch.sigmoid(out_gate_pre).view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 9️⃣ Second LayerNorm on the hidden dimension
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # apply the output gate
    out = out * out_gate

    # --------------------------------------------------------------
    # 10️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    out_flat = out.view(-1, hidden_dim)                     # (B·N·N, hidden_dim)
    out_proj = torch.nn.functional.linear(out_flat, to_out_w)  # (B·N·N, dim)
    out_proj = out_proj.view(B, N, N, dim).to(torch.float32)

    return out_proj
