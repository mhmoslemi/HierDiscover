"""
Outgoing TriMul (AlphaFold‑3) – optimized forward pass.

Key ideas
---------
1️⃣  Layer‑norm and the final linear projection stay in PyTorch (highly tuned cuBLAS).
2️⃣  The two projection‑+‑gate branches (left/right) are fused:
   a single FP16 matmul computes both the projection and its sigmoid gate
   (weights are concatenated on the output‑feature axis).
3️⃣  The heavy O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is implemented as a batched GEMM C = A @ Bᵀ with a custom Triton kernel.
   •  Tiles of 64/128/256 are chosen adaptively.
   •  Larger tiles for long sequences (≥1024 → 256×256) reduce kernel launch overhead.
   •  8 warps (256 threads) and 4 pipeline stages give good occupancy on H100.
   •  All arithmetic is FP16; accumulation uses FP32 for numerical stability.
4️⃣  The result is LayerNorm‑gated, projected back to the original dimensionality,
   and finally cast to float32 (the harness expects FP32).

The implementation respects the required signature:
    custom_kernel((input, mask, weights, config)) → Tensor[B, N, N, dim] (float32)
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B,C are fp16, accum in fp32)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # scalar sizes
    stride_am, stride_ai, stride_ak,   # strides for A (M,i,k)
    stride_bm, stride_bj, stride_bk,   # strides for B (M,j,k) – note B is accessed transposed
    stride_cm, stride_ci, stride_cj,   # strides for C (M,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Program IDs (i‑tile, j‑tile, batch×head)
    # ------------------------------------------------------------------
    pid_i = tl.program_id(0)   # tile along first N dimension (i)
    pid_j = tl.program_id(1)   # tile along second N dimension (j)
    pid_z = tl.program_id(2)   # batch*head index (M)

    # ------------------------------------------------------------------
    # Offsets for this tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K in tiles
    # ------------------------------------------------------------------
    for k_start in range(0, K, BLOCK_K):
        cur_k = k_start
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # --------------------------------------------------------------
        # Load A tile – shape (BLOCK_M, BLOCK_K)
        # --------------------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # --------------------------------------------------------------
        # Load B tile (already transposed) – shape (BLOCK_K, BLOCK_N)
        # --------------------------------------------------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # --------------------------------------------------------------
        # Mixed‑precision dot: fp16 × fp16 → fp32
        # --------------------------------------------------------------
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result tile (convert back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute ``out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`` via the
    Triton kernel above.  ``left`` and ``right`` are FP16 tensors of shape
    ``[B, N, N, H]`` (contiguous).  The result has the same shape and dtype.
    """
    B, N, _, H = left.shape

    # Rearrange to [M, N, N] where M = B*H.
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in element units (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # --------------------------------------------------------------
    # Choose tile sizes.
    #   N < 512      → 64×64 tiles (good load balance for small N)
    #   512 ≤ N <1024 → 128×128 tiles
    #   N ≥ 1024      → 256×256 tiles (fewer blocks, same total work)
    # --------------------------------------------------------------
    if N >= 1024:
        BLOCK_M = BLOCK_N = 256
        BLOCK_K = 64
        num_warps = 16
    elif N >= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8
    else:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch × hidden
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=4,  # a bit deeper pipeline for better prefetching
    )

    # Reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operator (no backward).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool or float, ignored if config['nomask']=True)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``; optional ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device as ``input_tensor``.
    """
    # -------------------------------------------------
    # Unpack
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣  First LayerNorm (float32 for numerical stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------
    # 2️⃣  Switch to FP16 for the heavy part
    # -------------------------------------------------
    x = x.to(torch.float16)

    # -------------------------------------------------
    # 3️⃣  Cast all model weights to FP16 once
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    left_proj_w   = left_proj_w.to(torch.float16)
    left_gate_w   = left_gate_w.to(torch.float16)
    right_proj_w  = right_proj_w.to(torch.float16)
    right_gate_w  = right_gate_w.to(torch.float16)
    out_gate_w    = out_gate_w.to(torch.float16)
    to_out_norm_w = to_out_norm_w.to(torch.float16)
    to_out_norm_b = to_out_norm_b.to(torch.float16)
    to_out_w      = to_out_w.to(torch.float16)

    # -------------------------------------------------
    # 4️⃣  Fuse projection + gate for the left side
    # -------------------------------------------------
    # Concatenate weights on the output‑feature axis: [2*hidden, dim]
    left_w_cat = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2*hidden, dim)

    left_cat = torch.nn.functional.linear(x, left_w_cat)  # [B, N, N, 2*hidden]  fp16
    left_proj = left_cat[..., :hidden_dim]                # [B, N, N, hidden]
    left_gate_logit = left_cat[..., hidden_dim:]          # [B, N, N, hidden]

    # -------------------------------------------------
    # 5️⃣  Fuse projection + gate for the right side (identical pattern)
    # -------------------------------------------------
    right_w_cat = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*hidden, dim)

    right_cat = torch.nn.functional.linear(x, right_w_cat)        # [B, N, N, 2*hidden] fp16
    right_proj = right_cat[..., :hidden_dim]                     # [B, N, N, hidden]
    right_gate_logit = right_cat[..., hidden_dim:]               # [B, N, N, hidden]

    # -------------------------------------------------
    # 6️⃣  Apply optional mask (broadcast on hidden dim)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # -------------------------------------------------
    # 7️⃣  Sigmoid gates (still FP16)
    # -------------------------------------------------
    left_gate = torch.sigmoid(left_gate_logit)
    right_gate = torch.sigmoid(right_gate_logit)

    # Fuse the gate into the projection inplace to avoid extra temporaries
    torch.mul(left_proj, left_gate, out=left_proj)   # left = left_proj * left_gate
    torch.mul(right_proj, right_gate, out=right_proj)   # right = right_proj * right_gate

    # -------------------------------------------------
    # 8️⃣  Compute the O(N³) contraction via Triton
    # -------------------------------------------------
    # At this point: left, right -> [B, N, N, hidden] (FP16, contiguous)
    left = left.contiguous()
    right = right.contiguous()
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, hidden] (FP16)

    # -------------------------------------------------
    # 9️⃣  Second LayerNorm (still FP16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 🔟  Output gating + final linear projection
    # -------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # [B,N,N,hidden] FP16
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)    # [B,N,N,dim] FP16

    # -------------------------------------------------
    # 1️⃣1️⃣ Cast back to float32 for the harness
    # -------------------------------------------------
    return out.to(torch.float32)
