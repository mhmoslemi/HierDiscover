```python
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel for batched matmul contraction
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_batch = tl.program_id(0)   # combined batch * hidden
    pid_i = tl.program_id(1)       # block row (i)
    pid_j = tl.program_id(2)       # block col (j)

    # compute original batch index and hidden index
    hidden = H
    batch_idx = pid_batch // hidden
    h_idx = pid_batch - batch_idx * hidden   # (pid_batch % hidden)

    # base offset (batch and hidden)
    # hidden dim is innermost, so offset_h = h_idx * stride_l_h (usually 1)
    offset = batch_idx * stride_l_batch + h_idx * stride_l_h

    # block start offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # predicate for i/j out of bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k_start in range(0, N, BLOCK_K):
        # k tile offsets
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (i,k)
        a_ptrs = left_ptr + offset + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask = mask_i[:, None] & mask_k[None, :],
                    other = 0.0).to(tl.float32)

        # Load right tile (k,j) . Note: right has layout (j,k) in memory,
        # we need (k,j) for matmul, so we address with k stride first then j stride.
        b_ptrs = right_ptr + offset + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        b = tl.load(b_ptrs,
                    mask = mask_k[:, None] & mask_j[None, :],
                    other = 0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(a, b)    # shape (BLOCK_M, BLOCK_N)

    # store result
    c_ptrs = out_ptr + offset + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask = mask_i[:, None] & mask_j[None, :])
```

    input_tensor, mask_tensor, weights, config = data
    ...
    # 0. Unpack
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

        bias=norm_bias
    )

    x = x.to(torch.float16)
    B, N, _, _ = x.shape
    M = B * N * N

    # 2. Linear projections, gating
    # Cast linear weights to fp16
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(torch.float16)

    # Flatten x for linear
    x_flat = x.reshape(-1, dim)   # shape (M, dim)
    # Linear
    left_proj = torch.nn.functional.linear(x_flat, left_proj_weight)   # (M, hidden_dim)
    right_proj = torch.nn.functional.linear(x_flat, right_proj_weight)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, out_gate_weight))

    # Reshape to (B,N,N,hidden)
    left_proj = left_proj.view(B, N, N, hidden_dim)
    right_proj = right_proj.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)
    right_gate = right_gate.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    left = left_proj * left_gate
    right = right_proj * right_gate

        right = right * mask

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel for batched TriMul contraction (outgoing)
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out = left @ right^T for each hidden channel.
    
    Shapes:
        left, right, out : (B, N, N, H)
    The kernel flattens the batch and hidden dimensions into a single
    launching axis so that each (b, h) pair is processed independently.
    """
    pid_batch = tl.program_id(0)  # combined batch*hidden index
    pid_i = tl.program_id(1)      # block row (i)
    pid_j = tl.program_id(2)      # block col (j)

    # Compute original batch and hidden indices
    hidden = H
    batch_idx = pid_batch // hidden
    hidden_idx = pid_batch - batch_idx * hidden   # pid_batch % hidden

    # Base offset for this (batch, hidden) pair; hidden is the innermost dim
    base = batch_idx * stride_l_batch + hidden_idx * stride_l_h

    # Tile offsets for i, j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k_start in range(0, N, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile A (i,k)
        a_ptrs = left_ptr + base \
                 + offs_i[:, None] * stride_l_i \
                 + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask = mask_i[:, None] & mask_k[None, :],
                    other = 0.0).to(tl.float32)

        # Load right tile B (k,j) – note the order (k first, then j)
        b_ptrs = right_ptr + base \
                 + offs_k[:, None] * stride_r_k \
                 + offs_j[None, :] * stride_r_i
        b = tl.load(b_ptrs,
                    mask = mask_k[:, None] & mask_j[None, :],
                    other = 0.0).to(tl.float32)

        # Accumulate the matrix product of the current tiles
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # Store the result back to the output tensor (FP16)
    c_ptrs = out_ptr + base \
             + offs_i[:, None] * stride_o_i \
             + offs_j[None, :] * stride_o_j
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask = mask_i[:, None] & mask_j[None, :])

    # ------------------------------------------------------------------
    x_flat = x.view(-1, dim)                     # (M, dim)

    right_proj = torch.nn.functional.linear(x_flat, right_proj_w)

    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, right_gate_w))

    out_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, out_gate_w))

    right_gate = right_gate.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------

    BLOCK_N = 64
    BLOCK_K = 64

    r_strides = right.stride()
    o_strides = out.stride()

    # Total number of independent (batch, hidden) matrices to multiply
    total_batches = B * hidden_dim

    grid = (
        total_batches,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,

        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix‑multiply (left @ rightᵀ) for every hidden channel.

    Arguments:
        left_ptr / right_ptr / out_ptr : pointers to tensors of shape (B,N,N,H)
        B, N, H                       : problem sizes
        stride_*_*                    : element‑wise strides for each dimension
        BLOCK_*                       : tiling sizes (compile‑time constants)
    """

    pid_batch = tl.program_id(0)          # combined (batch, hidden) index
    pid_i     = tl.program_id(1)          # block row (i)

    # ------------------------------------------------------------------
    return out.to(torch.float32)
