@triton.jit
def _project_gate_kernel(
    X_ptr, mask_ptr,
    Wlp_ptr, Wlg_ptr, Wrp_ptr, Wrg_ptr, Wog_ptr,
    L_out_ptr, R_out_ptr, O_out_ptr,
    M, dim, H,
    stride_xm, stride_xk,
    stride_mask_m,
    stride_lp_dim, stride_lp_h,
    stride_lg_dim, stride_lg_h,
    stride_rp_dim, stride_rp_h,
    stride_rg_dim, stride_rg_h,
    stride_og_dim, stride_og_h,
    stride_lout_m, stride_lout_h,
    stride_rout_m, stride_rout_h,
    stride_oout_m, stride_oout_h,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,
    WITH_MASK: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)
    # row offset
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    mask_valid = offs_m < M
    hidden_valid = offs_h < H

    # init accumulators
    acc_lproj = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_lgate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_rproj = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_rgate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_outg = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    num_k_tiles = tl.cdiv(dim, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        k_valid = offs_k < dim

        # load X tile
        x_ptrs = X_ptr + offs_m[:, None] * stride_xm + offs_k[None, :] * stride_xk
        x = tl.load(x_ptrs, mask=mask_valid[:, None] & k_valid[None, :], other=0.0) # fp16
        # cast to fp32 automatically in tl.dot (it does conversion)
        # left_proj weight tile
        wlp_ptrs = Wlp_ptr + offs_k[:, None] * stride_lp_dim + offs_h[None, :] * stride_lp_h
        wlp = tl.load(wlp_ptrs, mask=k_valid[:, None] & hidden_valid[None, :], other=0.0)
        # left_gate weight tile
        wlg_ptrs = Wlg_ptr + offs_k[:, None] * stride_lg_dim + offs_h[None, :] * stride_lg_h
        wlg = tl.load(wlg_ptrs, mask=k_valid[:, None] & hidden_valid[None, :], other=0.0)
        # right_proj weight tile
        wrp_ptrs = Wrp_ptr + offs_k[:, None] * stride_rp_dim + offs_h[None, :] * stride_rp_h
        wrp = tl.load(wrp_ptrs, mask=k_valid[:, None] & hidden_valid[None, :], other=0.0)
        # right_gate weight tile
        wrg_ptrs = Wrg_ptr + offs_k[:, None] * stride_rg_dim + offs_h[None, :] * stride_rg_h
        wrg = tl.load(wrg_ptrs, mask=k_valid[:, None] & hidden_valid[None, :], other=0.0)
        # out_gate weight tile
        wog_ptrs = Wog_ptr + offs_k[:, None] * stride_og_dim + offs_h[None, :] * stride_og_h
        wog = tl.load(wog_ptrs, mask=k_valid[:, None] & hidden_valid[None, :], other=0.0)

        acc_lproj += tl.dot(x, wlp)
        acc_lgate += tl.dot(x, wlg)
        acc_rproj += tl.dot(x, wrp)
        acc_rgate += tl.dot(x, wrg)
        acc_outg += tl.dot(x, wog)
    # finalize: apply sigmoid
    left_gate = 1.0 / (1.0 + tl.exp(-acc_lgate))
    right_gate = 1.0 / (1.0 + tl.exp(-acc_rgate))
    out_gate = 1.0 / (1.0 + tl.exp(-acc_outg))

    left = acc_lproj * left_gate
    right = acc_rproj * right_gate

    if WITH_MASK:
        mask_ptrs = mask_ptr + offs_m * stride_mask_m
        mask_vals = tl.load(mask_ptrs, mask=mask_valid, other=0.0) # fp16
        mask_vals_f = mask_vals.to(tl.float32)  # convert to fp32 for multiply? we can keep fp16 then upcast.
        # Need to broadcast across H dimension
        left = left * mask_vals[:, None]
        right = right * mask_vals[:, None]

    # store results
    l_out_ptrs = L_out_ptr + offs_m[:, None] * stride_lout_m + offs_h[None, :] * stride_lout_h
    r_out_ptrs = R_out_ptr + offs_m[:, None] * stride_rout_m + offs_h[None, :] * stride_rout_h
    o_out_ptrs = O_out_ptr + offs_m[:, None] * stride_oout_m + offs_h[None, :] * stride_oout_h

    tl.store(l_out_ptrs, left.to(tl.float16), mask=mask_valid[:, None] & hidden_valid[None, :])
    tl.store(r_out_ptrs, right.to(tl.float16), mask=mask_valid[:, None] & hidden_valid[None, :])
    tl.store(o_out_ptrs, out_gate.to(tl.float16), mask=mask_valid[:, None] & hidden_valid[None, :])

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    # Extract config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Move stuff to broadcast
    device = input_tensor.device

    # 1. LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # dtype float32

    # 2. Cast to fp16
    x = x.to(torch.float16)

    # 3. Prepare weights for projection kernel
    lp_w = weights["left_proj.weight"].to(torch.float16).t().contiguous()  # (dim, hidden)
    rp_w = weights["right_proj.weight"].to(torch.float16).t().contiguous()
    lg_w = weights["left_gate.weight"].to(torch.float16).t().contiguous()
    rg_w = weights["right_gate.weight"].to(torch.float16).t().contiguous()
    og_w = weights["out_gate.weight"].to(torch.float16).t().contiguous()

    # 4. Prepare optional mask
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
    else:
        mask_fp16 = None

    # 5. Compute left/right/out_gate via triton kernel
    left, right, out_gate = _project_and_gate(x, mask_fp16, lp_w, lg_w, rp_w, rg_w, og_w, dim, hidden_dim)

    # 6. O(N^3) contraction - use torch.bmm which is heavily optimised
    B, N, _, H = left.shape
    left_t = left.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)  # (B*H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)
    out = torch.bmm(left_t, right_t.transpose(1,2))
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # 7. Second LayerNorm
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # 8. Apply out gate
    out = out * out_gate

    # 9. Final linear projection
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)  # no bias

    # 10. Cast back to float32
    return out.to(torch.float32)
```

def _project_and_gate(x, mask, lp_w, lg_w, rp_w, rg_w, og_w, dim, hidden):
    B, N, N, _ = x.shape
    M = B * N * N
    # flatten
    x_flat = x.view(M, dim).contiguous()
    mask_flat = mask.view(M).contiguous() if mask is not None else None

    left_out = torch.empty_like(x_flat, dtype=torch.float16)  # (M, hidden)
    right_out = torch.empty_like(x_flat, dtype=torch.float16)
    out_gate_out = torch.empty_like(x_flat, dtype=torch.float16)

    # get strides
    stride_xm, stride_xk = x_flat.stride()   # (dim, 1)
    stride_lout_m, stride_lout_h = left_out.stride()
    stride_rout_m, stride_rout_h = right_out.stride()
    stride_ogout_m, stride_ogout_h = out_gate_out.stride()

    stride_mask_m = mask_flat.stride(0) if mask_flat is not None else 0

    stride_lp_dim, stride_lp_h = lp_w.stride()
    stride_lg_dim, stride_lg_h = lg_w.stride()
    stride_rp_dim, stride_rp_h = rp_w.stride()
    stride_rg_dim, stride_rg_h = rg_w.stride()
    stride_og_dim, stride_og_h = og_w.stride()

    BLOCK_M = 64
    BLOCK_H = 64
    BLOCK_K = 64

    grid = ((M + BLOCK_M - 1)//BLOCK_M,
            (hidden + BLOCK_H - 1)//BLOCK_H,
            1)

    # kernel call
    _project_gate_kernel[grid](
        x_flat,
        mask_flat if mask_flat is not None else 0,
        lp_w,
        lg_w,
        rp_w,
        rg_w,
        og_w,
        left_out,
        right_out,
        out_gate_out,
        # sizes
        M, dim, hidden,
        # strides
        stride_xm, stride_xk,
        stride_mask_m,
        stride_lp_dim, stride_lp_h,
        stride_lg_dim, stride_lg_h,
        stride_rp_dim, stride_rp_h,
        stride_rg_dim, stride_rg_h,
        stride_og_dim, stride_og_h,
        stride_lout_m, stride_lout_h,
        stride_rout_m, stride_rout_h,
        stride_ogout_m, stride_ogout_h,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
        WITH_MASK=1 if mask_flat is not None else 0,
        num_warps=8,
        num_stages=2
    )
    # reshape results
    left = left_out.view(B, N, N, hidden).contiguous()
    right = right_out.view(B, N, N, hidden).contiguous()
    out_gate = out_gate_out.view(B, N, N, hidden).contiguous()
    return left, right, out_gate

"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

Algorithm
---------
1️⃣  Layer‑norm the input (float32) → cast to fp16.
2️⃣  A fused Triton kernel computes four linear projections
    (left/right) together with their gates and the out‑gate in a
    single pass over the flattened input.  This reduces the memory
    traffic of the many torch.nn.functional.linear calls.
3️⃣  The heavy O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is performed as a batched matrix‑multiply C = A @ Bᵀ using
    torch.bmm (cuBLAS – fp16 × fp16 → fp32 accumulation).  This is
    far faster than a hand‑written Triton kernel for all realistic
    sequence lengths.
4️⃣  Layer‑norm on the hidden dimension, element‑wise out‑gate,
    final linear projection back to `dim`.
5️⃣  Cast the result to float32 (the harness expects float32).

The kernel uses block sizes 64×64×64 (rows × hidden × K) and 8 warps.
All operations after the first LN run in fp16, accumulation in fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fused projections + gating (+ optional mask)
#   Input   : X  – [M, dim]          (M = B * N * N)
#   Weights : Lp, Lg, Rp, Rg, Og – each [dim, hidden] (transposed Linear weights)
#   Output  : Left, Right – [M, hidden]   (proj * gate, mask applied)
#             OutGate    – [M, hidden]   (sigmoid)
# ----------------------------------------------------------------------
@triton.jit
def _project_gate_kernel(
    X_ptr,                # *fp16   [M, dim]
    MASK_ptr,             # *fp16   [M]      (optional)
    Lp_ptr, Lg_ptr,      # *fp16   [dim, hidden]
    Rp_ptr, Rg_ptr,      # *fp16   [dim, hidden]
    Og_ptr,               # *fp16   [dim, hidden]
    L_out_ptr, R_out_ptr, Og_out_ptr,   # *fp16  [M, hidden]
    M, DIM, HIDDEN,                       # i32 sizes
    stride_xm, stride_xk,                # strides for X
    stride_mask_m,                       # stride for MASK
    stride_lp_dim, stride_lp_h,          # strides for Lp
    stride_lg_dim, stride_lg_h,          # strides for Lg
    stride_rp_dim, stride_rp_h,          # strides for Rp
    stride_rg_dim, stride_rg_h,          # strides for Rg
    stride_og_dim, stride_og_h,          # strides for Og
    stride_lout_m, stride_lout_h,        # strides for L_out
    stride_rout_m, stride_rout_h,        # strides for R_out
    stride_ogout_m, stride_ogout_h,      # strides for Og_out
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_K: tl.constexpr,
    WITH_MASK: tl.constexpr,             # 0/1 compile‑time flag
):
    pid_m = tl.program_id(0)   # tile along rows (M)
    pid_h = tl.program_id(1)   # tile along hidden dimension

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_m = offs_m < M
    mask_h = offs_h < HIDDEN

    # ------------------------------------------------------------------
    # Accumulators (fp32)
    # ------------------------------------------------------------------
    acc_lp = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_lg = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_rp = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_rg = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_og = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    num_k_tiles = tl.cdiv(DIM, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < DIM

        # ------------------------------------------------------------------
        # Load X tile  (shape [BLOCK_M, BLOCK_K])
        # ------------------------------------------------------------------
        x_ptrs = X_ptr + offs_m[:, None] * stride_xm + offs_k[None, :] * stride_xk
        x = tl.load(
            x_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )   # fp16 → promoted to fp32 in tl.dot

        # ------------------------------------------------------------------
        # Load weight tiles
        # ------------------------------------------------------------------
        lp_ptrs = Lp_ptr + offs_k[:, None] * stride_lp_dim + offs_h[None, :] * stride_lp_h
        lp = tl.load(lp_ptrs, mask=mask_k[:, None] & mask_h[None, :], other=0.0)

        lg_ptrs = Lg_ptr + offs_k[:, None] * stride_lg_dim + offs_h[None, :] * stride_lg_h
        lg = tl.load(lg_ptrs, mask=mask_k[:, None] & mask_h[None, :], other=0.0)

        rp_ptrs = Rp_ptr + offs_k[:, None] * stride_rp_dim + offs_h[None, :] * stride_rp_h
        rp = tl.load(rp_ptrs, mask=mask_k[:, None] & mask_h[None, :], other=0.0)

        rg_ptrs = Rg_ptr + offs_k[:, None] * stride_rg_dim + offs_h[None, :] * stride_rg_h
        rg = tl.load(rg_ptrs, mask=mask_k[:, None] & mask_h[None, :], other=0.0)

        og_ptrs = Og_ptr + offs_k[:, None] * stride_og_dim + offs_h[None, :] * stride_og_h
        og = tl.load(og_ptrs, mask=mask_k[:, None] & mask_h[None, :], other=0.0)

        # ------------------------------------------------------------------
        # Accumulate GEMM products
        # ------------------------------------------------------------------
        acc_lp += tl.dot(x, lp)   # [M, H]
        acc_lg += tl.dot(x, lg)
        acc_rp += tl.dot(x, rp)
        acc_rg += tl.dot(x, rg)
        acc_og += tl.dot(x, og)

    # ----------------------------------------------------------------------
    # Apply sigmoid gates (still fp32)
    # ----------------------------------------------------------------------
    left_gate = 1.0 / (1.0 + tl.exp(-acc_lg))
    right_gate = 1.0 / (1.0 + tl.exp(-acc_rg))
    out_gate = 1.0 / (1.0 + tl.exp(-acc_og))

    left = acc_lp * left_gate
    right = acc_rp * right_gate

    # ----------------------------------------------------------------------
    # Optional mask (broadcast over hidden dimension)
    # ----------------------------------------------------------------------
    if WITH_MASK:
        mask_ptrs = MASK_ptr + offs_m * stride_mask_m
        mask_val = tl.load(mask_ptrs, mask=mask_m, other=0.0)   # fp16
        mask_val_f = mask_val.to(tl.float32)
        left = left * mask_val_f[:, None]
        right = right * mask_val_f[:, None]

    # ----------------------------------------------------------------------
    # Store results (fp16)
    # ----------------------------------------------------------------------
    l_out_ptrs = L_out_ptr + offs_m[:, None] * stride_lout_m + offs_h[None, :] * stride_lout_h
    tl.store(l_out_ptrs, left.to(tl.float16), mask=mask_m[:, None] & mask_h[None, :])

    r_out_ptrs = R_out_ptr + offs_m[:, None] * stride_rout_m + offs_h[None, :] * stride_rout_h
    tl.store(r_out_ptrs, right.to(tl.float16), mask=mask_m[:, None] & mask_h[None, :])

    og_out_ptrs = Og_out_ptr + offs_m[:, None] * stride_ogout_m + offs_h[None, :] * stride_ogout_h
    tl.store(og_out_ptrs, out_gate.to(tl.float16), mask=mask_m[:, None] & mask_h[None, :])


def _project_and_gate_triton(
    x_fp16: torch.Tensor,
    mask_fp16: torch.Tensor | None,
    w_lp_T: torch.Tensor,
    w_lg_T: torch.Tensor,
    w_rp_T: torch.Tensor,
    w_rg_T: torch.Tensor,
    w_og_T: torch.Tensor,
) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
    """
    Compute left, right and out‑gate tensors with a single Triton kernel.
    All tensors are fp16; accumulations are fp32.
    """
    B, N, _, HIDDEN = x_fp16.shape
    DIM = x_fp16.shape[-1]
    M = B * N * N

    # ----- flatten input -------------------------------------------------
    x_flat = x_fp16.view(M, DIM).contiguous()
    if mask_fp16 is not None:
        mask_flat = mask_fp16.view(M).contiguous()
    else:
        # dummy mask – never read (WITH_MASK = 0)
        mask_flat = torch.empty((1,), dtype=torch.float16, device=x_fp16.device)

    # ----- allocate outputs -----------------------------------------------
    left_out = torch.empty_like(x_flat, dtype=torch.float16)   # (M, hidden)
    right_out = torch.empty_like(x_flat, dtype=torch.float16)
    og_out = torch.empty_like(x_flat, dtype=torch.float16)

    # ----- strides (element counts) ---------------------------------------
    stride_xm, stride_xk = x_flat.stride()
    stride_mask_m = mask_flat.stride(0) if mask_fp16 is not None else 0

    stride_lp_dim, stride_lp_h = w_lp_T.stride()
    stride_lg_dim, stride_lg_h = w_lg_T.stride()
    stride_rp_dim, stride_rp_h = w_rp_T.stride()
    stride_rg_dim, stride_rg_h = w_rg_T.stride()
    stride_og_dim, stride_og_h = w_og_T.stride()

    stride_lout_m, stride_lout_h = left_out.stride()
    stride_rout_m, stride_rout_h = right_out.stride()
    stride_ogout_m, stride_ogout_h = og_out.stride()

    # ----- launch parameters -----------------------------------------------
    BLOCK_M = 64
    BLOCK_H = 64
    BLOCK_K = 64
    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (HIDDEN + BLOCK_H - 1) // BLOCK_H,
        1,
    )

    _project_gate_kernel[grid](
        x_flat,
        mask_flat,
        w_lp_T,
        w_lg_T,
        w_rp_T,
        w_rg_T,
        w_og_T,
        left_out,
        right_out,
        og_out,
        M,
        DIM,
        HIDDEN,
        stride_xm,
        stride_xk,
        stride_mask_m,
        stride_lp_dim,
        stride_lp_h,
        stride_lg_dim,
        stride_lg_h,
        stride_rp_dim,
        stride_rp_h,
        stride_rg_dim,
        stride_rg_h,
        stride_og_dim,
        stride_og_h,
        stride_lout_m,
        stride_lout_h,
        stride_rout_m,
        stride_rout_h,
        stride_ogout_m,
        stride_ogout_h,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
        WITH_MASK=0 if mask_fp16 is None else 1,
        num_warps=8,
        num_stages=2,
    )

    left = left_out.view(B, N, N, HIDDEN).contiguous()
    right = right_out.view(B, N, N, HIDDEN).contiguous()
    out_gate = og_out.view(B, N, N, HIDDEN).contiguous()
    return left, right, out_gate


def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input tensor [B, N, N, dim] (float32)
        torch.Tensor,                     # mask tensor  [B, N, N]   (float32 / bool)
        Dict[str, torch.Tensor],          # weights
        Dict[str, Any],                   # config (must contain dim, hidden_dim)
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients), implemented
    with a fused projection/gate Triton kernel and a cuBLAS batched GEMM for
    the O(N³) contraction.

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ first LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ cast to fp16 for the bulk of the computation
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ prepare FP16 weight tensors (transpose for GEMM)
    # ------------------------------------------------------------------
    w_lp_T = weights["left_proj.weight"].to(torch.float16).t().contiguous()
    w_rp_T = weights["right_proj.weight"].to(torch.float16).t().contiguous()
    w_lg_T = weights["left_gate.weight"].to(torch.float16).t().contiguous()
    w_rg_T = weights["right_gate.weight"].to(torch.float16).t().contiguous()
    w_og_T = weights["out_gate.weight"].to(torch.float16).t().contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ optional mask (broadcasted over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).contiguous()
    else:
        mask_fp16 = None

    # ------------------------------------------------------------------
    # 5️⃣ fused projection + gate kernel → left/right/out_gate
    # ------------------------------------------------------------------
    left, right, out_gate = _project_and_gate_triton(
        x,
        mask_fp16,
        w_lp_T,
        w_lg_T,
        w_rp_T,
        w_rg_T,
        w_og_T,
    )   # shapes: [B, N, N, hidden_dim] (fp16)

    # ------------------------------------------------------------------
    # 6️⃣ heavy O(N³) contraction: batched mat‑mul (left @ rightᵀ)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    left_b = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, N, N)
    right_b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out = torch.bmm(left_b, right_b.transpose(1, 2))                    # (B*H, N, N)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()       # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7️⃣ second LayerNorm (still fp16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ out‑gate modulation
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)      # (B, N, N, dim) fp16

    # ------------------------------------------------------------------
    # 🔟 cast result back to float32 as required by the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
