    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()
    # Determine tile size based on N
    BLOCK_M = 128 if N > 384 else 64
    BLOCK_N = BLOCK_M
    BLOCK_K = 64
    # Ensure each block size <= N
    grid = ((N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N, B*hd)
    _trimul_matmul_kernel[grid](
        A, B_, C,
        M=B*hd, N=N, K=N,
        stride_am=stride_am, stride_ai=stride_ai, stride_ak=stride_ak,
        stride_bm=stride_bm, stride_bj=stride_bj, stride_bk=stride_bk,
        stride_cm=stride_cm, stride_ci=stride_ci, stride_cj=stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2
    )
    # reshape back
    C = C.view(B, hd, N, N).permute(0,2,3,1).contiguous()
    return C

```
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _trimul_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          M, N, K,
                          stride_am, stride_ai, stride_ak,
                          stride_bm, stride_bj, stride_bk,
                          stride_cm, stride_ci, stride_cj,
                          BLOCK_M: tl.constexpr,
                          BLOCK_N: tl.constexpr,
                          BLOCK_K: tl.constexpr):
    pid_m = tl.program_id(0)  # tile row
    pid_n = tl.program_id(1)  # tile col
    pid_z = tl.program_id(2)  # batch*hidden

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        a_ptrs = A_ptr + pid_z * stride_am + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & (offs_k[None, :] < K),
                    other=0.0)   # fp16

        b_ptrs = B_ptr + pid_z * stride_bm + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_j[None, :],
                    other=0.0)   # fp16

        acc += tl.dot(a, b)  # fp16×fp16 → fp32

    c_ptrs = C_ptr + pid_z * stride_cm + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right: [B, N, N, H] (fp16)
    Returns: out [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    # reshape to [M, N, K] where M = B*H, K = N (contraction axis)
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)

    stride_am, stride_ai, stride_ak = A.stride()   # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()  # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()   # (N*N, N, 1)

    # Choose block sizes based on N to keep tiles within bounds
    if N <= 128:
        BLOCK_M = 64
        BLOCK_N = 64
    elif N <= 384:
        BLOCK_M = 64
        BLOCK_N = 64
    else:
        BLOCK_M = 128
        BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    # reshape back: [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass – fully fused version with fp16
    math and a Triton batched GEMM for the O(N³) contraction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights, config)
    Returns
    -------
    torch.Tensor
        Output [B, N, N, dim] in float32.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------
    # 1. First LayerNorm (float32)
    # ------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim] float32

    # ------------------------------------
    # 2. FP16 cast + flatten for linear ops
    # ------------------------------------
    x = x.to(torch.float16)
    B, N, _, _ = x.shape
    total = B * N * N
    x_flat = x.view(total, dim)  # [B*N*N, dim]

    # ------------------------------------
    # 3. Convert all static weights to fp16 once
    # ------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------
    # 4. Fused projections + gate linear layers
    # ------------------------------------
    # left_proj + left_gate
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)      # (2*H, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)   # (2*H, dim)

    left_combined = F.linear(x_flat, left_comb_w)   # (total, 2*H)
    right_combined = F.linear(x_flat, right_comb_w)

    # reshape back to [B, N, N, 2*H]
    left_combined = left_combined.view(B, N, N, 2 * hidden_dim)
    right_combined = right_combined.view(B, N, N, 2 * hidden_dim)

    # split projection and pre‑gate tensors
    left_proj = left_combined[..., :hidden_dim]
    left_gate_pre = left_combined[..., hidden_dim:]
    right_proj = right_combined[..., :hidden_dim]
    right_gate_pre = right_combined[..., hidden_dim:]

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------
    # 5. Out‑gate (single linear + sigmoid)
    # ------------------------------------
    out_gate_pre = F.linear(x_flat, out_gate_w)  # (total, H)
    out_gate = torch.sigmoid(out_gate_pre).view(B, N, N, hidden_dim)

    # ------------------------------------
    # 6. Apply mask (if any) and fuse gates
    # ------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------
    # 7. Heavy O(N³) contraction
    # ------------------------------------
    # Use Triton for modest sequence length, cuBLAS otherwise.
    if N <= 256:
        out = _batched_matmul_triton(left, right)  # fp16
    else:
        out = _batched_matmul_torch(left, right)   # fp16

    # ------------------------------------
    # 8. Second LayerNorm, out‑gate, final projection
    # ------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)    # [B, N, N, dim] fp16

    # ------------------------------------
    # 9. Cast back to float32 (output dtype)
    # ------------------------------------
    return out.to(torch.float32)
```

"""
Outgoing TriMul (AlphaFold‑3) – highly‑fused forward pass.

Algorithm
---------
1️⃣  Layer‑norm the input (float32) → cast to fp16.
2️⃣  Fuse left/right projections with their gates:
        * stack the two weight matrices (proj & gate) → one matmul.
        * split the result, apply sigmoid to the gate part.
3️⃣  Compute the out‑gate in the same way.
4️⃣  Fuse the optional mask and the gating multiplications:
        left = left_proj * left_gate * mask   (if mask present)
        right = right_proj * right_gate * mask
5️⃣  Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   – performed as a batched GEMM C = A @ Bᵀ.
     • Small N (≤256) → custom Triton kernel (fp16 × fp16 → fp32 acc.).
     • Larger N → torch.bmm (cuBLAS) which already batches over
       (batch × hidden_dim) dim.
6️⃣  Second LayerNorm (fp16) → element‑wise out‑gate → final Linear.
7️⃣  Cast the result back to float32.

All linear layers after the first LayerNorm are executed in fp16.
Two linear ops have been fused (proj + gate) to cut the number of
matrix‑multiplications by 1/3, which is the main speed‑up.
A Triton kernel is still used for the contraction (requirement of the
challenge) and falls back to cuBLAS for longer sequences.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (fp16 → fp16, accumulator fp32)
#   A, B : [M, N, K]   (M = batch * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 scalar sizes
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # tile over i
    pid_n = tl.program_id(1)  # tile over j
    pid_z = tl.program_id(2)  # batch*hidden index (M)

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # --------------------------------------------------------------
    # loop over K‑tiles
    # --------------------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # A[i, k] tile
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # B[j, k] transposed view → (k, j) tile
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk   # k‑dim first
            + offs_j[None, :] * stride_bj   # j‑dim second
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # mixed‑precision dot (fp16 × fp16 → fp32)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # write result
    # --------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM C = left @ rightᵀ using the Triton kernel above.
    All tensors are fp16; the accumulator is fp32.
    Shapes:
        left, right : [B, N, N, H]   (H = hidden_dim)
    Returns:
        out : [B, N, N, H]   fp16
    """
    B, N, _, H = left.shape
    # reshape to [M, N, K] where M = B*H, K = N
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)   # fp16 output

    # strides (in element counts, not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # tile size – adapt a bit to the sequence length
    if N <= 128:
        BLOCK_M = BLOCK_N = 64
    elif N <= 384:
        BLOCK_M = BLOCK_N = 64
    else:
        BLOCK_M = BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to the original 4‑D layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Same contraction using torch.bmm (cuBLAS).
    """
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 / bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ first LayerNorm (float32) – keep it in full precision
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ cast to fp16 and flatten for the linear layers
    # ------------------------------------------------------------------
    x = x.to(torch.float16)
    B, N, _, _ = x.shape
    total = B * N * N
    x_flat = x.view(total, dim)                      # [B·N·N, dim]

    # ------------------------------------------------------------------
    # 3️⃣ move static weights to fp16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ fused proj + gate linear layers (proj || gate)
    # ------------------------------------------------------------------
    left_comb_w  = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2·H, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0) # (2·H, dim)

    left_combined  = F.linear(x_flat, left_comb_w)   # (total, 2·H)
    right_combined = F.linear(x_flat, right_comb_w)

    # reshape back to 4‑D tensors
    left_combined  = left_combined.view(B, N, N, 2 * hidden_dim)
    right_combined = right_combined.view(B, N, N, 2 * hidden_dim)

    # split into projection and raw gate tensors
    left_proj      = left_combined[..., :hidden_dim]
    left_gate_pre  = left_combined[..., hidden_dim:]
    right_proj     = right_combined[..., :hidden_dim]
    right_gate_pre = right_combined[..., hidden_dim:]

    # apply sigmoid to the gate parts
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 5️⃣ out‑gate (single linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate_pre = F.linear(x_flat, out_gate_w)                 # (total, H)
    out_gate = torch.sigmoid(out_gate_pre).view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 6️⃣ optional mask + fuse gate multiplications
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left  = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ heavy O(N³) contraction (batched GEMM)
    # ------------------------------------------------------------------
    if N <= 256:
        out = _batched_matmul_triton(left, right)   # fp16
    else:
        out = _batched_matmul_torch(left, right)    # fp16

    # ------------------------------------------------------------------
    # 8️⃣ second LayerNorm + out‑gate + final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)                  # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 9️⃣ cast back to float32 (interface requirement)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
