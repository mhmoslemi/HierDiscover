```python
@triton.jit
def _gate_and_sigmoid_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    out_gate_ptr,
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    total_elements,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    left_proj = tl.load(left_proj_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)
    right_proj = tl.load(right_proj_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask)
    out_gate_raw = tl.load(out_gate_ptr + offs, mask=mask)

    # sigmoid
    left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))
    out_gate_sig = 1.0 / (1.0 + tl.exp(-out_gate_raw))

    left_out = left_proj * left_gate_sig
    right_out = right_proj * right_gate_sig

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)
    tl.store(out_gate_out_ptr + offs, out_gate_sig, mask=mask)
```

"""
TriMul (outgoing) – fused‑kernel Triton implementation
=====================================================

The forward pass follows the reference implementation exactly but
optimises the two most expensive sources of latency:

1️⃣  **Fused linear projections** – the five weight matrices
    (left/right projections, three gates) are concatenated into a
    single `(5*hidden, dim)` matrix.  One GEMM (fp16) therefore replaces
    five separate matmuls, cutting the number of memory reads of the
    input tensor from *5×* to *1×*.

2️⃣  **Fused gating** – a custom Triton kernel computes the two
    element‑wise sigmoids, the multiplications with the projected
    tensors, and the output‑gate sigmoid in **one** pass, avoiding
    three extra torch kernels (``sigmoid`` + ``mul``).

The remaining O(N³) contraction
    `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
is performed with a highly‑optimised batched GEMM (cuBLAS) as it
remains the dominant cost on an H100.  All other ops (LayerNorms,
final projection) stay in FP16, and the result is cast back to FP32.

Both the optional binary mask and the multi‑batch (B∈{1,2}) case are
handled.  The implementation works for all test configurations listed
in the problem statement and uses Triton for at least one non‑trivial
kernel (the gating kernel).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – fused gating (two sigmoids, three element‑wise muls)
# ----------------------------------------------------------------------
@triton.jit
def _gating_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    out_gate_raw_ptr,
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    size,
    stride_lp, stride_lg, stride_rp, stride_rg, stride_og,
    stride_lo, stride_ro, stride_og_out,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    # Load inputs (fp16)
    lp = tl.load(left_proj_ptr + offs * stride_lp, mask=mask)
    lg = tl.load(left_gate_ptr + offs * stride_lg, mask=mask)
    rp = tl.load(right_proj_ptr + offs * stride_rp, mask=mask)
    rg = tl.load(right_gate_ptr + offs * stride_rg, mask=mask)
    og_raw = tl.load(out_gate_raw_ptr + offs * stride_og, mask=mask)

    # Sigmoid = 1/(1+exp(-x))
    lg_sig = 1.0 / (1.0 + tl.exp(-lg))
    rg_sig = 1.0 / (1.0 + tl.exp(-rg))
    og_sig = 1.0 / (1.0 + tl.exp(-og_raw))

    # Apply gates
    left_out = lp * lg_sig
    right_out = rp * rg_sig

    # Store results
    tl.store(left_out_ptr + offs * stride_lo, left_out, mask=mask)
    tl.store(right_out_ptr + offs * stride_ro, right_out, mask=mask)
    tl.store(out_gate_out_ptr + offs * stride_og_out, og_sig, mask=mask)


def _apply_gating(
    left_proj: torch.Tensor,
    left_gate: torch.Tensor,
    right_proj: torch.Tensor,
    right_gate: torch.Tensor,
    out_gate_raw: torch.Tensor,
) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
    """
    Fuse sigmoids and element‑wise multiplications with a Triton kernel.
    All input tensors must have identical shape `[B, N, N, H]` and be
    contiguous in memory (fp16). Returns ``left``, ``right`` and
    ``out_gate`` tensors of the same shape (fp16).
    """
    assert (
        left_proj.shape
        == left_gate.shape
        == right_proj.shape
        == right_gate.shape
        == out_gate_raw.shape
    ), "All gating tensors must share shape"

    total = left_proj.numel()
    block = 1024  # trade‑off between launch overhead and occupancy
    grid = (triton.cdiv(total, block),)

    # Allocate output buffers
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)
    out_gate_out = torch.empty_like(out_gate_raw)

    # All tensors are contiguous → stride = 1
    stride = 1
    _gating_kernel[grid](
        left_proj, left_gate,
        right_proj, right_gate,
        out_gate_raw,
        left_out, right_out, out_gate_out,
        total,
        stride, stride, stride, stride, stride,
        stride, stride, stride,
        BLOCK_SIZE=block,
        num_warps=4,
    )
    return left_out, right_out, out_gate_out


# ----------------------------------------------------------------------
# Batched matrix multiplication for the O(N³) contraction
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute ``out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]``.
    Arguments are FP16 tensors of shape ``[B, N, N, H]``.
    Returns a tensor of the same shape (FP16).
    """
    B, N, _, H = left.shape
    # Reinterpret as (B*H, N, N) – left on the left side of GEMM
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    # Batched GEMM with transposed right operand
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point – forward pass of the outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``,
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → cast to fp16 for the rest
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32
    x_fp16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all five linear projections into a single GEMM
    # ------------------------------------------------------------------
    # Cast weights once to fp16
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # Stack the five matrices (order matters for later slicing)
    fused_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # shape [(5*hidden), dim]  fp16

    # Linear projection – one huge GEMM
    # Shape after linear: [B, N, N, 5*hidden] (fp16)
    fused = F.linear(x_fp16, fused_weight)  # bias is zero in the reference

    # ------------------------------------------------------------------
    # 3️⃣ Reshape to separate the 5 groups and prepare gating
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    fused = fused.view(B, N, N, 5, hidden_dim)  # [B, N, N, 5, H]

    left_proj    = fused[..., 0, :].contiguous()
    right_proj   = fused[..., 1, :].contiguous()
    left_gate_raw = fused[..., 2, :].contiguous()
    right_gate_raw = fused[..., 3, :].contiguous()
    out_gate_raw  = fused[..., 4, :].contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Fuse sigmoids and element‑wise multiplications (Triton)
    # ------------------------------------------------------------------
    left_gated, right_gated, out_gate = _apply_gating(
        left_proj, left_gate_raw, right_proj, right_gate_raw, out_gate_raw
    )

    # ------------------------------------------------------------------
    # 5️⃣ Optional binary mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_gated = left_gated * mask_fp16
        right_gated = right_gated * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction – batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    # out shape: [B, N, N, hidden] (fp16)
    out = _batched_matmul_transpose_fp16(left_gated, right_gated)

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (on hidden dimension)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # ------------------------------------------------------------------
    # 8️⃣ Final gating + projection back to original dimension
    # ------------------------------------------------------------------
    out = out * out_gate  # element‑wise out‑gate (fp16)
    to_out_w = weights["to_out.weight"].to(torch.float16)  # shape [dim, hidden]
    out = F.linear(out, to_out_w)  # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to FP32 (expected output type)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
