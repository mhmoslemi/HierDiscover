"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

The heavy O(N³) contraction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
is implemented as a fused Triton GEMM that works directly on the
[batch, seq, seq, hidden] layout (no explicit permutation/copy).
For each (i‑tile, j‑tile, batch) the kernel iterates over the hidden
dimension and over the K‑tiles, loading the required 2‑D slices for the
current hidden head with fp16 precision and accumulating in fp32 via
`tl.dot`.  This eliminates the costly transposition step of previous
implementations while keeping full tensor‑core utilisation.

All surrounding operations (LayerNorms, linear projections, gating,
masking) are performed with high‑performance PyTorch primitives in
fp16.  The final result is cast back to float32 for the harness.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
#   left, right, out : [B, N, N, H] (fp16)
#   • Accumulator is fp32.
#   • Tiles: BLOCK_M × BLOCK_N output, BLOCK_K reduction.
#   • Hidden dimension is iterated inside the kernel to avoid any data
#     layout permutation.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    batch, N, H, K,                    # scalar sizes
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,   # left strides
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,   # right strides
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute the outgoing TriMul contraction for a single batch."""
    # --------------------------------------------------------------
    #  Program IDs
    # --------------------------------------------------------------
    pid_i = tl.program_id(0)      # tile along the first sequence dimension (i)
    pid_j = tl.program_id(1)      # tile along the second sequence dimension (j)
    pid_b = tl.program_id(2)      # batch index

    # --------------------------------------------------------------
    #  Offsets for i / j tiles
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    #  Loop over hidden heads (no extra copy of the tensor)
    # --------------------------------------------------------------
    h = 0
    while h < H:
        # accumulator for the current head (fp32)
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # ----------------------------------------------------------
        #  Loop over the K‑dimension (sequence length)
        # ----------------------------------------------------------
        k = 0
        while k < K:
            offs_k = k + tl.arange(0, BLOCK_K)
            mask_k = offs_k < K

            # ----- load left slice  (shape BLOCK_M × BLOCK_K) -----
            a_ptrs = (
                A_ptr
                + pid_b * stride_a_batch
                + offs_i[:, None] * stride_a_i
                + offs_k[None, :] * stride_a_k
                + h * stride_a_h
            )
            a = tl.load(
                a_ptrs,
                mask=mask_i[:, None] & mask_k[None, :],
                other=0.0,
            )   # fp16

            # ----- load right slice (shape BLOCK_K × BLOCK_N) -----
            b_ptrs = (
                B_ptr
                + pid_b * stride_b_batch
                + offs_k[:, None] * stride_b_k
                + offs_j[None, :] * stride_b_j
                + h * stride_b_h
            )
            b = tl.load(
                b_ptrs,
                mask=mask_k[:, None] & mask_j[None, :],
                other=0.0,
            )   # fp16

            # ----- fused fp16 × fp16 → fp32 dot (tensor‑core) -----
            acc += tl.dot(a, b)

            k += BLOCK_K
        # ----------------------------------------------------------
        #  Store the result slice for head h
        # ----------------------------------------------------------
        c_ptrs = (
            C_ptr
            + pid_b * stride_c_batch
            + offs_i[:, None] * stride_c_i
            + offs_j[None, :] * stride_c_j
            + h * stride_c_h
        )
        tl.store(
            c_ptrs,
            acc.to(tl.float16),
            mask=mask_i[:, None] & mask_j[None, :],
        )
        h += 1


def _trimul_outgoing_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the outgoing TriMul contraction on fp16 tensors using the Triton kernel
    defined above.  ``left`` and ``right`` have shape [B, N, N, H] (contiguous).

    Returns
    -------
    out : torch.Tensor
        Tensor of shape [B, N, N, H] (fp16) – same layout as the inputs.
    """
    B, N, _, H = left.shape
    # output buffer
    out = torch.empty_like(left)

    # strides (in element units, not bytes)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h = right.stride()
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h = out.stride()

    # Choose tile sizes – larger tiles for long sequences
    if N >= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B,                               # batch dimension
    )

    _trimul_outgoing_kernel[grid](
        left, right, out,
        B, N, H, N,                     # batch, N, hidden, K (=N)
        stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
        stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
        stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operator (no backward).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default ``True``).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ---------- unpack ----------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # ---------- 1️⃣ first LayerNorm ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ---------- 2️⃣ cast to fp16 ----------
    x = x.to(torch.float16)

    # ---------- 3️⃣ cast model weights ----------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ---------- 4️⃣ linear projections ----------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,hidden]

    # ---------- 5️⃣ optional mask ----------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ---------- 6️⃣ gating ----------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    torch.mul(left, left_gate, out=left)   # fuse gate in‑place
    torch.mul(right, right_gate, out=right)

    # ---------- 7️⃣ heavy O(N³) contraction via Triton ----------
    left = left.contiguous()
    right = right.contiguous()
    out = _trimul_outgoing_fp16(left, right)   # [B,N,N,hidden] fp16

    # ---------- 8️⃣ second LayerNorm ----------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ---------- 9️⃣ final gating + projection ----------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # ---------- 🔟 cast back to float32 ----------
    return out.to(torch.float32)
