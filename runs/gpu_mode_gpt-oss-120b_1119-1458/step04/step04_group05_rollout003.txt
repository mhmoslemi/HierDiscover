import triton
import triton.language as tl

@triton.jit
def trimul_bmm_kernel(A_ptr, B_ptr, C_ptr,
                      M, N, K,
                      stride_am, stride_an, stride_ak,
                      stride_bm, stride_bn, stride_bk,
                      stride_cm, stride_cn, stride_ck,
                      BLOCK_M: tl.constexpr,
                      BLOCK_N: tl.constexpr,
                      BLOCK_K: tl.constexpr):
    # Load tile indices
    pid_z = tl.program_id(0) # batch*hidden dimension index
    pid_m = tl.program_id(1) # tile row index (i)
    pid_n = tl.program_id(2) # tile col index (j)

    # compute offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A and B tiles
        a = tl.load(A_ptr + pid_z * stride_am
                    + offs_i[:, None] * stride_an
                    + offs_k[None, :] * stride_ak, mask=True, other=0.0)

def _trimul_bmm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_an, stride_ak,
    stride_bm, stride_bn, stride_bk,
    stride_cm, stride_cn, stride_ck,

    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,

    pid_z = tl.program_id(0)  # batch*head dimension
    pid_m = tl.program_id(1)  # row tile index (i)
    pid_n = tl.program_id(2)  # col tile index (j)

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

"""
TriMul (outgoing) – fast forward implementation using Triton for the O(N³) contraction.

The heavy part of the module is the tensor contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

which can be expressed as a batched matrix multiplication
C = A @ Bᵀ with A,B ∈ ℝ^{M×N×N} where M = B·H (batch × hidden_dim) and N = seq_len.
The kernel below performs this GEMM in FP16 (tensor‑core) with FP32 accumulation,
operating on tiles of size `BLOCK=64`.  All other operations (LayerNorm, gating,
linear projections) are done with highly‑optimized PyTorch primitives.
The final result is returned as FP32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel for C = A @ Bᵀ  (batched GEMM)
#   A : (M, N, K)   – here K == N
#   B : (M, K, N)   – Bᵀ already transposed
#   C : (M, N, N)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_bmm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_an, stride_ak,
    stride_bm, stride_bn, stride_bk,
    stride_cm, stride_cn, stride_ck,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)      # index over (batch * hidden_dim)
    pid_i = tl.program_id(1)      # tile row (i)
    pid_j = tl.program_id(2)      # tile col (j)

    # i, j indices inside the tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32 (to keep numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the reduction dimension K (which equals N)
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)

        # A[i, k]  (BLOCK_M × BLOCK_K)
        a_ptr = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_an
            + k_off[None, :] * stride_ak
        )
        a = tl.load(a_ptr, mask=True, other=0.0)          # fp16

        # Bᵀ[k, j] = B[k, j]  (BLOCK_K × BLOCK_N)
        b_ptr = (
            B_ptr
            + pid_z * stride_bm
            + k_off[:, None] * stride_bn
            + offs_j[None, :] * stride_bk
        )
        b = tl.load(b_ptr, mask=True, other=0.0)          # fp16

        # Tensor‑core multiply‑accumulate
        acc += tl.dot(a, b)                               # fp16 * fp16 → fp32

    # Write the result tile back (cast to fp16)
    c_ptr = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_cn
        + offs_j[None, :] * stride_ck
    )
    tl.store(c_ptr, acc.to(tl.float16), mask=True)


def _next_multiple(x: int, multiple: int) -> int:
    """Round `x` up to the nearest multiple of `multiple`."""
    return ((x + multiple - 1) // multiple) * multiple


def _batched_matmul_triton(
    left: torch.Tensor,
    right: torch.Tensor,
    tile: int = 64,
    k_tile: int = 64,
) -> torch.Tensor:
    """
    Execute the batched GEMM C = left @ rightᵀ with a Triton kernel.
    `left` and `right` are FP16 tensors of shape [B, N, N, H] (`H` = hidden_dim).
    Returns a tensor of the same shape (still FP16) containing the contraction.
    """
    B, N, _, H = left.shape
    N_pad = _next_multiple(N, tile)

    # Pad the spatial dimensions to a multiple of `tile` so the kernel can
    # omit any bound checks (the extra entries are zero and will be discarded later).
    if N_pad != N:
        pad = (0, 0, 0, N_pad - N, 0, N_pad - N)          # (H left, H right, N2 left, N2 right, N1 left, N1 right)
        left = torch.nn.functional.pad(left, pad, value=0.0)
        right = torch.nn.functional.pad(right, pad, value=0.0)

    # Pack batch and hidden dimensions: (M, N_pad, N_pad) with M = B*H
    left_perm = left.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N) → A
    right_perm = right.permute(0, 3, 2, 1).contiguous()    # (B, H, N, N) → Bᵀ
    M = B * H
    A = left_perm.view(M, N_pad, N_pad)
    B_ = right_perm.view(M, N_pad, N_pad)

    # Allocate output
    C = torch.empty_like(A)                               # fp16

    # Strides in element units (not bytes)
    stride_am, stride_an, stride_ak = A.stride()
    stride_bm, stride_bn, stride_bk = B_.stride()
    stride_cm, stride_cn, stride_ck = C.stride()

    # Grid: (batch*head, tiles_i, tiles_j)
    grid = (
        M,
        (N_pad + tile - 1) // tile,
        (N_pad + tile - 1) // tile,
    )

    # Launch the kernel
    _trimul_bmm_kernel[grid](
        A,
        B_,
        C,
        M,
        N_pad,
        N_pad,
        stride_am,
        stride_an,
        stride_ak,
        stride_bm,
        stride_bn,
        stride_bk,
        stride_cm,
        stride_cn,
        stride_ck,
        BLOCK_M=tile,
        BLOCK_N=tile,
        BLOCK_K=k_tile,
        num_warps=8,
        num_stages=4,
    )

    # Reshape back to [B, N, N, H] and slice off any padding.
    C = C.view(B, H, N_pad, N_pad).permute(0, 2, 3, 1).contiguous()
    if N_pad != N:
        C = C[:, :N, :N, :]
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor[B, N, N] (float32 / bool / None)
        - weights_dict : dict with keys matching the reference PyTorch module
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor[B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1️⃣ Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2️⃣ First LayerNorm (keep FP32 for numerical safety, then cast)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    ).to(torch.float16)                         # FP16 for the heavy part

    # ------------------------------------------------------------------
    # 3️⃣ Cast all model weights once to FP16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections & gating (all in FP16)
    # ------------------------------------------------------------------
    B, N, _, _ = x_norm.shape
    flat = x_norm.view(-1, dim)                     # (B·N·N, dim)

    # Projections
    left = torch.nn.functional.linear(flat, left_proj_w).view(B, N, N, hidden_dim)
    right = torch.nn.functional.linear(flat, right_proj_w).view(B, N, N, hidden_dim)

    # Optional mask (broadcast over hidden_dim)
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_fp16
        right = right * mask_fp16

    # Gating (sigmoid) – still FP16
    left_gate = torch.sigmoid(torch.nn.functional.linear(flat, left_gate_w)).view(
        B, N, N, hidden_dim
    )
    right_gate = torch.sigmoid(torch.nn.functional.linear(flat, right_gate_w)).view(
        B, N, N, hidden_dim
    )
    out_gate = torch.sigmoid(torch.nn.functional.linear(flat, out_gate_w)).view(
        B, N, N, hidden_dim
    )

    # Fuse gating into the projected tensors
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ O(N³) contraction – batched matmul via Triton
    # ------------------------------------------------------------------
    out = _batched_matmul_triton(left, right, tile=64, k_tile=64)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm, output gate and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, dim)  FP16

    # ------------------------------------------------------------------
    # 7️⃣ Cast back to FP32 (the caller expects float32)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
