@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch_stride_a, stride_am, stride_ak,
    batch_stride_b, stride_bk, stride_bn,
    batch_stride_c, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
    # Compute batch ID, block indices
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # maybe use fp16 accumulation?
    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A block [M, K] and B block [K, N]
        A = tl.load(A_ptr + batch_id * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak, mask=offs_m[:, None] < M and offs_k[None, :] < K, other=0.0)

        # Compute dot product
        acc += tl.dot(A, B) # but dot expects shape (BLOCK_M, BLOCK_K) and (BLOCK_K, BLOCK_N) and returns (BLOCK_M, BLOCK_N)
    # Write back
    # Cast accumulator to output dtype, maybe fp16
    C = acc.to(tl.float16)
    tl.store(C_ptr + batch_id * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn, C, mask=offs_m[:, None] < M and offs_n[None, :] < N)

Define Triton kernel:

```
@triton.jit
def batched_gemm_kernel(A_ptr, B_ptr, C_ptr,
                        stride_A_batch, stride_A_m, stride_A_k,
                        stride_B_batch, stride_B_k, stride_B_n,
                        stride_C_batch, stride_C_m, stride_C_n,
                        M, N, K,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Bounds check
        mask_m = offs_m < M
        mask_n = offs_n < N
        mask_k = offs_k < K

        # Load A and B
        a = tl.load(A_ptr + batch_id * stride_A_batch
                              + offs_m[:, None] * stride_A_m
                              + offs_k[None, :] * stride_A_k,
                    mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(B_ptr + batch_id * stride_B_batch
                              + offs_k[:, None] * stride_B_k
                              + offs_n[None, :] * stride_B_n,
                    mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute dot product: accumulate a * b
        acc += tl.dot(a, b)  # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N)

    # Write out result
    c = acc.to(tl.float16)
    tl.store(C_ptr + batch_id * stride_C_batch
                     + offs_m[:, None] * stride_C_m
                     + offs_n[None, :] * stride_C_n,
             c,
             mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)
    device = input_tensor.device

    # First LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor,
                                       (dim,),
                                       weight=weights["norm.weight"],
                                       bias=weights["norm.bias"])
    x_fp16 = x.to(torch.float16)

    # Combined projection (left_proj, right_proj, left_gate, right_gate, out_gate)
    # Stack weights
    big_weight = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16)  # shape (5*hidden_dim, dim)

    # Flatten x and apply linear
    B, N, _, _ = x_fp16.shape
    x_flat = x_fp16.view(-1, dim)   # shape (B*N*N, dim)
    proj = torch.nn.functional.linear(x_flat, big_weight)  # shape (B*N*N, 5*hidden_dim)
    proj = proj.view(B, N, N, 5, hidden_dim)  # shape (B,N,N,5,hidden)

    # Split
    left = proj[..., 0, :]  # B,N,N,hidden
    right = proj[..., 1, :]
    left_gate = proj[..., 2, :]
    right_gate = proj[..., 3, :]
    out_gate = proj[..., 4, :]

    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # B,N,N,1
        left = left * mask_fp16
        right = right * mask_fp16

    # Sigmoid gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Compute out_hidden via batched GEMM: out = left @ right^T for each hidden slice
    # Prepare tensors for Triton kernel
    # left_perm: B, hidden, N, N   = left.permute(0,3,1,2)
    # right_perm: B, hidden, N, N = right.permute(0,3,2,1) (transpose to (k,j) order)
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,2,1).contiguous()
    # Flatten batch: B*hidden as batch_total
    batch_total = B * hidden_dim
    left_mat = left_perm.view(batch_total, N, N)
    right_mat = right_perm.view(batch_total, N, N)
    # Allocate output
    out_hidden = torch.empty_like(left_mat)   # same dtype and device (fp16)

    # call Triton kernel
    _batched_matmul_triton(left_mat, right_mat, out_hidden, N)

    # Reshape back to B,N,N,hidden
    out_hidden = out_hidden.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()

    # Second layernorm
    out_hidden = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # Apply out_gate
    out_hidden = out_hidden * out_gate

    # Final linear projection to dim
    out = torch.nn.functional.linear(out_hidden, weights["to_out.weight"].to(torch.float16))

    # Cast to float32
    return out.to(torch.float32)

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_A_batch, stride_A_m, stride_A_k,
    stride_B_batch, stride_B_k, stride_B_n,
    stride_C_batch, stride_C_m, stride_C_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a = tl.load(A_ptr + batch_id * stride_A_batch
                           + offs_m[:, None] * stride_A_m
                           + offs_k[None, :] * stride_A_k,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(B_ptr + batch_id * stride_B_batch
                           + offs_k[:, None] * stride_B_k
                           + offs_n[None, :] * stride_B_n,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # accumulate product
        acc += tl.dot(a, b)

    # store
    c = acc.to(tl.float16)
    tl.store(C_ptr + batch_id * stride_C_batch
                       + offs_m[:, None] * stride_C_m
                       + offs_n[None, :] * stride_C_n,
             c,
             mask=mask_m[:, None] & mask_n[None, :])
```

"""
Optimized “outgoing” TriMul forward pass (AlphaFold‑3).

Algorithm
---------
1️⃣ Layer‑norm the input (float32 → fp16).  
2️⃣ Fuse the four linear projections and the three gate‑linears into a single
   matrix‑multiply:  X·W_big where W_big has shape (5·hidden, dim).  The result
   is reshaped to [B, N, N, 5, hidden] and sliced into
   left_proj, right_proj, left_gate, right_gate, out_gate.  
3️⃣ Optional mask (cast to fp16) is applied to left/right before gating.  
4️⃣ Gating = sigmoid(linear) and in‑place multiplication with the projected
   tensors.  
5️⃣ Core O(N³) contraction is a batched GEMM
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   This is implemented as a custom Triton kernel (batched mat‑mul with
   C = A @ Bᵀ) that works on fp16 tensors.  
6️⃣ Hidden‑dim layer‑norm, multiplication by out‑gate and final linear
   projection back to the original dimension.  
7️⃣ Cast the result back to float32.

The heavy linear work is performed with a single GEMM, reducing kernel launch
overheads, while the O(N³) contraction is carried out by a fused Triton kernel.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel: batched GEMM C = A @ Bᵀ
#   A: [batch, M, K]   (fp16)
#   B: [batch, K, N]   (fp16)
#   C: [batch, M, N]   (fp16)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_A_batch, stride_A_m, stride_A_k,
    stride_B_batch, stride_B_k, stride_B_n,
    stride_C_batch, stride_C_m, stride_C_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m   = tl.program_id(1)
    pid_n   = tl.program_id(2)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K tiles
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a = tl.load(
            A_ptr
            + batch_id * stride_A_batch
            + offs_m[:, None] * stride_A_m
            + offs_k[None, :] * stride_A_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            B_ptr
            + batch_id * stride_B_batch
            + offs_k[:, None] * stride_B_k
            + offs_n[None, :] * stride_B_n,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate a @ b
        acc += tl.dot(a, b)

    # Store result (down‑cast to fp16)
    c = acc.to(tl.float16)
    tl.store(
        C_ptr
        + batch_id * stride_C_batch
        + offs_m[:, None] * stride_C_m
        + offs_n[None, :] * stride_C_n,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_gemm_triton(A: torch.Tensor, B: torch.Tensor, C: torch.Tensor) -> None:
    """
    Compute C = A @ Bᵀ for a batch of matrices using the Triton kernel above.
    A: [B, M, K]   fp16
    B: [B, K, N]   fp16
    C: [B, M, N]   fp16 (output)
    All tensors must be on the same CUDA device.
    """
    assert A.is_cuda and B.is_cuda and C.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16 and C.dtype == torch.float16

    batch, M, K = A.shape
    _, Kb, N = B.shape
    assert K == Kb, "Inner dimensions must match"

    stride_A_batch, stride_A_m, stride_A_k = A.stride()
    stride_B_batch, stride_B_k, stride_B_n = B.stride()
    stride_C_batch, stride_C_m, stride_C_n = C.stride()

    # Tunable tile sizes – work well for H100 fp16 tensor cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_kernel[
        grid
    ](
        A,
        B,
        C,
        stride_A_batch,
        stride_A_m,
        stride_A_k,
        stride_B_batch,
        stride_B_k,
        stride_B_n,
        stride_C_batch,
        stride_C_m,
        stride_C_n,
        M,
        N,
        K,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )


# ----------------------------------------------------------------------
# Entrypoint required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor[B, N, N]       (float32, bool or all‑ones)
        - weights_dict : mapping name → torch.Tensor (float32, on same device as input)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast once to fp16
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 3️⃣ Single fused linear projection for all 5 heads
    #    (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices along the output‑feature axis
    big_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape (5*hidden_dim, dim)

    B, N, _, _ = x_fp16.shape
    # Flatten [B, N, N, dim] → [B*N*N, dim] for a single GEMM
    x_flat = x_fp16.view(-1, dim)                     # (B*N*N, dim)
    proj = torch.nn.functional.linear(x_flat, big_weight)  # (B*N*N, 5*hidden_dim)

    # Reshape to [B, N, N, 5, hidden_dim] – a memory‑view, no additional copy
    proj = proj.view(B, N, N, 5, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Slice the fused projection
    # ------------------------------------------------------------------
    left      = proj[..., 0, :]  # [B, N, N, hidden]
    right     = proj[..., 1, :]
    left_gate = proj[..., 2, :]
    right_gate= proj[..., 3, :]
    out_gate  = proj[..., 4, :]

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left  = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # Fuse gates (in‑place multiplication)
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ O(N³) contraction via custom Triton batched GEMM
    #
    #    out_hidden[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    # ------------------------------------------------------------------
    # Rearrange tensors to batch‑wise GEMM: A = left[i,k], B = right[k,j]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, hidden, N, N] (k‑j order)

    batch_total = B * hidden_dim
    left_mat  = left_perm.view(batch_total, N, N)   # [B*hidden, N, N]
    right_mat = right_perm.view(batch_total, N, N)  # [B*hidden, N, N]
    out_mat   = torch.empty_like(left_mat)          # fp16 output placeholder

    _batched_gemm_triton(left_mat, right_mat, out_mat)

    # Restore the original 4‑D layout: [B, N, N, hidden]
    out_hidden = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension (fp16)
    # ------------------------------------------------------------------
    out_hidden = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate and final projection back to original dim
    # ------------------------------------------------------------------
    out_hidden = out_hidden * out_gate
    out = torch.nn.functional.linear(
        out_hidden, weights["to_out.weight"].to(torch.float16)
    )  # [B, N, N, dim]  (fp16)

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to float32 as required by the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
