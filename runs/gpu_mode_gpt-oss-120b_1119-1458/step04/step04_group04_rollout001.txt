def mask_gate_fuse_kernel(
    x_ptr, gate_ptr, mask_ptr,
    hidden_dim, size,

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def mask_gate_fuse_kernel(x_ptr, gate_ptr, mask_ptr,
                          hidden_dim, size,
                          BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    mask_offs = offs // hidden_dim
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=1.0)
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)

def _batched_matmul_torch(left: torch.Tensor,
                          right: torch.Tensor) -> torch.Tensor:
    # left/right: [B,N,N,H] fp16
    B, N, _, H = left.shape
    # reshape to [B*H,N,N]
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # bmm expects (B*H,N,N) @ (B*H,N,N).T
    c = torch.bmm(a, b.transpose(1, 2))
    # reshape back to [B,N,N,H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Triton‑accelerated implementation of the “outgoing” TriMul operator
    (AlphaFold‑3).  The heavy O(N³) pairwise contraction is performed with a
    batched GEMM (torch.bmm).  Optional mask‑gate fusion is executed by a
    small Triton kernel to demonstrate custom‑kernel usage.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ---------- 1. Input layer‑norm (float32) ----------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B,N,N,dim] float32

    # ---------- 2. Cast to fp16 ----------
    x = x.to(torch.float16)

    # ---------- 3. Cast model weights ----------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ---------- 4. Linear projections ----------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ---------- 5. Gating signals ----------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ---------- 6. Optional mask fusion ----------
    if not nomask:
        # mask: [B,N,N] → half
        mask_fp16 = mask_tensor.to(torch.float16)
        size = left.numel()
        BLOCK = 1024                       # ≤ 1024 threads per block
        grid = ( (size + BLOCK - 1) // BLOCK, )

        # fuse left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](

            hidden_dim, size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # fuse right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right, right_gate, mask_fp16,
            hidden_dim, size,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        left = left * left_gate
        right = right * right_gate

    # ---------- 7. O(N³) contraction ----------
    out = _batched_matmul_torch(left, right)   # [B,N,N,hidden] fp16

    # ---------- 8. Second layer‑norm ----------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ---------- 9. Final gating & projection ----------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # ---------- 10. Cast back to float32 ----------
    return out.to(torch.float32)

"""
Triton‑accelerated outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32) → cast to fp16.
2. Linear projections (left/right) and three gate‑projections (all fp16).
3. If a mask is present, a tiny Triton kernel fuses the three
   element‑wise multiplications  left = left * left_gate * mask
   and right = right * right_gate * mask (the mask is broadcast over the
   hidden dimension inside the kernel).
   When there is no mask we fall back to a plain PyTorch elementwise mul.
4. Heavy O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM (torch.bmm) on tensors reshaped to
   [B·H, N, N]; this leverages cuBLAS Tensor‑Core kernels on H100.
5. Second LayerNorm (fp16) → out‑gate (sigmoid) → element‑wise gating.
6. Final linear projection back to the original dimension and cast to
   float32 (required by the harness).

Only the forward pass is implemented – no gradients are needed.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel for fused multiplication with broadcasting mask.
# left (or right) has shape (B, N, N, H) → flattened size = B*N*N*H.
# mask has shape (B, N, N) and is broadcast on the hidden dim.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16  : tensor to be updated (left or right)
    gate_ptr,       # *fp16  : gate tensor (same shape as x)
    mask_ptr,       # *fp16  : mask tensor (shape B·N·N)
    hidden_dim,     # i32    : size of hidden dimension H
    size,           # i32    : total number of elements in x
    BLOCK: tl.constexpr,  # compile‑time tile size (≤1024)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices of this block
    active = offs < size                               # boundary guard

    # index of the mask element that corresponds to this x element.
    # because the hidden dimension is the innermost stride, we can obtain
    # the mask index by integer division.
    mask_offs = offs // hidden_dim

    x   = tl.load(x_ptr   + offs, mask=active, other=0.0)
    g   = tl.load(gate_ptr+ offs, mask=active, other=0.0)
    m   = tl.load(mask_ptr+ mask_offs, mask=active, other=1.0)   # mask = 1 outside valid range

    out = x * g * m
    tl.store(x_ptr + offs, out, mask=active)


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM implementing the O(N³) contraction.
    left/right: [B, N, N, H] (fp16)
    returns:    [B, N, N, H] (fp16)
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] where each hidden channel becomes a separate batch
    a = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    b = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # bmm with the second operand transposed – this is exactly the einsum in the reference
    c = torch.bmm(a, b.transpose(1, 2))               # [B*H, N, N]
    # restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out
