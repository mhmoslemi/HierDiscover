    B, N, _, H = left.shape
    # left: [B,N,N,H]
    # right: [B,N,N,H]
    # Convert to shape (B*H,N,N) as described
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()
    left_flat = left_perm.view(-1, N, N)
    right_flat = right_perm.view(-1, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # constants
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # compute strides
    M = N
    K = N
    N_dim = N

    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    batch_stride_a = left_flat.stride(0)
    batch_stride_b = right_flat.stride(0)
    batch_stride_c = out_flat.stride(0)

    grid = ( (M + BLOCK_M - 1)//BLOCK_M,
             (N_dim + BLOCK_N - 1)//BLOCK_N,
             left_flat.shape[0] )  # batch_h

        left_flat,
        right_flat,
        out_flat,
        M, N_dim, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=3,
    )
    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

"""
custom_kernel – Efficient “outgoing” TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  **First LayerNorm (float32)** – stabilises the input.
2️⃣  **FP16 conversion** – the heavy linear & gate operations are performed in half‑precision.
3️⃣  **Fused projection + gate** – `left_proj` and `left_gate` (resp. `right_proj`,
    `right_gate`) are computed with a single `Linear` by concatenating the two weight
    matrices; the sigmoid gate is applied on‑the‑fly.
4️⃣  **Optional mask** – broadcasted and multiplied after gating (order‑preserving).
5️⃣  **Batched GEMM C = A @ Bᵀ** – the O(N³) contraction is performed by a custom
    Triton kernel that computes many independent matrix‑multiplications
    `C[b·h] = left[b·h] @ right[b·h]ᵀ`.  The kernel works on FP16 inputs
    and accumulates in FP32 for numerical safety.
6️⃣  **Second LayerNorm over the hidden dimension** (still FP16).
7️⃣  **Output gating** and **final projection** back to the original `dim`.
8️⃣  **Cast back to float32** for the harness.

The heavy matrix multiplication is the dominant cost; using an explicitly
written Triton kernel removes the extra memory traffic of materialising the
intermediate `left`/`right` tensors for each hidden slice and fully exploits
Tensor‑Cores on H100.  All other operations are inexpensive and stay in Torch.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel – batched GEMM C = A @ Bᵀ
#   A : [BH, M, K]   (fp16)
#   B : [BH, K, N]   (fp16)  – already transposed (right side)
#   C : [BH, M, N]   (fp16)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)          # index over B * hidden_dim
    pid_m = tl.program_id(0)              # output row block
    pid_n = tl.program_id(1)              # output col block

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the start of the tile for A and B
    a_ptrs = (
        A
        + pid_batch * batch_stride_a
        + offs_m[:, None] * stride_am
        + offs_k[None, :] * stride_ak
    )
    b_ptrs = (
        B
        + pid_batch * batch_stride_b
        + offs_k[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )

    # Accumulator in float32 (Tensor‑Core friendly)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks of size BLOCK_K
    num_k = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k):
        a = tl.load(
            a_ptrs,
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )
        acc += tl.dot(a, b)            # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write out the result, casting back to fp16
    c = acc.to(tl.float16)
    c_ptrs = (
        C
        + pid_batch * batch_stride_c
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        c,
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication `out = left @ rightᵀ` implemented in Triton.
    Input / output are fp16 tensors of shape [B, N, N, H].
    """
    B, N, _, H = left.shape
    # (B, H, N, N) – note the right side must be transposed on the last two axes
    left_blk = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_blk = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    # Output buffer
    out_blk = torch.empty_like(left_blk)

    # ------------------------------------------------------------------
    # Strides in element units (not bytes)
    # ------------------------------------------------------------------
    stride_am = left_blk.stride(1)     # stride for M dimension
    stride_ak = left_blk.stride(2)     # stride for K dimension

    stride_bk = right_blk.stride(1)    # stride for K dimension in B
    stride_bn = right_blk.stride(2)    # stride for N dimension in B

    stride_cm = out_blk.stride(1)      # stride for M dimension in C
    stride_cn = out_blk.stride(2)      # stride for N dimension in C

    batch_stride_a = left_blk.stride(0)
    batch_stride_b = right_blk.stride(0)
    batch_stride_c = out_blk.stride(0)

    # ------------------------------------------------------------------
    # Launch parameters – these were found to be a good trade‑off on H100.
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _batched_matmul_kernel[grid](
        left_blk,
        right_blk,
        out_blk,
        N,               # M
        N,               # N
        N,               # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,     # enough warps for H100 Tensor‑Core utilisation
        num_stages=3,
    )

    # Reshape back to the original 4‑D layout
    out = (
        out_blk.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor[B, N, N]       (float32 / bool, may be all‑ones)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``; optional ``nomask``.

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the compute‑heavy portion
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gate weights (2*h × dim) to reduce memory traffic
    # ------------------------------------------------------------------
    left_w_proj = weights["left_proj.weight"].to(torch.float16)
    left_w_gate = weights["left_gate.weight"].to(torch.float16)
    right_w_proj = weights["right_proj.weight"].to(torch.float16)
    right_w_gate = weights["right_gate.weight"].to(torch.float16)

    left_concat = torch.cat([left_w_proj, left_w_gate], dim=0)   # (2H, dim)
    right_concat = torch.cat([right_w_proj, right_w_gate], dim=0)  # (2H, dim)

    # ------------------------------------------------------------------
    # 4️⃣ Compute projection + gate in a single Linear call
    # ------------------------------------------------------------------
    left_all = torch.nn.functional.linear(x_fp16, left_concat)   # [B, N, N, 2H]
    right_all = torch.nn.functional.linear(x_fp16, right_concat)

    left_proj = left_all[..., :hidden_dim]
    left_gate = torch.sigmoid(left_all[..., hidden_dim:])
    right_proj = right_all[..., :hidden_dim]
    right_gate = torch.sigmoid(right_all[..., hidden_dim:])

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcasted on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B, N, N]; broadcast to [B, N, N, 1] and cast to fp16
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ Output gate (same as reference)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_fp16, weights["out_gate.weight"].to(torch.float16))
    )  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7️⃣ O(N³) contraction via custom Triton batched GEMM (A @ Bᵀ)
    # ------------------------------------------------------------------
    out = _batched_matmul_triton(left, right)  # [B, N, N, H] in fp16

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension (fp16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(torch.float16)
    )  # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to float32 (required by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
