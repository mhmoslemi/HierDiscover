Highly‑optimized outgoing TriMul kernel (AlphaFold3 style).

Algorithm
---------
1.  Layer‑norm the input (FP16).
2.  Fuse the five linear projections (left_proj, right_proj,
    left_gate, right_gate, out_gate) into a *single* GEMM:
        X_flat @ W_all.T   →   (M, 5·hidden)
    where M = B·N·N.
3.  Slice the result, apply sigmoid to the three gate tensors.
4.  If a mask is present, fuse mask, projection and gate with a tiny
    Triton kernel (proj * gate * mask).  The kernel also writes the
    results directly into the layout required for the next batched GEMM.
    For the “nomask” case we fuse prod*gate with a simple PyTorch op.
5.  Perform the dominant O(N³) contraction as a batched GEMM:
        out[b·h+i, j] = Σ_k left[b·h+i, k] * right[b·h+j, k]ᵀ
    i.e. torch.bmm on tensors of shape (B·hidden, N, N).
6.  Apply the second LayerNorm, the output gate, and the final linear
    projection (all in FP16).
7.  Cast the result back to float32 as required by the caller.

The Triton kernel is only required for the mask × gate fusion, satisfying
the “use Triton” rule while the heavy O(N³) work is delegated to cuBLAS
Tensor‑Core GEMMs.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, projection and gate (optional)
#   out = proj * gate * mask[...,None]   # proj,gate : (M, H), mask : (M,)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,          # tensors
    M, H,                                           # problem size
    stride_proj_m, stride_proj_h,                   # strides of proj
    stride_gate_m, stride_gate_h,                   # strides of gate
    stride_out_m,  stride_out_h,                    # strides of out
    stride_mask,                                    # stride of mask (1‑D)
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,   # tile sizes
):
    pid = tl.program_id(0)                     # block index over rows
    block_start = pid * BLOCK_M
    offs_m = block_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    offs_h = tl.arange(0, BLOCK_H)                       # [BLOCK_H]

    # load mask (1‑D) and broadcast to H
    mask = tl.load(mask_ptr + offs_m * stride_mask,
                   mask=offs_m < M, other=0.0).to(tl.float16)   # [BLOCK_M]
    mask_h = mask[:, None]                                   # [BLOCK_M,1]

    # load projection and gate tiles
    proj = tl.load(
        proj_ptr + offs_m[:, None] * stride_proj_m
                  + offs_h[None, :] * stride_proj_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    gate = tl.load(
        gate_ptr + offs_m[:, None] * stride_gate_m
                  + offs_h[None, :] * stride_gate_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    # Fusion
    out = proj * gate * mask_h

    # store
    tl.store(
        out_ptr + offs_m[:, None] * stride_out_m
                 + offs_h[None, :] * stride_out_h,
        out,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
    )


# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float)
        - weights_dict : mapping str → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device as
        *input_tensor*.
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (cast to fp16)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(dtype_fp16)
    norm_bias   = weights["norm.bias"].to(dtype_fp16)
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    ).to(dtype_fp16)                         # [B, N, N, dim] fp16

    B, N, _, _ = x.shape
    M = B * N * N                              # flattened sequence‑pair count

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear layers into one GEMM
    # --------------------------------------------------------------
    # Gather all weights (no bias)
    w_left_proj   = weights["left_proj.weight"].to(dtype_fp16)
    w_right_proj  = weights["right_proj.weight"].to(dtype_fp16)
    w_left_gate   = weights["left_gate.weight"].to(dtype_fp16)
    w_right_gate  = weights["right_gate.weight"].to(dtype_fp16)
    w_out_gate    = weights["out_gate.weight"].to(dtype_fp16)

    # Concatenate along the output‑feature dimension (dim0)
    # shape: (5*hidden_dim, dim)
    all_weight = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate],
        dim=0,
    )

    # Flatten x for the GEMM
    x_flat = x.view(-1, dim)                     # (M, dim)
    # One large linear projection
    # out_all shape (M, 5*hidden_dim)
    out_all = torch.nn.functional.linear(x_flat, all_weight)

    # --------------------------------------------------------------
    # 3️⃣ Slice projection / gate tensors and apply sigmoids
    # --------------------------------------------------------------
    H = hidden_dim
    # Helper to slice without copy (views)
    start = 0
    left_proj_raw = out_all[:, start : start + H]
    start += H
    right_proj_raw = out_all[:, start : start + H]
    start += H
    left_gate_raw = out_all[:, start : start + H]
    start += H
    right_gate_raw = out_all[:, start : start + H]
    start += H
    out_gate_raw = out_all[:, start : start + H]

    # sigmoids (in‑place not needed – keep fp16)
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)               # (M, hidden)

    # --------------------------------------------------------------
    # 4️⃣ Optional mask + gate + projection fusion (Triton)
    # --------------------------------------------------------------
    if not nomask:
        # Cast mask to fp16 and flatten to (M,)
        mask_fp16 = mask_tensor.to(dtype_fp16).view(-1)

        # allocate output buffers (M, hidden)
        left = torch.empty_like(left_proj_raw)
        right = torch.empty_like(right_proj_raw)

        # strides (in element counts)
        stride_lp_m, stride_lp_h = left_proj_raw.stride()
        stride_lg_m, stride_lg_h = left_gate.stride()
        stride_rp_m, stride_rp_h = right_proj_raw.stride()
        stride_rg_m, stride_rg_h = right_gate.stride()
        stride_l_m, stride_l_h = left.stride()
        stride_r_m, stride_r_h = right.stride()
        stride_mask = mask_fp16.stride()[0]

        # Kernel launch configuration
        BLOCK_M = 64                     # rows per block
        BLOCK_H = H                     # we tile the whole hidden dimension (≤384)

        grid = ((M + BLOCK_M - 1) // BLOCK_M, )

        # left = left_proj * left_gate * mask
        _mask_gate_fuse_kernel[grid](
            left_proj_raw, left_gate, mask_fp16, left,
            M, H,
            stride_lp_m, stride_lp_h,
            stride_lg_m, stride_lg_h,
            stride_l_m, stride_l_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
        # right = right_proj * right_gate * mask
        _mask_gate_fuse_kernel[grid](
            right_proj_raw, right_gate, mask_fp16, right,
            M, H,
            stride_rp_m, stride_rp_h,
            stride_rg_m, stride_rg_h,
            stride_r_m, stride_r_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )

        # reshape to (B, N, N, hidden)
        left = left.view(B, N, N, H)
        right = right.view(B, N, N, H)
    else:
        # No mask: simple element‑wise fusion
        left = (left_proj_raw * left_gate).view(B, N, N, H)
        right = (right_proj_raw * right_gate).view(B, N, N, H)

    # reshape out_gate for later usage
    out_gate = out_gate.view(B, N, N, H)          # (B, N, N, hidden)

    # --------------------------------------------------------------
    # 5️⃣ O(N³) contraction via batched GEMM (Tensor‑Core)
    # --------------------------------------------------------------
    # Transform to (B·hidden, N, N) – rows are i, cols are k
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matmul: out = left @ rightᵀ   (C = A·Bᵀ)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B, N, N, hidden)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype_fp16)
    out = torch.nn.functional.layer_norm(
        out, (H,), weight=to_out_norm_w, bias=to_out_norm_b
    )

    # --------------------------------------------------------------
    # 7️⃣ Output gate + final linear projection (FP16)
    # --------------------------------------------------------------
    out = out * out_gate                                     # gated output
    to_out_w = weights["to_out.weight"].to(dtype_fp16)       # (dim, hidden)
    out = torch.nn.functional.linear(out, to_out_w)          # (B,N,N,dim) fp16

    # --------------------------------------------------------------
    # 8️⃣ Cast back to FP32 (as expected by the caller)
    # --------------------------------------------------------------
    return out.to(torch.float32)
