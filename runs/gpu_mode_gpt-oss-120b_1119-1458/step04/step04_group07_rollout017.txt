from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : [M, N, K]   (M = batch * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
#   All tensors are fp16, accumulation in fp32.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # scalar sizes
    stride_am, stride_ai, stride_ak,   # strides for A (M,i,k)
    stride_bm, stride_bj, stride_bk,   # strides for B (M,j,k)
    stride_cm, stride_ci, stride_cj,   # strides for C (M,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # program IDs (i‑tile, j‑tile, batch‑hidden index)
    # ------------------------------------------------------------------
    pid_i = tl.program_id(0)   # tile along first N dimension (i)
    pid_j = tl.program_id(1)   # tile along second N dimension (j)
    pid_z = tl.program_id(2)   # batch * hidden index (M)

    # ------------------------------------------------------------------
    # Compute offsets for this tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension (sequence dim) in tiles
    # ------------------------------------------------------------------
    for k_start in range(0, K, BLOCK_K):
        cur_k = k_start
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile (M, i, k)  → shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # Load B tile (M, j, k)  → shape (BLOCK_K, BLOCK_N)  (transposed view)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Mixed‑precision dot product (fp16×fp16 → fp32)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result (convert back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(
    left: torch.Tensor, right: torch.Tensor
) -> torch.Tensor:
    """
    Compute ``out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`` via the
    Triton kernel above.  ``left`` and ``right`` must be half‑precision
    contiguous tensors with shape ``[B, N, N, H]``.
    """
    B, N, _, H = left.shape

    # reshape to [B*H, N, N] – each hidden head becomes an independent batch
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A, dtype=torch.float16)

    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # select tile size – 128×128 works best for N ≥ 512,
    # otherwise use 64×64 tiles.
    if N >= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden
    )
    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operator (bias‑free, no grads).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]      (float32)
        - mask_tensor  : torch.Tensor [B, N, N]          (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack input
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype_fp16 = torch.float16

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                                            # [B, N, N, dim]  float32

    # -------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy compute
    # -------------------------------------------------
    x = x.to(dtype_fp16)

    # -------------------------------------------------
    # 3️⃣ Fuse all linear projections (proj + gate + out_gate) into ONE matmul
    # -------------------------------------------------
    # Cast all required weights once to fp16
    w_left_proj = weights["left_proj.weight"].to(dtype_fp16)
    w_left_gate = weights["left_gate.weight"].to(dtype_fp16)
    w_right_proj = weights["right_proj.weight"].to(dtype_fp16)
    w_right_gate = weights["right_gate.weight"].to(dtype_fp16)
    w_out_gate = weights["out_gate.weight"].to(dtype_fp16)

    # Concatenate along the output‑feature dimension:
    # order -> left_proj | left_gate | right_proj | right_gate | out_gate
    weight_concat = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate],
        dim=0,
    )  # shape [5*hidden_dim, dim]

    # Single fused linear: shape [B, N, N, 5*hidden_dim]
    fused = torch.nn.functional.linear(x, weight_concat)

    H = hidden_dim
    # Slice the concatenated result
    left_proj = fused[..., 0:H]                                   # [B,N,N,H]
    left_gate = torch.sigmoid(fused[..., H:2 * H])                # [B,N,N,H]
    right_proj = fused[..., 2 * H:3 * H]                          # [B,N,N,H]
    right_gate = torch.sigmoid(fused[..., 3 * H:4 * H])           # [B,N,N,H]
    out_gate = torch.sigmoid(fused[..., 4 * H:5 * H])            # [B,N,N,H]

    # -------------------------------------------------
    # 4️⃣ Apply gating and optional mask (still fp16)
    # -------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    if not nomask:
        mask_fp16 = mask_tensor.to(dtype_fp16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction (batched GEMM) – stays in fp16
    # -------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = _batched_matmul_transpose_fp16(left, right)          # [B,N,N,H] fp16

    # -------------------------------------------------
    # 6️⃣ Second LayerNorm (still fp16)
    # -------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(dtype_fp16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(dtype_fp16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # -------------------------------------------------
    # 7️⃣ Final gating (out_gate) and projection back to dim
    # -------------------------------------------------
    out = out * out_gate                                         # shape [B,N,N,H] fp16
    to_out_weight = weights["to_out.weight"].to(dtype_fp16)     # shape [dim, H]
    out = torch.nn.functional.linear(out, to_out_weight)       # [B,N,N,dim] fp16

    # -------------------------------------------------
    # 8️⃣ Cast back to float32 for the external API
    # -------------------------------------------------
    return out.to(torch.float32)
```
