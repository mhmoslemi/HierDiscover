"""
Outgoing TriMul kernel (AlphaFold‑3) – fused linear projections + BMM.

Algorithm
---------
1️⃣  Layer‑norm (float32) the input tensor and cast to fp16.
2️⃣  Fuse the five linear projections (left_proj, left_gate, right_proj,
    right_gate, out_gate) into a single matrix‑multiply using a custom
    Triton kernel.  The fused weight has shape (5·hidden_dim, dim) and
    the kernel computes `Y = X @ W^T` in half‑precision with fp32
    accumulation.
3️⃣  Slice the fused result, apply `sigmoid` to the three gate slices,
    optionally multiply by the mask and the gates:
        left  = left_proj  * left_gate  * mask
        right = right_proj * right_gate * mask
4️⃣  Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is performed as a batched GEMM (`torch.bmm`), i.e. a matrix‑
    multiply for each head (`B·hidden_dim` batches).  This uses
    cuBLAS (Tensor‑Core) and is far faster than a python loop.
5️⃣  Second Layer‑norm (fp16) on the hidden dimension, element‑wise
    multiplication by the out‑gate, and a final linear projection back
    to `dim`.
6️⃣  Cast the final output to float32.

All heavy arithmetic is in fp16 (with fp32 accumulation) to exploit
Tensor‑Cores on H100.  The custom Triton kernel is used for the fused
linear projection, ensuring the large input tensor is read only once.
"""

import math
from typing import Tuple, Dict, Any

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused linear projection (Y = X @ W^T) for the 5×gate
# ----------------------------------------------------------------------
@triton.jit
def _fused_linear_kernel(
    X_ptr,          # *fp16  [M, K]   (input after LN)
    W_ptr,          # *fp16  [K, N]   (weight^T, where N=5·hidden_dim)
    Y_ptr,          # *fp16  [M, N]   (output)
    M, N, K,        # i32 dimensions
    stride_xm, stride_xk,          # X strides (M, K)
    stride_wk, stride_wn,          # W strides (K, N)
    stride_ym, stride_yn,          # Y strides (M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: Y = X @ W  (fp16 inputs, fp32 accumulation)."""
    pid_m = tl.program_id(0)  # tile along M dimension
    pid_n = tl.program_id(1)  # tile along N dimension

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load X tile (M x K)
        x_ptrs = X_ptr + offs_m[:, None] * stride_xm + offs_k[None, :] * stride_xk
        x = tl.load(
            x_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # Load W tile (K x N)
        w_ptrs = W_ptr + offs_k[:, None] * stride_wk + offs_n[None, :] * stride_wn
        w = tl.load(
            w_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )  # fp16

        # Mixed‑precision dot (fp16 × fp16 → fp32)
        acc += tl.dot(x, w)

    # Write back
    y_ptrs = Y_ptr + offs_m[:, None] * stride_ym + offs_n[None, :] * stride_yn
    tl.store(
        y_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _fused_linear_triton(
    X: torch.Tensor,          # [M, K]  fp16, contiguous
    W_T: torch.Tensor,        # [K, N]  fp16, contiguous (W transposed)
) -> torch.Tensor:
    """Wrapper around the fused linear Triton kernel."""
    assert X.is_contiguous() and W_T.is_contiguous()
    M, K = X.shape
    _, N = W_T.shape

    Y = torch.empty((M, N), dtype=torch.float16, device=X.device)

    # strides in element counts (not bytes)
    stride_xm, stride_xk = X.stride()
    stride_wk, stride_wn = W_T.stride()
    stride_ym, stride_yn = Y.stride()

    # Tunable tile sizes – chosen to fit H100 Tensor‑Core SMEM budget
    BLOCK_M = 256
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _fused_linear_kernel[grid](
        X,
        W_T,
        Y,
        M,
        N,
        K,
        stride_xm,
        stride_xk,
        stride_wk,
        stride_wn,
        stride_ym,
        stride_yn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    return Y


# ----------------------------------------------------------------------
# Helper: batched matrix‑multiply for the heavy contraction
# ----------------------------------------------------------------------
def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """Batched GEMM C = A @ Bᵀ using torch.bmm (fp16×fp16→fp32)."""
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)          # [B*H, N, N]
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)        # [B*H, N, N]
    # (B*H, N, N) @ (B*H, N, N)ᵀ -> (B*H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()            # [B, N, N, H]
    return C

        - config_dict  : contains ``dim`` and ``hidden_dim``.

    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]   float32

    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],

    # Reshape back to 4‑D tensor and split into the five components
    Y = Y_flat.view(B, N, N, 5 * hidden_dim)        # [B, N, N, 5*H]
    left_proj, left_gate_pre, right_proj, right_gate_pre, out_gate_pre = torch.split(
        Y, hidden_dim, dim=-1
    )  # each [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 5️⃣ Gates (sigmoid) and optional mask
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction (batched GEMM)
    # ------------------------------------------------------------------
    # Using cuBLAS via torch.bmm – works well for all sequence lengths
    out = _batched_matmul_torch(left, right)  # [B, N, N, hidden_dim] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )  # fp16

    out = out * out_gate  # element‑wise gating (fp16)

    to_out_weight = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_weight)  # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to float32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
