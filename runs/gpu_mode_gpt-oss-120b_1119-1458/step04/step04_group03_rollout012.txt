"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

Algorithm
---------
1️⃣  Layer‑norm the input (fp32) → cast to fp16.
2️⃣  Fuse the two projection linear layers (proj + gate) per side:
    * weight matrices are concatenated, the fused matmul produces both
      the projection and its gate in a single GEMM.
3️⃣  Apply mask (if present) only to the projections, apply sigmoid to the
    gates and multiply (proj * gate).
4️⃣  Compute the heavy O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    as a batched GEMM (A @ Bᵀ).  For seq_len ≤ 256 we use a handcrafted
    Triton kernel (64×64×128 tiles, mixed‑precision fp16→fp32 accumulation);
    for larger sizes we fall back to cuBLAS via torch.bmm.
5️⃣  Second Layer‑norm (fp16) → element‑wise out‑gate (sigmoid) → final
    linear projection back to `dim`.
6️⃣  Cast the result to float32.

The implementation reduces the number of large GEMM calls from six to five
by concatenating projection/gate weights, and uses a low‑overhead Triton
kernel for the contraction on small sequences.  All heavy work stays in
fp16 with fp32 accumulation for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (fp16 → fp16, accumulator fp32)
#   A, B : [M, N, K] (M = B*hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]   (the second operand will be read transposed)
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)  (j is the column after transpose)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel for TriMul contraction.
    Computes C[m,i,j] = Σ_k A[m,i,k] * B[m,j,k]   (note B is accessed transposed).
    """
    pid_m = tl.program_id(0)   # i‑tile index
    pid_n = tl.program_id(1)   # j‑tile index
    pid_z = tl.program_id(2)   # batch‑*‑hidden index (m)

    # Tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator (fp32) – each thread computes a BLOCK_M×BLOCK_N tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & (offs_k[None, :] < K),
                    other=0.0)          # fp16

        # Load B tile transposed: we need B[m, j, k] → we read B[m, k, j]
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk   # k dimension first
            + offs_j[None, :] * stride_bj   # j dimension second
        )
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_j[None, :],
                    other=0.0)          # fp16

        # Mixed‑precision dot (fp16 × fp16 → fp32)
        acc += tl.dot(a, b)

    # Store result back to C
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right (both [B, N, N, H]) using the Triton kernel.
    Returns tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    # reshape to [M, N, N] where M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)          # fp16 output

    # strides in element counts (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()   # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()  # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()   # (N*N, N, 1)

    # Tuned tile sizes – 64×64×128 works well on H100 for fp16 MMA
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 128

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right using torch.bmm (cuBLAS, fp16×fp16→fp32 accumulation).
    """
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)          # [M,N,N]
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)        # [M,N,N]
    # Batched mat‑mul with transposed second operand
    C = torch.bmm(A, B_.transpose(1, 2))                                 # [M,N,N]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()              # [B,N,N,H]
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 / bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 2️⃣ first LayerNorm (float32) – keeps numerical stability
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 3️⃣ cast to fp16 for the rest of the pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ fuse projection + gate weights (both are dim → hidden_dim)
    # ------------------------------------------------------------------
    # left side: [proj, gate] concatenated -> (2*hidden, dim)
    left_w_concat = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    right_w_concat = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)

    # out‑gate (dim → hidden_dim)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # other parameters
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 5️⃣ flatten input for the fused linear projections
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    x_flat = x.reshape(-1, dim)                     # (B*N*N, dim)

    # ------------------------------------------------------------------
    # 6️⃣ fused left & its gate
    # ------------------------------------------------------------------
    left_concat = F.linear(x_flat, left_w_concat)   # (B*N*N, 2*hidden)
    # split
    left_proj = left_concat[:, :hidden_dim].view(B, N, N, hidden_dim)
    left_gate = left_concat[:, hidden_dim:].view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 7️⃣ fused right & its gate
    # ------------------------------------------------------------------
    right_concat = F.linear(x_flat, right_w_concat)  # (B*N*N, 2*hidden)
    right_proj = right_concat[:, :hidden_dim].view(B, N, N, hidden_dim)
    right_gate = right_concat[:, hidden_dim:].view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 8️⃣ out‑gate (dim → hidden_dim)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_w)).view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 9️⃣ optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 10️⃣ apply sigmoid to the gate tensors and fuse with projections
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate      # [B, N, N, hidden]
    right = right_proj * right_gate   # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 11️⃣ heavy O(N³) contraction
    # ------------------------------------------------------------------
    seq_len = N
    if seq_len <= 256:
        # Small sequences – use handcrafted Triton kernel
        out = _batched_matmul_triton(left, right)      # [B, N, N, hidden] (fp16)
    else:
        # Larger sequences – cuBLAS batched gemm (very fast)
        out = _batched_matmul_torch(left, right)       # [B, N, N, hidden] (fp16)

    # ------------------------------------------------------------------
    # 12️⃣ second LayerNorm on hidden dimension (still fp16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 13️⃣ gate the contracted tensor and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate                                 # element‑wise gating
    out = F.linear(out.reshape(-1, hidden_dim), to_out_w) \
          .view(B, N, N, dim)                           # fp16

    # ------------------------------------------------------------------
    # 14️⃣ cast back to fp32 (required by the harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
