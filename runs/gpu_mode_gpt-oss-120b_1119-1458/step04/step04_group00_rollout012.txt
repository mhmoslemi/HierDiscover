import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def mask_gate_fuse_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                          M, H,
                          stride_proj_m, stride_proj_h,
                          stride_gate_m, stride_gate_h,
                          stride_out_m, stride_out_h,
                          stride_mask,
                          BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = tl.arange(0, BLOCK_H)
    mask = tl.load(mask_ptr + offs_m * stride_mask,
                   mask=offs_m < M, other=0.0).to(tl.float16)
    mask_h = mask[:, None]  # broadcast over H

    proj = tl.load(proj_ptr + offs_m[:, None] * stride_proj_m + offs_h[None, :] * stride_proj_h,
                   mask=(offs_m[:, None] < M) & (offs_h[None, :] < H), other=0.0)
    gate = tl.load(gate_ptr + offs_m[:, None] * stride_gate_m + offs_h[None, :] * stride_gate_h,
                   mask=(offs_m[:, None] < M) & (offs_h[None, :] < H), other=0.0)

    out = proj * gate * mask_h

    tl.store(out_ptr + offs_m[:, None] * stride_out_m + offs_h[None, :] * stride_out_h,
             out, mask=(offs_m[:, None] < M) & (offs_h[None, :] < H))

"""Optimized outgoing TriMul forward kernel using fused linear projections and
Triton elementwise mask‑gate fusion.  The heavy O(N³) contraction is performed
with a batched cuBLAS GEMM (`torch.bmm`).  First LayerNorm and all linear
layers run in FP16; final result is cast back to FP32.

Key optimizations:
- Concatenate the projection and gate weight matrices so each position’s
  projection and gate are computed with a single matrix multiplication,
  cutting the number of GEMMs from four to two.
- Fuse mask, projection, and gate elementwise multiplication in a small
  Triton kernel (only required when a mask is present).
- All tensors are kept in half‑precision throughout the compute‑heavy path.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: element‑wise fusion of proj * gate * mask (if present)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    M, H,
    stride_proj_m, stride_proj_h,
    stride_gate_m, stride_gate_h,
    stride_out_m, stride_out_h,
    stride_mask,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)

    start_m = pid * BLOCK_M
    offs_m = start_m + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    offs_h = tl.arange(0, BLOCK_H)                     # [BLOCK_H]

    # Load mask and broadcast
    mask = tl.load(mask_ptr + offs_m * stride_mask,
                   mask=offs_m < M, other=0.0).to(tl.float16)
    mask_h = mask[:, None]  # [BLOCK_M, 1]

    # Load projection and gate tiles
    proj = tl.load(
        proj_ptr + offs_m[:, None] * stride_proj_m
                  + offs_h[None, :] * stride_proj_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    gate = tl.load(
        gate_ptr + offs_m[:, None] * stride_gate_m
                  + offs_h[None, :] * stride_gate_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )

    out = proj * gate * mask_h

    tl.store(
        out_ptr + offs_m[:, None] * stride_out_m
                 + offs_h[None, :] * stride_out_h,
        out,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
    )


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]     (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
        - weights_dict : model weights (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the input device.
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    ).to(torch.float16)   # [B, N, N, dim] in fp16

    B, N, _, _ = x.shape
    M = B * N * N   # total number of pairwise positions

    # ------------------------------------------------------------------
    # 2️⃣ Cast all (bias‑free) linear weights to fp16 once
    # ------------------------------------------------------------------
    # Concatenate projection and gate weights for left and right
    left_concat_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16)
    right_concat_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)

    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Project + gate (single GEMM per side)
    # ------------------------------------------------------------------
    x_flat = x.reshape(-1, dim)   # shape (M, dim)

    # left side: first hidden_dim rows = projection, next = gate
    left_concat = torch.nn.functional.linear(x_flat, left_concat_w)  # (M, 2*hidden)
    left_proj = left_concat[:, :hidden_dim]          # (M, hidden)
    left_gate = torch.sigmoid(left_concat[:, hidden_dim:])  # (M, hidden)

    # right side
    right_concat = torch.nn.functional.linear(x_flat, right_concat_w)
    right_proj = right_concat[:, :hidden_dim]
    right_gate = torch.sigmoid(right_concat[:, hidden_dim:])

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask + element‑wise fuse (Triton)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape -> flat fp16 vector of length M
        mask_fp16 = mask_tensor.to(torch.float16).reshape(-1)   # (M,)

        # Allocate output buffers for fused results
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # Strides (in element counts)
        stride_lp_m, stride_lp_h = left_proj.stride()
        stride_lg_m, stride_lg_h = left_gate.stride()
        stride_lo_m, stride_lo_h = left.stride()
        stride_rp_m, stride_rp_h = right_proj.stride()
        stride_rg_m, stride_rg_h = right_gate.stride()
        stride_ro_m, stride_ro_h = right.stride()
        stride_mask = mask_fp16.stride()[0]

        BLOCK_M = 64
        BLOCK_H = hidden_dim
        grid = ( (M + BLOCK_M - 1) // BLOCK_M, )

        # left = left_proj * left_gate * mask
        _mask_gate_fuse_kernel[grid](
            left_proj, left_gate, mask_fp16, left,
            M, hidden_dim,
            stride_lp_m, stride_lp_h,
            stride_lg_m, stride_lg_h,
            stride_lo_m, stride_lo_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
        # right = right_proj * right_gate * mask
        _mask_gate_fuse_kernel[grid](
            right_proj, right_gate, mask_fp16, right,
            M, hidden_dim,
            stride_rp_m, stride_rp_h,
            stride_rg_m, stride_rg_h,
            stride_ro_m, stride_ro_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Release intermediate tensors to save memory
    del left_proj, left_gate, right_proj, right_gate, left_concat, right_concat

    # Reshape to [B, N, N, hidden]
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ O(N³) contraction via batched GEMM (Tensor‑Core)
    # ------------------------------------------------------------------
    # Move hidden dim to batch dim for batched bmm
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # Re‑assemble to [B, N, N, hidden]

"""
Optimized outgoing TriMul forward kernel (no gradients).

Algorithm
---------
1️⃣   Layer‑norm the input (float32 → fp16).
2️⃣   Fuse each side’s projection and gate into a *single* linear layer by
     concatenating the two weight matrices (proj + gate).  This cuts the
     number of large GEMMs from four to two.
3️⃣   Apply the optional mask together with the element‑wise
     prod(proj, gate) in a tiny Triton kernel, saving separate
     PyTorch broadcasts.
4️⃣   Perform the dominant O(N³) contraction with a batched GEMM
     (`torch.bmm`).  For each hidden channel we compute
     out[i,j] = Σₖ left[i,k] * right[j,k]  →  left @ rightᵀ.
5️⃣   Second LayerNorm on the contracted tensor.
6️⃣   Compute the output‑gate (sigmoid of a linear on the normalized input)
     and finally project back to the original `dim`.
7️⃣   Cast the result back to float32 (the original API type).

All heavy linear work and the contraction run in fp16, exploiting H100
Tensor‑Cores.  A single Triton kernel is used for the mask‑gate fusion,
satisfying the “use Triton” requirement while keeping the critical path
GPU‑bound.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask) fused for a flattened tensor.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    M, H,
    stride_proj_m, stride_proj_h,
    stride_gate_m, stride_gate_h,
    stride_out_m, stride_out_h,
    stride_mask,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)

    start_m = pid * BLOCK_M
    offs_m = start_m + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    offs_h = tl.arange(0, BLOCK_H)                   # [BLOCK_H]

    # Load mask (1‑D) and broadcast to hidden dimension
    mask = tl.load(mask_ptr + offs_m * stride_mask,
                   mask=offs_m < M, other=0.0).to(tl.float16)
    mask_h = mask[:, None]                           # [BLOCK_M, 1]

    # Load projection and gate tiles
    proj = tl.load(
        proj_ptr + offs_m[:, None] * stride_proj_m
                  + offs_h[None, :] * stride_proj_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    gate = tl.load(
        gate_ptr + offs_m[:, None] * stride_gate_m
                  + offs_h[None, :] * stride_gate_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    # Fuse
    out = proj * gate * mask_h

    tl.store(
        out_ptr + offs_m[:, None] * stride_out_m
                 + offs_h[None, :] * stride_out_h,
        out,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
    )


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32 or bool)
        - weights_dict : dict with the model weights (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device as the input.
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    ).to(torch.float16)   # shape [B, N, N, dim] in fp16

    B, N, _, _ = x.shape
    M = B * N * N                     # total pair‑wise positions (flattened)

    # ------------------------------------------------------------------
    # 2️⃣ Cast linear weights to fp16; concatenate projection & gate weights
    # ------------------------------------------------------------------
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                      # (2*hidden, dim)
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                      # (2*hidden, dim)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Projection + gate (one GEMM per side)
    # ------------------------------------------------------------------
    x_flat = x.reshape(-1, dim)                     # (M, dim)

    # Left side: first hidden rows = projection, next hidden rows = gate
    left_concat = torch.nn.functional.linear(x_flat, left_concat_w)   # (M, 2*hidden)
    left_proj = left_concat[:, :hidden_dim]               # (M, hidden)
    left_gate = torch.sigmoid(left_concat[:, hidden_dim:])  # (M, hidden)

    # Right side
    right_concat = torch.nn.functional.linear(x_flat, right_concat_w)
    right_proj = right_concat[:, :hidden_dim]
    right_gate = torch.sigmoid(right_concat[:, hidden_dim:])

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask + element‑wise fuse (Triton)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).reshape(-1)   # (M,)
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # Strides (in element counts, not bytes)
        stride_lp_m, stride_lp_h = left_proj.stride()
        stride_lg_m, stride_lg_h = left_gate.stride()
        stride_lo_m, stride_lo_h = left.stride()
        stride_rp_m, stride_rp_h = right_proj.stride()
        stride_rg_m, stride_rg_h = right_gate.stride()
        stride_ro_m, stride_ro_h = right.stride()
        stride_mask = mask_fp16.stride()[0]

        BLOCK_M = 64
        BLOCK_H = hidden_dim
        grid = ((M + BLOCK_M - 1) // BLOCK_M,)

        # left = left_proj * left_gate * mask
        _mask_gate_fuse_kernel[grid](
            left_proj, left_gate, mask_fp16, left,
            M, hidden_dim,
            stride_lp_m, stride_lp_h,
            stride_lg_m, stride_lg_h,
            stride_lo_m, stride_lo_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
        # right = right_proj * right_gate * mask
        _mask_gate_fuse_kernel[grid](
            right_proj, right_gate, mask_fp16, right,
            M, hidden_dim,
            stride_rp_m, stride_rp_h,
            stride_rg_m, stride_rg_h,
            stride_ro_m, stride_ro_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Release intermediate tensors early to free memory
    del left_proj, left_gate, right_proj, right_gate, left_concat, right_concat

    # Reshape to [B, N, N, hidden]
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ O(N³) contraction via batched GEMM (Tensor‑Core)
    # ------------------------------------------------------------------
    # Bring hidden dimension into the batch axis for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Compute out[i,j] = Σₖ left[i,k] * right[j,k]  → left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Output gate + final projection
    # ------------------------------------------------------------------
    out_gate_flat = torch.nn.functional.linear(x_flat, out_gate_w)   # (M, hidden)
    out_gate = torch.sigmoid(out_gate_flat).view(B, N, N, hidden_dim)

    out = torch.nn.functional.linear(out, to_out_w)                 # (B,N,N,dim) in fp16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to fp32 for the caller
    # ------------------------------------------------------------------
    return out.to(torch.float32)
