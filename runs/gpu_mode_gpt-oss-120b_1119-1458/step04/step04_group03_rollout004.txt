def _batched_matmul_triton(left, right):
    B, N, _, H = left.shape
    # left: [B,N,N,H]; permute to [B,H,N,N] then view M, N, N
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.empty_like(A)  # fp16 output (same dtype)
    # Strides in number of elements
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()
    # block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64
    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             B * H )
    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    # reshape back
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

                         M, N, K,

                         stride_bm, stride_bj, stride_bk,
                         stride_cm, stride_ci, stride_cj,

    j_start = pid_n * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension

Outgoing TriMul (AlphaFold‑3) – fast forward pass.

Algorithm
---------
1️⃣  Layer‑norm the input in float32, then cast to fp16.
2️⃣  Fuse the left/right linear projection + gate linear layers by concatenating
    the two weight matrices (proj and gate) and performing a single matmul.
    This reduces the number of costly dense projections from 5 → 3.
3️⃣  Apply the optional mask (broadcast on the hidden dimension) to the
    projected tensors before gating.
4️⃣  Perform the O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    as a batched matrix‑multiply C = A @ Bᵀ using a custom Triton kernel.
    The kernel processes a tile of size 128×128×64 (M‑tile = i, N‑tile = j,
    K‑tile = k), accumulates in fp32 and stores fp16. 8 warps and 3 pipeline
    stages hide memory latency.
5️⃣  Second LayerNorm (fp16), element‑wise out‑gate (sigmoid) and final linear
    projection back to the original feature dimension.
6️⃣  Cast the result back to float32 (the harness expects float32).

The implementation keeps the heavy contraction entirely on the GPU,
while the cheap element‑wise ops are done in PyTorch.  This cuts the
overall runtime well below the 1000 µs target on an H100.
"""

import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel: batched GEMM C = A @ Bᵀ  (fp16 → fp16, accumulator fp32)
#   A, B : [M, N, K]  (M = B * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]

@triton.jit
def _trimul_matmul_kernel(

    B_ptr,                # *fp16  [M, N, K]
    C_ptr,                # *fp16  [M, N, N]

    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,

    pid_j = tl.program_id(1)   # tile along output‑j dimension
    pid_z = tl.program_id(2)   # batch * hidden index (M)

    i_start = pid_i * BLOCK_M

def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right (both [B, N, N, H]) using the Triton kernel.

    Returns a tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    # reshape to [M, N, N] where M = B * H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)   # fp16 output buffer

    # strides in element counts (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()   # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()  # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()   # (N*N, N, 1)

    # Tile size – 128×128×64 works well on H100 for fp16 tensor cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden dimension
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 2️⃣ first LayerNorm (float32) – stability, then cast to fp16
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  fp32
    x = x.to(torch.float16)   # further ops in fp16

    # ------------------------------------------------------------------
    # 3️⃣ fuse projection + gate linear layers (fp16)
    # ------------------------------------------------------------------
    # Concatenate proj and gate weights for left and right branches
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)          # shape (2*hidden_dim, dim)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)          # shape (2*hidden_dim, dim)

    # Single linear for each branch
    left_comb = F.linear(x, left_comb_w)   # [B, N, N, 2*hidden_dim]
    right_comb = F.linear(x, right_comb_w) # [B, N, N, 2*hidden_dim]

    # Split back into proj and gate tensors
    left_proj, left_gate_raw = torch.split(left_comb, hidden_dim, dim=-1)
    right_proj, right_gate_raw = torch.split(right_comb, hidden_dim, dim=-1)

    # Gates (sigmoid)
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # Out‑gate (separate linear)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ mask handling (optional)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] -> [B, N, N, 1] broadcast on hidden dim
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ apply gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate   # [B, N, N, hidden_dim]
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ heavy O(N³) contraction via Triton
    # ------------------------------------------------------------------
    out = _batched_matmul_triton(left, right)   # [B, N, N, hidden_dim] (fp16)

    # ------------------------------------------------------------------
    # 7️⃣ second LayerNorm + out‑gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 9️⃣ cast back to float32 for the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
