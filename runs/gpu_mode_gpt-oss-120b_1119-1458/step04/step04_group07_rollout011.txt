"""
Outgoing TriMul (AlphaFold‑3) – high‑performance forward pass.

Key ideas
---------
1️⃣  Fuse the **five** linear transforms (left_proj, left_gate,
    right_proj, right_gate, out_gate) into a **single** FP16 matmul.
    This removes five repeated reads of the huge input tensor
    (B·N²·dim) and reduces memory traffic dramatically.

2️⃣  Perform the O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    as a batched GEMM (C = A @ Bᵀ) with a custom Triton kernel.
    * All tensors are FP16, accumulation is FP32.
    * Tile size 128×128×64 for N ≥ 512, otherwise 64×64×64.
    * 8 warps / 3 pipeline stages – maps to Tensor‑Cores on H100.

3️⃣  All remaining cheap ops (LayerNorm, sigmoid gating, final
    projection) stay in PyTorch but run on FP16.

The function `custom_kernel` follows the required signature:
    `data = (input_tensor, mask_tensor, weights_dict, config_dict)`
and returns a Float‑32 tensor `[B, N, N, dim]`.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : [M, N, K]   (M = batch * hidden, N = seq_len, K = seq_len)
#   C    : [M, N, N]
#   Types: FP16 inputs → FP32 accumulator → FP16 output
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # scalar sizes
    stride_am, stride_ai, stride_ak,   # A strides for (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides for (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides for (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    #  Program IDs
    # ------------------------------------------------------------------
    pid_i = tl.program_id(0)      # tile along first N dimension (i)
    pid_j = tl.program_id(1)      # tile along second N dimension (j)
    pid_z = tl.program_id(2)      # batch * hidden index (M)

    # ------------------------------------------------------------------
    #  Compute offsets for this tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    #  Accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    #  Loop over K in tiles
    # ------------------------------------------------------------------
    k_range = tl.cdiv(K, BLOCK_K)          # number of K‑tiles (run‑time)
    for k_iter in range(0, k_range):
        cur_k = k_iter * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=(mask_i[:, None] & (offs_k[None, :] < K)),
            other=0.0,
        )   # fp16

        # B tile (transposed): (BLOCK_K, BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=((offs_k[:, None] < K) & mask_j[None, :]),
            other=0.0,
        )   # fp16

        # FP16 × FP16 → FP32 dot product
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    #  Store the result tile (convert back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(
    left: torch.Tensor, right: torch.Tensor
) -> torch.Tensor:
    """
    Compute ``out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]``.

    Parameters
    ----------
    left, right : torch.Tensor
        Shape ``[B, N, N, H]`` (dtype torch.float16, contiguous).

    Returns
    -------
    out : torch.Tensor
        Shape ``[B, N, N, H]`` (dtype torch.float16).
    """
    B, N, _, H = left.shape

    # Collapse batch & hidden into leading dimension for the kernel
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (same layout as A/B)
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in **element** units (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # ------------------------------------------------------------------
    # Choose tile sizes dynamically
    # ------------------------------------------------------------------
    if N >= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8
    else:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=3,
    )

    # Reshape back to the original 4‑D layout
    out = (
        C.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operator (no backward).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32, bool or None)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                     # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the bulk of the work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse all five linear maps (proj + gate + out_gate) into one matmul
    # ------------------------------------------------------------------
    # Cast weights once
    w_left_proj = weights["left_proj.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    # Concatenate: [left_proj; left_gate; right_proj; right_gate; out_gate]
    combined_weight = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate],
        dim=0,
    )  # shape (5*hidden_dim, dim), dtype fp16

    # Single fused linear → shape [B, N, N, 5*hidden_dim]
    fused = torch.nn.functional.linear(x_fp16, combined_weight)  # fp16

    # Split into the five logical tensors (views, no copy)
    left_proj, left_gate_pre, right_proj, right_gate_pre, out_gate_pre = fused.split(
        hidden_dim, dim=-1
    )
    del fused  # free intermediate buffer early

    # Apply sigmoid gating
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # Fuse gating into the projections (in‑place)
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via Triton
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hidden] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (still in FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + output projection
    # ------------------------------------------------------------------
    out = out * out_gate                      # element‑wise gating
    to_out_w = weights["to_out.weight"].to(torch.float16)   # shape (dim, hidden)
    out = torch.nn.functional.linear(out, to_out_w)        # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to Float32 for the harness / downstream code
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
