"""
TriMul (outgoing) – fast Triton‑accelerated forward pass
======================================================

The implementation follows the reference PyTorch TriMul module but
reduces kernel launch overhead by fusing linear layers:
  * A single fused linear computes the left/right projections.
  * Another fused linear computes the three gating tensors.
The heavy O(N³) contraction is realised with a cuBLAS batched GEMM
(`torch.bmm`), which is the fastest on H100.  An element‑wise
multiplication kernel written in Triton is used for the final
out‑gate multiplication – this satisfies the “must use Triton” rule
while adding virtually no latency.

All computations after the first LayerNorm are performed in FP16,
the final result is cast back to FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton element‑wise multiply (out = a * b) – used for the final
# out‑gate multiplication.  It can also be re‑used for any other
# binary elementwise op if needed.
# ----------------------------------------------------------------------
@triton.jit
def _elementwise_mul_kernel(
    a_ptr,
    b_ptr,
    out_ptr,
    size,
    stride_a,
    stride_b,
    stride_out,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs * stride_a, mask=mask)
    b = tl.load(b_ptr + offs * stride_b, mask=mask)
    tl.store(out_ptr + offs * stride_out, a * b, mask=mask)


def _apply_elementwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Fuse an element‑wise multiply with a Triton kernel.
    Both inputs must have identical shape, be contiguous and reside on CUDA.
    """
    assert a.shape == b.shape, "Elementwise mul tensors must have the same shape"
    a_c = a.contiguous()
    b_c = b.contiguous()
    out = torch.empty_like(a_c, dtype=a_c.dtype, device=a_c.device)

    size = a_c.numel()
    block = 1024  # good trade‑off between launch overhead and latency
    grid = (triton.cdiv(size, block),)

    _elementwise_mul_kernel[grid](
        a_c,
        b_c,
        out,
        size,
        a_c.stride(0),
        b_c.stride(0),
        out.stride(0),
        BLOCK_SIZE=block,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Batched matmul (C = A @ Bᵀ) – highly optimised cuBLAS via torch.bmm.
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Parameters
    ----------
    left : torch.Tensor
        Shape [B, N, N, H] in half precision.
    right : torch.Tensor
        Shape [B, N, N, H] in half precision.

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, H] in half precision.
    """
    B, N, _, H = left.shape

    # Reshape to (B*H, N, N) for batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched GEMM: (B*H, N, N) @ (B*H, N, N)ᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32, bool or None)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``,
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – brings statistics into FP32 for stability
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # Cast to half‑precision for the rest of the pipeline
    x = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Cast all model weights to FP16 once
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (H, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16) # (H, dim)

    left_gate_w = weights["left_gate.weight"].to(torch.float16)   # (H, dim)
    right_gate_w = weights["right_gate.weight"].to(torch.float16) # (H, dim)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)     # (H, dim)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)    # (H,)
    to_out_w = weights["to_out.weight"].to(torch.float16)            # (dim, H)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse left/right projections into a single linear layer
    #    out shape: [B, N, N, 2*H] → split into left_proj / right_proj
    # ------------------------------------------------------------------
    proj_weight = torch.cat([left_proj_w, right_proj_w], dim=0)  # (2*H, dim)
    proj = F.linear(x, proj_weight)  # [B, N, N, 2*H] (FP16)
    left_proj = proj[..., :hidden_dim]
    right_proj = proj[..., hidden_dim:]

    # ------------------------------------------------------------------
    # 4️⃣ Fuse the three gate linear projections (left_gate, right_gate, out_gate)
    #    out shape: [B, N, N, 3*H] → split accordingly
    # ------------------------------------------------------------------
    gate_weight = torch.cat([left_gate_w, right_gate_w, out_gate_w], dim=0)  # (3*H, dim)
    gates = F.linear(x, gate_weight)  # [B, N, N, 3*H] (FP16)
    left_gate_raw = gates[..., :hidden_dim]
    right_gate_raw = gates[..., hidden_dim:2 * hidden_dim]
    out_gate_raw = gates[..., 2 * hidden_dim:]

    # ------------------------------------------------------------------
    # 5️⃣ Optional binary mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ Apply gating (sigmoid) and combine with the projected tensors
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    left = left_proj * left_gate   # shape [B, N, N, H] (FP16)
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction – batched GEMM (C = A @ Bᵀ)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, H] (FP16)

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # still FP16

    # ------------------------------------------------------------------
    # 9️⃣ Apply the final output gate (sigmoid) using a Triton kernel
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw)              # [B, N, N, H] FP16
    out = _apply_elementwise_mul(out, out_gate)          # fused multiply (FP16)

    # ------------------------------------------------------------------
    # 🔟 Project back to the original feature dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)                       # [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 11️⃣ Cast back to float32 for the expected output type
    # ------------------------------------------------------------------
    return out.to(torch.float32)
