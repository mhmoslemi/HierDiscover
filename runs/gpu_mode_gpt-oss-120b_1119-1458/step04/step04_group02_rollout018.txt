"""
custom_kernel – Outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Layer‑norm the input (float32 for numerical stability).
2️⃣ Cast to fp16 for the heavy part of the forward pass.
3️⃣ Fuse the linear projection and gating weights (proj + gate) for the
   left and right branches, performing a single matmul per side and then
   applying a sigmoid to the gate half.
4️⃣ (Optional) Apply the mask to the projected tensors.
5️⃣ Compute the O(N³) contraction out[b,i,j,h] = Σₖ left[b,i,k,h] *
   right[b,j,k,h] with a custom Triton kernel that treats the hidden
   dimension as an extra batch dimension (total batches = B·H).  The
   kernel is a fused, batched GEMM that loads A = left and B = rightᵀ
   in tiles and uses `tl.dot` (Tensor‑Core friendly fp16→fp32
   accumulation).
6️⃣ Cast the result back to float32, apply a second Layer‑norm over the
   hidden dimension, multiply by the output‑gate and finally project back
   to the original feature dimension.
7️⃣ Return the float32 output.

All heavy tensor arithmetic runs in fp16, with fp32 accumulation for
numeric stability.  A single Triton kernel replaces the torch.bmm
batched matmul, and the left/right projection+gate are fused to halve
the number of linear passes.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Batched GEMM kernel: computes C = A @ Bᵀ for many independent batches.
# Shapes:
#   A : [B, N, N, H]  (fp16) – accessed as a 3‑D tensor (batch, i, k)
#   B : [B, N, N, H]  (fp16) – accessed as (batch, j, k) then transposed
#   C : [B, N, N, H]  (fp16) – result (batch, i, j)
# The hidden dimension H is flattened into the batch dimension:
#   total_batch = B * H.
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    BATCH, N, H,                       # B, sequence length, hidden dim
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_k, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Tiled batched GEMM (A @ Bᵀ) with BATCH = B*H.

    - a_ptr/b_ptr point to tensors of shape [B,N,N,H] (contiguous).
    - c_ptr points to the output of the same shape.
    """
    # ------------------------------------------------------------------
    # Program IDs: batch (combined B×H), row tile, col tile
    # ------------------------------------------------------------------
    pid_batch = tl.program_id(0)      # 0 … B*H-1
    pid_m = tl.program_id(1)          # tile over i dimension
    pid_n = tl.program_id(2)          # tile over j dimension

    # ------------------------------------------------------------------
    # Decompose pid_batch → (batch_index, hidden_index)
    # ------------------------------------------------------------------
    hidden = H
    b_idx = pid_batch // hidden       # 0 … B‑1
    h_idx = pid_batch % hidden        # 0 … H‑1

    # Base pointers for the current (b, h) slice
    a_batch = a_ptr + b_idx * stride_a_batch + h_idx * stride_a_h
    b_batch = b_ptr + b_idx * stride_b_batch + h_idx * stride_b_h
    c_batch = c_ptr + b_idx * stride_c_batch + h_idx * stride_c_h

    # ------------------------------------------------------------------
    # Offsets within the N×N matrix
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # rows (i)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # cols (j)

    # Mask for out‑of‑bounds rows / cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k in blocks
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------- Load A block (i,k) ----------------------------------
        a_ptrs = a_batch + (offs_i[:, None] * stride_a_i) + (offs_k[None, :] * stride_a_k)
        a_mask = mask_i[:, None] & mask_k[None, :]
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # ------- Load B block (j,k) ----------------------------------
        # B is used transposed, we therefore load a (j,k) tile and transpose it
        b_ptrs = b_batch + (offs_j[:, None] * stride_b_i) + (offs_k[None, :] * stride_b_k)
        b_mask = mask_j[:, None] & mask_k[None, :]
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)
        b = tl.trans(b)                     # now shape (BLOCK_K, BLOCK_N)

        # ------- Dot product -----------------------------------------
        acc += tl.dot(a, b)                 # fp16→fp32 accumulation

    # ------------------------------------------------------------------
    # Store the result back to C (cast to fp16)
    # ------------------------------------------------------------------
    c_ptrs = c_batch + (offs_i[:, None] * stride_c_i) + (offs_j[None, :] * stride_c_k)
    c_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the outgoing TriMul contraction with the Triton kernel.
    Input/Output tensors are fp16 with shape [B, N, N, H].
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    total_batches = B * H

    # Pointers (in element units, not bytes)
    a_ptr = left.data_ptr()
    b_ptr = right.data_ptr()
    out = torch.empty_like(left)               # fp16 output
    c_ptr = out.data_ptr()

    # Strides in element units
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h = right.stride()
    stride_c_batch, stride_c_i, stride_c_k, stride_c_h = out.stride()

    # Tile sizes – tuned for fp16 Tensor‑cores on H100
    BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32

    grid = (
        total_batches,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_matmul_outgoing_kernel[grid](
        a_ptr,
        b_ptr,
        c_ptr,
        B,
        N,
        H,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_a_h,
        stride_b_batch,
        stride_b_i,
        stride_b_k,
        stride_b_h,
        stride_c_batch,
        stride_c_i,
        stride_c_k,
        stride_c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=5,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operator – no gradients.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor[B, N, N]      (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Compute the output‑gate (float32) – needed after the contraction
    # ------------------------------------------------------------------
    out_gate_weight = weights["out_gate.weight"]
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3️⃣ Cast to fp16 for the heavy part
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse projection + gating weight matrices (fp16)
    # ------------------------------------------------------------------
    # Left side
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)          # [2*H, dim]
    left_comb = torch.nn.functional.linear(x_fp16, left_comb_w)        # [B,N,N,2*H]
    left_proj, left_gate_pre = left_comb.split([hidden_dim, hidden_dim], dim=-1)
    # Right side
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)       # [2*H, dim]
    right_comb = torch.nn.functional.linear(x_fp16, right_comb_w)      # [B,N,N,2*H]
    right_proj, right_gate_pre = right_comb.split([hidden_dim, hidden_dim], dim=-1)

    # ------------------------------------------------------------------
    # 5️⃣ Apply mask (if present) before gating, as in the reference code
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B,N,N]; broadcast over hidden dim
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ Activate gates and fuse them with the projections
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)               # [B,N,N,H]
    right_gate = torch.sigmoid(right_gate_pre)             # [B,N,N,H]

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ O(N³) contraction via the custom Triton kernel
    # ------------------------------------------------------------------
    out_fp16 = _batched_matmul_triton(left, right)        # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to float32 for the final LayerNorm & projection
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm over the hidden dimension (float32)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # [B,N,N,H] (float32)

    # ------------------------------------------------------------------
    # 10️⃣ Apply the output‑gate and final projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_w)        # [B,N,N,dim] (float32)

    return out
