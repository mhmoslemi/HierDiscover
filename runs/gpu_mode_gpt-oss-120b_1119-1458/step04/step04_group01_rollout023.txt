        b_mask = (offs_j < N)[None, :] & (offs_k < K)[:, None]
        b_ptrs = b_base + offs_k[:, None] * col_stride + offs_j[None, :] * row_stride
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Store result
    c_mask = (offs_i < M)[:, None] & (offs_j < N)[None, :]
    c_ptrs = c_base + offs_i[:, None] * out_row_stride + offs_j[None, :] * out_col_stride
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def _batched_contraction_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using a custom Triton kernel.
    left/right must be FP16 and have shape [B, N, N, H].
    Returns tensor of same shape and dtype.
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Strides for left/right (same)
    batch_stride = left.stride(0)    # B dimension stride
    row_stride = left.stride(1)      # i dimension stride (N*H)
    col_stride = left.stride(2)      # k dimension stride (H)
    hidden_stride = left.stride(3)   # h dimension stride (1)

    # Strides for output (same layout)
    out_batch_stride = out.stride(0)
    out_row_stride = out.stride(1)
    out_col_stride = out.stride(2)
    out_hidden_stride = out.stride(3)

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)

    _trimul_contract_kernel[grid](
        left, right, out,
        N, N, N, H,
        batch_stride, row_stride, col_stride, hidden_stride,
        out_batch_stride, out_row_stride, out_col_stride, out_hidden_stride,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Fast forward‑only implementation of the outgoing TriMul operation.
    The heavy O(N³) contraction is performed by a custom Triton kernel.
    All linear projections are fused into a single matrix multiply to cut
    memory traffic, and computations are carried out in FP16 wherever possible.
    Returns a float32 tensor of shape [B, N, N, dim].
    """
    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # ------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32
    x = x.to(torch.float16)  # subsequent ops in fp16

    # ------------------------------
    # 2️⃣ Fuse all linear projections and gates into a single matmul
    #    (weights are concatenated along the output dimension)
    # ------------------------------
    # Cast and concatenate all required weight matrices once
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    # all have shape (hidden_dim, dim)
    cat_weight = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(torch.float16)  # shape (5*hidden_dim, dim)

    # Single linear projection
    proj = F.linear(x, cat_weight)  # [B, N, N, 5*hidden_dim] in fp16

    # Slice the concatenated results
    h = hidden_dim
    left_proj = proj[..., 0 : h]
    left_gate_pre = proj[..., h : 2 * h]
    right_proj = proj[..., 2 * h : 3 * h]
    right_gate_pre = proj[..., 3 * h : 4 * h]
    out_gate_pre = proj[..., 4 * h : 5 * h]

    # Apply sigmoid to gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # ------------------------------
    # 3️⃣ Optional binary mask (broadcast over hidden dimension)
    # ------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------
    # 4️⃣ Apply gated multiplicative update (the “TriMul” core)
    # ------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Heavy O(N³) contraction via Triton
    out = _batched_contraction_triton(left, right)  # [B,N,N,hidden_dim] fp16

    # ------------------------------
    # 5️⃣ Second LayerNorm, output gate and final projection
    # ------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    out = out * out_gate
    out = F.linear(out, to_out_w)  # back to dim, still fp16

    # Cast back to float32 for the final output
    return out.to(torch.float32)

"""
Fast outgoing TriMul (αFold3) forward pass.

Algorithm
---------
1️⃣ Layer‑norm the input pair tensor (float32 → float16).
2️⃣ Fuse **all** linear projections and gating logits into *one* matmul
   by concatenating the five weight matrices (left_proj, left_gate,
   right_proj, right_gate, out_gate).  This reduces the massive
   input‑tensor read from O(5) to O(1) and eliminates several temporary
   tensors.
3️⃣ Apply the binary mask (if present) and sigmoid to the gate logits.
4️⃣ Compute the O(N³) multiplicative update
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
   with a custom Triton kernel that works on the original
   `[B, N, N, H]` layout (no costly `permute+contiguous` copy).
   The kernel treats each hidden slice as an independent batch
   (total batches = B·H) and performs a tiled FP16 GEMM with
   FP32 accumulation.
5️⃣ Second LayerNorm, output‑gate multiplication and final linear
   projection back to the original `dim`.  All remaining ops stay in
   FP16; the final result is cast to float32.

The heavy contraction is the only place where Triton is used – the
kernel achieves high arithmetic intensity on H100 tensor‑cores and
avoids the extra memory traffic of the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched contraction C[b,i,j,h] = Σ_k A[b,i,k,h] * B[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_contract_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K, H,                     # matrix sizes (M=N=seq_len), hidden dim
    batch_stride, row_stride, col_stride, hidden_stride,
    out_batch_stride, out_row_stride, out_col_stride, out_hidden_stride,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program identifiers
    pid_m = tl.program_id(0)          # block row index   (i)
    pid_n = tl.program_id(1)          # block col index   (j)
    pid_batch = tl.program_id(2)      # combined batch   (b·H)

    # decode batch‑and‑hidden indices
    b_idx = pid_batch // H            # batch dimension   (0 .. B‑1)
    h_idx = pid_batch % H             # hidden dimension  (0 .. H‑1)

    # base pointers for this (b,h) slice
    a_base = a_ptr + b_idx * batch_stride + h_idx * hidden_stride
    b_base = b_ptr + b_idx * batch_stride + h_idx * hidden_stride
    c_base = c_ptr + b_idx * out_batch_stride + h_idx * out_hidden_stride

    # offsets of the current output tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # tiled reduction over the contraction axis k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # ---- read tile from A (shape BLOCK_M × BLOCK_K) ----
        a_mask = (offs_i < M)[:, None] & (offs_k < K)[None, :]
        a_ptrs = a_base + offs_i[:, None] * row_stride + offs_k[None, :] * col_stride
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # ---- read transposed tile from B (shape BLOCK_K × BLOCK_N) ----
        b_mask = (offs_k < K)[:, None] & (offs_j < N)[None, :]
        # B_T rows = k, cols = j  →  pointer = (k * col_stride) + (j * row_stride)
        b_ptrs = b_base + offs_k[:, None] * col_stride + offs_j[None, :] * row_stride
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # ---- accumulate ----
        acc += tl.dot(a, b)   # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N)

    # ---- write‑back ----
    c_mask = (offs_i < M)[:, None] & (offs_j < N)[None, :]
    c_ptrs = c_base + offs_i[:, None] * out_row_stride + offs_j[None, :] * out_col_stride
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def _batched_contraction_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the O(N³) update using the Triton kernel.
    Arguments
    ----------
    left, right : torch.Tensor
        Float‑16 tensors of shape [B, N, N, H].
    Returns
    -------
    torch.Tensor
        Float‑16 tensor of the same shape containing the contracted result.
    """

    out = torch.empty_like(left)

    # layout strides (the same for left/right/out)
    batch_stride = left.stride(0)      # B‑dimension stride = N*N*H
    row_stride   = left.stride(1)      # i‑dimension stride = N*H
    col_stride   = left.stride(2)      # k‑dimension stride = H
    hidden_stride = left.stride(3)     # h‑dimension stride = 1

    out_col_stride   = out.stride(2)
    out_hidden_stride = out.stride(3)

    BLOCK_N = 128
    BLOCK_K = 32

    # launch grid: (tiles over rows, tiles over cols, total (B·H) slices)
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * H)

        left,
        right,
        out,
        N, N, N, H,
        batch_stride,
        row_stride,
        col_stride,
        hidden_stride,

        out_col_stride,
        out_hidden_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out

    # ------------------------------------------------------------------
