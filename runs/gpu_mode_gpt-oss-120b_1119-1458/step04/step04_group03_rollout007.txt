def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    nomask = config.get('nomask', True)

    # compute layernorm (float32)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor,
        (dim,), weight=norm_weight, bias=norm_bias)
    # cast to half
    x = x.to(torch.float16)

    # prepare fused weight
    fused_weight = torch.cat([
        weights["left_proj.weight"],  # shape hidden, dim
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(torch.float16) # shape (5*hidden, dim)

    # linear

    fused_out = torch.nn.functional.linear(x, fused_weight) # shape [B,N,N,5*hidden]
    # split
    left = fused_out[..., 0:hidden_dim]
    right = fused_out[..., hidden_dim:2*hidden_dim]
    left_gate_logits = fused_out[..., 2*hidden_dim:3*hidden_dim]
    right_gate_logits = fused_out[..., 3*hidden_dim:4*hidden_dim]
    out_gate_logits = fused_out[..., 4*hidden_dim:5*hidden_dim]

    #mask: apply mask if not nomask
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    #gates
    left_gate = torch.sigmoid(left_gate_logits)
    right_gate = torch.sigmoid(right_gate_logits)
    out_gate = torch.sigmoid(out_gate_logits)

    left = left * left_gate
    right = right * right_gate

    # contraction (batched GEMM)
    N = input_tensor.shape[1]
    # choose algorithm
    if N <= 256:
        out = _batched_matmul_triton(left, right)
    else:
        out = _batched_matmul_torch(left, right)

    # second layernorm (half)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate

    # linear back to dim
    out = torch.nn.functional.linear(out,
        weights["to_out.weight"].to(torch.float16))

    # cast to float32 and return
    return out.to(torch.float32)

    # Equivalent to (left_perm @ right_perm^T) across K dimension.

---------
1️⃣ Layer‑norm the 4‑D input tensor (float32) → cast to fp16.
2️⃣ Fuse the four linear projections (left‑proj, right‑proj, left‑gate,
   right‑gate) and the out‑gate into a *single* half‑precision GEMM
   (weight matrix of shape (5·hidden, dim)).  The result is sliced into
   `left`, `right`, `left_gate`, `right_gate`, `out_gate`.
3️⃣ Optional mask is broadcast and fused with `left`/`right`.
4️⃣ Apply sigmoid to the three gate tensors and fuse them with the
   projected tensors.
5️⃣ Heavy contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is implemented as a batched GEMM `C = A @ Bᵀ` where
   A = left.permute(0,3,1,2).view(M,N,N) and B = right.permute(...).
   * For sequence lengths ≤ 256 a handcrafted Triton kernel (64×64×64
     tiles, 8 warps, 2 stages) is used.
   * For larger N the highly‑optimized cuBLAS implementation via
     `torch.bmm` is used.
6️⃣ Second LayerNorm (fp16) → element‑wise out‑gate → final linear
   projection back to `dim`.
7️⃣ Cast the result to float32 (required by the harness).

The kernel runs on H100 (Tensor‑Core friendly), accumulates in fp32,
stores in fp16 and is fully vectorised over the hidden‑dimension batch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (fp16 → fp16, accumulator fp32)
#   A, B : [M, N, K]   (M = batch * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_matmul_kernel(
    A_ptr,                # *fp16  [M, N, K]
    B_ptr,                # *fp16  [M, N, K]
    C_ptr,                # *fp16  [M, N, N]
    M, N, K,              # i32 sizes
    stride_am, stride_ai, stride_ak,   # strides for A (M, i, k)
    stride_bm, stride_bj, stride_bk,   # strides for B (M, j, k)
    stride_cm, stride_ci, stride_cj,   # strides for C (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel for the TriMul contraction."""
    pid_i = tl.program_id(0)   # tile along output‑row dimension (i)
    pid_j = tl.program_id(1)   # tile along output‑col dimension (j)
    pid_z = tl.program_id(2)   # batch*hidden index (M)

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Accumulator (fp32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K‑tiles
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)

        # A tile (i, k)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (cur_k[None, :] < K),
            other=0.0,
        )  # fp16

        # B tile (j, k) – we need the transposed view (k, j)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + cur_k[:, None] * stride_bk          # k dimension first
            + offs_j[None, :] * stride_bj          # j dimension second
        )
        b = tl.load(
            b_ptrs,
            mask=(cur_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # Mixed‑precision dot product
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write back
    # --------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract ``left`` and ``right`` (both [B, N, N, H]) using the custom
    Triton GEMM kernel.  Returns a tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    # Rearrange to [M, N, N] where M = B * H
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    A = left_perm.view(B * H, N, N)
    B_ = right_perm.view(B * H, N, N)

    C = torch.empty_like(A)  # fp16 output

    # Strides in element counts (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tile size – 64×64×64 works well on H100 for fp16 Tensor‑Core MMA
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden
    )

    _tri_mul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Reshape back to the original layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract ``left`` and ``right`` using torch.bmm (cuBLAS).  Input
    tensors are float16; the accumulation is performed in float32
    automatically.
    """
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (gradient‑free).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 / bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 1️⃣ Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 2️⃣ First LayerNorm (float32) – stability
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float32)
    norm_bias = weights["norm.bias"].to(torch.float32)

    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 3️⃣ Cast to fp16 for the rest of the pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse the five linear projections into a single GEMM
    # ------------------------------------------------------------------
    #   left_proj, right_proj, left_gate, right_gate, out_gate
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape (5*hidden_dim, dim)

    # Linear: [B, N, N, dim] @ (5*hidden, dim).T  →  [B, N, N, 5*hidden]
    fused_out = F.linear(x, fused_weight)  # fp16

    # Slice the fused output
    left = fused_out[..., 0:hidden_dim]                                 # proj L
    right = fused_out[..., hidden_dim:2 * hidden_dim]                    # proj R
    left_gate_logit = fused_out[..., 2 * hidden_dim:3 * hidden_dim]     # gate L
    right_gate_logit = fused_out[..., 3 * hidden_dim:4 * hidden_dim]    # gate R
    out_gate_logit = fused_out[..., 4 * hidden_dim:5 * hidden_dim]      # gate out

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast on hidden dim) and gating
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    left_gate = torch.sigmoid(left_gate_logit)
    right_gate = torch.sigmoid(right_gate_logit)
    out_gate = torch.sigmoid(out_gate_logit)

    # Fuse gates with projected tensors
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction: batched mat‑mul left ⋅ rightᵀ
    # ------------------------------------------------------------------
    seq_len = input_tensor.shape[1]
    if seq_len <= 256:
        out = _batched_matmul_triton(left, right)      # fp16
    else:
        out = _batched_matmul_torch(left, right)       # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm → out‑gate → final linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))  # fp16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to float32 (the harness expects float32)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
