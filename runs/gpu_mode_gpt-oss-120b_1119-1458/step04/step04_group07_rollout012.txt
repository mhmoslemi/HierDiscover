def trimul_outgoing_kernel(

    B, N, H,

):

"""
Outgoing TriMul (AlphaFold‑3) – highly optimized Triton implementation.

The core O(N³) contraction
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
is expressed as a batched GEMM (A @ Bᵀ) for each hidden head `h`.
The implementation avoids any expensive permutation or copy:
* left/right are kept in their native layout [B, N, N, H] (FP16).
* A custom Triton kernel directly indexes batch, hidden, i, j, k dimensions
  using the original strides, accumulating in FP32 for numerical stability.
* Tiling with 128×128×64 (or 64×64×64) tiles and tensor‑core dot products
  yields very high throughput on H100.

All surrounding operations (LayerNorms, linear projections, gating)
are performed with the usual PyTorch primitives in FP16, and the final
output is returned as FP32. The kernel accounts for optional masking.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : [B, N, N, H] (FP16)
#   C    : [B, N, N, H] (FP16, accumulator FP32)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    A_ptr,               # *fp16  [B, N, N, H]
    B_ptr,               # *fp16  [B, N, N, H]
    C_ptr,               # *fp16  [B, N, N, H]
    batch_size, N, hidden_dim,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,   # A strides
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,   # B strides
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)   # tile along first N dimension (i)
    pid_j = tl.program_id(1)   # tile along second N dimension (j)
    pid_z = tl.program_id(2)   # combined batch*hidden index

    # ------------------------------------------------------------------
    # Decode combined index into batch and hidden channel
    # ------------------------------------------------------------------
    hid_idx = pid_z % hidden_dim
    batch_idx = pid_z // hidden_dim

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Base pointers for the selected batch and hidden channel
    # ------------------------------------------------------------------
    a_base = A_ptr + batch_idx * stride_a_batch + hid_idx * stride_a_h
    b_base = B_ptr + batch_idx * stride_b_batch + hid_idx * stride_b_h
    c_base = C_ptr + batch_idx * stride_c_batch + hid_idx * stride_c_h

    # ------------------------------------------------------------------
    # Accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K (which equals N)
    # ------------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile : shape (BLOCK_M, BLOCK_K)  (i, k)
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # Load B tile (transposed) : shape (BLOCK_K, BLOCK_N)  (k, j)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_j
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )  # fp16

        # Mixed‑precision dot (fp16 × fp16 → fp32)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store output tile (cast back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_tri_mul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the heavy contraction for outgoing TriMul:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Parameters
    ----------
    left, right : torch.Tensor
        Shape [B, N, N, H] (fp16, contiguous).

    Returns
    -------
    out : torch.Tensor
        Shape [B, N, N, H] (fp16, same device as inputs).
    """
    B, N, _, H = left.shape

    # Output buffer
    out = torch.empty_like(left, dtype=torch.float16)

    # Strides in element counts
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h = right.stride()
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h = out.stride()

    # Tile sizes – large tiles for big seq lengths
    if N >= 512:
        BLOCK_M = BLOCK_N = 128
        num_warps = 8
    else:
        BLOCK_M = BLOCK_N = 64
        num_warps = 4
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden
    )

    _trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_a_h,
        stride_b_batch,
        stride_b_j,
        stride_b_k,
        stride_b_h,
        stride_c_batch,
        stride_c_i,
        stride_c_j,
        stride_c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=3,
    )

    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operator (no backwards).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------
    # 2️⃣ Cast to fp16 for the rest of the pipeline
    # -------------------------------------------------
    x = x.to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Cast all linear weights once to fp16
    # -------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # -------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B, N, N, hidden_dim]

    # -------------------------------------------------
    # 5️⃣ Optional mask (broadcast on hidden dim)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 6️⃣ Gating (sigmoid) – fp16
    # -------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply gates in‑place to reduce memory traffic
    torch.mul(left, left_gate, out=left)
    torch.mul(right, right_gate, out=right)

    # ------------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via Triton (no layout permutation!)
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = _batched_tri_mul_fp16(left, right)   # [B, N, N, hidden_dim] fp16

    # -------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 9️⃣ Final gating + projection (fp16)
    # -------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] fp16

    # -------------------------------------------------
    # 🔟 Cast back to float32 for the harness
    # -------------------------------------------------
    return out.to(torch.float32)
```
