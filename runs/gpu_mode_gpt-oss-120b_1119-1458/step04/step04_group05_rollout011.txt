def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)
    device = input_tensor.device

    # 1. First layer_norm
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias).to(torch.float16)

    # 2. Fuse projection and gating linear transforms
    # Stack weight tensors; all bias=False
    fused_weight = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16)  # shape (5*hidden_dim, dim)

    # flatten x
    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim) # (B*N*N, dim)

    # compute fused linear (1 matmul)
    fused_out = torch.nn.functional.linear(x_flat, fused_weight) # (B*N*N, 5*H)
    # Now reshape
    fused_out = fused_out.view(B, N, N, hidden_dim*5)

    # Section slices:
    left_proj_raw = fused_out[..., :hidden_dim]      # (B,N,N,H)
    left_gate_raw = fused_out[..., hidden_dim:2*hidden_dim]
    right_proj_raw = fused_out[..., 2*hidden_dim:3*hidden_dim]
    right_gate_raw = fused_out[..., 3*hidden_dim:4*hidden_dim]
    out_gate_raw = fused_out[..., 4*hidden_dim:5*hidden_dim]

    # apply mask if needed (mask is (B,N,N) bool/float)
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1) # (B,N,N,1)
        left_proj_raw = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    # gating
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # apply gates
    left = left_proj_raw * left_gate
    right = right_proj_raw * right_gate

    # free fused_out maybe; but keep.

    # 3. Batch matmul
    # left shape [B,N,N,H], right shape [B,N,N,H]
    # For N<=512 use Triton batched, else torch bmm.

    if N <= 512:
        # use _batched_matmul_triton
        out = _batched_matmul_triton(left, right, tile=128, k_tile=64)
    else:
        # rearrange to (B*H,N,N)
        left_mat = left.permute(0,3,1,2).contiguous().view(B*hidden_dim, N, N)
        right_mat = right.permute(0,3,2,1).contiguous().view(B*hidden_dim, N, N)
        out_mat = torch.bmm(left_mat, right_mat)  # (B*H,N,N)
        out = out_mat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()

    # 4. Second LayerNorm, gating, final linear
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate

    to_out_weight = weights["to_out.weight"].to(torch.float16) # shape (dim, hidden_dim)
    out_flat = out.reshape(-1, hidden_dim)  # (B*N*N, hidden_dim)
    out_proj = torch.nn.functional.linear(out_flat, to_out_weight)  # (B*N*N, dim)
    out_proj = out_proj.view(B, N, N, dim)

    # cast to float32
    return out_proj.to(torch.float32)

"""
TriMul (outgoing) – fast forward implementation.
================================================

The outgoing TriMul contracts the pairwise tensor `x` ∈ ℝ^{B×N×N×C}
into an output of the same shape using

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]          (1)

where `left` and `right` are linear projections of `x` (with gating
and optional masking). The dominant O(N³·H) operation (1) is performed
as a batched matrix‑multiply A @ Bᵀ with A = left and B = right.
For N ≤ 512 a custom Triton kernel is used (Tensor‑Core‐enabled FP16
dot‑product, accumulation in FP32). For larger N we fall back on
cuBLAS‑backed `torch.bmm`.

All linear projections are fused into a *single* FP16 GEMM:
the five weight matrices (`left_proj`, `left_gate`, `right_proj`,
`right_gate`, `out_gate`) are concatenated along the output dimension,
so the input tensor is read only once. This reduces memory traffic
by ~5× compared with the reference implementation.

The overall forward pass therefore consists of:
1️⃣ LayerNorm on the input (float32 → fp16)
2️⃣ One fused projection + gating + optional masking
3️⃣ Batched matmul (Triton or cuBLAS)
4️⃣ Second LayerNorm, output‑gate, final linear projection
5️⃣ Cast back to float32

The implementation is gradient‑free (forward only) and works for
both masked and un‑masked inputs.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ
#   A : [M, N, K]   (M = B*H, N = seq_len (padded), K = seq_len (padded))
#   B : [M, K, N]   (already transposed, i.e. Bᵀ)
#   C : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bk, stride_bj,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)            # batch*head dimension
    pid_m = tl.program_id(0)            # tile over i (first N dim)
    pid_n = tl.program_id(1)            # tile over j (second N dim)

    # Offsets for i and j within the tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (to keep precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ---- load A[i,k] ----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # ---- load Bᵀ[k,j] (i.e. B[k,j]) ----
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # dot‑product (fp16×fp16 → fp32) and accumulate
        acc += tl.dot(a, b)

    # write result (cast back to fp16)
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(
    left: torch.Tensor,
    right: torch.Tensor,
    tile: int = 128,
    k_tile: int = 64,
) -> torch.Tensor:
    """
    Batched matrix multiplication `out = left @ rightᵀ` using the Triton kernel
    above. `left` and `right` are FP16 tensors of shape [B, N, N, H].
    Returns an FP16 tensor of the same shape.
    """
    B, N, _, H = left.shape

    # Pad to a multiple of `tile` for a clean kernel launch
    pad_len = (tile - (N % tile)) % tile
    if pad_len != 0:
        N_pad = N + pad_len
        left_pad = left.new_zeros((B, N_pad, N_pad, H), dtype=torch.float16)
        right_pad = right.new_zeros((B, N_pad, N_pad, H), dtype=torch.float16)
        left_pad[:, :N, :N, :] = left
        right_pad[:, :N, :N, :] = right
    else:
        left_pad = left
        right_pad = right
        N_pad = N

    # Pack batch and head dimension, and arrange right as transposed
    left_perm = left_pad.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right_perm = right_pad.permute(0, 3, 2, 1).contiguous()       # (B, H, N, N) == Bᵀ

    A = left_perm.view(B * H, N_pad, N_pad)       # (M, N, K)
    B_ = right_perm.view(B * H, N_pad, N_pad)    # (M, K, N)

    C = torch.empty_like(A)  # fp16 output buffer

    # strides expressed in **elements**
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bk, stride_bj = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    grid = (
        (N_pad + tile - 1) // tile,   # pid_m dimension
        (N_pad + tile - 1) // tile,   # pid_n dimension
        B * H,                        # pid_z dimension
    )
    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,            # M (unused inside the kernel)
        N_pad,            # N
        N_pad,            # K
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bk,
        stride_bj,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=tile,
        BLOCK_N=tile,
        BLOCK_K=k_tile,
        num_warps=8,
        num_stages=3,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N_pad, N_pad).permute(0, 2, 3, 1).contiguous()
    if pad_len != 0:
        C = C[:, :N, :N, :]
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (gradient‑free).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32/bool) or None
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2️⃣ first LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    ).to(torch.float16)   # shape [B, N, N, dim] in FP16

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 3️⃣ fuse all five input projections into a single matmul
    # ------------------------------------------------------------------
    # All 5 linear layers have bias=False → just stack the weight matrices
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],   # (H, dim)
            weights["left_gate.weight"],   # (H, dim)
            weights["right_proj.weight"],  # (H, dim)
            weights["right_gate.weight"],  # (H, dim)
            weights["out_gate.weight"],    # (H, dim)
        ],
        dim=0,  # concatenate on the output‑feature axis
    ).to(torch.float16)   # shape (5*H, dim)

    # flatten x -> (B·N·N, dim)
    x_flat = x.reshape(-1, dim)

    # a single fused linear pass
    fused_out = torch.nn.functional.linear(x_flat, fused_weight)          # (B·N·N, 5*H)
    fused_out = fused_out.view(B, N, N, hidden_dim * 5)                  # (B, N, N, 5*H)

    # slice the projections & gates
    h0 = hidden_dim
    left_proj_raw   = fused_out[...,        :h0]          # (B, N, N, H)
    left_gate_raw   = fused_out[...,   h0:2*h0]          # (B, N, N, H)
    right_proj_raw  = fused_out[..., 2*h0:3*h0]          # (B, N, N, H)
    right_gate_raw  = fused_out[..., 3*h0:4*h0]          # (B, N, N, H)
    out_gate_raw    = fused_out[..., 4*h0:5*h0]          # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣ optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if (mask_tensor is not None) and (not nomask):
        # mask shape (B, N, N) → (B, N, N, 1) for broadcasting
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left_proj_raw = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ gating (sigmoid) and fuse into projected tensors
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    left  = left_proj_raw * left_gate   # shape (B, N, N, H)
    right = right_proj_raw * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ heavy O(N³) contraction
    # ------------------------------------------------------------------
    if N <= 512:
        # Triton‑based batched GEMM (left @ rightᵀ)
        out = _batched_matmul_triton(left, right, tile=128, k_tile=64)
    else:
        # Fall back to cuBLAS‑backed batched GEMM for very large N
        left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
        right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)
        out_mat = torch.bmm(left_mat, right_mat)                       # (B*H, N, N)
        out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ second LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # final linear: hidden_dim → dim
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out_flat = out.reshape(-1, hidden_dim)                 # (B·N·N, hidden_dim)
    out_proj = torch.nn.functional.linear(out_flat, to_out_w)   # (B·N·N, dim)

    out_proj = out_proj.view(B, N, N, dim).to(torch.float32)

    return out_proj
