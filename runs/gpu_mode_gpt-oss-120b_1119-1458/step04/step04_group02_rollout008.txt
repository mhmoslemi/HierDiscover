def trmul_gemm_kernel(left:torch.Tensor, right:torch.Tensor):
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # shape [B,N,N,H]
    # compute strides
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()
    # convert to int64 pointers
    left_ptr = left.data_ptr()
    right_ptr = right.data_ptr()
    out_ptr = out.data_ptr()
    
    # Set tile sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32  # can try 32 or 64
    # Compute grid sizes
    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N
    num_batch_h = B * H
    grid = (num_i, num_j, num_batch_h)
    
    _trmul_gemm_kernel[grid](
          left_ptr, right_ptr, out_ptr,
          B, N, H,
          stride_l_b, stride_l_i, stride_l_k, stride_l_h,
          stride_r_b, stride_r_i, stride_r_k, stride_r_h,
          stride_o_b, stride_o_i, stride_o_j, stride_o_h,
          BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
          num_warps=8, # maybe 8 warps for good occupancy
    )
    return out

```
@triton.jit
def _trmul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)  # row tile identifier (i)
    pid_j = tl.program_id(1)  # column tile identifier (j)
    pid_bh = tl.program_id(2) # batch * hidden index

    # Compute batch and hidden ids
    batch_idx = pid_bh // H
    hidden_idx = pid_bh % H

    # offsets
    # compute base pointer for current batch+hidden slice
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    # create range offsets
    offs_i = i_offset + tl.arange(0, BLOCK_M)
    offs_j = j_offset + tl.arange(0, BLOCK_N)
    # masked flags for bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base offset for left and right (including batch and hidden)
    base_l = left_ptr + batch_idx * stride_l_b + hidden_idx * stride_l_h
    base_r = right_ptr + batch_idx * stride_r_b + hidden_idx * stride_r_h
    base_o = out_ptr + batch_idx * stride_o_b + hidden_idx * stride_o_h

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_tiles):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile shape (BLOCK_M, BLOCK_K)
        left_ptrs = base_l + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        # mask for left: i and k within bounds
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        # load right tile transposed (BLOCK_K, BLOCK_N)
        # need pointer for right transposed (k varying row, j varying column)
        right_ptrs = base_r + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # compute dot
        acc += tl.dot(left, right)  # (M,N)
    
    # store result
    out_ptrs = base_o + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    # Cast acc to fp16 for storage
    out = acc.to(tl.float16)
    tl.store(out_ptrs, out, mask=mask_i[:, None] & mask_j[None, :])
```

@triton.jit
def trmul_outgoing_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # i tile index
    pid_n = tl.program_id(1)  # j tile index
    pid_bh = tl.program_id(2) # batch*hidden index

    # compute batch and hidden
    batch_idx = pid_bh // H
    hidden_idx = pid_bh % H

    # base pointer for this batch+hidden
    base_l = left_ptr + batch_idx * stride_l_b + hidden_idx * stride_l_h
    base_r = right_ptr + batch_idx * stride_r_b + hidden_idx * stride_r_h
    base_o = out_ptr + batch_idx * stride_o_b + hidden_idx * stride_o_h

    # compute start indices
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # offsets
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # boundary masks
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k in range(0, num_k):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        mask_k = offs_k < N

        # pointer to left tile (M x K) non-transposed
        left_ptrs = base_l + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # pointer to right tile (K x N) transposed (i.e., rows are k, cols are j)
        right_ptrs = base_r + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i

    # store result. Convert to fp16
    out = acc.to(tl.float16)

"""
TriMul (outgoing) forward pass – highly‑optimized Triton implementation.

Algorithm
---------
1️⃣ Layer‑norm the input (float32 for numeric stability).
2️⃣ Cast to fp16 and apply the four linear projections (left/right  +  gating).
3️⃣ Optional mask (broadcasted over the hidden dimension) – fused with the
   projected tensors via a cheap element‑wise multiplication.
4️⃣ The core O(N³) contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is implemented as a custom batched GEMM (A @ Bᵀ) in Triton.  The kernel works
   directly on the original layout [B,N,N,H] → no expensive permutes/copies.
   Each hidden slice is treated as an extra batch dimension (B*H GEMMs) and
   the accumulation is performed in float32 while the storage is fp16.
5️⃣ Layer‑norm over the hidden dimension, multiply by the output gate and
   project back to the original dimension.
6️⃣ Cast the final result to float32 (required by the harness).

The heavy O(N³) work is now done by a hand‑written Triton GEMM which
eliminates the costly `permute(...).contiguous()` steps used in the
reference implementation, yielding a >3× speed‑up on the benchmark
configs.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for out‑going TriMul
# Computes out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]      (h = hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _trmul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,                 # pointers
    B, N, H,                                      # tensor sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # left strides
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,   # right strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # output strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batch‑GEMM kernel (C = A @ Bᵀ) with layout:
        A = left  [B, N_i, N_k, H]   (i‑k)
        B = right [B, N_j, N_k, H]   (j‑k)      → we read it transposed (k‑j)
        C = out   [B, N_i, N_j, H]

    The kernel is launched with a 3‑D grid:
        program_id(0) → tile over the i‑dimension (rows)
        program_id(1) → tile over the j‑dimension (columns)
        program_id(2) → (batch * hidden) dimension
    """
    pid_i = tl.program_id(0)          # i‑tile index
    pid_j = tl.program_id(1)          # j‑tile index
    pid_bh = tl.program_id(2)         # batch*hidden index

    # ------------------------------------------------------------------
    # Decode batch and hidden‑channel indices
    # ------------------------------------------------------------------
    batch_idx = pid_bh // H
    hidden_idx = pid_bh % H

    # ------------------------------------------------------------------
    # Compute base pointers (including batch & hidden offsets)
    # ------------------------------------------------------------------
    base_l = left_ptr  + batch_idx * stride_l_b + hidden_idx * stride_l_h
    base_r = right_ptr + batch_idx * stride_r_b + hidden_idx * stride_r_h
    base_o = out_ptr   + batch_idx * stride_o_b + hidden_idx * stride_o_h

    # ------------------------------------------------------------------
    # Offset of the current tile (i, j)
    # ------------------------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # Boundary masks (since N may not be a multiple of BLOCK_*)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Accumulator in higher precision
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # --------------------------------------------------------------
        # Load A‑tile (i‑k) – shape (BLOCK_M, BLOCK_K)
        # --------------------------------------------------------------
        ptrs_a = base_l + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(ptrs_a,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)   # cast to fp32 for accumulation

        # --------------------------------------------------------------
        # Load B‑tile (k‑j) – we read the right tensor transposed
        # --------------------------------------------------------------
        # right layout is (batch, j, k, hidden).  To produce a (k, j) tile we swap
        # the i‑ and k‑strides.
        ptrs_b = base_r + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        b = tl.load(ptrs_b,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # --------------------------------------------------------------
        # Multiply‑accumulate
        # --------------------------------------------------------------
        acc += tl.dot(a, b)    # (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # Store the result (cast back to fp16)
    # ------------------------------------------------------------------
    out = acc.to(tl.float16)
    ptrs_o = base_o + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(ptrs_o, out, mask=mask_i[:, None] & mask_j[:, None])


# ----------------------------------------------------------------------
# Wrapper that invokes the Triton kernel
# ----------------------------------------------------------------------
def _trmul_gemm_fp16(
    left: torch.Tensor, right: torch.Tensor
) -> torch.Tensor:
    """
    Batched GEMM C = A @ Bᵀ for the outgoing TriMul operation.
    All tensors must be contiguous fp16 with shape [B, N, N, H].
    Returns a tensor of the same shape (fp16) containing the contracted result.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape

    # Allocate output tensor
    out = torch.empty_like(left, dtype=torch.float16)

    # Strides (in element counts, not bytes)
    lsb, lsi, lsk, lsh = left.stride()
    rsb, rsi, rsk, rsh = right.stride()
    osb, osi, osj, osh = out.stride()

    # Tunable tile sizes – these work well on H100 for fp16
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # small K tile fits into shared memory, good for fp16

    grid = (
        ( (N + BLOCK_M - 1) // BLOCK_M, ),
        ( (N + BLOCK_N - 1) // BLOCK_N, ),
        ( B * H, ),
    )
    # Triton expects the arguments as integers (pointer = int64)
    _trmul_gemm_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, H,
        lsb, lsi, lsk, lsh,
        rsb, rsi, rsk, rsh,
        osb, osi, osj, osh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,        # 8 warps per block works well for fp16 on H100
        num_stages=3,       # double‑buffering of shared memory
    )
    return out


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor[B, N, N]       (float32 / bool) or None
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for stability)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]   (float32)

    # --------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part of the forward pass
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Cast model weights once to fp16 (avoids repeated casts)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # --------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x_fp16, left_proj_w)   # [B,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x_fp16, right_proj_w)  # same shape

    # --------------------------------------------------------------
    # 5️⃣ Optional mask (broadcasted over the hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        # Cast mask to fp16 and broadcast
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # --------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – fp16
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_w))

    # Fuse linear output with gates (in‑place multiplication)
    left = left_proj * left_gate
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 7️⃣ O(N³) contraction via the custom Triton GEMM kernel
    # --------------------------------------------------------------
    # left/right are already fp16, contiguous, shape [B,N,N,hidden_dim]
    out_hidden = _trmul_gemm_fp16(left, right)    # [B,N,N,hidden_dim] (fp16)

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension (fp16)
    # --------------------------------------------------------------
    out_hidden = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 9️⃣ Final gating + projection back to the original dimension
    # --------------------------------------------------------------
    out_hidden = out_hidden * out_gate
    out = torch.nn.functional.linear(out_hidden, to_out_w)   # [B,N,N,dim] (fp16)

    # --------------------------------------------------------------
    # 10️⃣ Cast to float32 as required by the harness
    # --------------------------------------------------------------
    return out.to(torch.float32)
