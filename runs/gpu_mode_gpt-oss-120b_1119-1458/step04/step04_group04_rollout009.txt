    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    # compute mask index (broadcast across hidden dimension)
    mask_idx = offs // hidden_dim
    mask_n = size // hidden_dim
    mask_valid = mask_idx < mask_n
    m = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)

def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for TriMul contraction: C = A @ Bᵀ.
    Input tensors are fp16, shape [B, N, N, H].
    Returns fp16 tensor of same shape.
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N]
    a = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    b = right.permute(0, 3, 1, 2).reshape(B * H, N, N)
    c = torch.bmm(a, b.transpose(1, 2))
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Outgoing `TriMul` forward pass (AlphaFold‑3).

    Steps
    -----
    1. Layer‐norm the 4‑D input (float32) and cast to fp16.
    2. Fuse *three* linear projections (left_proj, left_gate, out_gate) with a
       single matmul, and similarly fuse right_proj & right_gate.
    3. Apply sigmoid to the gate tensors.
    4. If a mask is supplied, fuse ``tensor * gate * mask`` in‑place via a

    5. Perform the heavy O(N³) contraction as a batched GEMM

    6. Second layer‑norm over the hidden dimension, element‑wise out‑gate,

    7. Cast the result back to float32 for the harness.
    """

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) → fp16
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2️⃣ Fuse linear projections:

    #    right_side = [right_proj, right_gate]          → (2*hidden, dim)
    # -----------------------------------------------------------------
    left_w = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["out_gate.weight"]

    right_w = torch.cat([
        weights["right_proj.weight"],
        weights["right_gate.weight"]

    # fused matmul, shape [..., 3*hidden] and [..., 2*hidden]
    left_fused = torch.nn.functional.linear(x, left_w)   # [B,N,N,3*hidden]
    right_fused = torch.nn.functional.linear(x, right_w) # [B,N,N,2*hidden]

    # split into logical tensors
    left_proj, left_gate_raw, out_gate_raw = torch.split(left_fused, hidden_dim, dim=-1)
    right_proj, right_gate_raw = torch.split(right_fused, hidden_dim, dim=-1)

    # 3️⃣ Sigmoid gates
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # -----------------------------------------------------------------
    # 4️⃣ Optional mask + gate fusion via Triton
    # -----------------------------------------------------------------
    if not nomask:
        # flatten mask for broadcasting across hidden dim
        mask_fp16 = mask_tensor.to(torch.float16).reshape(-1)  # [B*N*N]
        size = left_proj.numel()                               # B*N*N*hidden
        BLOCK = 1024

            mask_fp16.data_ptr(),

            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right_proj = right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            size,
            hidden_dim,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # no mask → simple element‑wise fusion
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched GEMM (cuBLAS)
    # -----------------------------------------------------------------
    out = _batched_matmul_torch(left_proj, right_proj)   # [B,N,N,hidden] fp16

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm (fp16) → out‑gate → final projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 7️⃣ Cast back to float32
    # -----------------------------------------------------------------
    return out.to(torch.float32)

"""
Outgoing TriMul (AlphaFold‑3) forward pass – Triton‑accelerated.

Algorithm
---------
1️⃣ First LayerNorm (float32) → cast to fp16.
2️⃣ Fuse three linear projections on the “left” side (left_proj, left_gate, out_gate)
   and two projections on the “right” side (right_proj, right_gate) into two
   matmuls by concatenating the weight matrices.  This cuts the number of
   GEMMs from five to two.
3️⃣ Apply sigmoid to the three gate tensors.
4️⃣ If a mask is supplied, fuse the element‑wise computation
   `tensor = tensor * gate * mask` with a tiny Triton kernel that broadcasts
   the mask along the hidden dimension; otherwise fuse only `tensor * gate`.
5️⃣ Heavy O(N³) contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM `C = A @ Bᵀ` using torch.bmm (cuBLAS Tensor‑Core)
   on tensors of shape [B·H, N, N].
6️⃣ Second LayerNorm (fp16) → element‑wise out‑gate → final linear projection
   back to `dim`.
7️⃣ Cast the result to float32 for the harness.

At least one operation (mask‑gate fusion) is implemented in a Triton kernel,
while the computationally dominant contraction uses cuBLAS for maximal throughput.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (tensor * gate * mask) in‑place.
# The mask is a 1‑D tensor of shape [B·N·N] and is broadcast over the
# hidden dimension inside the kernel.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,                # *fp16 : tensor to be updated (size = B*N*N*H)
    gate_ptr,             # *fp16 : gate tensor (same shape as x)
    mask_ptr,             # *fp16 : mask tensor, length = B*N*N (broadcasted)
    size,                 # i32   : total number of elements in x/gate
    hidden_dim,           # i32   : hidden dimension (gate dimension)
    BLOCK: tl.constexpr,  # compile‑time tile size (≤1024 threads per block)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < size

    # Load tensor and gate
    x = tl.load(x_ptr + offs, mask=valid, other=0.0)
    g = tl.load(gate_ptr + offs, mask=valid, other=0.0)

    # Compute mask index = element index // hidden_dim (broadcast)
    mask_idx = offs // hidden_dim
    mask_len = size // hidden_dim
    mask_valid = mask_idx < mask_len
    m = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)

    # Fused multiplication
    out = x * g * m

    tl.store(x_ptr + offs, out, mask=valid)


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the TriMul contraction.
    Inputs are fp16 tensors of shape [B, N, N, H] and the output has the
    same shape.  Internally we reshape to [B·H, N, N] and use torch.bmm.
    """
    B, N, _, H = left.shape
    # [B·H, N, N]
    a = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    b = right.permute(0, 3, 1, 2).reshape(B * H, N, N)
    # GEMM with transposed right operand
    c = torch.bmm(a, b.transpose(1, 2))
    # Restore original layout
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out
