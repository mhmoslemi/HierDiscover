    b = pos // (N*N)

    j = rem % N

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # 1) LayerNorm using Triton
    # Convert weight & bias to fp16
    norm_weight_fp16 = weights["norm.weight"].to(torch.float16)
    norm_bias_fp16 = weights["norm.bias"].to(torch.float16)
    x_norm_fp16 = _layer_norm_triton(
        input_tensor,
        norm_weight_fp16,
        norm_bias_fp16,
        eps,
        dim
    ) # shape B,N,N,dim half

    # 2) Fused linear for left+gate and right+gate
    left_fused_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    right_fused_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)

    # compute fused linear
    left_concat = torch.nn.functional.linear(x_norm_fp16, left_fused_weight) # shape B,N,N,2*hidden_dim
    right_concat = torch.nn.functional.linear(x_norm_fp16, right_fused_weight) # shape B,N,N,2*hidden_dim

    # split
    left_proj_raw = left_concat[..., :hidden_dim]
    left_gate_raw = left_concat[..., hidden_dim:]
    right_proj_raw = right_concat[..., :hidden_dim]
    right_gate_raw = right_concat[..., hidden_dim:]

    # mask if needed (mask shape [B,N,N]), broadcast to hidden_dim dimension
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # shape B,N,N,1
        left_proj_raw = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    # gating via sigmoid
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # apply gating
    left = left_proj_raw * left_gate
    right = right_proj_raw * right_gate

    # 3) out_gate from x_norm
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    out_gate_raw = torch.nn.functional.linear(x_norm_fp16, out_gate_weight)  # shape B,N,N,hidden_dim
    out_gate = torch.sigmoid(out_gate_raw)

    # 4) O(N³) contraction via batched matmul
    # Use batched GEMM
    # Note: we need to reshape left/right to [B*hidden_dim, N, N] and right transposed for matmul.
    B, N, _, H = left.shape

    # reshape for batch
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)
    out_flat = torch.bmm(left_flat, right_flat)  # shape: [B*H, N, N] -> fp16

    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape B,N,N,H

    # 5) second layernorm over hidden_dim
    to_out_norm_weight_fp16 = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias_fp16 = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight_fp16,
        bias=to_out_norm_bias_fp16,
    )  # still fp16 by default as input fp16

    # 6) gating with out_gate and final projection
    out = out * out_gate
    to_out_weight_fp16 = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_weight_fp16)  # shape B,N,N,dim

    # 7) cast to float32
    return out.to(torch.float32)

@triton.jit
def _layer_norm_kernel(
    out_ptr, inp_ptr,
    weight_ptr, bias_ptr,
    D: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_D: tl.constexpr
):
    row = tl.program_id(0)  # between 0 and P-1
    # base pointers
    out_row_ptr = out_ptr + row * D
    inp_row_ptr = inp_ptr + row * D

    sum_val = tl.float32(0.0)
    sum_sq_val = tl.float32(0.0)
    d = 0
    while d < D:
        cur = d + tl.arange(0, BLOCK_D)
        mask = cur < D
        x = tl.load(inp_row_ptr + cur, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x)
        sum_sq_val += tl.sum(x * x)
        d += BLOCK_D

    mean = sum_val / D
    var = sum_sq_val / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # second pass: write normalized values with weight and bias
    d = 0
    while d < D:
        cur = d + tl.arange(0, BLOCK_D)
        mask = cur < D
        x = tl.load(inp_row_ptr + cur, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(weight_ptr + cur, mask=mask, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + cur, mask=mask, other=0.0).to(tl.float32)
        out_f32 = ((x - mean) * inv_std) * w + b
        out_f16 = out_f32.to(tl.float16)
        tl.store(out_row_ptr + cur, out_f16, mask=mask)
        d += BLOCK_D

@triton.jit
def _layer_norm_kernel(
    out_ptr, inp_ptr,
    weight_ptr, bias_ptr,
    DIM: tl.constexpr,
    EPS: tl.constexpr,
    BLOCK_D: tl.constexpr
):
    row = tl.program_id(0)

    # Compute base pointers
    out_row_ptr = out_ptr + row * DIM
    inp_row_ptr = inp_ptr + row * DIM

    # First pass: compute mean and variance
    sum_val = tl.float32(0.0)
    sum_sq_val = tl.float32(0.0)
    offs = 0
    while offs < DIM:
        cur_offs = offs + tl.arange(0, BLOCK_D)
        mask = cur_offs < DIM
        x = tl.load(inp_row_ptr + cur_offs, mask=mask, other=0.0)
        x_f = x.to(tl.float32)
        sum_val += tl.sum(x_f, axis=0)
        sum_sq_val += tl.sum(x_f * x_f, axis=0)
        offs += BLOCK_D

    mean = sum_val / DIM
    var = sum_sq_val / DIM - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # Second pass: store normalized result with weight & bias
    offs = 0
    while offs < DIM:
        cur_offs = offs + tl.arange(0, BLOCK_D)
        mask = cur_offs < DIM
        x = tl.load(inp_row_ptr + cur_offs, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(weight_ptr + cur_offs, mask=mask, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + cur_offs, mask=mask, other=0.0).to(tl.float32)
        out_f = ((x - mean) * inv_std) * w + b
        out_h = out_f.to(tl.float16)
        tl.store(out_row_ptr + cur_offs, out_h, mask=mask)
        offs += BLOCK_D
```

"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Layer‑norm the input (float32) – implemented in a tiny Triton kernel that
    directly writes the affine‑normalized result in FP16, thus avoiding an
    extra cast step.
2️⃣  Fuse the four linear projects (left/right and their gates) into two
    large GEMM calls:
        left_fused = Linear(x_norm, [left_proj; left_gate])
        right_fused = Linear(x_norm, [right_proj; right_gate])
    The concatenated output is split into projection / gate tensors.
3️⃣  Apply the optional input mask (broadcasted), the sigmoid gates and the
    element‑wise gate × projection fusion.
4️⃣  Compute the O(N³) contraction as a batched GEMM with the cuBLAS
    dispatcher (torch.bmm) – this is the heaviest work and already runs at
    near‑Tensor‑Core speed on an H100.
5️⃣  Layer‑norm over the hidden dimension, multiply by the output gate
    (computed from the same normalized input) and project back to the
    original feature dimension.
6️⃣  Cast the final result back to float32.

Only the first Layer‑Norm is written in Triton (as required), all heavy
matrix‑multiplications stay on cuBLAS for maximal throughput.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton Layer‑Norm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def _layer_norm_fp16_kernel(
    out_ptr,               # *fp16   -> normalized output (flattened)
    inp_ptr,               # *fp32   -> input tensor (flattened)
    weight_ptr,            # *fp16   -> LN weight   (dim)
    bias_ptr,              # *fp16   -> LN bias     (dim)
    DIM: tl.constexpr,     # size of the feature dimension
    EPS: tl.constexpr,     # numerical epsilon
    BLOCK_D: tl.constexpr, # threads per block (covers a chunk of DIM)
):
    """
    One CUDA program (block) processes a single row of the input
    [B·N·N, DIM].  Threads cooperate to compute mean / variance, then
    write the affine‑normalized values in fp16.
    """
    row = tl.program_id(0)                     # row index = b·N·N + i·N + j

    # base pointers for this row
    out_row_ptr = out_ptr + row * DIM
    inp_row_ptr = inp_ptr + row * DIM

    # ------------------------------------------------------------------
    # 1️⃣  Compute mean and variance (float32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq_val = tl.float32(0.0)

    # each thread loads one element per iteration
    offs = tl.arange(0, BLOCK_D)                # thread id inside the block

    # loop over the feature dimension
    start = 0
    while start < DIM:
        idx = start + offs                      # absolute column index
        mask = idx < DIM
        x = tl.load(inp_row_ptr + idx, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x)                    # scalar reduction inside the thread
        sum_sq_val += tl.sum(x * x)
        start += BLOCK_D

    mean = sum_val / DIM
    var = sum_sq_val / DIM - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # ------------------------------------------------------------------
    # 2️⃣  Write normalized values, applying weight & bias
    # ------------------------------------------------------------------
    start = 0
    while start < DIM:
        idx = start + offs
        mask = idx < DIM
        x = tl.load(inp_row_ptr + idx, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(weight_ptr + idx, mask=mask, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + idx, mask=mask, other=0.0).to(tl.float32)

        out_f32 = ((x - mean) * inv_std) * w + b
        out_f16 = out_f32.to(tl.float16)

        tl.store(out_row_ptr + idx, out_f16, mask=mask)
        start += BLOCK_D


def _layer_norm_fp16(x: torch.Tensor,
                    weight: torch.Tensor,
                    bias: torch.Tensor,
                    eps: float = 1e-5) -> torch.Tensor:
    """
    Apply LayerNorm to a [B, N, N, D] tensor (float32) and obtain an
    fp16 tensor with the affine transformation baked in.
    """
    B, N, _, D = x.shape
    total_rows = B * N * N

    # flatten to [total_rows, D] – must be contiguous for pointer arithmetic
    x_flat = x.view(total_rows, D).contiguous()
    out_flat = torch.empty_like(x_flat, dtype=torch.float16)

    # cast weight & bias once
    weight_h = weight.to(torch.float16).contiguous()
    bias_h = bias.to(torch.float16).contiguous()

    # raw pointers (element‑wise, not byte strides)
    out_ptr = out_flat.data_ptr()
    inp_ptr = x_flat.data_ptr()
    weight_ptr = weight_h.data_ptr()
    bias_ptr = bias_h.data_ptr()

    # Simple configuration: 128 threads per block (multiple of 32)
    BLOCK_D = 128

    grid = (total_rows,)
    _layer_norm_fp16_kernel[grid](
        out_ptr,
        inp_ptr,
        weight_ptr,
        bias_ptr,
        DIM=D,
        EPS=eps,
        BLOCK_D=BLOCK_D,
        # Triton launch configuration
        num_warps=4,
        num_stages=2,
    )
    return out_flat.view(B, N, N, D)


# ----------------------------------------------------------------------
# Batched GEMM (left @ rightᵀ) – uses cuBLAS via torch.bmm
# ----------------------------------------------------------------------
def _batched_matmul_transpose(left: torch.Tensor,
                              right: torch.Tensor) -> torch.Tensor:
    """
    left/right: [B, N, N, H] (fp16)
    Returns:    [B, N, N, H] (fp16) where out = left @ rightᵀ (batch‑wise).
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)
    out_flat = torch.bmm(left_flat, right_flat)  # cuBLAS tensor‑core GEMM
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor,
                torch.Tensor,
                Dict[str, torch.Tensor],
                Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor[B, N, N]       (bool / float32)
        - weights_dict : name → torch.Tensor (float32, on same device)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm (float32 → fp16) – Triton implementation
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = _layer_norm_fp16(input_tensor, norm_weight, norm_bias)  # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 2️⃣  Fuse left/right projections and their gates
    # ------------------------------------------------------------------
    # left + left_gate
    left_fused_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    # right + right_gate
    right_fused_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)

    # Linear calls (fp16 in, fp16 out)
    left_concat = torch.nn.functional.linear(x, left_fused_w)   # [B, N, N, 2*H]
    right_concat = torch.nn.functional.linear(x, right_fused_w) # [B, N, N, 2*H]

    # Split projection / gate tensors
    left_proj = left_concat[..., :hidden_dim]                # [B,N,N,H]
    left_gate_raw = left_concat[..., hidden_dim:]            # [B,N,N,H]
    right_proj = right_concat[..., :hidden_dim]
    right_gate_raw = right_concat[..., hidden_dim:]

    # ------------------------------------------------------------------
    # 3️⃣  Optional mask (broadcasted on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣  Gating (sigmoid) and fuse projection × gate
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    left = left_proj * left_gate       # [B,N,N,H] fp16
    right = right_proj * right_gate   # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 5️⃣  Compute out_gate (still using the normalized input)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))   # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 6️⃣  O(N³) contraction via batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose(left, right)   # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 7️⃣  Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # fp16 in / fp16 out

    # ------------------------------------------------------------------
    # 8️⃣  Apply out_gate and project back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)   # shape [dim, hidden_dim]
    out = torch.nn.functional.linear(out, to_out_w)        # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 9️⃣  Cast back to float32 as required by the harness
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
