def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # layernorm weight & bias as float32
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # first layernorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    # cast to half
    x = x.to(torch.float16)

    # fused weight for 5 linear projections (no bias)
    fused_weight = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16)  # shape [5*hidden_dim, dim]

    # fused linear: shape [B,N,N,5*hidden_dim]
    fused = F.linear(x, fused_weight)  # (B,N,N,5H) fp16

    # split
    H = hidden_dim
    left_raw = fused[..., :H]
    left_gate_raw = fused[..., H:2*H]
    right_raw = fused[..., 2*H:3*H]
    right_gate_raw = fused[..., 3*H:4*H]
    out_gate_raw = fused[..., 4*H:5*H]

    # free fused if needed after using (preserve memory) 
    # not needed but can delete
    del fused

    # gating
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # apply gating and mask
    left = left_raw * left_gate
    right = right_raw * right_gate
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16
        del mask_fp16
    # free raw and gate raw to release memory
    del left_raw, left_gate_raw, right_raw, right_gate_raw, out_gate_raw

    # heavy contraction using triton kernel (or torch's bmm)
    B = left.shape[0]
    N = left.shape[1]
    H = hidden_dim

    # choose kernel path: for all sizes for simplicity
    # Use triton kernel if N <= 512 maybe else torch.bmm for better performance
    if N <= 512:
        out = _batched_matmul_triton(left, right)
    else:
        out = _batched_matmul_torch(left, right)

    # second layernorm
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # out_gate multiplication
    out = out * out_gate
    del out_gate

    # final linear projection
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)

    # cast back to float32
    return out.to(torch.float32)

def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor):
    B, N, _, H = left.shape
    # left shape [B,N,N,H]
    # permute to shape [M,N,N] where M = B*H
    A = left.permute(0,3,1,2).contiguous().view(B*H, N, N)
    B_ = right.permute(0,3,1,2).contiguous().view(B*H, N, N)

    C = torch.empty_like(A) # fp16

    # get strides
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    BLOCK_K = 64
    # Choose block sizes based on N
    if N <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
    elif N <= 512:
        BLOCK_M = 128
        BLOCK_N = 128
    else:
        BLOCK_M = 128
        BLOCK_N = 128

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B*H,
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B*H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    # reshape output back
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C
```

"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

Algorithm
---------
1️⃣  Layer‑norm the input (float32) → cast to fp16.
2️⃣  Fuse the five linear projections (left/right and three gates) into a single
    matmul:  (left_proj, left_gate, right_proj, right_gate, out_gate) =
    x @ W_fusedᵀ   (fp16).
3️⃣  Split the fused result, apply sigmoid to the gate slices and (optionally)
    the mask to left/right tensors.
4️⃣  Heavy contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is performed as a batched GEMM C = A @ Bᵀ.
    • For small sequence lengths (≤256) a custom Triton kernel
      (64×64×64 tiles) is used.
    • For larger N we fall back to torch.bmm (cuBLAS) which is
      heavily optimized on H100.
5️⃣  Second LayerNorm (fp16) → element‑wise out‑gate (sigmoid) → final
    Linear back to `dim`.
6️⃣  Cast to float32 (required by the harness).

The implementation dramatically reduces memory traffic by fusing the five
linear layers into one matmul and keeps a Triton kernel for the O(N³)
contraction on the small‑N regime.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (fp16 → fp16, accumulator fp32)
#   A, B : [M, N, K]  (M = B·hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K] (will be read as transposed)
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 dimensions
    stride_am, stride_ai, stride_ak,   # A strides (M,i,k)
    stride_bm, stride_bj, stride_bk,   # B strides (M,j,k) – note the transposed view
    stride_cm, stride_ci, stride_cj,   # C strides (M,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel for the TriMul contraction."""
    pid_m = tl.program_id(0)   # tile along i dimension
    pid_n = tl.program_id(1)   # tile along j dimension
    pid_z = tl.program_id(2)   # batch·hidden index (M)

    # ----------- tile offsets ----------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ------- load A tile (i, k) ---------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # ------- load B tile (k, j) – transposed view ----------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk   # k dimension first (row)
            + offs_j[None, :] * stride_bj   # j dimension second (col)
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # mixed‑precision dot: fp16 × fp16 → fp32 accumulate
        acc += tl.dot(a, b)

    # ----------- store result ------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(
    left: torch.Tensor,
    right: torch.Tensor,
    *,
    block_m: int = 64,
    block_n: int = 64,
    block_k: int = 64,
) -> torch.Tensor:
    """
    Contract left and right (both [B, N, N, H]) using the Triton kernel.
    Returns tensor of shape [B, N, N, H] (fp16).
    """
    B, N, _, H = left.shape
    # reshape to [M, N, N] where M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)          # fp16 output

    # strides in element units
    stride_am, stride_ai, stride_ak = A.stride()   # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()  # (N*N, N, 1) – note ordering for transpose
    stride_cm, stride_ci, stride_cj = C.stride()   # (N*N, N, 1)

    grid = (
        (N + block_m - 1) // block_m,   # i‑tiles
        (N + block_n - 1) // block_n,   # j‑tiles
        B * H,                          # batch·hidden
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
        num_warps=8,
        num_stages=2,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right using torch.bmm (cuBLAS).
    All tensors are fp16; accumulation is fp32 automatically.
    """
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)          # [M,N,N]
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)        # [M,N,N]

    # Bmm with transposed second operand (Bᵀ)
    C = torch.bmm(A, B_.transpose(1, 2))                                 # [M,N,N]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()              # [B,N,N,H]
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]],
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 / bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # 1️⃣ unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # 2️⃣ first LayerNorm (float32)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # --------------------------------------------------------------
    # 3️⃣ FP16 cast for the rest of the pipeline
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4️⃣ Fuse the five linear projections into a single matmul
    # --------------------------------------------------------------
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape [5*hidden_dim, dim]

    # (B,N,N,5*H) fp16
    fused = F.linear(x, fused_weight)

    # --------------------------------------------------------------
    # 5️⃣ Slice and apply sigmoid gates
    # --------------------------------------------------------------
    H = hidden_dim
    left_raw      = fused[..., 0:H]               # [B,N,N,H]
    left_gate_raw = fused[..., H:2 * H]
    right_raw     = fused[..., 2 * H:3 * H]
    right_gate_raw = fused[..., 3 * H:4 * H]
    out_gate_raw  = fused[..., 4 * H:5 * H]

    # free fused early to save peak memory
    del fused

    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    # --------------------------------------------------------------
    # 6️⃣ Apply mask (if any) and fuse gate multiplications
    # --------------------------------------------------------------
    left = left_raw * left_gate
    right = right_raw * right_gate

    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16
        del mask_fp16

    # discard intermediate tensors no longer needed
    del left_raw, left_gate_raw, right_raw, right_gate_raw, out_gate_raw

    # --------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction
    # --------------------------------------------------------------
    seq_len = input_tensor.shape[1]

    # Choose implementation based on seq_len (triton for very small, torch.bmm otherwise)
    if seq_len <= 256:
        # use a modest tile size for very small N
        out = _batched_matmul_triton(
            left, right, block_m=64, block_n=64, block_k=64
        )
    else:
        # for all larger N the cuBLAS path is faster
        out = _batched_matmul_torch(left, right)

    # --------------------------------------------------------------
    # 8️⃣ second LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate               # element‑wise gating
    del out_gate                        # free

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)      # [B,N,N,dim] (fp16)

    # --------------------------------------------------------------
    # 9️⃣ cast back to float32 (expected by the harness)
    # --------------------------------------------------------------
    return out.to(torch.float32)
```
