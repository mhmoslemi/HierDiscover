"""
Highly‑optimized forward‐only Triton‑accelerated implementation of the
“outgoing” TriMul operator (AlphaFold3).  The main ideas are:

* Fuse the five linear layers (two projections, three gates) into a single
  matrix multiplication to cut down on memory traffic.
* Fuse the projection → mask → sigmoid(gate) → elementwise multiplication
  into two Triton kernels (with/without mask).  This avoids allocating
  intermediate tensors and eliminates an explicit torch.sigmoid call.
* Use a cuBLAS batched GEMM (torch.bmm) for the O(N³) contraction
  ``out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`` – the fastest
  approach on modern GPUs such as the H100.
* Fuse the final output‑gate (sigmoid) into a third Triton kernel.
* Keep the heavy arithmetic in FP16 (half) and cast back to FP32 only for
  the final result.  All layer‑norms use the half‑precision weights.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mul_kernel(
    proj_ptr,     # half*   (B·N·N·H)
    gate_ptr,     # half*   (B·N·N·H)
    out_ptr,      # half*   (B·N·N·H)
    size,         # i32
    stride_proj,  # i32
    stride_gate,  # i32
    stride_out,   # i32
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    p = tl.load(proj_ptr + offs * stride_proj, mask=mask)
    g = tl.load(gate_ptr + offs * stride_gate, mask=mask)
    # sigmoid(g) = 1/(1+exp(-g))
    sig = 1.0 / (1.0 + tl.exp(-g))
    tl.store(out_ptr + offs * stride_out, p * sig, mask=mask)


@triton.jit
def _proj_gate_mask_mul_kernel(
    proj_ptr,      # half*   (B·N·N·H)
    gate_ptr,      # half*   (B·N·N·H)
    mask_ptr,      # half*   (B·N·N)   – broadcasted over H
    out_ptr,       # half*   (B·N·N·H)
    size,          # i32
    hidden_dim,    # i32   – H
    stride_proj,   # i32
    stride_gate,   # i32
    stride_out,    # i32
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    p = tl.load(proj_ptr + offs * stride_proj, mask=mask)
    g = tl.load(gate_ptr + offs * stride_gate, mask=mask)

    # compute mask index = offs // H  (broadcast H times)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask)   # 0/1 in half

    sig = 1.0 / (1.0 + tl.exp(-g))
    tl.store(out_ptr + offs * stride_out, p * m * sig, mask=mask)


@triton.jit
def _out_gate_mul_sigmoid_kernel(
    norm_ptr,      # half*   (B·N·N·H)
    gate_ptr,      # half*   (B·N·N·H)
    out_ptr,       # half*   (B·N·N·H)
    size,          # i32
    stride_norm,   # i32
    stride_gate,   # i32
    stride_out,    # i32
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    n = tl.load(norm_ptr + offs * stride_norm, mask=mask)
    g = tl.load(gate_ptr + offs * stride_gate, mask=mask)
    sig = 1.0 / (1.0 + tl.exp(-g))
    tl.store(out_ptr + offs * stride_out, n * sig, mask=mask)


# ----------------------------------------------------------------------
# Helper wrappers around the kernels
# ----------------------------------------------------------------------
def _apply_fused_proj_gate(proj: torch.Tensor, gate: torch.Tensor) -> torch.Tensor:
    """proj * sigmoid(gate)  (no mask).  All tensors are half‑precision."""
    total = proj.numel()
    out = torch.empty_like(proj)
    block = 65536
    grid = (triton.cdiv(total, block),)

    proj_flat = proj.contiguous().view(-1)
    gate_flat = gate.contiguous().view(-1)
    out_flat = out.view(-1)

    _proj_gate_mul_kernel[grid](
        proj_flat,
        gate_flat,
        out_flat,
        total,
        1, 1, 1,
        BLOCK_SIZE=block,
        num_warps=4,
    )
    return out


def _apply_fused_proj_gate_mask(
    proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor, hidden_dim: int
) -> torch.Tensor:
    """
    proj * mask * sigmoid(gate), where ``mask`` has shape [B,N,N] (half)
    and is broadcast over the hidden dimension.
    """
    total = proj.numel()
    out = torch.empty_like(proj)
    block = 65536
    grid = (triton.cdiv(total, block),)

    proj_flat = proj.contiguous().view(-1)
    gate_flat = gate.contiguous().view(-1)
    mask_flat = mask.contiguous().view(-1)          # shape B·N·N
    out_flat = out.view(-1)

    _proj_gate_mask_mul_kernel[grid](
        proj_flat,
        gate_flat,
        mask_flat,
        out_flat,
        total,
        hidden_dim,
        1, 1, 1,
        BLOCK_SIZE=block,
        num_warps=4,
    )
    return out


def _apply_out_gate_mul(norm: torch.Tensor, gate: torch.Tensor) -> torch.Tensor:
    """norm * sigmoid(gate).  All tensors are half‑precision."""
    total = norm.numel()
    out = torch.empty_like(norm)
    block = 65536
    grid = (triton.cdiv(total, block),)

    norm_flat = norm.contiguous().view(-1)
    gate_flat = gate.contiguous().view(-1)
    out_flat = out.view(-1)

    _out_gate_mul_sigmoid_kernel[grid](
        norm_flat,
        gate_flat,
        out_flat,
        total,
        1, 1, 1,
        BLOCK_SIZE=block,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Batched matmul (C = A @ Bᵀ)
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    using a cuBLAS batched GEMM (torch.bmm).  All tensors are half‑precision.
    """
    B, N, _, H = left.shape
    # (B, H, N, N) → (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    # GEMM with right transposed
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)
    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry‑point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,
        torch.Tensor,
        Dict[str, torch.Tensor],
        Dict[str, Any],
    ]
) -> torch.Tensor:
    """
    Forward pass for the outgoing TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 / bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    x = x.to(torch.float16)            # [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Assemble combined weight for the five linear layers
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)    # (H, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    combined_weight = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    )  # (5*H, dim) in FP16

    # ------------------------------------------------------------------
    # 3️⃣ Single fused linear pass (produces all projections & gates)
    # ------------------------------------------------------------------
    combined = F.linear(x, combined_weight)  # [B, N, N, 5*H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Slice the combined tensor
    # ------------------------------------------------------------------
    slices = torch.split(combined, hidden_dim, dim=-1)
    left_proj_raw, right_proj_raw, left_gate_raw, right_gate_raw, out_gate_raw = slices

    # ------------------------------------------------------------------
    # 5️⃣ Apply mask & gates (fused Triton kernels)
    # ------------------------------------------------------------------
    if nomask:
        left = _apply_fused_proj_gate(left_proj_raw, left_gate_raw)   # [B,N,N,H]
        right = _apply_fused_proj_gate(right_proj_raw, right_gate_raw)
    else:
        # mask is [B,N,N] → broadcast over H
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)      # [B,N,N,1]
        left = _apply_fused_proj_gate_mask(
            left_proj_raw, left_gate_raw, mask_fp16.squeeze(-1), hidden_dim
        )
        right = _apply_fused_proj_gate_mask(
            right_proj_raw, right_gate_raw, mask_fp16.squeeze(-1), hidden_dim
        )

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via cuBLAS batched GEMM
    # ------------------------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B, N, N, H] FP16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (still FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Output gate (sigmoid) fused with multiplication
    # ------------------------------------------------------------------
    out = _apply_out_gate_mul(out, out_gate_raw)  # [B,N,N,H] FP16

    # ------------------------------------------------------------------
    # 9️⃣ Final projection back to dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, H)
    out = F.linear(out, to_out_w)                          # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 🔚 Cast back to FP32 for the final result
    # ------------------------------------------------------------------
    return out.to(torch.float32)
