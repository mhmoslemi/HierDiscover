@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute pointers
    a_ptrs = a_ptr + batch_idx * stride_a_batch + offs_m[:,None] * stride_a_m + offs_k[None,:] * stride_a_k
    b_ptrs = b_ptr + batch_idx * stride_b_batch + offs_k[:,None] * stride_b_k + offs_n[None,:] * stride_b_n
    # note: passing b stride to treat as transposed

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load tiles
        a = tl.load(a_ptrs, mask=(offs_m[:,None] < M) & (offs_k[None,:] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:,None] < K) & (offs_n[None,:] < N), other=0.0)
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)
        # update pointers
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # store result
    c_ptrs = c_ptr + batch_idx * stride_c_batch + offs_m[:,None] * stride_c_m + offs_n[None,:] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m[:,None] < M) & (offs_n[None,:] < N))
```

"""
TriMul (outgoing) – fast Triton‑fused implementation
===================================================

The forward pass consists of the following stages:

1️⃣  LayerNorm on the input (FP32 → FP16).
2️⃣  **Fused linear projection** – we pack the five linear layers
    (left_proj, left_gate, right_proj, right_gate, out_gate) into a
    single matrix‑multiply, saving two memory passes over the input.
3️⃣  Sigmoid gates for the three *gate* projections.
4️⃣  Optional mask‑gate fusion (Triton):  out = proj * gate * mask.
5️⃣  Batched GEMM (left @ rightᵀ) performed with a custom Triton kernel.
    This replaces the O(N³) `einsum` and avoids the overhead of many
    calls to `torch.bmm`.
6️⃣  LayerNorm on the hidden dimension.
7️⃣  Output‑gate multiplication and final linear projection to the original
    feature dimension.
8️⃣  Cast back to float32 (the expected output type).

All heavy‐compute kernels run in FP16 (Tensor‑Core friendly) and the
final result is converted to FP32 for downstream code.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: mask * proj * gate   (M = B·N·N, H = hidden_dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,    # tensors
    M, H,                                     # problem size
    stride_proj_m, stride_proj_h,             # strides of proj
    stride_gate_m, stride_gate_h,             # strides of gate
    stride_out_m, stride_out_h,               # strides of out
    stride_mask,                              # stride of mask (1‑D)
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)                     # block index over rows (M)
    block_start = pid * BLOCK_M
    offs_m = block_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    offs_h = tl.arange(0, BLOCK_H)                       # [BLOCK_H]

    # Load mask (1‑D) and broadcast to H
    mask = tl.load(mask_ptr + offs_m * stride_mask,
                   mask=offs_m < M, other=0.0).to(tl.float16)   # [BLOCK_M]
    mask_h = mask[:, None]                                   # [BLOCK_M, 1]

    # Load projection and gate tiles
    proj = tl.load(
        proj_ptr + offs_m[:, None] * stride_proj_m +
                  offs_h[None, :] * stride_proj_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    gate = tl.load(
        gate_ptr + offs_m[:, None] * stride_gate_m +
                  offs_h[None, :] * stride_gate_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    out = proj * gate * mask_h

    # Store result
    tl.store(
        out_ptr + offs_m[:, None] * stride_out_m +
                 offs_h[None, :] * stride_out_h,
        out,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
    )


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B,C are FP16)
#   • A, B: (batch, N, N)   row‑major
#   • C: (batch, N, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,                               # M=N=K here (square)
    stride_a_batch, stride_a_i, stride_a_k,       # A strides
    stride_b_batch, stride_b_k, stride_b_j,       # Bᵀ strides (k,j)
    stride_c_batch, stride_c_i, stride_c_j,       # C strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    batch_idx = tl.program_id(0)   # which (batch * hidden) matrix
    pid_i = tl.program_id(1)       # block row
    pid_j = tl.program_id(2)       # block col

    # offsets within the block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the start of the batch
    a_batch_ptr = a_ptr + batch_idx * stride_a_batch
    b_batch_ptr = b_ptr + batch_idx * stride_b_batch
    c_batch_ptr = c_ptr + batch_idx * stride_c_batch

    # Accumulator (float32 for numerics)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        # Load tiles of A and Bᵀ
        a = tl.load(
            a_batch_ptr + offs_i[:, None] * stride_a_i + (offs_k + k * BLOCK_K)[None, :] * stride_a_k,
            mask=(offs_i[:, None] < M) &
                 ((offs_k + k * BLOCK_K)[None, :] < K),
            other=0.0,
        )
        b = tl.load(
            b_batch_ptr + (offs_k + k * BLOCK_K)[:, None] * stride_b_k + offs_j[None, :] * stride_b_j,
            mask=((offs_k + k * BLOCK_K)[:, None] < K) &
                 (offs_j[None, :] < N),
            other=0.0,
        )
        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Write back C (cast to fp16)
    c = acc.to(tl.float16)
    tl.store(
        c_batch_ptr + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j,
        c,
        mask=(offs_i[:, None] < M) & (offs_j[None, :] < N),
    )


# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul block (no backward).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32/bool) or None
        - weights       : dict of weight tensors (float32)
        - config        : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device as the input.
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    ).to(torch.float16)                     # [B, N, N, dim]  FP16

    B, N, _, _ = x.shape
    M = B * N * N                             # flattened pair count

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear transforms into one GEMM
    # ------------------------------------------------------------------
    # Convert all linear weights to FP16 once
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (hidden, dim)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # concatenate weights => shape (5*hidden, dim)
    combined_w = torch.cat(
        [left_proj_w,
         left_gate_w,
         right_proj_w,
         right_gate_w,
         out_gate_w],
        dim=0,
    )                                          # (5*hidden, dim)

    # flatten input to (M, dim)
    x_flat = x.view(-1, dim)                  # (M, dim)

    # single linear pass
    combined = torch.nn.functional.linear(x_flat, combined_w)   # (M, 5*hidden)

    # split
    h = hidden_dim
    left_proj  = combined[:, 0 * h : 1 * h]
    left_gate  = combined[:, 1 * h : 2 * h]
    right_proj = combined[:, 2 * h : 3 * h]
    right_gate = combined[:, 3 * h : 4 * h]
    out_gate   = combined[:, 4 * h : 5 * h]

    # --------------------------------------------------------------
    # 3️⃣ Apply sigmoid to gate tensors (still FP16)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)     # (M, hidden)
    right_gate = torch.sigmoid(right_gate)    # (M, hidden)
    out_gate   = torch.sigmoid(out_gate).view(B, N, N, hidden_dim)  # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 4️⃣ Apply mask & gate fusion (Triton) – only if mask is present
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_flat = mask_tensor.to(torch.float16).view(-1)   # (M,)

        # allocate fused tensors
        left  = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # strides in element counts
        stride_lp_m, stride_lp_h = left_proj.stride()
        stride_lg_m, stride_lg_h = left_gate.stride()
        stride_rp_m, stride_rp_h = right_proj.stride()
        stride_rg_m, stride_rg_h = right_gate.stride()
        stride_lout_m, stride_lout_h = left.stride()
        stride_rout_m, stride_rout_h = right.stride()
        stride_mask = mask_flat.stride()[0]

        # launch config – tune for FP16 hidden ≤128
        BLOCK_M = 64
        BLOCK_H = hidden_dim                 # we tile the whole hidden dim

        grid = ((M + BLOCK_M - 1) // BLOCK_M,)

        # left = left_proj * left_gate * mask
        _mask_gate_fuse_kernel[grid](
            left_proj, left_gate, mask_flat, left,
            M, hidden_dim,
            stride_lp_m, stride_lp_h,
            stride_lg_m, stride_lg_h,
            stride_lout_m, stride_lout_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
        # right = right_proj * right_gate * mask
        _mask_gate_fuse_kernel[grid](
            right_proj, right_gate, mask_flat, right,
            M, hidden_dim,
            stride_rp_m, stride_rp_h,
            stride_rg_m, stride_rg_h,
            stride_rout_m, stride_rout_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=4, num_stages=2,
        )
    else:
        # No mask – simple element‑wise fuse in PyTorch
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # reshape back to [B, N, N, hidden]
    left  = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Batched GEMM: out = left @ rightᵀ   (FP16)
    # ------------------------------------------------------------------
    # Rearrange to (B*hidden, N, N) where each "batch" is a separate GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, hidden, N, N)

    batch_hidden = B * hidden_dim
    left_mat  = left_perm.view(batch_hidden, N, N)
    right_mat = right_perm.view(batch_hidden, N, N)

    # allocate output buffer
    out_mat = torch.empty_like(left_mat)                # (B*hidden, N, N)

    # strides (in elements) – required for the Triton kernel
    stride_a_batch, stride_a_i, stride_a_k = left_mat.stride()    # (N*N, N, 1)
    stride_b_batch, stride_b_k, stride_b_j = right_mat.stride()   # (N*N, 1, N)
    stride_c_batch, stride_c_i, stride_c_j = out_mat.stride()    # (N*N, N, 1)

    # kernel launch grid
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch_hidden, num_i, num_j)

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch_hidden, N, N, N,
        stride_a_batch, stride_a_i, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_j,
        stride_c_batch, stride_c_i, stride_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # reshape back to original 4‑D layout (B, N, N, hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm on the hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Output gate and final projection
    # ------------------------------------------------------------------
    out = out * out_gate                               # element‑wise gate (FP16)
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, hidden)
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim) FP16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 (as required by the model)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
