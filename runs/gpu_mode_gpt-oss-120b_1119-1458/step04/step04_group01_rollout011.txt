"""
TriMul (outgoing) – Triton‑accelerated implementation
====================================================

The forward pass follows the reference implementation exactly but with
the following performance‑critical changes:

1️⃣  The first LayerNorm is performed with PyTorch (float32 → float16).
2️⃣  The two projection‐linear layers and their corresponding gates are
    fused into a single matmul per side (e.g. ``[proj, gate] = x @
    [W_proj; W_gate]``).  This halves the number of large GEMMs.
3️⃣  Optional binary mask is broadcast and applied after gating using
    ordinary PyTorch element‑wise multiplication (the mask is cheap).
4️⃣  The heavy O(N³) contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is implemented as a *batched* matrix multiplication using a custom
    Triton kernel.  The kernel works directly on the original layout
    ``[B, N, N, H]`` (no reshaping / permuting) and uses Tensor‑Core
    ``tl.dot`` with FP16 inputs and FP32 accumulation.
5️⃣  The second LayerNorm, output gate and final linear projection are
    performed in FP16 with PyTorch and the result is cast back to
    float32.

Only the contraction (the O(N³) part) is written in Triton, satisfying
the “must use Triton” requirement while keeping the overall runtime
well below the target 1 ms for the test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel for the batched GEMM step:
#   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]   (h = hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_batch_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides for the tensors (in element units, not bytes)
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    pid0 = tl.program_id(0)   # combined batch * hidden index
    pid1 = tl.program_id(1)   # tile row (i) index
    pid2 = tl.program_id(2)   # tile col (j) index

    # Decode batch and head (hidden) index
    batch = pid0 // H
    head = pid0 % H

    # Base offsets for this (batch, head)
    left_base  = batch * stride_l_b + head * stride_l_h
    right_base = batch * stride_r_b + head * stride_r_h
    out_base   = batch * stride_o_b + head * stride_o_h

    # Row / column coordinates within the N×N matrix
    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32 (Tensor‑Core accumulation)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile: left[batch, i, k, head]   shape (BLOCK_M, BLOCK_K)
        a_ptr = left_ptr + left_base \
                + offs_i[:, None] * stride_l_i \
                + offs_k[None, :] * stride_l_j
        a = tl.load(a_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)                      # half → promoted to FP32 for dot

        # B^T tile: right[batch, j, k, head]   (transposed on‑the‑fly)
        # shape (BLOCK_K, BLOCK_N) where rows=k, cols=j
        b_ptr = right_ptr + right_base \
                + offs_j[None, :] * stride_r_i \
                + offs_k[:, None] * stride_r_j
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Matrix‑multiply‑accumulate: (M×K)·(K×N) → (M×N)
        acc += tl.dot(a, b)   # uses Tensor‑Cores (FP16·FP16 → FP32)

    # Write the result tile back to memory (cast to FP16)
    out_ptrs = out_ptr + out_base \
               + offs_i[:, None] * stride_o_i \
               + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def _batched_gemm_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the TriMul contraction using the Triton kernel above.
    left / right: tensors of shape [B, N, N, H], dtype=half.
    Returns a tensor of the same shape and dtype.
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Strides in element units (PyTorch returns strides in elements)
    lsb, lsi, lsj, lsh = left.stride()
    rsb, rsi, rsj, rsh = right.stride()
    osb, osi, osj, osh = out.stride()

    # Tunable tile sizes – chosen to fit well on H100 Tensor Cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                     # batch * hidden dimension
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _tri_mul_batch_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        lsb,
        lsi,
        lsj,
        lsh,
        rsb,
        rsi,
        rsj,
        rsh,
        osb,
        osi,
        osj,
        osh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,              # Sufficient for H100
        num_stages=2,
    )
    return out


# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operation (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → half)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32
    x = x_norm.to(torch.float16)  # cast once for the rest of the pipeline

    # ------------------------------------------------------------------
    # 2️⃣ Cast all linear weights once to FP16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    # fuse projection + gate for the *left* side
    left_proj_gate_w = torch.cat([left_proj_w, weights["left_gate.weight"].to(torch.float16)], dim=0)
    # fuse projection + gate for the *right* side
    right_proj_gate_w = torch.cat([right_proj_w, weights["right_gate.weight"].to(torch.float16)], dim=0)

    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projection + gate in a single matmul
    # ------------------------------------------------------------------
    # left side: [B,N,N,2*hidden] → split into proj and gate
    left_combined = F.linear(x, left_proj_gate_w)          # no bias
    left_proj = left_combined[..., :hidden_dim]           # [B,N,N,hidden]
    left_gate = torch.sigmoid(left_combined[..., hidden_dim:])  # [B,N,N,hidden]

    # right side: same pattern
    right_combined = F.linear(x, right_proj_gate_w)
    right_proj = right_combined[..., :hidden_dim]
    right_gate = torch.sigmoid(right_combined[..., hidden_dim:])

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden) – multiplied after gating
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Compute the heavy O(N³) contraction via a Triton GEMM kernel
    # ------------------------------------------------------------------
    # out_hidden shape = [B, N, N, hidden] (half)
    out_hidden = _batched_gemm_triton(left, right)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (hidden dimension), output gate, final projection
    # ------------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # still FP16

    # out-gate (computed from the original normalized x)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # [B,N,N,hidden] FP16

    out_hidden = out_hidden * out_gate                   # element‑wise gate

    # Final linear projection back to original dim
    out = F.linear(out_hidden, to_out_w)                # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 7️⃣ Cast back to float32 for the API contract
    # ------------------------------------------------------------------
    return out.to(torch.float32)
