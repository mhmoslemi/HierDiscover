"""
Highly‑optimised outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1.  First LayerNorm in fp16 (weights are cast to fp16).
2.  Fuse *all* linear projections (left/right projections + 3 gates) into a
    single matmul:  x @ W_cat.T  where   W_cat ∈ ℝ^{5·H×D}.
    The result is sliced into   left_proj, right_proj,
    left_gate_pre, right_gate_pre, out_gate_pre.
3.  Apply mask (if present) to the two projection tensors,
    apply the sigmoid gates and fuse the gates into the projections:
        left  = left_proj  * sigmoid(left_gate_pre)
        right = right_proj * sigmoid(right_gate_pre)
4.  Heavy O(N³) contraction  out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
    is performed by a custom Triton GEMM kernel that computes a batched
    matmul C = A @ Bᵀ for each head h (A/B are [B·H,N,N] fp16,
    accumulation in fp32 → final fp16).  Tile sizes are tuned
    (64×64×64 for small N, 128×128×64 for N≥512) and the kernel
    runs with 8 or 16 warps depending on N.
5.  Second LayerNorm (fp16) on the hidden dimension.
6.  Multiply by the pre‑computed out‑gate and a final linear projection
    back to the model dimension.
7.  Return fp32 output (as required by the harness).

All heavy work is done in fp16 to minimise memory traffic; accumulation
is fp32 for numerical stability.  The only Triton‑implemented part
is the O(N³) contraction, which dominates runtime.
"""

from typing import Tuple, Dict, Any

# --------------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (A/B = [M,N,K] ; C = [M,N,N])
# --------------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # scalar sizes
    stride_am, stride_ai, stride_ak,   # stride for A (M,i,k)
    stride_bm, stride_bj, stride_bk,   # stride for B (M,j,k)
    stride_cm, stride_ci, stride_cj,   # stride for C (M,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile
    pid_z = tl.program_id(2)          # batch*head index

    # ----------------------------------------------------------------------
    # Tile offsets
    # ----------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (fp32 for stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------------------------------------------------------
    # Loop over K dimension
    # ----------------------------------------------------------------------
    for k_start in range(0, K, BLOCK_K):
        cur_k = k_start
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # --- load a tile of A ------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # --- load a tile of B (transposed) ----------------------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # mixed‑precision matrix multiply (fp16 × fp16 → fp32)
        acc += tl.dot(a, b)

    # ----------------------------------------------------------------------
    # Store result (fp16)
    # ----------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] with a
    Triton kernel.  All tensors must be fp16 and contiguous.

    Parameters
    ----------
    left, right : torch.Tensor
        Shape (B, N, N, H)   fp16.

    Returns
    -------
    out : torch.Tensor
        Shape (B, N, N, H)   fp16.
    """
    B, N, _, H = left.shape

    # reshape to [B*H, N, N] – each head becomes a separate batch entry
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # strides (in elements)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # ----------------------------------------------------------------------
    # Choose tile sizes according to sequence length
    # ----------------------------------------------------------------------
    if N >= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 16
    else:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 8

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=3,
    )

    # reshape back to [B, N, N, H]
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operator (no gradient).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D]  (float32)
        - mask_tensor  : torch.Tensor [B, N, N]  (float32 or bool)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, D] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (fp16)
    # -------------------------------------------------
    norm_weight_fp16 = weights["norm.weight"].to(torch.float16)
    norm_bias_fp16 = weights["norm.bias"].to(torch.float16)
    x = torch.nn.functional.layer_norm(
        input_tensor.to(torch.float16),
        (dim,),
        weight=norm_weight_fp16,
        bias=norm_bias_fp16,
    )  # [B, N, N, D]  fp16

    # -------------------------------------------------
    # 2️⃣ Fuse all linear projections + gates into ONE matmul
    # -------------------------------------------------
    # Concatenate the five weight matrices (out_features dimension first)
    left_proj_w   = weights["left_proj.weight"]   # (H, D)
    right_proj_w  = weights["right_proj.weight"]  # (H, D)
    left_gate_w   = weights["left_gate.weight"]   # (H, D)
    right_gate_w  = weights["right_gate.weight"]  # (H, D)
    out_gate_w    = weights["out_gate.weight"]    # (H, D)

    cat_weight = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(torch.float16)                             # (5·H, D)

    # Single linear pass
    combined = torch.nn.functional.linear(x, cat_weight)   # [B, N, N, 5·H]

    # Slice the combined tensor
    H = hidden_dim
    proj_left      = combined[..., :H]                 # (B,N,N,H)
    proj_right     = combined[..., H:2 * H]            # (B,N,N,H)
    left_gate_pre  = combined[..., 2 * H:3 * H]        # (B,N,N,H)
    right_gate_pre = combined[..., 3 * H:4 * H]        # (B,N,N,H)
    out_gate_pre   = combined[..., 4 * H:]             # (B,N,N,H)

    # Gate activations
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)

    # -------------------------------------------------
    # 3️⃣ Apply optional mask (broadcast on hidden dim)
    # -------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        proj_left  = proj_left * mask_fp16
        proj_right = proj_right * mask_fp16

    # -------------------------------------------------
    # 4️⃣ Fuse gate into projection (in‑place)
    # -------------------------------------------------
    left  = proj_left * left_gate    # (B,N,N,H)
    right = proj_right * right_gate  # (B,N,N,H)

    # -------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via Triton
    # -------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = _batched_matmul_transpose_fp16(left, right)  # (B,N,N,H) fp16

    # -------------------------------------------------
    # 6️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # (B,N,N,H) fp16

    # -------------------------------------------------
    # 7️⃣ Final gating + projection back to model dim
    # -------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (D, H)
    out = torch.nn.functional.linear(out, to_out_w)        # (B,N,N,D) fp16

    # -------------------------------------------------
    # 8️⃣ Cast to fp32 for the harness
    # -------------------------------------------------
    return out.to(torch.float32)
```
