"""
TriMul (outgoing) – fast forward pass

Algorithm
---------
1️⃣  Layer‑norm the input (FP32) → cast to FP16.
2️⃣  Fuse linear projections & gating:
    * One linear layer produces (left_proj, left_gate, out_gate) simultaneously.
    * One linear layer produces (right_proj, right_gate) simultaneously.
   This halves the number of dense‑matrix passes over the huge
   [B, N, N, dim] tensor, saving memory bandwidth.
3️⃣  Optional mask is broadcast on the hidden dimension and fused
   with the projections.
4️⃣  Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is implemented as a batched GEMM C = A @ Bᵀ.
   * For small sequences (N ≤ 256) a handcrafted Triton kernel
     (_trimul_matmul_kernel) performs the GEMM with FP16 inputs,
     FP32 accumulation and tensor‑core dot.
   * For larger N we fall back to `torch.bmm`, which calls cuBLAS
     (highly tuned for big matrices).
5️⃣  Second Layer‑Norm (FP16) → element‑wise out‑gate (sigmoid) →
   final linear projection back to `dim`.
6️⃣  Cast back to FP32 (required by the harness).

The implementation respects the original PyTorch semantics, supports
optional masking and works for the full range of test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : [M, N, K]   (M = B * hidden_dim, N = seq_len, K = seq_len)
#   C    : [M, N, N]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # i32 sizes
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix‑multiply kernel for the TriMul contraction."""
    pid_m = tl.program_id(0)   # tile along i (output row)
    pid_n = tl.program_id(1)   # tile along j (output col)
    pid_z = tl.program_id(2)   # batch * hidden index (M)

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (higher precision for the sum)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ----------------- load A tile (i, k) --------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # <-- fp16

        # ----------------- load B tile (j, k) as transposed -------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk   # k dimension first (row)
            + offs_j[None, :] * stride_bj   # j dimension second (col)
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # <-- fp16

        # mixed‑precision matrix multiply: fp16 × fp16 → fp32 accumulate
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Store the result (cast back to fp16)
    # --------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right (both [B, N, N, H]) using a custom Triton GEMM.
    Returns a tensor of shape [B, N, N, H] (FP16).
    """
    B, N, _, H = left.shape
    # reshape to [M, N, N] where M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A)  # FP16 output buffer

    # strides in element counts (not in bytes)
    stride_am, stride_ai, stride_ak = A.stride()   # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()  # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()   # (N*N, N, 1)

    # Tile size – 64×64×64 works well on an H100 for FP16 tensor cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    # reshape back to the original layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def _batched_matmul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right using torch.bmm (cuBLAS).
    Operates in FP16 with FP32 accumulation (default for torch.bmm).
    """
    B, N, _, H = left.shape
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)          # [M,N,N]
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)        # [M,N,N]
    C = torch.bmm(A, B_.transpose(1, 2))                                 # [M,N,N]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()              # [B,N,N,H]
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (bool / float)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack & config
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 2️⃣ first LayerNorm (float32) – numerical stability
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 3️⃣ cast to FP16 for the heavy section
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ fuse linear projections & gating (single matmuls)
    # ------------------------------------------------------------------
    # left side: left_proj, left_gate, out_gate  (3 * hidden_dim outputs)
    left_combined_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape (3*hidden_dim, dim)

    # right side: right_proj, right_gate  (2 * hidden_dim outputs)
    right_combined_w = torch.cat(
        [
            weights["right_proj.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape (2*hidden_dim, dim)

    # Linear passes (FP16)
    left_combined = F.linear(x, left_combined_w)   # [B,N,N,3*H]
    right_combined = F.linear(x, right_combined_w)  # [B,N,N,2*H]

    # Split the combined tensors
    left_proj = left_combined[..., :hidden_dim]                     # (B,N,N,H)
    left_gate_pre = left_combined[..., hidden_dim:2 * hidden_dim]  # (B,N,N,H)
    out_gate_pre = left_combined[..., 2 * hidden_dim:]             # (B,N,N,H)

    right_proj = right_combined[..., :hidden_dim]                 # (B,N,N,H)
    right_gate_pre = right_combined[..., hidden_dim:]             # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5️⃣ activations (sigmoid) – still FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # ------------------------------------------------------------------
    # 6️⃣ optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B,N,N] → broadcast to hidden dimension
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 7️⃣ apply gates (in‑place to keep memory usage low)
    # ------------------------------------------------------------------
    left = left_proj * left_gate   # [B,N,N,H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 8️⃣ heavy O(N³) contraction
    # ------------------------------------------------------------------
    seq_len = input_tensor.shape[1]
    # Use Triton for very small N, cuBLAS (torch.bmm) otherwise.
    if seq_len <= 256:
        out = _batched_matmul_triton(left, right)   # FP16
    else:
        out = _batched_matmul_torch(left, right)    # FP16

    # ------------------------------------------------------------------
    # 9️⃣ second LayerNorm (FP16) → out‑gate → final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)  # [B,N,N,dim]  FP16

    # ------------------------------------------------------------------
    # 🔟 cast back to FP32 (required by the test harness)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
