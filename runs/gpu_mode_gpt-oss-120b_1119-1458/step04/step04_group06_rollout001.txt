"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

Algorithm
---------
1️⃣ Layer‑norm the input tensor (float32 for numeric stability).
2️⃣ Cast to fp16 – all heavy work runs in fp16 (Tensor‑core friendly).
3️⃣ Apply four linear projections (left, right, left_gate, right_gate,
   out_gate) and the final projection (to_out).  All weights are cast to fp16.
4️⃣ Optional mask (broadcasted on the hidden dimension) is applied to
   left/right before gating.
5️⃣ Fuse the sigmoid gates into the projections in‑place.
6️⃣ Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM C = A @ Bᵀ where A and B are reshaped to
   [B·H, N, N] (H = hidden_dim).  A custom Triton kernel performs the
   multiplication with fp16 operands and fp32 accumulation.
   For very short sequences (N ≤ 128) we fall back to `torch.einsum`
   to avoid kernel launch overhead.
7️⃣ Second LayerNorm (fp16) → out‑gate (sigmoid) → final linear projection
   back to the original `dim`.
8️⃣ Cast the result back to float32 (the harness expects FP32).

Only the O(N³) contraction is implemented in Triton; the remaining steps
use highly optimized PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : [M, N, K]   (M = B * H, N = seq_len, K = seq_len)
#   C    : [M, N, N]
#   Operands are fp16, accumulator is fp32.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # program‑size constants
    stride_am, stride_ai, stride_ak,   # A strides (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B strides (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C strides (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM with B transposed (A @ Bᵀ)."""
    pid_m = tl.program_id(0)   # tile on output i dimension
    pid_n = tl.program_id(1)   # tile on output j dimension
    pid_z = tl.program_id(2)   # batch * hidden index (M)

    # Tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile (M, i, k)  -> shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # fp16

        # Load B tile (M, j, k) – note the order (k, j) implements Bᵀ
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # fp16

        # Mixed‑precision dot: fp16 × fp16 → fp32 acc
        acc += tl.dot(a, b)

    # Store result tile (fp16)
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )

# ----------------------------------------------------------------------
# Helper: batched GEMM C = A @ Bᵀ for the TriMul contraction.
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] using a Triton kernel.

    Arguments
    ---------
    left, right : torch.Tensor  [B, N, N, H] (fp16, contiguous)

    Returns
    -------
    out : torch.Tensor [B, N, N, H] (fp16, contiguous)
    """
    B, N, _, H = left.shape

    # Pack (batch, hidden) into a single leading dimension: [B·H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (fp16)
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in element counts (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()     # (N*N, N, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()    # (N*N, N, 1)
    stride_cm, stride_ci, stride_cj = C.stride()    # (N*N, N, 1)

    # Tile size heuristic
    if N <= 128:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch × hidden
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # Restore original layout [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (bool or float32)
        - weights_dict : name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32, same device as input).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B,N,N,dim]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the rest of the pipeline
    # -----------------------------------------------------------------
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Cast model weights to fp16 on the target device
    # -----------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(device=device, dtype=torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(device=device, dtype=torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)
    to_out_w      = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # -----------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)    # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)   # [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast on hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] -> [B,N,N,1] for broadcasting
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # -----------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – fp16
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Fuse gates in‑place (decreases memory traffic)
    left.mul_(left_gate)
    right.mul_(right_gate)

    # -----------------------------------------------------------------
    # 7️⃣ Heavy O(N³) contraction via Triton (fp16 in, fp16 out)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    if N <= 128:
        # For very short sequences the Triton launch overhead dominates.
        out = torch.einsum('b i k h, b j k h -> b i j h', left, right)
    else:
        out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hidden] fp16

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm (fp16)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Final gating + projection back to original dim
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # -----------------------------------------------------------------
    # 🔟 Cast back to float32 (required by the harness)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
