```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for batch*hidden matmul (left @ right.T)
@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_ir, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_mat = tl.program_id(2)

    # Compute batch and hidden indices
    b = pid_mat // H
    h = pid_mat % H

    # Offsets for current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this (b, h) slice
    left_base = left_ptr + b * stride_bl + h * stride_hl
    right_base = right_ptr + b * stride_br + h * stride_hr
    out_base = out_ptr + b * stride_bo + h * stride_ho

    # Accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_blocks):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Left tile (M x K)
        left_ptrs = left_base + offs_i[:, None] * stride_il + offs_k[None, :] * stride_kl
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # Right tile (K x N) -> we want rows=k, cols=j
        right_ptrs = right_base + offs_k[:, None] * stride_kr + offs_j[None, :] * stride_ir
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # Multiply-accumulate
        acc += tl.dot(left, right)   # (M, K) @ (K, N) -> (M, N)

    # Store result
    out_ptrs = out_base + offs_i[:, None] * stride_io + offs_j[None, :] * stride_jo
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

def batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out = left @ right^T for each (batch, hidden) slice using Triton.
    left, right shape: [B, N, N, H] (torch.float16, contiguous)
    Returns tensor of same shape and dtype.
    """
    assert left.shape == right.shape, "Shapes mismatch"
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=left.dtype, device=left.device)
    # strides
    stride_bl, stride_il, stride_kl, stride_hl = left.stride()
    stride_br, stride_ir, stride_kr, stride_hr = right.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out.stride()
    # block sizes (tuned for H100 FP16)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N), B * H)
    tri_mul_matmul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_bl, stride_il, stride_kl, stride_hl,
        stride_br, stride_ir, stride_kr, stride_hr,
        stride_bo, stride_io, stride_jo, stride_ho,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out

def custom_kernel(
    data: tuple[torch.Tensor, torch.Tensor, dict, dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul layer.
    Implements layernorm, linear projections and gating in FP16,
    the O(N³) contraction using a fused Triton batched GEMM kernel,
    then a second LayerNorm, output gating and final projection.
    Returns a float32 tensor of shape [B, N, N, dim].
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # 1. First LayerNorm (float32)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Cast to FP16 for the rest
    x = x_norm.to(torch.float16)

    # Cast weights to FP16 once
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # 2. Linear projections (B, N, N, hidden)
    left_proj = F.linear(x, left_proj_w)   # shape [B,N,N,hidden]
    right_proj = F.linear(x, right_proj_w)

    # 3. Optional mask (broadcast over hidden)
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # 4. Gating
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # 5. Apply gates
    left = (left_proj * left_gate).contiguous()
    right = (right_proj * right_gate).contiguous()

    # 6. Heavy O(N³) contraction via Triton
    # out shape: [B, N, N, hidden] (fp16)
    out = batched_matmul_triton(left, right)

    # 7. Second LayerNorm over hidden dim
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # 8. Output gating and final projection
    out = out * out_gate
    out = F.linear(out, to_out_w)   # back to dim

    # 9. Cast back to float32
    return out.to(torch.float32)

"""
TriMul (outgoing) – highly‑optimised forward pass.

*   A first LayerNorm normalises the pairwise input `x` over the last
    dimension (`dim`).
*   Linear projections, gates and the optional binary mask are computed
    in FP16 for speed.
*   The dominant O(N³) contraction

        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    is performed by a custom Triton kernel.  The kernel treats the
    product of the batch and hidden dimensions (B·H) as an independent
    batch index and executes a tiled GEMM (left @ rightᵀ) with FP16
    arithmetic and an FP32 accumulator.  This avoids the costly
    reshape‐and‑copy needed by the naïve PyTorch implementation.
*   A second LayerNorm (over the hidden dimension), an output‑gate and a
    final projection back to the original feature dimension complete the
    module.  The final result is returned as float‑32.

The Triton kernel follows the classic block‑wise matmul pattern with
3‑D grid (i‑tile, j‑tile, batch·hidden) and is tuned for H100
Tensor‑Core performance (BLOCK_M=N_BLOCK, BLOCK_N=N_BLOCK,
BLOCK_K=32, 8 warps).
"""

import math
from typing import Tuple, Dict, Any

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched GEMM for the TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # scalar sizes
    stride_bl, stride_il, stride_kl, stride_hl,  # left  strides (b,i,k,h)
    stride_br, stride_ir, stride_kr, stride_hr,  # right strides (b,i,k,h)
    stride_bo, stride_io, stride_jo, stride_ho,  # out   strides (b,i,j,h)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched (B*H) matrix multiplication C = A @ Bᵀ with A,B shape (N,N)."""
    pid_i = tl.program_id(0)          # block row index   (i)
    pid_j = tl.program_id(1)          # block column index (j)
    pid_mat = tl.program_id(2)        # batch*hidden index

    # Decode batch and hidden indices from pid_mat
    b = pid_mat // H
    h = pid_mat % H

    # Offsets for the current output tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this (b,h) slice
    left_base  = left_ptr  + b * stride_bl + h * stride_hl
    right_base = right_ptr + b * stride_br + h * stride_hr
    out_base   = out_ptr   + b * stride_bo + h * stride_ho

    # Accumulator in FP32 to preserve numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension (summation axis)
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_blocks):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- left tile: shape (BLOCK_M, BLOCK_K) ----
        left_ptrs = left_base + offs_i[:, None] * stride_il + offs_k[None, :] * stride_kl
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )  # FP16

        # ---- right tile: we need (K, BLOCK_N) = rows=k, cols=j ----
        right_ptrs = right_base + offs_k[:, None] * stride_kr + offs_j[None, :] * stride_ir
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )  # FP16

        # Matrix multiply‑accumulate
        acc += tl.dot(left, right)  # (M,K) @ (K,N) -> (M,N)

    # Write the result (cast back to FP16)
    out_ptrs = out_base + offs_i[:, None] * stride_io + offs_j[None, :] * stride_jo
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute batch‑wise matrix products:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Parameters
    ----------
    left, right : torch.Tensor
        Shape [B, N, N, H], dtype torch.float16, contiguous.

    Returns
    -------
    torch.Tensor
        Tensor of the same shape and dtype as the inputs.
    """
    assert left.shape == right.shape, "left and right must have identical shapes"
    B, N, _, H = left.shape

    out = torch.empty_like(left, dtype=left.dtype, device=left.device)

    # stride extraction (int64)
    stride_bl, stride_il, stride_kl, stride_hl = left.stride()
    stride_br, stride_ir, stride_kr, stride_hr = right.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out.stride()

    # Tuned tile sizes for H100 FP16 Tensor‑Core performance
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        tl.cdiv(N, BLOCK_M),      # program_id(0): i‑tiles
        tl.cdiv(N, BLOCK_N),      # program_id(1): j‑tiles
        B * H,                    # program_id(2): (batch·hidden) matrices
    )

    _tri_mul_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_bl,
        stride_il,
        stride_kl,
        stride_hl,
        stride_br,
        stride_ir,
        stride_kr,
        stride_hr,
        stride_bo,
        stride_io,
        stride_jo,
        stride_ho,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,      # 8 warps = 256 threads per block, good for H100
    )
    return out


# ----------------------------------------------------------------------
# Public entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # Cast to FP16 for the heavy lifting
    x = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Cast model weights to FP16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)      # [B, N, N, hidden_dim]
    right_proj = F.linear(x, right_proj_w)     # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1]
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left_proj  = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – still in FP16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # [B, N, N, hidden_dim]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # [B, N, N, hidden_dim]

    # Apply hidden‑dim gates
    left  = left_proj * left_gate      # [B, N, N, hidden_dim]
    right = right_proj * right_gate    # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction – Triton batched GEMM (C = A @ Bᵀ)
    # ------------------------------------------------------------------
    # left/right are already FP16; we pass them directly to the Triton kernel.
    out = _batched_matmul_triton(left, right)   # [B, N, N, hidden_dim] (FP16)

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension (FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Output gating + final projection back to original dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_w)   # [B, N, N, dim]  FP16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 for the final output
    # ------------------------------------------------------------------
    return out.to(torch.float32)
