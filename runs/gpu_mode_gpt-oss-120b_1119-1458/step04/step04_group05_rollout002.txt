"""
TriMul (outgoing) – forward only, highly‑optimized
=================================================

The outgoing TriMul computes for each batch and hidden channel the
contraction

    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

where ``left`` and ``right`` are linear projections of the input tensor
followed by elementwise sigmoid gates and an optional mask.  The O(N³)
contraction is performed with a custom Triton kernel that batches the
B·H hidden maps as independent GEMMs (A @ Bᵀ).  All other operations are
executed in FP16 and the final result is returned in FP32.

Key speed‑ups compared to the naïve PyTorch reference:
* The four linear layers (proj + gate for left/right) are fused into a single
  half‑precision matmul.
* The mask is fused into the projected tensors before gating.
* A custom Triton batched matrix‑multiply replaces the expensive three‑nested
  loops.  For N>512 we fall back to cuBLAS‑backed ``torch.bmm`` which is
  already optimal on H100.
* All heavy tensors stay in FP16; only the cheap LayerNorms and the final
  output are cast back to FP32.

The kernel works for any batch size (B=1‥2), sequence length N up to 1024,
model dimension ``dim`` up to 384 and hidden dimension ``hidden_dim``
(up to 128 in the test suite).  It also correctly handles the presence or
absence of a mask.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A : (M, N, K)   – M = batch * hidden, N = i‑axis, K = k‑axis
#   B : (M, K, N)   – already transposed (k, j)
#   C : (M, N, N)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bk, stride_bj,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM A @ Bᵀ with FP16 inputs, FP32 accumulation."""
    pid_m = tl.program_id(0)                # tile along output i‑axis
    pid_n = tl.program_id(1)                # tile along output j‑axis
    pid_z = tl.program_id(2)                # batch * hidden index

    # i‑ and j‑coordinates of the current tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for numerical safety
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K in tiles of BLOCK_K
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # -------- load A[i,k] --------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # FP16

        # -------- load Bᵀ[k,j] (i.e. B[k,j]) --------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # FP16

        # dot product (fp16*fp16 → fp32) and accumulate
        acc += tl.dot(a, b)

    # store result as FP16
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_triton(
    left: torch.Tensor,
    right: torch.Tensor,
    tile: int = 128,
    k_tile: int = 64,
) -> torch.Tensor:
    """
    Wrapper for the Triton kernel above.
    `left` and `right` are FP16 tensors of shape [B, N, N, H] (no padding
    required when N is a multiple of `tile`; otherwise they are padded
    internally to avoid bounds checks).  The kernel returns a tensor of the
    same logical shape (still FP16).
    """
    B, N, _, H = left.shape

    # ------------------------------------------------------------------
    # Pad N to a multiple of `tile` to keep the kernel simple
    # ------------------------------------------------------------------
    pad_len = (tile - (N % tile)) % tile
    if pad_len != 0:
        N_pad = N + pad_len
        left_pad = left.new_zeros((B, N_pad, N_pad, H), dtype=torch.float16)
        right_pad = right.new_zeros((B, N_pad, N_pad, H), dtype=torch.float16)
        left_pad[:, :N, :N, :] = left
        right_pad[:, :N, :N, :] = right
    else:
        left_pad = left
        right_pad = right
        N_pad = N

    # --------------------------------------------------------------
    #   Pack (batch, hidden) into a single leading dimension
    #   and transpose `right` so that we compute A @ Bᵀ.
    # --------------------------------------------------------------
    # left : (B, N, N, H) → (B, H, N, N) → (M, N, K)
    left_perm = left_pad.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    # right : (B, N, N, H) → (B, H, K, J) where K=j‑axis, J=i‑axis?
    #   we need (B, H, K, J) = (B, H, N, N) with K being the summed axis.
    right_perm = right_pad.permute(0, 3, 2, 1).contiguous()       # (B, H, N, N) == (B, H, K, J)

    A = left_perm.view(B * H, N_pad, N_pad)       # (M, N, K)
    B_ = right_perm.view(B * H, N_pad, N_pad)     # (M, K, N)

    C = torch.empty_like(A)                       # FP16 output

    # strides in element units (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bk, stride_bj = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    grid = (
        (N_pad + tile - 1) // tile,   # pid_m
        (N_pad + tile - 1) // tile,   # pid_n
        B * H,                        # pid_z
    )
    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N_pad,
        N_pad,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bk,
        stride_bj,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=tile,
        BLOCK_N=tile,
        BLOCK_K=k_tile,
        num_warps=8,
        num_stages=3,
    )

    # Reshape back to [B, N, N, H] and trim padding if any
    C = C.view(B, H, N_pad, N_pad).permute(0, 2, 3, 1).contiguous()
    if pad_len != 0:
        C = C[:, :N, :N, :]
    return C


def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input tensor  [B, N, N, dim] (float32)
        torch.Tensor,                     # mask tensor   [B, N, N]   (float32/bool) or None
        Dict[str, torch.Tensor],          # model weights
        Dict[str, Any],                   # config (must contain dim, hidden_dim, optional nomask)
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (gradient‑free).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # 1️⃣ Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2️⃣ First LayerNorm (float32 → FP16)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # LayerNorm in FP32 for numerical stability, then cast to FP16
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    ).to(torch.float16)  # shape [B, N, N, dim] in half precision

    # --------------------------------------------------------------
    # 3️⃣ Cast model weights to FP16 (once)
    # --------------------------------------------------------------
    # Fuse the four linear layers (left_proj, left_gate, right_proj, right_gate)
    left_proj_w = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]        # (hidden_dim, dim)
    left_gate_w = weights["left_gate.weight"]          # (hidden_dim, dim)
    right_gate_w = weights["right_gate.weight"]        # (hidden_dim, dim)

    # Order: [left_proj, left_gate, right_proj, right_gate]
    combined_proj_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    ).to(torch.float16)  # (4*hidden_dim, dim)

    out_gate_w = weights["out_gate.weight"].to(torch.float16)          # (hidden_dim, dim)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)   # (hidden_dim,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)     # (hidden_dim,)

    to_out_w = weights["to_out.weight"].to(torch.float16)            # (dim, hidden_dim)

    # --------------------------------------------------------------
    # 4️⃣ Combined projection + gating
    # --------------------------------------------------------------
    B, N, _, _ = x.shape
    flat_x = x.view(-1, dim)                                        # (B·N·N, dim)

    # One fused matmul produces [left_proj, left_gate, right_proj, right_gate]
    combined = torch.nn.functional.linear(flat_x, combined_proj_w)   # (B·N·N, 4*hidden)
    combined = combined.view(B, N, N, 4 * hidden_dim)

    left_proj   = combined[..., 0:hidden_dim]                       # (B, N, N, H)
    left_gate_p = combined[..., hidden_dim:2 * hidden_dim]         # (B, N, N, H)
    right_proj  = combined[..., 2 * hidden_dim:3 * hidden_dim]     # (B, N, N, H)
    right_gate_p = combined[..., 3 * hidden_dim:4 * hidden_dim]   # (B, N, N, H)

    left_gate  = torch.sigmoid(left_gate_p)
    right_gate = torch.sigmoid(right_gate_p)

    if not nomask:
        # mask is broadcast over the hidden dimension
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)     # (B, N, N, 1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # Apply gates
    left  = left_proj * left_gate
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Output gate (computed from the same normalized input)
    # --------------------------------------------------------------
    out_gate_pre = torch.nn.functional.linear(flat_x, out_gate_w)   # (B·N·N, hidden)
    out_gate = torch.sigmoid(out_gate_pre).view(B, N, N, hidden_dim)  # (B, N, N, H)

    # --------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction (batched matmul)
    # --------------------------------------------------------------
    if N <= 512:
        # Use Triton for modest N (already includes a small amount of padding)
        out = _batched_matmul_triton(left, right, tile=128, k_tile=64)
    else:
        # For larger N cuBLAS‑backed batched GEMM is faster
        left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # (M, i, k)
        right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)  # (M, k, j)
        out_mat = torch.bmm(left_mat, right_mat)                                      # (M, i, j)
        out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()      # (B, i, j, H)

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm, gating, final projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # (B, N, N, dim) in FP16

    # --------------------------------------------------------------
    # 8️⃣ Cast back to FP32 (the API expects float32)
    # --------------------------------------------------------------
    return out.to(torch.float32)
