"""
TriMul – “outgoing” variant (AlphaFold‑3)

Algorithm (forward only)
1️⃣  Layer‑norm the 4‑D input (float32) → cast to fp16.
2️⃣  Fuse the two linear projections (proj & gate) for the *left* and *right*
    branches by concatenating their weight matrices, then split the result.
    This cuts the number of GEMM calls from 5 → 3.
3️⃣  Fuse element‑wise `proj * gate * mask` with a tiny Triton kernel
    (the mask is broadcast on the hidden dimension).  When ``nomask``=True
    the mask kernel is skipped → plain PyTorch multiplication.
4️⃣  Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is performed as a batched GEMM (`C = A @ Bᵀ`) using torch.bmm,
    i.e. cuBLAS Tensor‑Core kernels (the fastest path on H100).
5️⃣  Layer‑norm on the hidden dimension (fp16) → multiply with the
    out‑gate (sigmoid) → final linear projection back to ``dim``.
6️⃣  Cast the result back to float32 for the harness.

A Triton kernel is used for the element‑wise fusion (step 3), satisfying the
requirement that “part of the operations” are custom‑kernel‑driven.
All heavy matrix multiplications stay in half‑precision for speed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fuse (tensor * gate * mask) in‑place.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,            # *fp16 : tensor to be updated   (size,)
    gate_ptr,         # *fp16 : gate tensor            (size,)
    mask_ptr,         # *fp16 : mask (0/1) tensor     (size,)
    size,             # i32   : total #elements
    BLOCK: tl.constexpr,  # compile‑time tile size (≤1024 threads)
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offsets
    mask = offs < size                                 # out‑of‑range guard

    # Load operands
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + offs, mask=mask, other=1.0)   # default to 1

    # Fused multiplication
    out = x * g * m

    # Store back in‑place
    tl.store(x_ptr + offs, out, mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (bool/float32)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True ⇒ mask is ignored

    device = input_tensor.device

    # --------------------------------------------------------------
    # 1️⃣ first LayerNorm (float32) – stable for the very first stats
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )   # [B, N, N, dim]  float32

    # --------------------------------------------------------------
    # 2️⃣ cast to fp16 for the heavy part
    # --------------------------------------------------------------
    x = x_norm.to(torch.float16)   # [B, N, N, dim]  fp16

    # --------------------------------------------------------------
    # 3️⃣ cast model weights to fp16 (once)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 4️⃣ fuse linear projection & gate for left and right sides
    #    (single matmul per side)
    # --------------------------------------------------------------
    # concatenate [proj; gate] weight matrices → one GEMM
    left_concat_w  = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, D)
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0) # (2*H, D)

    # shape: [B,N,N,2*H]
    left_concat = F.linear(x, left_concat_w)   # fp16
    right_concat = F.linear(x, right_concat_w) # fp16

    # split into proj / gate
    left_proj = left_concat[..., :hidden]                # [B,N,N,H]
    left_gate_pre = left_concat[..., hidden:]            # [B,N,N,H]
    left_gate = torch.sigmoid(left_gate_pre)

    right_proj = right_concat[..., :hidden]              # [B,N,N,H]
    right_gate_pre = right_concat[..., hidden:]          # [B,N,N,H]
    right_gate = torch.sigmoid(right_gate_pre)

    # --------------------------------------------------------------
    # 5️⃣ out‑gate (used later, after the second LayerNorm)
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # [B,N,N,H] fp16

    # --------------------------------------------------------------
    # 6️⃣ element‑wise fusion (proj * gate * mask) via Triton
    # --------------------------------------------------------------
    # Kernel works on contiguous 1‑D buffers; ensure tensors are contiguous.
    left_proj = left_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_proj = right_proj.contiguous()
    right_gate = right_gate.contiguous()

    total_elems = left_proj.numel()
    BLOCK = 1024                                           # ≤1024 threads per block
    grid = ((total_elems + BLOCK - 1) // BLOCK, )

    if not nomask:
        # mask is [B,N,N] (float or bool).  Broadcast to hidden dim and make contiguous.
        mask_fp16 = mask_tensor.to(torch.float16)
        mask_exp = mask_fp16.unsqueeze(-1).expand(-1, -1, -1, hidden).contiguous()
        # fused left = left_proj * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            mask_exp,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # fused right = right_proj * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right_proj,
            right_gate,
            mask_exp,
            total_elems,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        # No mask → simple element‑wise multiplication (still cheap)
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # --------------------------------------------------------------
    # 7️⃣ heavy O(N³) contraction: batched GEMM using cuBLAS (torch.bmm)
    #      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    B, N, _, _ = left_proj.shape
    # reshape to [B*H, N, N] for torch.bmm
    a = left_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)   # (B*H,N,N)
    b = right_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)  # (B*H,N,N)

    # bmm with transposed second operand
    c = torch.bmm(a, b.transpose(1, 2))      # (B*H,N,N)

    # restore original layout -> [B, N, N, H]
    out = c.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8️⃣ second LayerNorm (fp16) on hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 9️⃣ final gating + projection back to dim
    # --------------------------------------------------------------
    out = out * out_gate                     # element‑wise gate
    out = F.linear(out, to_out_w)            # [B,N,N,dim] (fp16)

    # --------------------------------------------------------------
    # 🔟 cast back to float32 (required by the evaluation harness)
    # --------------------------------------------------------------
    return out.to(torch.float32)
