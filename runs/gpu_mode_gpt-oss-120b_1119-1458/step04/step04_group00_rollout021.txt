def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    a_ptrs = a_ptr + batch_idx * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    b_ptrs = b_ptr + batch_idx * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # accumulate in fp32 for precision

    for k in range(0, K, BLOCK_K):
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)
        acc += tl.dot(a, b, out_dtype=tl.float32)  # dot does fp16 multiply add returns fp32 typically
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k
    # Convert back to fp16 for storing
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + batch_idx * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, c, mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))
```

```
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernel for fused mask & gate (optional): could be omitted if we do in Pytorch.

# Triton kernel for batched GEMM (TriMul contraction)
@triton.jit
def trimul_outgoing_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr, num_stages: tl.constexpr
):
    batch_idx = tl.program_id(0)   # 0 … B*H-1
    pid_m = tl.program_id(1)      # block row
    pid_n = tl.program_id(2)      # block col

    # Recover (batch, hidden) from flat index
    b = batch_idx // H
    h = batch_idx % H

    # Compute base addresses for this (b,h) slice
    left_base = left_ptr + b * stride_l_batch + h * stride_l_h
    right_base = right_ptr + b * stride_r_batch + h * stride_r_h
    out_base = out_ptr + b * stride_o_batch + h * stride_o_h

    # Tile coordinates
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)    # rows i
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)    # cols j

    # Accumulator in FP32 for numerical safety
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension k
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k,
            mask=(offs_i[:, None] < N) & (offs_k[None, :] < N),
            other=0.0,
        )

        # Load right tile (already transposed): shape (BLOCK_K, BLOCK_N)
        b_tile = tl.load(
            right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i,
            mask=(offs_k[:, None] < N) & (offs_j[None, :] < N),
            other=0.0,
        )

        acc += tl.dot(a, b_tile)

    # Cast back to FP16 and store
    c = acc.to(tl.float16)
    tl.store(
        out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j,
        c,
        mask=(offs_i[:, None] < N) & (offs_j[None, :] < N),
    )
```

```
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    This implementation follows the AlphaFold3 ``TriMul`` forward
    but moves the heavy O(N³) contraction to a specialised Triton
    kernel that performs a batched GEMM directly on the original
    (B, N, N, hidden) layout, thus avoiding costly permute &
    copy operations. The linear projections and gating are
    performed in FP16 and the second LayerNorm + output projection
    are applied with standard PyTorch ops.
    """
    # Unpack arguments
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------- 0. First LayerNorm (FP16) --------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    ).to(torch.float16)   # [B, N, N, dim] FP16

    B, N, _, _ = x.shape

    # -------------------- 1. Cast linear weights to FP16 once -----------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------- 2. Linear projections & gates (FP16) ---------
    # flatten only the leading three dimensions for faster linear
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)

    left_proj = torch.nn.functional.linear(x_flat, left_proj_w)     # (M, hidden)
    right_proj = torch.nn.functional.linear(x_flat, right_proj_w)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, right_gate_w))

    # output gate is needed later (shape B,N,N,hidden)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, out_gate_w))
    out_gate = out_gate.view(B, N, N, hidden_dim)   # (B,N,N,hidden)

    # reshape intermediate tensors back to (B,N,N,hidden)
    left_proj = left_proj.view(B, N, N, hidden_dim)
    right_proj = right_proj.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)
    right_gate = right_gate.view(B, N, N, hidden_dim)

    # -------------------- 3. Apply optional mask and element‑wise gates
    if not nomask:
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    left = left_proj * left_gate
    right = right_proj * right_gate

    # -------------------- 4. O(N³) contraction via custom Triton kernel
    # output tensor (will be overwritten)
    out = torch.empty_like(left)

    # Strides of the tensors (in element counts, not bytes)
    s_l_b, s_l_i, s_l_k, s_l_h = left.stride()
    s_r_b, s_r_i, s_r_k, s_r_h = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out.stride()

    # Triton launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (B * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    trimul_outgoing_gemm_kernel[grid](
        left, right, out,
        B, hidden_dim, N,
        s_l_b, s_l_i, s_l_k, s_l_h,
        s_r_b, s_r_i, s_r_k, s_r_h,
        s_o_b, s_o_i, s_o_j, s_o_h,

"""
TriMul (outgoing) – fast Triton‑accelerated forward pass
========================================================

The implementation follows the reference PyTorch module but moves the
dominant O(N³) contraction to a custom Triton kernel.  The kernel works
directly on the original layout ``[B, N, N, hidden]`` and therefore
avoids the costly ``permute(...).contiguous()`` copies that a naïve
approach would need.  All linear layers and LayerNorms are performed in
FP16 while the final result is returned in FP32 as required by the
caller.

Key ideas
---------

* **First LayerNorm** – standard PyTorch (FP16) layer‑norm.
* **Projections & gates** – 5 linear layers + sigmoid gates, all in
  FP16.  The optional mask is fused by a simple element‑wise multiply.
* **Contraction** – computes
  ``out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`` via a
  batched GEMM Triton kernel.  The kernel iterates over the reduction
  dimension ``k`` and accumulates in FP32 to keep numerical stability.
* **Second LayerNorm, output gate and final projection** – back to
  PyTorch, still in FP16.
* **Return** – cast back to FP32.

The kernel uses a 3‑D grid: (batch·hidden, tile‑row, tile‑col).  Block
sizes are tuned for H100 Tensor‑Core pipelines (128×128 tiles with a
K‑tile of 32).  No additional libraries are required beyond PyTorch and
Triton 3.3.1.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the O(N³) contraction
#   out = left @ rightᵀ   where left/right have shape [B, N, N, H]
#   The kernel fuses the batch (B) and hidden (H) dimensions into a
#   single program_id (0).  No permutes are required – strides are used
#   directly.
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_gemm_kernel(
    left_ptr, right_ptr, out_ptr,                # tensor pointers
    B, H, N,                                     # problem size (compile‑time)
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,   # strides of left
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,   # strides of right
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,   # strides of out
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,  # tile sizes
    num_warps: tl.constexpr, num_stages: tl.constexpr
):
    # 0‑dim: flattened (batch, hidden)
    batch_idx = tl.program_id(0)
    pid_m    = tl.program_id(1)   # block row index
    pid_n    = tl.program_id(2)   # block col index

    # Recover batch and hidden indices from the flattened id
    bh = batch_idx // H           # batch index  (0 … B‑1)
    hh = batch_idx % H            # hidden index (0 … H‑1)

    # Base pointers for this (batch, hidden) slice
    left_base  = left_ptr  + bh * stride_l_batch  + hh * stride_l_h
    right_base = right_ptr + bh * stride_r_batch  + hh * stride_r_h
    out_base   = out_ptr   + bh * stride_o_batch  + hh * stride_o_h

    # Offsets for the current output tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # rows i
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # cols j

    # Accumulator (FP32 for accuracy)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)                # reduction axis

        # Load left tile   (i, k)
        a = tl.load(
            left_base
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k,
            mask=(offs_i[:, None] < N) & (offs_k[None, :] < N),
            other=0.0,
        )
        # Load right tile (k, j) – note that we read it already transposed
        b = tl.load(
            right_base
            + offs_k[:, None] * stride_r_k
            + offs_j[None, :] * stride_r_i,
            mask=(offs_k[:, None] < N) & (offs_j[None, :] < N),
            other=0.0,
        )
        # Multiply‑accumulate
        acc += tl.dot(a, b)          # FP16·FP16 → FP32 accumulation

    # Store the result (convert back to FP16)
    out = acc.to(tl.float16)
    tl.store(
        out_base
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j,
        out,
        mask=(offs_i[:, None] < N) & (offs_j[None, :] < N),
    )


# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple (input, mask, weights, config)
        input   – torch.Tensor of shape [B, N, N, dim] (float32)
        mask    – torch.Tensor of shape [B, N, N] (float32 or bool)
        weights – dict of model weights (float32)
        config  – dict containing ``dim`` and ``hidden_dim``.
                  Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device as *input*.
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    ).to(torch.float16)                         # [B, N, N, dim] (FP16)

    # ------------------------------------------------------------------
    # 2️⃣ Cast all linear weights to FP16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections & sigmoid gates (FP16)
    # ------------------------------------------------------------------
    x_flat = x.view(-1, dim)                # (B·N·N, dim)

    left_proj = torch.nn.functional.linear(x_flat, left_proj_w)    # (M, H)
    right_proj = torch.nn.functional.linear(x_flat, right_proj_w)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, right_gate_w))

    out_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, out_gate_w))
    out_gate = out_gate.view(B, N, N, hidden_dim)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣ Reshape back to 4‑D tensors
    # ------------------------------------------------------------------
    left_proj = left_proj.view(B, N, N, hidden_dim)
    right_proj = right_proj.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)
    right_gate = right_gate.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask + element‑wise fusion
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B, N, N]; broadcast over hidden dimension
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    left = left_proj * left_gate          # (B, N, N, H)
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ O(N³) contraction via custom Triton kernel
    # ------------------------------------------------------------------
    out = torch.empty_like(left)          # allocate output (FP16)

    # Strides (in element counts) – required by Triton
    s_l_b, s_l_i, s_l_k, s_l_h = left.stride()
    s_r_b, s_r_i, s_r_k, s_r_h = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out.stride()

    # Kernel launch configuration (tuned for H100)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                         # flattened (batch, hidden)
        (N + BLOCK_M - 1) // BLOCK_M,           # tile‑rows
        (N + BLOCK_N - 1) // BLOCK_N,           # tile‑cols
    )

    trimul_outgoing_gemm_kernel[grid](
        left, right, out,
        B, hidden_dim, N,
        s_l_b, s_l_i, s_l_k, s_l_h,
        s_r_b, s_r_i, s_r_k, s_r_h,
        s_o_b, s_o_i, s_o_j, s_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Output gate + final linear projection (FP16)
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # -> (B,N,N,dim) FP16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to FP32 (as required by the caller)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
