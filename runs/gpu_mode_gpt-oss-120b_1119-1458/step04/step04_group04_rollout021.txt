import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def mask_gate_fuse_kernel(
    x_ptr, gate_ptr, mask_ptr,
    size,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + offs, mask=mask, other=1.0)
    out = x * g * m
    tl.store(x_ptr + offs, out, mask=mask)

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Outgoing TriMul forward pass (no autograd).
    The implementation fuses all linear projections + gating into a single
    matmul, applies an optional mask with a tiny Triton kernel, and carries
    out the O(N³) contraction via a single strided batched GEMM (torch.bmm).
    The computation is performed largely in fp16 and the result is cast back
    to fp32.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------
    # 1. LayerNorm on the input (fp16)
    # ------------------------------------------------------------
    norm_w = weights["norm.weight"].to(torch.float16, device=device)
    norm_b = weights["norm.bias"].to(torch.float16, device=device)
    # cast to fp16 before LN to keep everything half‑precision
    x = input_tensor.to(torch.float16)
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------
    # 2. Fuse the five linear projections (left, right, left_gate,
    #    right_gate, out_gate) in a single GEMM.
    # ------------------------------------------------------------
    # concatenate the five weight matrices along the output dimension
    w_left   = weights["left_proj.weight"].to(torch.float16, device=device)
    w_right  = weights["right_proj.weight"].to(torch.float16, device=device)
    w_lgate  = weights["left_gate.weight"].to(torch.float16, device=device)
    w_rgate  = weights["right_gate.weight"].to(torch.float16, device=device)
    w_outgate= weights["out_gate.weight"].to(torch.float16, device=device)

    combined_weight = torch.cat(
        [w_left, w_right, w_lgate, w_rgate, w_outgate],
        dim=0,
    )  # shape (5*hidden_dim, dim)

    # combined linear
    y = torch.nn.functional.linear(x, combined_weight)  # [B, N, N, 5*hidden]
    B, N, _, _ = y.shape

    # split into the 5 tensors
    y = y.reshape(B, N, N, 5, hidden_dim)
    left, right, left_gate, right_gate, out_gate = torch.unbind(y, dim=3)

    # ------------------------------------------------------------
    # 3. Gating (sigmoid) – keep the tensors in fp16
    # ------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # ------------------------------------------------------------
    # 4. Fuse (left * left_gate * mask) and (right * right_gate * mask)
    #    with a tiny Triton kernel when a mask is present.
    # ------------------------------------------------------------
    if not nomask:
        # mask may be bool or float, ensure fp16 and broadcast on hidden dimension
        mask_fp16 = mask_tensor.to(torch.float16, device=device)
        size = left.numel()
        BLOCK = 1024
        grid = ( (size + BLOCK - 1) // BLOCK, )

        # left = left * left_gate * mask
        mask_gate_fuse_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            mask_fp16.data_ptr(),
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
        # right = right * right_gate * mask
        mask_gate_fuse_kernel[grid](
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            size,
            BLOCK=BLOCK,
            num_warps=4,
        )
    else:
        left  = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------
    # 5. O(N³) contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM: C = A @ Bᵀ.
    # ------------------------------------------------------------
    # reshape to [B*hidden, N, N] (contiguous for cuBLAS)
    left_cont = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_cont = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batched matmul : (B*H, N, N) @ (B*H, N, N)ᵀ
    out_cont = torch.bmm(left_cont, right_cont.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim]
    out = out_cont.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------
    # 6. Second LayerNorm (fp16)
    # ------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16, device=device)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16, device=device)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------
    # 7. Final gating and projection back to dim
    # ------------------------------------------------------------
    out = out * out_gate  # element‑wise gating (fp16)

    to_out_w = weights["to_out.weight"].to(torch.float16, device=device)
    out = torch.nn.functional.linear(out, to_out_w)  # [B, N, N, dim] fp16

    # ------------------------------------------------------------
    # 8. Cast to float32 as required by the harness
    # ------------------------------------------------------------
    return out.to(torch.float32)

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Tiny Triton kernel used to fuse mask, gate and element‑wise product.
# It works on a flattened view of a tensor of shape (B,N,N,hidden).
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated (size = total #elements)
    gate_ptr,       # *fp16 : gate tensor, same shape as x
    mask_ptr,       # *fp16 : mask tensor broadcasted on hidden dim
    size,           # i32   : total number of elements
    BLOCK: tl.constexpr,   # compile‑time tile size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask_off = offs < size

    x   = tl.load(x_ptr   + offs, mask=mask_off, other=0.0)
    g   = tl.load(gate_ptr+ offs, mask=mask_off, other=0.0)
    m   = tl.load(mask_ptr+ offs, mask=mask_off, other=1.0)   # 1.0 = “no‑mask”

    tl.store(x_ptr + offs, x * g * m, mask=mask_off)

      fused into a single GEMM,
    * an optional mask is applied with a custom Triton kernel,
    * the O(N³) contraction is carried out as a single batched GEMM
      (torch.bmm) after a cheap permute+contiguous copy,
    * a second LayerNorm, an output gate and a final linear projection
      bring the tensor back to the original feature dimension.
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (keep everything in fp16)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(torch.float16, device=device)
    norm_b = weights["norm.bias"].to(torch.float16, device=device)

    # cast once to fp16 and norm in‑place
    x = input_tensor.to(torch.float16)
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear layers into a single GEMM
    # ------------------------------------------------------------------
    w_left     = weights["left_proj.weight"].to(torch.float16, device=device)
    w_right    = weights["right_proj.weight"].to(torch.float16, device=device)
    w_lgate    = weights["left_gate.weight"].to(torch.float16, device=device)
    w_rgate    = weights["right_gate.weight"].to(torch.float16, device=device)
    w_outgate  = weights["out_gate.weight"].to(torch.float16, device=device)

    # shape: (5*hidden_dim, dim)
    combined_weight = torch.cat(
        [w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0

    y = torch.nn.functional.linear(x, combined_weight)
