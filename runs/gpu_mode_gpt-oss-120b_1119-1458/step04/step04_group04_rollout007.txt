"""
Outgoing TriMul (AlphaFold‑3) forward pass – Triton‑accelerated elementwise fusion.

Algorithm
---------
1. Layer‑norm the 4‑D input in float32 (more stable for the first norm).
2. Cast to float16 – all heavy work runs in half‑precision.
3. Linear projections (left/right) and three gating linear layers (all fp16).
4. Fuse element‑wise multiplication of left * left_gate * mask
   (right * right_gate * mask) with a tiny Triton kernel.  When ``nomask=True``
   the mask is a tensor of ones, so the same kernel still runs.
5. Heavy O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM (C = A @ Bᵀ) using torch.bmm, which
   maps to highly‑optimised cuBLAS Tensor‑Core kernels.
6. Second layer‑norm on the hidden dimension (fp16).
7. Apply the output gate (sigmoid) and a final linear projection back to ``dim``.
8. Cast the result back to float32 for the external harness.

At least one custom Triton kernel (the mask‑gate fusion) is used,
while the matrix multiplication leverages the GPU‑native cuBLAS library.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: fused element‑wise multiplication (tensor * gate * mask)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    x_ptr,          # *fp16 : tensor to be updated, shape (size,)
    gate_ptr,       # *fp16 : gate tensor, same shape as x
    mask_ptr,       # *fp16 : mask tensor, shape (B,N,N) – broadcast over H
    size,           # i32   : total number of elements (B·N·N·H)
    BLOCK: tl.constexpr,   # compile‑time tile size (≤1024)
    H: tl.constexpr,       # hidden dimension (compile‑time for fast division)
):
    pid = tl.program_id(0)                     # block id
    offs = pid * BLOCK + tl.arange(0, BLOCK)   # linear offsets handled by block
    active = offs < size                        # boundary mask

    # Load the three operands
    x = tl.load(x_ptr + offs, mask=active, other=0.0)          # fp16
    g = tl.load(gate_ptr + offs, mask=active, other=0.0)       # fp16

    # Compute index into the mask (broadcast on the hidden dimension)
    mask_idx = offs // H                                        # integer division
    m = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)   # fp16 (1 == no mask)

    # Fused multiplication
    out = x * g * m

    # Write result back in‑place
    tl.store(x_ptr + offs, out, mask=active)


# ----------------------------------------------------------------------
# Wrapper that performs the batched GEMM using torch.bmm (cuBLAS)
# ----------------------------------------------------------------------
def batched_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    where ``left`` and ``right`` have shape [B, N, N, H] (fp16).
    The operation is performed as a batched matmul over (B·H) matrices
    of shape (N, N) ↔ (N, N)ᵀ.
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    C = torch.bmm(A, B_.transpose(1, 2))            # [B*H, N, N]
    # restore original 4‑D layout
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point – the custom kernel expected by the harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no autograd).

    Parameters
    ----------
    data : (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : dict name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional ``nomask``

    Returns
    -------
    torch.Tensor : output tensor [B, N, N, dim] (float32, same device as input)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(device)
    norm_b = weights["norm.bias"].to(device)
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_w, bias=norm_b
    )  # [B, N, N, dim] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the rest of the pipeline
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (no bias) – all in fp16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(device).to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device).to(torch.float16)

    left = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_w) # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣ Gating signals (sigmoid)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(device).to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5️⃣ Mask + gate fusion via Triton (runs even when ``nomask=True``)
    # ------------------------------------------------------------------
    if nomask:
        # mask of ones – shape [B,N,N]; broadcasted inside the kernel
        mask_fp16 = torch.ones(
            input_tensor.shape[:3], dtype=torch.float16, device=device
        )
    else:
        mask_fp16 = mask_tensor.to(torch.float16)

    # Ensure contiguity for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_fp16 = mask_fp16.contiguous()

    total_elements = left.numel()
    BLOCK = 1024                                          # ≤ 1024 threads per CUDA block
    grid = ((total_elements + BLOCK - 1) // BLOCK,)

    # left = left * left_gate * mask
    mask_gate_fuse_kernel[grid](
        left.data_ptr(),
        left_gate.data_ptr(),
        mask_fp16.data_ptr(),
        total_elements,
        BLOCK=BLOCK,
        H=hidden_dim,
        num_warps=4,
    )
    # right = right * right_gate * mask
    mask_gate_fuse_kernel[grid](
        right.data_ptr(),
        right_gate.data_ptr(),
        mask_fp16.data_ptr(),
        total_elements,
        BLOCK=BLOCK,
        H=hidden_dim,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction: batched GEMM (A @ Bᵀ)
    # ------------------------------------------------------------------
    out = batched_gemm_fp16(left, right)   # [B, N, N, H], fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)

    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b
    )  # fp16

    # ------------------------------------------------------------------
    # 8️⃣ Output gating + final projection back to ``dim``
    # ------------------------------------------------------------------
    out = out * out_gate                # element‑wise gating (fp16)
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 9️⃣ Cast back to float32 for the external harness and return
    # ------------------------------------------------------------------
    return out.to(torch.float32)
