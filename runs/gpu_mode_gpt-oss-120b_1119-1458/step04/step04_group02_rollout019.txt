def trmul_outgoing_kernel(a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_batch,
    stride_i, stride_k, stride_j,
    stride_out_i, stride_out_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_head = tl.program_id(2)  # flatten batch*hidden dimension

    b_idx = pid_head // H
    h_idx = pid_head % H

    # offsets in elements
    head_offset = b_idx * stride_batch + h_idx
    # (Note: hidden stride_h = 1)
    base_a = a_ptr + head_offset
    base_b = b_ptr + head_offset
    base_c = c_ptr + head_offset

    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J

    i_offsets = i0 + tl.arange(0, BLOCK_I)
    j_offsets = j0 + tl.arange(0, BLOCK_J)

    mask_i = i_offsets < N
    mask_j = j_offsets < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = kk * BLOCK_K
        k_offsets = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < N

        # Load A tile: shape (BLOCK_I, BLOCK_K)
        a_ptrs = base_a + i_offsets[:, None] * stride_i + k_offsets[None, :] * stride_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        # Load B tile: shape (BLOCK_K, BLOCK_J), note B_T = right transposed
        b_ptrs = base_b + k_offsets[:, None] * stride_k + j_offsets[None, :] * stride_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

    # store

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    # Norm
    x = torch.nn.functional.layer_norm(input_tensor,
         (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x_f16 = x.to(torch.float16)
    # Linear projections: left, right, gates
    left = torch.nn.functional.linear(x_f16, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x_f16, weights["right_proj.weight"].to(torch.float16))
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_f16, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_f16, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_f16, weights["out_gate.weight"].to(torch.float16)))
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16
    left = left * left_gate
    right = right * right_gate
    # Contraction using Triton kernel
    out = _trmul_outgoing_kernel_launch(left, right)   # shape [B,N,N,H] fp16
    # To_out_norm layernorm (hidden dimension)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))
    # Cast back to float32
    return out.to(torch.float32)

```
@triton.jit
def trmul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_batch, stride_i, stride_k,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)

    # compute batch and hidden indices
    b_idx = pid_hb // H
    h_idx = pid_hb % H

    head_offset = b_idx * stride_batch + h_idx

    # base pointers
    a_base = a_ptr + head_offset
    b_base = b_ptr + head_offset
    c_base = c_ptr + head_offset

    # tile indices
    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J

    i_offsets = i0 + tl.arange(0, BLOCK_I)
    j_offsets = j0 + tl.arange(0, BLOCK_J)

    mask_i = i_offsets < N
    mask_j = j_offsets < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = kk * BLOCK_K
        k_offsets = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < N

        # Load A tile (i,k)
        a_ptrs = a_base + i_offsets[:, None] * stride_i + k_offsets[None, :] * stride_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # Load B tile (k,j) transposed (right)
        b_ptrs = b_base + k_offsets[:, None] * stride_k + j_offsets[None, :] * stride_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_base + i_offsets[:, None] * stride_i + j_offsets[None, :] * stride_k
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

```python
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# kernel
@triton.jit
def trmul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_batch, stride_i, stride_k,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_head = tl.program_id(2)

    # decode batch and hidden
    b_idx = pid_head // H
    h_idx = pid_head % H

    # compute base offset for this head
    head_offset = b_idx * stride_batch + h_idx

    # pointers to the beginning of each head's sub-tensor
    a_head = a_ptr + head_offset
    b_head = b_ptr + head_offset
    c_head = c_ptr + head_offset

    # tile start indices
    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J

    # offsets inside the tile
    i_off = i0 + tl.arange(0, BLOCK_I)
    j_off = j0 + tl.arange(0, BLOCK_J)

    mask_i = i_off < N
    mask_j = j_off < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over K axis
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = kk * BLOCK_K
        k_off = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # A tile (i,k)
        a_ptrs = a_head + i_off[:, None] * stride_i + k_off[None, :] * stride_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # B tile (k,j) = right transposed
        b_ptrs = b_head + k_off[:, None] * stride_k + j_off[None, :] * stride_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # store
    c_ptrs = c_head + i_off[:, None] * stride_i + j_off[None, :] * stride_k
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def _batched_trmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    assert left.shape == right.shape
    B, N, _, H = left.shape
    total_heads = B * H
    # strides in element counts
    stride_batch = left.stride(0)
    stride_i = left.stride(1)  # N*H
    stride_k = left.stride(2)  # H

    # allocate output
    out = torch.empty_like(left)

    # tile sizes (tuned for H100)
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
        total_heads,
    )
    trmul_outgoing_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B,
        N,
        H,
        stride_batch,
        stride_i,
        stride_k,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    return out

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Implements a fused batched GEMM for the O(N³) contraction using Triton.
    All heavy arithmetic is performed in fp16 (Tensor‑core friendly) while
    the surrounding LayerNorm and gating are kept in fp32 for numerical
    stability. The final result is cast back to fp32.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. First LayerNorm (float32)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # 2. Linear projections (float32) + gating (float32)
    # convert weights once
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # Left/right projections (float32)
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,H] fp32
    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,H] fp32

    # Optional mask (broadcast on hidden dim)
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [B,N,N,1], fp32
        left = left * mask
        right = right * mask

    # Gates (float32)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for the heavy O(N³) contraction
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # 3. O(N³) contraction using Triton batched GEMM
    out_fp16 = _batched_trmul_fp16(left_fp16, right_fp16)  # [B,N,N,H] fp16

    # Convert back to fp32 for subsequent LayerNorm & projection
    out = out_fp16.to(torch.float32)

    # 4. Second LayerNorm over hidden dim (float32)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # 5. Multiply by output gate and final linear projection
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp32

    return out

"""
custom_kernel – “outgoing” TriMul forward (AlphaFold‑3).

Algorithm
---------
1️⃣ Layer‑norm the input (float32 for stability).
2️⃣ Cast to fp16 and apply all four linear projections (left, right, left‑gate,
   right‑gate, out‑gate) in fp16.  The optional mask is fused as a cheap
   element‑wise multiplication.
3️⃣ Fuse the left‑gate/right‑gate by in‑place multiplication.
4️⃣ Contract the two projected tensors:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM C = A @ Bᵀ with A and B stored as
   [B,N,N,H].  A custom Triton kernel performs the contraction directly on
   the original layout (no permute/contiguous copy), using Tensor‑core‑friendly
   fp16 dot‑products and accumulating in fp32.
5️⃣ Cast the result back to fp32 and apply the second LayerNorm (hidden dim),
   the output gate and the final linear projection to the original dimension.
6️⃣ Return a float32 tensor of shape [B, N, N, dim].

The heavy O(N³) work is now done in a highly‑optimized Triton kernel,
eliminating the costly ``permute().contiguous()`` copies of the reference
implementation while still using mixed‑precision arithmetic for maximal
throughput on an H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the contraction
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _trmul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to left, right, output
    B, N, H,                           # dims
    stride_batch, stride_i, stride_k,  # strides (in elements) for the left tensor
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM where the batch dimension is B*H (flattened batch‑hidden)."""
    pid_i = tl.program_id(0)   # tile row (i)
    pid_j = tl.program_id(1)   # tile col (j)
    pid_h = tl.program_id(2)   # flattened head = batch*hidden

    # ------------------------------------------------------------------
    # Decode batch index and hidden‑channel index from flattened head
    # ------------------------------------------------------------------
    b_idx = pid_h // H
    h_idx = pid_h % H

    # Base offset of the current head (h varies as the fastest dimension, stride = 1)
    head_offset = b_idx * stride_batch + h_idx

    # Pointers to the start of the current head
    a_head = a_ptr + head_offset
    b_head = b_ptr + head_offset
    c_head = c_ptr + head_offset

    # ------------------------------------------------------------------
    # Tile coordinates
    # ------------------------------------------------------------------
    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J

    i_off = i0 + tl.arange(0, BLOCK_I)
    j_off = j0 + tl.arange(0, BLOCK_J)

    mask_i = i_off < N
    mask_j = j_off < N

    # ------------------------------------------------------------------
    # Accumulator (fp32 for numeric safety)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Reduction over k dimension
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = kk * BLOCK_K
        k_off = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # A tile: left[b,i,k,h]  (i‑k)
        a_ptrs = a_head + i_off[:, None] * stride_i + k_off[None, :] * stride_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)                       # fp16 → fp32

        # B tile: right transposed (k‑j) = right[b,j,k,h]
        b_ptrs = b_head + k_off[:, None] * stride_k + j_off[None, :] * stride_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        b = b.to(tl.float32)

        # Rank‑K dot product → (BLOCK_I, BLOCK_J) matrix
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result (cast back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = c_head + i_off[:, None] * stride_i + j_off[None, :] * stride_k
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def _batched_trmul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Launch the Triton kernel that computes the O(N³) contraction.
    left / right : [B, N, N, H]   (torch.float16, contiguous)
    Returns a new tensor of the same shape and dtype.
    """
    B, N, _, H = left.shape
    total_heads = B * H

    # Strides in element counts (not bytes)
    stride_batch = left.stride(0)   # N * N * H
    stride_i = left.stride(1)       # N * H
    stride_k = left.stride(2)       # H

    out = torch.empty_like(left)    # fp16 output buffer

    # Tile sizes – tuned for H100 Tensor‑core performance
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,   # i‑tiles
        (N + BLOCK_J - 1) // BLOCK_J,   # j‑tiles
        total_heads,                    # batch*hidden heads
    )

    _trmul_outgoing_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B,
        N,
        H,
        stride_batch,
        stride_i,
        stride_k,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=8,        # enough warps to keep the H100 busy
        num_stages=2,       # double‑buffering of loads
    )
    return out

# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul module (no gradients).
    All heavy arithmetic (the N³ contraction) is performed by a fused Triton
    kernel, while the surrounding LayerNorm, gating, and final projection are
    carried out with PyTorch in mixed precision.
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 and pre‑cast all model weights that will be used in the
    #    fp16 region (projections, gates, final linear)
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # Cast weights once
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"]          # keep fp32 for LN
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]               # keep fp32 for final linear

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (fp16)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x_fp16, left_proj_w)   # [B, N, N, H] fp16
    right = torch.nn.functional.linear(x_fp16, right_proj_w)  # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1] fp16
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_w))

    # Fuse the gates (in‑place multiplication)
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ O(N³) contraction via the custom Triton kernel
    # ------------------------------------------------------------------
    out_fp16 = _batched_trmul_fp16(left, right)   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Cast back to fp32 for the remaining (light‑weight) ops
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Output gating and final linear projection (still fp32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] fp32

    # ------------------------------------------------------------------
    # 10️⃣ Return the result (float32, same device as input)
    # ------------------------------------------------------------------
    return out
