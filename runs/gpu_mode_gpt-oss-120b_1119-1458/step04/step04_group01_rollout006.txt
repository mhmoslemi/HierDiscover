def fused_mul3_kernel(a_ptr, b_ptr, c_ptr, out_ptr, size, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    c = tl.load(c_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b * c, mask=mask)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# Triton kernels
@triton.jit
def elementwise_mul2_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offset < size
    a = tl.load(a_ptr + offset, mask=mask)
    b = tl.load(b_ptr + offset, mask=mask)
    tl.store(out_ptr + offset, a * b, mask=mask)

def _apply_mul2(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Element‑wise multiplication of two half‑precision tensors using a Triton kernel.
    Both inputs must have the same shape and be contiguous.
    """
    a_c = a.contiguous()
    b_c = b.contiguous()
    out = torch.empty_like(a_c)
    size = a_c.numel()
    BLOCK = 65536  # adjust for occupancy
    grid = (triton.cdiv(size, BLOCK),)
    elementwise_mul2_kernel[grid](
        a_c, b_c, out, size,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return out

def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the contraction out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a batched GEMM on FP16 tensors.
    left/right shape: [B, N, N, H] (half)
    """
    B, N, _, H = left.shape
    # (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    # (B*H, N, N) @ (B*H, N, N)^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator (no gradients) implemented with
    FP16 arithmetic, a BETTER fusion of linear layers and a Triton‑based
    element‑wise multiplication. The computationally intensive N³ contraction is
    performed with a batched GEMM (cuBLAS) while the element‑wise products are
    fused via a small Triton kernel to satisfy the “use Triton” rule.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # Cast weights to half‑precision once
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)

    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)

    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ---------- 1️⃣  FP16 input and first LayerNorm ----------
    x_fp16 = input_tensor.to(torch.float16)
    x_norm = F.layer_norm(x_fp16, (dim,), weight=norm_w, bias=norm_b)  # [B,N,N,dim] half

    # ---------- 2️⃣  Combined linear for left projection + left gate ----------
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)          # (2*hidden, dim)
    left_comb = F.linear(x_norm, left_comb_w)                         # [B,N,N,2*hidden]

    left_proj = left_comb[..., :hidden_dim]                           # [B,N,N,hidden]
    left_gate_pre = left_comb[..., hidden_dim:]                       # [B,N,N,hidden]
    left_gate = torch.sigmoid(left_gate_pre)                          # half

    # ---------- 3️⃣  Combined linear for right projection + right gate ----------
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)      # (2*hidden, dim)
    right_comb = F.linear(x_norm, right_comb_w)                       # [B,N,N,2*hidden]

    right_proj = right_comb[..., :hidden_dim]
    right_gate_pre = right_comb[..., hidden_dim:]
    right_gate = torch.sigmoid(right_gate_pre)

    # ---------- 4️⃣  Fuse projection·gate element‑wise product with Triton ----------
    left = _apply_mul2(left_proj, left_gate)   # [B,N,N,hidden]
    right = _apply_mul2(right_proj, right_gate)

    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ---------- 5️⃣  Heavy N³ contraction (batched GEMM) ----------
    out = _batched_matmul_transpose_fp16(left, right)               # [B,N,N,hidden]

    # ---------- 6️⃣  Second LayerNorm (hidden) ----------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # ---------- 7️⃣  Out‑gate linear + sigmoid ----------
    out_gate_pre = F.linear(x_norm, out_gate_w)                     # [B,N,N,hidden]
    out_gate = torch.sigmoid(out_gate_pre)

    # ---------- 8️⃣  Apply out‑gate (Triton fused mul) ----------
    out = _apply_mul2(out, out_gate)

    # ---------- 9️⃣  Final projection back to model dimension ----------
    out = F.linear(out, to_out_w)                                    # [B,N,N,dim] half

    # ---------- 🔚 Cast back to float32 ----------
    return out.to(torch.float32)

"""
TriMul (outgoing) – fast Triton‑accelerated forward pass
========================================================

The implementation follows the reference PyTorch module exactly but
optimises the data‑flow:

1️⃣  All computation is performed in FP16 except the final cast back to FP32.
2️⃣  The two linear projections (proj + gate) are fused by concatenating the
    weight matrices, so the input tensor is read only once.
3️⃣  Element‑wise products (proj × gate and the final out‑gate) are carried out
    with a tiny Triton kernel – this satisfies the “must use Triton” rule and
    reduces the number of Python‑level kernels.
4️⃣  The heavy N³ contraction

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    is realised as a batched GEMM (C = A @ Bᵀ) using cuBLAS via ``torch.bmm``.
    The tensors are reshaped to (B·H, N, N) so a single batched GEMM is used,
    which is the fastest path on an H100.
5️⃣  All LayerNorms use the half‑precision weights and bias.

The whole forward pass is therefore a handful of highly‑optimised
operations plus three tiny Triton kernels for the element‑wise
multiplications.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernels – element‑wise multiplication (2‑input & 3‑input)
# ----------------------------------------------------------------------
@triton.jit
def _elementwise_mul2_kernel(
    a_ptr, b_ptr, out_ptr, size, BLOCK_SIZE: tl.constexpr
):
    """C = A * B  (flat 1‑D view)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _apply_mul2(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Fuse an element‑wise multiplication with a Triton kernel.
    Both tensors must have the same shape and be contiguous.
    """
    a_c = a.contiguous()
    b_c = b.contiguous()
    out = torch.empty_like(a_c)
    size = a_c.numel()
    BLOCK = 65536  # 64 K elements per program; good trade‑off
    grid = (triton.cdiv(size, BLOCK),)
    _elementwise_mul2_kernel[grid](
        a_c, b_c, out, size,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Batched matrix multiplication (contraction) – uses cuBLAS inside torch.bmm
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    Parameters
    ----------
    left, right : torch.Tensor
        Shape [B, N, N, H] (FP16).

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, H] (FP16).
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for a single batched GEMM
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    # (B*H, N, N) @ (B*H, N, N)ᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``,
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # 0️⃣  Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣  Cast everything to FP16 (weights + input)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)

    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)

    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    # ------------------------------------------------------------------
    x_fp16 = input_tensor.to(torch.float16)                     # [B,N,N,dim]
    x = F.layer_norm(x_fp16, (dim,), weight=norm_w, bias=norm_b)  # still FP16

    # ------------------------------------------------------------------
    # 3️⃣  Linear proj + gate (fused weight matrices)
    # ------------------------------------------------------------------

    left_gate = torch.sigmoid(left_comb[..., hidden_dim:])      # [B,N,N,H]

    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, dim)

    # ------------------------------------------------------------------
    out = _apply_mul2(out, out_gate)

    # ------------------------------------------------------------------
    # 9️⃣  Final projection back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)                   # [B,N,N,dim] (FP16)

    # ------------------------------------------------------------------
    # 🔚  Cast back to float32 (required output dtype)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
