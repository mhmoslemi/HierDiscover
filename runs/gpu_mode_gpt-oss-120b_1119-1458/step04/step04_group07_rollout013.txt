"""
Outgoing TriMul (AlphaFold‑3) – fast forward pass.

Key ideas
---------
* Fuse the five linear + sigmoid gates that operate on the raw input into a single
  matrix multiplication (one huge weight of shape (5*hidden, dim)).  This removes
  the overhead of launching five separate GEMMs and reduces memory traffic.
* Use a custom Triton kernel for the O(N³) contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
  The kernel treats (batch * hidden) as the batch dimension and computes a
  batched GEMM C = A @ Bᵀ in fp16 → fp32 accumulate.  Tile size is
  128×128×64 for N ≥ 512, otherwise 64×64×64.  The kernel uses 8 warps and
  3 pipeline stages (configurable).  This is the only Triton component as required.
* All other ops (LayerNorm, gating, final projection) stay in PyTorch but are
  performed in fp16 to keep the bandwidth low.  The final result is cast back
  to float32 for the harness.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : [M, N, K]   (M = batch*hidden, N = seq_len, K = seq_len)
#   C    : [M, N, N]
#   All tensors are fp16, accumulator is fp32.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K]
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp16  [M, N, N]
    M, N, K,             # scalar sizes
    stride_am, stride_ai, stride_ak,   # A stride for (M, i, k)
    stride_bm, stride_bj, stride_bk,   # B stride for (M, j, k)
    stride_cm, stride_ci, stride_cj,   # C stride for (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM where each program computes one (BLOCK_M x BLOCK_N) tile of C.
    """
    pid_i = tl.program_id(0)   # tile along first N dimension (i)
    pid_j = tl.program_id(1)   # tile along second N dimension (j)
    pid_z = tl.program_id(2)   # batch * hidden index

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_start in range(0, K, BLOCK_K):
        cur_k = k_start
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile  (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )   # fp16

        # Load B tile transposed  (BLOCK_K, BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )   # fp16

        # Mixed‑precision dot (fp16 * fp16 → fp32)
        acc += tl.dot(a, b)

    # Store result tile
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute ``out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`` using the
    Triton kernel above.  ``left`` and ``right`` must be fp16 and contiguous.

    Parameters
    ----------
    left, right : torch.Tensor
        Shape ``[B, N, N, H]`` (fp16).

    Returns
    -------
    out : torch.Tensor
        Shape ``[B, N, N, H]`` (fp16).
    """
    B, N, _, H = left.shape
    # Merge batch and hidden dim -> M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    C = torch.empty_like(A, dtype=torch.float16)

    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Choose tile sizes
    if N >= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64 if N < 1024 else 128   # larger K‑tile for very large seq‑len
    else:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # Restore original layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operator (gradient‑free).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (float32, bool or None)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default ``True``).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 for numerical stability)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------
    # 2️⃣ Cast to fp16 for the rest of the pipeline
    # -------------------------------------------------
    x = x.to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Fuse all input‑to‑hidden linear projections (5 * hidden_dim)
    # -------------------------------------------------
    # Gather the five weight matrices (all have shape [hidden_dim, dim])
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # Concatenate along the output‑feature dimension
    fused_weight = torch.cat(
        [left_proj_w,
         left_gate_w,
         right_proj_w,
         right_gate_w,
         out_gate_w],
        dim=0,
    )  # shape (5*hidden_dim, dim)

    # Single matmul: [B,N,N,dim] @ [5*hidden, dim].T  ->  [B,N,N,5*hidden]
    proj_all = torch.nn.functional.linear(x, fused_weight)  # fp16

    # Split back to the five tensors
    proj_all = proj_all.view(input_tensor.shape[0],
                             input_tensor.shape[1],
                             input_tensor.shape[2],
                             5 * hidden_dim)  # keep original 4‑D layout

    left, left_gate, right, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=-1
    )  # each: [B, N, N, hidden]

    # -------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dim)
    # -------------------------------------------------
    if not nomask:
        # mask may be bool or float; cast to fp16 for multiplication
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 5️⃣ Gating (sigmoid) – fp16
    # -------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Fuse gate into projection (in‑place for memory efficiency)
    torch.mul(left, left_gate, out=left)
    torch.mul(right, right_gate, out=right)

    # -------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction (batched GEMM) in Triton
    # -------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    contracted = _batched_matmul_transpose_fp16(left, right)  # [B,N,N,hidden] fp16

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm over ``hidden_dim`` (fp16)
    # -------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    contracted = torch.nn.functional.layer_norm(
        contracted,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # -------------------------------------------------
    # 8️⃣ Final gating + projection back to ``dim``
    # -------------------------------------------------
    contracted = contracted * out_gate                      # element‑wise fp16
    to_out_w = weights["to_out.weight"].to(torch.float16)   # shape (dim, hidden)
    out = torch.nn.functional.linear(contracted, to_out_w)  # [B,N,N,dim] fp16

    # -------------------------------------------------
    # 9️⃣ Cast back to float32 for the harness
    # -------------------------------------------------
    return out.to(torch.float32)
```
