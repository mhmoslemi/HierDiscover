"""
TriMul (outgoing) – fast Triton‑augmented forward pass
=====================================================

* LayerNorm (FP16) → Linear projections (FP16) → Sigmoid gates
* Optional mask + gate fusion performed by a tiny Triton kernel
* The O(N³) contraction (the “einsum” in the reference) is carried out
  with a single ``torch.einsum`` on FP16 tensors; this avoids the costly
  permute‑and‑contiguous copy required for a batched GEMM.
* Second LayerNorm (FP16) → output‑gate → final linear projection (FP16)
* Result is cast back to float32 (the format expected by the caller).

The Triton kernel fuses the element‑wise operations
``proj * gate * mask`` for either the left or the right branch,
saving two separate mask‑broadcasts and one extra memory pass.
All heavy linear algebra stays in cuBLAS/Tensor‑Core (via ``torch.nn.functional.linear``
and ``torch.einsum``), which is optimal on H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _mask_gate_fuse_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,              # tensors
    M, H,                                               # problem size
    stride_proj_m, stride_proj_h,                       # proj strides
    stride_gate_m, stride_gate_h,                       # gate strides
    stride_out_m, stride_out_h,                         # out strides
    stride_mask,                                        # mask stride (1‑D)
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,       # tile sizes
):
    """
    out[m, h] = proj[m, h] * gate[m, h] * mask[m]

    The kernel processes ``BLOCK_M`` rows (flattened sequence‑pair indices)
    and the full hidden dimension ``BLOCK_H`` (≤ hidden_dim) per launch.
    """
    pid = tl.program_id(0)                      # linear block index
    block_start = pid * BLOCK_M
    offs_m = block_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    offs_h = tl.arange(0, BLOCK_H)                         # [BLOCK_H]

    # Load mask (broadcast over hidden)
    mask_val = tl.load(mask_ptr + offs_m * stride_mask,
                       mask=offs_m < M,
                       other=0.0).to(tl.float16)          # [BLOCK_M]
    mask_h = mask_val[:, None]                             # [BLOCK_M, 1]

    # Load projection and gate tiles
    proj = tl.load(
        proj_ptr + offs_m[:, None] * stride_proj_m
                + offs_h[None, :] * stride_proj_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    gate = tl.load(
        gate_ptr + offs_m[:, None] * stride_gate_m
                + offs_h[None, :] * stride_gate_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0,
    )
    # Fuse: element‑wise product with mask
    out = proj * gate * mask_h

    # Store
    tl.store(
        out_ptr + offs_m[:, None] * stride_out_m
                + offs_h[None, :] * stride_out_h,
        out,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
    )

    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]

    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32/bool) or None
        - weights_dict : mapping from layer name to weight tensors (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) on the same device as input.
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → FP16)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias   = weights["norm.bias"].to(torch.float16)
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    ).to(torch.float16)          # [B, N, N, dim] FP16

    B, N, _, _ = x.shape
    M = B * N * N                     # flattened pair count

    # ------------------------------------------------------------------
    # 2️⃣ Cast linear weights to FP16 once
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections & sigmoid gates (FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(-1, dim)                     # (M, dim)

    left_proj = torch.nn.functional.linear(x_flat, left_proj_w)    # (M, hidden)
    right_proj = torch.nn.functional.linear(x_flat, right_proj_w)

    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, left_gate_w)
    )                                 # (M, hidden)
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, right_gate_w)
    )

    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, out_gate_w)
    )                                 # (M, hidden)
    out_gate = out_gate.view(B, N, N, hidden_dim)  # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask + gate fusion (Triton)
    # ------------------------------------------------------------------
    if not nomask:
        # flattened mask -> (M,)
        mask_fp16 = mask_tensor.to(torch.float16).reshape(-1)

        # allocate destination buffers (same layout as proj tensors)
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # strides in element counts
        stride_lp_m, stride_lp_h = left_proj.stride()
        stride_lg_m, stride_lg_h = left_gate.stride()
        stride_rp_m, stride_rp_h = right_proj.stride()
        stride_rg_m, stride_rg_h = right_gate.stride()
        stride_o_m, stride_o_h = left.stride()      # same as proj stride
        stride_ro_m, stride_ro_h = right.stride()
        stride_mask = mask_fp16.stride()[0]

        # Kernel launch configuration
        BLOCK_M = 128                 # rows per block (tuneable)
        BLOCK_H = hidden_dim          # we tile the whole hidden dim (≤128)

        grid = ((M + BLOCK_M - 1) // BLOCK_M,)

        # left = left_proj * left_gate * mask
        _mask_gate_fuse_kernel[grid](
            left_proj, left_gate, mask_fp16, left,
            M, hidden_dim,
            stride_lp_m, stride_lp_h,
            stride_lg_m, stride_lg_h,
            stride_o_m,  stride_o_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=8, num_stages=2,
        )
        # right = right_proj * right_gate * mask
        _mask_gate_fuse_kernel[grid](
            right_proj, right_gate, mask_fp16, right,
            M, hidden_dim,
            stride_rp_m, stride_rp_h,
            stride_rg_m, stride_rg_h,
            stride_ro_m,  stride_ro_h,
            stride_mask,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
            num_warps=8, num_stages=2,
        )
        # reshape to 4‑D tensors
        left = left.view(B, N, N, hidden_dim)   # (B, N, N, hidden)
        right = right.view(B, N, N, hidden_dim)
    else:
        # No mask – simple element‑wise fuse in PyTorch
        left = (left_proj * left_gate).view(B, N, N, hidden_dim)
        right = (right_proj * right_gate).view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ O(N³) contraction: einsum over the k‑dimension
    #    (equivalent to left @ rightᵀ for each hidden slice)
    # ------------------------------------------------------------------
    # left, right: (B, N, N, hidden)
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)  # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Output gate + final linear projection (FP16)
    # ------------------------------------------------------------------
    out = out * out_gate
    # final linear: hidden → dim
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim) FP16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 (the expected output dtype)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
