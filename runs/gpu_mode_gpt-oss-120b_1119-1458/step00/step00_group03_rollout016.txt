def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    # optional epsilon
    eps = 1e-5

    # LayerNorm x
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # Linear proj
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(x.dtype)
        left = left * mask
        right = right * mask

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Convert to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # left_h shape [B,N,N,hidden_dim]
    # Permute to bring hidden_dim before N dims
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()
    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)

    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=left_flat.device)

    # Triton kernel launch
    import triton
    import triton.language as tl

    # define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Compute grid size
    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    tiles_per_batch = num_pid_m * num_pid_n
    total_programs = BH * tiles_per_batch

    trimul_gemm_kernel[total_programs](
        left_flat,
        right_flat,
        out_flat,
        N,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(1),
        right_flat.stride(2),
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        num_warps=4,  # adjust
    )
    # Reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,hidden_dim]

    # to_out_norm (LayerNorm on hidden_dim)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Gating
    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

    tile_idx = pid % tiles_per_batch

    pid_n = tile_idx % num_pid_n

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_n = offs_n < N

def custom_kernel(data):
    """
    Triton-accelerated TriMul forward pass (outgoing version).
    The kernel computes the batch-wise pairwise multiplication:
        out[b,i,j,c] = sum_k left[b,i,k,c] * right[b,j,k,c]
    where left/right are gated linear projections of the input.
    The rest of the TriMul operations (norms, linear layers, gates)
    are performed with PyTorch for simplicity.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # LayerNorm on last dim
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if not nomask:
        # mask shape [bs, N, N], broadcast on dim
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [bs, N, N, 1]
        left = left * mask
        right = right * mask

    # Gating mechanisms
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Convert to float16 for the heavy GEMM
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # left_h shape [B, N, N, hidden_dim]

    # Bring hidden_dim forward so that we can treat (B*hidden_dim) as batch dimension
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)

    # Output buffer (float32 accumulator)
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=left_flat.device)

    # Triton kernel launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Compute grid size: one program per (batch*hidden) tile
    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    tiles_per_batch = num_pid_m * num_pid_n
    total_programs = BH * tiles_per_batch

    trimul_gemm_kernel[total_programs](
        left_flat,
        right_flat,
        out_flat,
        N,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(1),
        right_flat.stride(2),
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        num_warps=8,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden_dim]

    # to_out_norm (LayerNorm over hidden_dim)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply output gating
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass accelerated with Triton.

The heavy part of TriMul is the tensor contraction
    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
where left/right are the gated linear projections of the input.
We reformulate the contraction as a batched matrix multiply:
    out_h = L_h @ R_h.T   (L_h, R_h ∈ ℝ^{N×N})
for every hidden channel c (here treated as an extra batch dimension).
A custom Triton GEMM kernel performs this multiplication in FP16
with FP32 accumulation; the rest of the TriMul pipeline
(LayerNorm, linear layers, gates, output LayerNorm and projection)
is executed with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_gemm_kernel(
    L_ptr,                      # half *  (BH, N, N)
    R_ptr,                      # half *  (BH, N, N)
    O_ptr,                      # float * (BH, N, N)
    N,                          # seq_len (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    stride_bL, stride_iL, stride_kL,
    stride_bR, stride_iR, stride_kR,
    stride_bO, stride_iO, stride_jO,
):
    """
    Batched GEMM: for each combined batch index (b * hidden_dim)
    compute out = left @ right.T  where left, right ∈ R^{N×N}.
    """
    pid = tl.program_id(0)

    # Tiling layout
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_pid_m * num_pid_n

    batch_idx = pid // tiles_per_batch
    tile_idx = pid % tiles_per_batch
    pid_m = tile_idx // num_pid_n
    pid_n = tile_idx % num_pid_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to the beginning of the current batch slice
    L_batch = L_ptr + batch_idx * stride_bL
    R_batch = R_ptr + batch_idx * stride_bR
    O_batch = O_ptr + batch_idx * stride_bO

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load L tile (M x K)
        L_ptrs = L_batch + offs_m[:, None] * stride_iL + offs_k[None, :] * stride_kL
        L = tl.load(L_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load R tile transposed (K x N)   we need R[k, j]
        # Original layout: R[j, k] with strides (stride_iR = N, stride_kR = 1)
        R_ptrs = R_batch + offs_k[:, None] * stride_kR + offs_n[None, :] * stride_iR
        R = tl.load(R_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(L, R)

    # Store the result
    O_ptrs = O_batch + offs_m[:, None] * stride_iO + offs_n[None, :] * stride_jO
    tl.store(O_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):

    Triton‑accelerated TriMul forward (outgoing) pass.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ---------- 1. LayerNorm on the channel dimension ----------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ---------- 2. Linear projections (no bias) ----------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ---------- 3. Optional mask ----------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)            # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ---------- 4. Gating ----------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- 5. Cast to half for the heavy GEMM ----------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape                     # left_h : [B, N, N, hidden_dim]

    # Bring hidden_dim forward so we can treat (B * hidden_dim) as batch
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)                # [BH, N, N]  (half)
    right_flat = right_perm.view(BH, N, N)              # [BH, N, N]  (half)

    # Output buffer for the GEMM (float32 accumulator)
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=left_flat.device)

    # ---------- 6. Triton GEMM launch ----------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    tiles_per_batch = num_pid_m * num_pid_n
    total_programs = BH * tiles_per_batch

    trimul_gemm_kernel[(total_programs,)](
        left_flat,
        right_flat,
        out_flat,
        N,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        # strides for L
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        # strides for R
        right_flat.stride(0),
        right_flat.stride(1),
        right_flat.stride(2),
        # strides for O
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        num_warps=8,          # good occupancy on H100
    )

    # ---------- 7. Reshape back to [B, N, N, hidden_dim] ----------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]

    # ---------- 8. Output LayerNorm ----------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ---------- 9. Output gating ----------
    out = out * out_gate

    # ---------- 10. Final linear projection back to original dim ----------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```
