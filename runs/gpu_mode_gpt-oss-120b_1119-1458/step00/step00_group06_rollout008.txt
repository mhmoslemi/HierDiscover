@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_an, stride_ak,
    stride_bm, stride_bn, stride_bk,
    stride_cm, stride_cn, stride_ck,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_b = tl.program_id(0)  # batch index
    pid_m = tl.program_id(1)  # block row
    pid_n = tl.program_id(2)  # block col

    # Compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + pid_b * stride_am + offs_m[:, None] * stride_an + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B: shape (BLOCK_K, BLOCK_N)
        b_ptrs = B_ptr + pid_b * stride_bm + offs_k[:, None] * stride_bn + offs_n[None, :] * stride_bk
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Compute dot-product
        acc += tl.dot(a, b)

    # Store to C
    c_ptrs = C_ptr + pid_b * stride_cm + offs_m[:, None] * stride_cn + offs_n[None, :] * stride_ck

def custom_kernel(data):
    """
    Custom Triton implementation of the TriMul “outgoing” update.
    Steps:
    1) Layer‑norm the input.
    2) Linear projections to a hidden dimension for left, right and three gates.
    3) Apply the mask (if present) and element‑wise sigmoid gates.
    4) Fuse the left/right matrices with a batched half‑precision matrix‑multiply
       implemented in Triton (the O(N³) einsum).  Each hidden channel is treated as a
       separate batch element, so the kernel computes M = B * hidden_dim matrix multiplies
       of size N × N in parallel.
    5) Apply a second LayerNorm, the output gate and a final linear projection back to `dim`.
    The function returns a tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    # unpack
    input_tensor, mask_tensor, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm weight / bias
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # Linear weights
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1. Input layer‑norm
    # torch.nn.functional.layer_norm expects normalized_shape without batch dims
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # 2. Projections (no bias)
    left_proj = F.linear(x, left_proj_weight)   # (bs, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_weight)

    # 3. Optional mask
    if not nomask:
        # mask shape (bs, N, N) -> (bs, N, N, 1)
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # 4. Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # 5. Apply gates
    # Cast to half for the heavy O(N³) part
    left = (left_proj * left_gate).to(torch.float16)
    right = (right_proj * right_gate).to(torch.float16)

    # 6. Prepare tensors for batched GEMM
    # Permute to (bs, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Transpose the last two dimensions of right for the dot‑product
    right_t = right_perm.transpose(-1, -2).contiguous()

    # Collapse batch and channel dimensions → M = bs * hidden_dim
    M = left_perm.shape[0] * left_perm.shape[1]
    N = left_perm.shape[2]        # seq_len
    K = N                         # same as N

    A = left_perm.reshape(M, N, K).contiguous()
    B = right_t.reshape(M, K, N).contiguous()

    # Output buffer
    C = torch.empty_like(A, dtype=torch.float16)

    # Triton launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (M,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    # Compute strides (int)
    stride_am = A.stride(0)
    stride_an = A.stride(1)
    stride_ak = A.stride(2)

    stride_bm = B.stride(0)
    stride_bn = B.stride(1)
    stride_bk = B.stride(2)

    stride_cm = C.stride(0)
    stride_cn = C.stride(1)
    stride_ck = C.stride(2)

    # launch kernel
    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_am, stride_an, stride_ak,
        stride_bm, stride_bn, stride_bk,
        stride_cm, stride_cn, stride_ck,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Convert back to float32 and reshape
    out_hidden = C.to(torch.float32).view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1)

    # 7. Second LayerNorm over hidden_dim
    out_norm = F.layer_norm(out_hidden, (hidden_dim,), weight=to_out_norm_weight,
                            bias=to_out_norm_bias, eps=1e-5)

    # 8. Apply output gate
    out_gated = out_norm * out_gate

    # 9. Final linear projection back to dim
    output = F.linear(out_gated, to_out_weight)

    return output

3. Apply the optional pairwise mask and the element‑wise gates.
4. Fuse the N³ einsum
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   by treating each hidden channel as an independent batch element and
   executing a packed half‑precision batched matrix‑multiply with Triton.
   For each (batch, hidden) pair we compute a matrix product of size
   (seq_len × seq_len).  The kernel accumulates in FP32 and stores FP16.
5. Reshape the result back to (bs, N, N, hidden_dim), apply a second
   LayerNorm, the output gate and a final linear projection to the original
   dimension.
The function returns a float‑32 tensor of shape [bs, seq_len, seq_len, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_an, stride_ak,
    stride_bm, stride_bn, stride_bk,
    stride_cm, stride_cn, stride_ck,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched matrix multiply C = A @ B   (A: M×N×K, B: M×K×N, C: M×N×N)
    Each program ID corresponds to (batch, block_row, block_col).
    """
    pid_b = tl.program_id(0)          # batch / channel index
    pid_m = tl.program_id(1)          # block row index
    pid_n = tl.program_id(2)          # block col index

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for kk in range(0, K, BLOCK_K):
        offs_k = kk + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (A_ptr
                  + pid_b * stride_am
                  + offs_m[:, None] * stride_an
                  + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = (B_ptr
                  + pid_b * stride_bm
                  + offs_k[:, None] * stride_bn
                  + offs_n[None, :] * stride_bk)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # FP32 accumulation
        acc += tl.dot(a, b)

    # Write back FP16 (the kernel’s output tensor is allocated as half)
    c_ptrs = (C_ptr
              + pid_b * stride_cm
              + offs_m[:, None] * stride_cn
              + offs_n[None, :] * stride_ck)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated implementation of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output with shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1. Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2. Load model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 3. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # (bs, N, N, dim)

    # ------------------------------------------------------------------
    # 4. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_weight)   # (bs, N, N, hidden_dim)
    right_proj = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 5. Optional pairwise mask
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape (bs, N, N) → (bs, N, N, 1) for broadcasting
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 6. Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 7. Apply gates; cast to FP16 for the heavy O(N³) part
    # ------------------------------------------------------------------
    left = (left_proj * left_gate).to(torch.float16)   # (bs, N, N, hidden_dim)
    right = (right_proj * right_gate).to(torch.float16)

    # ------------------------------------------------------------------
    # 8. Prepare data for batched GEMM
    #     left  : (bs, N, N, hidden_dim) → (bs, hidden_dim, N, N)
    #     right : (bs, N, N, hidden_dim) → (bs, hidden_dim, N, N) and then transpose K↔J
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (bs, hidden, N, N)  i‑k
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (bs, hidden, N, N)  j‑k
    right_t = right_perm.transpose(-1, -2).contiguous()  # (bs, hidden, N, N)  k‑j

    # Collapse batch & channel → M = bs * hidden_dim
    M = bs * hidden_dim
    N = seq_len
    K = seq_len

    A = left_perm.reshape(M, N, K).contiguous()   # (M, N, K)
    B = right_t.reshape(M, K, N).contiguous()    # (M, K, N)
    C = torch.empty_like(A, dtype=torch.float16)  # (M, N, N)

    # ------------------------------------------------------------------
    # 9. Triton kernel launch (batched half‑precision GEMM)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        M,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Compute strides (in elements, not bytes)
    stride_am = A.stride(0)
    stride_an = A.stride(1)
    stride_ak = A.stride(2)

    stride_bm = B.stride(0)
    stride_bn = B.stride(1)
    stride_bk = B.stride(2)

    stride_cm = C.stride(0)
    stride_cn = C.stride(1)
    stride_ck = C.stride(2)

    # Launch the kernel
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_am,
        stride_an,
        stride_ak,
        stride_bm,
        stride_bn,
        stride_bk,
        stride_cm,
        stride_cn,
        stride_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 10. Post‑process output of the kernel
    # ------------------------------------------------------------------
    out_hidden = (
        C.to(torch.float32)
        .view(bs, hidden_dim, seq_len, seq_len)
        .permute(0, 2, 3, 1)
    )  # (bs, N, N, hidden_dim)

    # Second LayerNorm over the hidden dimension
    out = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # Apply output gate and final linear projection back to `dim`
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # (bs, N, N, dim)

    return out
```
