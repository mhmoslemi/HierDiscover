@triton.jit
def fused_mul_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_j, stride_left_d,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_d,
    stride_right_b, stride_right_i, stride_right_j, stride_right_d,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_d,
    stride_mask_b, stride_mask_i, stride_mask_j, stride_mask_d,
    stride_lout_b, stride_lout_i, stride_lout_j, stride_lout_d,
    stride_rout_b, stride_rout_i, stride_rout_j, stride_rout_d,
    BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    total = B * N * N
    # compute batch, i, j from pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # offsets for each dimension
    i_offsets = tl.arange(0, BLOCK_D)  # d index

    # compute masks for out-of-range d
    d_mask = i_offsets < D

    # Load mask scalar (broadcast across D)
    mask_val = tl.load(mask_ptr + b*stride_mask_b + i*stride_mask_i + j*stride_mask_j, mask=False, other=1.0) # shape scalar

    # but mask is shape (B,N,N,1) with stride for d maybe stride_mask_d = ? Since last dim 1.
    # simplest treat mask_ptr as pointer to scalar per (b,i,j), ignoring d.

    # Loop over D
    for d_start in range(0, D, BLOCK_D):
        cur_d = d_start + i_offsets
        mask = cur_d < D

        # Load values
        l = tl.load(left_ptr + b*stride_left_b + i*stride_left_i + j*stride_left_j + cur_d*stride_left_d, mask=mask, other=0.0)
        l_gate = tl.load(left_gate_ptr + b*stride_lgate_b + i*stride_lgate_i + j*stride_lgate_j + cur_d*stride_lgate_d, mask=mask, other=0.0)
        r = tl.load(right_ptr + b*stride_right_b + i*stride_right_i + j*stride_right_j + cur_d*stride_right_d, mask=mask, other=0.0)
        r_gate = tl.load(right_gate_ptr + b*stride_rgate_b + i*stride_rgate_i + j*stride_rgate_j + cur_d*stride_rgate_d, mask=mask, other=0.0)

        # Compute out: multiply by gates and mask
        out_l = l * l_gate * mask_val
        out_r = r * r_gate * mask_val

        # store
        tl.store(left_out_ptr + b*stride_lout_b + i*stride_lout_i + j*stride_lout_j + cur_d*stride_lout_d, out_l, mask=mask)
        tl.store(right_out_ptr + b*stride_rout_b + i*stride_rout_i + j*stride_rout_j + cur_d*stride_rout_d, out_r, mask=mask)

import torch
import triton
import triton.language as tl

@triton.jit
def apply_proj_gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    num_rows, hidden_dim,
    stride_proj_row, stride_proj_col,
    stride_gate_row, stride_gate_col,
    stride_out_row, stride_out_col,
    BLOCK_D: tl.constexpr):
    row_idx = tl.program_id(0)
    if row_idx >= num_rows:
        return

    mask_val = tl.load(mask_ptr + row_idx)

    for d_start in range(0, hidden_dim, BLOCK_D):
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim

        proj = tl.load(proj_ptr + row_idx * stride_proj_row + offs_d * stride_proj_col,
                       mask=mask_d, other=0.0)
        gate = tl.load(gate_ptr + row_idx * stride_gate_row + offs_d * stride_gate_col,
                       mask=mask_d, other=0.0)

        tl.store(out_ptr + row_idx * stride_out_row + offs_d * stride_out_col,
                 out, mask=mask_d)

def custom_kernel(data):
    """
    TritMul outgoing implementation.
    Performs:
        x_norm = LayerNorm(x)
        left = (x_norm @ left_proj) * sigmoid(x_norm @ left_gate) * mask
        right = (x_norm @ right_proj) * sigmoid(x_norm @ right_gate) * mask
        out = Σ_k left[...,k,:] * right[...,k,:]
        out = LayerNorm(out) * sigmoid(x_norm @ out_gate)
        output = out @ to_out_weight
    The element‑wise gating and mask multiplication are fused in a Triton kernel.
    """
    import torch.nn.functional as F

    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    x_flat = x_norm.view(-1, dim)  # [B*N*N, dim]

    # ----- Linear projections -------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]  # [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]  # [dim, hidden_dim]

    left_proj = F.linear(x_flat, left_proj_weight)          # [B*N*N, hidden_dim]
    right_proj = F.linear(x_flat, right_proj_weight)        # [B*N*N, hidden_dim]
    left_gate = torch.sigmoid(F.linear(x_flat, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_flat, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_weight))  # [B*N*N, hidden_dim]

    # ----- Mask handling ------------------------------------------------------
    if mask_tensor is None or mask_tensor.numel() == 0:
        mask_flat = torch.ones(x_flat.shape[0], dtype=dtype, device=device)
    else:
        mask_flat = mask_tensor.view(-1).to(dtype)  # shape [B*N*N]

    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    num_rows = left_proj.shape[0]  # B * N * N

    stride_proj_row = left_proj.stride(0)
    stride_proj_col = left_proj.stride(1)
    stride_gate_row = left_gate.stride(0)
    stride_gate_col = left_gate.stride(1)
    stride_out_row = left_out.stride(0)
    stride_out_col = left_out.stride(1)

    BLOCK_D = 64
    grid = (num_rows,)

    # left side
    apply_proj_gate_mask_kernel[grid](
        left_proj,
        left_gate,
        mask_flat,
        left_out,
        num_rows,
        hidden_dim,
        stride_proj_row,
        stride_proj_col,
        stride_gate_row,
        stride_gate_col,
        stride_out_row,
        stride_out_col,
        BLOCK_D=BLOCK_D,
    )

        right_proj,
        right_gate,
        mask_flat,
        right_out,
        num_rows,
        hidden_dim,

        stride_proj_col,
        stride_gate_row,
        stride_gate_col,
        stride_out_row,
        stride_out_col,
        BLOCK_D=BLOCK_D,
    )

    left_out = left_out.view(B, N, N, hidden_dim)
    right_out = right_out.view(B, N, N, hidden_dim)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*hidden_dim, N, N]

    # Restore shape [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----- Output normalization, gating and final linear ----------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # [B, N, N, hidden_dim]

    out_gate = out_gate.view(B, N, N, hidden_dim)
    out = out * out_gate

    # Final linear projection back to `dim`
    out_flat = out.view(-1, hidden_dim)
    final = F.linear(out_flat, to_out_weight)  # [B*N*N, dim]
    final = final.view(B, N, N, dim)

    return final

"""
TriMul (outgoing) – AlphaFold3 core operation.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor.
2. Project to a hidden dimension `H` (named `hidden_dim`) with two linear
   layers (`left_proj` and `right_proj`) and compute two sigmoid gates
   (`left_gate`, `right_gate`).  Both the projected tensors and the gates
   are multiplied together and masked.  This element‑wise fusing is done
   in a custom Triton kernel.
3. The heavy “tri‑multiplicative” reduction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is performed as a batched GEMM:
        (B*H, N, N)  ×  (B*H, N, N)ᵀ  →  (B*H, N, N)
4. Layer‑norm the result (`to_out_norm`), apply a third sigmoid gate
   (`out_gate`) and finish with a linear projection back to `dim`.
All kernels run on the GPU; the rest of the work uses highly‑optimised
PyTorch BLAS calls.  The final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def apply_proj_gate_mask_kernel(
    proj_ptr,               # *[num_rows, hidden_dim]  – projected tensor
    gate_ptr,               # *[num_rows, hidden_dim]  – sigmoid gate
    mask_ptr,               # *[num_rows]               – 0/1 mask per (b,i,j)
    out_ptr,                # *[num_rows, hidden_dim]  – destination
    num_rows, hidden_dim,   # runtime sizes
    stride_proj_row, stride_proj_col,
    stride_gate_row, stride_gate_col,
    stride_mask_row,
    stride_out_row, stride_out_col,
    BLOCK_D: tl.constexpr,  # compile‑time tile size for the hidden dim
):
    row = tl.program_id(0)                       # one program per (b,i,j)
    if row >= num_rows:
        return

    # mask for the whole row (scalar 0.0 or 1.0)
    mask_val = tl.load(mask_ptr + row * stride_mask_row)

    # iterate over the hidden dimension in tiles
    for d_start in range(0, hidden_dim, BLOCK_D):
        d_off = d_start + tl.arange(0, BLOCK_D)                # [BLOCK_D]
        d_mask = d_off < hidden_dim

        # ----- load projection and gate ---------------------------------
        proj = tl.load(
            proj_ptr
            + row * stride_proj_row
            + d_off * stride_proj_col,
            mask=d_mask,
            other=0.0,
        )
        gate = tl.load(
            gate_ptr
            + row * stride_gate_row
            + d_off * stride_gate_col,
            mask=d_mask,
            other=0.0,
        )

        # ----- fuse: proj * gate * mask ---------------------------------
        out = proj * gate * mask_val

        # store result
        tl.store(
            out_ptr
            + row * stride_out_row
            + d_off * stride_out_col,
            out,
            mask=d_mask,
        )

            - weights     : dict with the model parameters
            - config      : dict containing 'dim', 'hidden_dim', ...

        torch.Tensor of shape [B, N, N, dim] (float32)
    """

    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # 1) Layer‑norm on the input (the "norm" layer)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                                   # [B, N, N, dim]

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(flat_x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(flat_x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(flat_x, out_gate_weight))  # [B*N*N, H]

        stride_gate_row,
        stride_gate_col,
        stride_mask_row,
        stride_out_row,
        stride_out_col,
        BLOCK_D=BLOCK_D,
    )

    apply_proj_gate_mask_kernel[grid](
        right_proj,
        right_gate,
        mask_flat,
        right_out,
        num_rows,
        hidden_dim,
        stride_proj_row,
        stride_proj_col,
        stride_gate_row,
        stride_gate_col,
        stride_mask_row,
        stride_out_row,
        stride_out_col,
        BLOCK_D=BLOCK_D,
    )

    # reshape back to 4‑D tensors
    left_out = left_out.view(B, N, N, hidden_dim)
    right_out = right_out.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5) Core TriMul reduction:
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched GEMM.
    # ------------------------------------------------------------------
    # (B, N, N, H) → (B*H, N, N) for both matrices
    left_mat = left_out.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right_out.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # batched matrix multiply (left × rightᵀ)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]

    # reshape to original 4‑D layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # now out has shape [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6) Output layer‑norm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]            # [dim, hidden_dim]

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )                                          # [B, N, N, H]

    # apply out_gate (already computed on the flattened input)
    out_gate = out_gate.view(B, N, N, hidden_dim)
    out = out * out_gate

    # final linear back to `dim`
    out_flat = out.view(-1, hidden_dim)               # [B*N*N, H]
    final = F.linear(out_flat, to_out_weight)        # [B*N*N, dim]
    final = final.view(B, N, N, dim)

    return final
```
