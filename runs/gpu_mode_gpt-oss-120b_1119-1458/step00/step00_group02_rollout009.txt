custom_kernel for the “outgoing” TriMul operator (AlphaFold3).

Algorithm
---------
1. Layer‑Norm the input tensor (last dimension = dim).
2. Linear projections:
   left  = x @ W_left   (→ hidden_dim)
   right = x @ W_right  (→ hidden_dim)
3. Optional mask (shape [B,N,N]) is applied to both projections.

   right = right * right_gate

   TILE_M × TILE_N block of the output for a specific (batch, hidden)
   pair.
6. Layer‑Norm on the hidden dimension of `out`.
7. Apply `out_gate` (broadcasted) and a final linear projection back to
   `dim`.

The kernel operates in FP32 (you may cast the projections to FP16 before
calling the kernel for a small speed‑up; the accumulator stays in FP32
to preserve accuracy).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM that computes Σ_k left[b,i,k,h] * right[b,k,j,h]
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute the cubic “TriMul” using a tiled GEMM.

    Arguments
    ---------
    left_ptr, right_ptr, out_ptr : pointers to the tensors.
    B, N, H                     : batch size, sequence length, hidden dim.
    stride_*                    : strides (in elements) for the 4‑D tensors.
    BLOCK_M, BLOCK_N, BLOCK_K   : tiling sizes (compile‑time constants).
    """
    pid_m = tl.program_id(0)            # block index over i (rows)
    pid_n = tl.program_id(1)            # block index over j (cols)
    pid_z = tl.program_id(2)            # combined batch * hidden index

    # decode batch and hidden index from the third program dimension
    b = pid_z // H                      # batch id
    h = pid_z %  H                      # hidden‑dim id

    # offsets for the i (row) and j (col) dimensions of the output tile
    i_off = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_off < N
    j_mask = j_off < N

    # accumulator for the current output tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # ------------------------------------------------------------------
        # Load a BLOCK_M × BLOCK_K tile from left  : left[b, i, k, h]
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_l_batch
            + i_off[:, None] * stride_l_i
            + k_off[None, :] * stride_l_k
            + h * stride_l_h
        )
        left_tile = tl.load(
            left_ptrs,
            mask=(i_mask[:, None] & k_mask[None, :]),
            other=0.0,
        )

        # ------------------------------------------------------------------
        # Load a BLOCK_K × BLOCK_N tile from right transposed:
        #   right_T[b, k, j, h] == right[b, j, k, h]
        # ------------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + b * stride_r_batch
            + k_off[:, None] * stride_r_k
            + j_off[None, :] * stride_r_i   # original i‑stride used for “j” after transpose
            + h * stride_r_h
        )
        right_tile = tl.load(
            right_ptrs,
            mask=(k_mask[:, None] & j_mask[None, :]),
            other=0.0,
        )

        # Accumulate the matrix product of the two tiles
        acc += tl.dot(left_tile, right_tile)   # (M×K) · (K×N) → (M×N)

    # ----------------------------------------------------------------------
    # Store the computed tile to the output tensor
    # ----------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_o_batch
        + i_off[:, None] * stride_o_i
        + j_off[None, :] * stride_o_j
        + h * stride_o_h
    )
    tl.store(out_ptrs, acc, mask=(i_mask[:, None] & j_mask[None, :]))


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (may be all‑ones)
        - weights      : dict of weight tensors (see docstring)
        - config       : dict containing 'dim' and 'hidden_dim'

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, on the same device as inputs)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # H in the paper

    # ------------------------------------------------------------------
    # Layer‑Norm on the input (last dimension = dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Apply mask (if present).  Mask broadcasted over the hidden dim.
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)  # [B,N,N,1]
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # Gates (all linear → sigmoid)
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = F.linear(x, left_gate_weight).sigmoid()   # [B,N,N,hidden_dim]
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate   = F.linear(x, out_gate_weight).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Optional mixed‑precision: cast projections to fp16 for the kernel
    # (the accumulator stays fp32 inside the kernel)
    # ------------------------------------------------------------------
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=x.device)

    # ------------------------------------------------------------------
    # Strides (in elements) for the three tensors
    # ------------------------------------------------------------------
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h = right_fp16.stride()
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # ------------------------------------------------------------------
    # Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid_x = (N + BLOCK_M - 1) // BLOCK_M
    grid_y = (N + BLOCK_N - 1) // BLOCK_N
    grid_z = B * hidden_dim                      # each (batch, h) pair gets its own program

    trimul_outgoing_kernel[ (grid_x, grid_y, grid_z) ](
        left_fp16,
        right_fp16,
        out,
        B, N, hidden_dim,
        stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
        stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
        stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Post‑processing: hidden‑dim LayerNorm, gating, final linear
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # [B,N,N,hidden_dim]

    # Apply the output gate (broadcasted over the hidden dim)
    out = out * out_gate

    # Final linear projection back to `dim`
    to_out_weight = weights["to_out.weight"]   # (dim, hidden_dim)
    out = F.linear(out, to_out_weight)         # [B,N,N,dim]

    return out
```
