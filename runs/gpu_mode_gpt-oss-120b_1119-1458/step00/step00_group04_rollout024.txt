"""
custom_kernel – fast “outgoing” TriMul (AlphaFold3) implementation.

Algorithm
---------
1. Layer‑norm the input tensor (last dimension = dim).
2. Linear projections → left/right (shape [B,N,N,hidden_dim]).
3. Linear projections for the three gates → left_gate, right_gate, out_gate
   and apply sigmoid.
4. Fuse mask, left_gate and right_gate with a tiny Triton kernel:
   out_left = left * mask * left_gate
   out_right = right * mask * right_gate
   (mask is broadcasted over the hidden dimension).
5. Compute the core triangle‑multiplicative update:
   out[b,i,j,d] = Σ_k out_left[b,i,k,d] * out_right[b,j,k,d]
   This is a batched matrix‑multiply per hidden channel, realised with
   torch.bmm on the (N×N) matrices after reshaping (highly optimised
   on H100).
6. Layer‑norm the result (hidden_dim), multiply by out_gate and apply the
   final linear projection back to the original dimension.

Only the element‑wise mask+gate fusion is done in Triton, satisfying the
requirement to use a custom kernel while keeping the heavy O(N³) part on
GPU‑accelerated torch kernels.
"""

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: fuse mask, left_gate / right_gate with the projected tensors.
# --------------------------------------------------------------------------- #
@triton.jit
def mask_gate_fuse_kernel(
    # tensors
    proj_ptr,          # left or right projection [B,N,N,hidden]
    gate_ptr,          # corresponding gate  [B,N,N,hidden]
    mask_ptr,          # mask [B,N,N] (float, 0/1)
    out_ptr,           # output buffer [B,N,N,hidden]
    # dimensions (compile‑time constants)
    B: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    # strides
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b,  stride_out_i,  stride_out_j,  stride_out_h,
    # block size for hidden dimension
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)                     # one program per (b,i,j)
    total = B * N * N
    if pid >= total:
        return

    # decode pid → (b,i,j)
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # base offsets for the three tensors (without hidden dimension)
    proj_base = b * stride_proj_b + i * stride_proj_i + j * stride_proj_j
    gate_base = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j
    mask_off  = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    out_base  = b * stride_out_b  + i * stride_out_i  + j * stride_out_j

    # scalar mask (broadcast over the hidden dimension)
    mask_val = tl.load(mask_ptr + mask_off)

    # loop over hidden dimension in tiles of BLOCK_H
    h_off = 0
    while h_off < H:
        cur = tl.arange(0, BLOCK_H)
        mask_h = (h_off + cur) < H                     # out‑of‑bounds mask
        # pointer arithmetic for the current hidden block
        proj_ptr_off = proj_ptr + proj_base + (h_off + cur) * stride_proj_h
        gate_ptr_off = gate_ptr + gate_base + (h_off + cur) * stride_gate_h
        out_ptr_off  = out_ptr  + out_base  + (h_off + cur) * stride_out_h

        proj_vec = tl.load(proj_ptr_off, mask=mask_h, other=0.0)
        gate_vec = tl.load(gate_ptr_off, mask=mask_h, other=0.0)

        # fused operation: proj * mask * gate
        out_vec = proj_vec * mask_val * gate_vec

        tl.store(out_ptr_off, out_vec, mask=mask_h)

        h_off += BLOCK_H


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim]
    """
    # --------------------------- unpack --------------------------- #
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim       = config["dim"]
    hidden    = config["hidden_dim"]
    nomask    = config.get("nomask", True)      # if True we ignore the mask

    # -------------------------- LayerNorm -------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )   # [B,N,N,dim]

    # -------------------------- Projections ------------------------ #
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------- Gates ---------------------------- #
    left_gate  = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate   = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )   # [B,N,N,H]

    # ---------------------------- Mask --------------------------- #
    B, N, _, _ = input_tensor.shape
    if nomask or mask_tensor is None:
        mask = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask = mask_tensor.to(dtype)

    # -------------------- Triton: mask+gate fusion -------------- #
    # Allocate output buffers
    left_fused  = torch.empty_like(left)
    right_fused = torch.empty_like(right)

    # Strides (converted to python ints)
    s_l_b, s_l_i, s_l_j, s_l_h = left.stride()
    s_r_b, s_r_i, s_r_j, s_r_h = right.stride()
    s_lg_b, s_lg_i, s_lg_j, s_lg_h = left_gate.stride()
    s_rg_b, s_rg_i, s_rg_j, s_rg_h = right_gate.stride()
    s_m_b, s_m_i, s_m_j = mask.stride()
    s_lo_b, s_lo_i, s_lo_j, s_lo_h = left_fused.stride()
    s_ro_b, s_ro_i, s_ro_j, s_ro_h = right_fused.stride()

    # Grid = one program per (b,i,j)
    total_elements = B * N * N
    grid = (total_elements,)

    BLOCK_H = 32   # hidden‑dimension tile size (fits in registers)

    # Fuse mask with left projection
    mask_gate_fuse_kernel[grid](
        left, left_gate, mask, left_fused,
        B, N, hidden,
        s_l_b, s_l_i, s_l_j, s_l_h,
        s_lg_b, s_lg_i, s_lg_j, s_lg_h,
        s_m_b, s_m_i, s_m_j,
        s_lo_b, s_lo_i, s_lo_j, s_lo_h,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # Fuse mask with right projection
    mask_gate_fuse_kernel[grid](
        right, right_gate, mask, right_fused,
        B, N, hidden,
        s_r_b, s_r_i, s_r_j, s_r_h,
        s_rg_b, s_rg_i, s_rg_j, s_rg_h,
        s_m_b, s_m_i, s_m_j,
        s_ro_b, s_ro_i, s_ro_j, s_ro_h,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # --------------------- Core TriMul (batched matmul) ------------------- #
    # Use fp16 for the heavy O(N³) work
    left_fused_f16  = left_fused.to(torch.float16)
    right_fused_f16 = right_fused.to(torch.float16)

    # reshape to (B*H, N, N) matrices
    B, N, _, H = left_fused_f16.shape
    left_mat  = left_fused_f16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_fused_f16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # out_mat[b*H + h] = left_mat[b*H + h] @ right_mat[b*H + h].T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B,N,N,H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------- Output norm -------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out, (hidden,), weight=to_out_norm_w, bias=to_out_norm_b
    )

    # -------------------------- Gate & final linear ------------------- #
    out = out * out_gate.to(out.dtype)                     # broadcast multiply
    out = torch.nn.functional.linear(out, weights["to_out.weight"])  # [B,N,N,dim]

    return out
