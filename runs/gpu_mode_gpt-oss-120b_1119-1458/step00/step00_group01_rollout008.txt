@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_bc, stride_a_i, stride_a_k,
    stride_b_bc, stride_b_k, stride_b_j,
    stride_c_bc, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(axis=0) # bc dimension (batch*hidden)
    pid_m = tl.program_id(1) # block row
    pid_n = tl.program_id(2) # block col

    # Compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for a and b blocks
    a_ptrs = a_ptr + pid * stride_a_bc + offs_m[:, None] * stride_a_i + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    b_ptrs = b_ptr + pid * stride_b_bc + tl.arange(0, BLOCK_K)[:, None] * stride_b_k + offs_n[None, :] * stride_b_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k*BLOCK_K + tl.arange(0,BLOCK_K) < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k*BLOCK_K + tl.arange(0,BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0)
        # dot on k dimension, but tl.dot expects shape (BLOCK_M, BLOCK_K) and (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)
        # advance pointers
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # write back
    c_ptrs = c_ptr + pid * stride_c_bc + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

```
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # extract dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = input_tensor
    # compute mean and var over last dim (dim)
    # we can use torch.nn.functional.layer_norm
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    # linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)
    # mask
    if mask is not None and mask.numel()>0:
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_
    # gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()
    left = left * left_gate
    right = right * right_gate
    # Prepare for kernel
    B, N, _, _ = left.shape
    # permute to (B, hidden_dim, N, N) and contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # transpose right for multiplication (j,k) -> (k,j)
    right_t = right_perm.transpose(-2, -1).contiguous()
    # flatten batch*hidden
    left_flat = left_perm.view(B*hidden_dim, N, N)
```

```python
import torch
import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_bc, stride_a_i, stride_a_k,
    stride_b_bc, stride_b_k, stride_b_j,
    stride_c_bc, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bc = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block ranges
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch*channel
    a_base = a_ptr + pid_bc * stride_a_bc
    b_base = b_ptr + pid_bc * stride_b_bc
    c_base = c_ptr + pid_bc * stride_c_bc

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)
        cur_k = k_offset + offs_k

        # mask for k within bounds
        mask_k = cur_k < K

        # load A[i,k]
        a_ptrs = a_base + offs_m[:, None] * stride_a_i + cur_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load B[k,j] (note B is already transposed, so its row index is k)
        b_ptrs = b_base + cur_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product over K
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_base + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    """
    Triton-accelerated implementation of the outgoing TriMul operation.
    Steps:
    1. LayerNorm over the feature dimension.
    2. Linear projections (left/right) without bias.
    3. Optional mask application.
    4. Gated projections using sigmoid gates.
    5. Core pairwise multiplication across the second sequence axis implemented
       as a batched GEMM via a custom Triton kernel (equivalent to the einsum
       '... i k d, ... j k d -> ... i j d').
    6. Final LayerNorm + output gate + linear projection back to `dim`.
    All heavy compute (step 5) is performed in a fused Triton kernel.
    """
    import torch
    import triton
    import triton.language as tl

    # Unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------- 1. LayerNorm -------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )  # shape [B, N, N, dim]

    # ------------------- 2. Linear projections -------------------
    left_proj_weight = weights["left_proj.weight"]        # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------- 3. Optional mask -------------------
    if mask is not None and mask.numel() > 0:

"""
Outgoing TriMul (Triangle Multiplicative Update) – Triton accelerated.

Algorithm
---------
1. Layer‑norm over the feature dimension `dim`.
2. Linear projections (left/right) without bias → shape [B, N, N, H].
3. Optional mask (broadcasted on H) and gating (sigmoid) applied to the
   projected tensors.
4. Core operation: out[i, j, d] = Σ_k left[i, k, d] * right[j, k, d]
   This is mathematically a batched GEMM:
       out_bc = left_bc @ right_bc^T   (bc = batch × hidden_dim)
   Implemented with a custom Triton kernel that fuses the K‑loop,
   uses 128×128 tiles and 64‑wide K‑tiles (tuned for H100).
5. Post‑process the result – LayerNorm over the hidden dimension,
   element‑wise multiplication with an output gate, and a final linear
   projection back to `dim`.

Only the heavy matrix multiplication (step 4) is performed in Triton;
all other ops use regular PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_bc, stride_a_i, stride_a_k,
    stride_b_bc, stride_b_k, stride_b_j,
    stride_c_bc, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched GEMM: for each ``bc`` (= batch × hidden_dim)
    compute C = A @ Bᵀ where A and B have shape (N, N).
    The tensors are stored as
        A[bc, i, k]   with strides (stride_a_bc, stride_a_i, stride_a_k)
        B[bc, k, j]   with strides (stride_b_bc, stride_b_k, stride_b_j)
    and the output C[bc, i, j] with strides (stride_c_bc, stride_c_i, stride_c_j).
    """
    pid_bc = tl.program_id(0)       # batch*hidden index
    pid_m  = tl.program_id(1)       # tile row
    pid_n  = tl.program_id(2)       # tile col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this (bc) slice
    a_base = a_ptr + pid_bc * stride_a_bc
    b_base = b_ptr + pid_bc * stride_b_bc
    c_base = c_ptr + pid_bc * stride_c_bc

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K steps
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K)
        cur_k = k_offset + offs_k
        mask_k = cur_k < K

        # Load A[i, k]
        a_ptrs = a_base + offs_m[:, None] * stride_a_i + cur_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B[k, j] (B is already transposed)
        b_ptrs = b_base + cur_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate dot product over K
        acc += tl.dot(a, b)

    # Write the tile back to C
    c_ptrs = c_base + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑backed forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim]
            - mask:        torch.Tensor, shape [B, N, N] (may be None)
            - weights:     dict of torch.Tensors containing all model weights
            - config:      dict with at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_f = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul via custom Triton GEMM kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left.shape == [B, N, N, hidden_dim]

    # Rearrange to (B, hidden_dim, N, N) – makes the batch*hidden axis contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Transpose the second operand so we compute A @ Bᵀ
    right_t = right_perm.transpose(-2, -1).contiguous()  # [B, H, N, N] with strides (bc, k, j)

    # Collapse batch and hidden dimensions: (B*H, N, N)
    left_flat = left_perm.view(B * hidden_dim, N, N)
    right_flat = right_t.view(B * hidden_dim, N, N)
    out_flat = torch.empty_like(left_flat)  # will hold [B*H, N, N]

    # Strides for the flattened tensors (int64)
    stride_a_bc, stride_a_i, stride_a_k = left_flat.stride()
    stride_b_bc, stride_b_k, stride_b_j = right_flat.stride()
    stride_c_bc, stride_c_i, stride_c_j = out_flat.stride()

    # Tile sizes – chosen for H100 L2/shared‑memory balance
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        N,
        N,
        stride_a_bc,
        stride_a_i,
        stride_a_k,
        stride_b_bc,
        stride_b_k,
        stride_b_j,
        stride_c_bc,
        stride_c_i,
        stride_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # works well on H100
    )

    # Reshape back to original 4‑D layout
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = torch.nn.functional.linear(out, weights["to_out.weight"])  # [B,N,N,dim]

    return output
