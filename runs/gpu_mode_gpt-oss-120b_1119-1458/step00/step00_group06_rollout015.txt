"""
TriMul (outgoing) – fast forward implementation.

The kernel fuses the element‑wise mask application and the multiplicative
gating (left_gate, right_gate) for the two projected tensors.  The heavy
matrix‑multiplication over the sequence dimension (`einsum('...ikd,...jkd->...ijd')`)
is performed with a highly‑optimized batched GEMM (`torch.bmm`).  LayerNorms
and linear layers are left to PyTorch (cuBLAS/cuDNN) which already exploit
Tensor‑Cores.  The overall dataflow:

1. Layer‑norm over the input channel dimension.
2. Linear projections → left/right (hidden_dim) and three gate pre‑activations.
3. Sigmoid on gate tensors.
4. Triton kernel: left = left * left_gate * mask,
               right = right * right_gate * mask   (mask broadcasted).
5. Batched matrix‑multiply: out[i,j] = Σ_k left[i,k]·right[j,k] for each hidden
   channel (implemented via `torch.bmm` on reshaped tensors).
6. Layer‑norm on the hidden dimension, multiply by out_gate, final linear
   projection back to the original channel dimension.
All operations stay on the GPU and the final tensor has dtype float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: mask + gate fusion                                            #
# --------------------------------------------------------------------------- #
@triton.jit
def _mask_gate_kernel(
    left_ptr, left_gate_ptr,    # [B,N,N,HD]
    right_ptr, right_gate_ptr,  # [B,N,N,HD]
    mask_ptr,                   # [B,N,N] (scalar broadcast)
    out_left_ptr, out_right_ptr,  # [B,N,N,HD] results
    B, N, HD,                       # compile‑time ints
    # strides for left/right tensors (same layout for proj & gate)
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    # mask strides
    stride_m_b, stride_m_i, stride_m_j,
    # output strides
    stride_ol_b, stride_ol_i, stride_ol_j, stride_ol_h,
    stride_or_b, stride_or_i, stride_or_j, stride_or_h,
    BLOCK_H: tl.constexpr,          # hidden‑dim tile size
):
    """Apply mask and element‑wise gating.
    One program processes one (b,i,j) element and loops over the hidden
    dimension in blocks of size BLOCK_H.
    """
    pid = tl.program_id(0)                # flattened element index
    total = B * N * N
    # guard – out‑of‑range ids may appear when grid is larger than needed
    if pid >= total:
        return

    # ---- decode (b,i,j) from pid ----------------------------------------
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # mask scalar (float32)
    mask_off = b * stride_m_b + i * stride_m_i + j * stride_m_j
    mask_val = tl.load(mask_ptr + mask_off)

    # base offsets for the 4‑D tensors (no hidden dim offset yet)
    base_l = b * stride_l_b + i * stride_l_i + j * stride_l_j
    base_r = b * stride_r_b + i * stride_r_i + j * stride_r_j
    base_ol = b * stride_ol_b + i * stride_ol_i + j * stride_ol_j
    base_or = b * stride_or_b + i * stride_or_i + j * stride_or_j

    # loop over hidden dimension
    for h_start in range(0, HD, BLOCK_H):
        cur_h = tl.arange(0, BLOCK_H)
        h_valid = (h_start + cur_h) < HD

        # ---- load tensors -------------------------------------------------
        off_h = (h_start + cur_h) * stride_l_h
        left   = tl.load(left_ptr   + base_l   + off_h, mask=h_valid, other=0.0)
        left_g = tl.load(left_gate_ptr + base_l + off_h, mask=h_valid, other=0.0)
        right  = tl.load(right_ptr  + base_r   + off_h, mask=h_valid, other=0.0)
        right_g= tl.load(right_gate_ptr + base_r + off_h, mask=h_valid, other=0.0)

        # ---- fused mask & gate -------------------------------------------
        left_out  = left * left_g * mask_val
        right_out = right * right_g * mask_val

        # ---- store ---------------------------------------------------------
        tl.store(out_left_ptr  + base_ol + off_h, left_out,  mask=h_valid)
        tl.store(out_right_ptr + base_or + off_h, right_out, mask=h_valid)


# --------------------------------------------------------------------------- #
# Public entry point                                                            #
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle‑Multiplicative update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) – will be broadcasted.
            - weights: dict of all linear / LayerNorm parameters.
            - config: {'dim': C, 'hidden_dim': H, 'nomask': bool, ...}

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------- #
    # unpack inputs
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype = input_tensor.dtype
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------- #
    # 1) Input LayerNorm over channel dim (C)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------- #
    # 2) Linear projections + gate pre‑activations (float32)
    #    We rely on torch’s highly‑optimized GEMM for these.
    # ------------------------------------------------------------------- #
    # projections
    left_proj   = F.linear(x, weights["left_proj.weight"])   # [B,N,N,HD]
    right_proj  = F.linear(x, weights["right_proj.weight"])

    # gates (pre‑sigmoid)
    left_gate_pre  = F.linear(x, weights["left_gate.weight"])
    right_gate_pre = F.linear(x, weights["right_gate.weight"])
    out_gate_pre   = F.linear(x, weights["out_gate.weight"])

    # sigmoid
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)   # used later, after to_out_norm

    # ------------------------------------------------------------------- #
    # 3) Optional mask handling
    # ------------------------------------------------------------------- #
    if nomask or mask is None:
        mask = torch.ones(B, N, N, dtype=dtype, device=device)
    else:
        mask = mask.to(dtype)

    # ------------------------------------------------------------------- #
    # 4) Fuse mask * left_gate/right_gate with the projected tensors
    #    using a Triton kernel.
    # ------------------------------------------------------------------- #
    # allocate output buffers
    left_out  = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    total_elements = B * N * N
    # strides (all tensors are contiguous in the last dimension)
    s_l = left_proj.stride()        # (b,i,j,hd)
    s_r = right_proj.stride()
    s_lg = left_gate.stride()
    s_rg = right_gate.stride()
    s_m = mask.stride()
    s_ol = left_out.stride()
    s_or = right_out.stride()

    # launch kernel; one program per (b,i,j) element
    _mask_gate_kernel[(total_elements,)](
        left_proj, left_gate,
        right_proj, right_gate,
        mask,
        left_out, right_out,
        B, N, hidden_dim,
        s_l[0], s_l[1], s_l[2], s_l[3],          # left stride
        s_r[0], s_r[1], s_r[2], s_r[3],          # right stride
        s_m[0], s_m[1], s_m[2],                  # mask stride
        s_ol[0], s_ol[1], s_ol[2], s_ol[3],      # out‑left stride
        s_or[0], s_or[1], s_or[2], s_or[3],      # out‑right stride
        BLOCK_H=32,
        num_warps=4,
    )

    # ------------------------------------------------------------------- #
    # 5) Batched matrix multiplication over the sequence dimension.
    #    left_out, right_out have shape [B,N,N,HD].
    #    We reshape to (B*HD, N, N) and perform bmm left @ rightᵀ.
    # ------------------------------------------------------------------- #
    # reshape to (B*HD, N, N)
    left_mat  = left_out.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right_out.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # core TriMul operation: Σ_k left[i,k] * right[j,k]  (=> left @ rightᵀ)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back‑to original layout [B,N,N,HD]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,HD]

    # ------------------------------------------------------------------- #
    # 6) Hidden‑dim LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate                                      # element‑wise gating
    output = F.linear(out, weights["to_out.weight"])          # back to C dim

    return output
