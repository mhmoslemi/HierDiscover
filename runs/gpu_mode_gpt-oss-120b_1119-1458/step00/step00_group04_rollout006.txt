First, let’s understand what TriMul "outgoing" version means. In AlphaFold, there are two variants: "incoming" and "outgoing", essentially swapping left and right. In the code given, they use einsum `'... i k d, ... j k d -> ... i j d'`. This corresponds to outgoing: left[... i k d] * right[... j k d] summed over k yields output[... i j d]. Actually incoming would be `'... k j d, ... k i d -> ... i j d'` maybe. Anyway, we implement as shown.

We need to merge a few operations:

- Input shape: [B, N, N, C] (dim = C)
- Mask shape: [B, N, N]; optionally no mask (i.e., mask all ones). Mask is broadcast to [B, N, N, 1] for gating multiplication.

- Steps:

  1) LayerNorm on input: per last dimension of size C. Compute mean and variance across last dim per each [B,N,N] element. We can do that with torch.nn.functional.layer_norm or with a Triton kernel for layer norm. However, the problem allows some steps to be done in PyTorch.

The kernel works on FP16 inputs (saving memory and using Tensor Cores)
while accumulating in FP32 for numerical stability.  The final output
is returned in FP32 as required.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def trmul_kernel(
    left_ptr,          # fp16 pointer, shape (BHD, N, N)
    right_ptr,         # fp16 pointer, shape (BHD, N, N)
    out_ptr,           # fp16 pointer, shape (BHD, N, N)
    N,                 # sequence length (scalar)
    # strides for left
    stride_left_bhd, stride_left_m, stride_left_k,
    # strides for right (note: right is (BHD, N, K) where K==N)
    stride_right_bhd, stride_right_n, stride_right_k,
    # strides for output
    stride_out_bhd, stride_out_m, stride_out_n,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_bhd = tl.program_id(0)          # batch*hidden index
    pid_m   = tl.program_id(1)          # tile row
    pid_n   = tl.program_id(2)          # tile column

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for the edge of the matrix
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    # --------------------------------------------------------------
    # Loop over K dimension
    # --------------------------------------------------------------
    for k in range(num_k_tiles):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # left block:   (BLOCK_M, BLOCK_K)
        left_addr = (
            left_ptr
            + pid_bhd * stride_left_bhd
            + offs_m[:, None] * stride_left_m
            + k_offset[None, :] * stride_left_k
        )
        mask_left = mask_m[:, None] & mask_k[None, :]
        left = tl.load(left_addr, mask=mask_left, other=0.0)

        # right block (transposed): (BLOCK_K, BLOCK_N)
        right_addr = (
            right_ptr
            + pid_bhd * stride_right_bhd
            + k_offset[:, None] * stride_right_k
            + offs_n[None, :] * stride_right_n
        )
        mask_right = mask_k[:, None] & mask_n[None, :]
        right = tl.load(right_addr, mask=mask_right, other=0.0)

        # Matrix‑multiply the two blocks; accumulate in fp32
        acc += tl.dot(left, right, out_dtype=tl.float32)

    # --------------------------------------------------------------
    # Store the result (convert back to fp16)
    # --------------------------------------------------------------
    out_addr = (
        out_ptr
        + pid_bhd * stride_out_bhd
        + offs_m[:, None] * stride_out_m
        + offs_n[None, :] * stride_out_n
    )
    mask_out = mask_m[:, None] & mask_n[None, :]
    tl.store(out_addr, acc.to(tl.float16), mask=mask_out)


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Perform the outgoing TriMul update.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask       : torch.Tensor of shape [B, N, N] (bool or float)
            - weights    : dict of all model parameters (torch.Tensors)
            - config     : dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        out: torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (applied only to left/right, not gates)
    # ------------------------------------------------------------------
    if not nomask:
        # broadcast mask to the last dimension
        mask_exp = mask.to(x.dtype).unsqueeze(-1)    # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections of normalized x)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³·C) contraction via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Move hidden dimension to the second axis and make tensors contiguous
    # Shape becomes (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch and hidden dims: (B*hidden_dim, N, N)
    bh = B * hidden_dim
    left_flat = left_perm.view(bh, N, N)
    right_flat = right_perm.view(bh, N, N)

    # Use FP16 inside the kernel (Tensor‑Core friendly)
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16)

    # Strides (in elements) for the three tensors
    stride_l_bhd, stride_l_m, stride_l_k = left_fp16.stride()
    stride_r_bhd, stride_r_n, stride_r_k = right_fp16.stride()
    stride_o_bhd, stride_o_m, stride_o_n = out_fp16.stride()

    # Block sizes – tuned for H100 Tensor‑Core matmul
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Launch grid: (batch*hidden, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (
        bh,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    trmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        N,
        stride_l_bhd,
        stride_l_m,
        stride_l_k,
        stride_r_bhd,
        stride_r_n,
        stride_r_k,
        stride_o_bhd,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim] (float32)
    out = (
        out_fp16.to(torch.float32)
        .view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gate and final projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm * out_gate
    out_final = F.linear(out_norm, weights["to_out.weight"])

    return out_final
