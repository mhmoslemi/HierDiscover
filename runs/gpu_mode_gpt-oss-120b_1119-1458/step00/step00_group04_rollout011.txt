Define kernel signature:

```
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # Compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # K offset
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        # Load A and B with masking
        a_ptrs = A_ptr + pid_batch * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = B_ptr + pid_batch * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        a = tl.load(a_ptrs, mask=(offs_m < M) & (offs_k < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K) & (offs_n < N), other=0.0)
        # Multiply-accumulate
        acc += tl.dot(a, b)
    # Write out
    c_ptrs = C_ptr + pid_batch * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m < M) & (offs_n < N))
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if present
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).type_as(left) # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Prepare for batched matmul
    B, N, _, _ = left.shape  # left shape [B,N,N, hidden_dim]
    # permute and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, hidden_dim, N, N]
    total_batch = B * hidden_dim
    M = N
    K = N
    N_out = N

    left_flat = left_perm.view(total_batch, M, K).to(torch.float16)
    right_flat = right_perm.view(total_batch, K, N_out).to(torch.float16)

    out_flat = torch.empty((total_batch, M, N_out), dtype=torch.float16, device=device)

    # compute strides
    stride_a0, stride_a1, stride_a2 = left_flat.stride()
    stride_b0, stride_b1, stride_b2 = right_flat.stride()
    stride_c0, stride_c1, stride_c2 = out_flat.stride()

    # launch kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (total_batch,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N_out + BLOCK_N - 1) // BLOCK_N)

    batched_matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        M, N_out, K,
        stride_a0, stride_a1, stride_a2,
        stride_b0, stride_b1, stride_b2,
        stride_c0, stride_c1, stride_c2,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # Synchronize (launch is async; but we can rely on torch synchronization later)

    # reshape out
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # shape [B,N,N,hidden_dim] (fp16)
    out = out.type_as(x)  # convert to float32
    # post layernorm & gating & final linear
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

```
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # create mask for boundary
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # compute pointer offsets
        a_ptr = A_ptr + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
        b_ptr = B_ptr + pid_batch * stride_b_batch + offs_k[:, None] * stride_b_row + offs_n[None, :] * stride_b_col

        # load a and b with mask
        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float16)
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float16)

        # accumulate
        acc += tl.dot(a, b)

    # write result
    c_ptr = C_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    tl.store(c_ptr, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass implemented with a custom Triton batched GEMM.
The heavy triple‐summation 
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] 
is expressed as a batch of matrix multiplications:
    (i,k) @ (k,j) → (i,j) for every (b,d) pair.
A Triton kernel computes these B·hidden_dim independent GEMMs efficiently.
All other operations (LayerNorm, linear projections, gating, and final
projection) are done with PyTorch (float32) while the GEMM runs in float32
inside the kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batch of (M×K) @ (K×N) → (M×N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    M, N, K,                      # matrix dimensions
    stride_a_batch, stride_a_m, stride_a_k,   # strides for A (batch, row, col)
    stride_b_batch, stride_b_k, stride_b_n,   # strides for B (batch, row, col)
    stride_c_batch, stride_c_m, stride_c_n,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes C = A @ B where A: (batch, M, K), B: (batch, K, N).
    """

    pid_batch = tl.program_id(0)               # which (b,d) pair
    pid_m = tl.program_id(1)                   # block row
    pid_n = tl.program_id(2)                   # block column

    # --------------------------------------------------------------
    # Block offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in tiles of size BLOCK_K
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointer arithmetic
        a_ptrs = (
            A_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        b_ptrs = (
            B_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )

        # Load with masking
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write results
    # --------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    # Unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # Model hyper‑parameters
    dim = config["dim"]          # input / output channel dimension
    hidden_dim = config["hidden_dim"]   # projection dimension
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Normalisation and linear projections (float32)
    # ------------------------------------------------------------------
    # LayerNorm on the input (dim‑last)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 2) Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape: [B, N, N] → [B, N, N, 1]
        mask_exp = mask.unsqueeze(-1).type_as(left)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 3) Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4) Prepare tensors for the batched GEMM
    #    left  : (B, N, N, H) → (B, H, N, N) → (B*H, N, N)
    #    right : (B, N, N, H) → (B, H, N, N) with (k,j) order → (B*H, N, N)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Permute + make contiguous so that we obtain dense (batch, H, N, N) tensors
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)  = (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # (B, H, N, N)  = (k,j)

    # Collapse the first two dimensions: total_batch = B * H
    total_batch = B * hidden_dim
    left_mat = left_perm.view(total_batch, N, N)               # (total_batch, M=N, K=N)
    right_mat = right_perm.view(total_batch, N, N)            # (total_batch, K=N, N=N)

    # Output buffer
    out_mat = torch.empty_like(left_mat)                       # (total_batch, N, N)

    # ------------------------------------------------------------------
    # 5) Launch Triton batched GEMM kernel
    # ------------------------------------------------------------------
    # Block sizes – tuned for H100 (fp32 matmul)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Compute strides (in elements, not bytes)
    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_mat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()

    grid = (
        total_batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,           # M
        N,           # N
        N,           # K
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,        # good default for fp32 on H100
    )
    # Ensure kernel has finished before proceeding
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 6) Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7) Post‑processing: LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # broadcasted gate
    out = F.linear(out, weights["to_out.weight"])   # back to dim

    return out
