def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape

    device = input_tensor.device

    # Get config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # config may have dim (c?), hidden_dim (c_z?) but follow.

    # Compute LayerNorm: because we can't easily compute mean and variance across dim using Triton (they said not needed), we can use torch.nn.functional.layer_norm with weight/bias.

    # Use torch.layer_norm directly:
    # x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # but we must have weight shape (dim).

    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # shape (B,N,N,dim)

    # Linear projections:
    # left_proj = torch.einsum('bnkd,hd->bnkh', x_norm, left_proj_weight) maybe use torch.nn.functional.linear

    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # left_gate, right_gate, out_gate:
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # Apply mask if exists. mask param shape (B,N,N). 
    # config maybe contains "nomask". If no mask, mask tensor can be None or a boolean. We'll default to all ones if mask is None or config['nomask'] is True.

    if mask is not None and mask.numel() > 0:
        mask_expanded = mask.unsqueeze(-1)  # shape B,N,N,1
        left = left * mask_expanded
        right = right * mask_expanded

    # Apply gates:
    left = left * left_gate
    right = right * right_gate

    # Now compute out_ = einsum('...i k d, ...j k d -> ... i j d')
    # Use Triton kernel.
    # Input tensors left,right shape (B,N,N,hidden_dim). We'll ensure they are contiguous in channel dimension? They are contiguous normally.

    out = tri_mul_outgoing_kernel(left, right)

        # will define tri_mul_outgoing_kernel that returns output tensor shape (B,N,N,hidden_dim)

    # Post-processing
    out_normed = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out_normed = out_normed * out_gate  # elementwise, broadcast along hidden dim (i.e., shape (B,N,N, hidden_dim))
    # final linear:

def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    # Program ID
    pid_b = tl.program_id(2)  # third axis for batch
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # compute start offsets for i, j
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    # Determine valid range using mask if i < N, j < N
    i_range = i_start + tl.arange(0, BLOCK_I)
    j_range = j_start + tl.arange(0, BLOCK_J)

    # compute pointer base for left and right
    # left indices: [b, i, k, h]; we need to iterate over k, d
    # out index: [b, i, j, h]

    # Initialize accumulator across hidden dims: shape (BLOCK_I, BLOCK_J, BLOCK_H)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # Loop over k
    for k in range(0, N, BLOCK_K):
        # compute k range
        k_range = k + tl.arange(0, BLOCK_K)
        # mask if k >= N
        k_mask = k_range < N

        # Load left tile: shape (BLOCK_I, BLOCK_K, BLOCK_H)
        # left_ptr_offset = b * stride_l_b + i_range[:,None,None] * stride_l_i + k_range[None,:,None] * stride_l_k + h_range[None,None,:] * stride_l_h

            k_range[None, :, None] * stride_r_k + \
            h_range[None, None, :] * stride_r_h

@triton.jit
def tri_mul_outgoing_kernel_impl(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr
):
    # Program IDs
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_b = tl.program_id(2)

    # Calculate start indices
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    # Range indices
    i_idx = i_start + tl.arange(0, BLOCK_I)
    j_idx = j_start + tl.arange(0, BLOCK_J)

    # Bounds
    i_mask = i_idx < N
    j_mask = j_idx < N

    # Global pointer for output (base)
    out_ptrs_base = out_ptr + pid_b * stride_o_b

    # Loop over hidden dimension blocks
    for h_start in range(0, H, BLOCK_H):
        h_idx = h_start + tl.arange(0, BLOCK_H)
        h_mask = h_idx < H

        # accumulator
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

        # Loop over k dimension
        for k_start in range(0, N, BLOCK_K):
            k_idx = k_start + tl.arange(0, BLOCK_K)
            k_mask = k_idx < N

            # Create mask for loading left and right
            # left: shape (I, K, H)
            # broadcast mask to shape (I, K, H)
            mask_left = i_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :]

            # Pointers for left
            left_ptrs = left_ptr + pid_b * stride_l_b + \
                i_idx[:, None, None] * stride_l_i + \
                k_idx[None, :, None] * stride_l_k + \
                h_idx[None, None, :] * stride_l_h
            left_tile = tl.load(left_ptrs, mask=mask_left, other=0.0).to(tl.float32)

            # right transposed: shape (K, J, H)
            mask_right = j_mask[None, :, None] & k_mask[:, None, None] & h_mask[None, None, :]

            right_ptrs = right_ptr + pid_b * stride_r_b + \
                k_idx[:, None, None] * stride_r_k + \
                j_idx[None, :, None] * stride_r_i + \
                h_idx[None, None, :] * stride_r_h
            right_tile_T = tl.load(right_ptrs, mask=mask_right, other=0.0).to(tl.float32)

            # Compute contribution
            # left_tile: (I,K,H), right_tile_T: (K,J,H)
            # Compute out for each hidden channel using dot
            for dh in range(0, BLOCK_H):
                # Only compute if h_mask[dh] true (? this helps avoid wasted compute)
                # Could compute unconditional but internal mask can be zero
                left_h = left_tile[:, :, dh]   # (I,K)
                right_h_T = right_tile_T[:, :, dh]  # (K,J)
                # dot
                out_h = tl.dot(left_h, right_h_T)  # (I,J)
                acc[:, :, dh] += out_h

        # Write accumulator to output
        out_ptrs = out_ptrs_base + \
            i_idx[:, None, None] * stride_o_i + \
            j_idx[None, :, None] * stride_o_j + \
            h_idx[None, None, :] * stride_o_h
        # mask for storing: combination of i_mask, j_mask, h_mask
        mask_store = i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :]
        tl.store(out_ptrs, acc, mask=mask_store)

"""
TriMul (outgoing) – fast forward implementation.

The heavy part of the module is the ternary multiplication:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We first apply LayerNorm, linear projections and sigmoid gates in
regular PyTorch (these are cheap compared to the N³ work).  The
core einsum is off‑loaded to a fused Triton kernel that computes a
blocked matrix‑multiply for each hidden dimension slice:

    left_slice  : (BLOCK_I, BLOCK_K)
    right_slice : (BLOCK_K, BLOCK_J)   (right is loaded transposed)
    out_slice   : (BLOCK_I, BLOCK_J)

The kernel iterates over the sequence dimension K in tiles and over
the hidden dimension in blocks (BLOCK_H).  All arithmetic is performed
in float32; the final result is stored as float32 and the remaining
LayerNorm/linear steps are handled by PyTorch.

This implementation works for any batch size (1‑2), sequence length
up to 1024 and hidden dimensions up to 128, handling optional masks.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: outgoing TriMul (sum over k)
# ----------------------------------------------------------------------
@triton.jit
def triton_tri_mul_outgoing_kernel(
    left_ptr,                     # *f32   [B, N, N, H]
    right_ptr,                    # *f32   [B, N, N, H]
    out_ptr,                      # *f32   [B, N, N, H]
    B, N, H,                      # meta-params
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    # --------------------------------------------------------------
    # program IDs
    # --------------------------------------------------------------
    pid_i = tl.program_id(0)               # block index over output row   (i)
    pid_j = tl.program_id(1)               # block index over output col   (j)
    pid_b = tl.program_id(2)               # batch index

    # --------------------------------------------------------------
    # index ranges for this block
    # --------------------------------------------------------------
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    i_idx = i_start + tl.arange(0, BLOCK_I)
    j_idx = j_start + tl.arange(0, BLOCK_J)

    i_mask = i_idx < N
    j_mask = j_idx < N

    # ------------------------------------------------------------------
    # loop over hidden dimension in chunks of BLOCK_H
    # ------------------------------------------------------------------
    h_offset = 0
    while h_offset < H:
        h_idx = h_offset + tl.arange(0, BLOCK_H)
        h_mask = h_idx < H

        # accumulator for this (i,j) tile and hidden chunk
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

        # ------------------------------------------------------------------
        # loop over the reduction dimension k
        # ------------------------------------------------------------------
        k_offset = 0
        while k_offset < N:
            k_idx = k_offset + tl.arange(0, BLOCK_K)
            k_mask = k_idx < N

            # --------------------------
            # left tile  : (I, K, H)
            # --------------------------
            mask_left = i_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :]
            left_ptrs = (
                left_ptr
                + pid_b * stride_l_b
                + i_idx[:, None, None] * stride_l_i
                + k_idx[None, :, None] * stride_l_k
                + h_idx[None, None, :] * stride_l_h
            )
            left_tile = tl.load(left_ptrs, mask=mask_left, other=0.0).to(tl.float32)

            # --------------------------
            # right tile (transposed) : (K, J, H)
            #   we treat rows = k, cols = j
            # --------------------------
            mask_right = j_mask[None, :, None] & k_mask[:, None, None] & h_mask[None, None, :]
            right_ptrs = (
                right_ptr
                + pid_b * stride_r_b
                + k_idx[:, None, None] * stride_r_k   # k as first dimension
                + j_idx[None, :, None] * stride_r_i   # j as second dimension
                + h_idx[None, None, :] * stride_r_h
            )
            right_tile_T = tl.load(right_ptrs, mask=mask_right, other=0.0).to(tl.float32)

            # --------------------------
            # compute dot products per hidden slice
            # --------------------------
            for dh in range(0, BLOCK_H):
                # slice views (2‑D tensors)
                left_h  = left_tile[:, :, dh]      # (I, K)
                right_h = right_tile_T[:, :, dh]   # (K, J)
                # matrix multiplication (I,K) @ (K,J) -> (I,J)
                out_h = tl.dot(left_h, right_h)    # (I, J)
                acc[:, :, dh] += out_h

            k_offset += BLOCK_K

        # --------------------------------------------------------------
        # store the accumulated block
        # --------------------------------------------------------------
        out_ptrs = (
            out_ptr
            + pid_b * stride_o_b
            + i_idx[:, None, None] * stride_o_i
            + j_idx[None, :, None] * stride_o_j
            + h_idx[None, None, :] * stride_o_h
        )
        mask_store = i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :]
        tl.store(out_ptrs, acc, mask=mask_store)

        h_offset += BLOCK_H


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def tri_mul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right: [B, N, N, H]   (float32)
    returns out with same shape.
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # ensure contiguous for predictable strides
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # strides in element counts (as required by Triton)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tunable block sizes – they fit comfortably in the H100 register file
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 8                       # 8 * 32 * 32 = 8192 float32 registers ≈ 32 KB

    grid = (
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
        B,
    )
    triton_tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    # ------------------------------------------------------------------
    # configuration
    # ------------------------------------------------------------------
    dim = config["dim"]               # input & output channel dimension
    hidden_dim = config["hidden_dim"] # inner (c_z) dimension

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # -> [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Apply mask if present (mask shape [B,N,N])
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        # expand to [B,N,N,1] then broadcast over hidden_dim
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # [B,N,N,hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core tri‑mul (outgoing) – fused Triton kernel
    # ------------------------------------------------------------------
    out = tri_mul_outgoing_triton(left, right)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm over hidden_dim, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # back to [B,N,N,dim]

    return out
