@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program_id(0) -> i block
    # program_id(1) -> j block
    # program_id(2) -> slice id (b*H + h)
    pid_i = tl.program_id(axis=0)
    pid_j = tl.program_id(axis=1)
    pid_s = tl.program_id(axis=2)  # slice index
    # compute batch and hidden idx
    b = pid_s // H
    h = pid_s % H

    # compute row and col offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # limit within N
    mask_i = i_offset < N
    mask_j = j_offset < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dim
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offset
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask for K
        mask_k = k_offset < N

        # compute pointers
        # left: [b,i,k,h]
        left_ptrs = left_ptr + b * stride_l_b + h * stride_l_h + i_offset[:, None]*stride_l_i + k_offset[None, :]*stride_l_k
        # right: [b,j,k,h]  Note: second dimension is j
        right_ptrs = right_ptr + b * stride_r_b + h * stride_r_h + j_offset[:, None]*stride_r_j + k_offset[None, :]*stride_r_k

        # load tiles
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        right_tile = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # dot product
        acc += tl.dot(left_tile, right_tile)  # results dtype float32 after accumulation?

        # Note: tl.dot returns same dtype as input? If left_tile dtype is fp16 and accumulator is fp32, we can cast.

    # store result: out shape [b,i,j,h] but using out_strides accordingly
    out_ptrs = out_ptr + b * stride_o_b + h * stride_o_h + i_offset[:, None]*stride_o_i + j_offset[None, :]*stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = torch.float16  # compute in FP16 for performance

    # configs
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm
    norm_weight = weights["norm.weight"].to(dtype).to(device)
    norm_bias = weights["norm.bias"].to(dtype).to(device)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)
    x = x.to(dtype)

    # linear proj weights
    left_proj_w = weights["left_proj.weight"].to(dtype).to(device)
    right_proj_w = weights["right_proj.weight"].to(dtype).to(device)
    left_gate_w = weights["left_gate.weight"].to(dtype).to(device)
    right_gate_w = weights["right_gate.weight"].to(dtype).to(device)

def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_s = tl.program_id(2)

    b = pid_s // H
    h = pid_s % H

    # offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointers
        left_ptrs = left_ptr \
            + b * stride_l_b \
            + h * stride_l_h \
            + offs_i[:, None] * stride_l_i \
            + offs_k[None, :] * stride_l_k

        right_ptrs = right_ptr \
            + b * stride_r_b \
            + h * stride_r_h \
            + offs_j[:, None] * stride_r_j \
            + offs_k[None, :] * stride_r_k

        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        right_tile = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        + b * stride_o_b \
        + h * stride_o_h \
        + offs_i[:, None] * stride_o_i \
        + offs_j[None, :] * stride_o_j

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The core of the operator is the 4‑dimensional einsum
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
which can be expressed as a batched matrix multiplication
    out[b, h] = left[b, h] @ right[b, h].T .
We fuse this O(N³) computation in a custom Triton kernel that
processes tiles of the (i, j) output plane while iterating over the
summation dimension k.  All surrounding operations (LayerNorm,
linear projections, sigmoid gates) are performed with PyTorch in
FP16 for bandwidth‑limited stages; the final result is cast back to
FP32.

Supported shapes:
    input : [B, N, N, D]   (B∈{1,2}, N∈{128,256,512,1024}, D∈{128,384})
    hidden_dim : H (normally 128)

The function expects a tuple (input, mask, weights, config) and returns
the processed tensor of shape [B, N, N, D] (FP32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""

    pid_i = tl.program_id(0)                # block index over i
    pid_j = tl.program_id(1)                # block index over j
    pid_s = tl.program_id(2)                # combined batch * head index

    # decode batch and head (hidden) dimension from pid_s
    b = pid_s // H
    h = pid_s % H

    # offsets within the N dimension
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the summation dimension k
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, i, k, h]  (shape M x K)
        left_ptrs = left_ptr \
            + b * stride_l_b \
            + h * stride_l_h \
            + offs_i[:, None] * stride_l_i \
            + offs_k[None, :] * stride_l_k
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)                # fp16

        # right[b, j, k, h] (shape N x K)
        right_ptrs = right_ptr \
            + b * stride_r_b \
            + h * stride_r_h \
            + offs_j[:, None] * stride_r_j \
            + offs_k[None, :] * stride_r_k
        right_tile = tl.load(right_ptrs,
                             mask=mask_j[:, None] & mask_k[None, :],
                             other=0.0)                # fp16

        # promote to FP32 before the dot to avoid overflow in fp16
        acc += tl.dot(left_tile.to(tl.float32),
                      right_tile.to(tl.float32))

    # store the result (cast back to fp16)
    out_ptrs = out_ptr \
        + b * stride_o_b \
        + h * stride_o_h \
        + offs_i[:, None] * stride_o_i \
        + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def _trimul_einsum_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches the Triton kernel.
    left / right : [B, N, N, H] (FP16)
    returns out  : same shape and dtype (FP16)
    """
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in element units
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # tile sizes – tuned for H100 (FP16)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑blocks
        triton.cdiv(N, BLOCK_N),          # j‑blocks
        B * H                              # batch * head dimension
    )

    _trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel for the
    O(N³) pairwise multiplication.  All other layers (LayerNorm, Linear,
    sigmoid) are executed with PyTorch in FP16; the final output is FP32.
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype_fp16 = torch.float16

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension of the pairwise tensor
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(dtype_fp16).to(device)
    norm_b = weights["norm.bias"].to(dtype_fp16).to(device)

    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_w,
                     bias=norm_b,
                     eps=1e-5)
    x = x.to(dtype_fp16)   # work in FP16 from here on

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(dtype_fp16).to(device)
    right_proj_w = weights["right_proj.weight"].to(dtype_fp16).to(device)

    left = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # Optional mask (binary 0/1) – broadcast over the hidden dim
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(dtype_fp16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating pathways (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(dtype_fp16).to(device)
    right_gate_w = weights["right_gate.weight"].to(dtype_fp16).to(device)
    out_gate_w = weights["out_gate.weight"].to(dtype_fp16).to(device)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core TriMul operation – fused batched GEMM via Triton
    # ------------------------------------------------------------------
    out = _trimul_einsum_triton(left, right)     # [B, N, N, hidden_dim] FP16

    # ------------------------------------------------------------------
    # Output post‑processing
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype_fp16).to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype_fp16).to(device)

    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)

    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(dtype_fp16).to(device)
    out = F.linear(out, to_out_w)          # [B, N, N, dim] FP16

    # Cast back to FP32 for the model’s expected precision
    out = out.to(torch.float32)
    return out
