@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_l_i, stride_l_k, stride_l_h,
    stride_r_i, stride_r_k, stride_r_h,
    stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    using block-sparse GEMM, accumulating in float32.
    """
    # Compute block coordinates
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # offsets for i, j
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    i = i_base + tl.arange(0, BLOCK_M)
    j = j_base + tl.arange(0, BLOCK_N)

    # mask out-of-range indices
    mask_i = i < N
    mask_j = j < N

    # Initialize accumulator for each hidden tile later
    # We'll loop over hidden blocks inside.
    # accumulate per hidden block and store after loops.
    out_ptrs = out_ptr + i[:, None] * stride_o_i + j[None, :] * stride_o_j  # shape (M,N)

    # for each hidden block
    for hb in range(0, H, BLOCK_H):
        # compute hidden indices
        h = hb + tl.arange(0, BLOCK_H)
        mask_h = h < H

        # accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # iterate over k dimension
        for kk in range(0, N, BLOCK_K):
            # compute k indices
            k = kk + tl.arange(0, BLOCK_K)
            mask_k = k < N

            # Load left tile: shape (BLOCK_M, BLOCK_K)
            # left_ptr at (i, k, h)
            left_off = i[:, None] * stride_l_i + k[None, :] * stride_l_k + h[None, None] * stride_l_h
            left = tl.load(left_ptr + left_off, mask=mask_i[:, None] & mask_k[None, :] & mask_h[None, None], other=0.0)
            # Cast to float32
            left = left.to(tl.float32)

            # Load right tile: shape (BLOCK_N, BLOCK_K) where we load (j, k, h)
            right_off = j[:, None] * stride_r_i + k[None, :] * stride_r_k + h[None, None] * stride_r_h
            right = tl.load(right_ptr + right_off, mask=mask_j[:, None] & mask_k[None, :] & mask_h[None, None], other=0.0)
            right = right.to(tl.float32)

            # Compute dot product (M,N) accumulate: left (M,K) dot right (N,K) transpose
            # Use tl.dot with trans_b=True => (M,K) @ (K,N) -> (M,N)
            acc += tl.dot(left, right, trans_b=True)

        # store accumulator to out: need to write to out_ptr with proper hidden offset
        out_off = out_ptrs + h[None, None] * stride_o_h  # broadcast to M,N
        tl.store(out_off, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :] & mask_h[None, None])

def trimul_outgoing_kernel(left_ptr, right_ptr, out_ptr,
                           N, H,
                           stride_l_i, stride_l_k, stride_l_h,
                           stride_r_i, stride_r_k, stride_r_h,

                           BLOCK_N: tl.constexpr,
                           BLOCK_K: tl.constexpr,
                           BLOCK_H: tl.constexpr):
    """ Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h].
        left and right are of shape (N, N, H) for a single batch.
        Only the major compute (K-loop) is accelerated.
    """
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    i = i_base + tl.arange(0, BLOCK_M)
    j = j_base + tl.arange(0, BLOCK_N)
    # bounds
    mask_i = i < N
    mask_j = j < N

    # iterate over hidden blocks
    for hb in range(0, H, BLOCK_H):
        h = hb + tl.arange(0, BLOCK_H)
        mask_h = h < H

        # accumulator in fp32
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # loop over K dimension
        for kk in range(0, N, BLOCK_K):
            k = kk + tl.arange(0, BLOCK_K)
            mask_k = k < N

            # left tile (M,K)
            left_off = i[:, None] * stride_l_i + k[None, :] * stride_l_k + h[None, None] * stride_l_h
            left = tl.load(left_ptr + left_off,
                           mask=mask_i[:, None] & mask_k[None, :] & mask_h[None, None],
                           other=0.0).to(tl.float32)

            # right tile (N,K)
            right_off = j[:, None] * stride_r_i + k[None, :] * stride_r_k + h[None, None] * stride_r_h
            right = tl.load(right_ptr + right_off,
                            mask=mask_j[:, None] & mask_k[None, :] & mask_h[None, None],
                            other=0.0).to(tl.float32)

            # GEMM: (M,K) @ (K,N)^T -> (M,N)
            acc += tl.dot(left, right, trans_b=True)

        out_off = i[:, None] * stride_o_i + j[None, :] * stride_o_j + h[None, None] * stride_o_h
        # Convert acc back to FP16 for storage
        tl.store(out_ptr + out_off,

def custom_kernel(data):
    """
    Triton implementation of the outgoing TriMul operation.
    Takes a tuple (input_tensor, mask, weights, config) and returns the processed tensor.
    The heavy pairwise multiplication is performed by a custom Triton kernel.
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    # Determine dimensions
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Cast to fp16 for speed (except for final output)
    # Keep a copy of original dtype for final casting
    orig_dtype = input_tensor.dtype
    compute_dtype = torch.float16 if input_tensor.is_floating_point() and input_tensor.dtype != torch.float64 else torch.float32
    # We'll cast everything to compute_dtype (float16 is typical)
    x = input_tensor.to(compute_dtype)

    # LayerNorm over dim
    norm_weight = weights["norm.weight"].to(compute_dtype)
    norm_bias = weights["norm.bias"].to(compute_dtype)

    # Compute mean and variance across last dim
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # Linear projections
    left_proj_weight = weights["left_proj.weight"].to(compute_dtype)
    right_proj_weight = weights["right_proj.weight"].to(compute_dtype)
    left = F.linear(x_norm, left_proj_weight)  # shape (bs, N, N, hidden_dim)
    right = F.linear(x_norm, right_proj_weight)

    # Apply mask if provided
    if mask is not None:
        # mask shape (bs, N, N) -> expand
        mask_exp = mask.to(compute_dtype).unsqueeze(-1)  # (bs, N, N, 1)
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate_weight = weights["left_gate.weight"].to(compute_dtype)
    right_gate_weight = weights["right_gate.weight"].to(compute_dtype)
    out_gate_weight = weights["out_gate.weight"].to(compute_dtype)

    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Now compute the heavy einsum using Triton.
    # Allocate output buffer
    out = torch.empty((bs, seq_len, seq_len, hidden_dim), dtype=compute_dtype, device=x.device)

    # Determine stride layout for a single batch view (contiguous)
    # Ensure left/right/out are contiguous
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # Strides for each dimension (i, k, h)
    stride_l_i = left.stride(1)  # stride for dimension i (first N)
    stride_l_k = left.stride(2)  # stride for second N (k)
    stride_l_h = left.stride(3)  # stride for hidden dim

    stride_r_i = right.stride(1)
    stride_r_k = right.stride(2)
    stride_r_h = right.stride(3)

    stride_o_i = out.stride(1)
    stride_o_j = out.stride(2)
    stride_o_h = out.stride(3)

    # Kernel block sizes (tuned)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_H = 16

    # Launch kernel for each batch
    for b in range(bs):
        left_ptr = left[b].data_ptr()
        right_ptr = right[b].data_ptr()
        out_ptr = out[b].data_ptr()
        grid = (triton.cdiv(seq_len, BLOCK_M), triton.cdiv(seq_len, BLOCK_N))
        trimul_outgoing_kernel[grid](
            left_ptr, right_ptr, out_ptr,
            seq_len, hidden_dim,
            stride_l_i, stride_l_k, stride_l_h,
            stride_r_i, stride_r_k, stride_r_h,
            stride_o_i, stride_o_j, stride_o_h,
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
        )
    # Ensure kernel finished
    torch.cuda.synchronize()

    # Post-processing: to_out_norm + gating + linear to dim
    # LayerNorm over hidden_dim
    to_out_norm_weight = weights["to_out_norm.weight"].to(compute_dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(compute_dtype)

    # compute mean and var over hidden_dim dim=-1
    out_mean = out.mean(dim=-1, keepdim=True)
    out_var = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - out_mean) / torch.sqrt(out_var + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # Multiply by out_gate (same shape)
    out_norm = out_norm * out_gate

    # Final linear
    to_out_weight = weights["to_out.weight"].to(compute_dtype)
    # shape (dim, hidden_dim)
    final = F.linear(out_norm, to_out_weight)  # (bs, N, N, dim)

    # Cast back to original dtype (float32 for evaluation)
    final = final.to(torch.float32)

    return final

"""
custom_kernel for the outgoing TriMul operation (AlphaFold3).

Algorithm
---------
1. Layer‑normalize the input tensor over the channel dimension (dim).
2. Project the normalized tensor to a hidden dimension with two linear layers
   (left_proj and right_proj) and apply an optional mask.
3. Compute three gate vectors (left_gate, right_gate, out_gate) with a sigmoid
   activation and multiply the projected tensors by the corresponding gates.
4. **Heavy step** – compute
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   for every batch, pair (i,j) and hidden channel h.
   This is a batched matrix‑multiply for each hidden channel:
        out_h = left_h @ right_h.T
   The computation is performed by a fused Triton kernel that loops over the
   sequence dimension `k` in tiles and accumulates in float32.  The kernel is
   launched with a 3‑D grid (i‑tile, j‑tile, hidden‑channel) so that each
   program instance handles one hidden channel.
5. Apply a second LayerNorm over the hidden dimension, multiply by `out_gate`,
   and project back to the original channel dimension with a final linear layer.
6. Return the result as float32.

The Triton kernel fuses the O(N³·H) workload and avoids materialising
the large intermediate tensor that a naïve `torch.einsum` would create.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,           # const * left  [N, N, H]   (single batch slice)
    right_ptr,          # const * right [N, N, H]
    out_ptr,            #      * out   [N, N, H]
    N,                  # sequence length (int)
    stride_l_i, stride_l_k, stride_l_h,   # strides for left
    stride_r_i, stride_r_k, stride_r_h,   # strides for right
    stride_o_i, stride_o_j, stride_o_h,   # strides for out
    BLOCK_M: tl.constexpr,   # tile size in i dimension
    BLOCK_N: tl.constexpr,   # tile size in j dimension
    BLOCK_K: tl.constexpr    # tile size in k dimension
):
    """Compute a single hidden‑channel matrix product using tiled GEMM.

    Each program instance (pid_i, pid_j, pid_h) computes:
        out[i0:i0+M, j0:j0+N, h] = Σ_k left[i0:i0+M, k, h] @ right[j0:j0+N, k, h].T
    """
    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile
    pid_h = tl.program_id(2)          # hidden channel index

    # Global offsets for this tile
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N
    h = pid_h                          # scalar hidden index

    # Index vectors (might be out‑of‑bounds, masked later)
    i = i_base + tl.arange(0, BLOCK_M)
    j = j_base + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    for kk in range(0, N, BLOCK_K):
        k = kk + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # --------------------------------------------------------------
        # Load left[i, k, h]  ->  shape (M, K)
        # --------------------------------------------------------------
        left_off = i[:, None] * stride_l_i + k[None, :] * stride_l_k + h * stride_l_h
        left = tl.load(
            left_ptr + left_off,
            mask=mask_i[:, None] & mask_k[None, :],
            other=tl.float16(0.0)
        )
        left = left.to(tl.float32)

        # --------------------------------------------------------------
        # Load right[j, k, h] -> shape (N, K)
        # --------------------------------------------------------------
        right_off = j[:, None] * stride_r_i + k[None, :] * stride_r_k + h * stride_r_h
        right = tl.load(
            right_ptr + right_off,
            mask=mask_j[:, None] & mask_k[None, :],
            other=tl.float16(0.0)
        )
        right = right.to(tl.float32)

        # --------------------------------------------------------------
        # GEMM: (M,K) @ (K,N).T  ->  (M,N)
        # --------------------------------------------------------------
        acc += tl.dot(left, right, trans_b=True)

    # --------------------------------------------------------------
    # Write the result back (store as FP16 to save memory)
    # --------------------------------------------------------------
    out_off = i[:, None] * stride_o_i + j[None, :] * stride_o_j + h * stride_o_h
    tl.store(
        out_ptr + out_off,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :]
    )


# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated implementation of the outgoing TriMul module.
    Input:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Output:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    # --------------------------------------------------------------
    # Basic shapes / config
    # --------------------------------------------------------------
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # Choose compute dtype (FP16 for speed, keep FP32 for final output)
    # --------------------------------------------------------------
    compute_dtype = torch.float16 if input_tensor.dtype != torch.float64 else torch.float32
    x = input_tensor.to(compute_dtype)

    # --------------------------------------------------------------
    # ----- LayerNorm over `dim` ------------------------------------
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(compute_dtype)
    norm_b = weights["norm.bias"].to(compute_dtype)

    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_w + norm_b                     # (bs, N, N, dim)

    # --------------------------------------------------------------
    # Linear projections (no bias)
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(compute_dtype)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(compute_dtype)

    left = F.linear(x_norm, left_proj_w)      # (bs, N, N, hidden_dim)
    right = F.linear(x_norm, right_proj_w)

    # --------------------------------------------------------------
    # Optional mask (mask shape: [bs, N, N])
    # --------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(compute_dtype).unsqueeze(-1)   # (bs, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # Gates (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(compute_dtype)
    right_gate_w = weights["right_gate.weight"].to(compute_dtype)
    out_gate_w = weights["out_gate.weight"].to(compute_dtype)

    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))   # (bs, N, N, hidden_dim)

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # ---- Heavy pairwise multiplication via Triton -----------------
    # --------------------------------------------------------------
    # Allocate output buffer
    out = torch.empty((bs, seq_len, seq_len, hidden_dim),
                      dtype=compute_dtype,
                      device=x.device)

    # Make sure tensors are contiguous for simple stride handling
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # Tile sizes (tuned for H100, FP16)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32   # 32‑wide tiles give good occupancy on H100

    # Launch one kernel per batch slice
    for b in range(bs):
        left_block = left[b]      # shape (N, N, hidden_dim)
        right_block = right[b]
        out_block = out[b]

        # Strides for a single batch slice (int64)
        stride_l_i = left_block.stride(0)
        stride_l_k = left_block.stride(1)
        stride_l_h = left_block.stride(2)

        stride_r_i = right_block.stride(0)
        stride_r_k = right_block.stride(1)
        stride_r_h = right_block.stride(2)

        stride_o_i = out_block.stride(0)
        stride_o_j = out_block.stride(1)
        stride_o_h = out_block.stride(2)

        # Grid: (tiles_i, tiles_j, hidden_dim)
        grid = (
            triton.cdiv(seq_len, BLOCK_M),
            triton.cdiv(seq_len, BLOCK_N),
            hidden_dim,
        )

        trimul_outgoing_kernel[grid](
            left_block.data_ptr(),
            right_block.data_ptr(),
            out_block.data_ptr(),
            seq_len,
            stride_l_i, stride_l_k, stride_l_h,
            stride_r_i, stride_r_k, stride_r_h,
            stride_o_i, stride_o_j, stride_o_h,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )
    # Ensure kernels have finished before moving on
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # ----- Second LayerNorm over hidden dimension -------------------
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(compute_dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(compute_dtype)

    out_mean = out.mean(dim=-1, keepdim=True)
    out_var = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - out_mean) / torch.sqrt(out_var + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    # --------------------------------------------------------------
    # ----- Apply output gate and final projection -------------------
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                     # (bs, N, N, hidden_dim)

    to_out_w = weights["to_out.weight"].to(compute_dtype)   # (dim, hidden_dim)
    final = F.linear(out_norm, to_out_w)               # (bs, N, N, dim)

    # Cast back to float32 for the model output
    return final.to(torch.float32)
