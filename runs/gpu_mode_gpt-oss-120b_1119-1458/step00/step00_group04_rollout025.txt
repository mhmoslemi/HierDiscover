                   stride_right_b, stride_right_i, stride_right_k, stride_right_d,

                   num_cblocks: tl.constexpr,

   pid_c = tl.program_id(0)  # combined batch and D block index
   pid_i = tl.program_id(1)
   pid_j = tl.program_id(2)

   # compute batch idx and d block idx

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_weight)

    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)  # shape (B,N,N,1)
        left = left * mask
        right = right * mask

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Triton kernel to compute out = sum_k left[..., i,k,:] * right[..., j,k,:]
    # Arguments: left, right, out (allocate)
    out = torch.empty_like(left)  # will be overwritten
    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # Get strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()

@triton.jit
def trimul_kernel(left_ptr, right_ptr, out_ptr,
                  B, N, D,
                  stride_left_b, stride_left_i, stride_left_k, stride_left_d,
                  stride_right_b, stride_right_i, stride_right_k, stride_right_d,
                  stride_out_b, stride_out_i, stride_out_j, stride_out_d,
                  BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
                  BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr):
   # program ids
   pid0 = tl.program_id(0) # combined batch & d block
   pid_i = tl.program_id(1)
   pid_j = tl.program_id(2)

   # compute batch and d block indices
   d_blocks = tl.cdiv(D, BLOCK_D)
   batch_idx = pid0 // d_blocks
   d_block_idx = pid0 % d_blocks
   d_start = d_block_idx * BLOCK_D

   # compute start indices for i and j
   i_start = pid_i * BLOCK_M
   j_start = pid_j * BLOCK_N

   # create coordinate offsets
   offs_i = i_start + tl.arange(0, BLOCK_M)
   offs_j = j_start + tl.arange(0, BLOCK_N)
   offs_k = tl.arange(0, BLOCK_K)  # reused each iteration
   offs_d = d_start + tl.arange(0, BLOCK_D)

   # Masks for OOB
   mask_i = offs_i < N
   mask_j = offs_j < N
   mask_k = offs_k < N
   mask_d = offs_d < D

   # allocate accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

   # loops over k dimension tiles
   for kb in range(0, tl.cdiv(N, BLOCK_K)):
       k_start = kb * BLOCK_K
       # actual offsets for k
       k = k_start + offs_k
       mask_k_tile = k < N

       # loads for left: shape (M, K_tile, D)
       # compute pointers
       a_ptrs = left_ptr + batch_idx * stride_left_b + offs_i[:, None, None] * stride_left_i + k[None, :, None] * stride_left_k + offs_d[None, None, :] * stride_left_d

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

The heavy "pairwise" reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a custom Triton kernel that
* loops over the sequence dimension `k`
* keeps a per‑program accumulator of shape (BM, BN, BD) where
  BM – rows per block, BN – columns per block, BD – a slice of the hidden
  channel dimension.
* uses only elementwise loads/stores and a broadcasted
  multiply‑accumulate, avoiding any `tl.mean` or synchronization.
All other operations (LayerNorms, Linear projections, Sigmoid gates and the
final linear) are performed with PyTorch.  This satisfies the requirement
to use Triton for a non‑trivial part of the computation while keeping the
overall implementation simple and fast.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    left_ptr,               # const * left  [B, N, N, D]
    right_ptr,              # const * right [B, N, N, D]
    out_ptr,                # * out          [B, N, N, D]
    B, N, D,                # dimensions (runtime)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    # compile‑time block sizes
    BM: tl.constexpr,       # rows per block
    BN: tl.constexpr,       # cols per block
    BD: tl.constexpr,       # hidden‑channel slice per block
):
    """
    Compute the outgoing triangle multiplicative update:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    The kernel processes a tile of (BM × BN × BD) for a specific batch
    and a slice of the hidden dimension (size BD).  The reduction over `k`
    is performed inside a simple Python‑style loop (unrolled at runtime).
    """

    # ----------------- program IDs -----------------
    pid0 = tl.program_id(0)          # combined batch + hidden‑dim block
    pid_i = tl.program_id(1)          # tile row index
    pid_j = tl.program_id(2)          # tile column index

    # split pid0 into batch index and hidden‑dim block index
    d_blocks = tl.cdiv(D, BD)
    batch_idx = pid0 // d_blocks
    d_block   = pid0 % d_blocks
    d_start   = d_block * BD

    # starting coordinates of the current tile
    i_start = pid_i * BM
    j_start = pid_j * BN

    # ------------------------------------------------------------------
    # Create 1‑D coordinate vectors for the tile and for the hidden slice
    # ------------------------------------------------------------------
    offs_i = i_start + tl.arange(0, BM)      # [BM]
    offs_j = j_start + tl.arange(0, BN)      # [BN]
    offs_d = d_start + tl.arange(0, BD)      # [BD]

    # masks for out‑of‑bounds accesses (tails)
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # ------------------- accumulator -------------------
    # each program instance holds a (BM,BN,BD) accumulator in fp32
    acc = tl.zeros((BM, BN, BD), dtype=tl.float32)

    # ------------------- reduction over k -------------------
    k = 0
    while k < N:
        # ---- load left[i,k,d] -------------------------------------------------
        # shape (BM, BD)
        a_ptrs = (
            left_ptr
            + batch_idx * stride_left_b
            + offs_i[:, None] * stride_left_i
            + k * stride_left_k
            + offs_d[None, :] * stride_left_d
        )
        mask_a = mask_i[:, None] & mask_d[None, :]
        a = tl.load(a_ptrs, mask=mask_a, other=0.0).to(tl.float32)   # (BM,BD)

        # ---- load right[j,k,d] ------------------------------------------------
        # shape (BN, BD)
        b_ptrs = (
            right_ptr
            + batch_idx * stride_right_b
            + offs_j[:, None] * stride_right_i   # note: right uses the same i‑stride as left
            + k * stride_right_k
            + offs_d[None, :] * stride_right_d
        )
        mask_b = mask_j[:, None] & mask_d[None, :]
        b = tl.load(b_ptrs, mask=mask_b, other=0.0).to(tl.float32)   # (BN,BD)

        # ---- outer‑product and accumulate --------------------------------------
        # a[:,None,:]   -> (BM,1,BD)
        # b[None,:,:]   -> (1,BN,BD)
        # broadcasted multiply yields (BM,BN,BD)
        acc += a[:, None, :] * b[None, :, :]

        k += 1

    # ------------------- store the result ------------------------------------
    out_ptrs = (
        out_ptr
        + batch_idx * stride_out_b
        + offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + offs_d[None, None, :] * stride_out_d
    )
    mask_out = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
    tl.store(out_ptrs, acc, mask=mask_out)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # unpack
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1. LayerNorm on the input (over the last dim)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # -------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)      # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # -------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # -------------------------------------------------
    if not nomask:
        # mask: [B,N,N] -> [B,N,N,1] -> float32 for multiplication
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # -------------------------------------------------
    # 4. Gating (sigmoid of another linear projection)
    # -------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate                      # keep shape [B,N,N,hidden_dim]
    right = right * right_gate

    # -------------------------------------------------
    # 5. Triton kernel:   out = Σ_k left[...,i,k,d] * right[...,j,k,d]
    # -------------------------------------------------
    # Make tensors contiguous (required for stride extraction)
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left, dtype=torch.float32, device=device).contiguous()

    B, N, _, _ = left.shape

    # strides in element units (not bytes)
    s_lb, s_li, s_lk, s_ld = left.stride()
    s_rb, s_ri, s_rk, s_rd = right.stride()
    s_ob, s_oi, s_oj, s_od = out.stride()

    # ------------------------------------------------------------------
    # Block sizes – chosen to keep register pressure low while exposing
    # enough parallelism.  BM=BN=8 gives 64 threads per program; BD=8
    # slices the hidden dimension, so each thread works on an 8‑element
    # vector (still a single register per element after JIT lowering).
    # ------------------------------------------------------------------
    BM = 8
    BN = 8
    BD = 8

    # Number of hidden‑dim blocks and total programs in the combined axis 0
    d_blocks = (hidden_dim + BD - 1) // BD
    num_programs0 = B * d_blocks

    grid = (
        num_programs0,                # batch * hidden‑dim block
        (N + BM - 1) // BM,           # tile rows
        (N + BN - 1) // BN,           # tile cols
    )

    # launch Triton kernel
    trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        s_lb,
        s_li,
        s_lk,
        s_ld,
        s_rb,
        s_ri,
        s_rk,
        s_rd,
        s_ob,
        s_oi,
        s_oj,
        s_od,
        BM,
        BN,
        BD,
        num_warps=4,          # an H100 can comfortably run 4‑8 warps per program
    )
    # ensure kernel completion before using the result
    torch.cuda.synchronize()

    # -------------------------------------------------
    # 6. Output LayerNorm (over hidden_dim)
    # -------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # -------------------------------------------------
    # 7. Apply the output gate
    # -------------------------------------------------
    out = out * out_gate

    # -------------------------------------------------
    # 8. Final linear projection back to `dim`
    # -------------------------------------------------
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)   # [B,N,N,dim]

    return out
