def tri_mul_kernel(out_ptr, left_ptr, right_ptr,
                   B, N, H,
                   stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                   stride_left_b, stride_left_i, stride_left_k, stride_left_h,
                   stride_right_b, stride_right_j, stride_right_k, stride_right_h,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    b = pid // H
    h = pid % H

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for i,j out-of-range
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Coordinate offsets for out
    out_ptrs = out_ptr + b*stride_out_b + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j + h*stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile (i, k)
        left_ptrs = left_ptr + b*stride_left_b + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k + h*stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right tile transposed (k, j), load as (k, j)
        right_ptrs = right_ptr + b*stride_right_b + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j + h*stride_right_h
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Multiply and accumulate: left (M,K) * right (K,N) -> (M,N)
        acc += tl.dot(left, right)

    # Apply mask at out positions (just for i,j)
    # But output should be float32. Write
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward pass implemented with Triton.

The heavy O(N³·C) contraction
    out[b,i,j,c] = Σₖ left[b,i,k,c] * right[b,j,k,c]
is performed by a custom Triton kernel that computes,
for every (batch, hidden) slice, a batched matrix multiplication
    Left(b,c)  ∈ ℝ^{N×N}   (i‑k)
    Right(b,c) ∈ ℝ^{N×N}   (j‑k)
as  out = Left @ Rightᵀ .
All other operations (LayerNorm, Linear projections, gating,
final LayerNorm + Linear) are executed with PyTorch for simplicity.

The kernel tiles the (i, j) output plane with BLOCK_M×BLOCK_N
tiles and iterates over k in BLOCK_K steps, using
`tl.dot` (Tensor‑core friendly) and accumulating in float32.
Masks (if present) are applied before the kernel so the kernel
doesn’t need to handle them.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the core contraction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    out_ptr, left_ptr, right_ptr,      # pointers
    B, N, H,                           # sizes
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Grid: (B*H, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    """
    pid0 = tl.program_id(0)        # combined batch/hidden index
    pid_i = tl.program_id(1)       # tile row (i)
    pid_j = tl.program_id(2)       # tile col (j)

    # ------------------------------------------------------------------
    # decode batch & hidden indices
    # ------------------------------------------------------------------
    b = pid0 // H
    h = pid0 % H

    # ------------------------------------------------------------------
    # offsets for the current tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers to the output tile (B,N,N,H)
    out_tile_ptr = (
        out_ptr
        + b * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + h * stride_out_h
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k dimension
    # ------------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left:  (i,k) tile        shape (BLOCK_M, BLOCK_K)
        left_tile_ptr = (
            left_ptr
            + b * stride_left_b
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + h * stride_left_h
        )
        left = tl.load(left_tile_ptr,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right: (k,j) tile – we load the transposed view
        # original layout is (j,k); we want (k,j) for the dot
        right_tile_ptr = (
            right_ptr
            + b * stride_right_b
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j
            + h * stride_right_h
        )
        right = tl.load(right_tile_ptr,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # dot: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        # left/right are fp16, tl.dot accumulates in fp32
        acc += tl.dot(left, right)

    # ------------------------------------------------------------------
    # Write the result
    # ------------------------------------------------------------------
    tl.store(out_tile_ptr, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask         : torch.Tensor [B, N, N]      (bool/float)
        - weights      : dict of torch.Tensor containing:
            * "norm.weight", "norm.bias"
            * "left_proj.weight", "right_proj.weight"
            * "left_gate.weight", "right_gate.weight", "out_gate.weight"
            * "to_out_norm.weight", "to_out_norm.bias"
            * "to_out.weight"
        - config       : dict with at least:
            * "dim"         – input channel dimension
            * "hidden_dim" – internal projection dimension
            * "nomask"      – bool, True ⇒ ignore mask
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # LayerNorm on the input (over the last dimension)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    # These functions treat the final dimension as the feature dimension.
    left_proj = F.linear(x, weights["left_proj.weight"])      # [B,N,N,hidden]
    right_proj = F.linear(x, weights["right_proj.weight"])    # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # Apply mask if required (mask shape: [B,N,N])
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # ensure mask is float32 and broadcastable
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B,N,N,1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # Gating mechanisms
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()   # [B,N,N,hidden]
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left_proj * left_gate   # [B,N,N,hidden]
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # Cast to fp16 for the heavy contraction (Tensor‑cores friendly)
    # ------------------------------------------------------------------
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    B, N, _, _ = left.shape

    # Output placeholder (float32)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # ------------------------------------------------------------------
    # Prepare strides for Triton launch
    # ------------------------------------------------------------------
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()

    # Tile sizes – chosen to respect the 1024‑thread limit per program.
    # 128×128 tile uses 256 threads (each thread handles a 8×8 sub‑tile via tl.dot).
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # good for fp16 Tensor‑cores

    # Grid: (B*hidden, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # ------------------------------------------------------------------
    # Launch Triton kernel
    # ------------------------------------------------------------------
    _tri_mul_kernel[grid](
        out,
        left,
        right,
        B,
        N,
        hidden_dim,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_j,
        stride_right_k,
        stride_right_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # H100 works well with 4‑warp blocks
    )
    # ``out`` is now [B,N,N,hidden_dim] (float32)

    # ------------------------------------------------------------------
    # Final LayerNorm + gating + projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                               # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])      # -> [B,N,N,dim]

    return out
