@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,  # for a
    stride_bk, stride_bn, stride_bb, # for b
    stride_cm, stride_cn, stride_cb, # for c
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # k offset loop
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        # compute pointers with k dimension increment
        a_ptrs = a_ptr + batch * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        # load
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)
        # accumulate
        acc += tl.dot(a, b)
    # store result
    c_ptrs = c_ptr + batch * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])

def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

                    other=0.0)

                    mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
                    other=0.0)

        a = a.to(tl.float32)

    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]

                                            normalized_shape=(dim,),

    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))

        mask = mask.to(left.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    left = left * left_gate
    right = right * right_gate

    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)
    
    # Reorder for batch matmul: shape (B, hidden_dim, N, N) and (B, hidden_dim, N, N) transposed on last two dims for right.

"""
TriMul (outgoing) forward pass implemented with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor over the feature dimension (`dim`).
2. Linear projections:
   * left  = Linear(x, left_proj_weight)
   * right = Linear(x, right_proj_weight)
3. Gating (sigmoid):
   * left_gate  = Sigmoid(Linear(x, left_gate_weight))
   * right_gate = Sigmoid(Linear(x, right_gate_weight))
   * out_gate   = Sigmoid(Linear(x, out_gate_weight))
4. If a mask is supplied, multiply `left` and `right` by the mask
   (broadcasted over the hidden dimension).  Then apply the gates:
   `left  = left  * left_gate`
   `right = right * right_gate`.
5. Compute the cubic‑interaction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is exactly a batched matrix multiplication
   `out = left @ rightᵀ` for every hidden channel `d`.
   The heavy N³ work is performed in FP16 by a fused Triton kernel.
6. Reshape back to `[B,N,N,hidden_dim]` and convert to FP32.
7. Apply a second Layer‑norm (`to_out_norm`), multiply by `out_gate`,
   and finish with a linear projection `to_out` (hidden_dim → dim).

Only the N³ matmul is done in Triton; the remaining operations use
standard PyTorch kernels (which are already highly optimized).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for many (batch * hidden_dim) matrices
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,   # A dimensions: (batch, M, K)
    stride_bk, stride_bn, stride_bb,   # B dimensions: (batch, K, N)
    stride_cm, stride_cn, stride_cb,   # C dimensions: (batch, M, N)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m   = tl.program_id(1)
    pid_n   = tl.program_id(2)

    # Coordinates of the block we are working on
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Pointers with proper strides and batch offset
        a_ptrs = a_ptr + batch_id * stride_ab + \
                 offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch_id * stride_bb + \
                 offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load tiles, masking out‑of‑bounds elements
        a = tl.load(a_ptrs,
                    mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
                    other=0.0)

        # Promote to FP32, perform the dot product and accumulate
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)

    # Write results back (cast to the output dtype – half)
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + batch_id * stride_cb + \
             offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             c,
             mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])


def _batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    a: (B, M, K)   – FP16
    b: (B, K, N)   – FP16
    returns (B, M, N) – FP16
    """
    assert a.shape[0] == b.shape[0]
    batch, M, K = a.shape
    _, _, N = b.shape

    # Output allocation
    c = torch.empty((batch, M, N), dtype=a.dtype, device=a.device)

    # Strides (in elements, not bytes)
    stride_ab, stride_am, stride_ak = a.stride()
    stride_bb, stride_bk, stride_bn = b.stride()
    stride_cb, stride_cm, stride_cn = c.stride()

    # Tunable tile sizes – work well on H100 for the examined shapes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        a, b, c,
        M, N, K,
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return c


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [batch, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # `nomask` indicates whether a mask should be applied
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input layer‑norm (over the last dimension)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape [B, N, N, dim] – float32

    # ------------------------------------------------------------------
    # 2) Linear projections (dim → hidden_dim)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)    # [B, N, N, H]
    right = F.linear(x, right_proj_weight) # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3) Gating (sigmoid) – also dim → hidden_dim
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 4) Apply optional mask and the element‑wise gates
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask: [B, N, N] → broadcast over hidden_dim
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) N³ multiplication via a batched GEMM in FP16
    # ------------------------------------------------------------------
    # Cast to half‑precision for the heavy matmul
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Re‑order to (batch, hidden, N, N) and flatten (batch*hidden) → (B*H, N, N)
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_h.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] (K is now the first spatial axis)

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(-1, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(-1, N, N) # (B*H, N, N)

    # Core computation: out_flat = left_flat @ right_flat (k‑summation)
    out_flat = _batched_matmul(left_flat, right_flat)  # FP16 result

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6) Second layer‑norm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    # Convert to FP32 for the remaining (low‑cost) ops
    out = out.to(torch.float32)

    # Layer‑norm over the hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # Multiply by the output gate (broadcasted)
    out = out * out_gate

    # Final projection hidden_dim → dim
    out = F.linear(out, to_out_weight)  # bias is absent in the reference implementation

    return out
