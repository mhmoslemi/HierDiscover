                         stride_bb, stride_bh,

    pid_n = tl.program_id(1)
    pid_s = tl.program_id(2)

    b_ptr = B_ptr + batch_id * stride_bb + hidden_id * stride_bh
    c_ptr = C_ptr + batch_id * stride_cb + hidden_id * stride_ch

"""
TriMul (outgoing) forward pass with Triton‑accelerated pairwise contraction.

The module performs:
1. LayerNorm on the input tensor (last dim = dim).
2. Two linear projections (left / right) → shape [B, N, N, H].
3. Optional elementwise masking.
4. Three sigmoidal gates (left, right, out) computed from the normalized input.
5. Elementwise gating of left/right.
6. Core contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication L_h @ R_hᵀ for each hidden channel h.
   This O(N³) step is executed by a custom Triton kernel.
7. LayerNorm over the hidden dimension, multiplication by out_gate, and
   a final linear projection back to dim.

All operations except the contraction are performed with PyTorch for simplicity,
while the heavy O(N³) contraction is fused in Triton for speed.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    A_ptr,                     # left  tensor,  shape [B, H, N, N]
    B_ptr,                     # right tensor, shape [B, H, N, N]
    C_ptr,                     # output tensor, shape [B, H, N, N]
    N,                         # sequence length
    hidden_dim,                # number of hidden channels (H)
    batch_size,                # batch size (B)
    stride_ab, stride_ah,      # strides for batch and hidden dimensions of A
    stride_am, stride_ak,      # strides for M (=i) and K (=k) dimensions of A
    stride_bb, stride_bh,      # strides for batch and hidden dimensions of B
    stride_bm, stride_bk,      # strides for M (=j) and K (=k) dimensions of B
    stride_cb, stride_ch,      # strides for batch and hidden dimensions of C
    stride_cm, stride_cn,      # strides for M (=i) and N (=j) dimensions of C
    BLOCK_M: tl.constexpr,     # tile size in i dimension
    BLOCK_N: tl.constexpr,     # tile size in j dimension
    BLOCK_K: tl.constexpr,     # tile size in k dimension
):
    """Compute out = left @ rightᵀ for each (batch, hidden) slice."""
    pid_m = tl.program_id(0)        # i‑tile
    pid_n = tl.program_id(1)        # j‑tile
    pid_s = tl.program_id(2)        # flattened (batch, hidden) slice

    # ----------------------------------------------------------------------
    # Resolve batch index b and hidden index h from the flattened slice id.
    # ----------------------------------------------------------------------
    b = pid_s // hidden_dim
    h = pid_s % hidden_dim

    # Base pointers for this slice (add batch & hidden offsets)
    a_base = A_ptr + b * stride_ab + h * stride_ah
    b_base = B_ptr + b * stride_bb + h * stride_bh
    c_base = C_ptr + b * stride_cb + h * stride_ch

    # ----------------------------------------------------------------------
    # Compute tile coordinates in the i (M) and j (N) dimensions.
    # ----------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to the start of the current tiles (k‑dimension will be iterated)
    a_ptrs = a_base + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_base + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bm

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------------------------------------------------------
    # Loop over the reduction dimension k.
    # ----------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # Load tiles with proper masking (out‑of‑bounds loads → zero)
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Accumulate the partial matrix product
        acc += tl.dot(a, b)

        # Advance pointers to the next k‑block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ----------------------------------------------------------------------
    # Write the result tile back to C.
    # ----------------------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with the model parameters (see code)
            - config: dict with at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, same device as input)
    """
    # ------------------------------------------------------------------
    # Unpack inputs and basic shapes
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # usually 128
    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (left / right)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask = mask.to(dtype=dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoids of linear transforms of the normalized input)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction using Triton: out = left @ rightᵀ per hidden channel
    # ------------------------------------------------------------------
    # Rearrange to [B, H, N, N] for the kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    out_perm = torch.empty_like(left_perm)  # will hold [B, H, N, N]

    # Tile sizes – tuned for H100 (FP32, good occupancy)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        bs * hidden_dim,
    )

    _tri_mul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        N,
        hidden_dim,
        bs,
        # strides for A (left)
        left_perm.stride(0), left_perm.stride(1),    # batch, hidden
        left_perm.stride(2), left_perm.stride(3),    # i (M), k
        # strides for B (right)
        right_perm.stride(0), right_perm.stride(1),  # batch, hidden
        right_perm.stride(2), right_perm.stride(3),  # j (M), k
        # strides for C (output)
        out_perm.stride(0), out_perm.stride(1),      # batch, hidden
        out_perm.stride(2), out_perm.stride(3),      # i (M), j (N)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Convert back to [B, N, N, H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm → gating → final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
