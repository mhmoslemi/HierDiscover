def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data

    bs, N, _, dim = input_tensor.shape

    device = input_tensor.device
    dtype = input_tensor.dtype

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5

    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # shape [bs, N, N, hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # mask
    if not config.get("nomask", False):
        mask = mask_tensor.to(x_norm.dtype).unsqueeze(-1)  # [bs,N,N,1]
        left = left * mask
        right = right * mask
    else:
        # No mask: skip

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_half = left.to(torch.float16).contiguous()
    right_half = right.to(torch.float16).contiguous()
    # Output placeholder half, then cast back
    out_half = torch.empty((bs, N, N, config["hidden_dim"]), dtype=torch.float16, device=device)

    # Strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_half.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right_half.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_half.stride()

    # Determine block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Launch kernel
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        bs * config["hidden_dim"]
    )
    trimul_kernel[grid](
        left_ptr=left_half,
        right_ptr=right_half,
        out_ptr=out_half,
        B=bs,
        N=N,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert to float32
    out = out_half.to(torch.float32)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(out, (config["hidden_dim"],), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Multiply by out_gate
    out_norm = out_norm * out_gate

    # Final linear
    to_out_weight = weights["to_out.weight"]
    out_final = torch.nn.functional.linear(out_norm, to_out_weight)  # shape [bs,N,N,dim]

    return out_final

"""
TriMul (outgoing) forward pass with a fused Triton kernel.
---------------------------------------------------------

The TriMul operator works on a 4‑D tensor [B, N, N, C] (C = dim).
The forward consists of:

1. Layer‑norm over the last dimension.
2. Linear projections → left/right (shape [B,N,N,H]) where H = hidden_dim.
3. Optional mask multiplication.
4. Gating (sigmoid(linear(...))) and element‑wise gating of left/right.
5. Core triangular multiplicative update:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]      (Eq. 1)
   This is exactly a batched matrix‑multiply per hidden channel:
       out_h = left_h @ right_hᵀ   (B×H independent matmuls).
   We implement Eq. 1 with a custom Triton kernel that:
   • treats each (batch, hidden) pair as a separate program (grid‑z).
   • tiles the i‑ and j‑dimensions (BLOCK_M × BLOCK_N) and iterates
     over k tiles (BLOCK_K) using Tensor‑Core friendly fp16 loads.
   • accumulates in fp32 for accuracy.
6. Layer‑norm over the hidden dimension, element‑wise multiplication by the
   output gate, and a final linear projection back to dim.

All heavy arithmetic (step 5) runs on the GPU via Triton; the remaining
operations are performed with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: one independent matmul per (batch, hidden) pair.
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides for left tensor [B, N, N, H]
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right tensor [B, N, N, H] (j‑dim is second axis)
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    # strides for output tensor [B, N, N, H]
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matmul for the TriMul core:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    """
    pid_i = tl.program_id(0)  # tile index over i dimension
    pid_j = tl.program_id(1)  # tile index over j dimension
    pid_z = tl.program_id(2)  # combined batch * hidden index

    # ------------------------------------------------------------------
    # Decode batch and hidden indices from pid_z
    # ------------------------------------------------------------------
    hidden_idx = pid_z % H               # which hidden channel
    batch_idx  = pid_z // H               # which batch

    # ------------------------------------------------------------------
    # Starting offsets for this tile
    # ------------------------------------------------------------------
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    k0 = 0  # will be updated inside the loop

    # Base pointer that already includes batch and hidden offsets
    base_left  = left_ptr \
               + batch_idx * stride_left_b \
               + hidden_idx * stride_left_h
    base_right = right_ptr \
               + batch_idx * stride_right_b \
               + hidden_idx * stride_right_h
    base_out   = out_ptr \
               + batch_idx * stride_out_b \
               + hidden_idx * stride_out_h

    # ------------------------------------------------------------------
    # Accumulator (fp32) for the tile
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # Global k offset for this tile
        k_offset = k * BLOCK_K

        # ------------------------------------------------------------------
        # Compute pointers for the current k‑tile
        #   left  : shape (BLOCK_M, BLOCK_K)   – rows i, cols k
        #   right : shape (BLOCK_K, BLOCK_N)   – rows k, cols j
        # ------------------------------------------------------------------
        i_idx = tl.arange(0, BLOCK_M)[:, None]          # (M,1)
        k_idx = tl.arange(0, BLOCK_K)[None, :]          # (1,K)
        j_idx = tl.arange(0, BLOCK_N)[None, :]          # (1,N)

        # Masks for out‑of‑bounds loads
        mask_a = (i0 + i_idx) < N & (k_offset + k_idx) < N
        mask_b = (j0 + j_idx) < N & (k_offset + k_idx.T) < N

        # Pointers for left (i,k)
        a_ptrs = base_left \
               + (i0 + i_idx) * stride_left_i \
               + (k_offset + k_idx) * stride_left_k
        # Load fp16 and apply mask
        a = tl.load(a_ptrs, mask=mask_a, other=0.0).to(tl.float16)

        # Pointers for right (k,j) – note the transposition w.r.t. the
        # original shape, we load with k as the leading dimension.
        b_ptrs = base_right \
               + (k_offset + k_idx) * stride_right_k \
               + (j0 + j_idx) * stride_right_j
        b = tl.load(b_ptrs, mask=mask_b, other=0.0).to(tl.float16)

        # Tensor‑core matmul: (M, K) @ (K, N) → (M, N)
        acc += tl.dot(a, b)   # acc stays fp32

    # ------------------------------------------------------------------
    # Write results back to memory (fp32 → fp16 stored as fp32 for now)
    # ------------------------------------------------------------------
    i_range = i0 + tl.arange(0, BLOCK_M)[:, None]
    j_range = j0 + tl.arange(0, BLOCK_N)[None, :]

    mask_store = (i_range < N) & (j_range < N)

    out_ptrs = base_out \
            + i_range * stride_out_i \
            + j_range * stride_out_j

    # Store in fp32 (the downstream PyTorch code will cast to float32 anyway)
    tl.store(out_ptrs, acc, mask=mask_store)

# ----------------------------------------------------------------------
# Custom entry point used by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.
    Arguments:
        data = (input, mask, weights, config)
        input   : torch.Tensor [B, N, N, dim]
        mask    : torch.Tensor [B, N, N] (may be all‑ones when config["nomask"] is True)
        weights : dict of weight tensors (see body for required keys)
        config  : dict with keys "dim", "hidden_dim", "nomask"
    Returns:
        torch.Tensor [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    B, N, _, _ = input_tensor.shape
    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])    # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask multiplication (mask broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core triangular multiplicative update (Triton kernel)
    #    Convert to fp16 before the kernel to exploit Tensor‑Cores.
    # ------------------------------------------------------------------
    left_h  = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    out_h = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Strides (in elements, not bytes)
    sL_b, sL_i, sL_k, sL_h = left_h.stride()
    sR_b, sR_j, sR_k, sR_h = right_h.stride()
    sO_b, sO_i, sO_j, sO_h = out_h.stride()

    # Tile sizes – chosen to fit H100 Tensor‑Core registers
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,               # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,               # j‑tiles
        B * hidden_dim                              # (batch, hidden) programs
    )

    trimul_kernel[grid](
        left_h, right_h, out_h,
        B, N, hidden_dim,
        sL_b, sL_i, sL_k, sL_h,
        sR_b, sR_j, sR_k, sR_h,
        sO_b, sO_i, sO_j, sO_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Convert back to fp32 for the remaining FP‑32 pipeline
    out = out_h.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Post‑processing: layer‑norm over hidden dim, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])   # final projection to dim

    return out
