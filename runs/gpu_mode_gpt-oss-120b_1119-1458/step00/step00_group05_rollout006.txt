@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    batch = pid_bh // H
    head = pid_bh % H

    # Compute global offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for i and j in bounds
    mask_i = i_offset < N
    mask_j = j_offset < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for kb in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load left tile (i,k)
        left_ptrs = left_ptr + batch * stride_left_batch \
                   + i_offset[:, None] * stride_left_i \
                   + k_offset[None, :] * stride_left_k \
                   + head * stride_left_h
        left = tl.load(left_ptrs, mask = mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile (j,k) => we need right[j, k] but we treat as B_T tile (k,j)
        right_ptrs = right_ptr + batch * stride_right_batch \
                    + j_offset[None, :] * stride_right_i \
                    + k_offset[:, None] * stride_right_k \
                    + head * stride_right_h
        # Note shape (BLOCK_K, BLOCK_N)
        right = tl.load(right_ptrs, mask = mask_k[:, None] & mask_j[None, :], other=0.0)

        # Compute dot product
        acc += tl.dot(left, right)

    # Write output
    out_ptrs = out_ptr + batch * stride_out_batch \
               + i_offset[:, None] * stride_out_i \
               + j_offset[None, :] * stride_out_j \
               + head * stride_out_h
    tl.store(out_ptrs, acc, mask = mask_i[:, None] & mask_j[None, :])
```

"""
custom_kernel: fast forward implementation of the “outgoing” TriMul operator.

Algorithm
---------
1. Layer‐norm the input tensor (shape [B,N,N,D]).
2. Linear projection to the hidden dimension H (no bias) → left, right.
3. Optional mask (broadcasted on the hidden dimension) is applied.
4. Compute three sigmoid gates (left‑gate, right‑gate, out‑gate) from the
   normalized input and multiply them into left, right and the later output.
5. Heavy part – the cubic “einsum”:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed as a batched GEMM (left @ rightᵀ) in a fused Triton kernel.
   The kernel processes a tile of size BLOCK_M×BLOCK_N for each (batch,head)
   pair, iterating over K in blocks of size BLOCK_K.
6. Layer‑norm the intermediate result over the hidden dimension, apply the
   out‑gate, then a final linear projection back to D.
All steps except the GEMM are done with regular PyTorch ops; the Triton
kernel removes the O(N³) Python loops and lets the H100 execute the matrix
multiplications with tensor‑core speed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul cubic term
# ---------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    # pointers
    left_ptr,          # float32*  [B, N, N, H]
    right_ptr,         # float32*  [B, N, N, H]
    out_ptr,           # float32*  [B, N, N, H]
    # problem size
    B, N, H,
    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Computes, for each batch b and hidden index h:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    as a tiled GEMM using tensor‑core friendly blocks.
    """
    pid_i = tl.program_id(0)               # block row (i)
    pid_j = tl.program_id(1)               # block col (j)
    pid_bh = tl.program_id(2)              # combined batch * hidden index

    # decode batch and hidden index from the third dimension
    batch = pid_bh // H
    head  = pid_bh % H

    # -----------------------------------------------------------------
    # Offsets for the output tile (i, j)
    # -----------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over the reduction dimension k
    # -----------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_base = kb * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------------------------------------------------------------
        # Load left tile  : shape (BLOCK_M, BLOCK_K)
        #   left[batch, i, k, head]
        # -------------------------------------------------------------
        left_ptrs = left_ptr \
            + batch * stride_left_b \
            + head  * stride_left_h \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # -------------------------------------------------------------
        # Load right tile : shape (BLOCK_K, BLOCK_N)
        #   right[batch, j, k, head]   -> we need the transpose (k,j)
        # -------------------------------------------------------------
        right_ptrs = right_ptr \
            + batch * stride_right_b \
            + head  * stride_right_h \
            + offs_j[None, :] * stride_right_i \
            + offs_k[:, None] * stride_right_k
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # -------------------------------------------------------------
        # Multiply‑accumulate
        # -------------------------------------------------------------
        acc += tl.dot(left, right)   # (M,K)·(K,N) → (M,N)

    # -----------------------------------------------------------------
    # Store the result
    # -----------------------------------------------------------------
    out_ptrs = out_ptr \
        + batch * stride_out_b \
        + head  * stride_out_h \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ---------------------------------------------------------------
# Entry point expected by the evaluator
# ---------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor   [B, N, N, D]
        - mask_tensor  : torch.Tensor   [B, N, N]  (may be all‑ones)
        - weights_dict : mapping of weight names → torch.Tensor
        - config_dict  : contains at least "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32, on the same device).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]          # hidden dimension (often 128)

    # -----------------------------------------------------------------
    # 1. Input layer‑norm
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     (D,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]   # (H, D)
    right_proj_w = weights["right_proj.weight"] # (H, D)

    left  = F.linear(x, left_proj_w)   # [B,N,N,H]
    right = F.linear(x, right_proj_w)  # [B,N,N,H]

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dimension)
    # -----------------------------------------------------------------
    if mask_tensor is not None and not config.get("nomask", False):
        # mask is float32/float16 after conversion
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections of the normalized input)
    # -----------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # [B,N,N,H]

    left  = left  * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Batched GEMM: out = left @ rightᵀ   (the O(N³) part)
    # -----------------------------------------------------------------
    # Output buffer (float32) – the kernel accumulates in FP32.
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # Triton expects strides in element‐units.
    stride_left_b = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_h = left.stride(3)

    stride_right_b = right.stride(0)
    stride_right_i = right.stride(1)   # this dimension corresponds to the “j” index
    stride_right_k = right.stride(2)
    stride_right_h = right.stride(3)

    stride_out_b = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)

    # Tile sizes – chosen to fit well on H100 tensor‑cores.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks over i dimension
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks over j dimension
        B * H                           # one kernel instance per (batch, hidden)
    )

    trimul_gemm_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,                     # good default for H100
    )

    # -----------------------------------------------------------------
    # 6. Output layer‑norm, gating, final linear projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out,
                            (H,),
                            weight=to_out_norm_w,
                            bias=to_out_norm_b,
                            eps=1e-5)

    out_norm = out_norm * out_gate                # apply the output gate

    to_out_w = weights["to_out.weight"]           # (D, H)
    output = F.linear(out_norm, to_out_w)         # final projection → [B,N,N,D]

    # Output is already float32 on the correct device
    return output
```

**Explanation of key design choices**

* **LayerNorm & gating in PyTorch** – These operations are memory‑bound and cheap compared with the cubic term; keeping them in native PyTorch lets us reuse highly‑optimized kernels and avoids extra Triton bookkeeping.

* **Batched GEMM kernel** – By flattening the `(batch, hidden)` axes into a single launch dimension (`pid_bh`) we launch `B·H` independent matrix multiplications of size `N×N`. The kernel tiles the output with `BLOCK_M×BLOCK_N` blocks and iterates over `k` in `BLOCK_K`‑sized chunks, using `tl.dot` (tensor‑core friendly) and fused masking for out‑of‑bounds indices.

* **Stride‑aware indexing** – The kernel works for any contiguous layout (the tensors are produced by PyTorch’s default contiguous layout) and respects the true stride values, so it supports the four‑dimensional layout `[B, N, N, H]` without extra transposes.

* **Mixed‑precision friendliness** – The kernel reads the inputs as `float32` (the safest choice for the heavy‑tailed Cauchy distribution) and accumulates in `float32`. If memory pressure becomes an issue, one can switch `left` / `right` to `float16` and change `dtype=tl.float16` in the load calls; the dot product will still accumulate in FP32 on H100.

* **Scalability** – The grid size automatically adapts to the sequence length (`N`) and hidden dimension (`H`). Even for the largest test case (`N=1024`, `B=1`, `H=128`) the kernel launches 128 × 1 = 128 independent GEMMs, each tiled, which fits comfortably on an H100.

The function `custom_kernel` conforms to the required signature and can be used directly as the entry point for the evaluation harness.
