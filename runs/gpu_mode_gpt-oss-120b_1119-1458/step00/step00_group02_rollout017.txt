def trimul_einsum(left, right):
    B,N,N,H = left.shape
    assert right.shape == (B,N,N,H)
    out = torch.empty_like(left)  # same shape dtype float32? We'll allocate same dtype.
    # Cast to float16 for speed
    left_flat = left.to(torch.float16).contiguous()
    right_flat = right.to(torch.float16).contiguous()
    out_flat = out.to(torch.float16).contiguous()
    # compute strides
    left_ptr = left_flat
    right_ptr = right_flat
    out_ptr = out_flat

    # get strides as Python ints: left.stride() returns tuple of longs; we need multiply by element size? Triton expects strides in elements, not bytes.
    ls = left_flat.stride()  # (stride_b, stride_i, stride_k, stride_d)
    rs = right_flat.stride()
    os = out_flat.stride()

    # Ensure contiguous so strides will be correct.

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B*H)
    trimul_mul_kernel[grid](
        left_ptr, right_ptr, out_ptr,
        B, N, H,
        ls[0], ls[1], ls[2], ls[3],
        rs[0], rs[1], rs[2], rs[3],
        os[0], os[1], os[2], os[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # Convert out back to float32
    out = out_flat.to(torch.float32)
    return out

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def trimul_mul_kernel(left_ptr, right_ptr, out_ptr,
                      B, N, H,
                      stride_bl, stride_il, stride_ik, stride_id,
                      stride_br, stride_jr, stride_kr, stride_rd,
                      stride_bo, stride_io, stride_jo, stride_od,
                      BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # program ids
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_bc = tl.program_id(2) # batch*channel

    # derive batch and hidden dim index
    b = pid_bc // H
    d = pid_bc % H

    # compute start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for current batch and channel
    left_base = left_ptr + b * stride_bl + d * stride_id
    right_base = right_ptr + b * stride_br + d * stride_rd
    out_base = out_ptr + b * stride_bo + d * stride_od

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k_start in range(0, N, BLOCK_K):
        cur_k = tl.minimum(BLOCK_K, N - k_start)
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile (M x K)
        left_ptrs = left_base + (offs_m[:, None] * stride_il) + (offs_k[None, :] * stride_ik)
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)  # dtype inferred as fp16

        # load right tile transposed (K x N)   # B is row, J is column after transpose
        right_ptrs = right_base + (offs_k[:, None] * stride_kr) + (offs_n[None, :] * stride_jr)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)

        # dot: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_base + (offs_m[:, None] * stride_io) + (offs_n[None, :] * stride_jo)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def trimul_einsum(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the TriMul multiplicative update:
    out[i,j,d] = sum_k left[i,k,d] * right[j,k,d]
    Implemented as a batched GEMM via a Triton kernel.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    # output tensor (float32)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # Ensure contiguity
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # strides in elements
    l_strides = left.stride()
    r_strides = right.stride()
    o_strides = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            B * H)

    trimul_mul_kernel[grid](
        left, right, out,
        B, N, H,
        l_strides[0], l_strides[1], l_strides[2], l_strides[3],
        r_strides[0], r_strides[1], r_strides[2], r_strides[3],
        o_strides[0], o_strides[1], o_strides[2], o_strides[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    return out

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the outgoing TriMul module.
    The function mirrors the PyTorch reference implementation but replaces the
    O(N³) einsum with a custom Triton kernel that performs a batched matrix
    multiplication for each hidden channel.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

        input_tensor,

    left = left * left_gate

    left_fp16 = left.to(torch.float16)

        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # ----- Step 7: Final linear projection back to dim -----
    # Keep everything in float32 for the final linear
    output = F.linear(out, weights["to_out.weight"])
    return output

"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm
---------
1. Layer‑norm the input tensor over the feature dimension (dim).
2. Linear projections (left/right) without bias → shape [B,N,N,hidden_dim].
3. (Optional) Apply the pairwise mask to the projected tensors.
4. Compute three sigmoid gates from the normalized input and apply them:
   left = left * left_gate , right = right * right_gate .
5. Core multiplicative update:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched GEMM (A @ Bᵀ) for each (b,d) and is implemented
   in a custom Triton kernel for high throughput.
6. Layer‑norm the result over the hidden dimension, apply the output gate,
   and finish with a linear projection back to the original feature dimension.
The heavy O(N³·hidden_dim) step (5) is the only part written in Triton; the rest
uses torch.nn.functional for simplicity and correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul einsum
# --------------------------------------------------------------
@triton.jit
def trimul_mul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # batch size, seq len, hidden dim
    stride_bl, stride_il, stride_ik, stride_id,   # left strides (B,i,k,d)
    stride_br, stride_jr, stride_kr, stride_rd,   # right strides (B,j,k,d)
    stride_bo, stride_io, stride_jo, stride_od,   # out strides (B,i,j,d)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # 2‑D grid over i (rows) and j (cols); 3‑rd axis over batch*hidden
    pid_m = tl.program_id(0)    # block row   (i)
    pid_n = tl.program_id(1)    # block col   (j)
    pid_bc = tl.program_id(2)   # combined batch & hidden index

    # ------------------------------------------------------------------
    # Decode batch index b and hidden index d from pid_bc
    # ------------------------------------------------------------------
    b = pid_bc // H
    d = pid_bc % H

    # ------------------------------------------------------------------
    # Offsets for the current block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i indices
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j indices

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Base pointers for this (b,d) slice
    # ------------------------------------------------------------------
    left_base  = left_ptr  + b * stride_bl + d * stride_id
    right_base = right_ptr + b * stride_br + d * stride_rd
    out_base   = out_ptr   + b * stride_bo + d * stride_od

    # accumulator in fp32 (more accurate than fp16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    for start_k in range(0, N, BLOCK_K):
        # k indices for this tile
        offs_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- left tile : shape (M, K) ----
        # left[i, k]  -> i dimension uses stride_il, k uses stride_ik
        left_ptrs = left_base + (offs_m[:, None] * stride_il) + (offs_k[None, :] * stride_ik)
        left_tile = tl.load(left_ptrs,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0)            # fp16 values

        # ---- right tile transposed : shape (K, N) ----
        # right[j, k]  -> k uses stride_kr, j uses stride_jr
        # we load it as (k, j) to obtain the transpose needed for A @ Bᵀ
        right_ptrs = right_base + (offs_k[:, None] * stride_kr) + (offs_n[None, :] * stride_jr)
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_n[None, :],
                             other=0.0)           # fp16 values

        # Matrix multiplication for this tile (M×K)·(K×N) → (M×N)
        acc += tl.dot(left_tile, right_tile)   # acc is fp32

    # ------------------------------------------------------------------
    # Store the accumulated block
    # ------------------------------------------------------------------
    out_ptrs = out_base + (offs_m[:, None] * stride_io) + (offs_n[None, :] * stride_jo)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def trimul_einsum(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper around the Triton kernel that computes
        out[i, j, d] = Σ_k left[i, k, d] * right[j, k, d]
    left/right must be of shape [B, N, N, H] and have dtype torch.float16.
    Returns a float32 tensor of shape [B, N, N, H].
    """
    assert left.shape == right.shape, "left and right must have identical shapes"
    B, N, _, H = left.shape

    # Allocate output (float32 for higher accuracy)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # Ensure contiguity – Triton expects simple strided tensors
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # Strides in element‑units
    l_s = left.stride()
    r_s = right.stride()
    o_s = out.stride()

    # Tunable block sizes (fit well on H100, can be adjusted)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # 3‑D launch grid: (i‑blocks, j‑blocks, batch*hidden)
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    trimul_mul_kernel[grid](
        left, right, out,
        B, N, H,
        l_s[0], l_s[1], l_s[2], l_s[3],
        r_s[0], r_s[1], r_s[2], r_s[3],
        o_s[0], o_s[1], o_s[2], o_s[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out


# --------------------------------------------------------------
# Entry point used by the evaluator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module
    (the same semantics as the PyTorch reference implementation).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the feature dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])     # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional masking (mask shape [B, N, N])
    # ------------------------------------------------------------------
    if not nomask:
        # cast mask to same dtype and broadcast on the hidden dimension
        mask_f = mask.to(left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid of a linear projection)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) TriMul core – use Triton kernel
    # ------------------------------------------------------------------
    # Cast to half for the kernel (reduces bandwidth, still accurate)
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    # out is float32
    out = trimul_einsum(left_h, right_h)   # shape [B,N,N,hidden_dim], dtype float32

    # ------------------------------------------------------------------
    # 6) Output layer‑norm + output gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate   # element‑wise gate (both float32)

    # ------------------------------------------------------------------
    # 7) Final projection back to original dimension
    # ------------------------------------------------------------------
    output = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]
    return output
