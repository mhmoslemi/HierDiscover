@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)  # block row index
    pid_j = tl.program_id(1)  # block col index
    pid_bh = tl.program_id(2) # combined batch*hidden

    # decode batch and hidden index
    num_h_blocks = tl.cdiv(H, 1)  # we treat each hidden as separate program
    # but we want block size for hidden dimension = 1
    b = pid_bh // H
    h = pid_bh % H

    # Offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # For left:
    # Base pointer: left_ptr + b*stride_l_b + h*stride_l_h
    left_base = left_ptr + b*stride_l_b + h*stride_l_h
    # left submatrix block: rows i, col k
    # offset for row i: i * stride_l_i
    # offset for col k: k * stride_l_k
    # We'll compute block across k with loop.

    # For right (transposed):
    # Base pointer: right_ptr + b*stride_r_b + h*stride_r_h
    right_base = right_ptr + b*stride_r_b + h*stride_r_h

    # Output pointer: out_ptr + b*stride_o_b + h*stride_o_h
    out_base = out_ptr + b*stride_o_b + h*stride_o_h

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offset
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + (offs_i[:, None] * stride_l_i) + (offs_k[None, :] * stride_l_k)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right transposed block: shape (BLOCK_K, BLOCK_N) - note that we treat original right as (j,k) but we need (k,j)
        # right transposed: row = k, col = j
        right_ptrs = right_base + (offs_k[:, None] * stride_r_k) + (offs_j[None, :] * stride_r_j)
        # This loads right[k, j] values where original right[j,k].
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    # Store result
    out_ptrs = out_base + (offs_i[:, None] * stride_o_i) + (offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

import triton
import triton.language as tl

@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,

    This is equivalent to a batched matrix multiplication left[...,h] @ right[...,h].T.

    pid_j = tl.program_id(1)

"""
TriMul “outgoing” forward pass (AlphaFold‑3) with a custom Triton kernel.

The heavy part of the operation is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is mathematically a batched matrix multiplication
    left[...,d] @ right[...,d].T
for every hidden channel *d*.  The kernel below computes this
operation for all batches and hidden channels in a single launch.
All remaining steps (LayerNorm, linear projections, gating and the final
projection) are performed with regular PyTorch calls.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].

    This is equivalent to a batched GEMM per hidden channel:
        left[...,h] @ right[...,h].T
    where the batch dimension is (B * H).  The kernel iterates over the
    K‑dimension in tiles (BLOCK_K) and accumulates in float32.
    """
    pid_i = tl.program_id(0)      # block row   (i)
    pid_j = tl.program_id(1)      # block col   (j)
    pid_bh = tl.program_id(2)     # combined batch & hidden index

    # ----------------------------------------------------------------------
    # Decode batch (b) and hidden‑channel (h) from the 3rd program ID
    # ----------------------------------------------------------------------
    b = pid_bh // H               # integer division → batch index
    h = pid_bh % H                # remainder        → hidden index

    # ----------------------------------------------------------------------
    # Block offsets for the output matrix (i,j)
    # ----------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ----------------------------------------------------------------------
    # Base pointers for the current (batch, hidden) slice
    # ----------------------------------------------------------------------
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------------------------------------------------------
    # Loop over the K dimension (the summed axis)
    # ----------------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- left block (M × K) -------------------------------------------------
        # layout of left: (b, i, k, h)
        left_ptrs = left_base + (offs_i[:, None] * stride_l_i) + (offs_k[None, :] * stride_l_k)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ----- right block transposed (K × N) ------------------------------------
        # layout of right: (b, j, k, h) → we need (k, j) for the transpose
        right_ptrs = right_base + (offs_k[:, None] * stride_r_k) + (offs_j[None, :] * stride_r_j)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # ----- Multiply‑accumulate ------------------------------------------------
        acc += tl.dot(left, right)               # (M×K)·(K×N) → (M×N)

    # ----------------------------------------------------------------------
    # Write the result
    # ----------------------------------------------------------------------
    out_ptrs = out_base + (offs_i[:, None] * stride_o_i) + (offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul module (outgoing version).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on the same device).
    """
    # ----------------------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]                 # C  (input/output feature size)
    hidden_dim = config["hidden_dim"]   # C_h (intermediate size)

    # ----------------------------------------------------------------------
    # Shortcut for the weight tensors
    # ----------------------------------------------------------------------
    norm_weight            = weights["norm.weight"]
    norm_bias              = weights["norm.bias"]
    left_proj_weight       = weights["left_proj.weight"]
    right_proj_weight      = weights["right_proj.weight"]
    left_gate_weight       = weights["left_gate.weight"]
    right_gate_weight      = weights["right_gate.weight"]
    out_gate_weight        = weights["out_gate.weight"]
    to_out_norm_weight     = weights["to_out_norm.weight"]
    to_out_norm_bias       = weights["to_out_norm.bias"]
    to_out_weight          = weights["to_out.weight"]

    # ----------------------------------------------------------------------
    # 1) LayerNorm on the input tensor (over the last dimension)
    # ----------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # shape [B, N, N, dim]

    # ----------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ----------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_weight)   # → [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)  # → [B,N,N,hidden_dim]

    # ----------------------------------------------------------------------
    # 3) Optional mask (broadcast over the hidden dimension)
    # ----------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).type_as(left)   # [B,N,N,1]
        left  = left * mask_exp
        right = right * mask_exp

    # ----------------------------------------------------------------------
    # 4) Gating signals (sigmoid of linear layers)
    # ----------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply the gates
    left = left * left_gate          # [B,N,N,hidden_dim]
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 5) Heavy part – the N³ einsum – via Triton
    # ----------------------------------------------------------------------
    B, N, _, H = left.shape                        # H == hidden_dim
    assert H == hidden_dim, "hidden dimension mismatch"

    # Allocate the intermediate result
    out_inter = torch.empty((B, N, N, H), dtype=left.dtype, device=left.device)

    # Strides in element‑units (not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_inter.stride()

    # Choose reasonable tile sizes (tuned for H100, FP32)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑blocks
        triton.cdiv(N, BLOCK_N),          # j‑blocks
        B * H,                            # batch × hidden programs
    )

    _trimul_matmul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out_inter.data_ptr(),
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # ----------------------------------------------------------------------
    # 6) Post‑processing: LayerNorm → gate → final linear projection
    # ----------------------------------------------------------------------
    # LayerNorm on the hidden dimension
    out_norm = torch.nn.functional.layer_norm(
        out_inter,
        (H,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )   # shape [B,N,N,hidden_dim]

    # Apply the output gate
    out_norm = out_norm * out_gate

    # Final linear projection back to `dim`
    out = torch.nn.functional.linear(out_norm, to_out_weight)   # [B,N,N,dim]

    return out
