@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_i, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    num_blocks_d = tl.cdiv(D, BLOCK_D)
    b = pid_z // num_blocks_d
    d_block = pid_z % num_blocks_d

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    d0 = d_block * BLOCK_D

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_d = d0 + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    for k0 in range(0, N, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BM, BK, BD)
        left_ptrs = left_ptr + b * stride_l_b + (offs_i[:, None, None] * stride_l_i) + (offs_k[None, :, None] * stride_l_k) + (offs_d[None, None, :] * stride_l_d)
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0).to(tl.float32)

        # right block: shape (BN, BK, BD)
        right_ptrs = right_ptr + b * stride_r_b + (offs_j[:, None, None] * stride_r_i) + (offs_k[None, :, None] * stride_r_k) + (offs_d[None, None, :] * stride_r_d)
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0).to(tl.float32)

        # compute sum over k
        for kk in range(0, BLOCK_K):
            # Broadcast left[:, kk, :] -> (BM, 1, BD)
            # Broadcast right[:, kk, :] -> (1, BN, BD)
            acc += left[:, kk, :][:, None, :] * right[:, kk, :][None, :, :]
    # store result
    out_ptrs = out_ptr + b * stride_o_b + (offs_i[:, None, None] * stride_o_i) + (offs_j[None, :, None] * stride_o_j) + (offs_d[None, None, :] * stride_o_d)
    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])

"""
TriMul "outgoing" implementation using Triton for the heavy O(N³·C) contraction.
The algorithm follows the reference PyTorch implementation:

1. Layer‑norm on the input (dim = C).
2. Linear projections to a hidden dimension H for left/right tensors.
3. Gating signals (sigmoid of linear projections) and optional mask.
4. Elementwise gating & masking are performed in FP32 (torch for simplicity).
5. The core operation
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is executed by a custom Triton kernel that tiles the (i,j,d) output
   space and reduces over k.  The kernel works in FP16 for the two
   inputs and accumulates in FP32, then writes the result in FP16.
6. Layer‑norm on the hidden dimension, gated by the out‑gate signal,
   and a final linear projection back to the original dimension.

The kernel fuses the N³ reduction across the k‑axis while keeping the
hidden dimension vectorized (BLOCK_D).  This yields a highly parallel
implementation that fits well on an H100 GPU.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel:  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,               # half*  [B,N,N,H]
    right_ptr,              # half*  [B,N,N,H]
    out_ptr,                # half*  [B,N,N,H]
    B, N, H,                # int
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,  # tile size for i‑axis
    BLOCK_N: tl.constexpr,  # tile size for j‑axis
    BLOCK_K: tl.constexpr,  # tile size for reduction (k)
    BLOCK_H: tl.constexpr,  # tile size for hidden dimension
):
    pid_i = tl.program_id(0)                # i‑tile
    pid_j = tl.program_id(1)                # j‑tile
    pid_z = tl.program_id(2)                # combines batch and hidden‑dim tile

    # ------------------------------------------------------------------
    # Decode batch index and hidden‑dim tile from pid_z
    # ------------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    b = pid_z // num_h_tiles                # batch index (0 ≤ b < B)
    h_tile = pid_z % num_h_tiles             # hidden‑dim tile index
    # ------------------------------------------------------------------
    # Compute block start indices
    # ------------------------------------------------------------------
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    h0 = h_tile * BLOCK_H

    # ------------------------------------------------------------------
    # Offsets for each dimension (range within the tile)
    # ------------------------------------------------------------------
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_h = h0 + tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # ------------------------------------------------------------------
    # Accumulator (FP32) for the tile (i, j, h)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction axis k in blocks of size BLOCK_K
    # ------------------------------------------------------------------
    for k0 in range(0, N, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -----------------------------
        # Load left[b, i, k, h]
        # -----------------------------
        # Shape: (BLOCK_M, BLOCK_K, BLOCK_H)
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + offs_i[:, None, None] * stride_l_i
            + offs_k[None, :, None] * stride_l_k
            + offs_h[None, None, :] * stride_l_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)

        # -----------------------------
        # Load right[b, j, k, h]
        # -----------------------------
        # Shape: (BLOCK_N, BLOCK_K, BLOCK_H)
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + offs_j[:, None, None] * stride_r_i
            + offs_k[None, :, None] * stride_r_k
            + offs_h[None, None, :] * stride_r_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)

        # --------------------------------------------------------------
        # Accumulate over k:  acc[i,j,h] += left[i,kk,h] * right[j,kk,h]
        # --------------------------------------------------------------
        for kk in range(0, BLOCK_K):
            # left_kk: (BLOCK_M, BLOCK_H)
            # right_kk: (BLOCK_N, BLOCK_H)
            l = left[:, kk, :]          # (M, H)
            r = right[:, kk, :]         # (N, H)

            # Broadcast to (M, N, H) and accumulate
            # l[:, None, :] -> (M,1,H); r[None, :, :] -> (1,N,H)
            acc += l[:, None, :] * r[None, :, :]

    # ------------------------------------------------------------------
    # Write the tile back to global memory
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + offs_i[:, None, None] * stride_o_i
        + offs_j[None, :, None] * stride_o_j
        + offs_h[None, None, :] * stride_o_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )


# ----------------------------------------------------------------------
# Entry point expected by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Implements the forward pass of the TriMul "outgoing" operator.
    Heavy k‑reduction is carried out by a Triton kernel; all other
    linear layers, LayerNorm, and sigmoids are performed with PyTorch.
    """
    input_tensor, mask_tensor, weights, config = data

    # ------------------------------------------------------------------
    # Extract configuration
    # ------------------------------------------------------------------
    dim = config["dim"]                # input channel dimension C
    hidden_dim = config["hidden_dim"]  # hidden dimension H
    eps = 1e-5

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Unpack weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # First LayerNorm over the input dimension (C)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: [B, N, N, hidden_dim]
    left = F.linear(x, left_proj_weight)      # (B,N,N,H)
    right = F.linear(x, right_proj_weight)   # (B,N,N,H)

    # ------------------------------------------------------------------
    # Gating signals (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # Apply optional mask (broadcast to hidden_dim)
    # ------------------------------------------------------------------
    if mask_tensor is None or config.get("nomask", False):
        mask = torch.ones_like(left[..., :1], dtype=left.dtype, device=device)
    else:
        # mask shape: [B, N, N]; expand to [B, N, N, 1] for broadcasting
        mask = mask_tensor.unsqueeze(-1).type_as(left)

    # ------------------------------------------------------------------
    # Element‑wise masking + gating (performed in FP32 for stability)
    # ------------------------------------------------------------------
    left = left * mask * left_gate
    right = right * mask * right_gate

    # ------------------------------------------------------------------
    # Cast to half precision before the heavy contraction
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # Output tensor for the contraction (FP16)
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Strides for the Triton kernel (in element units)
    # ------------------------------------------------------------------
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_fp16.stride()

    B, N, _, _ = left_fp16.shape  # batch, seq_len, seq_len, hidden_dim

    # ------------------------------------------------------------------
    # Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_H = 4   # hidden‑dim tiling (fits well in registers)

    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    num_h_tiles = (hidden_dim + BLOCK_H - 1) // BLOCK_H
    grid_z = B * num_h_tiles
    grid = (grid_i, grid_j, grid_z)

    # ------------------------------------------------------------------
    # Launch the Triton kernel
    # ------------------------------------------------------------------
    trimul_einsum_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # Convert back to FP32 for the remaining LayerNorm, gating, and final projection
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # Second LayerNorm on the hidden dimension (H)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps
    )

    # ------------------------------------------------------------------
    # Multiply by the out‑gate signal and project back to original dimension
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # shape: [B, N, N, dim]

    return out
