                                            weight=norm_weight,

    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

```
import torch
import triton
import triton.language as tl

@triton.jit
def trimul_out_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out = sum_k left[...,i,k,h] * right[...,j,k,h] for each batch b and hidden h.
    left and right are fp16; output is fp32.
    """
    pid_i = tl.program_id(axis=0)
    pid_j = tl.program_id(axis=1)
    pid_hb = tl.program_id(axis=2)   # combined batch * hidden

    # decode batch and hidden
    b = pid_hb // H
    h = pid_hb % H

    # offsets for i and j axes
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for i,j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k blocks
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + \
            b * stride_left_b + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k + \
            h * stride_left_h
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right tile as (BLOCK_K, BLOCK_N) so we can dot with a
        b_ptrs = right_ptr + \
            b * stride_right_b + \
            offs_k[:, None] * stride_right_k + \
            offs_j[None, :] * stride_right_i + \
            h * stride_right_h
        b_tile = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(a, b_tile)

    # store result
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j + \
        h * stride_out_h
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

    1. LayerNorm on dim.
    2. Linear projections for left/right and gates (left_gate, right_gate, out_gate).
    3. Mask multiplication (if provided).
    4. Multiply left/right by their respective gates.
    5. Compute pairwise interaction by summing over the second sequence dimension using a
       custom Triton GEMM kernel (equivalent to einsum '... i k d, ... j k d -> ... i j d').
    6. Apply a second LayerNorm, multiply by out_gate, and final linear projection.
    All heavy compute (step 5) runs in fp16 for left/right and accumulates in fp32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Weight extraction
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1. LayerNorm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # 2. Linear projections (float32)
    left = torch.nn.functional.linear(x_norm, left_proj_weight)      # (B,N,N,hidden)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    if mask is not None and mask.numel() > 0:

        left = left * mask
        right = right * mask

    # 4. Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for the kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_fp16.shape  # validate shapes
    # Prepare output buffer (float32)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute grid
    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_m, num_n, B * hidden_dim)

    # Strides (int64)
    left_stride_b, left_stride_i, left_stride_k, left_stride_h = left_fp16.stride()
    right_stride_b, right_stride_i, right_stride_k, right_stride_h = right_fp16.stride()
    out_stride_b, out_stride_i, out_stride_j, out_stride_h = out.stride()

    # Launch kernel
    trimul_out_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        hidden_dim,
        left_stride_b,
        left_stride_i,
        left_stride_k,
        left_stride_h,
        right_stride_b,
        right_stride_i,
        right_stride_k,
        right_stride_h,
        out_stride_b,
        out_stride_i,
        out_stride_j,
        out_stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # 6. to_out_norm (LayerNorm over hidden dim)
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    # Multiply by out_gate (already float32)
    out_norm = out_norm * out_gate

    # Final projection back to original dim
    final_out = torch.nn.functional.linear(out_norm, to_out_weight)  # (B,N,N,dim)
    return final_out

"""
TriMul (outgoing) forward with a Triton‑accelerated pairwise interaction.

Algorithm
---------
1. Layer‑normalize the 4‑D input (B, N, N, dim).
2. Linear projections:
   * left  = x·W_left      (B,N,N,hidden)
   * right = x·W_right
   * left_gate  = sigmoid(x·W_left_gate)
   * right_gate = sigmoid(x·W_right_gate)
   * out_gate   = sigmoid(x·W_out_gate)  (same shape as the final output)
3. Optional pairwise mask (B,N,N,1) is applied to *left* and *right* only.
4. Gating:  left  ← left  * left_gate
           right ← right * right_gate
5. Compute the expensive “tri‑multiplicative” reduction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication (one GEMM per hidden channel) and
   is performed by a custom Triton kernel in fp16 (accumulating in fp32).
6. Layer‑normalize *out* over the hidden dimension, multiply by *out_gate*,
   and apply the final linear projection back to *dim*.

The Triton kernel fuses the reduction over the third axis (k) for every
batch‑hidden pair, dramatically reducing kernel launch overhead compared to a
naïve triple‑loop implementation.
"""

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched GEMM for the triangular multiplication
# --------------------------------------------------------------
@triton.jit
def trimul_out_kernel(
    left_ptr,          # fp16 pointer, shape (B,N,N,H)
    right_ptr,         # fp16 pointer, shape (B,N,N,H)
    out_ptr,           # fp32 pointer, shape (B,N,N,H)
    B, N, H,           # runtime sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Computes out = Σ_k left[...,i,k,h] * right[...,j,k,h].

    Each program instance handles a tile (BLOCK_M × BLOCK_N) of the output
    for a single (batch, hidden) pair.
    """
    pid_i = tl.program_id(axis=0)          # tile index over i (rows)
    pid_j = tl.program_id(axis=1)          # tile index over j (cols)
    pid_hb = tl.program_id(axis=2)         # combined batch * hidden index

    # decode batch and hidden index
    b = pid_hb // H
    h = pid_hb % H

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in blocks
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load left tile (M × K) ----
        a_ptrs = (
            left_ptr
            + b * stride_left_b
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + h * stride_left_h
        )
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)                     # fp16 input

        # ---- load right tile as (K × N) ----
        # note the order of offsets so that the tile is transposed w.r.t. k
        b_ptrs = (
            right_ptr
            + b * stride_right_b
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_i
            + h * stride_right_h
        )
        b_tile = tl.load(b_ptrs,
                         mask=mask_k[:, None] & mask_j[None, :],
                         other=0.0)                 # fp16 input

        # ---- dot product (M×K)·(K×N) -> (M×N) ----
        acc += tl.dot(a, b_tile)   # accumulates in fp32

    # store result
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------
# Public entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict containing all model parameters
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Load weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 2. Layer‑norm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # (B, N, N, dim), dtype=float32

    # ------------------------------------------------------------------
    # 3. Linear projections (float32) -----------------------------------
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)   # (B,N,N,hidden)
    right = torch.nn.functional.linear(x, right_proj_weight)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 4. Optional mask (applied only to left/right)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask = mask.to(x.dtype).unsqueeze(-1)               # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5. Apply gating
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Cast to fp16 and allocate output buffer (float32)
    # ------------------------------------------------------------------
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_fp16.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # --------------------------------------------------------------
    # 7. Triton launch configuration
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_m, num_n, B * hidden_dim)  # (i_tiles, j_tiles, batch*hidden)

    # strides (as returned by .stride()) -------------------------------------------------
    lsb, lsi, lsk, lsh = left_fp16.stride()
    rsb, rsi, rsk, rsh = right_fp16.stride()
    osb, osi, osj, osh = out.stride()

    # --------------------------------------------------------------
    # 8. Run the Triton kernel (fp16×fp16 → fp32)
    # --------------------------------------------------------------
    trimul_out_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        hidden_dim,
        lsb, lsi, lsk, lsh,
        rsb, rsi, rsk, rsh,
        osb, osi, osj, osh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 9. Second LayerNorm over hidden dimension, out_gate, final linear
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate            # apply out gate (float32)

    final = torch.nn.functional.linear(out, to_out_weight)   # (B,N,N,dim)
    return final
