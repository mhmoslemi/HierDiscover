```python
import torch
import triton
import triton.language as tl

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on last dim
    # Compute mean and var across dim
    # Using torch.nn.functional.layer_norm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if present
    if mask is not None:
        # mask shape [B,N,N]; unsqueeze to [B,N,N,1] and broadcast
        mask = mask.unsqueeze(-1).type_as(left)
        left = left * mask
        right = right * mask

    # gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()   # shape [B,N,N,hidden_dim]
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()
    left = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape   # B,N,N,hidden_dim

    # reshape to [B*hidden_dim, N, N] with groups as dimension
    left_flat = left.permute(0,3,1,2).contiguous().view(B*hidden_dim, N, N)
    right_flat = right.permute(0,3,1,2).contiguous().view(B*hidden_dim, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # launch Triton kernel
    # compute strides
    a_stride_g, a_stride_i, a_stride_j = left_flat.stride()
    b_stride_g, b_stride_i, b_stride_j = right_flat.stride()
    c_stride_g, c_stride_i, c_stride_j = out_flat.stride()

    # constants
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B*hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    matmul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        a_stride_g,
        a_stride_i,
        a_stride_j,
        b_stride_g,
        b_stride_i,
        b_stride_j,
        c_stride_g,
        c_stride_i,
        c_stride_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )
    # reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0,2,3,1)   # shape [B,N,N,hidden_dim]

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # gating with out_gate (broadcast)
    out = out * out_gate

    # final linear
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)   # shape [B,N,N,dim]

    return out
```

"""
TriMul (outgoing) – Triton‑accelerated forward pass

The operator receives a 4‑D tensor `x` of shape [B, N, N, D] and a (optional) binary
mask of shape [B, N, N].  The forward consists of:

1. Layer‑norm over the feature dimension D.
2. Two bias‑free linear projections (left / right) → hidden dimension H.
3. Optional mask multiplication.
4. Three sigmoid gates (left, right, out) applied to the projected tensors.
5. **Core O(N³·H) computation**:
       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]      (i‑j outer, k‑inner)
   This is a batched matrix multiplication for every (b, h) pair:
       out_g = left_g @ right_gᵀ   where g = b*H + h.
   The heavy reduction over *k* is performed by a Triton kernel that
   loads the two operands in FP16, accumulates in FP32 and writes the result
   as FP32 (keeps full precision for the subsequent layer‑norm).
6. Layer‑norm over the hidden dimension, multiplication by the out‑gate,
   and a final linear projection back to dimension D.

The Triton kernel computes many independent NxN matrix products in parallel,
tiling the M, N and K dimensions with 64‑element blocks (chosen to map well to
tensor‑cores on H100).  All surrounding logic (norms, linear layers, gating,
mask handling) is expressed in PyTorch for simplicity; only the cubic part is
offloaded to Triton.
"""

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel for the batched matrix multiplication   (outgoing version) #
# --------------------------------------------------------------------------- #
@triton.jit
def _triton_matmul_outgoing_kernel(
    a_ptr,                # left  matrix (group, i, k)  – FP16
    b_ptr,                # right matrix (group, j, k) – FP16
    c_ptr,                # output (group, i, j)       – FP32
    N,                    # sequence length
    stride_a_g, stride_a_i, stride_a_k,
    stride_b_g, stride_b_k, stride_b_j,   # note swapped i/k for the transposed view
    stride_c_g, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # 3‑D grid: (group, block_row, block_col)
    pid_g = tl.program_id(0)   # flattened batch * hidden group
    pid_m = tl.program_id(1)   # block row index (i)
    pid_n = tl.program_id(2)   # block col index (j)

    # ----- tile offsets ----------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- iterate over K dimension (the summed axis) -----------------------
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A: (group, i, k) – row‑major
        a_ptrs = a_ptr + pid_g * stride_a_g \
                         + offs_m[:, None] * stride_a_i \
                         + offs_k[None, :] * stride_a_k
        # B transposed: we need Bᵀ[k, j] = B[j, k]
        # Original B layout: (group, j, k)
        b_ptrs = b_ptr + pid_g * stride_b_g \
                         + offs_k[:, None] * stride_b_k \
                         + offs_n[None, :] * stride_b_j

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0, dtype=tl.float16)

        # FP16 × FP16 → accumulate in FP32
        acc += tl.dot(a, b)

    # ----- write result (FP32) ---------------------------------------------
    c_ptrs = c_ptr + pid_g * stride_c_g \
                     + offs_m[:, None] * stride_c_i \
                     + offs_n[None, :] * stride_c_j
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Entry point – custom Kernel callable from the evaluation harness            #
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, D] (float32, on the same device as inputs).
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------- #
    # 1. Configuration & weight extraction
    # ------------------------------------------------------------------- #
    dim = config["dim"]                       # input feature dimension D
    hidden_dim = config["hidden_dim"]         # intermediate hidden dimension H
    eps = 1e-5

    # layer‑norm parameters
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # linear projection weights (bias‑free)
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # output projection & its norm
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # ------------------------------------------------------------------- #
    # 2. Input normalisation
    # ------------------------------------------------------------------- #
    # Torch's LayerNorm works over the last dimension (`dim`).
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps
    )   # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------- #
    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left  = torch.nn.functional.linear(x, left_proj_w)    # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_w)   # [B,N,N,H]

    # ------------------------------------------------------------------- #
    # 4. Optional mask application
    # ------------------------------------------------------------------- #
    if not config.get("nomask", False):
        # mask is broadcasted on the hidden dimension
        mask = mask_tensor.unsqueeze(-1).to(x.dtype)     # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------- #
    # 5. Gating (sigmoid, bias‑free)
    # ------------------------------------------------------------------- #
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()   # [B,N,N,H]
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()  # [B,N,N,H]
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()    # [B,N,N,H]

    left  = left  * left_gate
    right = right * right_gate

    # Cast projected tensors to FP16 for the heavy O(N³) kernel
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------- #
    # 6. Prepare tensors for the Triton kernel:
    #    shape -> (group, N, N) where group = B * hidden_dim
    # ------------------------------------------------------------------- #
    B, N, _, _ = left.shape
    num_groups = B * hidden_dim

    # [B, N, N, H] -> [B, H, N, N] -> contiguous -> (group, N, N)
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(num_groups, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(num_groups, N, N)

    # Allocate output in FP32 (full precision for subsequent norm)
    out_flat = torch.empty((num_groups, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------- #
    # 7. Compute the cubic reduction with Triton
    # ------------------------------------------------------------------- #
    # Strides expressed in element counts (torch gives strides in that form)
    s_a_g, s_a_i, s_a_k = left_flat.stride()
    s_b_g, s_b_j, s_b_k = right_flat.stride()   # original layout: (group, j, k)

    # For the transposed view inside the kernel we need:
    #   stride_b_k = stride for the original i‑dimension (here = s_b_j)
    #   stride_b_j = stride for the original j‑dimension (here = s_b_k)
    s_b_k = s_b_j
    s_b_j = s_b_k

    s_c_g, s_c_i, s_c_j = out_flat.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        num_groups,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _triton_matmul_outgoing_kernel[
        grid
    ](
        left_flat,
        right_flat,
        out_flat,
        N,
        s_a_g, s_a_i, s_a_k,
        s_b_g, s_b_k, s_b_j,
        s_c_g, s_c_i, s_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------- #
    # 8. Reshape kernel output back to [B, N, N, H] (float32)
    # ------------------------------------------------------------------- #
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, H]

    # ------------------------------------------------------------------- #
    # 9. Final normalisation, out‑gate, and projection back to D
    # ------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps
    )
    out = out * out_gate

    # Final linear (bias‑free)
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, D]

    return out
