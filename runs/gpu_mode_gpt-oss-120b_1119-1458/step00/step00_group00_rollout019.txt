def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # x shape (B,N,N,dim)
    B, N, _, _ = x.shape

    # LayerNorm on dim
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # fuse mask and gates using Triton if mask is not None else apply directly
    if mask is not None:
        # mask shape (B,N,N) -> ensure float32
        mask_f = mask.to(x.dtype)  # Cast to same dtype
        # allocate output tensors for gated left/right
        left_gated = torch.empty_like(left)
        right_gated = torch.empty_like(right)
        # get strides
        left_strides = left.stride()
        right_strides = right.stride()
        left_gate_strides = left_gate.stride()
        right_gate_strides = right_gate.stride()
        mask_strides = mask_f.stride()
        BLOCK_SIZE = 1024  # tune
        grid = lambda meta: (triton.cdiv(B*N*N*hidden_dim, meta['BLOCK_SIZE']),)
        fuse_masks_gates_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_f,
            out_left_ptr=left_gated,
            out_right_ptr=right_gated,
            B=B, N=N, D=hidden_dim,
            stride_left_bi=left_strides[0],
            stride_left_i=left_strides[1],
            stride_left_j=left_strides[2],
            stride_left_d=left_strides[3],
            stride_right_bi=right_strides[0],
            stride_right_i=right_strides[1],
            stride_right_j=right_strides[2],
            stride_right_d=right_strides[3],
            stride_gate_bi=left_gate_strides[0],
            stride_gate_i=left_gate_strides[1],
            stride_gate_j=left_gate_strides[2],
            stride_gate_d=left_gate_strides[3],
            stride_mask_bi=mask_strides[0],
            stride_mask_i=mask_strides[1],
            stride_mask_j=mask_strides[2],
            BLOCK_SIZE=BLOCK_SIZE
        )
        left = left_gated
        right = right_gated
    else:
        # no mask; just multiply with gates
        left = left * left_gate
        right = right * right_gate

    # Now compute output using batched matrix multiplication
    # reshape to (B*hidden_dim, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous()  # shape (B, hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous()
    # compute out_flat = left_flat @ right_flat.transpose(-2, -1)
    out_flat = torch.matmul(left_flat, right_flat.transpose(-2, -1))
    # shape (B, hidden_dim, N, N)
    out = out_flat.permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,hidden_dim)

    # apply to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # apply out_gate
    out = out * out_gate

    # final linear transform
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

import torch
import triton
import triton.language as tl

@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    B, N, D,
    stride_left_bi, stride_left_i, stride_left_j, stride_left_d,
    stride_right_bi, stride_right_i, stride_right_j, stride_right_d,
    stride_gate_bi, stride_gate_i, stride_gate_j, stride_gate_d,
    stride_mask_bi, stride_mask_i, stride_mask_j,
    BLOCK_SIZE: tl.constexpr,
):
    """
    In-place kernel that fuses:
        left = left * mask * left_gate
        right = right * mask * right_gate
    on a tensor of shape (B, N, N, D).
    mask has shape (B, N, N) and is broadcast over the channel dimension.
    """
    pid = tl.program_id(0)
    total_elements = B * N * N * D
    start = pid * BLOCK_SIZE
    offs = tl.arange(0, BLOCK_SIZE)
    idx = start + offs
    mask_valid = idx < total_elements

    # decode linear index into 4-D coordinates (b, i, j, d)
    b = idx // (N * N * D)
    rem = idx % (N * N * D)
    i = rem // (N * D)
    rem2 = rem % (N * D)
    j = rem2 // D
    d = rem2 % D

    # compute flat offsets for each tensor
    left_offset = (b * stride_left_bi + i * stride_left_i +
                   j * stride_left_j + d * stride_left_d)
    right_offset = (b * stride_right_bi + i * stride_right_i +
                    j * stride_right_j + d * stride_right_d)
    left_gate_offset = (b * stride_gate_bi + i * stride_gate_i +
                        j * stride_gate_j + d * stride_gate_d)
    right_gate_offset = left_gate_offset   # same layout as left_gate/right_gate
    mask_offset = b * stride_mask_bi + i * stride_mask_i + j * stride_mask_j

    # load values (mask defaults to 1.0 when not present)
    left_val = tl.load(left_ptr + left_offset, mask=mask_valid)
    right_val = tl.load(right_ptr + right_offset, mask=mask_valid)
    left_gate_val = tl.load(left_gate_ptr + left_gate_offset, mask=mask_valid)
    right_gate_val = tl.load(right_gate_ptr + right_gate_offset, mask=mask_valid)
    mask_val = tl.load(mask_ptr + mask_offset, mask=mask_valid, other=1.0)

    # fused multiply
    left_out = left_val * left_gate_val * mask_val
    right_out = right_val * right_gate_val * mask_val

    # store back (in‑place)
    tl.store(left_ptr + left_offset, left_out, mask=mask_valid)
    tl.store(right_ptr + right_offset, right_out, mask=mask_valid)

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” update.
    The expensive element‑wise mask/gate fusion is performed by a custom
    Triton kernel, while the core N×N×N matrix multiplication is
    delegated to a batched cuBLAS GEMM (via torch.matmul).
    """
    # unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape
    B = batch_size
    N = seq_len
    H = hidden_dim

    # -----------------------------------------------------------------
    # 1) LayerNorm over the channel dimension (`dim`)
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Linear projections
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Gating values (sigmoid of a linear projection)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # -----------------------------------------------------------------
    # 4) Optional mask; fuse mask+gate+proj with Triton
    # -----------------------------------------------------------------
    apply_mask = (mask is not None) and (not config.get("nomask", False))
    if apply_mask:
        # ensure mask dtype matches the projection dtype
        mask_f = mask.to(left.dtype)
        # launch Triton kernel – we operate in‑place on `left` and `right`
        BLOCK_SIZE = 1024
        total = B * N * N * H
        grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)
        _mask_gate_fuse_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_f,
            B=B,
            N=N,
            D=H,
            stride_left_bi=left.stride(0),
            stride_left_i=left.stride(1),
            stride_left_j=left.stride(2),
            stride_left_d=left.stride(3),
            stride_right_bi=right.stride(0),
            stride_right_i=right.stride(1),
            stride_right_j=right.stride(2),
            stride_right_d=right.stride(3),
            stride_gate_bi=left_gate.stride(0),
            stride_gate_i=left_gate.stride(1),
            stride_gate_j=left_gate.stride(2),
            stride_gate_d=left_gate.stride(3),
            stride_mask_bi=mask_f.stride(0),
            stride_mask_i=mask_f.stride(1),
            stride_mask_j=mask_f.stride(2),
            BLOCK_SIZE=BLOCK_SIZE,
        )
    else:
        # No mask – just multiply by the gates
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Core triangle multiplication: batched matrix multiply
    #    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    #    This is equivalent to (left[...,d] @ right[...,d].T) per channel.
    # -----------------------------------------------------------------
    # reshape to (B*H, N, N) and use torch.matmul (cuBLAS)
    left_flat = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous()
    # matrix multiply with the second operand transposed on its last two dims
    out_flat = torch.matmul(left_flat, right_flat.transpose(-2, -1))
    # reshape back to (B, N, N, H)
    out = out_flat.permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6) Output LayerNorm, gating and final linear projection
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

# -*- coding: utf-8 -*-
"""
TriMul (outgoing) forward pass with a lightweight Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension (`dim`).
2. Linear projections produce `left` and `right` (shape [B,N,N,H]).
3. Compute three sigmoid gates from the normalized input:
   * `left_gate`, `right_gate` – applied to `left`/`right`
   * `out_gate`               – applied after the triangle multiplication.
4. If a mask is supplied, fuse ``mask * gate`` into `left` and `right`
   using a small Triton kernel (in‑place, no extra memory copies).
   When no mask is present we simply multiply by the gates.
5. Core triangle multiplication:
   For each channel `h`, `out[...,h] = left[...,h] @ right[...,h].T`.
   This is realized as a batched GEMM via `torch.matmul`,
   which calls cuBLAS and is highly optimized on H100.
6. Apply a second LayerNorm over the hidden dimension, multiply by
   `out_gate`, and project back to the original dimension with a final
   linear layer.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    B, N, D,
    stride_left_bi, stride_left_i, stride_left_j, stride_left_d,
    stride_right_bi, stride_right_i, stride_right_j, stride_right_d,
    stride_gate_bi, stride_gate_i, stride_gate_j, stride_gate_d,
    stride_mask_bi, stride_mask_i, stride_mask_j,
    BLOCK_SIZE: tl.constexpr,
):
    """
    In‑place fusion of mask and gates:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
    Input tensors have shape (B, N, N, D); mask has shape (B, N, N) and is
    broadcast over the channel dimension.
    """
    pid = tl.program_id(0)
    total = B * N * N * D
    start = pid * BLOCK_SIZE
    offs = tl.arange(0, BLOCK_SIZE)
    idx = start + offs
    valid = idx < total

    # Decode linear index -> (b,i,j,d)
    b = idx // (N * N * D)
    rem = idx % (N * N * D)
    i = rem // (N * D)
    rem2 = rem % (N * D)
    j = rem2 // D
    d = rem2 % D

    # Offsets for each tensor
    left_off   = (b * stride_left_bi   + i * stride_left_i   +
                  j * stride_left_j   + d * stride_left_d)
    right_off  = (b * stride_right_bi  + i * stride_right_i  +
                  j * stride_right_j  + d * stride_right_d)
    left_gate_off  = (b * stride_gate_bi   + i * stride_gate_i   +
                      j * stride_gate_j   + d * stride_gate_d)
    right_gate_off = left_gate_off          # same layout for the two gates
    mask_off   = b * stride_mask_bi + i * stride_mask_i + j * stride_mask_j

    # Load values
    l_val      = tl.load(left_ptr   + left_off,    mask=valid)
    r_val      = tl.load(right_ptr  + right_off,   mask=valid)
    lg_val     = tl.load(left_gate_ptr  + left_gate_off,  mask=valid)
    rg_val     = tl.load(right_gate_ptr + right_gate_off, mask=valid)
    mask_val   = tl.load(mask_ptr + mask_off, mask=valid, other=1.0)

    # Fusion
    l_out = l_val * lg_val * mask_val
    r_out = r_val * rg_val * mask_val

    # Store back in‑place
    tl.store(left_ptr   + left_off,   l_out, mask=valid)
    tl.store(right_ptr  + right_off,  r_out, mask=valid)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of model parameters (see code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` and
                         optional ``nomask`` flag.

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] – the updated pair representation.
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape
    H = hidden_dim

    # -----------------------------------------------------------------
    # 1) LayerNorm over the channel dimension `dim`
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Compute sigmoid gates from the normalized input
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 4) Optional mask: fuse mask * gate * projection with Triton
    # -----------------------------------------------------------------
    apply_mask = (mask is not None) and (not config.get("nomask", False))
    if apply_mask:
        # mask must be the same dtype as the projections
        mask_f = mask.to(left.dtype)

        # Launch fused kernel (in‑place updates of `left` & `right`)
        BLOCK_SIZE = 1024
        total = B * N * N * H
        grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

        _mask_gate_fuse_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_f,
            B=B,
            N=N,
            D=H,
            stride_left_bi=left.stride(0),
            stride_left_i=left.stride(1),
            stride_left_j=left.stride(2),
            stride_left_d=left.stride(3),
            stride_right_bi=right.stride(0),
            stride_right_i=right.stride(1),
            stride_right_j=right.stride(2),
            stride_right_d=right.stride(3),
            stride_gate_bi=left_gate.stride(0),
            stride_gate_i=left_gate.stride(1),
            stride_gate_j=left_gate.stride(2),
            stride_gate_d=left_gate.stride(3),
            stride_mask_bi=mask_f.stride(0),
            stride_mask_i=mask_f.stride(1),
            stride_mask_j=mask_f.stride(2),
            BLOCK_SIZE=BLOCK_SIZE,
        )
    else:
        # No mask – simple element‑wise gate multiplication
        left = left * left_gate
        right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Core triangle multiplication:
    #    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    #    This is a batched GEMM (left @ right.T) per hidden channel.
    # -----------------------------------------------------------------
    # Rearrange to (B, H, N, N) -> treat B*H as batch dimension for cuBLAS
    left_flat  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous()

    # Matrix multiply with the second operand transposed on its last two axes
    out_flat = torch.matmul(left_flat, right_flat.transpose(-2, -1))  # (B, H, N, N)

    # Restore original ordering (B, N, N, H)
    out = out_flat.permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate, final projection
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
