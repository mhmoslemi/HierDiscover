"""
TriMul (outgoing) optimized with Triton
---------------------------------------

The implementation follows the reference PyTorch module but off‑loads the
computationally heavy pairwise multiplication

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

to a custom Triton kernel.  The kernel evaluates one output tile
(i‑block × j‑block) for a given `(batch, hidden)` pair.  Each thread in the
tile computes one element of the output (i,j) and accumulates a dot‑product
over the ``k`` dimension using a simple outer‑product accumulation.
Only the summation over ``k`` is executed on the GPU; the layer‑norms,
linear projections and gating are performed with regular PyTorch
operations.

The kernel arguments include explicit strides, so the same code works for
any batch size, sequence length and hidden dimension that appear in the
test suite.  The outer Python function ``custom_kernel`` prepares the
tensors, launches the kernel and finishes the forward pass (normalisation
and final linear projection).  No gradients are computed.

The implementation is compatible with Triton 3.3.1 on an H100 and works
for both masked and un‑masked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: computes   out = Σ_k left[...,i,k,d] * right[...,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,            # const * left  (B,N,N,H)
    right_ptr,           # const * right (B,N,N,H)
    out_ptr,             # * out (B,N,N,H)

    # tensor metadata
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Grid layout
    #   pid_i : tile index over the i‑axis (rows)
    #   pid_j : tile index over the j‑axis (cols)
    #   pid_bh: packed (batch, hidden) index
    # ------------------------------------------------------------------
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # ------------------------------------------------------------------
    # Compute actual i, j coordinates for the tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Decode batch and hidden‑channel index from pid_bh
    # ------------------------------------------------------------------
    bh = pid_bh
    b = bh // H                 # batch index
    h = bh % H                  # hidden (channel) index

    # ------------------------------------------------------------------
    # Base pointers for this (batch, hidden) slice
    # ------------------------------------------------------------------
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # ------------------------------------------------------------------
    # Accumulator for the output tile (float32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction axis k in blocks of size BLOCK_K
    # ------------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k):
        # k indices for this block
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # --------------------------------------------------------------
        # Load a BLOCK_M × BLOCK_K tile from left  (i,k)
        # --------------------------------------------------------------
        a_ptrs = left_base \
                 + offs_i[:, None] * stride_left_i \
                 + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)   # (BLOCK_M, BLOCK_K)

        # --------------------------------------------------------------
        # Load a BLOCK_N × BLOCK_K tile from right (j,k)
        # --------------------------------------------------------------
        # We need right[j,k] → we store it with row=j, col=k.
        b_ptrs = right_base \
                 + offs_j[:, None] * stride_right_j \
                 + offs_k[None, :] * stride_right_k
        b = tl.load(b_ptrs,
                    mask=mask_j[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)   # (BLOCK_N, BLOCK_K)

        # --------------------------------------------------------------
        # Accumulate Σ_k left[i,k] * right[j,k] for this k‑block.
        #   For each k element we compute the outer product a[:,k] ⊙ b[:,k].
        # --------------------------------------------------------------
        for s in range(0, BLOCK_K):
            a_col = a[:, s]               # (BLOCK_M,)
            b_col = b[:, s]               # (BLOCK_N,)
            # outer product and accumulate
            acc += a_col[:, None] * b_col[None, :]

    # ------------------------------------------------------------------
    # Write the tile back to memory (respecting bounds)
    # ------------------------------------------------------------------
    out_ptrs = out_base \
               + offs_i[:, None] * stride_out_i \
               + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Python entry‑point: builds the full TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : Tuple containing
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask_tensor   : torch.Tensor of shape [B, N, N] (may be all‑ones)
        - weights       : dict of model parameters (see keys below)
        - config        : dict with "dim", "hidden_dim", and "nomask" flag

    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32) – the processed pairwise
    representation.
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Optional mask handling – mask may be omitted (nomask=True)
    # ------------------------------------------------------------------
    nomask = config.get("nomask", False)
    if nomask or mask_tensor is None:
        mask = torch.ones((B, N, N), dtype=input_tensor.dtype, device=input_tensor.device)
    else:
        mask = mask_tensor
    mask = mask.unsqueeze(-1)                          # [B,N,N,1]

    # ------------------------------------------------------------------
    # Helper to fetch weight tensors
    # ------------------------------------------------------------------
    w = weights
    # LayerNorm over the input channel dimension
    norm_weight = w["norm.weight"]
    norm_bias   = w["norm.bias"]
    # Linear projections
    left_proj_w   = w["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_w  = w["right_proj.weight"]
    # Gating linear layers (no bias)
    left_gate_w   = w["left_gate.weight"]
    right_gate_w  = w["right_gate.weight"]
    out_gate_w    = w["out_gate.weight"]
    # Output norm and final projection
    to_out_norm_w = w["to_out_norm.weight"]
    to_out_norm_b = w["to_out_norm.bias"]
    to_out_w      = w["to_out.weight"]      # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and gating values
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)                     # [B,N,N,hidden_dim]
    right_proj = F.linear(x, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 3. Apply mask and gates
    # ------------------------------------------------------------------
    left  = left_proj * mask
    right = right_proj * mask

    left  = left * left_gate
    right = right * right_gate

    # Ensure contiguity for the Triton kernel
    left  = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 4. Triton kernel: pairwise multiplication over the third axis
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim),
                      dtype=torch.float32,
                      device=input_tensor.device)

    # Strides in elements (not bytes) – required by Triton
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tile sizes – chosen to keep the accumulator (BLOCK_M×BLOCK_N) within
    # register limits while still providing good occupancy.
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    # Grid dimensions
    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    # Combine batch and hidden dimension into a single program‑id axis
    grid_bh = B * hidden_dim

    tri_mul_outgoing_kernel[(grid_i, grid_j, grid_bh)](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                # reasonable default for H100
    )

    # ------------------------------------------------------------------
    # 5. Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=eps)

    # ------------------------------------------------------------------
    # 6. Apply output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_w)          # [B,N,N,dim]

    return out
