@triton.jit
def tril_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    # program ids for output tiles (i,j) and hidden dim? We'll incorporate hidden dim inside.
    pid_b = tl.program_id(0) # batch index
    pid_i = tl.program_id(1) # row block index
    pid_j = tl.program_id(2) # col block index

    # offsets for block start i and j
    i_offsets = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offsets = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out-of-bound i, j
    mask_i = i_offsets < N
    mask_j = j_offsets < N

    # allocate accumulator for hidden dim
    # shape (BLOCK_M, BLOCK_N, BLOCK_H)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute actual k offsets
        k_offsets = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < N

        # left[batch, i, k, h] -> shape (BLOCK_M, BLOCK_K, BLOCK_H)
        left_ptrs = left_ptr + pid_b * stride_left_b + i_offsets[:, None, None] * stride_left_i + k_offsets[None, :, None] * stride_left_k + tl.arange(0, BLOCK_H)[None, None, :] * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0) # shape (BLOCK_M, BLOCK_K, BLOCK_H)

        # right[batch, j, k, h] -> shape (BLOCK_N, BLOCK_K, BLOCK_H)
        right_ptrs = right_ptr + pid_b * stride_right_b + j_offsets[:, None, None] * stride_right_i + k_offsets[None, :, None] * stride_right_k + tl.arange(0, BLOCK_H)[None, None, :] * stride_right_h

        # Need right transposed dimension: we have right as (j,k,h). Want to multiply left[i,k,h] * right[j,k,h]
        # For each h, we need sum over k: left[i,k,h] * right[j,k,h]
        # So we can multiply left and right broadcasting over i and j dimensions.

        # Expand left to (BLOCK_M, 1, BLOCK_K, BLOCK_H)
        # Expand right to (1, BLOCK_N, BLOCK_K, BLOCK_H)
        # Then multiply and sum over k

        # Use broadcasting: left unsqueezed, right unsqueezed:

@triton.jit
def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_K: tl.constexpr, H: tl.constexpr
):
    # program ids
    pid_b = tl.program_id(0)   # batch
    pid_i = tl.program_id(1)   # i index
    pid_j = tl.program_id(2)   # j index

    # bounds check
    if pid_i >= N or pid_j >= N:
        return

    # offsets
    i = pid_i
    j = pid_j
    b = pid_b

    # compute base pointer for left: left[b, i, :, :] i fixed, k varies
    # compute base pointer for right: right[b, j, :, :] j fixed, k varies

    # accumulate across k
    acc = tl.zeros((H,), dtype=tl.float32)

    # hidden dim offsets
    h_offsets = tl.arange(0, H)

    # loop over k with tile size BLOCK_K
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        # compute actual length of this tile
        cur_k = tl.minimum(N - k_start, BLOCK_K)
        # create range
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        # mask for k beyond N
        mask_k = k_offsets < N

        # pointer to left: shape (BLOCK_K, H)
        left_ptrs = left_ptr + b*stride_left_b + i*stride_left_i + k_offsets[:, None]*stride_left_k + h_offsets[None, :] * stride_left_h
        left_tile = tl.load(left_ptrs, mask=mask_k[:, None] & (h_offsets < H), other=0.0) # shape (BLOCK_K, H)

        # pointer to right
        right_ptrs = right_ptr + b*stride_right_b + j*stride_right_i + k_offsets[:, None]*stride_right_k + h_offsets[None,:] * stride_right_h
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & (h_offsets < H), other=0.0)

        # elementwise multiply left_tile and right_tile, sum across k dimension
        # Could use tl.sum(left_tile * right_tile, axis=0)
        acc += tl.sum(left_tile * right_tile, axis=0)

    # store result
    out_ptrs = out_ptr + b*stride_out_b + i*stride_out_i + j*stride_out_j + h_offsets*stride_out_h
    tl.store(out_ptrs, acc, mask=h_offsets < H)

@triton.jit
def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_K: tl.constexpr,
    H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    if pid_i >= N or pid_j >= N:
        return
    # Compute base pointers for this batch
    left_base = left_ptr + pid_b * stride_left_b
    right_base = right_ptr + pid_b * stride_right_b
    out_base = out_ptr + pid_b * stride_out_b

    i = pid_i
    j = pid_j

    # Offsets for i and j
    i_offset = i * stride_left_i
    j_offset = j * stride_right_i

    # Hidden offsets
    h_offsets = tl.arange(0, H)

    # Initialize accumulator
    acc = tl.zeros((H,), dtype=tl.float32)

    # Loop over k
    for k_start in range(0, tl.cdiv(N, BLOCK_K)):
        # compute actual offset vector for k within block
        k = k_start * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k < N

        # left pointer for this tile (shape: (BLOCK_K, H))
        left_ptrs = left_base + i_offset + k[:, None] * stride_left_k + h_offsets[None, :] * stride_left_h
        left_tile = tl.load(left_ptrs, mask=k_mask[:, None] & (h_offsets < H), other=0.0).to(tl.float32)  # cast to float32

        # right pointer for this tile (shape: (BLOCK_K, H))
        right_ptrs = right_base + j_offset + k[:, None] * stride_right_k + h_offsets[None, :] * stride_right_h
        right_tile = tl.load(right_ptrs, mask=k_mask[:, None] & (h_offsets < H), other=0.0).to(tl.float32)

        # accumulate
        acc += tl.sum(left_tile * right_tile, axis=0)

    # Store result
    out_ptrs = out_base + i * stride_out_i + j * stride_out_j + h_offsets * stride_out_h
    tl.store(out_ptrs, acc.to(tl.float16) if out_ptr.dtype == tl.float16 else acc, mask=h_offsets < H)

"""
TriMul “outgoing” kernel (AlphaFold‑3 style).

The forward pass consists of:
  1. Layer‑norm over the channel dimension `dim`.
  2. Linear projections to a hidden dimension `hidden_dim` (left / right).
  3. Optional binary mask (broadcasted) and three sigmoid‑gates.
  4. A triangular multiplication:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
     (the heavy N³·hidden_dim work‑load).  This reduction is implemented
     with a Triton kernel where each program instance computes the full
     hidden‑vector for a single (b,i,j) pair, looping over the k‑dimension.
  5. Layer‑norm on the hidden dimension, element‑wise multiplication by the
     output‑gate, and a final linear projection back to `dim`.

All heavy arithmetic (step 4) runs on H100 in fp16 for speed, while
accumulation is performed in fp32 to preserve numeric quality.
The rest of the pipeline uses PyTorch (which already benefits from
high‑performance cuBLAS kernels for the linear layers).
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: compute the outgoing triangle multiplication.
#   left  : [B, N, N, H]   (fp16)
#   right : [B, N, N, H]   (fp16)
#   out   : [B, N, N, H]   (fp16, written)
#   out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
# ----------------------------------------------------------------------
@triton.jit
def trmul_outgoing_kernel(
    left_ptr,          # *fp16
    right_ptr,         # *fp16
    out_ptr,           # *fp16
    B, N, H,           # runtime sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_K: tl.constexpr,   # block size along the reduction axis k
    H: tl.constexpr,         # hidden dimension (compile‑time constant)
):
    pid_b = tl.program_id(0)          # batch
    pid_i = tl.program_id(1)          # row index i
    pid_j = tl.program_id(2)          # column index j

    # Guard against over‑launch (when N is not a multiple of the grid size)
    if pid_i >= N or pid_j >= N:
        return

    # ------------------------------------------------------------------
    # Base pointers for this (b,i) and (b,j) pair.
    # ------------------------------------------------------------------
    left_base  = left_ptr  + pid_b * stride_left_b
    right_base = right_ptr + pid_b * stride_right_b
    out_base   = out_ptr   + pid_b * stride_out_b

    # Offsets for the fixed i / j within the batch.
    i_off = pid_i * stride_left_i
    j_off = pid_j * stride_right_i

    # ------------------------------------------------------------------
    # Registers for the hidden‑vector accumulator (fp32 for numerics).
    # ------------------------------------------------------------------
    h_idx = tl.arange(0, H)                     # [H]
    acc   = tl.zeros((H,), dtype=tl.float32)    # accumulator

    # ------------------------------------------------------------------
    # Loop over k in tiles of size BLOCK_K.
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tile in range(num_k_tiles):
        start_k = tile * BLOCK_K
        k_idx   = start_k + tl.arange(0, BLOCK_K)       # [BLOCK_K]
        k_mask  = k_idx < N                              # out‑of‑bounds guard

        # --------------------------------------------------------------
        # Load left[b,i,k,:]  → shape (BLOCK_K, H)
        # --------------------------------------------------------------
        left_ptrs = left_base + i_off \
                    + k_idx[:, None] * stride_left_k \
                    + h_idx[None, :] * stride_left_h
        left_tile = tl.load(left_ptrs,
                            mask=k_mask[:, None],
                            other=0.0).to(tl.float32)   # fp16 → fp32

        # --------------------------------------------------------------
        # Load right[b,j,k,:] → shape (BLOCK_K, H)
        # --------------------------------------------------------------
        right_ptrs = right_base + j_off \
                     + k_idx[:, None] * stride_right_k \
                     + h_idx[None, :] * stride_right_h
        right_tile = tl.load(right_ptrs,
                             mask=k_mask[:, None],
                             other=0.0).to(tl.float32)

        # --------------------------------------------------------------
        # Element‑wise product and reduction over k.
        # --------------------------------------------------------------
        acc += tl.sum(left_tile * right_tile, axis=0)   # shape (H,)

    # ------------------------------------------------------------------
    # Store result back to memory (fp16 store, keep fp32 accumulation).
    # ------------------------------------------------------------------
    out_ptrs = out_base \
               + pid_i * stride_out_i \
               + pid_j * stride_out_j \
               + h_idx * stride_out_h
    tl.store(out_ptrs, acc.to(tl.float16))


# ----------------------------------------------------------------------
# Entrypoint expected by the evaluator.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N]   (bool / 0‑1 float)
        - weights      : dict of torch.Tensor (all weight/bias tensors)
        - config       : dict containing ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]          # == H (always 128 in tests)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # x : [B, N, N, dim] (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left  = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted) – the reference code multiplies
    #     the mask *before* the gates.
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply gates
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Prepare tensors for the Triton kernel (fp16 for speed)
    # ------------------------------------------------------------------
    left  = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()
    # Output buffer (fp16) – will be cast back to fp32 later.
    out = torch.empty(
        (left.shape[0], left.shape[1], left.shape[2], hidden_dim),
        dtype=torch.float16,
        device=device,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Triton launch
    # ------------------------------------------------------------------
    B, N = left.shape[0], left.shape[1]   # note: left.shape = [B,N,N,H]
    # Strides in elements (not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Grid: one program per (b,i,j) pair.
    grid = (B, N, N)   # (batch, i, j)

    # Block size along k (tuned for the H100, 32 gives good occupancy)
    BLOCK_K = 32

    trmul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_K=BLOCK_K,
        H=hidden_dim,
        num_warps=4,               # modest number of warps for fp16 work‑load
    )

    # Back to fp32 for the remaining layers
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Hidden‑dim layer‑norm, out‑gate, final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # Apply the output gate (same shape as out)
    out = out * out_gate

    # Final linear projection back to `dim`
    to_out_weight = weights["to_out.weight"]   # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)

    return out
