"""
custom_kernel: Highly‑optimised forward of the TriMul “outgoing” operator.

Algorithm
---------
1. Layer‑norm the input tensor `x` over the last dimension (`dim`).
2. Linear projections → `left` and `right` (dim → hidden_dim, no bias).
3. (Optional) Apply the pairwise mask.
4. Compute three sigmoidal gates (`left_gate`, `right_gate`, `out_gate`) from the
   same normalized input.
5. Fuse the gates:  left  = left  * left_gate,  right = right * right_gate.
6. Cast `left`/`right` to float16 and rearrange to shape (B, H, N, N).
   The core operation is a batched matrix multiply:
       out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
   i.e. a GEMM per (batch,head) pair with matrices of size N×N.
   This is performed by a custom Triton kernel that a) tiles the work,
   b) uses `tl.dot` (tensor‑core friendly) and c) accumulates in float32.
7. Reshape the result back to (B, N, N, hidden_dim) and cast to float32.
8. Layer‑norm the result over `hidden_dim`, multiply by `out_gate`,
   and project back to the original `dim` with a final linear layer.
9. Return the tensor of shape (B, N, N, dim) (float32).

The kernel is written for mixed‑precision (fp16 computation, fp32 accumulation)
and automatically works for any batch size `B∈{1,2}`, sequence length `N`,
and hidden dimension up to 128.  It respects the `nomask` flag in the config.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the inner TriMul multiplication.
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A,                     # [batch, M, K]  left  matrix (fp16)
    B,                     # [batch, K, N]  right matrix (fp16) – already transposed
    C,                     # [batch, M, N]  output (fp16)
    M, N, K,               # matrix sizes (all == seq_len)
    stride_ab, stride_am, stride_ak,   # strides for A
    stride_bb, stride_bk, stride_bn,   # strides for B
    stride_cb, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   (A: M×K, B: K×N) for many batches."""
    pid_b = tl.program_id(2)                     # batch index (B * hidden_dim)
    pid_m = tl.program_id(0)                     # block row
    pid_n = tl.program_id(1)                     # block column

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds accesses
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tile in global memory
    a_ptrs = A + pid_b * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B + pid_b * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0).to(tl.float16)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float16)
        acc += tl.dot(a, b)          # FP32 accumulation
        # Advance to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store the result, casting back to fp16
    c_ptrs = C + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

    Forward pass of the TriMul “outgoing” operator using a custom Triton kernel

    x, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (over the last dimension)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # 2. Linear projections (dim → hidden_dim)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # 3. (Optional) Masking – mask shape [B,N,N]
    # ------------------------------------------------------------------
    if not nomask and mask is not None:

        left = left * mask_
        right = right * mask_

    # 4. Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # 5. Fuse gates
    left = left * left_gate          # [B,N,N,hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Prepare for the batched GEMM (cast to fp16, permute, transpose)
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_h = right.to(torch.float16).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]
    # Transpose the last two dimensions of `right` -> K × N for matmul
    right_h_T = right_h.transpose(-2, -1).contiguous()                # [B, H, N, N] (K,N)

    B_ = right_h_T.view(batch_cnt, N, N)                # [batch_cnt, K, N]

    C = torch.empty_like(A, dtype=torch.float16, device=device)   # [batch_cnt, M, N]

    stride_ab, stride_am, stride_ak = A.stride()        # batch, M, K
    stride_bb, stride_bk, stride_bn = B_.stride()       # batch, K, N

    BLOCK_N = 64
    BLOCK_K = 64

        (N + BLOCK_M - 1) // BLOCK_M,   # blocks over rows (i)
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks over columns (j)
        batch_cnt,                      # one block per (batch, head)
    )

    _batched_gemm_kernel[grid](
        A,
        B_,
        C,
        M=N,
        N=N,
        K=N,
        stride_ab=stride_ab,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bb=stride_bb,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cb=stride_cb,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8. Reshape back to (B, N, N, hidden_dim) and cast to FP32
    # ------------------------------------------------------------------
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------

        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # [B,N,N,hidden_dim]  FP32

    out = out * out_gate                               # element‑wise gating

    # Final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])       # [B,N,N,dim]

    return out
```
