```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_k, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    # Get program IDs
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)
    # Compute batch and channel block index
    num_h_blocks = tl.cdiv(H, BLOCK_H)
    b = pid_bh // num_h_blocks
    h_block = pid_bh % num_h_blocks

    # Ensure b < B
    if b >= B:
        return

    # Compute offsets
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    h0 = h_block * BLOCK_H

    # Offsets
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_h = h0 + tl.arange(0, BLOCK_H)

    # Mask for within bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # Loop over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block: shape (M, K, H)
        left_ptrs = left_ptr + b * stride_left_b \
            + offs_i[:, None, None] * stride_left_i \
            + offs_k[None, :, None] * stride_left_k \
            + offs_h[None, None, :] * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0)

        # Load right block (transposed): shape (K, N, H)
        right_ptrs = right_ptr + b * stride_right_b \
            + offs_k[:, None, None] * stride_right_k \
            + offs_j[None, :, None] * stride_right_j \
            + offs_h[None, None, :] * stride_right_h
        right = tl.load(right_ptrs, mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :], other=0.0)

        # Now compute dot for each h
        # left: (M,K,H); right: (K,N,H)
        # We'll iterate over h internal
        for h in range(0, BLOCK_H):
            # Check h bound
            if h0 + h >= H:
                continue
            # left slice shape (M,K)
            left_slice = left[:, :, h]  # (M,K)
            right_slice = right[:, :, h]  # (K,N)
            # dot
            acc[:, :, h] += tl.dot(left_slice, right_slice)

    # Write output
    out_ptrs = out_ptr + b * stride_out_b \
        + offs_i[:, None, None] * stride_out_i \
        + offs_j[None, :, None] * stride_out_j \
        + offs_h[None, None, :] * stride_out_h
    # Store only within bounds
    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

def mask_gate_fusion(
    proj_ptr, mask_ptr, gate_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_H: tl.constexpr
):
    b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N
    for start_h in range(0, H, BLOCK_H):
        offs_h = start_h + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        # compute pointers
        proj_ptrs = proj_ptr + b*stride_proj_b + offs_i[:,None,None]*stride_proj_i + offs_j[None,:,None]*stride_proj_j + offs_h[None,None,:]*stride_proj_h

        gate_ptrs = gate_ptr + b*stride_gate_b + offs_i[:,None,None]*stride_gate_i + offs_j[None,:,None]*stride_gate_j + offs_h[None,None,:]*stride_gate_h

        proj = tl.load(proj_ptrs, mask=mask_i[:,None,None] & mask_j[None,:,None] & mask_h[None,None,:], other=0.0)
        mask_val = tl.load(mask_ptrs, mask=mask_i[:,None] & mask_j[None,:], other=0.0)
        mask_val = mask_val[:, :, None]  # broadcast
        gate = tl.load(gate_ptrs, mask=mask_i[:,None,None] & mask_j[None,:,None] & mask_h[None,None,:], other=0.0)

@triton.jit
def mask_gate_fuse_kernel(
    left_proj_ptr, right_proj_ptr,
    mask_ptr,
    left_gate_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_h,
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_outl_b, stride_outl_i, stride_outl_j, stride_outl_h,
    stride_outr_b, stride_outr_i, stride_outr_j, stride_outr_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_H: tl.constexpr
):
    b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    for offset_h in range(0, H, BLOCK_H):
        offs_h = offset_h + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load left_proj and right_proj
        left_ptrs = left_proj_ptr + b*stride_lp_b + offs_i[:,None,None]*stride_lp_i + offs_j[None,:,None]*stride_lp_j + offs_h[None,None,:]*stride_lp_h
        right_ptrs = right_proj_ptr + b*stride_rp_b + offs_i[:,None,None]*stride_rp_i + offs_j[None,:,None]*stride_rp_j + offs_h[None,None,:]*stride_rp_h

        left_val = tl.load(left_ptrs, mask=mask_i[:,None,None] & mask_j[None,:,None] & mask_h[None,None,:], other=0.0)
        right_val = tl.load(right_ptrs, mask=mask_i[:,None,None] & mask_j[None,:,None] & mask_h[None,None,:], other=0.0)

        # Load mask
        mask_ptrs = mask_ptr + b*stride_mask_b + offs_i[:,None]*stride_mask_i + offs_j[None,:]*stride_mask_j
        mask_val = tl.load(mask_ptrs, mask=mask_i[:,None] & mask_j[None,:], other=0.0)
        mask_val = mask_val[:, :, None]  # broadcast to (M,N,1)

        # Load gates
        left_gate_ptrs = left_gate_ptr + b*stride_lg_b + offs_i[:,None,None]*stride_lg_i + offs_j[None,:,None]*stride_lg_j + offs_h[None,None,:]*stride_lg_h
        right_gate_ptrs = right_gate_ptr + b*stride_rg_b + offs_i[:,None,None]*stride_rg_i + offs_j[None,:,None]*stride_rg_j + offs_h[None,None,:]*stride_rg_h

        left_gate = tl.load(left_gate_ptrs, mask=mask_i[:,None,None] & mask_j[None,:,None] & mask_h[None,None,:], other=0.0)
        right_gate = tl.load(right_gate_ptrs, mask=mask_i[:,None,None] & mask_j[None,:,None] & mask_h[None,None,:], other=0.0)

        # Compute
        out_left = left_val * mask_val * left_gate
        out_right = right_val * mask_val * right_gate

        # Store
        out_left_ptrs = out_left_ptr + b*stride_outl_b + offs_i[:,None,None]*stride_outl_i + offs_j[None,:,None]*stride_outl_j + offs_h[None,None,:]*stride_outl_h
        out_right_ptrs = out_right_ptr + b*stride_outr_b + offs_i[:,None,None]*stride_outr_i + offs_j[None,:,None]*stride_outr_j + offs_h[None,None,:]*stride_outr_h

        tl.store(out_left_ptrs, out_left, mask=mask_i[:,None,None] & mask_j[None,:,None] & mask_h[None,None,:])
        tl.store(out_right_ptrs, out_right, mask=mask_i[:,None,None] & mask_j[None,:,None] & mask_h[None,None,:])
```

"""
TriMul (outgoing) implementation with a fused mask‑gate Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B, N, N, C]).
2. Linear projections:   left = W_l·x , right = W_r·x .
3. Compute three sigmoidal gates from the same normalized `x`:
   left_gate, right_gate, out_gate.
4. Fuse mask (if present), projection and gate multiplication with a
   Triton kernel:
       left_gated  = left  * mask * left_gate
       right_gated = right * mask * right_gate
   The kernel processes a tile of (i, j, h) per thread‑block and loops
   over the hidden dimension in small blocks to keep register pressure low.
5. For each hidden channel the operation
       out[..., i, j, h] = Σ_k left_gated[..., i, k, h] * right_gated[..., j, k, h]
   is exactly a batched matrix multiplication (left @ right^T).  After
   transposing `right_gated` we reshape the tensors to (B*H, N, N) and
   invoke `torch.bmm`, which uses cuBLAS/Tensor‑cores on H100.
6. Apply a second LayerNorm (`to_out_norm`), multiply by `out_gate`,
   and linearly project back to the original channel dimension with
   `to_out`.
7. Return the final tensor of shape [B, N, N, C].

The heavy element‑wise part (mask & gates) is executed in Triton,
while the cubic‑complexity reduction (the true TriMul) leverages the
highly‑optimized batched GEMM in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: fuse mask, projection and gate multiplication.
# -------------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    # input tensors
    proj_ptr,                # left or right projection               [B, N, N, H]
    mask_ptr,                # mask (no channel dim)                  [B, N, N]
    gate_ptr,                # corresponding gate                     [B, N, N, H]
    # output tensor
    out_ptr,                 # gated & masked result                 [B, N, N, H]
    # sizes
    B, N, H,
    # strides (in elements)
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,   # tile for i dimension
    BLOCK_N: tl.constexpr,   # tile for j dimension
    BLOCK_H: tl.constexpr,   # tile for hidden dimension
):
    # program ids
    b = tl.program_id(0)                 # batch
    pid_i = tl.program_id(1)             # i‑tile
    pid_j = tl.program_id(2)             # j‑tile

    # i / j offsets for this tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # loop over hidden dimension in blocks
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # ----- load projection -----
        proj_ptrs = proj_ptr \
            + b * stride_proj_b \
            + offs_i[:, None, None] * stride_proj_i \
            + offs_j[None, :, None] * stride_proj_j \
            + offs_h[None, None, :] * stride_proj_h
        proj = tl.load(
            proj_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # ----- load gate -----
        gate_ptrs = gate_ptr \
            + b * stride_gate_b \
            + offs_i[:, None, None] * stride_gate_i \
            + offs_j[None, :, None] * stride_gate_j \
            + offs_h[None, None, :] * stride_gate_h
        gate = tl.load(
            gate_ptrs,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # ----- load mask (broadcast over hidden) -----
        mask_ptrs = mask_ptr \
            + b * stride_mask_b \
            + offs_i[:, None] * stride_mask_i \
            + offs_j[None, :] * stride_mask_j
        m = tl.load(
            mask_ptrs,
            mask=mask_i[:, None] & mask_j[None, :],
            other=0.0,
        )
        m = m[:, :, None]   # broadcast to (M, N, 1)

        # ----- fused multiply -----
        out_val = proj * m * gate

        # ----- store result -----
        out_ptrs = out_ptr \
            + b * stride_out_b \
            + offs_i[:, None, None] * stride_out_i \
            + offs_j[None, :, None] * stride_out_j \
            + offs_h[None, None, :] * stride_out_h
        tl.store(
            out_ptrs,
            out_val,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


# -------------------------------------------------------------------------
# Custom kernel entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
              - input_tensor: torch.Tensor, shape [B, N, N, C]
              - mask_tensor: torch.Tensor, shape [B, N, N] (may be None)
              - weights_dict: mapping of weight names to torch.Tensors
              - config_dict: must contain "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # -----------------------------------------------------------------
    # unpack
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]  # c_z in the paper

    # -----------------------------------------------------------------
    # fetch model parameters
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # LayerNorm over the channel dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(C,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, C]

    # -----------------------------------------------------------------
    # Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj = F.linear(x, left_proj_weight)   # [B, N, N, hidden]
    right_proj = F.linear(x, right_proj_weight) # [B, N, N, hidden]

    # -----------------------------------------------------------------
    # Gating values (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B, N, N, hidden]
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, N, N, hidden]

    # -----------------------------------------------------------------
    # Prepare mask (float, broadcastable)
    # -----------------------------------------------------------------
    if mask_tensor is None:
        mask = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask = mask_tensor.to(dtype=dtype)
    # ensure contiguous for kernel stride extraction
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask = mask.contiguous()

    # -----------------------------------------------------------------
    # Allocate tensors for the fused result
    # -----------------------------------------------------------------
    left_gated = torch.empty_like(left_proj)
    right_gated = torch.empty_like(right_proj)

    # -----------------------------------------------------------------
    # Triton launch: fuse mask, projection and gate multiplication
    # -----------------------------------------------------------------
    # tile sizes – chosen to keep register pressure low on H100
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_H = 8

    grid = (
        B,                                   # batch
        (N + BLOCK_M - 1) // BLOCK_M,        # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,        # j‑tiles
    )

    _mask_gate_fuse_kernel[grid](
        # pointers
        left_proj, mask, left_gate, left_gated,
        # sizes
        B, N, hidden_dim,
        # strides (elementwise)
        *left_proj.stride(),
        *mask.stride(),
        *left_gate.stride(),
        *left_gated.stride(),
        # compile‑time constants
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_H=BLOCK_H,
    )
    _mask_gate_fuse_kernel[grid](
        right_proj, mask, right_gate, right_gated,
        B, N, hidden_dim,
        *right_proj.stride(),
        *mask.stride(),
        *right_gate.stride(),
        *right_gated.stride(),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_H=BLOCK_H,
    )

    # -----------------------------------------------------------------
    # Core TriMul: batched GEMM (left @ right^T) per hidden channel
    # -----------------------------------------------------------------
    # reshape to (B*H, N, N) – left as (M, K), right transposed as (K, N)
    left_mat = left_gated.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # swap the second and third axes of right_gated so that we have (K, J)
    right_mat = right_gated.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # batched matrix multiplication (cublas / TensorCore)
    out_mat = torch.bmm(left_mat, right_mat)   # [B*H, N, N]

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # Output layer‑norm, gate, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B, N, N, C]

    return out
