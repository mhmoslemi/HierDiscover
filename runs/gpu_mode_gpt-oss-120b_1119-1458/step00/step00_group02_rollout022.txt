    right_fp16 = right.to(torch.float16)
    # Allocate output tensor
    out = torch.empty_like(left_fp16, dtype=torch.float32, device=device)  # shape [B,N,N,hidden_dim]

    B, N, N2, D = left_fp16.shape
    assert N == N2
    # Pointer arguments
    left_ptr = left_fp16
    right_ptr = right_fp16
    out_ptr = out

    # Get strides
    stride_b = left_fp16.stride(0)
    stride_i = left_fp16.stride(1)
    stride_k = left_fp16.stride(2)
    stride_d = left_fp16.stride(3)

    out_stride_b = out_ptr.stride(0)
    out_stride_i = out_ptr.stride(1)
    out_stride_j = out_ptr.stride(2)
    out_stride_d = out_ptr.stride(3)

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32  # depends on N
    # Compute grid
    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_z = B * hidden_dim  # each (b,d) pair
    grid = (num_pid_m, num_pid_n, total_z)

    # Launch kernel
    tri_mul_kernel[grid](
        left_ptr,
        right_ptr,
        out_ptr,
        B, N, hidden_dim,
        stride_b, stride_i, stride_k, stride_d,
        out_stride_b, out_stride_i, out_stride_j, out_stride_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4
    )
    # Now out is [B,N,N,hidden_dim] float32
    # Apply to_out_norm (layernorm over hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Multiply by out_gate
    out_gate = out_gate.to(out_norm.dtype)
    out_norm = out_norm * out_gate

    # Final linear projection to dim
    to_out_weight = weights["to_out.weight"]
    out_final = torch.nn.functional.linear(out_norm, to_out_weight)

    return out_final
```

"""
TriMul forward operator (outgoing version) with a fused Triton kernel.

The heavy pairwise update
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched matrix multiplication:
    left:  (B,D,N,N) → (B*D,N,N)
    right: (B,D,N,N) → (B*D,N,N) and transposed on the second N‑axis.
Each (b,d) pair runs its own matmul kernel in Triton, using fp16
loads and Tensor‑Core accumulation in fp32.  The rest of the
TriMul pipeline (LayerNorms, linear projections, gating and the final
output projection) is performed with PyTorch (float32) for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul for a single (batch, hidden) slice.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    left_ptr,          # *fp16 left  [B,N,N,D]
    right_ptr,         # *fp16 right [B,N,N,D]
    out_ptr,           # *fp32 out   [B,N,N,D]
    B, N, D,
    stride_b, stride_i, stride_k, stride_d,          # strides for left/right
    out_stride_b, out_stride_i, out_stride_j, out_stride_d,  # strides for out
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # ----- program IDs -------------------------------------------------
    pid_m = tl.program_id(0)          # tile row (i)
    pid_n = tl.program_id(1)          # tile col (j)
    pid_z = tl.program_id(2)          # combines batch and hidden channel

    # decode batch index and hidden channel index
    d = pid_z % D               # hidden dimension index
    b = pid_z // D              # batch index

    # ----- tile offsets ------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for the current (b,d) slice (no k offset yet)
    a_base = left_ptr \
        + b * stride_b \
        + d * stride_d \
        + offs_m[:, None] * stride_i          # shape (BLOCK_M, 1)

    b_base = right_ptr \
        + b * stride_b \
        + d * stride_d \
        + offs_n[:, None] * stride_i          # shape (BLOCK_N, 1)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k_tiles):
        k_offset = k_iter * BLOCK_K

        # valid k mask for this tile
        cur_k_valid = (k_offset + offs_k) < N

        # ----- load left tile (M x K) ---------------------------------
        a = tl.load(
            a_base + (k_offset + offs_k)[None, :] * stride_k,
            mask=mask_m[:, None] & cur_k_valid[None, :],
            other=0.0
        ).to(tl.float16)

        # ----- load right tile (N x K) --------------------------------
        b_tile = tl.load(
            b_base + (k_offset + offs_k)[:, None] * stride_k,
            mask=mask_n[:, None] & cur_k_valid[None, :],
            other=0.0
        ).to(tl.float16)

        # transpose right to (K x N) for dot
        b_tile = tl.trans(b_tile, (1, 0))

        # ----- accumulate ------------------------------------------------
        acc += tl.dot(a, b_tile)      # dot --> fp32 accumulation

    # ----- store result -------------------------------------------------
    out_ptrs = out_ptr \
        + b * out_stride_b \
        + d * out_stride_d \
        + offs_m[:, None] * out_stride_i \
        + offs_n[None, :] * out_stride_j

    store_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(out_ptrs, acc, mask=store_mask)


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim] (float32)
            - mask_tensor:  [B, N, N]   (bool or float, may be None)
            - weights_dict: the model weights (torch tensors)
            - config_dict:  {"dim": dim, "hidden_dim": hidden_dim}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # LayerNorm on the input (last dim = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_ = mask.to(x.dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # Gating values (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast projections to fp16 for the Triton kernel
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # output tensor (float32) for the kernel
    out = torch.empty_like(left_fp16, dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Strides (must be plain Python ints)
    # ------------------------------------------------------------------
    stride_b = left_fp16.stride(0)
    stride_i = left_fp16.stride(1)
    stride_k = left_fp16.stride(2)
    stride_d = left_fp16.stride(3)

    out_stride_b = out.stride(0)
    out_stride_i = out.stride(1)
    out_stride_j = out.stride(2)
    out_stride_d = out.stride(3)

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32  # good balance for fp16 Tensor Cores

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_z = B * hidden_dim  # one program ID per (batch, hidden)

    grid = (num_pid_m, num_pid_n, total_z)

    # launch kernel
    tri_mul_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        hidden_dim,
        stride_b, stride_i, stride_k, stride_d,
        out_stride_b, out_stride_i, out_stride_j, out_stride_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # Output post‑processing
    # ------------------------------------------------------------------
    # LayerNorm on hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Multiply by the output gating values
    out = out * out_gate

    # Final linear back to `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out
