import triton
import triton.language as tl

@triton.jit
def trmul_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_j,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    EPS: tl.constexpr
):
    # Compute batch index
    batch_idx = tl.program_id(axis=2)   # combined batch index (b*H + h)
    # Compute block indices for M and N
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute mask for valid rows and columns
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Compute base pointers for this batch
    left_batch_ptr = left_ptr + batch_idx * stride_left_batch
    right_batch_ptr = right_ptr + batch_idx * stride_right_batch
    out_batch_ptr = out_ptr + batch_idx * stride_out_batch

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block (M x K)
        left_ptrs = left_batch_ptr + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right block transposed (K x N) -> using right as (K,N)
        # row stride = stride_right_k (original second dim stride), col stride = stride_right_j (original first dim stride)
        right_ptrs = right_batch_ptr + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Perform dot product
        acc += tl.dot(left, right)  # shape (M,N)

    # Write back if within bounds
    out_ptrs = out_batch_ptr + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

import torch
import triton
import triton.language as tl

def custom_kernel(data):
    """
    Triton-accelerated implementation of the “outgoing” TriMul operator.
    The kernel computes, for each batch and hidden channel h,
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    which is the core N³ contraction of the TriMul module.
    All other operations (LayerNorm, linear projections, gating)
    are performed with PyTorch for simplicity; only the cubic contraction
    is off‑loaded to a Triton kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------------
    # 1. LayerNorm on the input channel dimension
    # ------------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------------
    if not nomask and mask is not None:
        # ensure mask is float32 same dtype as left/right
        mask_f = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------------
    # 5. Cubic contraction via Triton
    # ------------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = _trimul_contract_triton(left, right, bs, N, hidden_dim)

    # ------------------------------------------------------------------------
    # 6. Output LayerNorm, gate and final linear
    # ------------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

def _trimul_contract_triton(left: torch.Tensor, right: torch.Tensor, batch: int, N: int, H: int) -> torch.Tensor:
    """
    Triton kernel that computes the pairwise multiplication‑accumulate:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    Input tensors must be contiguous and have shape (B, N, N, H) and dtype float32.
    """
    assert left.is_cuda and right.is_cuda
    assert left.dtype == torch.float32 and right.dtype == torch.float32
    B, n1, n2, h = left.shape
    assert n1 == N and n2 == N and h == H
    device = left.device
    out = torch.empty_like(left, device=device)  # same dtype and shape

    # Strides (in elements) for each dimension
    # shape order: (batch, i, k, h)
    l_stride_b, l_stride_i, l_stride_k, l_stride_h = left.stride()
    r_stride_b, r_stride_j, r_stride_k, r_stride_h = right.stride()
    o_stride_b, o_stride_i, o_stride_j, o_stride_h = out.stride()

    # Define block sizes (tuned for H100)
    BLOCK_M = 64   # rows i
    BLOCK_N = 64   # cols j
    BLOCK_K = 64   # reduction over k

    # Grid: one program per (i‑tile, j‑tile, batch*channel)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    @triton.jit
    def _kernel(
        left_ptr, right_ptr, out_ptr,
        B, N, H,
        l_stride_b, l_stride_i, l_stride_k,
        r_stride_b, r_stride_k, r_stride_j,
        o_stride_b, o_stride_i, o_stride_j,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
        # --------------------------------------------------------------------
        #  Program IDs
        # --------------------------------------------------------------------
        pid_m = tl.program_id(axis=0)  # i‑tile
        pid_n = tl.program_id(axis=1)  # j‑tile
        pid_batch = tl.program_id(axis=2)  # combined batch*channel index

        # --------------------------------------------------------------------
        #  Decompose combined index into batch (b) and channel (h)
        # --------------------------------------------------------------------
        b = pid_batch // H
        h = pid_batch % H

        # --------------------------------------------------------------------
        #  Tile offsets
        # --------------------------------------------------------------------
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_m = offs_m < N
        mask_n = offs_n < N

        # Base pointers for this (b, h)
        left_base = left_ptr + b * l_stride_b + h * l_stride_h
        right_base = right_ptr + b * r_stride_b + h * r_stride_h
        out_base = out_ptr + b * o_stride_b + h * o_stride_h

        # Accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # --------------------------------------------------------------------
        #  Reduction over k dimension (the second axis of left, second axis of right)
        # --------------------------------------------------------------------
        for k in range(0, tl.cdiv(N, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # Load left tile (M x K) – indexing (i, k)
            ptr_left = left_base + offs_m[:, None] * l_stride_i + offs_k[None, :] * l_stride_k
            left_tile = tl.load(ptr_left, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

            # Load right tile transposed (K x N)
            #   row   = k  (original second axis)
            #   col   = j  (original first axis)
            ptr_right = right_base + offs_k[:, None] * r_stride_k + offs_n[None, :] * r_stride_j
            right_tile = tl.load(ptr_right, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

            # Multiply‑accumulate
            acc += tl.dot(left_tile, right_tile)

        # --------------------------------------------------------------------
        #  Write back result
        # --------------------------------------------------------------------
        ptr_out = out_base + offs_m[:, None] * o_stride_i + offs_n[None, :] * o_stride_j
        tl.store(ptr_out, acc, mask=mask_m[:, None] & mask_n[None, :])

    # Launch the kernel
    _kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B,
        N,
        H,
        l_stride_b,
        l_stride_i,
        l_stride_k,
        r_stride_b,
        r_stride_k,
        r_stride_j,
        o_stride_b,
        o_stride_i,
        o_stride_j,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
    )
    return out

"""
TriMul (outgoing) – Triton‑accelerated core contraction
=====================================================

The TriMul module (AlphaFold‑3, Chai, …) computes a dense N³ interaction
between two projected pair‑wise representations.  All linear layers,
gating and LayerNorms are executed with regular PyTorch ops; the only
cubic part

    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

is off‑loaded to a custom Triton kernel.  The kernel treats every
(b, h) pair as an independent matrix‑multiply problem (M=N rows,
N columns, K=N reduction) and thus runs a batched GEMM with
`tl.dot`.  The implementation works for any batch size (1‑2), any
sequence length up to 1024 and hidden dimensions up to 384, matching the
test specifications.

The public entry point is :func:`custom_kernel`, which receives a tuple
``(input, mask, weights, config)`` and returns the processed tensor of
shape ``[B, N, N, dim]``.
"""

import torch
import triton
import triton.language as tl


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32 on the same device).
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N]  ->  [B, N, N, 1]  -> broadcast to hidden_dim
        mask_f = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid of a linear projection)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"]
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"]
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"]
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cubic contraction via Triton
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = _trimul_contract_triton(left, right, B, N, hidden_dim)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gate and final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out


def _trimul_contract_triton(left: torch.Tensor,
                            right: torch.Tensor,
                            B: int,
                            N: int,
                            H: int) -> torch.Tensor:
    """
    Triton kernel implementing

        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Input tensors must be contiguous, float32 and have shape (B, N, N, H).
    The returned tensor has the same shape and dtype.
    """
    assert left.is_cuda and right.is_cuda
    assert left.dtype == torch.float32 and right.dtype == torch.float32
    assert left.shape == (B, N, N, H)
    assert right.shape == (B, N, N, H)

    device = left.device
    out = torch.empty_like(left, device=device)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # Strides (in elements) for each tensor
    #   shape order: (batch, dim0, dim1, channel)
    # ------------------------------------------------------------------
    l_stride_b, l_stride_i, l_stride_k, l_stride_h = left.stride()
    r_stride_b, r_stride_j, r_stride_k, r_stride_h = right.stride()
    o_stride_b, o_stride_i, o_stride_j, o_stride_h = out.stride()

    # ------------------------------------------------------------------
    # Tunable block sizes (empirically good for H100)
    # ------------------------------------------------------------------
    BLOCK_M = 64   # rows   (i dimension)
    BLOCK_N = 64   # cols   (j dimension)
    BLOCK_K = 64   # reduction over k

    # Grid: (tiles_i, tiles_j, batch * channel)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    @triton.jit
    def _kernel(
        left_ptr, right_ptr, out_ptr,
        B, N, H,
        l_stride_b, l_stride_i, l_stride_k, l_stride_h,
        r_stride_b, r_stride_k, r_stride_j, r_stride_h,
        o_stride_b, o_stride_i, o_stride_j, o_stride_h,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
    ):
        # ------------------------------------------------------------------
        # Identify this program
        # ------------------------------------------------------------------
        pid_m = tl.program_id(axis=0)                 # i‑tile
        pid_n = tl.program_id(axis=1)                 # j‑tile
        pid_batch = tl.program_id(axis=2)             # combined (b, h)

        # Decompose combined index into batch and hidden channel
        b = pid_batch // H
        h = pid_batch % H

        # ------------------------------------------------------------------
        # Tile offsets
        # ------------------------------------------------------------------
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_m = offs_m < N
        mask_n = offs_n < N

        # Base pointers for the (b, h) slice
        left_base  = left_ptr  + b * l_stride_b  + h * l_stride_h
        right_base = right_ptr + b * r_stride_b + h * r_stride_h
        out_base   = out_ptr   + b * o_stride_b   + h * o_stride_h

        # Accumulator for the tile (float32)
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # ------------------------------------------------------------------
        # Reduction over k
        # ------------------------------------------------------------------
        for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
            offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # left: (i, k)  → strides (l_stride_i, l_stride_k)
            left_ptrs = left_base + offs_m[:, None] * l_stride_i + offs_k[None, :] * l_stride_k
            left_tile = tl.load(
                left_ptrs,
                mask=mask_m[:, None] & mask_k[None, :],
                other=0.0,
            )

            # right (transposed): we need (k, j) where original layout is (j, k)
            # Row stride = r_stride_k (original k‑stride)
            # Col stride = r_stride_j (original j‑stride)
            right_ptrs = right_base + offs_k[:, None] * r_stride_k + offs_n[None, :] * r_stride_j
            right_tile = tl.load(
                right_ptrs,
                mask=mask_k[:, None] & mask_n[None, :],
                other=0.0,
            )

            # Accumulate the matrix product for this k‑tile
            acc += tl.dot(left_tile, right_tile)     # (M,N) = (M,K)·(K,N)

        # ------------------------------------------------------------------
        # Store the computed tile
        # ------------------------------------------------------------------
        out_ptrs = out_base + offs_m[:, None] * o_stride_i + offs_n[None, :] * o_stride_j
        tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

    # Launch
    _kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B,
        N,
        H,
        l_stride_b,
        l_stride_i,
        l_stride_k,
        l_stride_h,
        r_stride_b,
        r_stride_k,
        r_stride_j,
        r_stride_h,
        o_stride_b,
        o_stride_i,
        o_stride_j,
        o_stride_h,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
    )
    return out
