"""
tri_mul_outgoing_triton.py

Implementation of the “outgoing” TriMul operation from AlphaFold‑3 using a
custom Triton kernel for the N³ pairwise accumulation.  The kernel computes
`out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]` for each batch `b` and
hidden channel `d`.  All other linear/LN/gating steps are performed in
PyTorch; the heavy N³ contraction is off‑loaded to Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------------- #
# Triton kernel: computes the 4‑D tensor contraction described above.
# ----------------------------------------------------------------------------- #
@triton.jit
def trimul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # dimensions
    B, N, H,
    # strides for left  (batch, i, k, h)
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    # strides for right (batch, j, k, h)
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    # strides for out   (batch, i, j, h)
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    Each (b, h) pair is processed by a separate program instance.
    """
    pid_bh = tl.program_id(2)                # flattened batch‑channel id
    batch_idx = pid_bh // H                   # batch index
    h = pid_bh % H                            # hidden‑channel index

    pid_i = tl.program_id(0)                  # block row (i)
    pid_j = tl.program_id(1)                  # block col (j)

    # -------------------  compute start offsets ------------------- #
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # masks for the i/j dimensions (handle non‑divisible N)
    i_valid = i_start + tl.arange(0, BLOCK_M) < N
    j_valid = j_start + tl.arange(0, BLOCK_N) < N

    # base pointers that already include batch and hidden‑channel offsets
    left_base  = left_ptr  + batch_idx * stride_left_batch  + h * stride_left_h
    right_base = right_ptr + batch_idx * stride_right_batch + h * stride_right_h
    out_base   = out_ptr   + batch_idx * stride_out_batch   + h * stride_out_h

    # accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_tiles = tl.cdiv(N, BLOCK_K)

    for tile_k in range(num_tiles):
        k_start = tile_k * BLOCK_K

        k_valid = k_start + tl.arange(0, BLOCK_K) < N

        # -------------------  load left tile (i, k)  ------------------- #
        # left[i, k]   -> shape (BLOCK_M, BLOCK_K)
        offs_i = i_start + tl.arange(0, BLOCK_M)[:, None]          # (BM,1)
        offs_k_left = k_start + tl.arange(0, BLOCK_K)[None, :]     # (1,BK)
        left_ptrs = left_base + offs_i * stride_left_i + offs_k_left * stride_left_k
        left_tile = tl.load(left_ptrs,
                            mask=i_valid[:, None] & k_valid[None, :],
                            other=0.0)

        # -------------------  load right tile (k, j)  ------------------- #
        # right[j, k]  -> we need it transposed to (k, j)
        offs_j = j_start + tl.arange(0, BLOCK_N)[None, :]          # (1,BN)
        offs_k_right = k_start + tl.arange(0, BLOCK_K)[:, None]    # (BK,1)
        right_ptrs = right_base + offs_k_right * stride_right_k + offs_j * stride_right_i
        right_tile = tl.load(right_ptrs,
                             mask=k_valid[:, None] & j_valid[None, :],
                             other=0.0)

        # -------------------  matrix multiplication accumulation  ------------------- #
        # left_tile: (BM, BK), right_tile: (BK, BN) -> (BM, BN)
        acc += tl.dot(left_tile, right_tile)

    # -------------------  store result ------------------- #
    offs_i_out = i_start + tl.arange(0, BLOCK_M)[:, None]      # (BM,1)
    offs_j_out = j_start + tl.arange(0, BLOCK_N)[None, :]      # (1,BN)
    out_ptrs = out_base + offs_i_out * stride_out_i + offs_j_out * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=i_valid[:, None] & j_valid[None, :])

    Triton‑accelerated implementation of the N³ contraction.
    left / right: shape [B, N, N, H], contiguous float32.
    Returns tensor of the same shape.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides (in elements, not bytes)
    sL_b, sL_i, sL_k, sL_h = left.stride()
    sR_b, sR_i, sR_k, sR_h = right.stride()
    sO_b, sO_i, sO_j, sO_h = out.stride()

    # Tile sizes – tuned for H100, float32
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),     # i‑tiles
        triton.cdiv(N, BLOCK_N),     # j‑tiles
        B * H                        # batch × hidden‑channel
    )

    trimul_kernel[grid](
        left, right, out,
        B, N, H,
        sL_b, sL_i, sL_k, sL_h,
        sR_b, sR_i, sR_k, sR_h,
        sO_b, sO_i, sO_j, sO_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------------- #
# Main entry point expected by the evaluator
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold‑3 style).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on GPU).
    """
    input_tensor, mask, weights, config = data

    # --------------------------------------------------------------------- #
    # Extract configuration
    # --------------------------------------------------------------------- #
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------------- #
    # Load all parameters (they are already on the correct device)
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------------- #
    # LayerNorm over the feature dimension (dim)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # --------------------------------------------------------------------- #
    # Linear projections (no bias)
    # --------------------------------------------------------------------- #
    # shape: [B, N, N, hidden_dim]
    left = F.linear(x, left_proj_weight)    # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)

    # --------------------------------------------------------------------- #
    # Apply mask if it is present (mask shape: [B, N, N])
    # --------------------------------------------------------------------- #
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------------- #
    # Gating MLPs + sigmoid
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # N³ contraction via Triton kernel
    # --------------------------------------------------------------------- #
    # Ensure contiguous tensors for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = trimul_triton(left, right)   # shape [B,N,N,hidden_dim]

    # --------------------------------------------------------------------- #
    # Final LayerNorm on hidden dimension
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # --------------------------------------------------------------------- #
    # Apply output gate and final linear projection back to `dim`
    # --------------------------------------------------------------------- #
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # (B,N,N,dim)

    return out
