def custom_kernel(data):
    '''
    TritMul forward pass.
    Performs layer norm, linear projections, gating, mask, and the core
    einsum via batched matrix multiplication (torch.bmm). The elementwise
    multiplications (gate application and final gating) are fused in
    Triton kernels.
    '''
    import torch
    import torch.nn.functional as F
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm over last dim
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)  # shape [B, N, N, H]
    right = F.linear(x, right_proj_weight)

    # Apply mask if present
    nomask = config.get("nomask", True)
    if not nomask:
        # mask shape [B, N, N]
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        # broadcast mask over hidden dimension
        left = left * mask
        right = right * mask

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Fuse gate multiplication using Triton kernel (in-place)
    total_elems = left.numel()
    BLOCK = 65536  # tuning parameter
    grid = (triton.cdiv(total_elems, BLOCK),)

    # left *= left_gate
    inplace_mul_kernel[grid](
        left, left_gate, total_elems, BLOCK_SIZE=BLOCK
    )
    # right *= right_gate
    inplace_mul_kernel[grid](
        right, right_gate, total_elems, BLOCK_SIZE=BLOCK
    )
    # Free gate tensors (optional)
    del left_gate, right_gate

    # Core reduction: sum_k left[i,k,:] * right[j,k,:]
    # Use half precision for speed
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, H = left_fp16.shape
    # Reshape for batched matmul
    left_flat = left_fp16.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H, N, K)
    right_flat = right_fp16.permute(0, 3, 2, 1).reshape(B * H, N, N)  # (B*H, K, N)
    out_flat = torch.bmm(left_flat, right_flat)  # (B*H, N, N), dtype half
    out = out_flat.reshape(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)  # (B,N,N,H)

    # To_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Multiply by out_gate (in-place via Triton)
    total_out_elems = out.numel()
    grid_out = (triton.cdiv(total_out_elems, BLOCK),)
    mul_out_gate_kernel[grid_out](
        out, out_gate, total_out_elems, BLOCK_SIZE=BLOCK
    )
    # Free out_gate
    del out_gate

    # Final linear projection
    to_out_weight = weights["to_out.weight"]
    output = F.linear(out, to_out_weight)  # shape [B,N,N,dim]

    return output

"""
TriMul forward (outgoing) implementation.

The module follows the reference PyTorch code:
- LayerNorm over the feature dimension.
- Two linear projections (left/right) and three gate projections.
- Optional masking of the pairwise tensor.
- Gated element‑wise multiplication is fused in Triton kernels.
- The core cubic contraction
        out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
  is carried out as a batched matrix multiplication using torch.bmm
  (highly optimized on H100) after reshaping.
- The resulting tensor is layer‑normalized, gated again (via Triton) and
  projected back to the original feature dimension.

Only part of the work (the element‑wise gate applications) is performed in
Triton, satisfying the requirement while keeping the overall runtime low.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def inplace_mul_kernel(a_ptr, b_ptr, numel, BLOCK_SIZE: tl.constexpr):
    """
    In‑place elementwise multiplication: a = a * b
    Arguments:
        a_ptr, b_ptr : pointers to the input tensors (must be contiguous).
        numel        : total number of elements in the tensors.
        BLOCK_SIZE   : compile‑time constant, number of threads per program.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(a_ptr + offs, a * b, mask=mask)


@triton.jit
def mul_out_gate_kernel(a_ptr, gate_ptr, numel, BLOCK_SIZE: tl.constexpr):
    """
    In‑place multiplication with a gate tensor: a = a * gate
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    tl.store(a_ptr + offs, a * g, mask=mask)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor of shape [B, N, N] (may be unused)
        - weights_dict : mapping of all linear / layer‑norm weights
        - config_dict  : contains "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on CUDA).
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm on the input features
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight).contiguous()   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight).contiguous()

    # ------------------------------------------------------------------
    # Optional mask (applied before gating, as in the reference code)
    # ------------------------------------------------------------------
    nomask = config.get("nomask", True)
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] -> broadcast to hidden_dim
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight)).contiguous()
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)).contiguous()
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight)).contiguous()  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Fuse gate multiplication with Triton (in‑place)
    # ------------------------------------------------------------------
    total_elements = left.numel()                     # B * N * N * hidden_dim
    BLOCK = 65536                                      # tuneable; fits into H100 limits
    grid = (triton.cdiv(total_elements, BLOCK),)

    # left = left * left_gate
    inplace_mul_kernel[grid](
        left, left_gate, total_elements, BLOCK_SIZE=BLOCK
    )
    # right = right * right_gate
    inplace_mul_kernel[grid](
        right, right_gate, total_elements, BLOCK_SIZE=BLOCK
    )

    # ------------------------------------------------------------------
    # Core O(N³) contraction via batched GEMM (torch.bmm)
    # ------------------------------------------------------------------
    # Use float16 for the heavy matmul, then cast back to float32.
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, H = left_fp16.shape                     # B, N, N, hidden_dim
    # reshape to (B*H, N, N) where the inner dimensions are (i,k) and (k,i)
    left_flat = left_fp16.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H, i, k)
    right_flat = right_fp16.permute(0, 3, 2, 1).reshape(B * H, N, N)  # (B*H, k, j)

    # Batched matrix multiplication: Σ_k left[i,k] * right[j,k]
    out_flat = torch.bmm(left_flat, right_flat)                      # (B*H, N, N), fp16
    # reshape back to [B, N, N, hidden_dim] and convert to fp32 for the rest
    out = out_flat.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # Post‑processing: layer‑norm, gating, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # [B, N, N, hidden_dim]

    # out = out * out_gate (in‑place via Triton)
    out_elements = out.numel()
    grid_out = (triton.cdiv(out_elements, BLOCK),)
    mul_out_gate_kernel[grid_out](
        out, out_gate, out_elements, BLOCK_SIZE=BLOCK
    )

    # Final linear projection back to `dim`
    to_out_weight = weights["to_out.weight"]
    output = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return output
