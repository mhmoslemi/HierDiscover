@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, V: tl.constexpr, H_remainder: tl.constexpr):
   # program ids
   pid_bh = tl.program_id(2) # combined batch and hidden slice index
   batch_idx = pid_bh // tl.cdiv(H, V)
   slice_idx = pid_bh % tl.cdiv(H, V)
   h_start = slice_idx * V
   # compute i and j block ids
   pid_i = tl.program_id(0)
   pid_j = tl.program_id(1)

   # ranges
   i_range = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
   j_range = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

   # masks for i, j range
   mask_i = i_range < N
   mask_j = j_range < N

   # allocate accumulator
   acc = tl.zeros((BLOCK_I, BLOCK_J, V), dtype=tl.float32)

   for k in range(0, tl.cdiv(N, BLOCK_K)):
       k_range = k * BLOCK_K + tl.arange(0, BLOCK_K)
       mask_k = k_range < N

       # compute pointers for left and right tile
       left_ptrs = left_ptr + \
           batch_idx * stride_left_batch + \
           i_range[:, None, None] * stride_left_i + \
           k_range[None, :, None] * stride_left_k + \
           (h_start + tl.arange(0, V))[None, None, :] * stride_left_h

       right_ptrs = right_ptr + \
           batch_idx * stride_right_batch + \
           j_range[:, None, None] * stride_right_j + \
           k_range[None, :, None] * stride_right_k + \
           (h_start + tl.arange(0, V))[None, None, :] * stride_right_h

       # load
       left = tl.load(left_ptrs,
                      mask=mask_i[:, None, None] & mask_k[None, :, None] & ((h_start + tl.arange(0, V)) < H),
                      other=0.0)
       right = tl.load(right_ptrs,
                       mask=mask_j[:, None, None] & mask_k[None, :, None] & ((h_start + tl.arange(0, V)) < H),
                       other=0.0)

       # accumulate for each hidden slice
       for v in range(V):
           # left_slice shape (BLOCK_I, BLOCK_K)
           # right_slice shape (BLOCK_J, BLOCK_K)
           left_slice = left[:, :, v]   # (BLOCK_I, BLOCK_K)
           right_slice = right[:, :, v] # (BLOCK_J, BLOCK_K)
           # dot product across K
           acc[:, :, v] += tl.dot(left_slice, tl.trans(right_slice))

   # store result
   out_ptrs = out_ptr + \
       batch_idx * stride_out_batch + \
       i_range[:, None, None] * stride_out_i + \
       j_range[None, :, None] * stride_out_j + \
       (h_start + tl.arange(0, V))[None, None, :] * stride_out_h

   # mask store
   mask_ij = mask_i[:, None] & mask_j[None, :]
   mask_h = (h_start + tl.arange(0, V)) < H
   mask_store = mask_ij[:, :, None] & mask_h[None, None, :]

   tl.store(out_ptrs, acc, mask=mask_store)

"""
Triton implementation of the outgoing Triangle Multiplicative Update (TriMul) operation.
The kernel fuses the O(N^3) contraction `out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]`
by tiling the `(i,j)` output plane and the reduction dimension `k`. The hidden dimension
`d` is processed in small vectors (`V`) to keep register pressure low. All other
operations (LayerNorm, Linear projections, sigmoid gates, final LayerNorm and Linear)
are performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr,        # *float32
    right_ptr,       # *float32
    out_ptr,         # *float32
    B, N, H,         # runtime sizes
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, V: tl.constexpr):
    """
    Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    where `h` is processed in blocks of size `V`.
    """
    pid_i = tl.program_id(0)      # i block
    pid_j = tl.program_id(1)      # j block
    pid_z = tl.program_id(2)      # combined batch and hidden‑slice index

    # --------------------------------------------------------------
    # decode batch index and hidden‑slice index
    # --------------------------------------------------------------
    hidden_slices = tl.cdiv(H, V)
    batch_idx = pid_z // hidden_slices
    slice_idx = pid_z % hidden_slices
    h_start = slice_idx * V                # first hidden index handled by this program

    # --------------------------------------------------------------
    # ranges for the i, j and k tiles
    # --------------------------------------------------------------
    i_off = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_off = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = i_off < N
    mask_j = j_off < N

    # accumulator of shape (BLOCK_I, BLOCK_J, V)
    acc = tl.zeros((BLOCK_I, BLOCK_J, V), dtype=tl.float32)

    # --------------------------------------------------------------
    # tiled reduction over k
    # --------------------------------------------------------------
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # pointer arithmetic for the three tensors
        left_ptrs = left_ptr + \
            batch_idx * stride_l_batch + \
            i_off[:, None, None] * stride_l_i + \

            k_off[None, :, None] * stride_r_k + \
            (h_start + tl.arange(0, V))[None, None, :] * stride_r_h

        # mask combines bounds on i, j, k and the hidden dimension
        mask_h = (h_start + tl.arange(0, V)) < H
        load_mask_left  = mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]
        load_mask_right = mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]

        left_tile  = tl.load(left_ptrs,  mask=load_mask_left,  other=0.0)
        right_tile = tl.load(right_ptrs, mask=load_mask_right, other=0.0)

        # ----------------------------------------------------------
        # accumulate over the V hidden channels
        # ----------------------------------------------------------
        for v in range(V):
            # each slice is a (BLOCK_I, BLOCK_K) / (BLOCK_J, BLOCK_K) matrix
            L = left_tile[:, :, v]    # (BLOCK_I, BLOCK_K)
            R = right_tile[:, :, v]   # (BLOCK_J, BLOCK_K)
            # matmul L @ Rᵀ -> (BLOCK_I, BLOCK_J)
            acc[:, :, v] += tl.dot(L, tl.trans(R))

    # store the result
    # --------------------------------------------------------------
    out_ptrs = out_ptr + \
        batch_idx * stride_o_batch + \
        i_off[:, None, None] * stride_o_i + \
        j_off[None, :, None] * stride_o_j + \
        (h_start + tl.arange(0, V))[None, None, :] * stride_o_h

    store_mask_h = (h_start + tl.arange(0, V)) < H
    store_mask = mask_i[:, None] & mask_j[None, :] & store_mask_h[None, None, :]
    tl.store(out_ptrs, acc, mask=store_mask)

    - mask:   Optional tensor of shape [B, N, N] (float or bool)

    # ------------------------------------------------------------------

    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------

    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))

    right = right * right_gate

    BLOCK_J = 64

    hidden_slices = (hidden_dim + V - 1) // V

    # strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()

    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7) Final linear projection back to original dimension
    # ------------------------------------------------------------------
    output = F.linear(out, weights["to_out.weight"])

    return output

"""
Outgoing TriMul (Triangle multiplicative update) – Triton accelerated version.

The heavy O(N³·C) contraction
    out[b,i,j,c] = Σₖ left[b,i,k,c] * right[b,j,k,c]
is performed by a custom Triton kernel that tiles the (i,j) output plane
and the reduction dimension k.  The channel dimension c (called `hidden_dim`
in the code) is processed in small vector chunks (`V`) to keep register
pressure low. All other operations (LayerNorms, Linear projections,
sigmoids, final projection) are expressed with vanilla PyTorch for
readability and correctness.

The kernel works for any batch size (≤2), sequence length (≤1024) and
hidden dimension that is a multiple or not of `V` (the kernel masks
out‑of‑range entries).  The final result has dtype float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,               # *float32
    right_ptr,              # *float32
    out_ptr,                # *float32
    B, N, H,                # runtime sizes
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, V: tl.constexpr):
    """
    Compute:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    The hidden dimension `h` is sliced into chunks of size `V`.
    """
    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile
    pid_z = tl.program_id(2)          # combined batch‑and‑hidden‑slice

    # --------------------------------------------------------------
    # Decode batch index and hidden‑slice index
    # --------------------------------------------------------------
    hidden_slices = tl.cdiv(H, V)
    batch_idx = pid_z // hidden_slices
    slice_idx = pid_z % hidden_slices
    h_start = slice_idx * V                     # first hidden index for this program

    # --------------------------------------------------------------
    # Index ranges for this tile
    # --------------------------------------------------------------
    i_off = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_off = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = i_off < N
    mask_j = j_off < N

    # Accumulator (BLOCK_I, BLOCK_J, V)
    acc = tl.zeros((BLOCK_I, BLOCK_J, V), dtype=tl.float32)

    # --------------------------------------------------------------
    # Reduction over k (tiled)
    # --------------------------------------------------------------
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # Pointers with the proper offsets
        left_ptrs = left_ptr + \
            batch_idx * stride_l_batch + \
            i_off[:, None, None] * stride_l_i + \
            k_off[None, :, None] * stride_l_k + \
            (h_start + tl.arange(0, V))[None, None, :] * stride_l_h

        right_ptrs = right_ptr + \
            batch_idx * stride_r_batch + \
            j_off[:, None, None] * stride_r_j + \
            k_off[None, :, None] * stride_r_k + \
            (h_start + tl.arange(0, V))[None, None, :] * stride_r_h

        # Masks for out‑of‑bounds loads
        mask_h = (h_start + tl.arange(0, V)) < H
        load_mask_left  = mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]
        load_mask_right = mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]

        left_tile  = tl.load(left_ptrs,  mask=load_mask_left,  other=0.0)
        right_tile = tl.load(right_ptrs, mask=load_mask_right, other=0.0)

        # ----------------------------------------------------------
        # Accumulate over the V hidden channels
        # ----------------------------------------------------------
        for v in range(V):
            L = left_tile[:, :, v]   # (BLOCK_I, BLOCK_K)
            R = right_tile[:, :, v]  # (BLOCK_J, BLOCK_K)
            # L @ Rᵀ  → (BLOCK_I, BLOCK_J)
            acc[:, :, v] += tl.dot(L, tl.trans(R))

    # --------------------------------------------------------------
    # Store the result
    # --------------------------------------------------------------
    out_ptrs = out_ptr + \
        batch_idx * stride_o_batch + \
        i_off[:, None, None] * stride_o_i + \
        j_off[None, :, None] * stride_o_j + \
        (h_start + tl.arange(0, V))[None, None, :] * stride_o_h

    store_mask_h = (h_start + tl.arange(0, V)) < H
    store_mask = mask_i[:, None] & mask_j[None, :] & store_mask_h[None, None, :]
    tl.store(out_ptrs, acc, mask=store_mask)
