@triton.jit
def trimul_matmul_kernel(A_ptr, B_ptr, C_ptr,
                         B, N, H,
                         stride_a_batch, stride_a_i, stride_a_k, stride_a_d,
                         stride_b_batch, stride_b_i, stride_b_k, stride_b_d,
                         stride_c_batch, stride_c_i, stride_c_j, stride_c_d,
                         BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_b = tl.program_id(2)  # batch*hidden index
    pid_m = tl.program_id(0)  # tile i
    pid_n = tl.program_id(1)  # tile j
    
    # Compute actual batch and hidden index
    hidden_dim = H
    b_idx = pid_b // hidden_dim
    d_idx = pid_b % hidden_dim
    
    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)  # for each iteration inside loop
    
    # mask for valid m,n
    mask_m = offs_m < N
    mask_n = offs_n < N
    
    # pointer to start of A and B for this batch/hidden
    a_base = A_ptr + b_idx * stride_a_batch + d_idx * stride_a_d
    b_base = B_ptr + b_idx * stride_b_batch + d_idx * stride_b_d
    c_base = C_ptr + b_idx * stride_c_batch + d_idx * stride_c_d
    
    # Load coordinates for output block indices
    # Offsets for A: (i,k)
    # Offsets for B (transposed): (k,j)
    # Use pointer arithmetic each iteration:
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K
        # compute K offsets
        k_offs = cur_k + offs_k
        mask_k = k_offs < N
        
        # pointers
        a_ptrs = a_base + offs_m[:, None] * stride_a_i + k_offs[None, :] * stride_a_k
        b_ptrs = b_base + k_offs[:, None] * stride_b_k + offs_n[None, :] * stride_b_i  # note stride_b_i is for column j
        # load values
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)
        # dot product (multiply-add)
        acc += tl.dot(a, b)
    # Store result
    c_ptrs = c_base + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor over the last dimension (dim).
2. Linear projections (no bias) to a hidden dimension `hidden_dim`:
   * left  = x_norm @ left_proj.weight
   * right = x_norm @ right_proj.weight
3. Optional binary mask (broadcasted over hidden_dim) is applied to left/right
   before the gates.
4. Gating (sigmoid) computed from separate linear layers and applied:
   left  = left  * left_gate
   right = right * right_gate
5. The expensive contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is performed as a batched matrix‑multiply for each hidden channel.
   By flattening the (batch, hidden) dimensions we obtain a standard
   GEMM A @ Bᵀ for every channel, which is executed efficiently by the
   Triton kernel `trimul_matmul_kernel`.
6. Layer‑norm over the hidden dimension, multiplication by `out_gate`,
   and a final linear projection to the original `dim`.

The kernel operates on half‑precision inputs (saving bandwidth) while
accumulating in float32 to preserve numeric quality.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_matmul_kernel(
    # pointers
    A_ptr: tl.pointer(tl.float16),  # left tensor   (B,N,N,hidden)
    B_ptr: tl.pointer(tl.float16),  # right tensor  (B,N,N,hidden)
    C_ptr: tl.pointer(tl.float32),  # output tensor (B,N,N,hidden)
    # problem size
    B: tl.int64,
    N: tl.int64,
    H: tl.int64,
    # strides
    stride_a_batch, stride_a_i, stride_a_k, stride_a_d,
    stride_b_batch, stride_b_i, stride_b_k, stride_b_d,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_d,
    # block sizes (compile‑time constants)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched (batch*hidden) matrix multiplication:
    C[b*h, i, j] = Σ_k A[b*h, i, k] * B[b*h, j, k]   (B = batch, H = hidden)
    The kernel treats every (batch, hidden) pair as an independent batch.
    """
    pid_m = tl.program_id(0)      # tile over the i‑dimension
    pid_n = tl.program_id(1)      # tile over the j‑dimension
    pid_bh = tl.program_id(2)     # flattened (batch, hidden) index

    # recover original batch and hidden indices
    hidden_dim = H
    b_idx = pid_bh // hidden_dim
    d_idx = pid_bh % hidden_dim

    # offsets for the current tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (batch, hidden) slice
    a_base = A_ptr + b_idx * stride_a_batch + d_idx * stride_a_d
    b_base = B_ptr + b_idx * stride_b_batch + d_idx * stride_b_d
    c_base = C_ptr + b_idx * stride_c_batch + d_idx * stride_c_d

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the k‑dimension in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for the current K‑tile
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_i

        # load tiles, casting to fp32 for the dot
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # accumulate the matrix product for this tile
        acc += tl.dot(a, b)     # (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # write back the result
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator using a Triton kernel
    for the N³ contraction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim]
            - mask: torch.Tensor, shape [B, N, N] (may be ignored)
            - weights: dict of torch.Tensors (model parameters)
            - config: dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # --------------------------------------------------------------------- #
    # Unpack arguments
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    # --------------------------------------------------------------------- #
    # 1️⃣ LayerNorm over the last dimension
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------------- #
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------------- #
    left_proj = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden]
    right_proj = F.linear(x, weights["right_proj.weight"]) # [B, N, N, hidden]

    # --------------------------------------------------------------------- #
    # 3️⃣ Optional mask (broadcasted over hidden_dim)
    # --------------------------------------------------------------------- #
    if not nomask and mask is not None:
        mask = mask.to(dtype=left_proj.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # --------------------------------------------------------------------- #
    # 4️⃣ Gating (sigmoid) and gated projection
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,hidden]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # [B,N,N,hidden]

    left = left_proj * left_gate   # [B,N,N,hidden]
    right = right_proj * right_gate

    # --------------------------------------------------------------------- #
    # 5️⃣ Cast to half‑precision for the heavy contraction
    # --------------------------------------------------------------------- #
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape

    # --------------------------------------------------------------------- #
    # 6️⃣ Allocate output tensor for the hidden‑dim contraction
    # --------------------------------------------------------------------- #
    out_hidden = torch.empty((B, N, N, hidden_dim),
                             dtype=torch.float32,
                             device=device)

    # --------------------------------------------------------------------- #
    # 7️⃣ Prepare strides for Triton kernel
    # --------------------------------------------------------------------- #
    stride_left = left_h.stride()   # (batch, i, k, d)
    stride_right = right_h.stride()
    stride_out = out_hidden.stride()

    # --------------------------------------------------------------------- #
    # 8️⃣ Launch Triton kernel (batched GEMM per hidden channel)
    # --------------------------------------------------------------------- #
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32   # tuned for H100
    grid = (tl.cdiv(N, BLOCK_M),    # tiles over i
            tl.cdiv(N, BLOCK_N),    # tiles over j
            B * hidden_dim)         # (batch, hidden) flattened

    trimul_matmul_kernel[grid](
        left_h,
        right_h,
        out_hidden,
        B,
        N,
        hidden_dim,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Ensure kernel completion before proceeding
    torch.cuda.synchronize()

    # --------------------------------------------------------------------- #
    # 9️⃣ Output LayerNorm over hidden_dim, gating, final linear projection
    # --------------------------------------------------------------------- #
    out_norm = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm * out_gate               # apply output gate
    out = F.linear(out_norm, weights["to_out.weight"])   # back to [B,N,N,dim]

    return out
