def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    b = pid0 // H
    h = pid0 % H

    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    # Bound checks
    i_mask = i_offset + tl.arange(0, BLOCK_M) < N
    j_mask = j_offset + tl.arange(0, BLOCK_N) < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        # indices.
        i_idx = i_offset + tl.arange(0, BLOCK_M)
        k_idx = k_offset + tl.arange(0, BLOCK_K)
        j_idx = j_offset + tl.arange(0, BLOCK_N)

        # compute masks
        k_mask = k_idx < N

        # left block [M, K]

        # accumulate
        acc += tl.dot(left, right, out_dtype=tl.float32)

"""
TriMul (outgoing) kernel.

The forward pass consists of:
1. LayerNorm on the input tensor (shape [B, N, N, C]).
2. Two linear projections (left/right) and three gating linear layers
   (left_gate, right_gate, out_gate). The projections are optionally masked.
3. Element‑wise gated tensors `left` and `right` of shape [B, N, N, H].
4. Core contraction:
       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   which can be written as a batched matrix multiplication
   `out = left @ right.transpose(-1, -2)` per hidden channel.
   This O(N³·H) operation is implemented in Triton, loading tiles
   of the K‑dimension and using `tl.dot` with half‑precision inputs
   and a float32 accumulator.
5. LayerNorm over the hidden dimension, multiplication by `out_gate`,
   and a final linear projection back to the original channel size.

Only the heavy contraction (step 4) runs on Triton; all other
operations are performed with regular PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the 4‑D contraction
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,               # const __half *
    right_ptr,              # const __half *
    out_ptr,                # float *
    B, N, H,                # runtime ints
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out_{b,i,j,h} = sum_k left_{b,i,k,h} * right_{b,j,k,h}."""
    pid0 = tl.program_id(0)          # encodes (batch, hidden)
    pid_i = tl.program_id(1)         # i‑tile
    pid_j = tl.program_id(2)         # j‑tile

    # decode batch and hidden index
    b = pid0 // H
    h = pid0 % H

    # tile origins
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    # 1‑D ranges
    i_range = tl.arange(0, BLOCK_M)
    j_range = tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    # bounds for the current tile
    i_mask = i_offset + i_range < N
    j_mask = j_offset + j_range < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for kk in range(num_k_tiles):
        k_offset = kk * BLOCK_K
        k_idx = k_offset + k_range
        k_mask = k_idx < N

        # ------------------------------------------------------------------
        # Load a (BLOCK_M, BLOCK_K) tile from left  : left[b, i, k, h]
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + (i_offset + i_range)[:, None] * stride_left_i
            + k_idx[None, :] * stride_left_k
            + h * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=i_mask[:, None] & k_mask[None, :],
            other=0.0,
        )  # dtype == __half

        # ------------------------------------------------------------------
        # Load a (BLOCK_K, BLOCK_N) tile from right transposed:
        #    right_T[k, j] = right[b, j, k, h]   (j is the output column)
        # ------------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + k_idx[:, None] * stride_right_k
            + (j_offset + j_range)[None, :] * stride_right_i
            + h * stride_right_h
        )
        right_T = tl.load(
            right_ptrs,
            mask=k_mask[:, None] & j_mask[None, :],
            other=0.0,
        )  # dtype == __half, shape (K, N)

        # dot product accumulation (M×K)·(K×N) → (M×N)
        acc += tl.dot(left, right_T, out_dtype=tl.float32)

    # ----------------------------------------------------------------------
    # Store the result tile
    # ----------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + (i_offset + i_range)[:, None] * stride_out_i
        + (j_offset + j_range)[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])

# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    # ------------------------------------------------------------------
    # Extract configuration
    # ------------------------------------------------------------------
    dim = config["dim"]            # input / output channel dimension C
    hidden_dim = config["hidden_dim"]  # hidden dimension H
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Extract weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input layer norm   (over the last dimension = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_weight)    # [B, N, N, H]
    right_proj = F.linear(x, right_proj_weight)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Optional masking (mask shape: [B, N, N])
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(left_proj.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4. Gating values (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B, N, N, H]
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5. Apply gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate      # [B, N, N, H]
    right = right_proj * right_gate   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6. Cast to half precision for the heavy contraction
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    B, N, _, _ = left_h.shape

    # Output tensor of the contraction (float32)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 7. Launch Triton kernel
    # ------------------------------------------------------------------
    # Block sizes – tuned for H100 (fits in L2 and registers)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                         # (batch * hidden) programs
        (N + BLOCK_M - 1) // BLOCK_M,           # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,           # j‑tiles
    )

    # Strides (in elements, not bytes)
    ls = left_h.stride()
    rs = right_h.stride()
    os = out.stride()

    trimul_outgoing_kernel[grid](
        left_h,
        right_h,
        out,
        B,
        N,
        hidden_dim,
        ls[0], ls[1], ls[2], ls[3],
        rs[0], rs[1], rs[2], rs[3],
        os[0], os[1], os[2], os[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8. Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias
    )

    # ------------------------------------------------------------------
    # 9. Apply the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10. Final linear projection back to `dim` channels
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # bias is False in the original module

    return out
