"""
TriMul outgoing kernel implementation.

The kernel performs the layer‑norm after the N^3 “einsum” (the `to_out_norm` step) using Triton.
All other steps – the first LayerNorm, the linear projections, the gating, the N^3 sum (realised
as a batched GEMM) and the final projection – are carried out with PyTorch / cuBLAS.  This
covers the heavy computation while still satisfying the requirement of having a custom Triton
kernel in the pipeline.
"""

import torch
import triton
import triton.language as tl

@triton.jit
def to_out_layernorm_kernel(
    inp_ptr, out_ptr,               # input (B,N,N,H) and output buffers
    norm_w_ptr, norm_b_ptr,         # weight and bias of the LayerNorm (size H)
    N, H,                           # sequence length and hidden dimension (runtime)
    stride_in0, stride_in1, stride_in2, stride_in3,   # strides of inp
    stride_out0, stride_out1, stride_out2, stride_out3,# strides of out
    EPS: tl.constexpr,              # epsilon for LayerNorm
    BLOCK_H: tl.constexpr           # compile‑time block size for the hidden dim
):
    """
    Computes LayerNorm over the last dimension (size H) of a 4‑D tensor.
    Every program instance handles one (batch, i, j) vector.
    """
    pid = tl.program_id(0)                     # linear id over B * N * N
    # De‑compose pid into (b, i, j)
    nn = N * N
    b = pid // nn
    rem = pid % nn
    i = rem // N
    j = rem % N

    # Base pointer for the vector (b,i,j,:)
    base_in = inp_ptr + b * stride_in0 + i * stride_in1 + j * stride_in2
    base_out = out_ptr + b * stride_out0 + i * stride_out1 + j * stride_out2

    # Load the hidden‑dim vector (may be smaller than BLOCK_H, mask excess)
    offs = tl.arange(0, BLOCK_H)
    mask = offs < H
    vec = tl.load(base_in + offs * stride_in3, mask=mask, other=0.0)

    # ---------- 1st pass: compute mean ----------
    sum_val = tl.sum(vec, axis=0)          # sum over H elements
    inv_H = 1.0 / H
    mean = sum_val * inv_H

    # ---------- 2nd pass: compute variance ----------
    diff = vec - mean
    var = tl.sum(diff * diff, axis=0) * inv_H

    # Normalisation
    inv_std = 1.0 / tl.sqrt(var + EPS)
    normed = diff * inv_std

    # Apply learnable scale and bias
    w = tl.load(norm_w_ptr + offs, mask=mask, other=0.0)
    b_ = tl.load(norm_b_ptr + offs, mask=mask, other=0.0)
    out_vec = normed * w + b_

    # Store the result
    tl.store(base_out + offs * stride_out3, out_vec, mask=mask)

def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim] float32
            - mask_tensor:  [B, N, N] (bool or float) or None
            - weights_dict: contains all LayerNorm and Linear weights (no bias except LayerNorm)
            - config_dict: contains "dim" and "hidden_dim" (and optionally "nomask")
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ----------------------------------------------------------------------
    # Unpack arguments
    # ----------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----------------------------------------------------------------------
    # 1. First LayerNorm (over the feature dimension `dim`)
    # ----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ----------------------------------------------------------------------
    # 2. Linear projections and gating (all done with cuBLAS)
    # ----------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]   # shape (dim, hidden_dim)

    # Linear maps: (B,N,N,dim) -> (B,N,N,hidden_dim)
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Gates (pre‑sigmoid)
    left_gate_pre = torch.nn.functional.linear(x, left_gate_weight)
    right_gate_pre = torch.nn.functional.linear(x, right_gate_weight)
    out_gate_pre = torch.nn.functional.linear(x, out_gate_weight)

    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    out_gate = out_gate_pre.sigmoid()   # will be used after `to_out_norm`

    # ----------------------------------------------------------------------
    # 3. Optional mask
    # ----------------------------------------------------------------------
    if mask_tensor is not None and not config.get("nomask", False):
        # mask shape: [B,N,N] -> broadcast to hidden dim
        mask_exp = mask_tensor.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ----------------------------------------------------------------------
    # 4. Apply gates
    # ----------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 5. N^3 contraction:   out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
    #    Realised as a batched GEMM over the hidden dimension.
    # ----------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) and compute (B*H) independent matrix multiplies
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # out_mat = left_mat @ right_mat^T   -> shape (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------------------
    # 6. LayerNorm over hidden dimension (custom Triton kernel)
    # ----------------------------------------------------------------------
    out_norm = torch.empty_like(out)

    # Strides needed by the kernel (int64)
    s0, s1, s2, s3 = out.stride()   # (B,N,N,H) contiguous -> (N*N*H, N*H, H, 1)
    so0, so1, so2, so3 = out_norm.stride()

    total_positions = B * N * N
    grid = (total_positions,)

    to_out_layernorm_kernel[grid](
        out,
        out_norm,
        to_out_norm_weight,
        to_out_norm_bias,
        N,
        H,
        s0, s1, s2, s3,
        so0, so1, so2, so3,
        eps,
        BLOCK_H=H,
        num_warps=4,
    )

    # ----------------------------------------------------------------------
    # 7. Apply the output gate (element‑wise)
    # ----------------------------------------------------------------------
    out_norm = out_norm * out_gate

    # ----------------------------------------------------------------------
    # 8. Final linear projection back to `dim`
    # ----------------------------------------------------------------------
    # Collapse the first three dimensions for a single linear call
    out_flat = out_norm.view(B * N * N, H)                     # (B*N*N, H)
    final_flat = torch.nn.functional.linear(out_flat, to_out_weight)  # (B*N*N, dim)
    final = final_flat.view(B, N, N, -1)  # (B, N, N, dim)

    return final

"""
TriMul “outgoing” forward pass (AlphaFold‑3 style).

The expensive N³ contraction (the `einsum` in the reference implementation) is
performed as a batched GEMM with cuBLAS.  All remaining operations are carried
out with PyTorch except the second LayerNorm (the `to_out_norm` step), which
is implemented as a tiny Triton kernel that normalises the hidden dimension
for every (batch, i, j) entry.  This satisfies the requirement of having a
custom Triton kernel while keeping the overall runtime low.

The pipeline:
1. LayerNorm over the input feature dimension (`dim`).
2. Linear projections (`left`, `right`) and three sigmoid gates.
3. Optional pairwise mask.
4. Apply the left/right gates.
5. Compute  out[b,i,j,:] = Σₖ left[b,i,k,:] * right[b,j,k,:]
   → realised as a batched matrix multiplication over the hidden dimension.
6. LayerNorm over the hidden dimension (custom Triton kernel).
7. Multiply with the output gate.
8. Final linear projection back to `dim`.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: LayerNorm over the hidden dimension (size H)
# ----------------------------------------------------------------------
@triton.jit
def to_out_layernorm_kernel(
    inp_ptr, out_ptr,          # input (B,N,N,H) and output buffers
    norm_w_ptr, norm_b_ptr,    # weight and bias of the LayerNorm (size H)
    N, H,                      # sequence length and hidden dimension (runtime)
    stride_in0, stride_in1, stride_in2, stride_in3,   # input strides
    stride_out0, stride_out1, stride_out2, stride_out3,# output strides
    EPS: tl.constexpr,         # epsilon for numerical stability
    BLOCK_H: tl.constexpr      # compile‑time block size (== H)
):
    """
    One program instance processes a single (batch, i, j) vector of length H.
    """
    pid = tl.program_id(0)                     # linear id over B * N * N

    # Decompose pid → (b, i, j)
    nn = N * N
    b  = pid // nn
    rem = pid % nn
    i = rem // N
    j = rem % N

    # Base pointers for the vector (b,i,j,:)
    base_in  = inp_ptr  + b * stride_in0  + i * stride_in1  + j * stride_in2
    base_out = out_ptr + b * stride_out0 + i * stride_out1 + j * stride_out2

    # Offsets for the hidden dimension
    offs = tl.arange(0, BLOCK_H)
    mask = offs < H

    # Load the hidden‑dim vector
    vec = tl.load(base_in + offs * stride_in3, mask=mask, other=0.0)

    # ---------- 1st pass: mean ----------
    sum_val = tl.sum(vec)          # sum over H elements
    inv_H   = 1.0 / H
    mean    = sum_val * inv_H

    # ---------- 2nd pass: variance ----------
    diff = vec - mean
    var  = tl.sum(diff * diff) * inv_H

    # Normalise
    inv_std = 1.0 / tl.sqrt(var + EPS)
    normed  = diff * inv_std

    # Apply learnable scale (weight) and bias
    w = tl.load(norm_w_ptr + offs, mask=mask, other=0.0)
    b_ = tl.load(norm_b_ptr + offs, mask=mask, other=0.0)
    out_vec = normed * w + b_

    # Store result
    tl.store(base_out + offs * stride_out3, out_vec, mask=mask)


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor of shape [B, N, N, dim]
            - mask_tensor  : torch.Tensor of shape [B, N, N] (or None)
            - weights_dict : mapping from weight names to torch.Tensors
            - config_dict  : must contain "dim", "hidden_dim" and optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps        = 1e-5

    # ------------------------------------------------------------------
    # 1. First LayerNorm over the feature dimension `dim`
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections + gates (performed with cuBLAS)
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]          # (dim, hidden_dim)

    # Projections: (B,N,N,dim) → (B,N,N,hidden_dim)
    left  = torch.nn.functional.linear(x, left_proj_weight)   # no bias
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Gates (pre‑sigmoid)
    left_gate_pre  = torch.nn.functional.linear(x, left_gate_weight)
    right_gate_pre = torch.nn.functional.linear(x, right_gate_weight)
    out_gate_pre   = torch.nn.functional.linear(x, out_gate_weight)

    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    out_gate   = out_gate_pre.sigmoid()        # will be used after `to_out_norm`

    # ------------------------------------------------------------------
    # 3. Optional mask (if supplied and not disabled by config)
    # ------------------------------------------------------------------
    if mask_tensor is not None and not config.get("nomask", False):
        mask_exp = mask_tensor.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Apply left/right gates
    # ------------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. N³ contraction via batched GEMM
    #    out[b,i,j,:] = Σₖ left[b,i,k,:] * right[b,j,k,:]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Rearrange to (B, H, N, N) → (B*H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_mat   = left_perm.view(B * H, N, N)
    right_mat  = right_perm.view(B * H, N, N)

    # Batched matrix multiplication (left @ rightᵀ)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. LayerNorm over the hidden dimension (custom Triton kernel)
    # ------------------------------------------------------------------
    out_norm = torch.empty_like(out)

    # Strides (int64) for the kernel
    s0, s1, s2, s3   = out.stride()        # (B,N,N,H) – typically (N*N*H, N*H, H, 1)
    so0, so1, so2, so3 = out_norm.stride()

    total_positions = B * N * N
    grid = (total_positions,)

    to_out_layernorm_kernel[grid](
        out,
        out_norm,
        to_out_norm_weight,
        to_out_norm_bias,
        N,
        H,
        s0, s1, s2, s3,
        so0, so1, so2, so3,
        eps,
        BLOCK_H=hidden_dim,      # compile‑time block size
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7. Apply the output gate (element‑wise)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    # Collapse leading dimensions for a single cuBLAS call
    out_flat = out_norm.view(B * N * N, hidden_dim)          # (B*N*N, H)
    final_flat = torch.nn.functional.linear(out_flat, to_out_weight)  # (B*N*N, dim)
    final = final_flat.view(B, N, N, -1)                     # (B, N, N, dim)

    return final
