def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data

    dim = config["dim"]; hidden_dim = config["hidden_dim"];  # from config

    # maybe also config includes "nomask": bool, else check.

    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Normalize
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Project left/right
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    if not config.get("nomask", False):
        # mask shape [B,N,N]; convert to float dtype same as x
        mask = mask_tensor.to(dtype=x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Compute triangular multiplication via GEMM with Triton
    # Rearrange tensors to shape [B*hidden, N, N]
    B, N, _, _ = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, hidden, N, N]
    # reshape
    batch_heads = B * hidden_dim
    left_flat = left_perm.view(batch_heads, N, N)
    right_flat = right_perm.view(batch_heads, N, N)
    # transpose right for GEMM
    right_T = right_flat.transpose(1, 2).contiguous()

    # Triton GEMM
    out_flat = tri_mul_gemm(left_flat, right_T)  # shape [batch_heads, N, N]

    # Reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,hidden]

    # to_out_norm (layernorm on hidden)
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # Apply out_gate scaling
    out = out * out_gate

    # Final linear to dim
    out = torch.nn.functional.linear(out, to_out_weight)  # bias None
    return out

"""
TriMul (outgoing) fused kernel.

This implementation follows the PyTorch reference but accelerates the
core N³ “einsum” using a custom Triton GEMM kernel.  The algorithm is:

1. Layer‑norm the input (dim‑wise).
2. Linear projections → left / right tensors.
3. Optional pairwise mask applied to left & right.
4. Gating (sigmoid of linear projections) applied element‑wise.
5. For every (batch, head) pair compute
       out[b, i, j, h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batch of N×N matrix products with the second operand
   transposed.  This is performed by the Triton kernel `trimul_gemm_kernel`.
6. Layer‑norm + out‑gate on the hidden dimension.
7. Final linear projection back to `dim`.

All tensors stay on GPU, the kernel works for any batch ∈ {1,2},
sequence length up to 1024 and hidden dimensions 128/384.
The kernel uses float32 accumulation (FP16 could be used for extra speed
but FP32 guarantees numerical stability for the test suite).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton GEMM kernel: C = A @ B   where A:[B, M, K], B:[B, K, N]
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    batch_id = tl.program_id(0)          # which matrix (batch*head)
    pid_m = tl.program_id(1)             # row tile
    pid_n = tl.program_id(2)             # col tile

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds (final partial tiles)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute base pointers for this matrix
    a_ptrs = (A_ptr +
              batch_id * stride_a_batch +
              offs_m[:, None] * stride_a_m +
              offs_k[None, :] * stride_a_k)

    b_ptrs = (B_ptr +
              batch_id * stride_b_batch +
              offs_k[:, None] * stride_b_k +
              offs_n[None, :] * stride_b_n)

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # load A and B tiles with proper masking
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & (offs_k[None, :] < K)),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=((offs_k[:, None] < K) & mask_n[None, :]),
                    other=0.0)
        # fused dot product
        acc += tl.dot(a, b)

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # write out the result
    c_ptrs = (C_ptr +
              batch_id * stride_c_batch +
              offs_m[:, None] * stride_c_m +
              offs_n[None, :] * stride_c_n)

    tl.store(c_ptrs, acc,
             mask=(mask_m[:, None] & mask_n[None, :]))


def trimul_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM: C = A @ B   (A: [B, M, K], B: [B, K, N])
    Works with any contiguous or non‑contiguous layout; strides are extracted.
    """
    assert A.is_cuda and B.is_cuda
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "Inner dimensions must match for GEMM"

    # output allocation
    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # strides as integers
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # block sizes – tuned for H100 (fits in shared memory)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (batch,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    trimul_gemm_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the TriMul “outgoing” operator.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        output tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # unpack
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (mask shape: [B,N,N])
    # ------------------------------------------------------------------
    if not nomask:
        # broadcast mask to the last dim
        mask = mask_tensor.to(dtype=x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul via Triton GEMM
    #    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    This is equivalent to (left_h).matmul((right_h).transpose(-2,-1))
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Bring the hidden dimension up front for easy reshaping
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, hidden, N, N]

    # Merge batch and hidden dimensions → treat each (b, h) as an independent GEMM
    batch_heads = B * hidden_dim
    left_flat = left_perm.view(batch_heads, N, N)        # [BH, N, N]
    right_flat = right_perm.view(batch_heads, N, N)      # [BH, N, N]

    # We need Bᵀ for the GEMM: (right_flat)ᵀ → shape [BH, N, N] with swapped axes
    right_T = right_flat.transpose(1, 2)  # non‑contiguous view, strides handled by Triton

    # Triton GEMM: out_flat = left_flat @ right_T
    out_flat = trimul_gemm(left_flat, right_T)  # [BH, N, N]

    # Restore original ordering: [B, N, N, hidden]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output layer‑norm, gating, and final projection back to `dim`
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # bias is None

    return out
