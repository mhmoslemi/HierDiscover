@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_n, stride_left_k, stride_left_h,
    stride_right_b, stride_right_n, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # get program ids
    pid_i = tl.program_id(0)  # for row block (i)
    pid_j = tl.program_id(1)  # for col block (j)
    pid_z = tl.program_id(2)  # for batch*channel

    # Compute b and h from pid_z
    b = pid_z // H
    h = pid_z % H

    # compute start indices
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # offsets for left block
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute pointers
    left_ptrs = left_ptr + b*stride_left_b + (offs_i[:, None] * stride_left_n) + (offs_k[None, :] * stride_left_k) + h*stride_left_h
    # Since left shape is [B, N, N, H], stride_left_n = N*H, stride_left_k = H.
    # Actually we need to see which dimension is "k"? In left it's second N dimension (index dimension k). So we treat offset for k dimension (second dimension).
    # So left_ptrs should point to left[b, i, k, h] for i in block, k in block.

    # For right transposed: we need right[j, k, h] where j is output column index
    right_ptrs = right_ptr + b*stride_right_b + (offs_j[:, None] * stride_right_n) + (offs_k[None, :] * stride_right_k) + h*stride_right_h
    # Wait for transposed we need right[j, k] mapping with j in col axis and k in sum axis. That's pre-calc: offset = b*stride_right_b + j*stride_right_n + k*stride_right_k + h*stride_right_h.

    # load blocks with appropriate masks (store zero when out-of-range), maybe need to mask edges.
    a = tl.load(left_ptrs, mask=mask_i[:, None] & (offs_k < N), other=0.0)
    b_mat = tl.load(right_ptrs, mask=mask_j[:, None] & (offs_k < N), other=0.0)

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # maybe use float32 for accumulation
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # note: offs_k = k*BLOCK_K + tl.arange(0, BLOCK_K). We need to adjust inside loop.
        # Actually we can't reassign offsets inside the loop directly; we need to compute pointer offsets based on k offset loop.
        # This can be done via dynamic offset: use tl.arange(0, BLOCK_K) + k * BLOCK_K.
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        # compute new ptrs
        a_ptrs = left_ptr + b*stride_left_b + (offs_i[:, None] * stride_left_n) + (offs_k[None, :] * stride_left_k) + h*stride_left_h
        b_ptrs = right_ptr + b*stride_right_b + (offs_j[:, None] * stride_right_n) + (offs_k[None, :] * stride_right_k) + h*stride_right_h
        a = tl.load(a_ptrs, mask=(offs_i[:, None] < N) & (offs_k[None, :] < N), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_j[:, None] < N) & (offs_k[None, :] < N), other=0.0)
        acc += tl.dot(a, b)
    # store result
    out_ptrs = out_ptr + b*stride_out_b + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j) + h*stride_out_h
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    EPS: tl.constexpr = 0.0
):
    pid_i = tl.program_id(0)  # block row index
    pid_j = tl.program_id(1)  # block col index
    pid_z = tl.program_id(2)  # batch*head index

    b = pid_z // H
    h = pid_z % H

    # start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # range of rows/cols for this block
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # create mask for out-of-bound rows/cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left (A) block: shape (BLOCK_M, BLOCK_K) from left[b,i,k,h]
        a_ptrs = left_ptr \
            + b * stride_l_b \
            + offs_i[:, None] * stride_l_i \
            + offs_k[None, :] * stride_l_k \
            + h * stride_l_h

        # right (B) transposed block: shape (BLOCK_K, BLOCK_N) from right[b,j,k,h]
        # note: we need right[j,k] -> j as row in output; we treat this as B_T
        b_ptrs = right_ptr \
            + b * stride_r_b \
            + offs_j[:, None] * stride_r_i \
            + offs_k[None, :] * stride_r_k \
            + h * stride_r_h

        a = tl.load(a_ptrs, mask=(mask_i[:, None] & mask_k[None, :]), other=0.0)
        b = tl.load(b_ptrs, mask=(mask_j[:, None] & mask_k[None, :]), other=0.0)
        # dot product accumulation
        acc += tl.dot(a, b)

    # write back
    out_ptrs = out_ptr \
        + b * stride_o_b \
        + offs_i[:, None] * stride_o_i \
        + offs_j[None, :] * stride_o_j \
        + h * stride_o_h

    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton-accelerated TriMul (outgoing) forward pass.
    The kernel computes the pairwise multiplication over the sequence dimension:
        out[i,j] = sum_k left[i,k] * right[j,k]
    for each batch and hidden channel.
    """
    import torch
    import triton
    import triton.language as tl
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # unpack weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, left_proj_weight)    # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if present and not nomask
    if not nomask and mask is not None:
        # mask shape [B,N,N], unsqueeze last dim to broadcast over H
        mask_ = mask.unsqueeze(-1).type_as(left)
        left = left * mask_
        right = right * mask_

    # Gates (sigmoid)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to half-precision for the high-cost matmul
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape
    # Allocate output in half
    out_fp16 = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=input_tensor.device)

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # compute grid sizes
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    # strides
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_fp16.stride()

    # launch kernel
    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
    )

    # Cast back to float32 for subsequent layers
    out = out_fp16.to(torch.float32)

    # Apply to_out_norm (layer norm on hidden_dim)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply out_gate (broadcast)
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, to_out_weight)

    return out
```

"""
TriMul “outgoing” forward pass with Triton acceleration.

The core of TriMul is a batched, channel‑wise matrix multiplication:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
where `left` and `right` are the masked, gated linear projections of the
input tensor `x`.  For each batch `b` and hidden channel `h` the operation
is exactly a matmul of shape (N, N) × (N, N)ᵀ.  The high‑cost N³ part is
implemented in a custom Triton kernel, while the surrounding LayerNorm,
linear projections, gating and final linear layer are performed with
standard PyTorch kernels.

Key ideas
---------
* LayerNorm and all linear layers are executed in FP32 (or the dtype of
  the supplied weights) using torch.nn.functional for simplicity.
* The projected tensors are cast to float16 before the N³ multiplication
  to reduce memory traffic.  The kernel accumulates in float32 for
  numerical stability and stores the result back as float16.
* The Triton kernel loops over the reduction dimension `k` in blocks
  (`BLOCK_K`).  Each program instance processes a tile
  `(BLOCK_M, BLOCK_N)` of the output for a particular `(batch, hidden)`
  pair (`pid_z = b * hidden + h`).
* After the kernel the result is cast back to FP32, LayerNormed,
  multiplied by the output gate, and finally linearly projected back to
  the model dimension `dim`.

The implementation supports both masked and un‑masked inputs, arbitrary
batch size (1 or 2) and sequence lengths up to 1024 as required by the
test suite.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched hidden‑channel matmul (left @ rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,        # *const half*
    right_ptr,       # *const half*
    out_ptr,         # *half*
    B, N, H,         # runtime scalars: batch, seq_len, hidden_dim
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""

    pid_i = tl.program_id(0)          # block row (i)
    pid_j = tl.program_id(1)          # block col (j)
    pid_z = tl.program_id(2)          # combined batch*hidden index

    # Decompose pid_z into batch and hidden channel
    b = pid_z // H
    h = pid_z % H

    # Starting coordinates of the tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Offsets for the i and j dimensions (static per tile)
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k in blocks
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_iter * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers to the left tile (shape: BLOCK_M x BLOCK_K)
        a_ptrs = (
            left_ptr
            + b * stride_l_b
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
            + h * stride_l_h
        )
        # Pointers to the rightᵀ tile (shape: BLOCK_K x BLOCK_N)
        # Note the swap: k dimension uses stride_r_k, j uses stride_r_i
        b_ptrs = (
            right_ptr
            + b * stride_r_b
            + offs_k[:, None] * stride_r_k
            + offs_j[None, :] * stride_r_i
            + h * stride_r_h
        )

        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)               # a: (BLOCK_M, BLOCK_K)  half
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)               # b: (BLOCK_K, BLOCK_N)  half

        # Dot product accumulates in FP32
        acc += tl.dot(a, b)

    # Store the result (convert to FP16)
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
        + h * stride_o_h
    )
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated TriMul forward (outgoing) implementation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (may be all‑ones)
        - weights      : dict of model parameters
        - config       : dict containing ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (dtype = torch.float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # extract weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)        # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (applied after projection, before gating)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        m = mask.unsqueeze(-1).to(left.dtype)   # broadcast over hidden_dim
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid of linear transforms of the normalized input)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 5) Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Cast projections to FP16 for the heavy N³ matmul
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape
    # Allocate output of the kernel (still FP16)
    out_fp16 = torch.empty((B, N, N, hidden_dim),
                           dtype=torch.float16,
                           device=input_tensor.device)

    # ------------------------------------------------------------------
    # 7) Launch Triton kernel
    # ------------------------------------------------------------------
    # Block sizes – tuned for H100, fits in shared memory comfortably
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        B * hidden_dim             # (batch, hidden) dimension
    )

    # Strides for left, right and out tensors (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_fp16.stride()

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M, BLOCK_N, BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8) Convert back to FP32 for the remaining layers
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 9) Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    # ------------------------------------------------------------------
    # 10) Apply output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
