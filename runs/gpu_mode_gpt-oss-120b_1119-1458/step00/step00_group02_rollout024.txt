```python
@triton.jit
def apply_gate_mask_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    B, N, H,
    stride_lp_b, stride_lp_i, stride_lp_k, stride_lp_h,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rp_b, stride_rp_i, stride_rp_k, stride_rp_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_m_b, stride_m_i, stride_m_k,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    BLOCK: tl.constexpr, HAS_MASK: tl.constexpr
):
    pid = tl.program_id(0)
    # compute linear index for elements (b,i,k)
    num_elements = B * N * N
    offset = pid * BLOCK + tl.arange(0, BLOCK)
    mask_in_range = offset < num_elements

    # compute (b,i,k) for each element
    b_idx = offset // (N * N)
    rem = offset % (N * N)
    i_idx = rem // N
    k_idx = rem % N

    # compute mask for each thread
    mask_val = tl.full((BLOCK,), 1.0, dtype=tl.float16)
    if HAS_MASK:
        # mask pointer addresses
        mask_ptrs = mask_ptr + b_idx * stride_m_b + i_idx * stride_m_i + k_idx * stride_m_k
        mask_val = tl.load(mask_ptrs, mask=mask_in_range, other=1.0)  # scalar per element

    # compute d offsets:
    d_offs = tl.arange(0, H)

    # left_proj load
    lp_ptrs = left_proj_ptr + b_idx * stride_lp_b + i_idx * stride_lp_i + k_idx * stride_lp_k + d_offs * stride_lp_h
    left_proj = tl.load(lp_ptrs, mask=mask_in_range[:, None], other=0.0)   # shape (BLOCK, H)

    # left_gate load
    lg_ptrs = left_gate_ptr + b_idx * stride_lg_b + i_idx * stride_lg_i + k_idx * stride_lg_k + d_offs * stride_lg_h
    left_gate = tl.load(lg_ptrs, mask=mask_in_range[:, None], other=0.0)

    # compute left_out = left_proj * left_gate * mask_val
    # broadcast mask_val (shape BLOCK) to (BLOCK, H)
    mask_val_vec = mask_val[:, None]
    left_out = left_proj * left_gate * mask_val_vec

    # store left_out
    lo_ptrs = left_out_ptr + b_idx * stride_out_b + i_idx * stride_out_i + k_idx * stride_out_k + d_offs * stride_out_h
    tl.store(lo_ptrs, left_out, mask=mask_in_range[:, None])

    # right_proj load
    rp_ptrs = right_proj_ptr + b_idx * stride_rp_b + i_idx * stride_rp_i + k_idx * stride_rp_k + d_offs * stride_rp_h
```

```python
def custom_kernel(data):
    """
    Triton-accelerated TriMul "outgoing" update.
    This kernel:
    - Applies LayerNorm to the input.
    - Projects input to a hidden dimension (left and right projections).
    - Computes gating values and applies mask (if any) using a Triton kernel.
    - Performs the core O(N^3) contraction using batched matrix multiplication.
    - Applies output LayerNorm, out-gate, and final linear projection.
    The heavy elementwise fusion of mask and gates is done in Triton.
    """
    import torch
    import triton
    import triton.language as tl

    # unpack
    input_tensor, mask, weights, config = data

    # Config params
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    device = input_tensor.device
    dtype = torch.float16   # use half precision for speed

    # Cast inputs to fp16
    x = input_tensor.to(dtype)

    # LayerNorm (input) using torch (built-in)
    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias = weights["norm.bias"].to(dtype)
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections (left/right) using torch.nn.functional.linear
    left_proj_weight = weights["left_proj.weight"].to(dtype)   # shape (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"].to(dtype)

    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Gates
    left_gate_weight = weights["left_gate.weight"].to(dtype)
    right_gate_weight = weights["right_gate.weight"].to(dtype)
    out_gate_weight = weights["out_gate.weight"].to(dtype)

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # If no mask, treat as ones. Optimization: we still need to provide mask tensor for kernel when HAS_MASK True
    has_mask = mask is not None and mask.numel() > 0 and not config.get("nomask", False)
    if has_mask:
        mask_fp = mask.to(dtype).unsqueeze(-1)  # shape [B,N,N,1] for broadcasting later
    else:
        mask_fp = None

    # Allocate final left and right after gating+mask
    left_fused = torch.empty_like(left)   # same shape and dtype
    right_fused = torch.empty_like(right)

    # Launch Triton kernel to fuse mask + gates
    # Compute strides
    B, N, _, H = left.shape  # H=hidden_dim
    # All tensors are contiguous after linear; ensure contiguity
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    if has_mask:
        mask_tensor = mask_fp.squeeze(-1).contiguous()  # shape [B,N,N]
    else:
        mask_tensor = torch.empty(0, device=device, dtype=dtype)  # placeholder

    # stride values
    stride_lp_b, stride_lp_i, stride_lp_k, stride_lp_h = left.stride()
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h = left_gate.stride()
    stride_rp_b, stride_rp_i, stride_rp_k, stride_rp_h = right.stride()
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h = right_gate.stride()
    stride_out_b, stride_out_i, stride_out_k, stride_out_h = left_fused.stride()
    if has_mask:
        stride_m_b, stride_m_i, stride_m_k = mask_tensor.stride()
    else:
        stride_m_b = stride_m_i = stride_m_k = 0

    BLOCK = 128  # tune block size
    grid = ( (B * N * N + BLOCK - 1) // BLOCK, )
    # Launch kernel
    apply_gate_mask_kernel[grid](
        left.data_ptr(),
        left_gate.data_ptr(),
        right.data_ptr(),
        right_gate.data_ptr(),
        mask_tensor.data_ptr() if has_mask else 0,
        left_fused.data_ptr(),
        right_fused.data_ptr(),
        B, N, H,
        stride_lp_b, stride_lp_i, stride_lp_k, stride_lp_h,
        stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
        stride_rp_b, stride_rp_i, stride_rp_k, stride_rp_h,
        stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
        stride_m_b, stride_m_i, stride_m_k,
        stride_out_b, stride_out_i, stride_out_k, stride_out_h,
        BLOCK=BLOCK,
        HAS_MASK=has_mask,
    )

    # Now compute contraction using batched GEMM
    # reshape to (B*H,N,N)
    left_bh = left_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_bh = right_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # Batched matrix multiplication: left * right^T
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))  # shape (B*H,N,N)

    # reshape back to (B,N,N,H)
    out = out_bh.reshape(B, H, N, N).permute(0, 2, 3, 1)  # shape (B,N,N,H)

    # Output LayerNorm
    to_out_norm_weight = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(dtype)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply out_gate (already computed in half)
    out = out * out_gate

    # Final linear projection back to original dim
    to_out_weight = weights["to_out.weight"].to(dtype)  # shape (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)  # shape (B,N,N,dim)

    # Cast back to float32 to match expected dtype
    out = out.to(torch.float32)

    return out
```

```
@triton.jit
def apply_gate_mask_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    B, N, H,
    stride_lp_b, stride_lp_i, stride_lp_k, stride_lp_h,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rp_b, stride_rp_i, stride_rp_k, stride_rp_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_m_b, stride_m_i, stride_m_k,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    BLOCK: tl.constexpr, CHUNK: tl.constexpr, HAS_MASK: tl.constexpr
):
    pid = tl.program_id(0)
    # Compute linear index range
    num_elements = B * N * N
    idx = pid * BLOCK + tl.arange(0, BLOCK)
    mask_valid = idx < num_elements

    # Decompose idx into b,i,k
    b_idx = idx // (N * N)
    rem = idx % (N * N)
    i_idx = rem // N
    k_idx = rem % N

    # Base pointers offsets for each thread (without hidden dim)
    base_lp = b_idx * stride_lp_b + i_idx * stride_lp_i + k_idx * stride_lp_k
    base_lg = b_idx * stride_lg_b + i_idx * stride_lg_i + k_idx * stride_lg_k
    base_rp = b_idx * stride_rp_b + i_idx * stride_rp_i + k_idx * stride_rp_k
    base_rg = b_idx * stride_rg_b + i_idx * stride_rg_i + k_idx * stride_rg_k
    base_out = b_idx * stride_out_b + i_idx * stride_out_i + k_idx * stride_out_k
    # mask pointer base
    if HAS_MASK:
        base_mask = b_idx * stride_m_b + i_idx * stride_m_i + k_idx * stride_m_k

    # Load mask value if needed
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + base_mask, mask=mask_valid, other=1.0)
    else:
        mask_val = tl.full((BLOCK,), 1.0, dtype=tl.float16)  # shape (BLOCK,)
    # broadcast mask_val for later multiplication
    mask_val = mask_val[:, None]  # shape (BLOCK,1)

    # Number of chunks over hidden dimension
    num_chunks = tl.cdiv(H, CHUNK)
    for chunk in range(0, num_chunks):
        d_start = chunk * CHUNK
        # range of d indices for this chunk
        d_offs = tl.arange(0, CHUNK) + d_start
        # Determine valid within hidden_dim
        d_mask = d_offs < H
        # final mask to apply on loads/store
        valid = mask_valid[:, None] & d_mask[None, :]

        # Load slices
        lp = tl.load(left_proj_ptr + base_lp[:, None] + d_offs[None, :] * stride_lp_h,
                     mask=valid, other=0.0)
        lg = tl.load(left_gate_ptr + base_lg[:, None] + d_offs[None, :] * stride_lg_h,
                     mask=valid, other=0.0)
        rp = tl.load(right_proj_ptr + base_rp[:, None] + d_offs[None, :] * stride_rp_h,
                     mask=valid, other=0.0)
        rg = tl.load(right_gate_ptr + base_rg[:, None] + d_offs[None, :] * stride_rg_h,
                     mask=valid, other=0.0)

        # Compute fused values
        left_val = lp * lg * mask_val
        right_val = rp * rg * mask_val

        # Store
        tl.store(left_out_ptr + base_out[:, None] + d_offs[None, :] * stride_out_h,
                 left_val, mask=valid)
        tl.store(right_out_ptr + base_out[:, None] + d_offs[None, :] * stride_out_h,
                 right_val, mask=valid)
```

"""
TriMul “outgoing” update – Triton‑accelerated forward pass.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B, N, N, C]).
2. Project the normalized tensor to a hidden dimension H with two linear layers:
   left_proj = x @ W_left   , right_proj = x @ W_right.
3. Compute three sigmoid gates (left_gate, right_gate, out_gate) with linear layers.
4. Fuse mask (if present), left/right gates and the projections with a Triton kernel.
   The kernel processes each (b,i,k) entry and updates the hidden dimension in
   chunks (default CHUNK=32) to keep register pressure low.
5. Perform the core O(N³) contraction using a batched GEMM:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is implemented as a batched matrix multiplication after reshaping:
        left  → (B·H, N, N)
        right → (B·H, N, N)ᵀ
6. Apply a second Layer‑norm over the hidden dimension, multiply by the out‑gate,
   and finally project back to the original dimension with a linear layer.
7. The result is returned in float32.

Only the element‑wise gating/mask fusion is executed in Triton, while the heavy
matrix multiplication leverages cuBLAS (via torch.bmm), which is optimal on H100.
Mixed‑precision (float16) is used throughout the forward pass for speed.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse (proj * gate) and optional mask for left/right tensors.
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_mask_kernel(
    # input tensors
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,                       # uint8/float16 mask (optional)
    # output tensors
    left_out_ptr, right_out_ptr,
    # problem size
    B, N, H,
    # strides for left_proj
    stride_lp_b, stride_lp_i, stride_lp_k, stride_lp_h,
    # strides for left_gate
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    # strides for right_proj
    stride_rp_b, stride_rp_i, stride_rp_k, stride_rp_h,
    # strides for right_gate
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    # strides for mask (if present)
    stride_m_b, stride_m_i, stride_m_k,
    # strides for output (same for left/right)
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    # compile‑time constants
    BLOCK: tl.constexpr,            # #threads per program (covers B*N*N)
    CHUNK: tl.constexpr,            # hidden‑dim chunk size (≤32‑64 works well)
    HAS_MASK: tl.constexpr,         # 1 → use mask, 0 → ignore mask
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 1‑D index over the (B, N, N) domain
    # --------------------------------------------------------------
    total = B * N * N
    idx = pid * BLOCK + tl.arange(0, BLOCK)               # shape (BLOCK,)
    valid_idx = idx < total

    # decode linear index → (b, i, k)
    b_idx = idx // (N * N)
    rem   = idx % (N * N)
    i_idx = rem // N
    k_idx = rem % N

    # --------------------------------------------------------------
    # Base offsets for each tensor (without hidden dim)
    # --------------------------------------------------------------
    base_lp = b_idx * stride_lp_b + i_idx * stride_lp_i + k_idx * stride_lp_k
    base_lg = b_idx * stride_lg_b + i_idx * stride_lg_i + k_idx * stride_lg_k
    base_rp = b_idx * stride_rp_b + i_idx * stride_rp_i + k_idx * stride_rp_k
    base_rg = b_idx * stride_rg_b + i_idx * stride_rg_i + k_idx * stride_rg_k
    base_out = b_idx * stride_out_b + i_idx * stride_out_i + k_idx * stride_out_k

    # --------------------------------------------------------------
    # Load mask value (scalar) if present, otherwise use 1.0
    # --------------------------------------------------------------
    if HAS_MASK:
        base_m = b_idx * stride_m_b + i_idx * stride_m_i + k_idx * stride_m_k
        mask_scalar = tl.load(mask_ptr + base_m, mask=valid_idx, other=1.0)   # (BLOCK,)
    else:
        mask_scalar = tl.full((BLOCK,), 1.0, dtype=tl.float16)               # (BLOCK,)

    # broadcast to hidden‑dim axis
    mask_scalar = mask_scalar[:, None]      # (BLOCK, 1)

    # --------------------------------------------------------------
    # Loop over hidden dimension in chunks to stay within register budget
    # --------------------------------------------------------------
    num_chunks = tl.cdiv(H, CHUNK)
    for chunk in range(0, num_chunks):
        d_start = chunk * CHUNK
        d_offs = tl.arange(0, CHUNK) + d_start                # (CHUNK,)
        d_valid = d_offs < H                                   # (CHUNK,)

        # combined mask for (BLOCK, CHUNK) elements
        active = valid_idx[:, None] & d_valid[None, :]

        # ----- left_proj -------------------------------------------------
        lp = tl.load(
            left_proj_ptr + base_lp[:, None] + d_offs[None, :] * stride_lp_h,
            mask=active, other=0.0)
        # ----- left_gate -------------------------------------------------
        lg = tl.load(
            left_gate_ptr + base_lg[:, None] + d_offs[None, :] * stride_lg_h,
            mask=active, other=0.0)
        # fused left output
        left_val = lp * lg * mask_scalar

        tl.store(
            left_out_ptr + base_out[:, None] + d_offs[None, :] * stride_out_h,
            left_val, mask=active)

        # ----- right_proj ------------------------------------------------
        rp = tl.load(
            right_proj_ptr + base_rp[:, None] + d_offs[None, :] * stride_rp_h,
            mask=active, other=0.0)
        # ----- right_gate ------------------------------------------------
        rg = tl.load(
            right_gate_ptr + base_rg[:, None] + d_offs[None, :] * stride_rg_h,
            mask=active, other=0.0)
        # fused right output
        right_val = rp * rg * mask_scalar

        tl.store(
            right_out_ptr + base_out[:, None] + d_offs[None, :] * stride_out_h,
            right_val, mask=active)

# User‑facing entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” update.
    Arguments:
        data = (input_tensor, mask_tensor, weight_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    import torch.nn.functional as F

    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    # Configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    has_mask = (mask_tensor is not None) and (not config.get("nomask", False))

    # ------------------------------------------------------------------
    # Cast everything to float16 for speed (final result will be float32)
    # ------------------------------------------------------------------
    dtype = torch.float16
    x = input_tensor.to(dtype)

    # ------------------------------------------------------------------
    # Input LayerNorm (equivalent to nn.LayerNorm(dim))
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
        eps=eps,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections to hidden_dim
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"].to(dtype))   # [B,N,N,H]
    right_proj = F.linear(x, weights["right_proj.weight"].to(dtype))

    # ------------------------------------------------------------------
    # Sigmoid gates (still in half precision)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"].to(dtype)).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"].to(dtype)).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"].to(dtype)).sigmoid()   # [B,N,N,H]

    # ------------------------------------------------------------------
    # Prepare buffers for gated+masked tensors
    # ------------------------------------------------------------------
    left_fused = torch.empty_like(left_proj)
    right_fused = torch.empty_like(right_proj)

    # Ensure contiguity – required for correct stride extraction
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    left_fused = left_fused.contiguous()
    right_fused = right_fused.contiguous()

    # Mask handling (float16, broadcasted later)
    if has_mask:
        mask_fp = mask_tensor.to(dtype).contiguous()            # [B, N, N]
    else:
        mask_fp = torch.empty(0, device=device, dtype=dtype)    # placeholder

    # ------------------------------------------------------------------
    # Extract strides (B,N,N,H layout is contiguous)
    # ------------------------------------------------------------------
    B, N, _, H = left_proj.shape
    stride_lp_b, stride_lp_i, stride_lp_k, stride_lp_h = left_proj.stride()
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h = left_gate.stride()
    stride_rp_b, stride_rp_i, stride_rp_k, stride_rp_h = right_proj.stride()
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h = right_gate.stride()
    stride_out_b, stride_out_i, stride_out_k, stride_out_h = left_fused.stride()

    if has_mask:
        stride_m_b, stride_m_i, stride_m_k = mask_fp.stride()
    else:
        stride_m_b = stride_m_i = stride_m_k = 0

    # ------------------------------------------------------------------
    # Launch Triton kernel to fuse mask, projections and gates
    # ------------------------------------------------------------------
    BLOCK = 128          # threads per program (covers B*N*N elements)
    CHUNK = 32           # hidden‑dim chunk size (fits in registers)
    grid = ( (B * N * N + BLOCK - 1) // BLOCK, )
    apply_gate_mask_kernel[grid](
        left_proj.data_ptr(),
        left_gate.data_ptr(),
        right_proj.data_ptr(),
        right_gate.data_ptr(),
        mask_fp.data_ptr() if has_mask else 0,
        left_fused.data_ptr(),
        right_fused.data_ptr(),
        B, N, H,
        stride_lp_b, stride_lp_i, stride_lp_k, stride_lp_h,
        stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
        stride_rp_b, stride_rp_i, stride_rp_k, stride_rp_h,
        stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
        stride_m_b, stride_m_i, stride_m_k,
        stride_out_b, stride_out_i, stride_out_k, stride_out_h,
        BLOCK=BLOCK,
        CHUNK=CHUNK,
        HAS_MASK=int(has_mask),
    )

    # ------------------------------------------------------------------
    # Core O(N³) contraction: batched GEMM over the hidden dimension
    # ------------------------------------------------------------------
    # Reshape tensors to (B*H, N, N) for torch.bmm
    left_bh = left_fused.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # [B*H,N,N]
    right_bh = right_fused.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N) # [B*H,N,N]

    # out_bh = left_bh @ right_bhᵀ  → shape (B*H, N, N)
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))

    # Restore original layout [B, N, N, H]
    out = out_bh.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Output LayerNorm, out‑gate, and final projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(dtype))   # [B,N,N,dim]

    # Return in float32 as prescribed by the reference implementation
    return out.to(torch.float32)
