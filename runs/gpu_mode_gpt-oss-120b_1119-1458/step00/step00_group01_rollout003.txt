@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # compute bh index
    bh = tl.program_id(0)  # range B*H
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    total_bh = B * H
    b = bh // H
    h = bh % H

    # offsets for i and j block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers base offsets for batch and channel
    base_left = left_ptr + b * stride_left_b + h * stride_left_h
    base_right = right_ptr + b * stride_right_b + h * stride_right_h
    base_out = out_ptr + b * stride_out_b + h * stride_out_h

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block (M,K)
        # left_ptr[ b,i,k,h]
        left_offset = offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(base_left + left_offset,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0, dtype=tl.float16)

        # Load right block (N,K)
        right_offset = offs_n[:, None] * stride_right_j + offs_k[None, :] * stride_right_k
        right = tl.load(base_right + right_offset,
                       mask=mask_n[:, None] & mask_k[None, :],
                       other=0.0, dtype=tl.float16)

        # Compute dot product over K, accumulate as float32
        acc += tl.dot(left, right, out_dtype=tl.float32)

    # Write out
    out_offset = offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(base_out + out_offset, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) implementation with a fused Triton kernel.

The kernel computes the Einstein‑summation
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

for all batches `b` and hidden channels `d`.  This is the
computational bottleneck of the TriMul module (an O(N³·H) operation).
The surrounding LayerNorm, linear projections, and gating are performed
with PyTorch for simplicity; the heavy reduction over the `k` dimension
is carried out in a custom Triton kernel that tiles the i‑ and j‑axes
and performs a dot‑product over blocks of `k`.  The kernel works on
FP16 inputs (to reduce memory traffic) while accumulating in FP32 for
numerical stability.

The overall algorithm:
1. Layer‑norm `x` over the feature dimension `dim`.
2. Linear projections → `left`, `right` (shape B,N,N,H).
3. Optional mask application.
4. Gating (sigmoid of linear projections) and element‑wise gating of
   `left` and `right`.
5. Triton kernel: batched matmul over the `k` dimension for each
   hidden channel → `out` (shape B,N,N,H).
6. Layer‑norm over the hidden dimension, gating by `out_gate`,
   final linear projection back to `dim`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,                # pointers
    B, N, H,                                     # dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute the TriMul contraction for a tile of size BLOCK_M × BLOCK_N.
    The reduction over K is tiled with BLOCK_K.
    """

    # 3‑D grid: (batch*hidden, tile_i, tile_j)
    pid_bh = tl.program_id(0)      # index = b * H + h
    pid_i  = tl.program_id(1)      # i‑tile
    pid_j  = tl.program_id(2)      # j‑tile

    # recover batch and hidden indices
    b = pid_bh // H
    h = pid_bh - b * H

    # offsets for the i‑ and j‑tiles
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds i / j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers that already include the batch and hidden offsets
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- left block: shape (BLOCK_M, BLOCK_K) -----
        # left[b, i, k, h]  ->  i*stride_i + k*stride_k
        left_offs = offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(
            left_base + left_offs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16
        )

        # ----- right block: shape (BLOCK_N, BLOCK_K) -----
        # right[b, j, k, h]  ->  j*stride_j + k*stride_k
        right_offs = offs_j[:, None] * stride_right_j + offs_k[None, :] * stride_right_k
        right = tl.load(
            right_base + right_offs,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16
        )

        # dot over K, accumulate in FP32
        acc += tl.dot(left, right, out_dtype=tl.float32)

    # ----- write result -----
    out_offs = offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(
        out_base + out_offs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :]
    )


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Configuration / shapes
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    B, N, _, C = input_tensor.shape
    assert C == dim, f"Input feature dimension {C} != expected dim {dim}"
    H = hidden_dim  # channel dimension after projection

    # ------------------------------------------------------------------
    # Load weights (all are stored as float32 tensors)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) LayerNorm over the feature dimension (`dim`)
    # ------------------------------------------------------------------
    # Compute mean / var across last dimension
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var  = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias   # broadcast over batch/seq

    # Cast to half precision for the heavy part (projected tensors)
    x_norm_fp16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x_norm_fp16, left_proj_weight.to(torch.float16))   # [B,N,N,H]
    right = F.linear(x_norm_fp16, right_proj_weight.to(torch.float16))  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    use_mask = (mask_tensor is not None) and (not config.get("nomask", False))
    if use_mask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left  = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4) Gating networks (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm_fp16, left_gate_weight.to(torch.float16)))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x_norm_fp16, right_gate_weight.to(torch.float16)))  # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x_norm_fp16, out_gate_weight.to(torch.float16)))    # [B,N,N,H]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) TriMul contraction via Triton kernel
    # ------------------------------------------------------------------
    # Prepare output tensor (FP16, same layout as inputs)
    out_fp16 = torch.empty_like(left, dtype=torch.float16, device=device)

    # Strides (in elements) for each tensor
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_fp16.stride()

    # Tile sizes – tuned for H100 with FP16
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Launch grid: (batch*hidden, n_tiles_i, n_tiles_j)
    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    trimul_kernel[grid](
        left, right, out_fp16,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure kernel completion before moving on
    torch.cuda.synchronize()

    # Convert back to FP32 for the remaining LayerNorm and linear projection
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm and gating
    # ------------------------------------------------------------------
    out_mean = out.mean(dim=-1, keepdim=True)
    out_var  = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - out_mean) * torch.rsqrt(out_var + eps)
    out = out * to_out_norm_weight + to_out_norm_bias   # broadcast over B,N,N

    # Apply the out_gate (already FP16 -> convert)
    out = out * out_gate.to(torch.float32)

    # ------------------------------------------------------------------
    # 7) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # bias is None in the reference

    return out
```
