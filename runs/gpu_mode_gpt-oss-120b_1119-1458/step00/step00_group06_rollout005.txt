"""
TriMul (outgoing) forward implementation with a fused batched‐matmul Triton kernel.

The module computes (with a LayerNorm and linear projections):
    left  = (x · W_left ) * mask * sigmoid(x·W_left_gate )
    right = (x · W_right) * mask * sigmoid(x·W_right_gate)

and then evaluates the “outgoing” message passing operation
    out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]      (einsum “… i k d, … j k d → … i j d”)

The inner sum is a batched matrix multiplication for each hidden channel:
    out_h = left_h @ right_hᵀ   for h = 0 … hidden_dim‑1

We fuse this computation in a single Triton kernel that processes many
( BLOCK_M × BLOCK_N ) output tiles across all (batch·hidden_dim) groups.
All heavy arithmetic is performed in float32 accumulators while the
input / output tensors are stored in float16 to reduce bandwidth.
The remaining linear / LayerNorm steps are performed with native PyTorch
(rather than Triton) for simplicity.

The kernel works for any batch size (≤2), sequence length (≤1024) and
hidden dimensions used in the test suite.  The final tensor is
returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ
# A : (G, M, K)   – left side   (G = batch * hidden_dim)
# B : (G, N, K)   – right side (we multiply by its transpose)
# C : (G, M, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A, B, C,                     # pointers
    M, N, K,                     # problem size (int32)
    stride_am, stride_ak, stride_ah,   # A strides: (M,K,group)
    stride_bk, stride_bn, stride_bh,   # B strides: (K,N,group) – note K comes first for the dot
    stride_cm, stride_cn, stride_ch,   # C strides: (M,N,group)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matmul C = A @ Bᵀ  (float16 inputs, float32 accumulation)."""
    pid = tl.program_id(0)

    # number of tiles in each dimension
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_group = num_tiles_m * num_tiles_n

    # decode group id and tile coordinates
    group = pid // tiles_per_group
    tile_id = pid % tiles_per_group
    pid_m = tile_id // num_tiles_n
    pid_n = tile_id % num_tiles_n

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this group
    a_ptrs = A + group * stride_ah + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B + group * stride_bh + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # boundary masks for the current K‑tile
        cur_k = offs_k
        mask_k = cur_k < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)          # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)          # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)                 # fp32 accumulation

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result in fp16 (final cast)
    c = acc.to(tl.float16)
    c_ptrs = C + group * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (x, mask, weights, config)
        x       – [B, N, N, dim] tensor (float32)
        mask    – [B, N, N]   tensor (float32) or None
        weights – dict of model parameters
        config  – dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # LayerNorm (the first norm)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Apply mask if present (the reference multiplies the mask after the projection)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask = mask.unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gates (sigmoid of linear projections of the *normalized* input)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the Triton batched matmul:
    #   left  : (B, N, N, H) -> (G, M, K)
    #   right : (B, N, N, H) -> (G, N, K)
    #   G = B * H, M = N, K = N
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # bring hidden dimension in front and make the tensor contiguous
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    # merge batch and hidden dims
    G = B * hidden_dim
    left = left.view(G, N, N)          # (G, M, K)
    right = right.view(G, N, N)        # (G, N, K)

    # Cast to fp16 for bandwidth‑limited kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = G * num_tiles_m * num_tiles_n

    # strides (int64) – note the order expected by the kernel
    stride_am = left_fp16.stride(1)     # M stride
    stride_ak = left_fp16.stride(2)     # K stride
    stride_ah = left_fp16.stride(0)     # group stride

    stride_bk = right_fp16.stride(2)    # K stride (inner dim)
    stride_bn = right_fp16.stride(1)    # N stride (row)
    stride_bh = right_fp16.stride(0)    # group stride

    stride_cm = out_fp16.stride(1)      # M stride
    stride_cn = out_fp16.stride(2)      # N stride
    stride_ch = out_fp16.stride(0)      # group stride

    # launch the kernel
    _batched_matmul_kernel[
        total_programs,
        # num_warps = 4 gives good occupancy on H100 for fp16 matmul
        num_warps=4,
        num_stages=2,
    ](
        left_fp16,
        right_fp16,
        out_fp16,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_ah,
        stride_bk,
        stride_bn,
        stride_bh,
        stride_cm,
        stride_cn,
        stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Convert back to float32 and restore original layout [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Final LayerNorm, out‑gate, and output linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # shape (B, N, N, dim)

    return out
