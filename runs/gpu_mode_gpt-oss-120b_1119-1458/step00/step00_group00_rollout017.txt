def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    eps = 1e-5

    # layernorm input
    x = input_tensor
    # x shape [B,N,N,dim]
    # compute mean and var across last dim dim
    # Use torch.nn.functional.layer_norm:
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # linear projections
    # left = linear(x_norm, left_proj_weight)
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # mask (if provided)
    if not config.get("nomask", False):
        mask = mask_tensor.to(x_norm.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # gate computations
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # compute einsum using triton kernel
    # def triton_tri_mul(left, right): returns output shape [B,N,N,hidden_dim] dtype same as left
    out = triton_tri_mul(left, right)

    # to_out_norm
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    out_norm = out_norm * out_gate

    # final linear
    out = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return out

def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl
    
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    eps = 1e-5

    # LayerNorm on input
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # Linear projections (Project and gate)
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    
    # Apply mask if present
    if not config.get("nomask", False):
        # mask shape: [B, N, N]; unsqueeze for broadcast
        mask = mask_tensor.to(dtype=x_norm.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # Compute gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Triangular multiplication via Triton
    out = tri_mul_triton(left, right)   # define below

    # Output layernorm
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate
    
    # Final projection
    out_final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])
    return out_final

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    # Derive batch and hidden dimension index from pid_z
    b = pid_z // H
    d = pid_z % H

    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension in tiles
    for k in range(0, N, BLOCK_K):
        k_idx = k + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # left tile (i, k)
        left_ptrs = left_ptr + b * stride_left_b \
                    + i[:, None] * stride_left_i \
                    + k_idx[None, :] * stride_left_k \
                    + d * stride_left_h

               + i[:, None] * stride_out_i \
               + j[None, :] * stride_out_j \
               + d * stride_out_h

"""
TriMul (outgoing) – Triton‑accelerated implementation
====================================================

The module computes, for each batch `b` and hidden channel `d`:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` are linearly projected, masked and gated versions of the
input tensor `x` of shape `[B, N, N, C]`.  The heavy cubic “einsum” is performed
with a custom Triton kernel that carries out a blocked GEMM over the `k`
dimension for every `(i,j,d)` tile.  The rest of the TriMul forward pass
(layer‑norms, linear projections, sigmoid gates) is done with regular
PyTorch kernels.

The kernel uses three compile‑time block sizes (default 32×32×32) and a 3‑D
grid:
    pid_x → i‑tile, pid_y → j‑tile, pid_z → (batch, hidden‑dim) tile.
All out‑of‑bounds accesses are masked to zero, allowing any `N` (including
non‑multiples of the block size).  The final output has dtype ``torch.float32``.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# -------------------------------------------------------------------------
# Triton kernel: block‑wise cubic contraction (the einsum in the reference)
# -------------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    Shapes:
        left  : [B, N, N, H]
        right : [B, N, N, H]
        out   : [B, N, N, H]
    """
    pid_i = tl.program_id(0)          # i‐tile
    pid_j = tl.program_id(1)          # j‐tile
    pid_z = tl.program_id(2)          # combined (batch, hidden)

    # ----- decode batch and hidden index from pid_z -----
    b = pid_z // H                     # batch index
    d = pid_z %  H                     # hidden channel

    # ----- i and j coordinates of the tile -----
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_off < N
    mask_j = j_off < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- iterate over k in BLOCK_K chunks -----
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        k_base = kb * BLOCK_K
        k_off = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # left tile : shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr \
            + b * stride_left_b \
            + i_off[:, None] * stride_left_i \
            + k_off[None, :] * stride_left_k \
            + d * stride_left_h
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile : shape (BLOCK_K, BLOCK_N)   (k first, j second)
        right_ptrs = right_ptr \
            + b * stride_right_b \
            + k_off[:, None] * stride_right_k \
            + j_off[None, :] * stride_right_j \
            + d * stride_right_h
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # accumulate the block GEMM
        acc += tl.dot(left, right)

    # ----- write back the result -----
    out_ptrs = out_ptr \
        + b * stride_out_b \
        + i_off[:, None] * stride_out_i \
        + j_off[None, :] * stride_out_j \
        + d * stride_out_h
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------------------
# Helper that launches the kernel
# -------------------------------------------------------------------------
def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton‑based implementation of the cubic contraction used in TriMul.
    Input / output shape: [B, N, N, H] (H = hidden_dim).
    """
    B, N, N2, H = left.shape
    assert N == N2, "left/right must be square on the sequence dimensions"

    out = torch.empty_like(left)          # same dtype & device

    # ----- strides (in elements, not bytes) -----
    sL_b, sL_i, sL_k, sL_h = left.stride()
    sR_b, sR_i, sR_k, sR_h = right.stride()
    sO_b, sO_i, sO_k, sO_h = out.stride()

    # ----- block sizes (tunable) -----
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    # ----- grid dimensions -----
    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid_z = B * H                         # batch * hidden channel

    tri_mul_kernel[(grid_i, grid_j, grid_z)](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=sL_b,
        stride_left_i=sL_i,
        stride_left_k=sL_k,
        stride_left_h=sL_h,
        stride_right_b=sR_b,
        stride_right_j=sR_i,               # j‑dimension of right = its 2nd dim stride
        stride_right_k=sR_k,
        stride_right_h=sR_h,
        stride_out_b=sO_b,
        stride_out_i=sO_i,
        stride_out_j=sO_k,                 # j‑dimension of out = its 3rd dim stride
        stride_out_h=sO_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# -------------------------------------------------------------------------
# Entry point expected by the grader
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (dtype float32).
    """
    # -----------------------------------------------------------------
    # unpack
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. Input LayerNorm   (over the last dimension = dim)
    # -----------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])      # -> [B,N,N,hidden_dim]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional masking (mask is [B,N,N]; broadcast over hidden_dim)
    # -----------------------------------------------------------------
    if not config.get("nomask", False):
        # mask may be bool or numeric; cast to same dtype as tensors
        mask = mask_tensor.to(dtype=x_norm.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid) – also linear, no bias
    # -----------------------------------------------------------------
    left_gate = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Cubic contraction (the heavy part) using Triton
    # -----------------------------------------------------------------
    out = tri_mul_triton(left, right)          # shape [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # 6. Output LayerNorm (over hidden_dim)
    # -----------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate

    # -----------------------------------------------------------------
    # 7. Final linear projection back to original dim
    # -----------------------------------------------------------------
    out_final = F.linear(out_norm, weights["to_out.weight"])   # -> [B,N,N,dim]

    return out_final
