def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config['hidden_dim']

    # Layer norm on input
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    # shape: (bs, N, N, hidden_dim)
    left = torch.nn.functional.linear(x, left_proj_weight)   # weight shape (hidden_dim, dim)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if present else use mask = 1
    if not config.get('nomask', True):
        # mask shape (bs,N,N) broadcast to hidden_dim
        mask = mask.unsqueeze(-1)  # (bs,N,N,1)
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # compute out = left @ right^T across N dimension (inner product over axis 2)
    # reshape to (bs*hidden_dim, N, N)
    left_reshaped = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_reshaped = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)

    # cast to fp16
    left_fp16 = left_reshaped.to(torch.float16)
    right_fp16 = right_reshaped.to(torch.float16)

    out_fp16 = torch.empty_like(left_fp16)  # shape (bs*hidden_dim, N, N)
    # launch triton kernel
    triton_matmul_kernel(left_fp16, right_fp16, out_fp16,
                         M=bs * hidden_dim,
                         N=N,
                         K=N,
                         stride_am=left_fp16.stride(0),
                         stride_ak=left_fp16.stride(1),
                         stride_bk=right_fp16.stride(1),
                         stride_bn=right_fp16.stride(0),
                         stride_cm=out_fp16.stride(0),
                         stride_cn=out_fp16.stride(1),
                         BLOCK_M=64, BLOCK_N=64, BLOCK_K=64)

    out = out_fp16.to(torch.float32)

    # reshape back to (bs, N, N, hidden_dim)
    out = out.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Apply to_out_norm and out_gate
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate

    # Final linear to dim
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

@triton.jit
def triton_trimul_kernel(A, B, C,
                         M, N, K,
                         stride_Am, stride_Ai, stride_Ak,
                         stride_Bm, stride_Bi, stride_Bk,
                         stride_Cm, stride_Ci, stride_Cj,
                         BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_m = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # row, col offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute base ptr for this m
    a_base = A + pid_m * stride_Am
    b_base = B + pid_m * stride_Bm
    c_base = C + pid_m * stride_Cm

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A block (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + (offs_i[:, None] * stride_Ai + offs_k[None, :] * stride_Ak)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load B block transposed (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + (offs_k[:, None] * stride_Bk + offs_j[None, :] * stride_Bi)

        acc += tl.dot(a, b, out_dtype=tl.float32)

    c_ptrs = c_base + (offs_i[:, None] * stride_Ci + offs_j[None, :] * stride_Cj)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
Custom Triton‐accelerated outgoing TriMul (Triangle Multiplicative Update).

The heavy pairwise multiplication
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batch of independent matrix multiplications
(left_d @ right_dᵀ) for every hidden channel `d` and batch element.
A single Triton kernel processes all (batch, hidden) channels in parallel
using a 3‑D grid (channel, row‑block, col‑block).  All surrounding
LayerNorms, Linear projections and gates are performed with PyTorch
(easily implemented in FP32).  The kernel works with fp16 inputs and
accumulates in fp32 for numerical stability, finally writing fp16
results that are cast back to fp32 for the remaining PyTorch pipeline.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_kernel(
    A, B, C,                     # pointers
    M, N, K,                     # matrix dimensions
    stride_Am, stride_Ai, stride_Ak,   # strides for A (M, i, k)
    stride_Bm, stride_Bi, stride_Bk,   # strides for B (M, j, k)
    stride_Cm, stride_Ci, stride_Cj,   # strides for C (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched Gram‑matrix multiplication (M independent N×N products)."""
    pid_m = tl.program_id(0)            # which (batch,hidden) pair
    pid_i = tl.program_id(1)            # row‑block index
    pid_j = tl.program_id(2)            # col‑block index

    # ------------------------------------------------------------------
    # Offsets within the N×N matrix for this block
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Base pointers for the selected (batch,hidden) slice
    # ------------------------------------------------------------------
    a_base = A + pid_m * stride_Am
    b_base = B + pid_m * stride_Bm
    c_base = C + pid_m * stride_Cm

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction axis k
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block   (BLOCK_M, BLOCK_K)   -> A[m, i, k]
        a_ptrs = a_base + (offs_i[:, None] * stride_Ai
                           + offs_k[None, :] * stride_Ak)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B block transposed (BLOCK_K, BLOCK_N) -> B[m, k, j]
        # Note: original B layout is (M, j, k); we need (k, j) for dot.
        b_ptrs = b_base + (offs_k[:, None] * stride_Bk
                           + offs_j[None, :] * stride_Bi)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Accumulate: a (M×K) @ b (K×N) -> (M×N)
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    # Write result back as FP16
    # ------------------------------------------------------------------
    c_ptrs = c_base + (offs_i[:, None] * stride_Ci
                       + offs_j[None, :] * stride_Cj)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel
    for the O(N³) pairwise multiplication.

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be ignored)
            - weights: dict of weight tensors
            - config: dict with keys "dim", "hidden_dim", "nomask" ...

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Model parameters (all are 2‑D weight matrices / 1‑D bias vectors)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]

    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Initial LayerNorm over the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional masking (mask shape [B,N,N])
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_expanded = mask.unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_expanded
        right = right * mask_expanded

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Batched Gram‑matrix multiplication via Triton
    #    left[b, i, k, d]  @  right[b, j, k, d]^T   -->  out[b, i, j, d]
    # ------------------------------------------------------------------
    B, N_seq, _, _ = left.shape
    M = B * hidden_dim          # total number of independent matrices
    N = N_seq
    K = N_seq

    # Rearrange tensors to (M, N, K) where M = B*hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_perm.view(M, N, K).to(torch.float16)
    right_mat = right_perm.view(M, N, K).to(torch.float16)

    # Output placeholder
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # Strides for the kernel
    stride_Am, stride_Ai, stride_Ak = left_mat.stride()
    stride_Bm, stride_Bi, stride_Bk = right_mat.stride()
    stride_Cm, stride_Ci, stride_Cj = out_mat.stride()

    # Tunable block sizes – work well for N up to 1024 on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        M,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _trimul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M, N, K,
        stride_Am, stride_Ai, stride_Ak,
        stride_Bm, stride_Bi, stride_Bk,
        stride_Cm, stride_Ci, stride_Cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to FP32 and restore original shape [B, N, N, hidden_dim]
    out = out_mat.to(torch.float32)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # maps hidden_dim -> dim

    return out
