import torch
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    A, B, C,
    stride_ab, stride_ai, stride_ak,
    stride_bb, stride_bk, stride_bj,
    stride_cb, stride_ci, stride_cj,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    pid_n = tl.program_id(1)
    # compute batch index and block row index
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    batch = pid // num_blocks_m
    pid_m = pid % num_blocks_m

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers
    a_ptrs = A + batch * stride_ab + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
    b_ptrs = B + batch * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bj

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        # Convert to float32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)
        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write out
    c_ptrs = C + batch * stride_cb + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul (outgoing) forward implementation with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B, N, N, D]).
2. Compute left/right projections (Linear w/out bias) to a hidden dimension H.
3. Apply the optional pair‑wise mask (broadcasted on H).
4. Compute three gating vectors (sigmoid(Linear(x))) and apply them:
   left  = left  * left_gate
   right = right * right_gate
   out_gate = sigmoid(Linear(x))  # used later
5. The expensive part is the Einstein sum:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is exactly a batched matrix multiplication A @ Bᵀ for every hidden
   channel h.  We fuse all hidden channels and the batch dimension into a
   single “batch” dimension (size = B*H) and launch a Triton kernel that
   computes C = A @ Bᵀ for each fused batch.
   * A, B are FP16 (to reduce bandwidth) and the accumulation is done in
     FP32.
   * The kernel tiles the M, N, K dimensions (default 64×64×64) and uses
     `tl.dot` which maps to tensor‑core instructions on H100.
6. Reshape the kernel result back to [B, N, N, H] (float‑32).
7. Apply a second LayerNorm on the hidden dimension, multiply by `out_gate`,
   and project back to the original dimension D with a final Linear layer.
8. Return the resulting tensor (dtype float32, shape [B, N, N, D]).

Only the O(N³) summation is performed on GPU via Triton; everything else
leverages fast PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    stride_ab, stride_ai, stride_ak,    # A strides: batch, row(i), col(k)
    stride_bb, stride_bk, stride_bj,    # B strides (treated as Bᵀ): batch, row(k), col(j)
    stride_cb, stride_ci, stride_cj,    # C strides: batch, row(i), col(j)
    M, N, K,                           # matrix sizes (all = seq_len)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM C = A @ Bᵀ   (A,B : half, C : float32)

    The batch dimension is fused with the hidden channel dimension.
    """
    pid = tl.program_id(0)
    pid_n = tl.program_id(1)

    # number of blocks that cover the M dimension
    num_blocks_m = tl.cdiv(M, BLOCK_M)

    # decode batch index and local block‑row index
    batch = pid // num_blocks_m
    pid_m = pid % num_blocks_m

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # batch offsets (in elements)
    a_batch_off = batch * stride_ab
    b_batch_off = batch * stride_bb
    c_batch_off = batch * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptr = A_ptr + a_batch_off \
                + offs_m[:, None] * stride_ai \
                + cur_k[None, :] * stride_ak
        a = tl.load(a_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Bᵀ tile: B is stored as (batch, K, N), we load (BLOCK_K, BLOCK_N)
        b_ptr = B_ptr + b_batch_off \
                + cur_k[:, None] * stride_bk \
                + offs_n[None, :] * stride_bj
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # inner product over K
        acc += tl.dot(a, b)

    # write back
    c_ptr = C_ptr + c_batch_off \
            + offs_m[:, None] * stride_ci \
            + offs_n[None, :] * stride_cj
    tl.store(c_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for each batch entry.
    A, B : [B*H, N, N] (float16)
    Returns C : [B*H, N, N] (float32)
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, N, K = A.shape
    assert B.shape == (batch, N, K)   # same shape before transpose
    C = torch.empty((batch, N, N), dtype=torch.float32, device=A.device)

    # strides in elements
    stride_ab, stride_ai, stride_ak = A.stride()        # (batch, i, k)
    stride_bb, stride_bi, stride_bk = B.stride()        # (batch, i, k)

    # We treat B as transposed: Bᵀ has strides (batch, k, j)
    stride_bk_T = stride_bi        # stride for the "K" dimension after transpose
    stride_bj_T = stride_bk        # stride for the "N" (j) dimension after transpose

    stride_cb, stride_ci, stride_cj = C.stride()

    # Tune tile sizes (empirically good for H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (batch * ((N + BLOCK_M - 1) // BLOCK_M)),  # combined batch*row blocks
        (N + BLOCK_N - 1) // BLOCK_N,              # column blocks
    )

    _batched_matmul_kernel[grid](
        A, B, C,
        stride_ab, stride_ai, stride_ak,
        stride_bb, stride_bk_T, stride_bj_T,
        stride_cb, stride_ci, stride_cj,
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # enough to keep tensor cores busy
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns: output tensor of shape [B, N, N, D] (float32)
    """
    # unpack
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask shape: [B, N, N] -> broadcast to [B, N, N, 1]
        mask = mask_tensor.to(dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy O(N³) multiplication via Triton
    # ------------------------------------------------------------------
    # Cast to half for the kernel (memory‑bandwidth friendly)
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # Fuse batch and hidden dimensions: shape -> (B*H, N, N)
    B, N, _, _ = left_f16.shape  # input shape: B, N, N, H
    left_reshaped = left_f16.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    right_reshaped = right_f16.permute(0, 3, 1, 2).contiguous().view(-1, N, N)

    # Batched matmul: C = left @ rightᵀ   (float32 output)
    out_reshaped = _batched_matmul(left_reshaped, right_reshaped)  # [B*H, N, N]

    # Restore original layout: [B, N, N, H] (float32)
    out = out_reshaped.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Post‑processing (LayerNorm, out_gate, final projection)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
