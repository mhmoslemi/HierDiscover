@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl: tl.constexpr, stride_bi: tl.constexpr, stride_bk: tl.constexpr, stride_bh: tl.constexpr,
    stride_rl: tl.constexpr, stride_ri: tl.constexpr, stride_rk: tl.constexpr, stride_rh: tl.constexpr,
    stride_ol: tl.constexpr, stride_oi: tl.constexpr, stride_oj: tl.constexpr, stride_oh: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program_id: encode group and tile indices.
    # total_tiles_per_group = ceil(N/BLOCK_M)*ceil(N/BLOCK_N)
    total_tiles_M = tl.cdiv(N, BLOCK_M)
    total_tiles_N = tl.cdiv(N, BLOCK_N)
    tiles_per_group = total_tiles_M * total_tiles_N
    pid = tl.program_id(0)
    group = pid // tiles_per_group
    tile_id = pid % tiles_per_group
    pid_m = tile_id // total_tiles_N
    pid_n = tile_id % total_tiles_N

    # Compute b and h from group id
    b = group // H
    h = group % H

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this group
    left_base = left_ptr + b * stride_bl + h * stride_bh
    right_base = right_ptr + b * stride_rl + h * stride_rh
    out_base = out_ptr + b * stride_ol + h * stride_oh

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A: left[i,k] shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + (offs_m[:, None] * stride_bi) + (offs_k[None, :] * stride_bk)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # B: right[j,k] transposed to (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + (offs_k[:, None] * stride_rk) + (offs_n[None, :] * stride_ri)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # store result
    out_ptrs = out_base + (offs_m[:, None] * stride_oi) + (offs_n[None, :] * stride_oj)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_bi, stride_bk, stride_bh,
    stride_rl, stride_ri, stride_rk, stride_rh,
    stride_ol, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # compute tile indices

    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_group = num_tiles_m * num_tiles_n
    pid = tl.program_id(0)
    group = pid // tiles_per_group
    tile_id = pid % tiles_per_group
    pid_m = tile_id // num_tiles_n
    pid_n = tile_id % num_tiles_n

    # decode group coords
    b = group // H
    h = group % H

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers
    left_base = left_ptr + b * stride_bl + h * stride_bh
    right_base = right_ptr + b * stride_rl + h * stride_rh
    out_base = out_ptr + b * stride_ol + h * stride_oh

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A tile
        a_ptrs = left_base + (offs_m[:, None] * stride_bi) + (offs_k[None, :] * stride_bk)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load B tile (transposed)
        b_ptrs = right_base + (offs_k[:, None] * stride_rk) + (offs_n[None, :] * stride_ri)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Convert to float32 if needed
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # store
    out_ptrs = out_base + (offs_m[:, None] * stride_oi) + (offs_n[None, :] * stride_oj)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) – fast Triton implementation.

The module takes a 4‑D tensor X ∈ ℝ^{B×N×N×C} and performs:

1. LayerNorm over the last dimension (C).
2. Two linear projections (left/right) to a hidden dimension H.
3. Optional pair‑wise mask.
4. Three sigmoid gates (left, right, out) applied element‑wise.
5. Core quadratic update:
      out_{b,i,j,:} = Σ_k left_{b,i,k,:} · right_{b,j,k,:}
   which is a batch of H independent GEMMs of size N×N (left × rightᵀ).
   This step is executed entirely in a custom Triton kernel.
6. LayerNorm over H, out‑gate, and final linear projection back to C.

Only the forward pass is required; gradients are not computed.
The Triton kernel fuses the N×N×H matrix multiplications and
avoids the O(N³) Python loops of the reference implementation.
"""

import math
import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel for the batched GEMM:  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,            # const *float
    right_ptr,           # const *float
    out_ptr,             # *float
    B, N, H,             # compile‑time constants
    stride_bl, stride_bi, stride_bk, stride_bh,   # left strides
    stride_rl, stride_ri, stride_rk, stride_rh,   # right strides
    stride_ol, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    One program instance computes a BLOCK_M×BLOCK_N tile of the
    output for a single (batch, hidden) pair.
    """
    # ------------------------------------------------------------------
    # 1) map program id -> (batch, hidden, tile_i, tile_j)
    # ------------------------------------------------------------------
    tiles_m = tl.cdiv(N, BLOCK_M)          # number of tiles along the i‑axis
    tiles_n = tl.cdiv(N, BLOCK_N)          # number of tiles along the j‑axis
    tiles_per_group = tiles_m * tiles_n     # tiles for one (b,h)

    pid = tl.program_id(0)
    group = pid // tiles_per_group          # which (batch, hidden) pair
    tile_id = pid % tiles_per_group

    pid_m = tile_id // tiles_n              # tile index along i
    pid_n = tile_id % tiles_n               # tile index along j

    # decode batch and hidden dimension from group
    b = group // H
    h = group % H

    # ------------------------------------------------------------------
    # 2) compute offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for the selected (b,h)
    left_base  = left_ptr  + b * stride_bl + h * stride_bh
    right_base = right_ptr + b * stride_rl + h * stride_rh
    out_base   = out_ptr   + b * stride_ol + h * stride_oh

    # accumulator in FP32 (even if inputs are FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3) loop over the reduction dimension k
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- left tile (M × K) -------------------------------------------------
        a_ptrs = left_base + (offs_m[:, None] * stride_bi) + (offs_k[None, :] * stride_bk)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        a = a.to(tl.float32)

        # ---- right tile (K × N) – note the transpose (k as rows, j as cols) ---
        b_ptrs = right_base + (offs_k[:, None] * stride_rk) + (offs_n[None, :] * stride_ri)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)
        b = b.to(tl.float32)

        # ---- matrix multiplication ------------------------------------------------
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) × (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # 4) write the tile back to global memory
    # ------------------------------------------------------------------
    out_ptrs = out_base + (offs_m[:, None] * stride_oi) + (offs_n[None, :] * stride_oj)
    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that wraps the Triton kernel – returns a tensor of shape [B,N,N,H]
# ----------------------------------------------------------------------
def triton_tri_mul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for TriMul:
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    Arguments:
        left, right – tensors of shape (B, N, N, H), contiguous, same dtype.
    Returns:
        out – tensor of shape (B, N, N, H), same dtype (float32 is used internally).
    """
    assert left.shape == right.shape
    B, N, N2, H = left.shape
    assert N == N2

    # Ensure contiguous memory layout (required for striding assumptions)
    left = left.contiguous()
    right = right.contiguous()

    # Output tensor
    out = torch.empty_like(left)

    # Extract strides (in elements, not bytes)
    stride_bl, stride_bi, stride_bk, stride_bh = left.stride()
    stride_rl, stride_ri, stride_rk, stride_rh = right.stride()
    stride_ol, stride_oi, stride_oj, stride_oh = out.stride()

    # ------------------------------------------------------------------
    # Choose tile sizes – tuned for H100 FP32 performance
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Number of tiles per (b,h) pair
    tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    total_tiles = tiles_m * tiles_n

    # One program instance per tile per (batch, hidden) pair
    grid = (B * H * total_tiles,)

    # Launch Triton kernel
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_bl,
        stride_bi,
        stride_bk,
        stride_bh,
        stride_rl,
        stride_ri,
        stride_rk,
        stride_rh,
        stride_ol,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # H100 benefits from 8 warps per block
        num_stages=3,         # sufficient for double‑buffering
    )
    return out


# ----------------------------------------------------------------------
# Entry point – matches the repository’s expected signature
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (the same shape as the input).
    """
    input_tensor, mask_tensor, weights, config = data

    # ------------------------------------------------------------------
    # Configuration
    # ------------------------------------------------------------------
    dim = config["dim"]               # C  (input / output channel size)
    hidden_dim = config["hidden_dim"] # H  (intermediate channel size)
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # Load model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    # shape: [B, N, N, dim]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    # Result shape: [B, N, N, hidden_dim]
    left = F.linear(x, left_proj_weight)
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → unsqueeze → [B, N, N, 1]
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Compute gates (sigmoid after a linear)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gating to the projected tensors
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core quadratic update – Triton fused GEMM
    # ------------------------------------------------------------------
    # out shape = [B, N, N, hidden_dim]
    out = triton_tri_mul(left, right)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    out = out * out_gate                # element‑wise gating (same shape)
    out = F.linear(out, to_out_weight)  # back to [B, N, N, dim]

    return out
