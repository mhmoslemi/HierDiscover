@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # i tile
    pid_n = tl.program_id(1)  # j tile
    pid_bh = tl.program_id(2) # batch * hidden

    # decode batch and head
    b = pid_bh // H
    h = pid_bh % H

    # offsets for this tile
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N
    k_start = 0  # will be updated inside loop

    # range indices
    i = i_start + tl.arange(0, BLOCK_M)
    j = j_start + tl.arange(0, BLOCK_N)

    # allocate accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        k = k_start + tl.arange(0, BLOCK_K)

        #=== load left block (M x K) ===
        # pointer base for left: per batch, per hidden
        left_base = left_ptr + b*stride_left_b + h*stride_left_h
        # compute offsets

        mask_k = k < N

                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0,
                       dtype=tl.float16)

        #=== load right block (K x N) ===

        right_base = right_ptr + b*stride_right_b + h*stride_right_h

                        other=0.0,

def custom_kernel(data):
    """
    Triton-accelerated implementation of the outgoing TriMul block.
    It follows the reference PyTorch code but off‑loads the large
    tensor contraction
        out[i,j] = Σ_k left[i,k] * right[j,k]
    to a fused triton kernel.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: (B,N,N) -> (B,N,N,1)
        mask = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Main contraction via Triton
    # ------------------------------------------------------------------
    # cast to fp16 for the kernel to reduce traffic
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out = trimul_einsum(left_fp16, right_fp16,
                        BLOCK_M=128, BLOCK_N=128, BLOCK_K=32)

    # ------------------------------------------------------------------
    # 6) Output layer‑norm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out

    right = right * right_gate

    # large contraction:
    #   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # implemented as a fused batched GEMM per hidden channel using Triton.

    out = LayerNorm(out)             # hidden dim H
    out = out * out_gate
    out = Linear(out)                # back to original dim

All linear layers are bias‑free, LayerNorm uses ε=1e‑5.
The heavy O(N³·H) contraction is off‑loaded to a custom Triton kernel
that treats every (batch, hidden) pair as an independent GEMM:
    (N×N) × (N×N)ᵀ → (N×N)
The kernel works in fp16 for the two inputs, accumulates in fp32,
and writes fp32 results.  It handles arbitrary batch size (≤2),
sequence length (≤1024) and hidden dimension (≤384).  The mask
is optional (controlled by the ``nomask`` flag in the config).

The public entry point is ``custom_kernel(data)`` where
``data = (input_tensor, mask, weights, config)`` as described
by the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel that computes the TriMul contraction:
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides for left  (b,i,k,h)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right (b,j,k,h) – we treat it as (k,j) in the kernel
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    # strides for out   (b,i,j,h)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)        # tile over i
    pid_n = tl.program_id(1)        # tile over j
    pid_bh = tl.program_id(2)       # combined batch * hidden index

    # decode batch (b) and hidden channel (h)
    b = pid_bh // H
    h = pid_bh % H

    # start indices of the current tile
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # index vectors for the tile (will be masked at the boundaries)
    i = i_start + tl.arange(0, BLOCK_M)
    j = j_start + tl.arange(0, BLOCK_N)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k in blocks of size BLOCK_K
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        k = k_start + tl.arange(0, BLOCK_K)

        # ----------------------- load left[i,k] -----------------------
        # Pointer base for this (b, h)
        left_base = left_ptr + b * stride_left_b + h * stride_left_h
        offs_i = i * stride_left_i                # (M,)
        offs_k = k * stride_left_k                # (K,)
        left_ptrs = left_base + offs_i[:, None] + offs_k[None, :]  # (M,K)

        mask_i = i < N
        mask_k = k < N
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # ----------------------- load right[j,k] (as K×N) -----------------------
        # We treat right as a (K,N) matrix: rows = k, cols = j
        right_base = right_ptr + b * stride_right_b + h * stride_right_h
        offs_k_r = k * stride_right_k                # (K,)
        offs_j = j * stride_right_j                  # (N,)
        right_ptrs = right_base + offs_k_r[:, None] + offs_j[None, :]  # (K,N)

        mask_j = j < N
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # ----------------------- dot product & accumulate -----------------------
        acc += tl.dot(left, right)   # (M,N) ← (M,K)·(K,N)

    # ----------------------- write output -----------------------
    out_base = out_ptr + b * stride_out_b + h * stride_out_h
    offs_i_out = i * stride_out_i
    offs_j_out = j * stride_out_j
    out_ptrs = out_base + offs_i_out[:, None] + offs_j_out[None, :]

    mask_i = i < N
    mask_j = j < N
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _trimul_einsum(left_fp16: torch.Tensor, right_fp16: torch.Tensor,
                   BLOCK_M: int = 128, BLOCK_N: int = 128, BLOCK_K: int = 32) -> torch.Tensor:
    """
    Wrapper that launches the Triton kernel.
    left_fp16, right_fp16: [B, N, N, H] fp16 tensors.
    Returns: out fp32 tensor of shape [B, N, N, H].
    """
    assert left_fp16.is_cuda and right_fp16.is_cuda
    B, N, _, H = left_fp16.shape
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left_fp16.device)

    # Pull raw pointers and strides
    left_ptr = left_fp16.data_ptr()
    right_ptr = right_fp16.data_ptr()
    out_ptr = out.data_ptr()
    s_l = left_fp16.stride()
    s_r = right_fp16.stride()
    s_o = out.stride()

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,                     # one program per (batch, hidden)
    )

    _trimul_einsum_kernel[grid](
        left_ptr,
        right_ptr,
        out_ptr,
        B,
        N,
        H,
        s_l[0],
        s_l[1],
        s_l[2],
        s_l[3],
        s_r[0],
        s_r[1],
        s_r[2],
        s_r[3],
        s_o[0],
        s_o[1],
        s_o[2],
        s_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul block.

    Arguments:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] float32 (on CUDA)
            - mask:        [B, N, N]   int/float (on CUDA)
            - weights: dict of torch tensors with the model parameters
            - config: dict containing at least:
                  "dim" (int), "hidden_dim" (int), and optional "nomask" (bool)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (mask shape: [B, N, N] -> [B, N, N, 1])
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)    # broadcastable
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of bias‑free linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Main contraction via Triton (fp16 for speed, fp32 accumulation)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Choose block sizes that fit H100 well; they can be tuned further.
    out = _trimul_einsum(left_fp16, right_fp16,
                         BLOCK_M=128, BLOCK_N=128, BLOCK_K=32)   # [B, N, N, hidden_dim] fp32

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating and final projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )  # [B, N, N, hidden_dim]

    out = out * out_gate                                 # gating
    out = F.linear(out, weights["to_out.weight"])        # [B, N, N, dim]
    return out
