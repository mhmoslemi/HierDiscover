def custom_kernel(data):
    input, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    # optional: if mask is None and config.get('nomask', False): set mask = torch.ones...
    if mask is None or (config.get('nomask', False) and mask.numel() == 0):
        mask = torch.ones(input.shape[:3], dtype=input.dtype, device=input.device)
    # Cast to float16 for kernel efficiency:
    x = input

    # LayerNorm: using weights:
    # norm_weight shape [dim]; norm_bias shape [dim]
    # Compute mean and var per last dim:
    # Use torch.nn.functional.layer_norm
    x = torch.nn.functional.layer_norm(x, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])
    # left projection:
    left = torch.nn.functional.linear(x, weights['left_proj.weight'])
    right = torch.nn.functional.linear(x, weights['right_proj.weight'])
    # mask (broadcast):
    mask_unsq = mask.unsqueeze(-1)  # shape [B,N,N,1]
    left = left * mask_unsq
    right = right * mask_unsq

    # Gates:
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights['left_gate.weight']))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights['right_gate.weight']))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights['out_gate.weight']))

    left = left * left_gate
    right = right * right_gate

    # Cast left/right to fp16 for performance
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # kernel execution:
    out_hidden = trmul_kernel(left_fp16, right_fp16,
                              B=input.shape[0], N=input.shape[1], H=hidden_dim,
                              ... strides ...)

    # Convert back to float32
    out_hidden = out_hidden.to(torch.float32)

    # to_out_norm:
    out_hidden = torch.nn.functional.layer_norm(out_hidden, (hidden_dim,), weight=weights['to_out_norm.weight'], bias=weights['to_out_norm.bias'])
    # gating:
    out_hidden = out_hidden * out_gate

    # final linear projection:
    out = torch.nn.functional.linear(out_hidden, weights['to_out.weight'])
    return out

    left_ptr, right_ptr, out_ptr,
    B, N, H,

    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    chan_idx = pid_bh % H

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

"""
TriMul (outgoing) implementation with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input over the last dimension (dim).
2. Linear projections `left_proj`, `right_proj` (no bias) → shape [B,N,N,H].
3. Optional mask multiplication (broadcasted on the hidden dimension).
4. Compute gating values with sigmoid‑activated linear layers and apply them to
   `left` and `right`.
5. Perform the expensive tensor contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batch of N×N matrix multiplications for each hidden channel d.
   This is implemented in Triton as a batched matmul where each (batch, d)
   pair is a separate program‑id.  Inputs are cast to fp16 for bandwidth,
   accumulation happens in fp32, and the result is stored as fp16.
6. Layer‑norm over the hidden dimension, multiply by the `out_gate`,
   and a final linear projection back to the original dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication per hidden channel.

    Computes, for each (b, d):
        out[b, :, :, d] = left[b, :, :, d] @ right[b, :, :, d].T
    where left/right are fp16 tensors of shape [B,N,N,H] and output
    is stored in fp16 (accumulated in fp32).
    """
    pid_bh = tl.program_id(0)      # combined batch*hidden index
    pid_m  = tl.program_id(1)      # block row index (i)
    pid_n  = tl.program_id(2)      # block col index (j)

    # Decompose combined index
    batch_idx = pid_bh // H
    chan_idx  = pid_bh % H

    # Offsets in the M (i) and N (j) dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Main K‑loop
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Pointers to the current tile of left and right
        left_ptrs = (
            left_ptr
            + batch_idx * stride_left_b
            + chan_idx   * stride_left_h
            + offs_m[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        right_ptrs = (
            right_ptr
            + batch_idx * stride_right_b
            + chan_idx   * stride_right_h
            + offs_n[:, None] * stride_right_j
            + offs_k[None, :] * stride_right_k
        )

        # Load with bounds checking
        mask_m = offs_m < N
        mask_n = offs_n < N
        mask_k = offs_k < N
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        right = tl.load(right_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # Convert to fp32 for accumulation
        left_f32 = left.to(tl.float32)
        right_f32 = right.to(tl.float32)

        acc += tl.dot(left_f32, right_f32)

    # Write the result back (store as fp16)
    out_ptrs = (
        out_ptr
        + batch_idx * stride_out_b
        + chan_idx   * stride_out_h
        + offs_m[:, None] * stride_out_i
        + offs_n[None, :] * stride_out_j
    )
    mask_m = offs_m < N
    mask_n = offs_n < N
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _batched_trmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper for the Triton kernel.
    `left`/`right` are fp16 tensors of shape [B,N,N,H].
    Returns fp16 tensor of the same shape.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    # Ensure contiguous memory layout
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left, dtype=torch.float16, device=left.device)

    # Strides in element units (not bytes)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_rj, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Tunable block sizes (empirically good for H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                              # combined batch * hidden
        (N + BLOCK_M - 1) // BLOCK_M,       # M blocks
        (N + BLOCK_N - 1) // BLOCK_N,       # N blocks
    )

    _trmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_lb,
        s_li,
        s_lk,
        s_lh,
        s_rb,
        s_rj,
        s_rk,
        s_rh,
        s_ob,
        s_oi,
        s_oj,
        s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # reasonable default for fp16 on H100
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor, shape [B, N, N] (may be None)
        - weights      : dict of torch.Tensors containing layer‑norm and linear weights
        - config       : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # If mask is omitted or the test specifies “nomask”, treat as all‑ones
    if mask is None or nomask:
        mask = torch.ones(
            (inp.shape[0], inp.shape[1], inp.shape[2]),
            dtype=inp.dtype,
            device=inp.device,
        )
    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    # weights["norm.weight"] shape [dim], bias shape [dim]
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and masking
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask (broadcast on hidden dimension)
    mask_unsq = mask.unsqueeze(-1)  # [B,N,N,1]
    left = left * mask_unsq
    right = right * mask_unsq

    # ------------------------------------------------------------------
    # 3. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Core contraction via Triton (fp16 for bandwidth)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_hidden_fp16 = _batched_trmul(left_fp16, right_fp16)  # [B,N,N,hidden_dim] fp16
    out_hidden = out_hidden_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 5. Post‑norm, out‑gate, final projection
    # ------------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate

    # Final linear back to dim
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
