def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    B, N, _, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    dim = config["dim"]
    eps = 1e-5
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Use torch.layer_norm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (C,), weight=norm_weight, bias=norm_bias, eps=eps)
    # Projections
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"]) # shape B,N,N,hidden_dim
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # Apply mask if provided (mask maybe None? config indicates "nomask"?)
    if mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1).to(input_tensor.dtype)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # Apply gating
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Compute out = matmul left @ right^T per hidden_dim
    # use Triton kernel for batched matmul
    out = tri_mul_einsum(left, right)  # returns shape B,N,N,hidden_dim

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_normed = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # gating after norm
    out_gated = out_normed * out_gate

    # final linear
    out_final = torch.nn.functional.linear(out_gated, weights["to_out.weight"])
    return out_final

```
def tri_mul_kernel(L_ptr, R_ptr, O_ptr,
                   B, N, H,
                   stride_l_b, stride_l_i, stride_l_k, stride_l_h,
                   stride_r_b, stride_r_i, stride_r_k, stride_r_h,
                   stride_o_b, stride_o_i, stride_o_j, stride_o_h,
                   BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr):
    # Program IDs
    pid_head = tl.program_id(0) # head = batch * H + h
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute batch and hidden index
    batch = pid_head // H
    hid = pid_head % H

    # compute start offsets
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    # compute offsets base
    # base pointer for left matrix for given batch and head
    L_ptrs_base = L_ptr + batch * stride_l_b + hid * stride_l_h
    R_ptrs_base = R_ptr + batch * stride_r_b + hid * stride_r_h
    O_ptr_base = O_ptr + batch * stride_o_b + hid * stride_o_h

    # offsets inside i,j dimensions
    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)

    # mask for out-of-bound i,j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # iterate over k dimension tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointer offsets for left: shape (BLOCK_I, BLOCK_K)
        # left[i,k] index offset = offs_i * stride_l_i + offs_k * stride_l_k
        L_ptrs = L_ptrs_base + (offs_i[:, None] * stride_l_i) + (offs_k[None, :] * stride_l_k)
        # load left_tile
        L = tl.load(L_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # compute pointer offsets for right transposed: we need shape (BLOCK_K, BLOCK_J)
        # index right[j,k] but we want (k,j) ordering; offset = offs_k * stride_r_k + offs_j * stride_r_i
        R_ptrs = R_ptrs_base + (offs_k[:, None] * stride_r_k) + (offs_j[None, :] * stride_r_i)
        R = tl.load(R_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # dot: (BLOCK_I, BLOCK_K) @ (BLOCK_K, BLOCK_J)
        acc += tl.dot(L, R)
    # store result
    O_ptrs = O_ptr_base + (offs_i[:, None] * stride_o_i) + (offs_j[None, :] * stride_o_j)
    tl.store(O_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) kernel for AlphaFold3/Evoformer.

The forward pass consists of:
1) LayerNorm on the channel dimension C.
2) Linear projections (left/right) and linear gates (left/right/out) followed by a sigmoid.
3) Optional pair mask is applied before gating.
4) The core O(N³) contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   implemented as a batched matrix‑multiply L @ Rᵀ per head (head = batch × hidden_dim)
   with a custom Triton kernel.
5) LayerNorm on the hidden dimension, out‑gate gating and a final linear projection.

Only the O(N³) contraction is Triton‑implemented; all other operations use
torch (highly‑optimized cuBLAS/cuDNN kernels). The kernel works on fp16 inputs
and accumulates in fp32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------------- #
# Triton kernel: out = left @ rightᵀ  (per head = batch * hidden_dim)
# ----------------------------------------------------------------------------- #
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # dimensions
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute, for each head h = b * H + d,
        out[i, j] = Σ_k left[i, k] * right[j, k]   (matrix‑multiply L @ Rᵀ)
    left/right/out have shape (B, N, N, H) in row‑major order.
    """
    pid_head = tl.program_id(0)          # head index = batch * H + hidden
    pid_i    = tl.program_id(1)          # tile row index
    pid_j    = tl.program_id(2)          # tile col index

    # --------------------------------------------------------------------- #
    # 1) Derive batch and hidden indices from flattened head id
    # --------------------------------------------------------------------- #
    batch = pid_head // H
    hid   = pid_head % H

    # --------------------------------------------------------------------- #
    # 2) Compute starting offsets for the current tile
    # --------------------------------------------------------------------- #
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    # Base pointers for the selected (batch, hidden) slice
    left_base  = left_ptr  + batch * stride_l_b  + hid * stride_l_h
    right_base = right_ptr + batch * stride_r_b + hid * stride_r_h
    out_base   = out_ptr   + batch * stride_o_b + hid * stride_o_h

    # Offsets within the tile (vector of indices)
    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------------- #
    # 3) Accumulate the dot‑product over K in blocks of BLOCK_K
    # --------------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----------- load left tile : shape (BLOCK_I, BLOCK_K) --------------
        left_ptrs = left_base \
                    + (offs_i[:, None] * stride_l_i) \
                    + (offs_k[None, :] * stride_l_k)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ----------- load right tile (transposed) : (BLOCK_K, BLOCK_J) -----
        # right[j, k] -> we load as (k, j) to get the required transpose
        right_ptrs = right_base \
                     + (offs_k[:, None] * stride_r_k) \
                     + (offs_j[None, :] * stride_r_i)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # ----------- multiply‑accumulate ------------------------------------
        acc += tl.dot(left, right)   # (BLOCK_I, BLOCK_K) @ (BLOCK_K, BLOCK_J)

    # --------------------------------------------------------------------- #
    # 4) Write the result (acc) back to global memory
    # --------------------------------------------------------------------- #
    out_ptrs = out_base \
               + (offs_i[:, None] * stride_o_i) \
               + (offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------------- #
# Helper that wraps the Triton kernel, handling tensor shape/strides
# ----------------------------------------------------------------------------- #
def tri_mul_einsum(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched contraction   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    Arguments:
        left, right : torch.Tensor of shape (B, N, N, H) and dtype torch.float16
    Returns:
        out : torch.Tensor of shape (B, N, N, H) and dtype torch.float16
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    # Ensure contiguity (required for correct stride interpretation)
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)   # fp16 output

    # Extract element‑wise strides (in number of elements, not bytes)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Tile sizes – chosen to fit H100 shared‑memory and register limits
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    # Grid: (head, i_tile, j_tile) where head = B * H
    num_heads = B * H
    num_i_tiles = (N + BLOCK_I - 1) // BLOCK_I
    num_j_tiles = (N + BLOCK_J - 1) // BLOCK_J
    grid = (num_heads, num_i_tiles, num_j_tiles)

    # Launch kernel
    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------------- #
# Main entry point required by the problem statement
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module, using a custom Triton kernel
    for the core O(N³) contraction.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape (B, N, N, dim) on the same device as the inputs.
    """
    # --------------------------------------------------------------------- #
    # Unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    # config fields
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # 1) LayerNorm on channel dimension (dim)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B, N, N, dim], float32

    # --------------------------------------------------------------------- #
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------------- #
    left_proj = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right_proj = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------------- #
    # 3) Optional mask (mask shape [B, N, N])
    # --------------------------------------------------------------------- #
    if mask_tensor is not None:
        # mask is broadcasted over the hidden dimension
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # --------------------------------------------------------------------- #
    # 4) Gates (sigmoid of linear projections)
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------------- #
    # 5) Apply gates
    # --------------------------------------------------------------------- #
    left = left_proj * left_gate          # (B,N,N,hidden_dim)
    right = right_proj * right_gate

    # Cast to fp16 before the heavy O(N³) contraction to reduce memory traffic
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 6) Core contraction via Triton kernel
    # --------------------------------------------------------------------- #
    out_fp16 = tri_mul_einsum(left_fp16, right_fp16)   # (B,N,N,hidden_dim), fp16
    out = out_fp16.to(torch.float32)                  # back to fp32 for the rest

    # --------------------------------------------------------------------- #
    # 7) to_out_norm (LayerNorm over hidden_dim)
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------------- #
    # 8) Apply out‑gate and final linear projection back to dim
    # --------------------------------------------------------------------- #
    out = out * out_gate                    # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])  # (B,N,N,dim)

    return out
