def elementwise_fuse_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N
    offs = tl.arange(0, BLOCK)
    mask = offs < H
    proj_offs = b * stride_proj_b + i * stride_proj_i + j * stride_proj_j + offs * stride_proj_h
    gate_offs = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + offs * stride_gate_h

"""
TriMul forward kernel (outgoing version).

Algorithm
---------
1) Layer‑normalize the input tensor `x` (shape B,N,N,Dim) using the supplied
   `norm.weight`/`norm.bias`.
2) Project the normalized tensor into the hidden dimension (H) with three
   linear layers:
        left_proj = x @ left_proj.weight.T
        right_proj = x @ right_proj.weight.T
        left_gate = sigmoid(x @ left_gate.weight.T)
        right_gate = sigmoid(x @ right_gate.weight.T)
        out_gate  = sigmoid(x @ out_gate.weight.T)
3) Fuse mask, projection and gate via a tiny Triton kernel:
        left  = left_proj * left_gate * mask   (mask broadcasted on H)
        right = right_proj * right_gate * mask
   (When `nomask=True` the mask is all‑ones.)
4) For each hidden channel compute a matrix‑multiply over the sequence
   dimension:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is performed as a batched GEMM (`torch.bmm`) on the tensor‑core
   CUDA backend:
        L_h = left.permute(0,3,1,2).reshape(B*H,N,N)
        R_h = right.permute(0,3,1,2).reshape(B*H,N,N)
        out_h = (L_h) @ (R_h).T
   and reshaped back to (B,N,N,H).
5) Layer‑norm the result with `to_out_norm.weight`/`bias`, multiply by
   `out_gate`, and finally linearly project back to the original dimension
   with `to_out.weight`.
6) Return a float32 tensor of shape (B,N,N,Dim).

The only Triton‑implemented part is step 3, which fuses three element‑wise
operations and avoids extra memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask, projection and gate (half‑precision)
# ----------------------------------------------------------------------
@triton.jit
def fuse_mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,          # pointers
    B, N, H,                                        # logical sizes
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_H: tl.constexpr                            # compile‑time block size
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # ----- decode (b,i,j) from linear pid -----
    b = pid // (N * N)
    idx = pid % (N * N)
    i = idx // N
    j = idx % N

    # ----- vector of hidden channels -----
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # ----- compute element pointers -----
    proj_ptrs = proj_ptr + b * stride_proj_b + i * stride_proj_i + j * stride_proj_j + offs_h * stride_proj_h
    gate_ptrs = gate_ptr + b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + offs_h * stride_gate_h
    out_ptrs  = out_ptr  + b * stride_out_b  + i * stride_out_i  + j * stride_out_j  + offs_h * stride_out_h

    # mask is a scalar per (b,i,j); broadcast to the hidden dimension
    mask_val = tl.load(mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j)

    # ----- load, compute and store -----
    proj = tl.load(proj_ptrs, mask=mask_h, other=0.0)
    gate = tl.load(gate_ptrs, mask=mask_h, other=0.0)
    out  = proj * gate * mask_val
    tl.store(out_ptrs, out, mask=mask_h)


# ----------------------------------------------------------------------
# Entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, Dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32) – may be None
        - weights_dict : dict of torch.Tensor (all weight/bias tensors)
        - config_dict  : dict with keys
            * "dim"        – input feature dimension
            * "hidden_dim" – hidden dimension after projection
            * "nomask"     – bool, True ⇒ ignore mask (use all‑ones)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, Dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, Dim = input_tensor.shape
    hidden = config["hidden_dim"]
    # ------------------------------------------------------------------
    # cast everything to half for speed (float16) – H100 tensor cores
    # ------------------------------------------------------------------
    # keep a float32 copy only for final output
    x = input_tensor
    # LayerNorm on the last dimension
    x = F.layer_norm(
        x,
        normalized_shape=(Dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    ).to(torch.float16)

    # --------------------------------------------------------------
    # Linear projections (half precision)
    # --------------------------------------------------------------
    # Helper to cast weight matrices once
    def _to_half(w):
        return w.to(device=device, dtype=torch.float16)

    left_proj_w  = _to_half(weights["left_proj.weight"])
    right_proj_w = _to_half(weights["right_proj.weight"])
    left_gate_w  = _to_half(weights["left_gate.weight"])
    right_gate_w = _to_half(weights["right_gate.weight"])
    out_gate_w   = _to_half(weights["out_gate.weight"])
    to_out_w     = _to_half(weights["to_out.weight"])

    # proj: (B,N,N,hidden)
    left_proj  = F.linear(x, left_proj_w)   # bias=False
    right_proj = F.linear(x, right_proj_w)

    # gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # Fuse mask, projection and gate using Triton
    # ------------------------------------------------------------------
    if config.get("nomask", False) or mask_tensor is None:
        mask = torch.ones((B, N, N), device=device, dtype=torch.float16)
    else:
        mask = mask_tensor.to(device=device, dtype=torch.float16)

    # allocate fused tensors
    left  = torch.empty_like(left_proj)   # (B,N,N,hidden)
    right = torch.empty_like(right_proj)

    # kernel launch configuration
    total_positions = B * N * N
    BLOCK_H = 128                     # hidden ≤ 128 in all provided configs
    grid = (total_positions,)

    # left side
    fuse_mask_gate_kernel[grid](
        left_proj, left_gate, mask, left,
        B, N, hidden,
        *left_proj.stride(),
        *left_gate.stride(),
        *mask.stride(),
        *left.stride(),
        BLOCK_H=BLOCK_H,
    )
    # right side
    fuse_mask_gate_kernel[grid](
        right_proj, right_gate, mask, right,
        B, N, hidden,
        *right_proj.stride(),
        *right_gate.stride(),
        *mask.stride(),
        *right.stride(),
        BLOCK_H=BLOCK_H,
    )
    # ------------------------------------------------------------------
    # Core TriMul reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # Implemented as a batched GEMM over the sequence dimension.
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for each side
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # batched matmul: (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, hidden)
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm, out_gate, final linear projection
    # ------------------------------------------------------------------
    # to_out_norm (LayerNorm over hidden dimension)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # multiply by out_gate (already half‑precision)
    out = out * out_gate

    # final linear projection back to Dim
    out = F.linear(out, to_out_w)   # bias=False

    # ------------------------------------------------------------------
    # Cast back to float32 for the return value
    # ------------------------------------------------------------------
    return out.to(dtype=torch.float32)
```
