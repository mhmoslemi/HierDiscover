def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          B, N, H,
                          stride_ah0, stride_ah1, stride_ah2, stride_ah3,
                          stride_bh0, stride_bh1, stride_bh2, stride_bh3,
                          stride_ch0, stride_ch1, stride_ch2, stride_ch3,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_head = tl.program_id(0)  # head index: 0..B*H-1
    pid_m = tl.program_id(1)     # block row index
    pid_n = tl.program_id(2)     # block col index

    # compute actual head indices: batch and head
    b = pid_head // H
    h = pid_head % H
    # compute offsets for A,B,C based on b,h
    a_ptr = A_ptr + b * stride_ah0 + h * stride_ah1
    b_ptr = B_ptr + b * stride_bh0 + h * stride_bh1
    c_ptr = C_ptr + b * stride_ch0 + h * stride_ch1

    # Offsets within matrices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block [M, K]
        a = tl.load(a_ptr + (offs_m[:, None] * stride_ah2 + offs_k[None, :] * stride_ah3),
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        # Load B block [K, N]
        b = tl.load(b_ptr + (offs_k[:, None] * stride_bh2 + offs_n[None, :] * stride_bh3),
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Convert to fp16 for dot if desired

        # Compute dot and accumulate (fp32)

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
import math

@triton.jit
def batched_matmul_kernel(
    A, B, C,
    BATCH, N, HEAD,
    stride_ah0, stride_ah1, stride_ah2, stride_ah3,
    stride_bh0, stride_bh1, stride_bh2, stride_bh3,
    stride_ch0, stride_ch1, stride_ch2, stride_ch3,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication for TriMul.
    Computes for each head (batch*head):
        C[b,h,i,j] = sum_k A[b,h,i,k] * B[b,h,k,j]
    where A and B are half-precision, C is float32.
    """
    # Identify which head / batch we are processing.
    pid_head = tl.program_id(0)  # 0 .. BATCH*HEAD-1
    pid_m = tl.program_id(1)     # block row
    pid_n = tl.program_id(2)     # block column

    # Decompose head index.
    batch_idx = pid_head // HEAD
    head_idx = pid_head % HEAD

    # Compute base pointers for this head.
    a_ptr = A + batch_idx * stride_ah0 + head_idx * stride_ah1
    b_ptr = B + batch_idx * stride_bh0 + head_idx * stride_bh1
    c_ptr = C + batch_idx * stride_ch0 + head_idx * stride_ch1

    # Offsets for the M and N dimensions.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension.
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block (M x K) and B block (K x N)
        a = tl.load(
            a_ptr + (offs_m[:, None] * stride_ah2 + offs_k[None, :] * stride_ah3),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        b = tl.load(
            b_ptr + (offs_k[:, None] * stride_bh2 + offs_n[None, :] * stride_bh3),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        # Compute dot product (float16 inputs, float32 accumulation)
        # The dot returns float16; casting to fp32 occurs during accumulation.
        acc += tl.dot(a, b)

    # Write back the result.
    c = acc
    tl.store(
        c_ptr + (offs_m[:, None] * stride_ch2 + offs_n[None, :] * stride_ch3),
        c,
        mask=mask_m[:, None] & mask_n[None, :]
    )
```

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.
    The heavy tensor contraction
        out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    is executed by a fused batched GEMM kernel.
    All other linear / normalization / gating steps are performed with
    PyTorch primitives.
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # --- LayerNorm on input -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # --- Linear projections -------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # --- Mask --------------------------------------------------------------
    if mask is not None:

"""Triton‑accelerated forward of the outgoing TriMul.

The core contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is implemented as a batched half‑precision GEMM:
    left -> (B, H, i, k)
    right -> (B, H, k, j)
    out   -> (B, H, i, j)
All remaining linear, LayerNorm and gating steps are performed with
PyTorch primitives. The kernel uses fp16 inputs and fp32 accumulation,
leveraging the tensor‑core units of an H100.
"""
import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    A, B, C,
    BATCH, N, HEAD,
    stride_ah0, stride_ah1, stride_ah2, stride_ah3,
    stride_bh0, stride_bh1, stride_bh2, stride_bh3,
    stride_ch0, stride_ch1, stride_ch2, stride_ch3,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C[b,h,i,j] = Σ_k A[b,h,i,k] * B[b,h,k,j].

    A, B : fp16  (B, H, N, N)   (row‑major)
    C    : fp32  (B, H, N, N)
    """
    pid_head = tl.program_id(0)   # 0 .. BATCH*HEAD-1
    pid_m = tl.program_id(1)      # block row
    pid_n = tl.program_id(2)      # block column

    # Decompose head index.
    batch_idx = pid_head // HEAD
    head_idx = pid_head % HEAD

    # Base pointers for this (batch, head) slice.
    a_ptr = A + batch_idx * stride_ah0 + head_idx * stride_ah1
    b_ptr = B + batch_idx * stride_bh0 + head_idx * stride_bh1
    c_ptr = C + batch_idx * stride_ch0 + head_idx * stride_ch1

    # Offsets inside the tile.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K.
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load tiles of shape (M, K) and (K, N).
        a = tl.load(
            a_ptr + (offs_m[:, None] * stride_ah2 + offs_k[None, :] * stride_ah3),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        b = tl.load(
            b_ptr + (offs_k[:, None] * stride_bh2 + offs_n[None, :] * stride_bh3),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )
        # fp16 × fp16 → fp32 accumulation.
        acc += tl.dot(a, b)

    # Write back the result tile.
    tl.store(
        c_ptr + (offs_m[:, None] * stride_ch2 + offs_n[None, :] * stride_ch3),
        acc,
        mask=mask_m[:, None] & mask_n[None, :]
    )

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton GEMM
    for the cubic contraction. Returns a tensor of shape
    [bs, seq_len, seq_len, dim] with dtype torch.float32.
    """
    x, mask, weights, config = data

    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 4. Gating
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Prepare tensors for the Triton kernel
    #    - cast to FP16 for bandwidth reduction
    #    - permute to (B, H, i, k) / (B, H, k, j)
    # -----------------------------------------------------------------
    left_h = left.to(torch.float16).permute(0, 3, 1, 2).contiguous()
    # right needs a transpose of its last two dimensions so that k is the
    # reduction axis and j is the output column.
    right_h = right.to(torch.float16).permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_h.shape

    # Output buffer (FP32)
    out_perm = torch.empty((B, H, N, N), dtype=torch.float32, device=device)

    # -----------------------------------------------------------------
    # 6. Triton batched GEMM
    # -----------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    total_heads = B * H
    grid = (
        total_heads,
        math.ceil(N / BLOCK_M),
        math.ceil(N / BLOCK_N),
    )

    batched_matmul_kernel[grid](
        left_h,
        right_h,
        out_perm,
        B,
        N,
        H,
        left_h.stride(0), left_h.stride(1), left_h.stride(2), left_h.stride(3),
        right_h.stride(0), right_h.stride(1), right_h.stride(2), right_h.stride(3),
        out_perm.stride(0), out_perm.stride(1), out_perm.stride(2), out_perm.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # -----------------------------------------------------------------
    # 7. Restore original layout (B,N,N,hidden_dim)
    # -----------------------------------------------------------------
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 8. Output LayerNorm, gate and final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out

The heavy cubic contraction

leveraging the Tensor‑Core units of an H100.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    A,               # left  tensor (fp16)  [B,H,N,N]  (i,k)
    B,               # right tensor (fp16)  [B,H,N,N]  (k,j)
    C,               # out   tensor (fp32)  [B,H,N,N]  (i,j)
    BATCH, N, HEAD,  # problem sizes
    stride_ah0, stride_ah1, stride_ah2, stride_ah3,   # A strides
    stride_bh0, stride_bh1, stride_bh2, stride_bh3,   # B strides
    stride_ch0, stride_ch1, stride_ch2, stride_ch3,   # C strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C[b,h,i,j] = Σ_k A[b,h,i,k] * B[b,h,k,j].

    A, B : fp16, row‑major, shape (B, H, N, N)
    C    : fp32, row‑major, shape (B, H, N, N)
    """
    pid_head = tl.program_id(0)   # 0 … BATCH*HEAD‑1
    pid_m = tl.program_id(1)      # tile row
    pid_n = tl.program_id(2)      # tile col

    # split the fused head index into batch and head
    batch_idx = pid_head // HEAD
    head_idx = pid_head % HEAD

    # base pointers for the current (batch, head) slice
    a_ptr = A + batch_idx * stride_ah0 + head_idx * stride_ah1
    b_ptr = B + batch_idx * stride_bh0 + head_idx * stride_bh1
    c_ptr = C + batch_idx * stride_ch0 + head_idx * stride_ch1

    # tile offsets inside the matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.to(x.dtype).unsqueeze(-1)         # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 4. Gating
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Prepare tensors for the Triton kernel
    #    - cast to fp16 for traffic reduction
    #    - permute to (B, H, i, k) and (B, H, k, j)
    # -----------------------------------------------------------------
    left_h = left.to(torch.float16).permute(0, 3, 1, 2).contiguous()   # (B,H,N,N) i,k
    right_h = right.to(torch.float16).permute(0, 3, 2, 1).contiguous()  # (B,H,N,N) k,j

    B, H, N, _ = left_h.shape

    # output buffer (FP32)
    out_perm = torch.empty((B, H, N, N), dtype=torch.float32, device=device)

    # -----------------------------------------------------------------
    # 6. Triton batched GEMM
    # -----------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    total_heads = B * H
    grid = (
        total_heads,
        math.ceil(N / BLOCK_M),
        math.ceil(N / BLOCK_N),
    )

    batched_matmul_kernel[grid](
        left_h,
        right_h,
        out_perm,
        B,
        N,
        H,
        left_h.stride(0),
        left_h.stride(1),
        left_h.stride(2),
        left_h.stride(3),
        right_h.stride(0),
        right_h.stride(1),
        right_h.stride(2),
        right_h.stride(3),
        out_perm.stride(0),
        out_perm.stride(1),
        out_perm.stride(2),
        out_perm.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # -----------------------------------------------------------------
    # 7. Restore the (B,N,N,hidden_dim) layout
    # -----------------------------------------------------------------
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # (B,N,N,hidden_dim)

    # -----------------------------------------------------------------
    # 8. Output LayerNorm, gating, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```
