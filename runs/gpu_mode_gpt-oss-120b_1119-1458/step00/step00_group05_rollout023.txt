    x, mask, weights, config = data
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on dim
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # Linear projections & gating
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

        right = right * mask_

    right = right * right_gate

    L = bs * hidden_dim

    dim = config["dim"]

    else:

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

        right = right * mask_

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))

    # Apply to_out_norm via Triton kernel

    out_norm = torch.empty_like(out)

    BLOCK_H = 32

"""
TriMul (outgoing) forward pass implemented with a mix of PyTorch and a custom
Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (dim‑wise) using the supplied `norm` weights.
2. Compute the five linear projections (left, right, left_gate, right_gate,
   out_gate) with `torch.nn.functional.linear`.  Apply a sigmoid to the three
   gate projections.
3. Optional mask multiplication (only when ``config["nomask"]`` is False).
4. Apply the per‑element gates: ``left *= left_gate`` and ``right *= right_gate``.
5. Perform the heavy triangular multiplicative update:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   using a fast `torch.einsum` (equivalent to a batch of matrix multiplications.
6. Fuse the post‑update LayerNorm (`to_out_norm`) with a custom Triton kernel.
   The kernel computes the mean/variance across the hidden dimension,
   applies the learnt scale/bias, and writes the normalized tensor.
7. Multiply by the `out_gate` (gate after the `to_out_norm`).
8. Final linear projection back to the original `dim` using `torch.nn.functional.linear`.

Only step 6 is performed in Triton; the rest uses highly‑optimised
PyTorch primitives.  This satisfies the requirement that *part* of the
operation be implemented in a kernel while keeping the overall
implementation fast and numerically identical to the reference code.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: LayerNorm across the hidden dimension (H) for each
# (batch, i, j) element.  It writes the normalized tensor into `out_norm`.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_hidden_kernel(
    inp_ptr,                     # const float*   (B,N,N,H)  input
    out_ptr,                     # float*         (B,N,N,H)  normalized output
    weight_ptr,                  # const float*   (H)       to_out_norm.weight
    bias_ptr,                    # const float*   (H)       to_out_norm.bias
    B, N, H,                     # int64 scalars
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,   # int64 stride of inp
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,# int64 stride of out
    eps,                         # float32 epsilon for LayerNorm
    BLOCK_SIZE: tl.constexpr,   # program (block) size in 1‑D
    BLOCK_H:   tl.constexpr,    # tile size for the hidden dimension
):
    pid  = tl.program_id(0)                     # block index
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)   # linear ids
    total = B * N * N
    mask = offs < total
    if tl.sum(mask) == 0:
        return

    # ---- Compute (b,i,j) coordinates for each thread in the block ----
    b_idx   = offs // (N * N)
    rem     = offs % (N * N)
    i_idx   = rem // N
    j_idx   = rem % N

    # ---- Base pointers for the hidden‑vector of each (b,i,j) ----
    inp_base  = inp_ptr  + b_idx * stride_in_b  + i_idx * stride_in_i  + j_idx * stride_in_j
    out_base  = out_ptr  + b_idx * stride_out_b + i_idx * stride_out_i + j_idx * stride_out_j

    # ---- First pass: compute mean and variance across H ----
    sum_val = tl.zeros([BLOCK_SIZE], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_SIZE], dtype=tl.float32)

    for off_h in range(0, H, BLOCK_H):
        cur_h   = off_h + tl.arange(0, BLOCK_H)            # (BLOCK_H,)
        hmask   = cur_h < H                                 # (BLOCK_H,)
        # load values: shape (BLOCK_SIZE, BLOCK_H)
        vals = tl.load(
            inp_base[:, None] + cur_h[None, :] * stride_in_h,
            mask = mask[:, None] & hmask[None, :],
            other = 0.0,
        )
        vals_f = vals.to(tl.float32)
        sum_val += tl.sum(vals_f, axis=1)      # (BLOCK_SIZE,)
        sum_sq  += tl.sum(vals_f * vals_f, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ---- Second pass: write normalized values (apply scale & bias) ----
    for off_h in range(0, H, BLOCK_H):
        cur_h   = off_h + tl.arange(0, BLOCK_H)
        hmask   = cur_h < H
        vals = tl.load(
            inp_base[:, None] + cur_h[None, :] * stride_in_h,
            mask = mask[:, None] & hmask[None, :],
            other = 0.0,
        )
        vals_f = vals.to(tl.float32)

        w = tl.load(weight_ptr + cur_h, mask=hmask, other=0.0)   # (BLOCK_H,)
        b = tl.load(bias_ptr   + cur_h, mask=hmask, other=0.0)   # (BLOCK_H,)

        # broadcasting mean/inv_std (BLOCK_SIZE,) to (BLOCK_SIZE, BLOCK_H)
        normed = ((vals_f - mean[:, None]) * inv_std[:, None]) * w[None, :] + b[None, :]
        tl.store(
            out_base[:, None] + cur_h[None, :] * stride_out_h,
            normed,
            mask = mask[:, None] & hmask[None, :],
        )


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[float32] of shape (B, N, N, dim)
        - mask         : torch.Tensor (bool/int) of shape (B, N, N)
        - weights      : dict mapping string names to weight tensors
        - config       : dict containing at least "dim", "hidden_dim",
                         and optionally "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape (B, N, N, dim), dtype float32.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim       = config["dim"]
    hidden    = config["hidden_dim"]
    nomask    = config.get("nomask", False)
    eps       = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (dim‑wise)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Linear projections for left/right and the three gates
    # ------------------------------------------------------------------
    left      = F.linear(x_norm, weights["left_proj.weight"])          # (B,N,N,hidden)
    right     = F.linear(x_norm, weights["right_proj.weight"])         # (B,N,N,hidden)

    left_gate  = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)   # (B,N,N,1)
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Apply gates to left/right
    # ------------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triangular multiplicative update (einsum over the k dimension)
    # ------------------------------------------------------------------
    # out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    out = torch.einsum("b i k d, b j k d -> b i j d", left, right)   # (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 6. to_out_norm (LayerNorm over hidden dim) – fused in a Triton kernel
    # ------------------------------------------------------------------
    B, N, _, H = out.shape
    out = out.contiguous()
    out_norm = torch.empty_like(out)

    # Triton launch configuration
    BLOCK_SIZE = 128          # number of (b,i,j) elements per program
    BLOCK_H    = 32           # tile size for the hidden dimension
    grid = ( (B * N * N + BLOCK_SIZE - 1) // BLOCK_SIZE, )

    layernorm_hidden_kernel[grid](
        inp_ptr        = out,
        out_ptr        = out_norm,
        weight_ptr     = weights["to_out_norm.weight"],
        bias_ptr       = weights["to_out_norm.bias"],
        B              = B,
        N              = N,
        H              = H,
        stride_in_b    = out.stride(0),
        stride_in_i    = out.stride(1),
        stride_in_j    = out.stride(2),
        stride_in_h    = out.stride(3),
        stride_out_b   = out_norm.stride(0),
        stride_out_i   = out_norm.stride(1),
        stride_out_j   = out_norm.stride(2),
        stride_out_h   = out_norm.stride(3),
        eps            = eps,
        BLOCK_SIZE     = BLOCK_SIZE,
        BLOCK_H        = BLOCK_H,
        num_warps      = 4,
    )

    # ------------------------------------------------------------------
    # 7. Multiply by out_gate (after to_out_norm)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    # Collapse the first three dims for an efficient matmul
    out_final = F.linear(out_norm.view(-1, hidden), weights["to_out.weight"])
    out_final = out_final.view(B, N, N, dim)      # (B,N,N,dim)

    return out_final
