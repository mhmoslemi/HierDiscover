def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_la_b, stride_la_i, stride_la_k, stride_la_h,
    stride_ra_b, stride_ra_j, stride_ra_k, stride_ra_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid_j = tl.program_id(1)
    pid_bc = tl.program_id(2)

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    d = pid_bc % H

    offs_i = i0 + tl.arange(0, BLOCK_M)

    mask_j = offs_j < N

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if not nomask:

        left = left * mask_exp
        right = right * mask_exp

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # cast to half for kernel
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    B, N, _, H = left_h.shape

    # allocate output half tensor
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_h.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right_h.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_h.stride()  # note: out dims same as left

    # Grid dimensions
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    trimul_outgoing_kernel[grid](
        left_h,
        right_h,
        out_h,
        B,
        N,
        H,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_la_b, stride_la_i, stride_la_k, stride_la_h,
    stride_ra_b, stride_ra_j, stride_ra_k, stride_ra_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Triton kernel for the outgoing triangular multiplicative update.
    Computes out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d] for each batch b and channel d.
    """
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bc = tl.program_id(2)

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    b = pid_bc // H
    d = pid_bc % H

    # coordinate offsets
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K blocks
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k0 = k_iter * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile (M, K)
        left_ptrs = left_ptr + b*stride_la_b + d*stride_la_h + offs_i[:, None]*stride_la_i + offs_k[None, :]*stride_la_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile (K, N) with k first, j second
        right_ptrs = right_ptr + b*stride_ra_b + d*stride_ra_h + offs_k[:, None]*stride_ra_k + offs_j[None, :]*stride_ra_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate
        acc += tl.dot(left, right).to(tl.float32)

    # store output
    out_ptrs = out_ptr + b*stride_out_b + d*stride_out_h + offs_i[:, None]*stride_out_i + offs_j[None, :]*stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul "outgoing" kernel (AlphaFold3)

This implementation follows the reference PyTorch module but off‑loads the
dominant O(N³·C) operation to a custom Triton kernel.  The kernel computes

    out[b,i,j,c] = Σₖ left[b,i,k,c] * right[b,j,k,c]

for every batch (b) and hidden channel (c).  The remaining LayerNorm,
linear and gating operations are performed with regular PyTorch kernels.
The heavy matrix‑multiplication is tiled (64×64) and executed in fp16
with fp32 accumulation for numerical stability.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,
    right_ptr,
    out_ptr,
    # compile‑time constants
    B: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    # strides (in elements)
    stride_la_b,
    stride_la_i,
    stride_la_k,
    stride_la_h,
    stride_ra_b,
    stride_ra_j,
    stride_ra_k,
    stride_ra_h,
    stride_out_b,
    stride_out_i,
    stride_out_j,
    stride_out_h,
    # block sizes (compile‑time constants)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Triton kernel for the outgoing triangular multiplicative update.

    Computes out[b,i,j,c] = Σₖ left[b,i,k,c] * right[b,j,k,c]
    for a single (batch, channel) pair per program instance.
    """
    pid_i = tl.program_id(0)      # tile over output rows  (i)
    pid_j = tl.program_id(1)      # tile over output cols  (j)
    pid_bc = tl.program_id(2)     # flattened batch*channel index

    # start offsets of the current tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # decode batch and hidden channel from the combined axis
    b = pid_bc // H
    c = pid_bc % H

    # offsets for the i and j dimensions
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    # iterate over the reduction dimension k
    for k_iter in range(num_k):
        k0 = k_iter * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # left tile: shape (BLOCK_M, BLOCK_K)  -> indices (i, k)
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_la_b
            + c * stride_la_h
            + offs_i[:, None] * stride_la_i
            + offs_k[None, :] * stride_la_k
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # ------------------------------------------------------------------
        # right tile: shape (BLOCK_K, BLOCK_N)  -> indices (k, j)
        # (note the swapped order so we can use a regular matmul)
        # ------------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + b * stride_ra_b
            + c * stride_ra_h
            + offs_k[:, None] * stride_ra_k
            + offs_j[None, :] * stride_ra_j
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )  # fp16

        # accumulate in fp32
        acc += tl.dot(left, right).to(tl.float32)

    # ----------------------------------------------------------------------
    # store the result (convert back to fp16)
    # ----------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + c * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the TriMul module with an accelerated Triton kernel.
    Input:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Output:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim is the last axis)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (mask shape [B,N,N])
    # ------------------------------------------------------------------
    if not nomask:
        mask_unsq = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_unsq
        right = right * mask_unsq

    # ------------------------------------------------------------------
    # 4. Gating networks (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 5. Apply gates to the projected tensors
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Cast to fp16 and launch the Triton kernel
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    B, N, _, H = left_h.shape  # H == hidden_dim

    # output buffer (fp16)
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # strides (in elements) – note the order for the “right” tensor:
    #   right shape = [B, N, N, H] → stride (b, j, k, h)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_h.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right_h.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_h.stride()

    # ------------------------------------------------------------------
    # Tiling configuration.  These values work well on an H100 for the
    # supported dimensions (N up to 1024, hidden_dim=128).
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,     # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,     # j‑tiles
        B * H,                            # one program per (batch, channel)
    )

    trimul_outgoing_kernel[grid](
        left_h,
        right_h,
        out_h,
        # compile‑time constants (must be passed as literals)
        B,
        N,
        H,
        # runtime strides
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_j,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        # block sizes (constexpr)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Back to FP32 for the remaining post‑processing
    out = out_h.to(torch.float32)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm (over hidden_dim) and gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
