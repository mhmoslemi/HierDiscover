def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    
    # everything as float16 for speed
    x = x.to(torch.float16)
    mask = mask.to(torch.float16)
    
    # layer norm
    x = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(torch.float16),
        bias=weights["norm.bias"].to(torch.float16),
        eps=1e-5
    )
    
    # linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x, weights["right_proj.weight"].to(torch.float16))
    
    # gating linear
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16)))

    # mask broadcast
    # shape for mask: [bs,N,N,1]
    mask = mask.unsqueeze(-1)  # [bs,N,N,1]
    
    # Apply mask and gates via Triton kernel
    left, right = triton_apply_mask_and_gate(left, right, mask, left_gate, right_gate)
    
    # Compute out = left @ right^T per hidden dim
    bs, N, _, _ = left.shape
    # reshape
    left_reshape = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_reshape = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    # perform batched matmul
    out_reshape = torch.bmm(left_reshape, right_reshape.transpose(1, 2))
    out = out_reshape.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    
    # to_out_norm and to_out
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=1e-5
    )
    out = out * out_gate  # out_gate shape [bs,N,N,hidden_dim]; elementwise product

    out = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))
    
    # Cast back to float32 for output

def gate_mul_kernel(

        weight=weights["norm.weight"].to(torch.float16),
        bias=weights["norm.bias"].to(torch.float16),
        eps=1e-5,
    )
    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x, weights["right_proj.weight"].to(torch.float16))
    # Apply mask if provided
    if mask is not None:
        mask = mask.to(torch.float16)
        # broadcast mask across hidden dim by unsqueeze
        mask = mask.unsqueeze(-1) # shape [bs,N,N,1]
        left = left * mask
        right = right * mask
    # Gates
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16))
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16))
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16))
    )

    apply_gate_mul(left, right, left_gate, right_gate)

    right_mat = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The implementation follows the reference PyTorch module:

1. Layer‑norm the input (last dim = dim).
2. Linear projections to a hidden dimension (left, right) and three
   sigmoid‑gates (left_gate, right_gate, out_gate).
3. Apply the binary mask (if provided) and multiply the projections by
   their respective gates.  The element‑wise gate multiplication is
   fused in a tiny Triton kernel (`gate_mul_kernel`).
4. Compute the triangular multiplicative update:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to a batched matrix‑multiply
   `left_d @ right_d.T` for each hidden channel d.
   We perform this with a single `torch.bmm` on a flattened batch of
   size B·hidden_dim – a call to cuBLAS that is highly optimized on H100.
5. Layer‑norm the hidden result, multiply by `out_gate`,
   and project back to the original dimension with a final linear
   layer.
All heavy element‑wise work (mask‑gate multiplication) is done in Triton,
while the dense N×N matrix multiplies use cuBLAS for maximum throughput.
The kernel works for both float16 and float32 inputs (internally we
use float16 for speed and cast the final result back to float32).
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: multiply left/right tensors by their respective gates.
# Each thread processes one element; the kernel works on flat tensors.
# ----------------------------------------------------------------------
@triton.jit
def gate_mul_kernel(
    left_ptr,          # *float16
    right_ptr,         # *float16
    left_gate_ptr,     # *float16
    right_gate_ptr,    # *float16
    num_elements: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """Element‑wise: left *= left_gate ; right *= right_gate."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < num_elements

    # Load values (dtype = float16)
    left = tl.load(left_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)

    # Apply gates
    left = left * lgate
    right = right * rgate

    # Store results back
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gate_mul(left: torch.Tensor,
                    right: torch.Tensor,
                    left_gate: torch.Tensor,
                    right_gate: torch.Tensor):
    """
    Calls `gate_mul_kernel` on the flattened tensors.
    All tensors must be contiguous float16.
    """
    assert left.is_contiguous() and right.is_contiguous()
    assert left_gate.is_contiguous() and right_gate.is_contiguous()
    numel = left.numel()
    BLOCK_SIZE = 1024  # fits in H100 max threads per block
    grid = lambda meta: (triton.cdiv(numel, meta['BLOCK_SIZE']),)

    gate_mul_kernel[grid](
        left, right,
        left_gate, right_gate,
        num_elements=numel,
        BLOCK_SIZE=BLOCK_SIZE,
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul (outgoing) operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim]
            - mask: torch.Tensor or None, shape [B, N, N] (binary)
            - weights: dict of model parameters (all tensors on the same device)
            - config: dict with keys "dim", "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # Work in FP16 for speed; final result will be cast back to FP32.
    x = input_tensor.to(torch.float16)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(torch.float16),
        bias=weights["norm.bias"].to(torch.float16),
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(
        x,
        weights["left_proj.weight"].to(torch.float16),
        bias=None,
    )
    right = torch.nn.functional.linear(
        x,
        weights["right_proj.weight"].to(torch.float16),
        bias=None,
    )

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(
            x,
            weights["left_gate.weight"].to(torch.float16),
            bias=None,
        )
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(
            x,
            weights["right_gate.weight"].to(torch.float16),
            bias=None,
        )
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(
            x,
            weights["out_gate.weight"].to(torch.float16),
            bias=None,
        )
    )

    # ------------------------------------------------------------------
    # 5. Fuse gate multiplication with a Triton kernel
    # ------------------------------------------------------------------
    # Ensure contiguous memory layout for the kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    _apply_gate_mul(left, right, left_gate, right_gate)

    # ------------------------------------------------------------------
    # 6. Triangular multiplicative update (batched GEMM)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    This is equivalent to (left_d @ right_d.T) per hidden channel d.
    # ------------------------------------------------------------------
    bs, N, _, _ = left.shape
    # Rearrange to (B*hidden_dim, N, N) for a single bmm call
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)

    # Batched matrix multiplication – cuBLAS handles fp16 very efficiently.
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Post‑processing: LayerNorm, out_gate, final linear
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=1e-5,
    )

    out = out * out_gate  # element‑wise gate

    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Cast back to float32 to match expected output dtype.
    return out.to(torch.float32)
