def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program id for i, j, h, and maybe batch dimension
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2) # combine batch and hidden dimension

    # compute batch index and hidden index from combined pid_hb:
    batch_id = pid_hb // H
    hid = pid_hb % H

    # compute offsets...
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    # dims: i < N, j < N

    # create mask for out-of-range i/j
    mask_i = i < N
    mask_j = j < N

    # pointers for left: shape (B,N,N,H)
    # left offset: b*stride_left_b + i*stride_left_i + k*stride_left_k + h*stride_left_h
    # right offset: b*stride_right_b + j*stride_right_j + k*stride_right_k + h*stride_right_h
    # out offset: b*stride_out_b + i*stride_out_i + j*stride_out_j + h*stride_out_h

    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Use tl.dot? tl.dot expects vectors length K, but we have block of K. Could do tl.dot per i,j pair but that's heavy.
        # Alternative: compute outer product across k and sum using matrix multiplication: left_block @ right_block.T
        # left shape (BLOCK_I, BLOCK_K); right shape (BLOCK_J, BLOCK_K) need transpose to (BLOCK_K, BLOCK_J)
        # Compute product as left_block @ right_block.T -> (BLOCK_I, BLOCK_J)
        acc += tl.dot(left, right, trans_b=True) # needs dot across k dimension with trans_b flag

"""
TriMul (outgoing) – Triton‑accelerated implementation.

The heavy part of the TriMul module is the cubic‑complex
einsum:

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` are tensors of shape
[B, N, N, H] (H = hidden_dim).  This is equivalent to a
batched matrix multiplication `left @ rightᵀ` for each
hidden channel `d`.  The kernel below computes this
operation in FP16 while accumulating in FP32, tiling the
N×N output with blocks of size (BLOCK_I, BLOCK_J) and
looping over K‑tiles of size BLOCK_K.

All other steps (LayerNorms, linear projections, gates,
and the final projection) are performed with PyTorch
operations.  The kernel is launched with a 3‑D grid:
(i‑tile, j‑tile, batch*hidden‑dim) – i.e. each hidden
channel of each batch is processed by an independent
program instance.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # meta parameters
    B, N, H,
    # strides (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time block sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication:
         out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
       left/right: [B, N, N, H] (FP16)
       out        : [B, N, N, H] (FP32)
    """

    pid_i = tl.program_id(0)                     # i‑tile index
    pid_j = tl.program_id(1)                     # j‑tile index
    pid_hb = tl.program_id(2)                    # combined batch‑hidden index

    batch_id = pid_hb // H                       # real batch index
    hid = pid_hb % H                             # hidden‑channel index

    # 1‑D offsets inside the N dimension
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # masks for out‑of‑range threads
    mask_i = i < N
    mask_j = j < N

    # -----------------------------------------------------------------
    # Pointers to the beginning of the current batch & hidden channel
    # -----------------------------------------------------------------
    left_base  = left_ptr  + batch_id * stride_left_b  + hid * stride_left_h
    right_base = right_ptr + batch_id * stride_right_b + hid * stride_right_h
    out_base   = out_ptr   + batch_id * stride_out_b   + hid * stride_out_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # -------------------------------------------------------------
    # Loop over K dimension in tiles
    # -------------------------------------------------------------
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ----- load left tile (i,k) -------------------------------------------------
        # shape: (BLOCK_I, BLOCK_K)
        left_offset = i[:, None] * stride_left_i + k[None, :] * stride_left_k
        left = tl.load(left_base + left_offset,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ----- load right tile transposed (k,j) ------------------------------------
        # we need right[j,k] -> treat as (k,j) for matmul
        # shape: (BLOCK_K, BLOCK_J)
        right_offset = k[:, None] * stride_right_k + j[None, :] * stride_right_j
        right = tl.load(right_base + right_offset,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # ----- block matrix multiplication -----------------------------------------
        # left: (BLOCK_I, BLOCK_K), right: (BLOCK_K, BLOCK_J)
        acc += tl.dot(left, right)             # result (BLOCK_I, BLOCK_J)

    # -----------------------------------------------------------------
    # Store the result
    # -----------------------------------------------------------------
    out_offset = i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_base + out_offset,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # -----------------------------------------------------------------
    # Extract model parameters (they are already on the correct device)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast on the hidden dimension)
    # -----------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] -> [B,N,N,1]
        mask = mask_tensor.to(dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating terms (sigmoid)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # -----------------------------------------------------------------
    # 5. Apply gates
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Triton kernel: cubic contraction (einsum)
    # -----------------------------------------------------------------
    # Cast to FP16 for reduced memory bandwidth; accumulation will be FP32
    left_h  = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    B, N, _, H = left_h.shape   # H == hidden_dim
    assert H == hidden_dim

    out = torch.empty_like(left_h, dtype=torch.float32)  # [B,N,N,H] FP32 accumulator

    # block sizes – empirically good on H100
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    # strides (in elements) of the contiguous tensors
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_h.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right_h.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,   # i‑tiles
        (N + BLOCK_J - 1) // BLOCK_J,   # j‑tiles
        B * H,                           # batch * hidden‑channel
    )

    trimul_outgoing_kernel[grid](
        left_h,
        right_h,
        out,
        B,
        N,
        H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 7. Output LayerNorm (over hidden_dim)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # [B,N,N,hidden_dim] float32

    # -----------------------------------------------------------------
    # 8. Apply output gate
    # -----------------------------------------------------------------
    out = out * out_gate

    # -----------------------------------------------------------------
    # 9. Final linear projection back to `dim`
    # -----------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_weight)  # [B,N,N,dim]

    return out

All other tensor operations (LayerNorms, linear layers, sigmoid gates, mask handling) are performed with standard PyTorch ops.
