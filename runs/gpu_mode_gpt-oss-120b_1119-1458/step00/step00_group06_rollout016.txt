import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for LayerNorm
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,
    dim, eps,
    stride_x_b, stride_x_i, stride_x_j, stride_x_d,
    stride_y_b, stride_y_i, stride_y_j, stride_y_d,
    stride_w, stride_b,
    N,                # seq_len
    BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    # Compute indices for the (b,i,j) instance
    # Use integer division
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # Base offsets for this instance in x and y
    off = b * stride_x_b + i * stride_x_i + j * stride_x_j
    off_y = b * stride_y_b + i * stride_y_i + j * stride_y_j

    # -----------------------------------------------------------------
    # First pass: accumulate sum and sum of squares across the feature dim.
    # -----------------------------------------------------------------
    sum_val = tl.zeros([1], dtype=tl.float32)
    sum_sq  = tl.zeros([1], dtype=tl.float32)

    offset_d = 0
    while offset_d < dim:
        col = tl.arange(0, BLOCK_D)
        cur = offset_d + col
        mask = cur < dim
        x = tl.load(x_ptr + off + cur * stride_x_d, mask=mask, other=0.0)
        sum_val += tl.sum(x, axis=0)
        sum_sq  += tl.sum(x * x, axis=0)
        offset_d += BLOCK_D

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean

    # -----------------------------------------------------------------
    # Second pass: write normalized output using the learned weight/bias.
    # -----------------------------------------------------------------
    offset_d = 0
    while offset_d < dim:
        col = tl.arange(0, BLOCK_D)
        cur = offset_d + col
        mask = cur < dim
        x = tl.load(x_ptr + off + cur * stride_x_d, mask=mask, other=0.0)
        w = tl.load(w_ptr + cur * stride_w, mask=mask, other=0.0)
        b = tl.load(b_ptr + cur * stride_b, mask=mask, other=0.0)

        x_hat = (x - mean) * tl.rsqrt(var + eps)
        y = x_hat * w + b
        tl.store(y_ptr + off_y + cur * stride_y_d, y, mask=mask)
        offset_d += BLOCK_D

def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm (Trion)
    x_norm = layer_norm_triton(x, weights["norm.weight"], weights["norm.bias"], eps)

    # linear projections
    left = F.linear(x_norm, weights["left_proj.weight"])  # shape B,N,N,hidden_dim
    right = F.linear(x_norm, weights["right_proj.weight"])

    # mask handling
    if not config.get("nomask", False) and mask is not None:
        # broadcast mask to hidden_dim
        mask_f = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_f
        right = right * mask_f

    # gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # triangle multiplication (batch GEMM)
    B, N, N, H = left.shape
    # reshape for batched GEMM
    left_ = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_ = right.permute(0, 3, 2, 1).reshape(B * H, N, N)  # transposed k dimension
    out = torch.bmm(left_, right_)  # shape (B*H, N, N)
    out = out.reshape(B, H, N, N).permute(0, 2, 3, 1)  # B,N,N,H

    # to_out_norm (LayerNorm over hidden_dim)
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # multiply with out_gate
    out = out * out_gate

    # final linear projection to dim
    out = F.linear(out, weights["to_out.weight"])

    return out
```

"""
TriMul (outgoing) forward pass with a custom Triton LayerNorm kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B,N,N,C]) using a
   Triton kernel that computes mean/variance per (b,i,j) slice and
   applies the learned scale & bias.
2. Linear projections (no bias) for the “left”, “right”, and three gates.
3. Optional pair‑wise mask (broadcast over the hidden dimension).
4. Gate the left and right tensors via a sigmoid.
5. Triangle‑multiplication (outgoing):
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is implemented as a very large batch of independent GEMMs:
      (B*H, N, N) @ (B*H, N, N).T   where H = hidden_dim.
   The GEMM is performed with `torch.bmm` (cuBLAS) which is highly
   optimized on H100.  Mixed‑precision (FP16) is used for the heavy
   multiplication to improve throughput while the final result is
   kept in FP32.
6. Apply a second LayerNorm (hidden dimension) with learnable scale &
   bias (`to_out_norm`), multiply by the output‑gate, and project back
   to the original channel dimension with a final linear layer.
7. Return the tensor of shape [B,N,N,C].

Only the first LayerNorm is written in Triton; the remaining
operations rely on PyTorch (torch.nn.functional) and fast cuBLAS
kernels.  This satisfies the requirement of “at least part of the
operations in a kernel” while keeping the implementation clear and
correct for all test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------------------
# Triton kernel: LayerNorm over the last dimension.
# ------------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,                # input, output, weight, bias pointers
    dim, eps,                                   # scalar arguments
    stride_x_b, stride_x_i, stride_x_j, stride_x_d,   # strides of x (input)
    stride_y_b, stride_y_i, stride_y_j, stride_y_d,   # strides of y (output)
    stride_w, stride_b,                         # strides of weight / bias (usually 1)
    seq_len,                                    # N (sequence length)
    BLOCK_D: tl.constexpr,                      # thread‑block size over the feature dimension
):
    """
    Compute LayerNorm for each element of shape (B, N, N, dim).

    Each program instance corresponds to a single (b, i, j) slice.
    """
    pid = tl.program_id(0)

    # -------------------------------------------------
    # Decode (b, i, j) from the linear program id.
    # -------------------------------------------------
    b = pid // (seq_len * seq_len)
    rem = pid % (seq_len * seq_len)
    i = rem // seq_len
    j = rem % seq_len

    # Base pointer for this slice.
    x_offset = b * stride_x_b + i * stride_x_i + j * stride_x_j
    y_offset = b * stride_y_b + i * stride_y_i + j * stride_y_j

    # -------------------------------------------------
    # First pass: compute mean and variance.
    # -------------------------------------------------
    sum_val = tl.zeros([1], dtype=tl.float32)
    sum_sq  = tl.zeros([1], dtype=tl.float32)

    offset = 0
    while offset < dim:
        col = tl.arange(0, BLOCK_D)
        cur = offset + col
        mask = cur < dim

        x = tl.load(x_ptr + x_offset + cur * stride_x_d, mask=mask, other=0.0)
        sum_val += tl.sum(x, axis=0)          # reduction over the block
        sum_sq  += tl.sum(x * x, axis=0)
        offset += BLOCK_D

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean

    # -------------------------------------------------
    # Second pass: write normalized output.
    # -------------------------------------------------
    offset = 0
    while offset < dim:
        col = tl.arange(0, BLOCK_D)
        cur = offset + col
        mask = cur < dim

        x = tl.load(x_ptr + x_offset + cur * stride_x_d, mask=mask, other=0.0)
        w = tl.load(w_ptr + cur * stride_w, mask=mask, other=0.0)
        b_ = tl.load(b_ptr + cur * stride_b, mask=mask, other=0.0)

        x_hat = (x - mean) * tl.rsqrt(var + eps)
        y = x_hat * w + b_
        tl.store(y_ptr + y_offset + cur * stride_y_d, y, mask=mask)

        offset += BLOCK_D


def layer_norm_triton(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Wrapper around ``layernorm_kernel``.
    Arguments:
        x      – input tensor of shape (B, N, N, C) (float32)
        weight – LayerNorm weight of shape (C,)
        bias   – LayerNorm bias   of shape (C,)
    Returns:
        Normalized tensor of the same shape as ``x``.
    """
    assert x.is_cuda and weight.is_cuda and bias.is_cuda
    B, N, _, C = x.shape
    out = torch.empty_like(x)

    # Strides in element units (not bytes)
    sx = x.stride()
    sy = out.stride()

    # launch one program per (b,i,j) slice
    grid = (B * N * N,)

    # BLOCK_D must be a compile‑time constant; we pick 128 (covers C≤384)
    layernorm_kernel[grid](
        x, out,
        weight, bias,
        C, eps,
        sx[0], sx[1], sx[2], sx[3],
        sy[0], sy[1], sy[2], sy[3],
        weight.stride(0), bias.stride(0),
        N,
        BLOCK_D=128,
    )
    return out


# ------------------------------------------------------------------------
# Entry point – the custom kernel used by the evaluation harness.
# ------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the “outgoing” Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights      : dict of weight tensors (no bias on linear layers)
        - config       : dict with keys "dim" and "hidden_dim" and "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------------
    # 1) LayerNorm (handled by a Triton kernel)
    # --------------------------------------------------------------------
    x_norm = layer_norm_triton(
        x,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
    )  # shape [B, N, N, dim]

    # --------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # --------------------------------------------------------------------
    # 3) Optional pair mask (broadcast over hidden dim)
    # --------------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)            # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------------
    # 4) Gates (sigmoid of linear projections)
    # --------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------
    # 5) Triangle multiplication (outgoing)
    #    Compute for each head d:  out[d] = left[d] @ right[d].T
    #    Use mixed precision (FP16) for the heavy GEMM.
    # --------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Cast to FP16 for the GEMM – this yields large speed‑ups on H100.
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Reshape to (B*H, N, N) where each head is an independent matrix.
    left_mat = left_fp16.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H, N, N)
    right_mat = right_fp16.permute(0, 3, 2, 1).reshape(B * H, N, N) # (B*H, N, N)

    # Batched matrix multiplication.  CuBLAS (or CUTLASS) will fuse this efficiently.
    out_fp16 = torch.bmm(left_mat, right_mat)   # (B*H, N, N)

    # Back to the original layout: (B, N, N, H)
    out = out_fp16.reshape(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------------
    # 6) Second LayerNorm over hidden dimension
    # --------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------------
    # 7) Output gate and final linear projection back to dim
    # --------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # shape [B, N, N, dim]

    return out
