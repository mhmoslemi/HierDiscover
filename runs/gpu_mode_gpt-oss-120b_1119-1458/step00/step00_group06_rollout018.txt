def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # input shape (B,N,N,dim)
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # LayerNorm
    x = x = (input_tensor - torch.mean(input_tensor, dim=-1, keepdim=True)) / torch.sqrt(torch.var(input_tensor, dim=-1, unbiased=False, keepdim=True) + 1e-5)
    # but we need weight and bias.
    # Better: use torch.nn.functional.layer_norm with weight and bias.
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)
    
    # Linear projections: left_proj = x @ left_proj_weight.T
    left = torch.nn.functional.linear(x, left_proj_weight)  # bias None
    right = torch.nn.functional.linear(x, right_proj_weight)
    
    # Apply mask (if not nomask)
    if mask is not None:
        mask_expanded = mask.unsqueeze(-1)  # shape (B,N,N,1)
        left = left * mask_expanded
        right = right * mask_expanded
    
    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()
    
    left = left * left_gate
    right = right * right_gate
    
    # Now compute einsum using Triton
    # Rearrange left and right for Triton
    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # shape B,H,N,N
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    
    # Flatten batch and hidden dim
    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)
    
    # Create output tensor
    out_flat = torch.empty_like(left_flat)
    
    # Compute strides:
    def get_strides(t):
        # tens: shape (BH,N,N). Strides in number of elements
        s0, s1, s2 = t.stride()
        return s0, s1, s2
    stride_lm, stride_ln, stride_lk = get_strides(left_flat)
    stride_rm, stride_rn, stride_rk = get_strides(right_flat)
    stride_om, stride_on, stride_ok = get_strides(out_flat)

    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (BH,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))
    
    triton_tri_mul_kernel[grid](
        left_flat, right_flat, out_flat,
        BH, N,
        stride_lm, stride_ln, stride_lk,
        stride_rm, stride_rn, stride_rk,
        stride_om, stride_on, stride_ok,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    
    # Reshape out back to (B,N,N,hidden_dim)
    out_perm = out_flat.view(B, hidden_dim, N, N)  # (B,H,N,N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

import torch
import triton
import triton.language as tl
import math

@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N,
    stride_lm, stride_ln, stride_lk,
    stride_rm, stride_rn, stride_rk,
    stride_om, stride_on, stride_ok,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b, i, j] = sum_k left[b, i, k] * right[b, k, j] for each batch b (including hidden dim).
    left and right are assumed to be of shape (M, N, N) where M = B * hidden_dim.
    """
    pid_batch = tl.program_id(0)   # batch*hidden index
    pid_i = tl.program_id(1)      # block row index
    pid_j = tl.program_id(2)      # block col index

    # Offsets for the output tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block: shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_ptr + pid_batch * stride_lm

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel for
    the expensive N^3 reduction.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm on the input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if required
    if not nomask:
        # mask shape: [B, N, N] -> broadcast to hidden_dim
        mask_exp = mask.unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # Gating values
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Make tensors contiguous for Triton
    left = left.contiguous()
    right = right.contiguous()

    # Permute to (B, hidden_dim, N, N) and flatten batch*hidden for the kernel
    B, N, _, _ = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch and hidden dimensions
    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)

    # Allocate output buffer
    out_flat = torch.empty_like(left_flat)

    # Strides (in elements)
    stride_lm, stride_ln, stride_lk = left_flat.stride()
    stride_rm, stride_rn, stride_rk = right_flat.stride()
    stride_om, stride_on, stride_ok = out_flat.stride()

    # Triton block sizes (tuned for H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    triton_tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        BH,
        N,
        stride_lm,
        stride_ln,
        stride_lk,
        stride_rm,
        stride_rn,
        stride_rk,
        stride_om,
        stride_on,
        stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Final hidden-to-output LayerNorm and linear projection
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

The heavy N³ reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched matrix multiply (Gram product) using Triton.
All other operations (layer‑norms, linear projections, gating, final linear)
are performed with PyTorch for simplicity.
"""

import math
import torch
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel: batched Gram matrix (left @ rightᵀ) over the
# sequence dimension.  Input tensors have shape (M, N, N) where
# M = batch_size * hidden_dim.  The kernel computes
#   out[m, i, j] = Σ_k left[m,i,k] * right[m,k,j]
# -------------------------------------------------------------
@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N,
    stride_lm, stride_ln, stride_lk,
    stride_rm, stride_rn, stride_rk,
    stride_om, stride_on, stride_ok,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)   # batch*hidden index
    pid_i = tl.program_id(1)       # tile row
    pid_j = tl.program_id(2)       # tile col

    # -------------------- tile offsets --------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 (more stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- reduction over k --------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_ptr + pid_batch * stride_lm
                     + offs_i[:, None] * stride_ln
                     + offs_k[None, :] * stride_lk,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )  # dtype matches input (fp16)

        # right block transposed: shape (BLOCK_K, BLOCK_N)
        # we need right[k, j] => stride_rk for k, stride_rn for j
        right = tl.load(
            right_ptr + pid_batch * stride_rm
                      + offs_k[:, None] * stride_rk
                      + offs_j[None, :] * stride_rn,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )  # dtype matches input (fp16)

        # dot produces fp16, cast to fp32 before accumulation
        acc += tl.dot(left, right).to(tl.float32)

    # -------------------- store result --------------------
    tl.store(
        out_ptr + pid_batch * stride_om
                + offs_i[:, None] * stride_on
                + offs_j[None, :] * stride_ok,
        acc,
        mask=mask_i[:, None] & mask_j[None, :]
    )

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [batch, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast on the hidden dimension)
    # -----------------------------------------------------------------
    if not nomask:
        m = mask.unsqueeze(-1)               # [B, N, N, 1]
        left = left * m
        right = right * m

    # -----------------------------------------------------------------
    # 4. Gating values (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # -----------------------------------------------------------------
    # 5. Apply gates
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Prepare tensors for the Triton kernel
    #    Shape transformation: (B, N, N, H) -> (B*H, N, N)
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    left = left.contiguous()
    right = right.contiguous()

    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Cast to fp16 for higher throughput; Triton kernel will accumulate in fp32
    left_half = left_perm.to(torch.float16)
    right_half = right_perm.to(torch.float16)

    BH = B * hidden_dim
    left_flat = left_half.view(BH, N, N)
    right_flat = right_half.view(BH, N, N)

    out_flat = torch.empty_like(left_flat)             # fp16 buffer

    # -----------------------------------------------------------------
    # 7. Launch Triton kernel
    # -----------------------------------------------------------------
    stride_lm, stride_ln, stride_lk = left_flat.stride()
    stride_rm, stride_rn, stride_rk = right_flat.stride()
    stride_om, stride_on, stride_ok = out_flat.stride()

    # Tuned block sizes for H100 (feel free to modify)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    triton_tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        BH,
        N,
        stride_lm,
        stride_ln,
        stride_lk,
        stride_rm,
        stride_rn,
        stride_rk,
        stride_om,
        stride_on,
        stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 8. Reshape back to (B, N, N, hidden_dim) and cast to fp32
    # -----------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # -----------------------------------------------------------------
    # 9. Hidden → output LayerNorm, gating and final linear
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
