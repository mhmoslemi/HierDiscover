def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, D = input_tensor.shape
    H = config["hidden_dim"]
    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (D,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Linear projections
    # Cast to half for speed if possible
    if x.dtype != torch.float16:
        x_half = x.half()
    else:
        x_half = x
    left_proj_weight = weights["left_proj.weight"].half()
    right_proj_weight = weights["right_proj.weight"].half()
    left = torch.nn.functional.linear(x_half, left_proj_weight) # shape [B,N,N,H]
    right = torch.nn.functional.linear(x_half, right_proj_weight)
    # mask
    if not config.get("nomask", True):
        mask = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask
    # Gates (compute in half)
    left_gate_weight = weights["left_gate.weight"].half()
    right_gate_weight = weights["right_gate.weight"].half()
    out_gate_weight = weights["out_gate.weight"].half()
    left_gate = torch.nn.functional.linear(x_half, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_half, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_half, out_gate_weight).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # core matmul compute out = left @ right.T per head using Triton
    # left,right dtype half
    out = trimul_outgoing_kernel_launcher(left, right)  # returns half
    # out is [B,N,N,H]

    # to_out_norm (layernorm across H)
    # convert out to float32 perhaps for stability
    out_fp32 = out.float()
    # Apply layernorm across hidden dimension (i.e., last dim = H)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_normed = torch.nn.functional.layer_norm(out_fp32, (H,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # multiply with out_gate (gate is half or maybe float16)
    out_gate_fp32 = out_gate.float()
    out_normed = out_normed * out_gate_fp32

    # final linear to original dim
    to_out_weight = weights["to_out.weight"]  # shape (dim, H)
    output = torch.nn.functional.linear(out_normed, to_out_weight)  # shape [B,N,N,dim]

    return output

@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # compute program ID mapping
    # Assume grid = (num_blocks_i * num_blocks_j * B * H)
    # pid = tl.program_id(0)
    # compute batch_head index: batch_head_idx = pid // (num_blocks_i * num_blocks_j)
    # and i and j ids.
    # For each batch_head_idx, get b = batch_head_idx // H, h = batch_head_idx % H
    num_i_blocks = tl.cdiv(N, BLOCK_M)
    num_j_blocks = tl.cdiv(N, BLOCK_N)
    total_blocks_per_batchh = num_i_blocks * num_j_blocks
    pid = tl.program_id(0)
    batch_head_idx = pid // total_blocks_per_batchh
    i_j_idx = pid % total_blocks_per_batchh
    i_block_id = i_j_idx // num_j_blocks
    j_block_id = i_j_idx % num_j_blocks

    # compute coordinate offsets
    off_i = i_block_id * BLOCK_M + tl.arange(0, BLOCK_M)
    off_j = j_block_id * BLOCK_N + tl.arange(0, BLOCK_N)

    # create mask to handle tail
    mask_i = off_i < N
    mask_j = off_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # use fp32 accumulation

    # Loop over K blocks
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_blocks):
        off_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = off_k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        # left offset: b*stride_l_b + h*stride_l_h + off_i[:, None]*stride_l_i + off_k[None, :]*stride_l_k
        # pointer arithmetic: left_ptr + off_i[:, None]*stride_l_i + off_k[None, :]*stride_l_k + b*stride_l_b + h*stride_l_h

        # To compute pointer:
        # We'll compute base pointer off for each batch-head outside loops.
        # But we need to get b and h values:
        b = batch_head_idx // H
        h = batch_head_idx % H

        # compute offsets
        # L base = left_ptr + b*stride_l_b + h*stride_l_h
        # L_ptr = L_base + off_i[:,None] * stride_l_i + off_k[None,:] * stride_l_k

        # Load left
        l_ptrs = left_ptr + b*stride_l_b + h*stride_l_h + off_i[:, None] * stride_l_i + off_k[None,:] * stride_l_k
        L = tl.load(l_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile for transposed B: we need B[k,j] i.e., right[b, j, k, h]
        # So offsets: right ptr + b*stride_r_b + h*stride_r_h + off_k[:, None]*stride_r_k + off_j[None,:]*stride_r_i
        # Note: stride_r_i corresponds to first N axis (=j), stride_r_k second axis (k)

        r_ptrs = right_ptr + b*stride_r_b + h*stride_r_h + off_k[:, None] * stride_r_k + off_j[None, :] * stride_r_i
        R = tl.load(r_ptrs, mask=mask_k[:, None] & mask_j[None,:], other=0.0)

        # multiply-accumulate
        # Here both L and R are half (float16) but accumulate in float32. Convert them to float16 then cast.
        # tl.dot expects same dtype floats? dot uses multiplication on float16->float32?
        # Use tl.dot directly: out addition with dot inside loops? For block matmul we can use tl.dot(L, R)
        # L shape (BLOCK_M, BLOCK_K), R shape (BLOCK_K, BLOCK_N)
        acc += tl.dot(L, R)

    # Write to output
    out_ptrs = out_ptr + b*stride_o_b + h*stride_o_h + off_i[:, None] * stride_o_i + off_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None,:])

def trimul_outgoing_kernel_launcher(left, right):
    # left and right are torch tensors on cuda, dtype half.
    B, N, _, H = left.shape
    out = torch.empty((B,N,N,H), dtype=left.dtype, device=left.device)
    # compute strides
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()  # out same layout
    # define block size, we might tune based on N.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32   # Must be power-of-two for performance

    # Compute grid dimensions:
    num_i_blocks = (N + BLOCK_M - 1) // BLOCK_M
    num_j_blocks = (N + BLOCK_N - 1) // BLOCK_N
    total = B * H * num_i_blocks * num_j_blocks
    grid = (total, )

    trimul_matmul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out

"""
TriMul (outgoing) – fast Triton implementation
================================================

This kernel implements the forward pass of the “outgoing” TriMul operator
used in AlphaFold‑3 and related protein‑structure models.

Algorithm
---------
1. Layer‑norm over the last dimension of the input tensor ``x``.
2. Two linear projections (left / right) from ``dim`` → ``hidden_dim``.
   An optional mask (shape [B,N,N]) is applied after the projection.
3. Three gating linear layers (left, right, out) with a ``sigmoid``
   non‑linearity. The left/right gates multiply the projected tensors.
4. Core tensor contraction (the “tri‑multiplicative” update):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each head ``h``.
   It is realized with a custom Triton kernel that:
   * processes each head (and batch) independently,
   * uses FP16 for input tensors,
   * accumulates in FP32 for numerical stability,
   * tiles the computation with BLOCK_M × BLOCK_N × BLOCK_K.
5. Layer‑norm over the hidden dimension (``hidden_dim``) and an output
   gate (element‑wise sigmoid from a linear layer) are applied.
6. Final linear projection back to the original dimension ``dim``.

All heavy work (step 4) is performed on the GPU with Triton; the remaining
operations use PyTorch’s highly‑optimised kernels. The result is a
float‑32 tensor of shape ``[B, N, N, dim]`` matching the reference
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul left @ rightᵀ for every head
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    # Pointers
    left_ptr, right_ptr, out_ptr,
    # Sizes
    B, N, H,
    # Strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    # Compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out = left @ rightᵀ   (shape [B,N,N,H])
    left  : [B, N, N, H]   (i,k)
    right : [B, N, N, H]   (j,k)    → we read it transposed as (k,j)
    out   : [B, N, N, H]   (i,j)
    """

    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # 1) Map program id to (batch, head, i‑block, j‑block)
    # ------------------------------------------------------------------
    num_i_blocks = tl.cdiv(N, BLOCK_M)
    num_j_blocks = tl.cdiv(N, BLOCK_N)
    blocks_per_head = num_i_blocks * num_j_blocks

    batch_head_idx = pid // blocks_per_head            # 0 … B*H-1
    block_idx      = pid % blocks_per_head

    i_block = block_idx // num_j_blocks
    j_block = block_idx %  num_j_blocks

    b = batch_head_idx // H          # batch index
    h = batch_head_idx %  H          # head index

    # ------------------------------------------------------------------
    # 2) Offsets for the current tile
    # ------------------------------------------------------------------
    offs_i = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # 3) Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 4) Loop over K dimension
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_blocks):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < N

        # ----------- left tile (M × K) ---------------------------------
        # left[b, i, k, h]  → stride: (b, i, k, h)
        left_ptrs = (
            left_ptr
            + b * stride_l_b + h * stride_l_h
            + offs_i[:, None] * stride_l_i
            + cur_k[None, :] * stride_l_k
        )
        L = tl.load(
            left_ptrs,
            mask=(mask_i[:, None] & mask_k[None, :]),
            other=0.0,
        )  # FP16

        # ----------- right tile (K × N)  (transposed) -------------------
        # rightᵀ[k, j] = right[b, j, k, h]  → stride: (b, j, k, h)
        right_ptrs = (
            right_ptr
            + b * stride_r_b + h * stride_r_h
            + cur_k[:, None] * stride_r_k   # k dimension first
            + offs_j[None, :] * stride_r_i   # j dimension second
        )
        R = tl.load(
            right_ptrs,
            mask=(mask_k[:, None] & mask_j[None, :]),
            other=0.0,
        )  # FP16

        # ----------- multiply‑accumulate --------------------------------
        acc += tl.dot(L, R)  # FP16 dot added into FP32 accumulator

    # ------------------------------------------------------------------
    # 5) Write result (cast back to FP16)
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_o_b + h * stride_o_h
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=(mask_i[:, None] & mask_j[None, :]),
    )


def trimul_matmul_launcher(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Launches the Triton kernel for the outgoing TriMul contraction.
    Both ``left`` and ``right`` must be CUDA tensors of dtype ``float16``
    with shape ``[B, N, N, H]``.
    Returns a ``float16`` tensor of the same shape.
    """
    assert left.is_cuda and right.is_cuda
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, _, H = left.shape

    # Output buffer
    out = torch.empty_like(left)

    # Strides in *elements* (not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tile sizes – chosen to match NVIDIA tensor‑core shape (16×16×16)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: one program per (batch, head, i‑block, j‑block)
    num_i_blocks = (N + BLOCK_M - 1) // BLOCK_M
    num_j_blocks = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = B * H * num_i_blocks * num_j_blocks
    grid = (total_blocks,)

    trimul_matmul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
              - input_tensor : torch.Tensor [B, N, N, dim]
              - mask_tensor  : torch.Tensor [B, N, N] (may be None)
              - weights_dict : dict of model parameters
              - config_dict  : dict with keys ``dim``, ``hidden_dim`` and ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (dtype float32)
    """
    # ------------------------------------------------------------------
    # 0) Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    no_mask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input layer‑norm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for the heavy linear projections
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # Linear weights (FP16)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Project to hidden dimension
    # ------------------------------------------------------------------
    left = F.linear(x_h, left_proj_w)   # [B,N,N,hidden_dim]  FP16
    right = F.linear(x_h, right_proj_w)  # [B,N,N,hidden_dim]  FP16

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if not no_mask:
        # mask: [B,N,N] → [B,N,N,1] → FP16
        mask_f = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5) Gating (linear + sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B,N,N,hidden_dim] FP16
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))    # used later, FP16

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core contraction: batched matmul using Triton
    # ------------------------------------------------------------------
    # left/right are now [B,N,N,hidden_dim] FP16
    out_hidden = trimul_matmul_launcher(left, right)  # FP16

    # ------------------------------------------------------------------
    # 7) Convert to FP32 for the final two layers
    # ------------------------------------------------------------------
    out_hidden_fp32 = out_hidden.float()                     # [B,N,N,hidden_dim]
    out_gate_fp32 = out_gate.float()

    # Output layer‑norm over hidden dimension
    out_normed = F.layer_norm(
        out_hidden_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply output gate
    out_normed = out_normed * out_gate_fp32

    # ------------------------------------------------------------------
    # 8) Final linear projection back to original dimension
    # ------------------------------------------------------------------
    output = F.linear(out_normed, weights["to_out.weight"])   # [B,N,N,dim]  float32

    return output
