@triton.jit
def batch_gemm_kernel(A_ptr, B_ptr, C_ptr,
                      B, N, H,
                      stride_Ab, stride_Ai, stride_Ak, stride_Ah,
                      stride_Bb, stride_Bj, stride_Bk, stride_Bh,
                      stride_Cb, stride_Ci, stride_Cj, stride_Ch,
                      BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_bh = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    b = pid_bh // H
    h = pid_bh % H

    # compute offsets for i,j
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # offsets for k loop
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offset range
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointer address for A: shape (M, K)
        A_ptrs = A_ptr + b * stride_Ab + h * stride_Ah + offs_i[:, None] * stride_Ai + offs_k[None, :] * stride_Ak
        # compute pointer address for B: shape (K, N)
        B_ptrs = B_ptr + b * stride_Bb + h * stride_Bh + offs_k[:, None] * stride_Bk + offs_j[None, :] * stride_Bj

        # load - use zeros for out-of-range positions
        a = tl.load(A_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)  # shape (M, K)
        b_ = tl.load(B_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)  # shape (K, N)

        acc += tl.dot(a, b_)
    # end for

    # write result
    C_ptrs = C_ptr + b * stride_Cb + h * stride_Ch + offs_i[:, None] * stride_Ci + offs_j[None, :] * stride_Cj
    tl.store(C_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # shapes
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config['hidden_dim']
    # Cast weight to input dtype
    dtype = input_tensor.dtype
    device = input_tensor.device
    # get weight tensors and cast
    def cast(t): return t.to(dtype=dtype, device=device)
    norm_weight = cast(weights['norm.weight'])
    norm_bias = cast(weights['norm.bias'])
    left_proj_weight = cast(weights['left_proj.weight'])
    right_proj_weight = cast(weights['right_proj.weight'])
    left_gate_weight = cast(weights['left_gate.weight'])
    right_gate_weight = cast(weights['right_gate.weight'])
    out_gate_weight = cast(weights['out_gate.weight'])
    to_out_norm_weight = cast(weights['to_out_norm.weight'])
    to_out_norm_bias = cast(weights['to_out_norm.bias'])
    to_out_weight = cast(weights['to_out.weight'])
    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)
    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)   # (B,N,N,hidden)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Mask handling
    if not config.get('nomask', True):
        mask = mask.to(dtype=dtype).unsqueeze(-1)  # shape (B,N,N,1)
        left = left * mask
        right = right * mask

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous for kernel
    left = left.contiguous()
    right = right.contiguous()
    # Compute outer product via Triton kernel
    out = triton_matmul(left, right, N=N, H=hidden_dim)  # pass N, hidden_dim automatically inferred.

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)
    # Gate
    out = out * out_gate
    # Final linear
    output = torch.nn.functional.linear(out, to_out_weight)

    return output

def triton_matmul(left, right):
    assert left.shape == right.shape
    B, N, _, H = left.shape
    out = torch.empty_like(left)
    # fetch strides
    stride_Ab, stride_Ai, stride_Ak, stride_Ah = left.stride()
    stride_Bb, stride_Bj, stride_Bk, stride_Bh = right.stride()
    stride_Cb, stride_Ci, stride_Cj, stride_Ch = out.stride()
    # Choose block sizes
    if N >= 512:
        BLOCK_M = 128
        BLOCK_N = 128
    else:
        BLOCK_M = 64
        BLOCK_N = 64
    BLOCK_K = 32   # fixed
    grid = ( (N + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N, B*H )

        stride_Cb, stride_Ci, stride_Cj, stride_Ch,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

@triton.jit
def batch_matmul_kernel(A_ptr, B_ptr, C_ptr,
                        B, N, H,
                        stride_Ab, stride_Ai, stride_Ak, stride_Ah,
                        stride_Bb, stride_Bj, stride_Bk, stride_Bh,
                        stride_Cb, stride_Ci, stride_Cj, stride_Ch,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_bh = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    b = pid_bh // H
    h = pid_bh % H

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        A_ptrs = A_ptr + b*stride_Ab + h*stride_Ah + offs_i[:, None] * stride_Ai + offs_k[None, :] * stride_Ak
        B_ptrs = B_ptr + b*stride_Bb + h*stride_Bh + offs_k[:, None] * stride_Bk + offs_j[None, :] * stride_Bj

        a = tl.load(A_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b_ = tl.load(B_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # ensure a,b_ are float32 for accumulation
        a = a.to(tl.float32)
        b_ = b_.to(tl.float32)

        acc += tl.dot(a, b_)

    C_ptrs = C_ptr + b*stride_Cb + h*stride_Ch + offs_i[:, None] * stride_Ci + offs_j[None, :] * stride_Cj

"""
TriMul outgoing operator implemented with a Triton batched GEMM kernel.

The heavy contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is a batch‑wise matrix multiplication per hidden channel (h).  
A custom Triton kernel tiles the i/j dimensions (BLOCK_M×BLOCK_N) and the
reduction dimension k (BLOCK_K), and processes each hidden channel
independently (grid dimension 2 = batch * hidden).  Layer‑norms, linear
projections, gating and masking are performed in PyTorch; the kernel
provides the O(N³·H) core that dominates runtime.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the batched matrix multiplication across the K axis
# ----------------------------------------------------------------------
@triton.jit
def batch_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    B, N, H,
    stride_Ab, stride_Ai, stride_Ak, stride_Ah,
    stride_Bb, stride_Bj, stride_Bk, stride_Bh,
    stride_Cb, stride_Ci, stride_Cj, stride_Ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C = A @ Bᵀ for each batch and hidden channel.
    A : (B, N, N, H)   – layout (b, i, k, h)
    B : (B, N, N, H)   – layout (b, j, k, h)
    C : (B, N, N, H)   – layout (b, i, j, h)
    The multiplication is performed over the ``k`` dimension:
        C[b,i,j,h] = Σ_k A[b,i,k,h] * B[b,j,k,h]
    """
    # ------------------------------------------------------------------
    # index the grid
    # ------------------------------------------------------------------
    pid_i = tl.program_id(0)   # i‑tile
    pid_j = tl.program_id(1)   # j‑tile
    pid_bh = tl.program_id(2)  # combined batch * hidden

    b = pid_bh // H            # batch index
    h = pid_bh % H             # hidden (channel) index

    # ------------------------------- offsets ---------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 irrespective of A/B dtype
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------- k‑loop ----------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_tiles):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A has shape (M, K)  --> (i, k)
        A_offset = (b * stride_Ab + h * stride_Ah
                    + offs_i[:, None] * stride_Ai
                    + offs_k[None, :] * stride_Ak)
        a = tl.load(A_ptr + A_offset,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # B has shape (K, N)  --> (k, j)
        B_offset = (b * stride_Bb + h * stride_Bh
                    + offs_k[:, None] * stride_Bk
                    + offs_j[None, :] * stride_Bj)
        b_tile = tl.load(B_ptr + B_offset,
                         mask=mask_k[:, None] & mask_j[None, :],
                         other=0.0)

        # Promote to FP32 for the dot‑product
        a = a.to(tl.float32)
        b_tile = b_tile.to(tl.float32)
        acc += tl.dot(a, b_tile)

    # ------------------------------- store ----------------------------
    C_offset = (b * stride_Cb + h * stride_Ch
                + offs_i[:, None] * stride_Ci
                + offs_j[None, :] * stride_Cj)
    tl.store(C_ptr + C_offset, acc, mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Helper that wraps the Triton kernel
# ----------------------------------------------------------------------
def triton_batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    ``left`` and ``right`` are tensors of shape (B, N, N, H).
    Returns ``out`` of the same shape where the contraction over the
    third dimension (K) is performed with a Triton kernel.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Get strides for the 4‑D tensors (batch, i, k, hidden)
    sAb, sAi, sAk, sAh = left.stride()
    sBb, sBj, sBk, sBh = right.stride()
    sCb, sCi, sCj, sCh = out.stride()

    # Choose tile sizes – larger tiles give higher throughput for
    # larger sequence lengths.
    BLOCK_M = 128 if N >= 512 else 64
    BLOCK_N = 128 if N >= 512 else 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j tiles
        B * H                           # batch × hidden
    )

    batch_gemm_kernel[grid](
        left, right, out,
        B, N, H,
        sAb, sAi, sAk, sAh,
        sBb, sBj, sBk, sBh,
        sCb, sCi, sCj, sCh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4, num_stages=3
    )
    return out

# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] – the result of the TriMul block.
    """
    torch.set_grad_enabled(False)  # the kernel does not need gradients

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Cast all weights to the same dtype / device as the input
    # ------------------------------------------------------------------
    def cast(w):
        return w.to(dtype=input_tensor.dtype, device=input_tensor.device)

    norm_weight = cast(weights["norm.weight"])
    norm_bias = cast(weights["norm.bias"])
    left_proj_weight = cast(weights["left_proj.weight"])
    right_proj_weight = cast(weights["right_proj.weight"])
    left_gate_weight = cast(weights["left_gate.weight"])
    right_gate_weight = cast(weights["right_gate.weight"])
    out_gate_weight = cast(weights["out_gate.weight"])
    to_out_norm_weight = cast(weights["to_out_norm.weight"])
    to_out_norm_bias = cast(weights["to_out_norm.bias"])
    to_out_weight = cast(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the ``dim`` axis
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,),
                     weight=norm_weight, bias=norm_bias, eps=1e-5)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)          # (B,N,N,hidden)
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Gating values (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 4) Optional mask – broadcast to the hidden dimension
    # ------------------------------------------------------------------
    if not nomask:
        # mask: (B, N, N) → (B, N, N, 1)
        mask_f = mask.to(dtype=input_tensor.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core O(N³·H) contraction performed by Triton
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = triton_batched_gemm(left, right)   # shape (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 7) Final LayerNorm + gating + linear projection back to ``dim``
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    out = out * out_gate                     # broadcasting (B,N,N,hidden)
    output = F.linear(out, to_out_weight)    # (B,N,N,dim)

    return output

    BLOCK_N = 128 if N >= 512 else 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H                           # batch × hidden
    )

    batch_gemm_kernel[grid](
        left, right, out,
        B, N, H,
        sAb, sAi, sAk, sAh,
        sBb, sBj, sBk, sBh,
        sCb, sCi, sCj, sCh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4, num_stages=3
    )
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul block.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] – the result of the TriMul module.
    """
    torch.set_grad_enabled(False)  # no gradients needed for the custom kernel

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Cast all weight tensors to the same dtype/device as the input
    # ------------------------------------------------------------------
    def cast(t):
        return t.to(dtype=input_tensor.dtype, device=input_tensor.device)

    norm_weight = cast(weights["norm.weight"])
    norm_bias = cast(weights["norm.bias"])
    left_proj_weight = cast(weights["left_proj.weight"])
    right_proj_weight = cast(weights["right_proj.weight"])
    left_gate_weight = cast(weights["left_gate.weight"])
    right_gate_weight = cast(weights["right_gate.weight"])
    out_gate_weight = cast(weights["out_gate.weight"])
    to_out_norm_weight = cast(weights["to_out_norm.weight"])
    to_out_norm_bias = cast(weights["to_out_norm.bias"])
    to_out_weight = cast(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1) Initial LayerNorm over the feature dimension ``dim``
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # (B, N, N, H)
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast to hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(dtype=input_tensor.dtype).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core contraction via Triton
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = triton_batched_gemm(left, right)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 7) Final LayerNorm, gating, and projection back to ``dim``
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                     # (B, N, N, H)
    output = F.linear(out, to_out_weight)    # (B, N, N, dim)

    return output
