```python
import triton
import triton.language as tl

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, hidden_dim,
    stride_left_i, stride_left_k, stride_left_d,
    stride_right_i, stride_right_k, stride_right_d,
    stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_d = tl.program_id(2)

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J
    d_base = pid_d * BLOCK_D

    # Range vectors
    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)
    d = d_base + tl.arange(0, BLOCK_D)

    mask_i = i < N
    mask_j = j < N
    mask_d = d < hidden_dim

    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        k_range = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # Load left tile: shape (BLOCK_I, BLOCK_K, BLOCK_D)
        left_ptrs = left_ptr + i[:, None, None] * stride_left_i \
                               + k_range[None, :, None] * stride_left_k \
                               + d[None, None, :] * stride_left_d
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                       other=0.0,
                       dtype=tl.float32)

        # Load right tile: shape (BLOCK_J, BLOCK_K, BLOCK_D)
        right_ptrs = right_ptr + j[:, None, None] * stride_right_i \
                                 + k_range[None, :, None] * stride_right_k \
                                 + d[None, None, :] * stride_right_d
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                        other=0.0,
                        dtype=tl.float32)

        # For each d slice compute dot and accumulate
        # left: (I, K, D); right: (J, K, D)
        # Use a loop over d dimension. Unroll for speed
        for dd in range(0, BLOCK_D):
            # mask for valid d
            if BLOCK_D > 1:
                # Add guard
                valid = mask_d[dd]
            else:
                valid = True
            # Compute dot only if valid
            a = left[:, :, dd]   # shape (I, K)
            b = right[:, :, dd]  # shape (J, K)
            # Use tl.dot with transposition on b
            acc[:, :, dd] += tl.dot(a, b, trans_b=True)

    # Store results
    out_ptrs = out_ptr + i[:, None, None] * stride_out_i \
                         + j[None, :, None] * stride_out_j \
                         + d[None, None, :] * stride_out_d
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])
```

#!/usr/bin/env python
"""
TriMul forward pass (outgoing version) optimized with Triton.

The heavy part of the TriMul operator is the 4‑D tensor contraction
    out[i, j, d] = Σ_k left[i, k, d] * right[j, k, d]
which is equivalent to a batch of GEMMs over the hidden dimension `d`.
The implementation below

* uses regular PyTorch for LayerNorms, linear projections and gating,
* casts the projected tensors to FP16,
* launches a Triton kernel that tiles the (i, j) output plane and a small
  block of the hidden dimension `d`.  Inside the kernel each `d`‑slice
  is multiplied with a Tensor‑Core‑accelerated `tl.dot`, accumulating in
  FP32.  The result is written back as FP32.

The kernel works for any batch size (≤2 in the test suite) and any
sequence length up to 1024.  Block sizes are chosen to give good
occupancy on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, HIDDEN,
    stride_li, stride_lk, stride_ld,          # left strides (i, k, d)
    stride_ri, stride_rk, stride_rd,          # right strides (i, k, d)
    stride_oi, stride_oj, stride_od,          # out strides (i, j, d)
    # -------------------- compile‑time constants --------------------
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Accumulate out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d].

    The kernel processes a tile of size (BLOCK_I, BLOCK_J) in the (i,j)
    plane and a sub‑block of the hidden dimension of size BLOCK_D.
    """
    pid_i = tl.program_id(0)    # tile on i‑axis
    pid_j = tl.program_id(1)    # tile on j‑axis
    pid_d = tl.program_id(2)    # tile on hidden‑dim axis

    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J
    d0 = pid_d * BLOCK_D

    i = i0 + tl.arange(0, BLOCK_I)
    j = j0 + tl.arange(0, BLOCK_J)
    d = d0 + tl.arange(0, BLOCK_D)

    mask_i = i < N
    mask_j = j < N
    mask_d = d < HIDDEN

    # accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the k dimension in blocks
    # ------------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k0 = k * BLOCK_K
        k_idx = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # ------------------- load left[i, k, d] --------------------
        left_ptrs = left_ptr \
            + i[:, None, None] * stride_li \
            + k_idx[None, :, None] * stride_lk \
            + d[None, None, :] * stride_ld
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] &
                            mask_k[None, :, None] &
                            mask_d[None, None, :],
                       other=0.0,
                       dtype=tl.float16)               # FP16 load

        # ------------------- load right[j, k, d] -------------------
        right_ptrs = right_ptr \
            + j[:, None, None] * stride_ri \
            + k_idx[None, :, None] * stride_rk \
            + d[None, None, :] * stride_rd
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None, None] &
                             mask_k[None, :, None] &
                             mask_d[None, None, :],
                        other=0.0,
                        dtype=tl.float16)               # FP16 load

        # ------------------- dot for each d slice -------------------
        # left  : (BLOCK_I, BLOCK_K, BLOCK_D)
        # right : (BLOCK_J, BLOCK_K, BLOCK_D)
        for dd in range(0, BLOCK_D):
            # Slice out a (BLOCK_I, BLOCK_K) and (BLOCK_J, BLOCK_K) matrix
            a = left[:, :, dd]          # (I, K)
            b = right[:, :, dd]         # (J, K)
            # tl.dot computes Σ_k a[i,k] * b[j,k]  (b is transposed internally)
            acc[:, :, dd] += tl.dot(a, b, trans_b=True)

    # ------------------------------------------------------------------
    # Store the result (FP32)
    # ------------------------------------------------------------------
    out_ptrs = out_ptr \
        + i[:, None, None] * stride_oi \
        + j[None, :, None] * stride_oj \
        + d[None, None, :] * stride_od
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None, None] &
                  mask_j[None, :, None] &
                  mask_d[None, None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor : torch.Tensor [B, N, N]  or None
            - weights_dict: {
                  "norm.weight": ..., "norm.bias": ...,
                  "left_proj.weight": ..., "right_proj.weight": ...,
                  "left_gate.weight": ..., "right_gate.weight": ...,
                  "out_gate.weight": ...,
                  "to_out_norm.weight": ..., "to_out_norm.bias": ...,
                  "to_out.weight": ...
              }
            - config_dict: {"dim": int, "hidden_dim": int, ...}
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = x.device
    dtype = x.dtype

    # ------------------------------------------------------------------
    # 1. LayerNorm over the input channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gating values (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if mask is None:
        mask_tensor = torch.ones_like(x[..., :1], dtype=dtype, device=device)
    else:
        mask_tensor = mask.to(dtype).unsqueeze(-1)  # shape [B, N, N, 1]

    # Apply mask and gates
    left = left * mask_tensor * left_gate
    right = right * mask_tensor * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to FP16 for the heavy contraction
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 6. Allocate output of the tri‑multiplication (float32)
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 7. Triton kernel launch parameters
    # ------------------------------------------------------------------
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32        # tile size along the reduction (k) dimension
    BLOCK_D = 8         # tile size for the hidden dimension

    grid_i = (N + BLOCK_I - 1) // BLOCK_I
    grid_j = (N + BLOCK_J - 1) // BLOCK_J
    grid_d = (hidden_dim + BLOCK_D - 1) // BLOCK_D

    # ------------------------------------------------------------------
    # 8. Run kernel per batch (B ≤ 2)
    # ------------------------------------------------------------------
    for b in range(B):
        left_b = left[b]          # shape [N, N, hidden_dim]
        right_b = right[b]        # shape [N, N, hidden_dim]
        out_b = out[b]            # shape [N, N, hidden_dim]

        # Strides are expressed in number of elements (not bytes)
        stride_li, stride_lk, stride_ld = left_b.stride()
        stride_ri, stride_rk, stride_rd = right_b.stride()
        stride_oi, stride_oj, stride_od = out_b.stride()

        trimul_kernel[(grid_i, grid_j, grid_d)](
            left_b,
            right_b,
            out_b,
            N,
            hidden_dim,
            stride_li, stride_lk, stride_ld,
            stride_ri, stride_rk, stride_rd,
            stride_oi, stride_oj, stride_od,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )

    # ------------------------------------------------------------------
    # 9. Final LayerNorm over the hidden dimension, gating and output linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                       # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])  # project back to dim

    return out
```
