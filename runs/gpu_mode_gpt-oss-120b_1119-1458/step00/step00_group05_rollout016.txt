@triton.jit
def apply_gate_and_mask_kernel(
    out_ptr, proj_ptr, gate_ptr, mask_ptr,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_d,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_d,
    stride_mask_b, stride_mask_i, stride_mask_j,
    N, H,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_D: tl.constexpr):
    
    pid = tl.program_id(0)
    batch_id = tl.program_id(1)
    
    # compute grid tile indices
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    i_tile = pid // num_j
    j_tile = pid % num_j

    i_off = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    j_off = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)
    d_off = tl.arange(0, BLOCK_D)

    # mask has no d dimension
    mask_ptrs = mask_ptr + batch_id * stride_mask_b + i_off[:, None] * stride_mask_i + j_off[None, :] * stride_mask_j
    mask = tl.load(mask_ptrs, mask=(i_off < N) & (j_off < N), other=0.0)

    # expand mask to (BLOCK_I, BLOCK_J, BLOCK_D)
    mask = mask[:, :, None]

    # load proj and gate
    proj_ptrs = proj_ptr + batch_id * stride_proj_b \
        + i_off[:, None, None] * stride_proj_i \
        + j_off[None, :, None] * stride_proj_j \
        + d_off[None, None, :] * stride_proj_d
    proj = tl.load(proj_ptrs, mask=(i_off[:, None, None] < N) & (j_off[None, :, None] < N) & (d_off[None, None, :] < H), other=0.0)

    gate_ptrs = gate_ptr + batch_id * stride_gate_b \
        + i_off[:, None, None] * stride_gate_i \
        + j_off[None, :, None] * stride_gate_j \
        + d_off[None, None, :] * stride_gate_d
    gate = tl.load(gate_ptrs, mask=(i_off[:, None, None] < N) & (j_off[None, :, None] < N) & (d_off[None, None, :] < H), other=0.0)

    # sigmoid
    gate = 1.0 / (1.0 + tl.exp(-gate))

    out = proj * gate * mask

    # store result
    out_ptrs = out_ptr + batch_id * stride_out_b \
        + i_off[:, None, None] * stride_out_i \
        + j_off[None, :, None] * stride_out_j \
        + d_off[None, None, :] * stride_out_d
    tl.store(out_ptrs, out, mask=(i_off[:, None, None] < N) & (j_off[None, :, None] < N) & (d_off[None, None, :] < H))

"""
TriMul (outgoing) – fast Triton‑fused gating & masking + cuBLAS batched GEMM.

Algorithm
---------
1. Layer‑norm the input 4‑D tensor.
2. Two linear projections produce LEFT and RIGHT tensors of shape
   (B, N, N, H) (H = hidden_dim).
3. Separate linear layers compute gate logits; a Triton kernel fuses
   *sigmoid*, *mask* (if present) and element‑wise multiplication:
        LEFT  = LEFT  * sigmoid(LEFT_GATE)  * MASK
        RIGHT = RIGHT * sigmoid(RIGHT_GATE) * MASK
4. The core O(N³·H) update   out[i,j,:] = Σₖ LEFT[i,k,:] * RIGHT[j,k,:]
   is realized as a batch of H independent matrix‑multiplications:
        LEFT  → (B·H, N, N)   (permute to (B, H, N, N))
        RIGHT → (B·H, N, N)   (permute + transpose k‑axis)
        out   = bmm(LEFT, RIGHT)   → (B·H, N, N)
   This uses the highly‑optimised cuBLAS GEMM.
5. A second Layer‑Norm (to_out_norm) is applied, followed by an
   output‑gate (sigmoid) and a final linear projection back to (dim).

The Triton kernel only handles the element‑wise part (step 3) – this is
the portion that torch.compile normally cannot fuse.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def apply_gate_and_mask_kernel(
    out_ptr, proj_ptr, gate_ptr, mask_ptr,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_d,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_d,
    stride_mask_b, stride_mask_i, stride_mask_j,
    N, H,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_D: tl.constexpr
):
    """
    out   = proj * sigmoid(gate) * mask
    Shapes: (B, N, N, H)
    """
    # program IDs: 0 -> (i_tile, j_tile), 1 -> batch
    pid = tl.program_id(0)
    batch = tl.program_id(1)

    # tile indices
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    i_tile = pid // num_j
    j_tile = pid % num_j

    # offsets inside the tile
    i_offset = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)
    d_offset = tl.arange(0, BLOCK_D)

    # masks for out‑of‑bounds
    i_mask = i_offset < N
    j_mask = j_offset < N
    d_mask = d_offset < H

    # ----- load mask (no channel dim) -----
    mask_ptrs = (mask_ptr
                 + batch * stride_mask_b
                 + i_offset[:, None] * stride_mask_i
                 + j_offset[None, :] * stride_mask_j)
    mask = tl.load(mask_ptrs, mask=i_mask[:, None] & j_mask[None, :], other=0.0)
    # broadcast to (i, j, d)
    mask = mask[:, :, None]

    # ----- load projection -----
    proj_ptrs = (proj_ptr
                 + batch * stride_proj_b
                 + i_offset[:, None, None] * stride_proj_i
                 + j_offset[None, :, None] * stride_proj_j
                 + d_offset[None, None, :] * stride_proj_d)
    proj = tl.load(proj_ptrs,
                   mask=(i_mask[:, None, None] &
                         j_mask[None, :, None] &
                         d_mask[None, None, :]),
                   other=0.0)

    # ----- load gate logits -----
    gate_ptrs = (gate_ptr
                 + batch * stride_gate_b
                 + i_offset[:, None, None] * stride_gate_i
                 + j_offset[None, :, None] * stride_gate_j
                 + d_offset[None, None, :] * stride_gate_d)
    gate = tl.load(gate_ptrs,
                   mask=(i_mask[:, None, None] &
                         j_mask[None, :, None] &
                         d_mask[None, None, :]),
                   other=0.0)

    # ----- sigmoid -----
    gate = 1.0 / (1.0 + tl.exp(-gate))

    # ----- apply mask and gate -----
    out = proj * gate * mask

    # ----- store result -----
    out_ptrs = (out_ptr
                + batch * stride_out_b
                + i_offset[:, None, None] * stride_out_i
                + j_offset[None, :, None] * stride_out_j
                + d_offset[None, None, :] * stride_out_d)
    tl.store(out_ptrs, out,
             mask=(i_mask[:, None, None] &
                   j_mask[None, :, None] &
                   d_mask[None, None, :]))


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            - input  : torch.Tensor, shape [B, N, N, dim]
            - mask   : torch.Tensor, shape [B, N, N]  (may be None)
            - weights: dict of parameter tensors
            - config : dict, must contain “dim” and “hidden_dim”.
    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------ #
    # unpack arguments
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------ #
    # fetch weights
    # ------------------------------------------------------------------ #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------ #
    # 1) LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------ #
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------ #
    # 2) Linear projections
    # ------------------------------------------------------------------ #
    left_proj = F.linear(x, left_proj_weight)          # (B,N,N,H)
    right_proj = F.linear(x, right_proj_weight)        # (B,N,N,H)

    # ------------------------------------------------------------------ #
    # 3) Gate logits (still in FP32)
    # ------------------------------------------------------------------ #
    left_gate_logits = F.linear(x, left_gate_weight)   # (B,N,N,H)
    right_gate_logits = F.linear(x, right_gate_weight) # (B,N,N,H)
    out_gate_logits = F.linear(x, out_gate_weight)     # (B,N,N,H)

    B, N, _, _ = left_proj.shape

    # ------------------------------------------------------------------ #
    # 4) Prepare mask tensor (float32, broadcasting on H)
    # ------------------------------------------------------------------ #
    if nomask or mask is None:
        mask_tensor = torch.ones((B, N, N), dtype=left_proj.dtype,
                                 device=left_proj.device)
    else:
        mask_tensor = mask.to(left_proj.dtype)

    # ------------------------------------------------------------------ #
    # 5) Apply mask + sigmoid gates using Triton (in‑place)
    # ------------------------------------------------------------------ #
    # Choose tile sizes that comfortably fit into shared memory.
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_D = 32

    # Grid: one 2‑D launch – (i_tile * j_tile, batch)
    grid = ( ( (N + BLOCK_I - 1) // BLOCK_I ) *
             ( (N + BLOCK_J - 1) // BLOCK_J ), B )

    # Apply to LEFT
    apply_gate_and_mask_kernel[
        grid,
        (BLOCK_I, BLOCK_J, BLOCK_D)
    ](
        left_proj, left_proj, left_gate_logits, mask_tensor,
        left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
        left_gate_logits.stride(0), left_gate_logits.stride(1),
        left_gate_logits.stride(2), left_gate_logits.stride(3),
        mask_tensor.stride(0), mask_tensor.stride(1), mask_tensor.stride(2),
        N, hidden_dim,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_D=BLOCK_D
    )

    # Apply to RIGHT
    apply_gate_and_mask_kernel[
        grid,
        (BLOCK_I, BLOCK_J, BLOCK_D)
    ](
        right_proj, right_proj, right_gate_logits, mask_tensor,
        right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
        right_gate_logits.stride(0), right_gate_logits.stride(1),
        right_gate_logits.stride(2), right_gate_logits.stride(3),
        mask_tensor.stride(0), mask_tensor.stride(1), mask_tensor.stride(2),
        N, hidden_dim,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_D=BLOCK_D
    )

    # ------------------------------------------------------------------ #
    # 6) Core O(N³·H) update – batched GEMM (cuBLAS)
    # ------------------------------------------------------------------ #
    # reshape to (B*H, N, N) for bmm
    left_flat = left_proj.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    # right needs the k‑axis transposed → (B*H, N, N)
    right_flat = right_proj.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)

    # batched matrix multiplication: Σₖ LEFT[i,k,:] * RIGHT[j,k,:]
    out_flat = torch.bmm(left_flat, right_flat)                  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 7) to_out_norm
    # ------------------------------------------------------------------ #
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    # ------------------------------------------------------------------ #
    # 8) output gate (sigmoid) and final linear projection
    # ------------------------------------------------------------------ #
    out_gate = torch.sigmoid(out_gate_logits)
    out = out * out_gate

    # final projection back to “dim”
    out = F.linear(out, to_out_weight)    # (B, N, N, dim)

    return out
