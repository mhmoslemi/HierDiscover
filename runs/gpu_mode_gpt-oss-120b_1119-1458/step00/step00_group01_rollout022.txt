def layer_norm(x, weight, bias, eps=1e-5):
    # x: shape [..., C]
    # weight shape [C]
    # bias shape [C]
    # compute mean and var across last dim
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    return (x - mean) / torch.sqrt(var + eps) * weight + bias

def trmul_matmul_kernel(A_ptr, B_ptr, C_ptr,
                        batch, N,
                        stride_a_batch, stride_a_i, stride_a_k,
                        stride_b_batch, stride_b_k, stride_b_j,
                        stride_c_batch, stride_c_i, stride_c_j,
                        BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    
    # Offsets for i and j within block
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    
    # Mask for bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    
    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)
    
    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N
        
        # Compute pointers for A (left): shape (BLOCK_I, BLOCK_K)
        # A[b, i, k] using strides: batch*stride_a_batch + i*stride_a_i + k*stride_a_k
        a_ptrs = A_ptr + batch_id * stride_a_batch + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        
        # Compute pointers for B_T (right transposed): we need B_T[k, j] = B[j, k]
        # Use original B strides: batch*stride_b_batch + j*stride_b_i + k*stride_b_j
        # To get B_T layout (k,j), we swap i and j. So we load using:
        # B_ptrs = B_ptr + batch_id * stride_b_batch + offs_k[:, None] * stride_b_j + offs_j[None, :] * stride_b_i
        b_ptrs = B_ptr + batch_id * stride_b_batch + offs_k[:, None] * stride_b_j + offs_j[None, :] * stride_b_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        # Now b shape (BLOCK_K, BLOCK_J)
        # Compute dot
        acc += tl.dot(a, b)
    
    # Store result
    c_ptrs = C_ptr + batch_id * stride_c_batch + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # layer norm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Compute core multiplication using Triton kernel
    B, N, _, _ = left.shape  # left shape [B,N,N,hidden_dim]

    # reshape for batched matmul per hidden_dim
    left_flat = left.permute(0,3,1,2).contiguous().view(-1, N, N)
    right_flat = right.permute(0,3,1,2).contiguous().view(-1, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    batch2 = left_flat.shape[0]

    # strides
    stride_a_batch = left_flat.stride(0)
    stride_a_i = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)

    stride_b_j = right_flat.stride(2)   # col stride (original j)

    stride_c_i = out_flat.stride(1)

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for batched matrix multiplication with B transposed.
@triton.jit
def batched_gemm_transpose_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,                    # matrix dimensions
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for a batch of matrices.

    A: (batch, M, K)
    B: (batch, N, K)  <-- stored as (batch, K, N) by swapping strides
    C: (batch, M, N)

    The kernel is launched with a 3‑D grid:
        (batch, ceil(M/BLOCK_M), ceil(N/BLOCK_N))
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # ------------------------------------------------------------------
    # Compute tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A[batch, m, k]
        a_ptrs = (A_ptr
                  + batch_id * stride_a_batch
                  + offs_m[:, None] * stride_a_m
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Bᵀ[batch, k, n] = B[batch, n, k]
        b_ptrs = (B_ptr
                  + batch_id * stride_b_batch
                  + offs_k[:, None] * stride_b_k   # row = k (original column)
                  + offs_n[None, :] * stride_b_n)  # col = n (original row)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ------------------------------------------------------------------
    # Store the result
    c_ptrs = (C_ptr
              + batch_id * stride_c_batch
              + offs_m[:, None] * stride_c_m
              + offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    TriMul “outgoing” forward pass implemented with a fused
    Triton kernel for the core pairwise multiplication.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: [B, N, N, C]      (float32)
            mask:         [B, N, N]        (bool or float)
            weights:      dict of model weights
            config:       dict containing "dim" and "hidden_dim"

    Returns:
        output tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # 3. Gating values (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # 4. Apply mask (if present)
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # 5. Combine projections with their gates
    left = left_proj * left_gate          # [B, N, N, hidden_dim]
    right = right_proj * right_gate       # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6. Core multiplicative update: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    We rewrite it as a batched matrix multiplication:
    #       for each (b, d):  out_d = left_d @ right_dᵀ
    B, N, _, _ = left.shape

    # Rearrange to (batch*hidden_dim, N, N) where rows = i, cols = k
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(-1, N, N)

    batch2 = left_flat.shape[0]   # B * hidden_dim

    # Allocate output buffer
    out_flat = torch.empty_like(left_flat, device=device)

    # Strides (in elements, not bytes)
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)   # stride for i dimension
    stride_a_k = left_flat.stride(2)   # stride for k dimension

    stride_b_batch = right_flat.stride(0)
    # For the transposed view of right we need:
    stride_b_k = right_flat.stride(2)  # original column stride becomes row stride after transpose
    stride_b_n = right_flat.stride(1)  # original row stride becomes column stride after transpose

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # Tunable tile sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (batch2,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    batched_gemm_transpose_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output layer‑norm, gate and final linear projection
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    out = out * out_gate  # gating

    # Final linear mapping back to original channel dimension
    to_out_weight = weights["to_out.weight"]
    output = F.linear(out, to_out_weight)

    return output

                  + offs_m[:, None] * stride_a_m
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Bᵀ[batch, k, n] = B[batch, n, k]
        b_ptrs = (B_ptr
                  + batch_id * stride_b_batch
                  + offs_k[:, None] * stride_b_k   # row = k (original column)
                  + offs_n[None, :] * stride_b_n)  # col = n (original row)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ------------------------------------------------------------------
    # Store the result tile
    c_ptrs = (C_ptr
              + batch_id * stride_c_batch
              + offs_m[:, None] * stride_c_m
              + offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask:         torch.Tensor [B, N, N] (bool/float) or None
            - weights:      dict with all model parameters
            - config:       dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # 3. Gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # 4. Apply optional mask
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # 5. Fuse projections with their gates
    left = left_proj * left_gate                     # [B, N, N, hidden_dim]
    right = right_proj * right_gate                  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6. Core multiplicative update via Triton GEMM
    B, N, _, _ = left.shape

    # Rearrange to (B*hidden_dim, N, N) – each slice is a square matrix.
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(-1, N, N)

    batch2 = left_flat.shape[0]       # B * hidden_dim
    out_flat = torch.empty_like(left_flat, device=device)

    # Strides (element strides, not byte strides)
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)        # i‑stride
    stride_a_k = left_flat.stride(2)        # k‑stride

    stride_b_batch = right_flat.stride(0)
    # For Bᵀ we swap the last two strides:
    stride_b_k = right_flat.stride(2)       # original column stride → row stride after transpose
    stride_b_n = right_flat.stride(1)       # original row stride    → column stride after transpose

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # Tile sizes – chosen to fit nicely in registers / shared memory.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (batch2,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    batched_gemm_transpose_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # a modest amount of warps works well on H100
    )

    # Reshape back to the original 4‑D layout
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate and final projection
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    out = out * out_gate                     # element‑wise gating

    to_out_weight = weights["to_out.weight"]  # (C, hidden_dim)
    output = F.linear(out, to_out_weight)    # back to [B, N, N, C]

    return output
