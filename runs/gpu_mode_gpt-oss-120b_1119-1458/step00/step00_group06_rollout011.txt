@triton.jit
def tri_mul_outgoing_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride, stride_ni, stride_nk,  # stride for a: batch, i, k
    stride_nj, # stride for b: batch, j, k
    N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    batch_id = pid // H
    h_id = pid % H

    # compute block row and col offsets
    block_i = tl.program_id(1) * BLOCK_M
    block_j = tl.program_id(2) * BLOCK_N

    # offsets for i and j coordinates within block
    i = block_i + tl.arange(0, BLOCK_M)
    j = block_j + tl.arange(0, BLOCK_N)

    # mask for out-of-bound
    mask_i = i < N
    mask_j = j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute pointers for A and B
    # Compute base pointers for the given batch and hidden dimension
    a_batch_ptr = a_ptr + batch_id * batch_stride + h_id
    b_batch_ptr = b_ptr + batch_id * batch_stride + h_id
    c_batch_ptr = c_ptr + batch_id * batch_stride + h_id

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        # Compute offsets for k tile
        k_idx = k_offset + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + i[:, None] * stride_ni + k_idx[None, :] * stride_nk
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B tile: shape (BLOCK_N, BLOCK_K) using j for rows
        b_ptrs = b_batch_ptr + j[:, None] * stride_ni + k_idx[None, :] * stride_nk
        b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # Compute
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_batch_ptr + i[:, None] * stride_ni + j[None, :] * stride_nj
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input pair representation (B, N, N, C).
2. Linear projections → left_proj, right_proj  (C → H) and three gates
   (left_gate, right_gate, out_gate) (C → H) followed by sigmoid.
3. Optional pair mask is applied to the projected tensors before gating.
4. Fuse the element‑wise gating:
      left  = left_proj  * left_gate
      right = right_proj * right_gate
5. Core update:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication where each hidden slice
   (H) is multiplied independently:   out_h = left_h @ right_hᵀ.
   The computation is performed by a custom Triton kernel that
   tiles the i/j dimensions (BLOCK_M × BLOCK_N) and the reduction
   dimension k (BLOCK_K).  The hidden dimension is folded into the
   batch axis, thus the kernel launches (B·H) independent GEMM
   instances.
6. Layer‑norm over the hidden dimension, apply out_gate and a final
   linear projection back to C.

The kernel works on FP32 tensors and returns FP32 output.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: batched GEMM for the outgoing TriMul update.
# -------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,                     # left  [B, N, N, H]
    b_ptr,                     # right [B, N, N, H]
    c_ptr,                     # out   [B, N, N, H]
    stride_batch,              # stride for batch dimension (B)
    stride_i,                  # stride for first N dimension (i or j)
    stride_k,                  # stride for second N dimension (k)
    stride_h,                  # stride for hidden dimension (should be 1)
    N,                         # sequence length
    hidden_dim,                # number of hidden slices (H)
    BLOCK_M: tl.constexpr,    # tile size for i
    BLOCK_N: tl.constexpr,    # tile size for j
    BLOCK_K: tl.constexpr,    # tile size for k (reduction)
):
    # -----------------------------------------------------------------
    # program ids:
    #   axis 0 : flattened batch * hidden   (pid0 = b*H + h)
    #   axis 1 : tile over i (rows)
    #   axis 2 : tile over j (cols)
    # -----------------------------------------------------------------
    pid0 = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # decode batch and hidden index
    batch_id = pid0 // hidden_dim
    h_id = pid0 % hidden_dim

    # start offsets for the current TILE
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # range vectors inside the tile
    i = i_base + tl.arange(0, BLOCK_M)
    j = j_base + tl.arange(0, BLOCK_N)

    # boundary masks (used for loading/storing)
    mask_i = i < N
    mask_j = j < N

    # -----------------------------------------------------------------
    # base pointers for this (batch, hidden) slice
    # -----------------------------------------------------------------
    a_base = a_ptr + batch_id * stride_batch + h_id * stride_h
    b_base = b_ptr + batch_id * stride_batch + h_id * stride_h
    c_base = c_ptr + batch_id * stride_batch + h_id * stride_h

    # accumulator for the tile (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # tiled reduction over k
    # -----------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tk in range(0, num_k_tiles):
        k_base = tk * BLOCK_K
        k = k_base + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ----- load a tile (i, k) -----
        a_ptrs = a_base + i[:, None] * stride_i + k[None, :] * stride_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # ----- load b tile (j, k) -----
        b_ptrs = b_base + j[:, None] * stride_i + k[None, :] * stride_k
        b = tl.load(b_ptrs,
                    mask=mask_j[:, None] & mask_k[None, :],
                    other=0.0)

        # ----- accumulate dot product over k (sum across the K dim) -----
        # tl.dot expects (M, K) and (N, K) and returns (M, N)
        acc += tl.dot(a, b)

    # -----------------------------------------------------------------
    # store the computed tile
    # -----------------------------------------------------------------
    c_ptrs = c_base + i[:, None] * stride_i + j[None, :] * stride_k
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------
# Entry point used by the test harness
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Perform the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys
            * dim          : input channel dimension C
            * hidden_dim   : hidden dimension H
            * nomask       : bool (True → ignore mask)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------- unpack ---------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype = input_tensor.dtype
    # Ensure contiguous layout for reliable stride assumptions
    x = input_tensor.contiguous()

    # ----------------------- LayerNorm -------------------------
    # norm.weight & norm.bias have shape (dim,)
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ----------------------- Projections -----------------------
    left_proj  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,H)
    right_proj = F.linear(x, weights["right_proj.weight"])

    # -------------------------- Mask ---------------------------
    if mask is not None and not nomask:
        # mask: (B,N,N) -> (B,N,N,1), broadcast to hidden dim
        mask_f = mask.to(dtype=dtype).unsqueeze(-1)
        left_proj  = left_proj * mask_f
        right_proj = right_proj * mask_f

    # -------------------------- Gates -------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates to the projected tensors
    left  = left_proj  * left_gate    # (B,N,N,H)
    right = right_proj * right_gate   # (B,N,N,H)

    # ------------------- Triton GEMM for core update ---------
    B, N, _, _ = left.shape  # left.shape == (B, N, N, H)

    # Strides for the 4‑D tensors (contiguous layout)
    stride_batch = left.stride(0)   # N * N * H
    stride_i      = left.stride(1)   # N * H
    stride_k      = left.stride(2)   # H
    stride_h      = left.stride(3)   # 1

    # Allocate output buffer
    out = torch.empty_like(left)

    # Tunable tile sizes – work well on H100 for the target shapes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                         # axis 0 : batch * hidden
        (N + BLOCK_M - 1) // BLOCK_M,           # axis 1 : i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,           # axis 2 : j‑tiles
    )

    # Launch the kernel
    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        stride_batch,
        stride_i,
        stride_k,
        stride_h,
        N,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # H100 benefits from higher warp count
    )
    # Synchronize to make sure Triton kernel has finished before
    # we continue with PyTorch ops.
    torch.cuda.synchronize()

    # -------------------- Output post‑processing -------------
    # Layer‑norm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # Apply the final gating
    out = out * out_gate

    # Final linear projection back to C
    output = F.linear(out, weights["to_out.weight"])

    return output
