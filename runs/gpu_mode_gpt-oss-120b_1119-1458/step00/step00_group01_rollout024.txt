def gated_kernel(in_ptr, gate_ptr, mask_ptr, out_ptr,
                B, N, H,
                stride_in_b, stride_in_i, stride_in_k, stride_in_h,
                stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
                stride_mask_b, stride_mask_i, stride_mask_k,
                stride_out_b, stride_out_i, stride_out_k, stride_out_h,
                BLOCK_H: tl.constexpr):
    pid0 = tl.program_id(0)  # flatten batch * hidden_block
    pid_i = tl.program_id(1)  # i index
    pid_k = tl.program_id(2)  # k index

    total_hidden_blocks = tl.cdiv(H, BLOCK_H)
    batch_idx = pid0 // total_hidden_blocks
    hid_block = pid0 % total_hidden_blocks
    # offset for hidden
    h_start = hid_block * BLOCK_H
    h_offsets = h_start + tl.arange(0, BLOCK_H)
    mask_h = h_offsets < H

    # bounds
    in_bounds = (batch_idx < B) & (pid_i < N) & (pid_k < N)

    # load mask scalar broadcast (float32)
    mask_offset = batch_idx * stride_mask_b + pid_i * stride_mask_i + pid_k * stride_mask_k
    mask_val = tl.load(mask_ptr + mask_offset, mask=in_bounds, other=0.0).to(tl.float32)

    # load input vector
    in_offset_base = batch_idx * stride_in_b + pid_i * stride_in_i + pid_k * stride_in_k
    in_ptrs = in_ptr + in_offset_base + h_offsets * stride_in_h
    inp = tl.load(in_ptrs, mask=mask_h, other=0.0).to(tl.float32)

    # load gate

    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()
    
    # mask handling: if no mask or config nomask True, create ones mask
    if mask is None or config.get("nomask", False):
        mask_tensor = torch.ones(x.shape[:3], dtype=x.dtype, device=x.device)
    else:
        mask_tensor = mask
    
    # apply mask and gates using Triton kernel
    left = gated_fwd(left_proj, left_gate, mask_tensor)    # left = left_proj * mask * left_gate
    right = gated_fwd(right_proj, right_gate, mask_tensor) # right = right_proj * mask * right_gate
    
    # Core TriMul: batched matmul across hidden dimension
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_mat = left.permute(0,3,1,2).reshape(B*H, N, N)
    right_mat = right.permute(0,3,1,2).reshape(B*H, N, N)
    # matrix multiplication with transpose on second operand
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0,2,3,1)
    
    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5
    )

    out = out * out_gate
    
    # final linear projection back to dim
    out = torch.nn.functional.linear(out, to_out_weight)  # shape B,N,N,dim
    
    return out

"""
TriMul (outgoing) kernel implementation.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension.
2. Linear projections generate “left” and “right” tensors (shape [B,N,N,H]).
3. Compute three sigmoid‑gates (left, right, output) from the normalized input.
4. Fuse mask, left‑gate and right‑gate with the corresponding projections using a
   custom Triton kernel (vectorised over the hidden dimension).
   The kernel processes one (batch, i, k) coordinate at a time and writes a
   vector of `BLOCK_H` hidden values, handling arbitrary hidden sizes by looping
   over hidden‑tiles.
5. The core multiplicative update is a batched matrix multiplication:
      out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
   By reshaping to (B*H, N, N) we can use the highly‑optimised cuBLAS GEMM
   (`torch.bmm`).  This yields the tensor `[B,N,N,H]`.
6. Apply a second Layer‑Norm on the hidden dimension, multiply by the output‑gate,
   and finally a linear projection back to the original channel dimension.

Only the gating (mask × gate) fusion is performed in Triton; the heavy
matrix‑multiply is delegated to cuBLAS, giving a good balance of correctness and
performance on an H100.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _gated_fuse_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_k, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    total_hb,                     # total hidden‑tiles per batch (runtime int)
    BLOCK_H: tl.constexpr,       # compile‑time hidden tile size
):
    """
    Computes: out = proj * gate * mask   (mask broadcasted over hidden)
    Kernel grid: (B * total_hb, N, N)
    """
    pid0 = tl.program_id(0)   # combines batch index and hidden‑tile index
    pid_i = tl.program_id(1)   # index i (or j for the right tensor)
    pid_k = tl.program_id(2)   # index k

    # Decode batch and hidden‑tile
    batch_idx = pid0 // total_hb
    hb_idx    = pid0 % total_hb
    h_start   = hb_idx * BLOCK_H

    # Offsets for the hidden vector within the tile
    h_offsets = h_start + tl.arange(0, BLOCK_H)
    mask_h    = h_offsets < H

    # Bounds for the spatial dimensions
    i_in_bounds = pid_i < N
    k_in_bounds = pid_k < N
    batch_in_bounds = batch_idx < B
    valid = i_in_bounds & k_in_bounds & batch_in_bounds

    # ----- Load scalar mask and broadcast -----
    mask_offset = batch_idx * stride_mask_b + pid_i * stride_mask_i + pid_k * stride_mask_k
    mask_val = tl.load(mask_ptr + mask_offset, mask=valid, other=0.0).to(tl.float32)

    # ----- Load projection and gate vectors -----
    proj_offset_base = batch_idx * stride_proj_b + pid_i * stride_proj_i + pid_k * stride_proj_k
    proj_ptrs = proj_ptr + proj_offset_base + h_offsets * stride_proj_h
    proj = tl.load(proj_ptrs, mask=mask_h, other=0.0).to(tl.float32)

    gate_offset_base = batch_idx * stride_gate_b + pid_i * stride_gate_i + pid_k * stride_gate_k
    gate_ptrs = gate_ptr + gate_offset_base + h_offsets * stride_gate_h
    gate = tl.load(gate_ptrs, mask=mask_h, other=0.0).to(tl.float32)

    # ----- Compute fused result -----
    out = proj * mask_val * gate

    # ----- Store -----
    out_offset_base = batch_idx * stride_out_b + pid_i * stride_out_i + pid_k * stride_out_k
    out_ptrs = out_ptr + out_offset_base + h_offsets * stride_out_h
    tl.store(out_ptrs, out, mask=mask_h)


def _apply_gated_fusion(proj, gate, mask):
    """
    Helper that launches the Triton kernel to compute
        proj * gate * mask   (mask broadcasted over the hidden axis)
    Arguments:
        proj: Tensor[B,N,N,H]  (float32)
        gate: Tensor[B,N,N,H]  (float32)
        mask: Tensor[B,N,N]    (float32, values 0 or 1)
    Returns:
        Tensor[B,N,N,H] = proj * gate * mask
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # Strides in element counts (torch uses element‑wise strides)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_out  = out.stride()
    s_mask = mask.stride()

    BLOCK_H = 64  # chosen to cover the largest hidden size with a few tiles
    total_hb = (H + BLOCK_H - 1) // BLOCK_H

    grid = (B * total_hb, N, N)   # (batch*hidden_tile, i, k)

    _gated_fuse_kernel[grid](
        proj, gate, mask, out,
        B, N, H,
        s_proj[0], s_proj[1], s_proj[2], s_proj[3],
        s_gate[0], s_gate[1], s_gate[2], s_gate[3],
        s_mask[0], s_mask[1], s_mask[2],
        s_out[0], s_out[1], s_out[2], s_out[3],
        total_hb,
        BLOCK_H=BLOCK_H,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask_tensor:  torch.Tensor of shape [B, N, N] (may be None)
            - weights:       dict of model weights
            - config:        dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # --------------------------------------------------------------------- #
    # Unpack inputs
    # --------------------------------------------------------------------- #
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------- #
    # Load model weights
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # --------------------------------------------------------------------- #
    # 1) First LayerNorm over the channel dimension
    # --------------------------------------------------------------------- #
    x_norm = torch.nn.functional.layer_norm(
        x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5
    )  # [B, N, N, dim]

    # --------------------------------------------------------------------- #
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------------- #
    left_proj  = torch.nn.functional.linear(x_norm, left_proj_weight)   # [B,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)  # [B,N,N,hidden_dim]

    # --------------------------------------------------------------------- #
    # 3) Compute gating sigmoids
    # --------------------------------------------------------------------- #
    left_gate  = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()   # [B,N,N,hidden_dim]
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()  # [B,N,N,hidden_dim]
    out_gate   = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()    # [B,N,N,hidden_dim]

    # --------------------------------------------------------------------- #
    # 4) Apply mask (if any) and fuse with gates via Triton
    # --------------------------------------------------------------------- #
    if mask is None or config.get("nomask", False):
        # Create an all‑ones mask of the correct dtype/device
        mask_tensor = torch.ones(x.shape[:3], dtype=x.dtype, device=x.device)
    else:
        mask_tensor = mask.to(x.dtype)  # ensure same dtype as the projections

    # Fuse mask, projection and gate for the left and right tensors
    left  = _apply_gated_fusion(left_proj, left_gate, mask_tensor)   # left = left_proj * left_gate * mask
    right = _apply_gated_fusion(right_proj, right_gate, mask_tensor) # right = right_proj * right_gate * mask

    # --------------------------------------------------------------------- #
    # 5) Core TriMul: batched matrix multiplication over the k‑dimension
    # --------------------------------------------------------------------- #
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N) where the inner matrix is (i,k) for left
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # out_mat[b*h, i, j] = Σ_k left_mat[b*h, i, k] * right_mat[b*h, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 6) Second LayerNorm + output gate
    # --------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5
    )
    out = out * out_gate

    # --------------------------------------------------------------------- #
    # 7) Final linear projection back to original channel dimension
    # --------------------------------------------------------------------- #
    out = torch.nn.functional.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
