    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    pid_n = tl.program_id(1)
    pid_batch = tl.program_id(2)

    # offsets

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_n = offs_n < N

        mask_k = offs_k < K

"""
TriMul (outgoing) forward kernel.

Algorithm
---------
1. Layer‑norm the input 4‑D tensor (B, N, N, C).
2. Linear projections → `left` and `right` of shape (B, N, N, H).
3. Apply optional pairwise mask and element‑wise sigmoid gates.
4. Compute the core “triangle” update:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each hidden channel.
   We fuse it in a custom Triton kernel `trilmul_gemm_kernel` that
   treats each (batch, hidden) pair as an independent GEMM of size
   N×N × N×N → N×N.
5. Layer‑norm on the hidden dimension, apply the output gate and a
   final linear projection back to the original channel size.
All heavy computation (step 4) is performed on the GPU with
Triton, while the surrounding point‑wise ops use PyTorch for
convenience.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trilmul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (all == seq_len)
    stride_am, stride_ak,              # strides for A (M‑dim, K‑dim)
    stride_bk, stride_bn,              # strides for B (K‑dim, N‑dim)
    stride_cm, stride_cn,              # strides for C (M‑dim, N‑dim)
    stride_batch_a, stride_batch_b, stride_batch_c,  # batch‑dim stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM:
        C_batch = A_batch @ B_batch   (A: M×K , B: K×N)
    Each batch corresponds to one original (batch, hidden) pair.
    All tensors are stored in row‑major order (C‑contiguous).
    """
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_b = tl.program_id(2)   # batch (original_batch * hidden_dim)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current batch
    a_batch = a_ptr + pid_b * stride_batch_a
    b_batch = b_ptr + pid_b * stride_batch_b
    c_batch = c_ptr + pid_b * stride_batch_c

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K‑dimension tiles
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M×K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B tile (K×N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Accumulate the GEMM product
        acc += tl.dot(a, b)

    # Write back the result (store as FP16 to save memory)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    TriMul forward (outgoing version).

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor of shape [B, N, N, C]
            - mask: torch.Tensor of shape [B, N, N] (may be ignored)
            - weights: dict of model parameters (torch.Tensors)
            - config: dict with keys 'dim', 'hidden_dim', 'nomask' ...

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input layer‑norm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, C]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional pairwise mask
    # -----------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] -> [B, N, N, 1] for broadcasting
        m = mask.to(input_tensor.dtype).unsqueeze(-1)
        left = left * m
        right = right * m

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate   # still float32
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core triangle update – batched GEMM on the (i,k) & (k,j) axes
    #    We cast to FP16 for bandwidth, the kernel accumulates in FP32.
    # -----------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute to obtain matrices A(i,k) and B(k,j) for each hidden channel
    # A: [B, H, N, N] with (i,k) layout
    # B: [B, H, N, N] with (k,j) layout (notice the transpose of the last two dims)
    A = left.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N]
    B = right.permute(0, 3, 2, 1).contiguous()         # [B, H, N, N]

    Bsz, H, N, _ = A.shape
    # Collapse batch & hidden into a single GEMM batch dimension
    A_flat = A.view(-1, N, N)   # [B*H, N, N]
    B_flat = B.view(-1, N, N)   # [B*H, N, N]

    # Allocate output buffer (FP16)
    C_flat = torch.empty_like(A_flat, dtype=torch.float16, device=A.device)

    # Tiling parameters – chosen for H100 (adjustable if needed)
    BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        A_flat.shape[0],                      # batch = B * hidden_dim
    )

    # Strides for the flattened tensors (row‑major)
    stride_am = N          # distance between successive rows of A (M‑dim)
    stride_ak = 1          # distance between successive columns of A (K‑dim)
    stride_bk = N          # distance between successive rows of B (K‑dim)
    stride_bn = 1          # distance between successive columns of B (N‑dim)
    stride_cm = N          # distance between successive rows of C (M‑dim)
    stride_cn = 1          # distance between successive columns of C (N‑dim)

    stride_batch_a = N * N
    stride_batch_b = N * N
    stride_batch_c = N * N

    trilmul_gemm_kernel[grid](
        A_flat,
        B_flat,
        C_flat,
        N, N, N,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        stride_batch_a,
        stride_batch_b,
        stride_batch_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H) and convert to FP32 for the remaining ops
    out = C_flat.view(Bsz, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, C]

    return out
