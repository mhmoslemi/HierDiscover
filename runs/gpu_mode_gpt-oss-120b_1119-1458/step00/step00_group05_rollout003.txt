import torch
import torch.nn.functional as F
import triton

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # LayerNorm on input
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Linear projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])
    # Apply mask if present
    if mask is not None:
        mask = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask
    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))
    left = left * left_gate
    right = right * right_gate

    # Cast to half for matmul
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, N2, H = left_h.shape
    assert N == N2
    # reshape to (B, H, N, N)
    left_permuted = left_h.permute(0, 3, 1, 2).contiguous()
    right_permuted = right_h.permute(0, 3, 1, 2).contiguous()
    # flatten batch*hidden
    BH = B * hidden_dim
    left_flat = left_permuted.reshape(BH, N, N)
    right_flat = right_permuted.reshape(BH, N, N)
    # transpose right for matmul: (BH, N, N) -> (BH, N, N) with swapped axes
    right_T = right_flat.transpose(-2, -1)  # shape (BH,N,N) with possibly noncontig strides

    # allocate output
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=left_flat.device)

    # launch kernel
    _matmul_kernel(left_flat, right_T, out_flat,
                   M=N, N=N, K=N,
                   BLOCK_M=64, BLOCK_N=64, BLOCK_K=32)

    # reshape out back to [B,N,N,H] (float32)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Apply to_out_norm and out_gate
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out = out * out_gate
    # final linear
    out = F.linear(out, weights["to_out.weight"])
    return out
```

@triton.jit
def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    a_batch_stride, a_m_stride, a_k_stride,
    b_batch_stride, b_k_stride, b_n_stride,
    c_batch_stride, c_m_stride, c_n_stride,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        a_ptrs = a_ptr + pid_batch * a_batch_stride + offs_m[:, None] * a_m_stride + offs_k[None, :] * a_k_stride
        b_ptrs = b_ptr + pid_batch * b_batch_stride + offs_k[:, None] * b_k_stride + offs_n[None, :] * b_n_stride

        # Load with boundary checks
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Promote to float32 if half
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # Write result
    c_ptrs = c_ptr + pid_batch * c_batch_stride + offs_m[:, None] * c_m_stride + offs_n[None, :] * c_n_stride
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)

"""
TriMul “outgoing” implementation with a custom Triton kernel.

The heavy part of TriMul is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched matrix multiplication
    out_d = left_d @ right_dᵀ   (for every hidden channel d)

We flatten the (batch, hidden) dimensions and run a fused
Batched‑GEMM kernel in Triton. All other operations (LayerNorm,
Linear projections, gates, final projection) are performed with
standard PyTorch primitives. The kernel works on FP32 tensors
(for exactness) but can be switched to FP16 by casting the
inputs before the launch.

The entry point must be called `custom_kernel` and accept a tuple
(input, mask, weights, config) as described in the prompt.
"""

import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) where
#   A : (batch, M, K)   – left  projected tensor
#   B : (batch, K, N)   – right projected tensor (already transposed)
#   C : (batch, M, N)   – output
# -------------------------------------------------------------
@triton.jit
def _triton_matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # problem size
    a_batch_stride, a_m_stride, a_k_stride,
    b_batch_stride, b_k_stride, b_n_stride,
    c_batch_stride, c_m_stride, c_n_stride,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program identifiers
    pid_batch = tl.program_id(0)          # batch * hidden channel
    pid_m = tl.program_id(1)              # row block
    pid_n = tl.program_id(2)              # col block

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # pointers for the current block of A and B
        a_ptrs = (a_ptr +
                  pid_batch * a_batch_stride +
                  offs_m[:, None] * a_m_stride +
                  offs_k[None, :] * a_k_stride)
        b_ptrs = (b_ptr +
                  pid_batch * b_batch_stride +
                  offs_k[:, None] * b_k_stride +
                  offs_n[None, :] * b_n_stride)

        # masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # load tiles (FP32)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # dot‑product accumulation
        acc += tl.dot(a, b)

    # write back the result
    c_ptrs = (c_ptr +
              pid_batch * c_batch_stride +
              offs_m[:, None] * c_m_stride +
              offs_n[None, :] * c_n_stride)
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)


def _batched_matmul_triton(a: torch.Tensor, b: torch.Tensor, out: torch.Tensor,
                           BLOCK_M=128, BLOCK_N=128, BLOCK_K=32):
    """
    Launch the Triton kernel for a batch of matrix multiplications.

    a : (B*H, M, K)   float32 (contiguous)
    b : (B*H, K, N)   float32 (may be a transposed view)
    out : (B*H, M, N) float32 (pre‑allocated)
    """
    BH, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "inner dimensions must match"

    grid = (BH,                      # batch * hidden
            triton.cdiv(M, BLOCK_M),  # row blocks
            triton.cdiv(N, BLOCK_N))  # col blocks

    _triton_matmul_kernel[grid](
        a, b, out,
        M, N, K,
        a.stride(0), a.stride(1), a.stride(2),
        b.stride(0), b.stride(1), b.stride(2),
        out.stride(0), out.stride(1), out.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )


# -------------------------------------------------------------
# Public entry point called by the evaluator
# -------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward (outgoing) – PyTorch + Triton implementation.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        torch.Tensor of shape [bs, seq_len, seq_len, dim]
    """
    # -----------------------------------------------------------------
    # unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]               # input/ output channel dimension
    hidden_dim = config["hidden_dim"] # projection dimension

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (normalize over the last dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) – produce [B, N, N, H]
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # → (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted over channel dim)
    # -----------------------------------------------------------------
    if mask_tensor is not None:
        # ensure dtype matches the projections
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4️⃣ Gates (sigmoid of linear layers) – shape (B,N,N,H)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Heavy pairwise multiplication via Triton (batched GEMM)
    # -----------------------------------------------------------------
    # reshape to (B, H, N, N) → flatten batch*hidden → (BH, N, N)
    B, N, _, H = left.shape  # H == hidden_dim
    # (B, H, N, N) contiguous layout
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * H
    left_flat = left_perm.view(BH, N, N)                     # (BH, N, N)
    right_flat = right_perm.view(BH, N, N)                   # (BH, N, N)
    # transpose right for A @ Bᵀ
    right_T = right_flat.transpose(-2, -1)  # shape (BH, N, N) view

    # allocate output buffer (float32) – accumulator is float32
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # launch the Triton kernel
    _batched_matmul_triton(left_flat, right_T, out_flat,
                           BLOCK_M=128, BLOCK_N=128, BLOCK_K=32)

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6️⃣ Output LayerNorm, gating and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
