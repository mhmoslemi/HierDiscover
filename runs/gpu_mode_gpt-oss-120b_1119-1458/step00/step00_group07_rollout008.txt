def custom_kernel(data):
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype
    bs, seq_len, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask (if not nomask)
    if not nomask:
        mask_unsq = mask.unsqueeze(-1).type_as(x_norm)
        left = left * mask_unsq
        right = right * mask_unsq

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Prepare for kernel
    # Permute and reshape to (B*hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = bs * hidden_dim
    left_flat = left_perm.view(BH, seq_len, seq_len)
    right_flat = right_perm.view(BH, seq_len, seq_len)

    # Output allocation
    out_flat = torch.empty_like(left_flat)

    # strides
    stride_a_batch, stride_a_i, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_i, stride_b_k = right_flat.stride()
    stride_c_batch, stride_c_i, stride_c_j = out_flat.stride()
    # Note: for left_flat shape (BH, N, N).
    # stride_i = left_flat.stride(1) = N, stride_k = left_flat.stride(2) = 1.
    # For right_flat similar.

    # define block size
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    # compute grid
    num_i = (seq_len + BLOCK_M - 1) // BLOCK_M
    num_j = (seq_len + BLOCK_N - 1) // BLOCK_N
    grid = (BH, num_i, num_j)

    # Launch kernel
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        BH,
        seq_len,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_b_batch,
        stride_b_i,
        stride_b_k,
        stride_c_batch,
        stride_c_i,
        stride_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B,N,N,hidden_dim]
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()

    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    # Apply out_gate
    out = out * out_gate

    # Final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

"""
TriMul forward (outgoing version) with a custom Triton kernel.

The heavy part of the TriMul operation is the tensor contraction
```
out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
```
where `left` and `right` have shape [B, N, N, H] (H = hidden_dim).
We fuse this N‑wise inner product with a tiled matrix‑multiply kernel.
The tensors are first flattened to shape [B*H, N, N] so that each
instance (batch*hidden) performs an ordinary matrix multiplication
`(N×N) @ (N×N).T`.  The kernel tiles the M (i) and N (j) dimensions
with a configurable block size and reduces over the K (k) dimension.
All other TriMul steps (LayerNorms, linear projections, gates,
final norm + linear) are performed with regular PyTorch ops.

The kernel works for arbitrary batch size (B∈{1,2}), sequence length
(N up to 1024) and hidden dimension (H up to 384) and respects the
optional mask (`nomask=False`).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel: out_flat[bh, i, j] = Σ_k left_flat[bh, i, k] * right_flat[bh, j, k]
# ----------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    A,               # left  [BH, N, N]  (float32)
    B,               # right [BH, N, N]  (float32)
    C,               # out   [BH, N, N]  (float32)
    BH: tl.constexpr,            # B * hidden_dim
    N: tl.constexpr,
    stride_A_batch,  # stride of dim 0 in A
    stride_A_i,      # stride of dim 1 (i) in A
    stride_A_k,      # stride of dim 2 (k) in A
    stride_B_batch,  # stride of dim 0 in B
    stride_B_i,      # stride of dim 1 (j) in B   (treated as i after transpose)
    stride_B_k,      # stride of dim 2 (k) in B
    stride_C_batch,  # stride of dim 0 in C
    stride_C_i,      # stride of dim 1 (i) in C
    stride_C_j,      # stride of dim 2 (j) in C
    BLOCK_M: tl.constexpr,      # tile size for i
    BLOCK_N: tl.constexpr,      # tile size for j
    BLOCK_K: tl.constexpr,      # tile size for k
):
    pid_batch = tl.program_id(0)   # index over BH = B*hidden_dim
    pid_i = tl.program_id(1)       # tile over i dimension
    pid_j = tl.program_id(2)       # tile over j dimension

    # -----------------------------------------------------------------
    # Compute block offsets
    # -----------------------------------------------------------------
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    i_mask = offs_i < N
    j_mask = offs_j < N

    # -----------------------------------------------------------------
    # Offsets for the batch*hidden dimension
    # -----------------------------------------------------------------
    batch_offset_A = pid_batch * stride_A_batch
    batch_offset_B = pid_batch * stride_B_batch
    batch_offset_C = pid_batch * stride_C_batch

    # -----------------------------------------------------------------
    # Accumulator (float32)
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over k dimension in tiles
    # -----------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        k_mask = offs_k < N

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = A + batch_offset_A \
                 + offs_i[:, None] * stride_A_i \
                 + offs_k[None, :] * stride_A_k
        a = tl.load(a_ptrs,
                    mask=i_mask[:, None] & k_mask[None, :],
                    other=0.0)

        # Load B block transposed: we need B_T[k, j] = B[j, k]
        # B layout is (BH, j, k).  With strides (batch, i=j, k)
        b_ptrs = B + batch_offset_B \
                 + offs_k[:, None] * stride_B_k \
                 + offs_j[None, :] * stride_B_i
        b = tl.load(b_ptrs,
                    mask=k_mask[:, None] & j_mask[None, :],
                    other=0.0)

        # Accumulate dot product over the K tile
        acc += tl.dot(a, b)

    # -----------------------------------------------------------------
    # Store the result
    # -----------------------------------------------------------------
    c_ptrs = C + batch_offset_C \
            + offs_i[:, None] * stride_C_i \
            + offs_j[None, :] * stride_C_j
    tl.store(c_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])


# -------------------------------------------------------------
# Entrypoint used by the evaluation harness
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on same device).
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    bs, seq_len, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # Helper to fetch a weight (ensure correct dtype/device)
    # -----------------------------------------------------------------
    def get(name):
        w = weights[name].to(device=device, dtype=dtype)
        return w

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=get("norm.weight"),
        bias=get("norm.bias")
    )

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x_norm, get("left_proj.weight"))
    right = F.linear(x_norm, get("right_proj.weight"))

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        # mask: [bs, seq_len, seq_len] -> [bs, seq_len, seq_len, 1]
        mask_b = mask.to(dtype=dtype).unsqueeze(-1)
        left = left * mask_b
        right = right * mask_b

    # -----------------------------------------------------------------
    # 4) Gating signals (sigmoid of linear maps)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm, get("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x_norm, get("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x_norm, get("out_gate.weight")))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Prepare data for the Triton kernel
    #    view as [BH, N, N] where BH = bs * hidden_dim
    # -----------------------------------------------------------------
    # Permute to bring hidden_dim next to batch for a simple view
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    BH = bs * hidden_dim
    left_flat  = left_perm.view(BH, seq_len, seq_len)
    right_flat = right_perm.view(BH, seq_len, seq_len)

    # Allocate output for the kernel
    out_flat = torch.empty_like(left_flat)

    # Strides for the flattened tensors (in elements, not bytes)
    stride_A_batch, stride_A_i, stride_A_k = left_flat.stride()
    stride_B_batch, stride_B_i, stride_B_k = right_flat.stride()
    stride_C_batch, stride_C_i, stride_C_j = out_flat.stride()

    # -----------------------------------------------------------------
    # 6) Launch the Triton kernel
    # -----------------------------------------------------------------
    # Tile sizes – chosen to give good occupancy on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_i = (seq_len + BLOCK_M - 1) // BLOCK_M
    num_j = (seq_len + BLOCK_N - 1) // BLOCK_N
    grid = (BH, num_i, num_j)

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        BH,
        seq_len,
        stride_A_batch,
        stride_A_i,
        stride_A_k,
        stride_B_batch,
        stride_B_i,
        stride_B_k,
        stride_C_batch,
        stride_C_i,
        stride_C_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 7) Reshape back to [B, N, N, H]
    # -----------------------------------------------------------------
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len) \
                 .permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # -----------------------------------------------------------------
    # 8) Output LayerNorm, gating, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=get("to_out_norm.weight"),
        bias=get("to_out_norm.bias")
    )
    out = out * out_gate
    out = F.linear(out, get("to_out.weight"))  # -> [B, N, N, dim]

    return out
