"""
custom_kernel: Triton‑accelerated forward pass of the “outgoing” TriMul operator.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B, N, N, C]).
2. Linear projections → left/right (shape [B, N, N, H]) and compute three
   gating tensors (sigmoid outputs, also shape [B, N, N, H]).
3. Apply optional mask and element‑wise gates to left/right.
4. Fuse the heavy contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply for each hidden channel h:
        left_(b,h)  : (N_i × N_k)   = left.permute(0,3,1,2)
        right_(b,h) : (N_k × N_j)   = right.permute(0,3,2,1)
   The kernel computes C = A @ B for each (b·h) batch using
   block‑wise GEMM (BLOCK_M/BLOCK_N = 64, BLOCK_K = 32) in FP16
   and accumulates in FP32 for precision.
5. Reshape the result back to [B, N, N, H] and cast to FP32.
6. Layer‑norm over the hidden dimension, multiply by the output gate,
   and a final linear projection back to the original channel size C.

Only the O(N³·H) contraction is implemented in Triton; all other
operations use PyTorch’s highly‑optimised kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _batch_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM for tensors of shape (batch, M, K) @ (batch, K, N).

    Each batch corresponds to one (original batch × hidden_dim) pair.
    Computation is performed in FP16 with FP32 accumulation.
    """
    pid = tl.program_id(0)

    # number of tiles per dimension
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # decompose pid into (batch, tile_m, tile_n)
    batch_idx = pid // (num_blocks_m * num_blocks_n)
    tile_rem = pid % (num_blocks_m * num_blocks_n)
    tile_m = tile_rem // num_blocks_n
    tile_n = tile_rem % num_blocks_n

    # offsets within the tile
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptr = (
            A_ptr
            + batch_idx * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + cur_k[None, :] * stride_a_k
        )
        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile: shape (BLOCK_K, BLOCK_N)
        b_ptr = (
            B_ptr
            + batch_idx * stride_b_batch
            + cur_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # GEMM on the tile
        acc += tl.dot(a, b)

    # Store result as FP16 (the rest of the model works in FP32)
    c_ptr = (
        C_ptr
        + batch_idx * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(c_ptr, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, C]   (float32)
        - mask        : torch.Tensor[B, N, N]       (bool/float)
        - weights     : dict mapping weight names to tensors
        - config      : dict containing ``dim`` and ``hidden_dim`` (and optionally ``nomask``)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----- 1. LayerNorm over last dim (C) ---------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, C]

    # ----- 2. Linear projections -------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])  # [B, N, N, H]

    # ----- 3. Optional mask -------------------------------------------------
    if not nomask and mask is not None:
        mask_ = mask.unsqueeze(-1).to(dtype)          # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ----- 4. Compute gates -------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = (left * left_gate).to(torch.float16)   # cast to FP16 for the kernel
    right = (right * right_gate).to(torch.float16)

    # ----- 5. Triton GEMM for the O(N³) contraction -------------------------
    B, N, _, _ = left.shape  # B, N, N, H

    # Re‑order tensors so that each hidden channel becomes a separate batch:
    #   left  -> (B, H, i, k)
    #   right -> (B, H, k, j)   (note the transposition of the last two axes)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]

    batch_h = B * hidden_dim

    left_flat = left_perm.view(batch_h, N, N)   # (batch_h, M=N, K=N)
    right_flat = right_perm.view(batch_h, N, N)  # (batch_h, K=N, N=N)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Strides (in elements) for the flattened tensors
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(1)  # K‑dimension stride
    stride_b_n = right_flat.stride(2)  # N‑dimension stride

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # Blocking parameters – chosen to fit H100 tensor‑core granularity
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch_h * num_blocks_m * num_blocks_n,)

    _batch_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_h,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----- 6. Reshape back to [B, N, N, H] ---------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ----- 7. Post‑processing (LayerNorm, output gate, final Linear) -------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate  # broadcasting over the hidden dimension
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
