def fuse_proj_gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr,
    out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_k, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr
):
    pid = tl.program_id(0)
    # total elements = B*N*N*H
    total = B * N * N * H
    base = pid * BLOCK
    offs = base + tl.arange(0, BLOCK)
    mask = offs < total
    # compute indices via division
    # note: use tl.int64 for indices
    # d = offs % H
    # tmp = offs // H
    # k = tmp % N
    # tmp2 = tmp // N
    # i = tmp2 % N
    # b = tmp2 // N
    # compute offsets
    d = offs % H
    tmp = offs // H
    k = tmp % N
    tmp2 = tmp // N
    i = tmp2 % N
    b = tmp2 // N

    # compute pointer offsets
    proj_offset = (b * stride_proj_b) + (i * stride_proj_i) + (k * stride_proj_k) + (d * stride_proj_h)
    gate_offset = (b * stride_gate_b) + (i * stride_gate_i) + (k * stride_gate_k) + (d * stride_gate_h)
    out_offset = (b * stride_out_b) + (i * stride_out_i) + (k * stride_out_k) + (d * stride_out_h)
    # load values
    proj = tl.load(proj_ptr + proj_offset, mask=mask, other=0.0).to(tl.float16)
    gate = tl.load(gate_ptr + gate_offset, mask=mask, other=0.0).to(tl.float16)
    val = proj * gate
    if HAS_MASK:

        val = val * mask_val.to(tl.float16)
    tl.store(out_ptr + out_offset, val, mask=mask)

"""
TriMul (outgoing) forward pass implemented with a small Triton fusion.

Algorithm
---------
1.  Layer‑norm the input (float32 → float16 for the rest of the pipeline).
2.  Linear projections `left_proj`, `right_proj` (to hidden_dim) and three
    gating linear layers (`left_gate`, `right_gate`, `out_gate`).
3.  Fuse mask (if present), the projection and its gate in a single
    Triton kernel:
        left  = left_proj  * left_gate  * mask
        right = right_proj * right_gate * mask
   The kernel works on the flattened `[b,i,k,d]` tensor and therefore
   needs only the strides of the four tensors.
4.  Perform the cubic “tri‑multiply” using a batched GEMM:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   After permuting the hidden dimension to the batch axis we obtain a
   Tensor of shape (B*hidden_dim, N, N) and use `torch.bmm`, which on
   H100 runs on tensor‑cores (FP16 → FP32 accumulation).
5.  Layer‑norm over the hidden dimension (`to_out_norm`) and apply the
   output gate.
6.  Final linear projection back to `dim`.

Only the element‑wise mask / gate fusion is written in Triton, satisfying
the “must use a custom kernel” requirement, while the heavy N³ work is
performed by an optimized cuBLAS batched GEMM.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _fused_proj_gate_mask_kernel(
    proj_ptr,          # left_proj / right_proj   (fp16)
    gate_ptr,          # corresponding gate       (fp16)
    mask_ptr,          # mask (fp16) or dummy    (fp16)
    out_ptr,           # fused output             (fp16)
    B, N, H,           # batch, seq_len, hidden
    # strides of the four tensors
    stride_proj_b, stride_proj_i, stride_proj_k, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    BLOCK: tl.constexpr,           # number of elements per program
    HAS_MASK: tl.constexpr,        # compile‑time flag
):
    """Fuse projection, gate and (optional) mask for a 4‑D tensor."""
    pid = tl.program_id(0)
    total = B * N * N * H
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # ---- decode flat index into (b,i,k,d) ----
    d = offs % H
    off1 = offs // H
    k = off1 % N
    off2 = off1 // N
    i = off2 % N
    b = off2 // N

    # ---- compute linear offsets ----
    proj_off = (b * stride_proj_b + i * stride_proj_i +
                k * stride_proj_k + d * stride_proj_h)
    gate_off = (b * stride_gate_b + i * stride_gate_i +
                k * stride_gate_k + d * stride_gate_h)
    out_off  = (b * stride_out_b  + i * stride_out_i +
                k * stride_out_k  + d * stride_out_h)

    proj = tl.load(proj_ptr + proj_off, mask=mask, other=tl.float16(0.0))
    gate = tl.load(gate_ptr + gate_off, mask=mask, other=tl.float16(0.0))
    val  = proj * gate

    if HAS_MASK:
        mask_off = b * stride_mask_b + i * stride_mask_i + k * stride_mask_k
        m = tl.load(mask_ptr + mask_off, mask=mask, other=tl.float16(0.0))
        val = val * m

    tl.store(out_ptr + out_off, val, mask=mask)


def custom_kernel(data):
    """
    Forward pass for the “outgoing” TriMul operation.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : [B, N, N, dim] (float32)
            - mask_tensor  : [B, N, N] (float32) – may be None if `nomask` is True
            - weights_dict : contains all model weight tensors
            - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool, ...}
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------ #
    # unpack
    # ------------------------------------------------------------------ #
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, N2, dim = input_tensor.shape
    assert N == N2, "pairwise tensor must be square"
    hidden_dim = config["hidden_dim"]
    use_mask = not config.get("nomask", True)

    # ------------------------------------------------------------------ #
    # 1) LayerNorm on the input (keep FP16 for the heavy part)
    # ------------------------------------------------------------------ #
    x = input_tensor.to(torch.float16)                     # [B,N,N,dim] fp16
    norm_w = weights["norm.weight"].to(x.dtype)
    norm_b = weights["norm.bias"].to(x.dtype)
    x = torch.nn.functional.layer_norm(
        x, (dim,), weight=norm_w, bias=norm_b, eps=1e-5
    )                                                       # fp16

    # ------------------------------------------------------------------ #
    # 2) Linear projections + gate linear layers (all FP16)
    # ------------------------------------------------------------------ #
    w_left_proj = weights["left_proj.weight"].to(x.dtype)
    w_right_proj = weights["right_proj.weight"].to(x.dtype)
    w_left_gate  = weights["left_gate.weight"].to(x.dtype)
    w_right_gate = weights["right_gate.weight"].to(x.dtype)
    w_out_gate   = weights["out_gate.weight"].to(x.dtype)

    left_proj = torch.nn.functional.linear(x, w_left_proj)          # [B,N,N,H]
    right_proj = torch.nn.functional.linear(x, w_right_proj)

    left_gate = torch.nn.functional.linear(x, w_left_gate).sigmoid()
    right_gate = torch.nn.functional.linear(x, w_right_gate).sigmoid()
    out_gate = torch.nn.functional.linear(x, w_out_gate).sigmoid()  # [B,N,N,H]

    # ------------------------------------------------------------------ #
    # 3) Fuse (proj * gate * mask) with Triton
    # ------------------------------------------------------------------ #
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    if use_mask:
        mask_fp16 = mask_tensor.to(x.dtype)          # [B,N,N]
    else:
        # dummy tensor – never read when HAS_MASK==0
        mask_fp16 = torch.empty(1, dtype=x.dtype, device=device)

    BLOCK = 1024

    total_elements = B * N * N * hidden_dim
    grid = (triton.cdiv(total_elements, BLOCK),)

    # left side
    _fused_proj_gate_mask_kernel[grid](
        left_proj, left_gate, mask_fp16, left,
        B, N, hidden_dim,
        left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        mask_fp16.stride(0) if use_mask else 0,
        mask_fp16.stride(1) if use_mask else 0,
        mask_fp16.stride(2) if use_mask else 0,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        BLOCK=BLOCK,
        HAS_MASK=1 if use_mask else 0,
    )

    # right side (identical kernel, same indexing)
    _fused_proj_gate_mask_kernel[grid](
        right_proj, right_gate, mask_fp16, right,
        B, N, hidden_dim,
        right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        mask_fp16.stride(0) if use_mask else 0,
        mask_fp16.stride(1) if use_mask else 0,
        mask_fp16.stride(2) if use_mask else 0,
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        BLOCK=BLOCK,
        HAS_MASK=1 if use_mask else 0,
    )

    # ------------------------------------------------------------------ #
    # 4) Tri‑multiplicative update  (batched GEMM on the hidden dim)
    # ------------------------------------------------------------------ #
    # bring hidden dim to the batch axis => (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(B * hidden_dim, N, N)    # (BH, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)  # (BH, N, N)

    # batched matrix multiplication: (BH,N,N) @ (BH,N,N)^T  -> (BH,N,N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------ #
    # 5) Post‑processing (LayerNorm, out‑gate, final linear)
    # ------------------------------------------------------------------ #
    # to_out_norm works on the hidden dimension
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float32)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float32)

    # convert to fp32 for the norm (more stable, negligible cost)
    out_fp32 = out.to(torch.float32)
    out_norm = torch.nn.functional.layer_norm(
        out_fp32, (hidden_dim,), weight=to_out_norm_w,
        bias=to_out_norm_b, eps=1e-5
    )                                                   # fp32

    out_norm = out_norm * out_gate.to(out_norm.dtype)   # gate (fp32)

    # final linear back to `dim`
    to_out_w = weights["to_out.weight"].to(out_norm.dtype)   # (dim, hidden_dim)
    final_out = torch.nn.functional.linear(out_norm, to_out_w)  # [B,N,N,dim]

    return final_out
