"""
TriMul "outgoing" implementation using a fused triton kernel for mask & gate application.
The heavy O(N^3) reduction is performed as a batched matrix multiplication over the
hidden dimension, which leverages cuBLAS via torch.bmm. The triton kernel fuses the
element‑wise mask and gate multiplications for both the left and right projected
tensors, reducing memory traffic compared with separate PyTorch ops.
"""

import torch
import triton
import triton.language as tl

# -----------------------------------------------------------------------
# Triton kernel: fuse mask broadcasting, left/right gate multiplication.
# -----------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    # data pointers
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    mask_ptr,
    # problem size
    B, N, H,
    # strides for left/right tensors (they are contiguous and identical)
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    # strides for gate tensors (identical to left)
    stride_g_b, stride_g_i, stride_g_j, stride_g_h,
    # strides for mask (3‑D)
    stride_m_b, stride_m_i, stride_m_j,
    # compile‑time block size
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place mask & gate fusion for left/right tensors.
    For each element (b,i,j,h) we compute:
        left[b,i,j,h]  = left[b,i,j,h]  * left_gate[b,i,j,h]  * mask[b,i,j]
        right[b,i,j,h] = right[b,i,j,h] * right_gate[b,i,j,h] * mask[b,i,j]
    """
    pid = tl.program_id(0)
    # linear index of the first element processed by this program
    block_start = pid * BLOCK_SIZE
    offs = tl.arange(0, BLOCK_SIZE)
    idx = block_start + offs
    total = B * N * N * H
    mask = idx < total

    # convert 1‑D idx into (b,i,j,h)
    b = idx // (N * N * H)
    rem = idx % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # offsets for the tensor reads / writes
    off_l   = b * stride_l_b + i * stride_l_i + j * stride_l_j + h * stride_l_h
    off_lg  = b * stride_g_b + i * stride_g_i + j * stride_g_j + h * stride_g_h
    off_r   = off_l           # right tensor has the same shape/strides
    off_rg  = off_lg          # right gate same layout

    off_mask = b * stride_m_b + i * stride_m_i + j * stride_m_j

    # load values (masked loads return 0 for out‑of‑bounds threads)
    left_val      = tl.load(left_ptr   + off_l,  mask=mask, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + off_lg, mask=mask, other=0.0)
    right_val     = tl.load(right_ptr  + off_r,  mask=mask, other=0.0)
    right_gate_val= tl.load(right_gate_ptr + off_rg, mask=mask, other=0.0)
    mask_val      = tl.load(mask_ptr   + off_mask, mask=mask, other=0.0)

    # fused multiplication
    left_out  = left_val  * left_gate_val  * mask_val
    right_out = right_val * right_gate_val * mask_val

    # store in‑place
    tl.store(left_ptr  + off_l, left_out,  mask=mask)
    tl.store(right_ptr + off_r, right_out, mask=mask)

# -----------------------------------------------------------------------

    """
    Implements the outgoing TriMul operator.

    ----------
    data : tuple
        (input_tensor, mask, weights, config)
    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    # -------------------------------------------------------------------
    # unpack config & shapes
    # -------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape  # input shape (B, N, N, dim)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # LayerNorm on the input (dim‑wise)
    # -------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(

        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -------------------------------------------------------------------
    # Linear projections (no bias)
    # -------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------------------------
    # Gates (Linear + sigmoid)
    # -------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # -------------------------------------------------------------------
    # Ensure contiguous layout for the fused kernel
    # -------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    out_gate = out_gate.contiguous()

    # mask may be None (nomask) – create an all‑ones mask in that case
    if mask is None:
        mask = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask = mask.to(dtype).contiguous()

    # -------------------------------------------------------------------
    # Fuse mask, left_gate, right_gate into left/right (in‑place) via Triton
    # -------------------------------------------------------------------
    total_elements = B * N * N * hidden_dim
    BLOCK_SIZE = 1024  # one thread per element, up to H100's max 1024 threads/block
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    _apply_mask_gate_kernel[grid](
        left,
        left_gate,
        right,
        right_gate,
        mask,
        B,
        N,
        hidden_dim,
        left.stride(0),
        left.stride(1),
        left.stride(2),
        left.stride(3),
        left_gate.stride(0),
        left_gate.stride(1),
        left_gate.stride(2),
        left_gate.stride(3),
        mask.stride(0),
        mask.stride(1),
        mask.stride(2),
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )
    # At this point `left` and `right` contain the masked & gated tensors.

    # -------------------------------------------------------------------
    # Batched GEMM over the hidden dimension: out[b, i, j, d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # -------------------------------------------------------------------
    # Rearrange to (B, hidden_dim, N, N) => view as (B*hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    # Compute the matrix product for each hidden slice
    out_mat = torch.bmm(left_mat, right_mat.transpose(-2, -1))
    # Reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------------------------
    # to_out_norm (layer‑norm over hidden_dim)
    # -------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # -------------------------------------------------------------------
    # Apply the output gate (broadcasted over the hidden dimension)
    # -------------------------------------------------------------------
    out = out * out_gate

    # -------------------------------------------------------------------
    # Final linear projection back to the original dimensionality
    # -------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure float32 output as required
    out = out.to(torch.float32)

    return out

"""
TriMul (outgoing) implementation.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension.
2. Linear projections → left, right (shape [..., N, N, H]).
3. Compute three gating tensors (left_gate, right_gate, out_gate) with a sigmoid.
4. Fuse mask, left_gate and right_gate into the projected tensors using a
   custom Triton kernel (in‑place).  For every element (b,i,j,h):
        left[b,i,j,h]  = left[b,i,j,h]  * left_gate[b,i,j,h]  * mask[b,i,j]
        right[b,i,j,h] = right[b,i,j,h] * right_gate[b,i,j,h] * mask[b,i,j]
5. The core O(N³) update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is exactly a batched matrix‑multiply over the hidden dimension:
        out[...,h] = left[...,h] @ right[...,h]ᵀ
   We perform it with `torch.bmm` (cuBLAS/Tensor‑Core) after casting to
   float16 for speed, then convert back to float32.
6. Layer‑norm the result over the hidden dimension, multiply by out_gate,
   and apply the final linear projection back to the original dimension.

Only step 4 is executed in Triton – the remaining work uses highly‑optimized
PyTorch / cuBLAS kernels.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _apply_mask_gate_kernel(
    # tensor pointers
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    mask_ptr,
    # problem size
    B, N, H,
    # strides for left/right tensors (contiguous layout)
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    # strides for gate tensors (identical layout)
    stride_g_b, stride_g_i, stride_g_j, stride_g_h,
    # strides for mask (3‑D)
    stride_m_b, stride_m_i, stride_m_j,
    # compile‑time constant
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place mask & gate fusion.
    For each (b,i,j,h):
        left[b,i,j,h]  = left[b,i,j,h]  * left_gate[b,i,j,h]  * mask[b,i,j]
        right[b,i,j,h] = right[b,i,j,h] * right_gate[b,i,j,h] * mask[b,i,j]
    """
    pid = tl.program_id(0)                     # linear block id
    block_start = pid * BLOCK_SIZE
    offs = tl.arange(0, BLOCK_SIZE)            # thread id within block
    idx = block_start + offs                    # linear index of element
    total = B * N * N * H
    mask = idx < total

    # de‑compose linear index into (b,i,j,h)
    b = idx // (N * N * H)
    rem = idx % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # compute offsets
    off_l   = b * stride_l_b + i * stride_l_i + j * stride_l_j + h * stride_l_h
    off_lg  = b * stride_g_b + i * stride_g_i + j * stride_g_j + h * stride_g_h
    off_r   = off_l                               # right uses same layout
    off_rg  = off_lg
    off_mask = b * stride_m_b + i * stride_m_i + j * stride_m_j

    # load values (out‑of‑bounds threads read 0)
    left_val      = tl.load(left_ptr   + off_l,   mask=mask, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + off_lg, mask=mask, other=0.0)
    right_val     = tl.load(right_ptr  + off_r,   mask=mask, other=0.0)
    right_gate_val= tl.load(right_gate_ptr + off_rg, mask=mask, other=0.0)
    mask_val      = tl.load(mask_ptr   + off_mask, mask=mask, other=0.0)

    # fused multiplication
    left_out  = left_val  * left_gate_val  * mask_val
    right_out = right_val * right_gate_val * mask_val

    # store back in‑place
    tl.store(left_ptr  + off_l, left_out,  mask=mask)
    tl.store(right_ptr + off_r, right_out, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [bs, N, N, dim]
        - mask         : torch.Tensor of shape [bs, N, N] (or None)
        - weights      : dict with all linear / layer‑norm parameters
        - config       : dict with keys ["dim", "hidden_dim"]

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the feature dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Gates (Linear + sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 4️⃣ Fuse mask & gates (in‑place) with Triton
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    out_gate = out_gate.contiguous()

    if mask is None:
        mask = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask = mask.to(dtype).contiguous()

    total_elements = B * N * N * hidden_dim
    BLOCK_SIZE = 1024                      # ≤ max threads per block on H100
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    _apply_mask_gate_kernel[grid](
        left,
        left_gate,
        right,
        right_gate,
        mask,
        B,
        N,
        hidden_dim,
        left.stride(0),
        left.stride(1),
        left.stride(2),
        left.stride(3),
        left_gate.stride(0),
        left_gate.stride(1),
        left_gate.stride(2),
        left_gate.stride(3),
        mask.stride(0),
        mask.stride(1),
        mask.stride(2),
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )
    # (Triton runs asynchronously on the same CUDA stream, next ops wait for it.)

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³) update: batched GEMM over the hidden dimension
    # ------------------------------------------------------------------
    # Cast to float16 to exploit Tensor‑Core kernels
    left_fp16  = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # (B, H, N, N)
    right_fp16 = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)  # (B, H, N, N)

    left_mat  = left_fp16.view(B * hidden_dim, N, N)     # (B*H, N, N)
    right_mat = right_fp16.view(B * hidden_dim, N, N)    # (B*H, N, N)

    # Matrix multiply for each hidden slice
    out_mat_fp16 = torch.bmm(left_mat, right_mat.transpose(-2, -1))

    # Back‑to‑float32 and reshape to (B, N, N, H)
    out = out_mat_fp16.to(torch.float32).view(B, hidden_dim, N, N)
    out = out.permute(0, 2, 3, 1).contiguous()           # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Output layer‑norm, out‑gate and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    out = out * out_gate                                 # broadcast over H

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure the required float32 output
    return out.to(torch.float32)
