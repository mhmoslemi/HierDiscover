    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    inv_std = torch.rsqrt(var + eps)
    x_norm = (input_tensor - mean) * inv_std
    # apply weight, bias

    left = F.linear(x_norm, weights["left_proj.weight"])  # shape [bs, N, N, hidden_dim]
    right = F.linear(x_norm, weights["right_proj.weight"])

    if not nomask:

        mask_unsq = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_unsq
        right = right * mask_unsq

    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Prepare output tensor
    out = torch.empty_like(left_h, dtype=torch.float16)  # shape same as left (B,N,N,hidden_dim)

    # Compute strides
    stride_left = left_h.stride()
    stride_right = right_h.stride()
    stride_out = out.stride()

    # kernel grid
    B, N, _, H = left_h.shape
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

             (N + BLOCK_N - 1)//BLOCK_N,
             B * H)

    triton_tri_mul_kernel[grid](
        left_h, right_h, out,
        B, N, H,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Convert back to float32 for subsequent layers
    out_f = out.to(torch.float32)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    # compute mean/var over hidden dim
    mean_o = out_f.mean(dim=-1, keepdim=True)
    var_o = out_f.var(dim=-1, unbiased=False, keepdim=True)
    inv_std_o = torch.rsqrt(var_o + eps)
    out_norm = (out_f - mean_o) * inv_std_o
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # Multiply by out_gate (needs broadcast)
    out_gate_f = out_gate.to(out_norm.dtype)  # shape B,N,N,hidden_dim
    out_norm = out_norm * out_gate_f

    # Final linear
    to_out_weight = weights["to_out.weight"]
    out_final = F.linear(out_norm, to_out_weight)   # shape [B,N,N,dim]
    return out_final

"""
custom_kernel: implements outgoing TriMul forward pass.
- LayerNorm on input.
- Linear projections and gating.
- Mask application.
- Triton kernel computes out = sum_k left[:,i,k,:] * right[:,j,k,:].
- LayerNorm on result, gating, final linear.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_i = tl.program_id(0)   # tile row dimension (i)
    pid_j = tl.program_id(1)   # tile col dimension (j)
    pid_hb = tl.program_id(2)  # combined batch and hidden index

    # decode combined index
    b = pid_hb // H
    h = pid_hb % H

    # start offsets for the M and N dimensions
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds rows/cols
    mask_i = i_offset < N
    mask_j = j_offset < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension (the summed axis)
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # ----- Load a tile of left (shape BLOCK_M x BLOCK_K) -----
        # layout: (B, i, k, H)
        a_ptrs = left_ptr \
                 + b * stride_lb \
                 + h * stride_lh \
                 + i_offset[:, None] * stride_li \
                 + k_offset[None, :] * stride_lk
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # ----- Load a tile of right transposed (shape BLOCK_K x BLOCK_N) -----
        # right layout: (B, j, k, H); we need (k, j)
        b_ptrs = right_ptr \
                 + b * stride_rb \
                 + h * stride_rh \
                 + k_offset[:, None] * stride_rk \
                 + j_offset[None, :] * stride_ri
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float16)

        # ----- Dot product over the K dimension -----
        acc += tl.dot(a, b)   # result is FP32

    # ----- Write the result -----
    c_ptrs = out_ptr \
             + b * stride_ob \
             + h * stride_oh \
             + i_offset[:, None] * stride_oi \
             + j_offset[None, :] * stride_oj
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Outgoing TriMul forward pass.
    Implements the heavy N^3 contraction using a custom Triton kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ---------- LayerNorm on input ----------
    # weight and bias are 1D tensors of size dim
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Compute mean/var across last dimension
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    inv_std = torch.rsqrt(var + eps)
    x = (input_tensor - mean) * inv_std
    x = x * norm_weight + norm_bias   # broadcast over ... dim

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if not nomask:
        # mask shape: [B, N, N]; broadcast to last dim
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ---------- Gating ----------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to half for the kernel ----------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Output placeholder (half)
    out_h = torch.empty_like(left_h, dtype=torch.float16, device=device)

    # ---------- Triton kernel launch ----------
    B, N, _, H = left_h.shape

    # Strides (int64)
    stride_left = left_h.stride()
    stride_right = right_h.stride()
    stride_out = out_h.stride()

    # Block sizes (tuneable)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    tri_mul_outgoing_kernel[grid](
        left_h,
        right_h,
        out_h,
        B, N, H,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure kernel finished
    torch.cuda.synchronize()

    # Cast back to FP32 for the remaining ops
    out = out_h.to(torch.float32)

    # ---------- to_out_norm ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    inv_std_o = torch.rsqrt(var_o + eps)
    out_norm = (out - mean_o) * inv_std_o
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # ---------- Apply out gate ----------
    out_norm = out_norm * out_gate

    # ---------- Final linear ----------
    to_out_weight = weights["to_out.weight"]
    output = F.linear(out_norm, to_out_weight)   # bias is False

    return output

"""
custom_kernel
-------------

Implements the *outgoing* TriMul forward pass described in the AlphaFold3 paper.
The heavy N³ contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

is performed by a custom Triton kernel.  The surrounding operations
(LayerNorm, linear projections, gating, final LayerNorm and output linear)
are carried out with regular PyTorch kernels.

The kernel tiles the output in a (BLOCK_M × BLOCK_N) block, iterates over the
summation dimension K in blocks of size BLOCK_K and accumulates in FP32 while
performing the inner products in FP16.  One program instance is launched for
every (batch, hidden‑channel) pair, i.e. grid = (⌈N/BM⌉, ⌈N/BN⌉, B·H).

All inputs are assumed to be on the GPU; the final output is FP32 with shape
[bs, seq_len, seq_len, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,      # *fp16   [B, N, N, H]
    right_ptr,     # *fp16   [B, N, N, H]
    out_ptr,       # *fp16   [B, N, N, H]
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,   # strides for left
    stride_rb, stride_ri, stride_rk, stride_rh,   # strides for right
    stride_ob, stride_oi, stride_oj, stride_oh,   # strides for out
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""

    pid_i = tl.program_id(0)          # tile row   (i)
    pid_j = tl.program_id(1)          # tile col   (j)
    pid_hb = tl.program_id(2)         # combined batch & hidden index

    # Decode batch (b) and hidden channel (h)
    b = pid_hb // H
    h = pid_hb % H

    # Tile offsets in the M and N dimensions
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_off < N
    mask_j = j_off < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension (summation axis)
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # --------------------------------------------------------------
        # Load a tile of LEFT: shape (BLOCK_M, BLOCK_K)
        # layout: (B, i, k, H)
        a_ptrs = left_ptr \
                 + b * stride_lb \
                 + h * stride_lh \
                 + i_off[:, None] * stride_li \
                 + k_off[None, :] * stride_lk
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # --------------------------------------------------------------
        # Load a tile of RIGHT transposed: shape (BLOCK_K, BLOCK_N)
        # Right layout: (B, j, k, H) → we need (k, j)
        b_ptrs = right_ptr \
                 + b * stride_rb \
                 + h * stride_rh \
                 + k_off[:, None] * stride_rk \
                 + j_off[None, :] * stride_ri
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float16)

        # --------------------------------------------------------------
        # Dot product over K (FP16 × FP16 → FP32)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write the block of results to OUT
    c_ptrs = out_ptr \
             + b * stride_ob \
             + h * stride_oh \
             + i_off[:, None] * stride_oi \
             + j_off[None, :] * stride_oj
    tl.store(c_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [bs, N, N, dim]
        - mask         : torch.Tensor of shape [bs, N, N] (may be all‑ones)
        - weights      : dict containing all required weight tensors
        - config       : dict with keys "dim", "hidden_dim",
                         optional "eps" (float) and "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, N, N, dim] (FP32)
    """
    # ----------------------------------------------
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ----------------------------------------------
    # LayerNorm on the input (last dimension)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # torch.nn.functional.layer_norm normalises over the last dimension
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ----------------------------------------------
    # Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # ----------------------------------------------
    # Optional mask (broadcasted on the hidden dimension)
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)      # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ----------------------------------------------
    # Gating (sigmoid of linear projections)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------
    # Cast to FP16 for the heavy contraction
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Allocate output tensor (FP16)
    out_h = torch.empty_like(left_h, dtype=torch.float16, device=device)

    # ----------------------------------------------
    # Triton kernel launch parameters
    B, N, _, H = left_h.shape
    stride_left = left_h.stride()    # (B,N,N,H) → (N*N*H, N*H, H, 1)
    stride_right = right_h.stride()
    stride_out = out_h.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles over i
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles over j
        B * H,                          # one program per (batch, hidden)
    )

    tri_mul_outgoing_kernel[grid](
        left_h,
        right_h,
        out_h,
        B, N, H,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()  # ensure kernel completion

    # ----------------------------------------------
    # Cast back to FP32 for the remaining layers
    out = out_h.to(torch.float32)

    # ----------------------------------------------
    # to_out_norm (LayerNorm over the hidden dimension)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    # ----------------------------------------------
    # Apply the output gate
    out = out * out_gate

    # ----------------------------------------------
    # Final linear projection (no bias)
    to_out_weight = weights["to_out.weight"]
    output = F.linear(out, to_out_weight)   # (B,N,N,dim)

    return output
