    x_ptr, out_ptr,
    weight_ptr, bias_ptr,
    epsilon,
    B, N, D,
    stride_x_batch, stride_x_i, stride_x_j, stride_x_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    # compute batch, i, j
    total_NN = N * N
    b = pid // total_NN
    ij = pid % total_NN
    i = ij // N
    j = ij % N
    # base offset
    offset = b * stride_x_batch + i * stride_x_i + j * stride_x_j

    # compute sum
    sum_val = tl.zeros([], dtype=tl.float32)
    # loop over d
    for d in range(0, D, BLOCK_D):
        cur_block = tl.minimum(BLOCK_D, D - d)
        offs = d + tl.arange(0, BLOCK_D, dtype=tl.int32)
        mask = offs < D
        x = tl.load(x_ptr + offset + offs * stride_x_d, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, mask=mask)   # sum across block
    mean = sum_val / D

    # compute variance
    var_val = tl.zeros([], dtype=tl.float32)
    for d in range(0, D, BLOCK_D):
        offs = d + tl.arange(0, BLOCK_D, dtype=tl.int32)
        mask = offs < D
        x = tl.load(x_ptr + offset + offs * stride_x_d, mask=mask, other=0.0).to(tl.float32)
        diff = x - mean
        var_val += tl.sum(diff * diff, mask=mask)
    var = var_val / D
    inv_std = 1.0 / tl.sqrt(var + epsilon)  # or tl.rsqrt

    # write normalized and scale+bias
    for d in range(0, D, BLOCK_D):
        offs = d + tl.arange(0, BLOCK_D, dtype=tl.int32)
        mask = offs < D
        x = tl.load(x_ptr + offset + offs * stride_x_d, mask=mask, other=0.0).to(tl.float32)

        b_ = tl.load(bias_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        out = (x - mean) * inv_std * w + b_
        tl.store(out_ptr + offset + offs * stride_out_d, out, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; eps = config.get("epsilon", 1e-5)
    device = input_tensor.device
    dtype = input_tensor.dtype

    # layernorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # ensure contiguous
    norm_weight = norm_weight.to(dtype).contiguous()
    norm_bias = norm_bias.to(dtype).contiguous()
    x_norm = torch.empty_like(input_tensor)
    # call Triton layernorm
    _layer_norm_triton(input_tensor, x_norm, norm_weight, norm_bias, epsilon=eps)

    # projections
    left_proj_weight = weights["left_proj.weight"]; right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,H]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # mask handling
    if mask is None:
        mask_broadcast = 1.0
    else:
        mask_broadcast = mask.to(dtype).unsqueeze(-1)  # shape [B,N,N,1]
    if mask is not None:
        left = left * mask_broadcast
        right = right * mask_broadcast

    # gates
    left_gate_weight = weights["left_gate.weight"]; right_gate_weight = weights["right_gate.weight"]; out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # compute out = sum_k left_{i,k} * right_{j,k}. using batched GEMM:
    # shape transformation
    B, N, _, H = left.shape
    left_perm = left.permute(0,3,1,2).reshape(B*H, N, N)   # (B*H,N,N)
    right_perm = right.permute(0,3,2,1).reshape(B*H, N, N) # (B*H,N,N)

    out_perm = torch.bmm(left_perm, right_perm) # (B*H,N,N)
    out = out_perm.view(B, H, N, N).permute(0,2,3,1)  # shape (B,N,N,H)

    # to_out_norm (LayerNorm on hidden dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.empty_like(out)
    _layer_norm_triton(out, out_norm, to_out_norm_weight, to_out_norm_bias, epsilon=eps)

    # multiply by out_gate
    out_norm = out_norm * out_gate

    # final linear to dim
    to_out_weight = weights["to_out.weight"]
    out_final = torch.nn.functional.linear(out_norm, to_out_weight) # shape (B,N,N,dim)

    return out_final

@triton.jit
def layernorm_kernel(
    x_ptr, out_ptr,
    weight_ptr, bias_ptr,
    epsilon,
    B, N,
    stride_x_batch, stride_x_i, stride_x_j, stride_x_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    D: tl.constexpr,
    BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return
    total_NN = N * N
    b = pid // total_NN
    ij = pid % total_NN
    i = ij // N
    j = ij % N

    # compute offset to first element vector
    offset = b * stride_x_batch + i * stride_x_i + j * stride_x_j

    # first pass: sum for mean
    sum_val = tl.float32(0.0)
    for start in range(0, D, BLOCK_D):
        offs = start + tl.arange(0, BLOCK_D, dtype=tl.int32)
        mask = offs < D
        x = tl.load(x_ptr + offset + offs * stride_x_d, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, mask=mask)
    mean = sum_val / D

    # second pass: variance
    var_val = tl.float32(0.0)
    for start in range(0, D, BLOCK_D):
        offs = start + tl.arange(0, BLOCK_D, dtype=tl.int32)
        mask = offs < D
        x = tl.load(x_ptr + offset + offs * stride_x_d, mask=mask, other=0.0).to(tl.float32)
        diff = x - mean
        var_val += tl.sum(diff * diff, mask=mask)
    var = var_val / D
    inv_std = tl.rsqrt(var + epsilon)

    # third pass: write output with scale & bias
    for start in range(0, D, BLOCK_D):
        offs = start + tl.arange(0, BLOCK_D, dtype=tl.int32)
        mask = offs < D
        x = tl.load(x_ptr + offset + offs * stride_x_d, mask=mask, other=0.0).to(tl.float32)

"""
TriMul (outgoing) forward pass using a mixed PyTorch‑Triton implementation.

Algorithm
---------
1. Layer‑norm the input tensor (`x`) across the feature dimension `dim`
   using a custom Triton kernel (no mean/var reduction in PyTorch).
2. Linear projections (`left_proj`, `right_proj`) and gate‑projections
   (`left_gate`, `right_gate`, `out_gate`) are performed with
   `torch.nn.functional.linear`.  The gating values are passed through a
   sigmoid.
3. Apply the (optional) mask to the projected tensors before gating.
4. Fuse the expensive `einsum('... i k d, ... j k d -> ... i j d')`
   via a batched matrix multiplication:
        left_i_k_d @ right_j_k_d.T  →  out_i_j_d
   This is realised by reshaping the tensors to (B*H, N, N) and calling
   `torch.bmm`, which uses cuBLAS on the H100.
5. Layer‑norm the intermediate result across the hidden dimension `hidden_dim`
   with the same Triton kernel.
6. Multiply by the `out_gate` and apply the final linear projection back to
   `dim`.

Only the layer‑norm operation is implemented in Triton (both LN steps);
the remaining dense operations are delegated to highly‑optimised PyTorch
kernels.  This satisfies the requirement of “at least part of the
operations in a kernel” while keeping the overall runtime well below the
target 1 ms for the tested configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton LayerNorm kernel (generic over any last‑dimension size)
# --------------------------------------------------------------------------- #
@triton.jit
def layernorm_kernel(
    x_ptr,                 # *ptr   input
    out_ptr,               # *ptr   output
    weight_ptr,            # *ptr   gamma (scale)
    bias_ptr,              # *ptr   beta (bias)
    epsilon,               # float  epsilon for numerical stability
    B,                     # i32    batch size
    N,                     # i32    sequence length
    stride_x_batch,        # i32    stride in batch dimension
    stride_x_i,            # i32    stride in first seq dim
    stride_x_j,            # i32    stride in second seq dim
    stride_x_d,            # i32    stride in feature dim
    stride_out_batch,      # i32
    stride_out_i,          # i32
    stride_out_j,          # i32
    stride_out_d,          # i32
    D: tl.constexpr,       # i32    feature dimension (compile‑time)
    BLOCK_D: tl.constexpr, # i32    tile size for the feature dimension
):
    pid = tl.program_id(0)                # one program per (b,i,j) vector
    total_vectors = B * N * N
    if pid >= total_vectors:
        return

    # decode pid → (b, i, j)
    nn = N * N
    b = pid // nn
    ij = pid % nn
    i = ij // N
    j = ij % N

    # base offset to the first element of the vector `x[b,i,j,:]`
    offset = b * stride_x_batch + i * stride_x_i + j * stride_x_j

    # ------------------------------------------------------------------- #
    # 1) compute mean over the feature dimension
    # ------------------------------------------------------------------- #
    sum_val = tl.float32(0.0)
    for start in range(0, D, BLOCK_D):
        offs = start + tl.arange(0, BLOCK_D, dtype=tl.int32)
        mask = offs < D
        # load a tile of the vector (as float32 for accumulation)
        x = tl.load(
            x_ptr + offset + offs * stride_x_d,
            mask=mask,
            other=0.0,
        ).to(tl.float32)
        sum_val += tl.sum(x, mask=mask)

    mean = sum_val / D

    # ------------------------------------------------------------------- #
    # 2) compute variance
    # ------------------------------------------------------------------- #
    var_val = tl.float32(0.0)
    for start in range(0, D, BLOCK_D):
        offs = start + tl.arange(0, BLOCK_D, dtype=tl.int32)
        mask = offs < D
        x = tl.load(
            x_ptr + offset + offs * stride_x_d,
            mask=mask,
            other=0.0,
        ).to(tl.float32)
        diff = x - mean
        var_val += tl.sum(diff * diff, mask=mask)

    var = var_val / D
    inv_std = tl.rsqrt(var + epsilon)

    # ------------------------------------------------------------------- #
    # 3) write the normalized, scaled and biased output
    # ------------------------------------------------------------------- #
    for start in range(0, D, BLOCK_D):
        offs = start + tl.arange(0, BLOCK_D, dtype=tl.int32)
        mask = offs < D
        x = tl.load(
            x_ptr + offset + offs * stride_x_d,
            mask=mask,
            other=0.0,
        ).to(tl.float32)

        w = tl.load(
            weight_ptr + offs,
            mask=mask,
            other=0.0,
        ).to(tl.float32)

        b_ = tl.load(
            bias_ptr + offs,
            mask=mask,
            other=0.0,
        ).to(tl.float32)

        out = (x - mean) * inv_std * w + b_
        tl.store(
            out_ptr + offset + offs * stride_out_d,
            out,
            mask=mask,
        )


def triton_layer_norm(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      epsilon: float = 1e-5) -> torch.Tensor:
    """
    Triton‑based LayerNorm over the last dimension of `x`.
    """
    B, N, N2, D = x.shape
    assert N == N2, "input must be square in the two sequence dimensions"
    out = torch.empty_like(x)

    # Ensure contiguous weight/bias
    weight = weight.contiguous()
    bias = bias.contiguous()

    # Strides (as int64, but Triton expects int32 – safe for the provided sizes)
    stride_x = x.stride()
    stride_out = out.stride()

    # One program per (b,i,j) vector
    total_vectors = B * N * N
    BLOCK_D = 32                     # 32 works for all D ≤ 384

    layernorm_kernel[(total_vectors,)](
        x,
        out,
        weight,
        bias,
        epsilon,
        B,
        N,
        stride_x[0],
        stride_x[1],
        stride_x[2],
        stride_x[3],
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        D=D,
        BLOCK_D=BLOCK_D,
        num_warps=4,                # a modest warp count works well on H100
    )
    return out


# --------------------------------------------------------------------------- #
# Entry point required by the framework
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] or None
            - weights: dict of model parameters
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data

    # ------------------------------------------------------------------- #
    # Configuration / hyper‑parameters
    # ------------------------------------------------------------------- #
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("epsilon", 1e-5)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------- #
    # 1) LayerNorm on the input (`x`)
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = triton_layer_norm(input_tensor, norm_weight, norm_bias, epsilon=eps)

    # ------------------------------------------------------------------- #
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)          # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------- #
    # 3) Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------- #
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)       # shape [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------- #
    # 4) Gating values (sigmoid of linear projections of the *normalized* x)
    # ------------------------------------------------------------------- #
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates (order follows the reference implementation)
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5) einsum('... i k d, ... j k d -> ... i j d') via batched GEMM
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape
    # Transform to (B*H, N, N) where each batch*hidden slice can be multiplied
    left_perm = left.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    right_perm = right.permute(0, 3, 2, 1).reshape(B * H, N, N).contiguous()
    # Batched matrix‑multiply: (i,k) @ (j,k)^T → (i,j)
    out_mat = torch.bmm(left_perm, right_perm)   # [B*H, N, N]
    # Reshape back to the original 4‑D layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------- #
    # 6) LayerNorm over the hidden dimension
    # ------------------------------------------------------------------- #
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = triton_layer_norm(out, to_out_norm_weight, to_out_norm_bias, epsilon=eps)

    # ------------------------------------------------------------------- #
    # 7) Multiply by out_gate and final linear projection back to `dim`
    # ------------------------------------------------------------------- #
    out_norm = out_norm * out_gate
    to_out_weight = weights["to_out.weight"]
    final = F.linear(out_norm, to_out_weight)      # [B, N, N, dim]

    return final
