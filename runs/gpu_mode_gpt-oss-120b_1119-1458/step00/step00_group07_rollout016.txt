@triton.jit
def apply_mask_gate_kernel(L_ptr, R_ptr, LG_ptr, RG_ptr, mask_ptr, L_out_ptr, R_out_ptr,
    B, N, hidden_dim,
    stride_L_b, stride_L_n1, stride_L_n2, stride_L_h,
    stride_R_b, stride_R_n1, stride_R_n2, stride_R_h,
    stride_LG_b, stride_LG_n1, stride_LG_n2, stride_LG_h,
    stride_RG_b, stride_RG_n1, stride_RG_n2, stride_RG_h,
    stride_mask_b, stride_mask_n1, stride_mask_n2,
    stride_Lout_b, stride_Lout_n1, stride_Lout_n2, stride_Lout_h,
    stride_Rout_b, stride_Rout_n1, stride_Rout_n2, stride_Rout_h,
    BLOCK: tl.constexpr):
    # compute linear index
    pid = tl.program_id(0)
    total = B * N * N * hidden_dim
    # if pid >= total: return
    # compute indices
    b = pid // (N * N * hidden_dim)
    rem = pid % (N * N * hidden_dim)
    i = rem // (N * hidden_dim)
    rem2 = rem % (N * hidden_dim)
    j = rem2 // hidden_dim
    d = rem2 % hidden_dim

    # compute pointers
    offset_L = b*stride_L_b + i*stride_L_n1 + j*stride_L_n2 + d*stride_L_h
    offset_R = b*stride_R_b + i*stride_R_n1 + j*stride_R_n2 + d*stride_R_h
    offset_LG = b*stride_LG_b + i*stride_LG_n1 + j*stride_LG_n2 + d*stride_LG_h
    offset_RG = b*stride_RG_b + i*stride_RG_n1 + j*stride_RG_n2 + d*stride_RG_h
    offset_mask = b*stride_mask_b + i*stride_mask_n1 + j*stride_mask_n2

    # load values
    left = tl.load(L_ptr + offset_L)
    right = tl.load(R_ptr + offset_R)
    left_gate = tl.load(LG_ptr + offset_LG)
    right_gate = tl.load(RG_ptr + offset_RG)
    mask_val = tl.load(mask_ptr + offset_mask) if mask_ptr != 0 else tl.full([1], 1.0, dtype=tl.float32)[0]
    # compute
    left_out = left * left_gate * mask_val
    right_out = right * right_gate * mask_val

    # store
    tl.store(L_out_ptr + offset_L, left_out)
    tl.store(R_out_ptr + offset_R, right_out)

@triton.jit
def apply_mask_gate_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_rows, hidden_dim,
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_c,
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_c,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_c,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_c,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_D: tl.constexpr):
    """
    Elementwise fused multiplication of left/right projections with their respective gates
    and mask. Each program processes a tile of hidden_dim elements for a specific (b,i,j) pair.
    """
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)

    # compute batch, i, j indices
    row = pid0
    b = row // (seq_len * seq_len)
    rem = row % (seq_len * seq_len)
    i = rem // seq_len
    j = rem % seq_len

    # compute base offsets for each tensor
    # offset for mask (scalar)
    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    mask_val = tl.load(mask_ptr + mask_offset) # broadcast

    # compute column offset for hidden dims
    c_base = pid1 * BLOCK_D
    c = c_base + tl.arange(0, BLOCK_D)

    # mask for valid hidden dim
    c_mask = c < hidden_dim

    # load left_proj slice
    lp_offset = b*stride_lp_b + i*stride_lp_i + j*stride_lp_j + c*stride_lp_c
    left = tl.load(left_proj_ptr + lp_offset, mask=c_mask, other=0.0)
    # load right_proj slice
    rp_offset = b*stride_rp_b + i*stride_rp_i + j*stride_rp_j + c*stride_rp_c
    right = tl.load(right_proj_ptr + rp_offset, mask=c_mask, other=0.0)
    # load gates
    lg_offset = b*stride_lg_b + i*stride_lg_i + j*stride_lg_j + c*stride_lg_c
    left_gate = tl.load(left_gate_ptr + lg_offset, mask=c_mask, other=0.0)
    rg_offset = b*stride_rg_b + i*stride_rg_i + j*stride_rg_j + c*stride_rg_c
    right_gate = tl.load(right_gate_ptr + rg_offset, mask=c_mask, other=0.0)

    # compute fused output
    left_out = left * left_gate * mask_val
    right_out = right * right_gate * mask_val

    # store
    tl.store(left_out_ptr + lp_offset, left_out, mask=c_mask)
    tl.store(right_out_ptr + rp_offset, right_out, mask=c_mask)

"""
TriMul “outgoing” implementation (forward only).

Algorithm
---------
1. Layer‑norm the input tensor `x` over the last dimension (dim).
2. Linear projections without bias:
       left  = x @ left_proj.weight.T
       right = x @ right_proj.weight.T
3. Compute three gating tensors (sigmoid of linear projections):
       left_gate  = sigmoid(x @ left_gate.weight.T)
       right_gate = sigmoid(x @ right_gate.weight.T)
       out_gate   = sigmoid(x @ out_gate.weight.T)
4. Fuse the mask, left/right gates and the projected tensors in a Triton kernel:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
5. Compute the “triangle” contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is performed as a batched GEMM:
       (B·D, N, N) × (B·D, N, N)ᵀ  →  (B·D, N, N)
6. Layer‑norm the result over the hidden dimension, multiply by `out_gate`
   and apply the final linear projection back to `dim`.

Only step 4 is executed on the GPU with a custom Triton kernel; all other
operations use PyTorch’s highly‑optimized primitives (including cuBLAS for the
batched GEMM). The kernel works for any batch size, sequence length and hidden
dimension supported by the test matrix.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _fused_mask_gate_kernel(
    # pointers
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    # runtime sizes
    seq_len, hidden_dim,
    # strides (in elements)
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_c,
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_c,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_c,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_c,
    stride_mask_b, stride_mask_i, stride_mask_j,
    # compile‑time tile size
    BLOCK_D: tl.constexpr,
):
    """
    Fused element‑wise multiplication:
        left  = left_proj  * left_gate  * mask
        right = right_proj * right_gate * mask
    Each program processes a (batch, i, j) slice and a BLOCK_D chunk of
    the hidden dimension.
    """
    pid0 = tl.program_id(0)         # encodes (b, i, j)
    pid1 = tl.program_id(1)         # encodes hidden‑dim block

    # ------------------------------------------------------------
    # decode (b, i, j) from linear index `pid0`
    # ------------------------------------------------------------
    row = pid0
    b = row // (seq_len * seq_len)
    rem = row % (seq_len * seq_len)
    i = rem // seq_len
    j = rem % seq_len

    # ------------------------------------------------------------
    # load the scalar mask value and broadcast it
    # ------------------------------------------------------------
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    mask_val = tl.load(mask_ptr + mask_off)

    # ------------------------------------------------------------
    # hidden‑dim offsets inside the tile
    # ------------------------------------------------------------
    c_start = pid1 * BLOCK_D
    c = c_start + tl.arange(0, BLOCK_D)
    c_valid = c < hidden_dim

    # ------------------------------------------------------------
    # load all required tensors
    # ------------------------------------------------------------
    lp_off = b * stride_lp_b + i * stride_lp_i + j * stride_lp_j + c * stride_lp_c
    left = tl.load(left_proj_ptr + lp_off, mask=c_valid, other=0.0)

    rp_off = b * stride_rp_b + i * stride_rp_i + j * stride_rp_j + c * stride_rp_c
    right = tl.load(right_proj_ptr + rp_off, mask=c_valid, other=0.0)

    lg_off = b * stride_lg_b + i * stride_lg_i + j * stride_lg_j + c * stride_lg_c
    left_gate = tl.load(left_gate_ptr + lg_off, mask=c_valid, other=0.0)

    rg_off = b * stride_rg_b + i * stride_rg_i + j * stride_rg_j + c * stride_rg_c
    right_gate = tl.load(right_gate_ptr + rg_off, mask=c_valid, other=0.0)

    # ------------------------------------------------------------
    # fused multiplication
    # ------------------------------------------------------------
    left_out = left * left_gate * mask_val
    right_out = right * right_gate * mask_val

    # ------------------------------------------------------------
    # write results
    # ------------------------------------------------------------
    tl.store(left_out_ptr + lp_off, left_out, mask=c_valid)
    tl.store(right_out_ptr + rp_off, right_out, mask=c_valid)


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N]  (may be all‑ones if nomask)
        - weights      : dict of model weights
        - config       : dict with keys: dim, hidden_dim, nomask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # -----------------------------------------------------------------
    # unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # ensure mask is present and has the correct dtype
    # -----------------------------------------------------------------
    if nomask or mask is None:
        mask = torch.ones(
            (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
            dtype=dtype,
            device=device,
        )
    else:
        # mask may be bool – convert to float
        mask = mask.to(dtype)

    # -----------------------------------------------------------------
    # layer‑norm on the input (over the last dimension `dim`)
    # -----------------------------------------------------------------
    eps = 1e-5
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj = F.linear(x_norm, weights["left_proj.weight"])
    right_proj = F.linear(x_norm, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # gating tensors (sigmoid of linear projections)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # fuse mask, left/right gates and projections with a Triton kernel
    # -----------------------------------------------------------------
    B, N, _, _ = x_norm.shape
    # allocate output tensors
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # stride extraction (in elements, not bytes)
    lp_str = left_proj.stride()
    rp_str = right_proj.stride()
    lg_str = left_gate.stride()
    rg_str = right_gate.stride()
    mask_str = mask.stride()

    total_rows = B * N * N
    BLOCK_D = 32  # hidden‑dim tile size (tunable)
    grid = (total_rows, (hidden_dim + BLOCK_D - 1) // BLOCK_D)

    _fused_mask_gate_kernel[grid](
        # pointers
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask,
        left,
        right,
        # runtime sizes
        N,
        hidden_dim,
        # strides
        lp_str[0],
        lp_str[1],
        lp_str[2],
        lp_str[3],
        rp_str[0],
        rp_str[1],
        rp_str[2],
        rp_str[3],
        lg_str[0],
        lg_str[1],
        lg_str[2],
        lg_str[3],
        rg_str[0],
        rg_str[1],
        rg_str[2],
        rg_str[3],
        mask_str[0],
        mask_str[1],
        mask_str[2],
        # compile‑time tile size
        BLOCK_D=BLOCK_D,
    )

    # -----------------------------------------------------------------
    # triangle contraction: Σ_k left[...,k,d] * right[...,k,d]
    # This is a batched GEMM over the hidden dimension.
    # -----------------------------------------------------------------
    # reshape to (B·D, N, N) for a single batched matmul
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, D, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, D, N, N]

    B_D = B * hidden_dim
    left_flat = left_t.view(B_D, N, N)               # [B·D, N, N]
    right_flat = right_t.view(B_D, N, N)             # [B·D, N, N]

    # batched matrix multiplication (left × rightᵀ)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # [B·D, N, N]

    # reshape back to [B, N, N, D]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # final layer‑norm over hidden_dim, gate, and final linear projection
    # -----------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, D]

    out_gated = out_norm * out_gate

    # final linear back to `dim`
    output = F.linear(out_gated, weights["to_out.weight"])

    return output
