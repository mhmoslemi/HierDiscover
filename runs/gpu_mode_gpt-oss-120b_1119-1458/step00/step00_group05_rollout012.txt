@triton.jit
def triton_trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    B, N, H,
    stride_A_b, stride_A_h, stride_A_i, stride_A_k,
    stride_B_b, stride_B_h, stride_B_k, stride_B_j,
    stride_C_b, stride_C_h, stride_C_i, stride_C_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    bh = tl.program_id(0) # combined batch*hidden index
    i_block = tl.program_id(1)
    j_block = tl.program_id(2)

    # decode b and h
    b = bh // H
    h = bh % H

    # compute base offsets for this block
    i_offset = i_block * BLOCK_M
    j_offset = j_block * BLOCK_N

    # tile offsets for k reduction
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute number of iterations over K dimension
    for k in range(0, N, BLOCK_K):
        # compute k offset
        k_offset = k

        # compute offsets for A and B
        a_ptrs = A_ptr + b * stride_A_b + h * stride_A_h + (i_offset + tl.arange(0, BLOCK_M))[:, None] * stride_A_i + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_A_k
        b_ptrs = B_ptr + b * stride_B_b + h * stride_B_h + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_B_k + (j_offset + tl.arange(0, BLOCK_N))[None, :] * stride_B_j

        # bounds checking
        mask_i = (i_offset + tl.arange(0, BLOCK_M)) < N
        mask_j = (j_offset + tl.arange(0, BLOCK_N)) < N
        mask_k = (k_offset + tl.arange(0, BLOCK_K)) < N

        # For loads: set out-of-domain to zero
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float16)

        # dot product
        acc += tl.dot(a, b)

    # write back
    c_ptrs = C_ptr + b * stride_C_b + h * stride_C_h + (i_offset + tl.arange(0, BLOCK_M))[:, None] * stride_C_i + (j_offset + tl.arange(0, BLOCK_N))[None, :] * stride_C_j

    mask_i = (i_offset + tl.arange(0, BLOCK_M)) < N
    mask_j = (j_offset + tl.arange(0, BLOCK_N)) < N
    mask = mask_i[:, None] & mask_j[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)
```

    right_proj_weight = weights["right_proj.weight"]

        right = right * mask_expanded

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # apply gates
    left = left * left_gate
    right = right * right_gate

    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    left_perm = left_fp16.permute(0, 3, 1, 2)

    right_perm = right_fp16.permute(0, 3, 2, 1)

    to_out_weight = weights["to_out.weight"]
    out_final = torch.nn.functional.linear(out_norm, to_out_weight)  # shape [B,N,N,dim]

"""
TriMul (outgoing) forward pass implemented with Triton.

Algorithm
---------
1. Layer‑norm `x` on the last dimension (dim).
2. Linear projections:
   * left  = x @ left_proj_weight
   * right = x @ right_proj_weight
3. Compute three sigmoid gates (left, right, output) from `x`.
4. Apply gates and (optional) pairwise mask:
   left  = left  * left_gate  * mask
   right = right * right_gate * mask
5. Compute the core tensor‑wise product
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication
   left[b,h] (N×N)  @  right[b,h].T (N×N).
   This is done by a custom Triton kernel that processes the
   batch‑hidden dimension (`bh = b*hidden_dim+h`) together.
6. Layer‑norm on the hidden dimension, element‑wise multiply by the
   output gate and a final linear projection back to `dim`.

Only the heavy N³ reduction (step 5) is executed in Triton; all other
operations use PyTorch’s highly‑optimized kernels.
"""
import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ B.T where A,B shape = (B,H,N,N)
#   A : rows = i, cols = k
#   B : rows = k, cols = j   (already transposed)
#   C : rows = i, cols = j
# ----------------------------------------------------------------------
@triton.jit
def triton_trimul_matmul_kernel(
    A_ptr,               # *fp16, shape (B,H,N,N) rows=i cols=k
    B_ptr,               # *fp16, shape (B,H,N,N) rows=k cols=j
    C_ptr,               # *fp16, shape (B,H,N,N) rows=i cols=j
    BATCH, N, HIDDEN,    # int32 scalars
    stride_A_b, stride_A_h, stride_A_i, stride_A_k,
    stride_B_b, stride_B_h, stride_B_k, stride_B_j,
    stride_C_b, stride_C_h, stride_C_i, stride_C_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Grid layout:
    #   pid0 : combined batch*hidden index  (bh = b * HIDDEN + h)
    #   pid1 : block row index i
    #   pid2 : block column index j
    # ------------------------------------------------------------------
    bh = tl.program_id(0)
    block_i = tl.program_id(1)
    block_j = tl.program_id(2)

    # decode batch and hidden id
    b = bh // HIDDEN
    h = bh % HIDDEN

    # start offsets for the current tile
    i0 = block_i * BLOCK_M
    j0 = block_j * BLOCK_N

    # accumulate in float32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension in BLOCK_K sized chunks
    for k_offset in range(0, N, BLOCK_K):
        # ------------------------------------------------------------------
        # Load A tile (M×K)   -> rows i, cols k
        # Load B tile (K×N)   -> rows k, cols j
        # ------------------------------------------------------------------
        a_i = i0 + tl.arange(0, BLOCK_M)
        a_k = k_offset + tl.arange(0, BLOCK_K)
        b_k = k_offset + tl.arange(0, BLOCK_K)
        b_j = j0 + tl.arange(0, BLOCK_N)

        # masks for out‑of‑bounds loads
        mask_a_i = a_i < N
        mask_a_k = a_k < N
        mask_b_k = b_k < N
        mask_b_j = b_j < N

        # pointer arithmetic for A
        a_ptrs = (A_ptr
                  + b * stride_A_b
                  + h * stride_A_h
                  + a_i[:, None] * stride_A_i
                  + a_k[None, :] * stride_A_k)
        # pointer arithmetic for B (already transposed)
        b_ptrs = (B_ptr
                  + b * stride_B_b
                  + h * stride_B_h
                  + b_k[:, None] * stride_B_k
                  + b_j[None, :] * stride_B_j)

        a = tl.load(a_ptrs,
                    mask=mask_a_i[:, None] & mask_a_k[None, :],
                    other=tl.float16(0.0))
        b = tl.load(b_ptrs,
                    mask=mask_b_k[:, None] & mask_b_j[None, :],
                    other=tl.float16(0.0))

        # dot produces fp32 accumulator
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write result C tile (M×N) → rows i, cols j
    # ------------------------------------------------------------------
    c_i = i0 + tl.arange(0, BLOCK_M)
    c_j = j0 + tl.arange(0, BLOCK_N)

    mask_c_i = c_i < N
    mask_c_j = c_j < N
    mask_c = mask_c_i[:, None] & mask_c_j[None, :]

    c_ptrs = (C_ptr
              + b * stride_C_b
              + h * stride_C_h
              + c_i[:, None] * stride_C_i
              + c_j[None, :] * stride_C_j)

    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (x, mask, weights, config)
            x      – torch.Tensor [B, N, N, C] (float32)
            mask   – torch.Tensor [B, N, N] (bool/float) or None
            weights– dict of torch.Tensors containing all layer parameters
            config – dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input layer‑norm on the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]     # (hidden_dim, dim)

    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3. Gating logits (sigmoid)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]   # (hidden_dim, dim)
    right_gate_w = weights["right_gate.weight"]  # (hidden_dim, dim)
    out_gate_w   = weights["out_gate.weight"]    # (hidden_dim, dim)

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 4. Apply gates and (optional) pairwise mask
    # ------------------------------------------------------------------
    left  = left  * left_gate
    right = right * right_gate

    if not nomask:
        # mask is [B,N,N]; broadcast over hidden dimension
        mask = mask.to(x.dtype).unsqueeze(-1)         # [B,N,N,1]
        left  = left  * mask
        right = right * mask
    # else: treat as all‑ones -> no operation needed

    # ------------------------------------------------------------------
    # 5. Core N³ reduction using Triton
    #    Convert to fp16 for the matmul kernel
    # ------------------------------------------------------------------
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Views required for A @ B.T
    #   A: rows=i, cols=k  → (B, hidden, N, N)
    #   B: rows=k, cols=j  → (B, hidden, N, N) with axes (k, j)
    A = left_fp16.permute(0, 3, 1, 2)   # (B, H, N, N)  rows=i, cols=k
    B = right_fp16.permute(0, 3, 2, 1)  # (B, H, N, N)  rows=k, cols=j

    B_batch, H_dim, N_dim, _ = A.shape
    # allocate output buffer (fp16) in the same layout as A (rows=i, cols=j)
    C = torch.empty_like(A, dtype=torch.float16)   # (B,H,N,N)

    # strides in elements (not bytes)
    sA_b, sA_h, sA_i, sA_k = A.stride()
    sB_b, sB_h, sB_k, sB_j = B.stride()
    sC_b, sC_h, sC_i, sC_j = C.stride()

    # tile sizes – chosen to fit H100 registers and shared memory
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B_batch * H_dim,                     # combined batch*hidden axis
        (N_dim + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N_dim + BLOCK_N - 1) // BLOCK_N    # j‑tiles
    )

    triton_trimul_matmul_kernel[grid](
        A, B, C,
        B_batch, N_dim, H_dim,
        sA_b, sA_h, sA_i, sA_k,
        sB_b, sB_h, sB_k, sB_j,
        sC_b, sC_h, sC_i, sC_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8, num_stages=2
    )
    torch.cuda.synchronize()

    # Recover shape (B,N,N,hidden_dim)
    out_hidden = C.permute(0, 2, 3, 1).to(torch.float32)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6. Output layer‑norm, gating and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_hidden = torch.nn.functional.layer_norm(
        out_hidden, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b, eps=1e-5
    )

    # element‑wise multiply by the output gate (same shape as out_hidden)
    out_hidden = out_hidden * out_gate

    to_out_w = weights["to_out.weight"]   # (dim, hidden_dim)
    out_final = torch.nn.functional.linear(out_hidden, to_out_w)  # [B,N,N,dim]

    return out_final
