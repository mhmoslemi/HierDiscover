def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl
    import torch.nn.functional as F

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"] # maybe config["hidden_dim"]

    # Preliminary layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Linear projections
    left = F.linear(x, weights["left_proj.weight"]) # shape [bs, N,N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask if needed
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1) # shape [bs,N,N,1]
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for kernel, compute heavy multiplication
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    # allocate output
    out_fp16 = torch.empty((bs, N, N, hidden_dim), dtype=torch.float16, device=device)

    # compute strides
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_fp16.stride()

    # Choose block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = ( (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N, bs * hidden_dim)

    # Define kernel
    @triton.jit
    def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                       B, N, H,
                       stride_l_b, stride_l_i, stride_l_k, stride_l_h,
                       stride_r_b, stride_r_j, stride_r_k, stride_r_h,
                       stride_o_b, stride_o_i, stride_o_j, stride_o_h,
                       BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
        pid_bh = tl.program_id(2)
        pid_i = tl.program_id(0)
        pid_j = tl.program_id(1)

        # Decompose batch and hidden
        b = pid_bh // H
        h = pid_bh % H

        i_offset = pid_i * BLOCK_M
        j_offset = pid_j * BLOCK_N

        # Offsets vectors
        offs_i = i_offset + tl.arange(0, BLOCK_M)
        offs_j = j_offset + tl.arange(0, BLOCK_N)

        mask_i = offs_i < N
        mask_j = offs_j < N

        # Base pointers for this batch and hidden
        base_left = left_ptr + b * stride_l_b + h * stride_l_h
        base_right = right_ptr + b * stride_r_b + h * stride_r_h
        base_out = out_ptr + b * stride_o_b + h * stride_o_h

        # Accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # Loop over k tiles
        for k in range(0, tl.cdiv(N, BLOCK_K)):
            k_offset = k * BLOCK_K
            offs_k = k_offset + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # Compute pointers
            a_ptrs = base_left + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
            b_ptrs = base_right + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j

            # Load with masks
            a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
            b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

            # Accumulate
            acc += tl.dot(a, b)

        # Store result
        out_ptrs = base_out + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
        tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    # launch kernel
    tri_mul_kernel[grid](
        left_fp16, right_fp16, out_fp16,
        bs, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Convert output to float32
    out = out_fp16.to(torch.float32)

    # Apply output gate
    out = out * out_gate

    # Apply output layernorm
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # Final linear projection
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
custom_kernel
==============

Implements the *outgoing* TriMul operation (AlphaFold‑3) with a fused
triton kernel for the cubic `einsum` step:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

The surrounding linear layers, LayerNorms and sigmoids are performed
with regular PyTorch ops.  The heavy `N³` computation is turned into a
batched matrix multiplication (`left @ rightᵀ`) across the hidden
dimension and executed in FP16 with FP32 accumulation for both speed
and numerical stability.

Supported shapes:
    - batch size `B`  ∈ {1,2}
    - sequence length `N` ∈ {128,256,512,1024}
    - input channel `dim` ∈ {128,384}
    - hidden dimension `hidden_dim` = 128 (as used in AlphaFold‑3)

The kernel works for both masked and un‑masked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM performing the tri‑multiplicative update
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Matrix‑multiply per hidden slice:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]   (FP16 inputs, FP32 acc)"""

    # ------------------------------------------------------------------
    # Program IDs
    # ------------------------------------------------------------------
    pid_bh = tl.program_id(2)          # combined batch*hidden index
    pid_i  = tl.program_id(0)          # row tile index (i)
    pid_j  = tl.program_id(1)          # column tile index (j)

    # ------------------------------------------------------------------
    # De‑compose batch (b) and hidden (h) from the flattened index
    # ------------------------------------------------------------------
    b = pid_bh // H
    h = pid_bh % H

    # ------------------------------------------------------------------
    # Starting offsets for this tile
    # ------------------------------------------------------------------
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    offs_i = i_offset + tl.arange(0, BLOCK_M)
    offs_j = j_offset + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Base pointers for this (b,h) slice
    # ------------------------------------------------------------------
    base_left  = left_ptr  + b * stride_l_b + h * stride_l_h
    base_right = right_ptr + b * stride_r_b + h * stride_r_h
    base_out   = out_ptr   + b * stride_o_b + h * stride_o_h

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in blocks
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers for the current K‑tile
        # A: left  [i,k]  (shape BLOCK_M × BLOCK_K)
        a_ptrs = base_left + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        # B: rightᵀ [k,j]  (shape BLOCK_K × BLOCK_N)
        b_ptrs = base_right + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j

        # Load with masking, promote to FP32 for the dot
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) · (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    out_ptrs = base_out + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point expected by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul module.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor or None, shape [B, N, N]
        - weights      : dict of model tensors (see keys below)
        - config       : dict containing at least ``dim`` and ``hidden_dim``.

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # Helper: LayerNorm using functional API
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # Optional mask (mask shape [B, N, N]; broadcast on hidden dim)
    # --------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        # mask is float (0/1); ensure same dtype as left/right
        mask = mask.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
        left  = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # Gating (sigmoid) – still in float32
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # Cast to FP16 for the cubic multiplication (saves bandwidth)
    # --------------------------------------------------------------
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16   = torch.empty((B, N, N, hidden_dim),
                             dtype=torch.float16,
                             device=device)

    # --------------------------------------------------------------
    # Strides (in element units) – needed by the Triton kernel
    # --------------------------------------------------------------
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_fp16.stride()

    # --------------------------------------------------------------
    # Kernel launch configuration
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # j‑tiles
        B * hidden_dim                         # batch‑hidden tiles
    )

    # --------------------------------------------------------------
    # Triton kernel launch
    # --------------------------------------------------------------
    _tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # a modest number works well on Hopper
    )

    # --------------------------------------------------------------
    # Cast back to FP32 for the remaining layers
    # --------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    # --------------------------------------------------------------
    # Output gating, layer‑norm and final projection (as in reference)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out

**Explanation of key choices**

* **Triton kernel** – Implements a batched GEMM where each hidden slice
  (`h`) is treated as an independent batch.  The kernel loads tiles of
  `left` (`A`) and `rightᵀ` (`B`) in FP16, promotes them to FP32, and
  accumulates with `tl.dot`.  Masks guard out‑of‑bounds loads and stores.

* **Mixed‑precision** – All expensive memory‑bound work (`left/right`
  and the cubic multiplication) runs in FP16, greatly reducing traffic
  on H100 while keeping a FP32 accumulator for numerical safety.

* **PyTorch for cheap ops** – LayerNorm, linear layers, and sigmoids are
  executed with the highly‑optimized torch kernels.  This keeps the
  Triton kernel focused on the only heavy `O(N³·hidden)` step.

* **Flexibility** – The implementation works for any `B ∈ {1,2}`,
  `N ∈ {128,…,1024}`, `dim ∈ {128,384}`, and respects the optional mask.
  The grid scales with `B·hidden_dim`, so the kernel automatically
  covers all required batch‑hidden slices.

The function `custom_kernel` can be called directly with the described
