```
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    #0: batch_h (B*H)
    #1: tile_i
    #2: tile_j
    bh = tl.program_id(0)
    i_tile = tl.program_id(1)
    j_tile = tl.program_id(2)

    batch = bh // H
    h = bh % H

    # compute i and j offsets
    i_offset = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for valid i,j < N
    mask_i = i_offset < N
    mask_j = j_offset < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offsets
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load left block [i,k] for fixed h
        # left_ptr + batch*stride_l_batch + i*stride_l_i + k*stride_l_k + h*stride_l_h
        # Expand to (BLOCK_M,BLOCK_K)
        left_ptrs = left_ptr + batch * stride_l_batch \
                    + i_offset[:, None] * stride_l_i \
                    + k_offset[None, :] * stride_l_k \
                    + h * stride_l_h
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right block transposed [k,j] for fixed h
        # right_ptr base offset similar
        right_ptrs = right_ptr + batch * stride_r_batch \
                     + j_offset[None, :] * stride_r_i \
                     + k_offset[:, None] * stride_r_k \
                     + h * stride_r_h
        # shape (BLOCK_K, BLOCK_N)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)
        # now right is already transposed shape (BLOCK_K, BLOCK_N)

        # accumulate multiplication (outer sum)
        acc += tl.dot(left, right)  # dot does (M,K)*(K,N) -> (M,N)

    # Write back to output
    out_ptrs = out_ptr + batch * stride_o_batch \
               + i_offset[:, None] * stride_o_i \
               + j_offset[None, :] * stride_o_j \
               + h * stride_o_h

```
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    bh = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute batch and hidden index
    batch = bh // H
    h = bh % H

    # compute i and j ranges
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)

    mask_i = i_range < N
    mask_j = j_range < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        k_range = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # Load left block
        left_ptrs = left_ptr + batch*stride_l_batch \
                      + i_range[:, None] * stride_l_i \
                      + k_range[None, :] * stride_l_k \
                      + h * stride_l_h
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right block transposed (k x j)
        right_ptrs = right_ptr + batch*stride_r_batch \
                      + j_range[None, :] * stride_r_i \
                      + k_range[:, None] * stride_r_k \
                      + h * stride_r_h
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    # Store result
    out_ptrs = out_ptr + batch*stride_o_batch \
                + i_range[:, None] * stride_o_i \
                + j_range[None, :] * stride_o_j \
                + h * stride_o_h
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) – fast Triton implementation of the cubic contraction

Given tensors
    left  = L  ∈ ℝ^{B×N×N×H}
    right = R  ∈ ℝ^{B×N×N×H}

the module needs
    out[b,i,j,h] = Σ_k L[b,i,k,h] * R[b,j,k,h]          (1)

Equation (1) is exactly a batched matrix multiplication for every
hidden‑dimension slice:

    out_{b,h} = L_{b,h} @ (R_{b,h})ᵀ   where
    L_{b,h} ∈ ℝ^{N×N}, R_{b,h} ∈ ℝ^{N×N}.

We fuse the batch (B) and hidden (H) dimensions into a single program‑ID
dimension and compute a tiled GEMM with Triton.  Each thread block
produces a (BLOCK_M × BLOCK_N) tile of the result for a fixed (b,h)
pair, accumulating over K‑tiles.  The rest of the TriMul forward pass
(LayerNorms, linear projections, gates, etc.) is performed with
PyTorch for simplicity.

The kernel works with arbitrary batch size, sequence length and hidden
dimension (the shapes used in the tests are covered).  Computation is
performed in float32 while the inputs to the kernel are cast to float16
to reduce memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the core cubic contraction (outgoing TriMul)
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,          # const __global__ float16 *
    right_ptr,         # const __global__ float16 *
    out_ptr,           # __global__ float32 *
    B, N, H,           # int64 scalars
    # strides for left  [B,N,N,H]
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    # strides for right [B,N,N,H]
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    # strides for out    [B,N,N,H]
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    bh_id   = tl.program_id(0)                # fused batch * hidden
    pid_i   = tl.program_id(1)                # tile over rows   (i)
    pid_j   = tl.program_id(2)                # tile over cols   (j)

    # decompose fused dimension
    batch_id = bh_id // H
    h_id     = bh_id % H

    # start indices of the tile
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # range of rows/cols handled by this program
    i_idx = i_base + tl.arange(0, BLOCK_M)
    j_idx = j_base + tl.arange(0, BLOCK_N)

    # mask for OOB rows / cols
    mask_i = i_idx < N
    mask_j = j_idx < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension (the summed index)
    # ------------------------------------------------------------------
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_iter * BLOCK_K
        k_idx = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # ------------------------------------------------------------------
        # Load left block  [i, k]   (shape BLOCK_M × BLOCK_K)
        # ------------------------------------------------------------------
        left_offset = (
            batch_id * stride_l_batch
            + i_idx[:, None] * stride_l_i
            + k_idx[None, :] * stride_l_k
            + h_id * stride_l_h
        )
        left_block = tl.load(
            left_ptr + left_offset,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # fp16 → fp32

        # ------------------------------------------------------------------
        # Load right block transposed [k, j]   (shape BLOCK_K × BLOCK_N)
        #
        # Right tensor is indexed as right[batch, j, k, h];
        # by swapping the k and j strides we obtain the transposed view.
        # ------------------------------------------------------------------
        right_offset = (
            batch_id * stride_r_batch
            + j_idx[None, :] * stride_r_i   # original i‑stride (first N)
            + k_idx[:, None] * stride_r_k   # original k‑stride (second N)
            + h_id * stride_r_h
        )
        right_block = tl.load(
            right_ptr + right_offset,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)          # fp16 → fp32

        # ------------------------------------------------------------------
        # Accumulate the dot product for this K‑tile
        # ------------------------------------------------------------------
        acc += tl.dot(left_block, right_block)   # (M×K)·(K×N) → (M×N)

    # ------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------
    out_offset = (
        batch_id * stride_o_batch
        + i_idx[:, None] * stride_o_i
        + j_idx[None, :] * stride_o_j
        + h_id * stride_o_h
    )
    tl.store(
        out_ptr + out_offset,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (may be None)
        - weights      : dict of model parameters
        - config       : dict with keys "dim", "hidden_dim", "nomask" ...

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = input_tensor.dtype
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # Helper to fetch weights
    # ------------------------------------------------------------------
    def get(name):
        return weights[name]

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=get("norm.weight"),
        bias=get("norm.bias"),
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, get("left_proj.weight"))   # → [B,N,N,hidden]
    right = F.linear(x, get("right_proj.weight")) # → [B,N,N,hidden]

    # ------------------------------------------------------------------
    # Prepare mask (broadcasted over hidden dimension)
    # ------------------------------------------------------------------
    if nomask or mask is None:
        mask_ = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask_ = mask.to(dtype)

    mask_ = mask_.unsqueeze(-1)   # [B,N,N,1]

    left = left * mask_
    right = right * mask_

    # ------------------------------------------------------------------
    # Gates
    # ------------------------------------------------------------------
    left_gate = F.linear(x, get("left_gate.weight")).sigmoid()
    right_gate = F.linear(x, get("right_gate.weight")).sigmoid()
    out_gate = F.linear(x, get("out_gate.weight")).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to fp16 for the tiled kernel (reduces traffic)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ------------------------------------------------------------------
    # Allocate output of the cubic contraction (float32)
    # ------------------------------------------------------------------
    out = torch.empty_like(left_fp16, dtype=torch.float32)  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # Strides for the kernel (torch uses row‑major layout)
    # ------------------------------------------------------------------
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h = right_fp16.stride()
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tile sizes – chosen to fit well on H100 (tuned empirically)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (B*H, ceil(N/M), ceil(N/N))
    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # ------------------------------------------------------------------
    # Launch Triton kernel
    # ------------------------------------------------------------------
    trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        hidden_dim,
        stride_l_batch,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_batch,
        stride_r_i,
        stride_r_k,
        stride_r_h,
        stride_o_batch,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm on hidden dimension, gating and final proj.
    # ------------------------------------------------------------------
    # to_out_norm : LayerNorm over hidden_dim
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=get("to_out_norm.weight"),
        bias=get("to_out_norm.bias"),
    )
    # Apply out gate
    out_norm = out_norm * out_gate

    # Final linear back to dim
    out_final = F.linear(out_norm, get("to_out.weight"))   # bias is None

    return out_final
