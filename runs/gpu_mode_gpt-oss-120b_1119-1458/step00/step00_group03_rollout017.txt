def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # compute batch index and head index
    batch_idx = pid_bh // H
    head_idx = pid_bh % H

    # compute offsets
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out-of-bounds
    i_mask = i_off < N
    j_mask = j_off < N

    # compute base offsets
    base = batch_idx * stride_b + head_idx  # since stride for hidden_dim is 1 (contiguous)
    # left: stride_i for i dimension, stride_k for k dimension
    # right: same but need to handle k and j dims swapped: row= k, col= j

    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        k_mask = cur_k < N

        # loads
        left_ptrs = left_ptr + base + i_off[:, None] * stride_i + cur_k[None, :] * stride_k
        # left shape (BLOCK_M, BLOCK_K)
        left_tile = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float32)

        # right pointer: row=cur_k (k), col=j_off (j)
        right_ptrs = right_ptr + base + cur_k[:, None] * stride_k + j_off[None, :] * stride_i
        # shape (BLOCK_K, BLOCK_N)
        right_tile = tl.load(right_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0).to(tl.float32)

        # matmul left_tile @ right_tile (size MxK times KxN)
        acc += tl.dot(left_tile, right_tile)
    # store result
    out_ptrs = out_ptr + base + i_off[:, None] * stride_i + j_off[None, :] * stride_k
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])
```

"""
TriMul (outgoing version) – Triton implementation

This kernel implements the heavy N³ contraction in the TriMul module:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

For each batch `b` and hidden dimension `h` we treat `left` and `right`
as two N×N matrices (the hidden dimension is sliced out) and compute
the matrix product `left @ rightᵀ`.  The surrounding LayerNorms,
Linear projections, gating and the final Linear are performed with
regular PyTorch ops; only the contraction is fused into a Triton kernel.

The kernel works on tensors with layout (B, N, N, H) (contiguous in memory:
stride = (N·N·H, N·H, H, 1)).  The program id space is:
   pid_i : block index over dimension i
   pid_j : block index over dimension j
   pid_bh: combined batch‑hidden index (B·H blocks)
Each block computes a tile of size BLOCK_M × BLOCK_N for a fixed (b, h)
pair, accumulating over the sequence dimension k in tiles of size BLOCK_K.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel that performs the core contraction
# --------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # runtime values
    stride_b, stride_i, stride_k,           # strides of the (B,N,N,H) tensors
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # ------------------------------------------------------------------
    # 1) identify which (batch, hidden) slice we are processing
    # ------------------------------------------------------------------
    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile
    pid_bh = tl.program_id(2)         # combined batch·hidden index

    batch_idx = pid_bh // H
    hidden_idx = pid_bh % H

    # ------------------------------------------------------------------
    # 2) compute tile offsets for i and j dimensions
    # ------------------------------------------------------------------
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_off < N
    mask_j = j_off < N

    # base address for the selected (b, h) slice (hidden dim stride = 1)
    base = batch_idx * stride_b + hidden_idx

    # ------------------------------------------------------------------
    # 3) accumulator initialised to zero (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 4) loop over the reduction dimension k
    # ------------------------------------------------------------------
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # left tile   : shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + base \
                    + i_off[:, None] * stride_i \
                    + cur_k[None, :] * stride_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile  : shape (BLOCK_K, BLOCK_N)  (k is the row, j the column)
        right_ptrs = right_ptr + base \
                     + cur_k[:, None] * stride_k \
                     + j_off[None, :] * stride_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # matrix multiplication for the current k‑tile
        # left   (M×K)  @  right (K×N)  ->  (M×N)
        acc += tl.dot(left, right)

    # ------------------------------------------------------------------
    # 5) write the result tile back to global memory
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + base \
               + i_off[:, None] * stride_i \
               + j_off[None, :] * stride_k   # stride of j‑dim = stride_k (hidden dim is innermost)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------
# Entry point – the full forward pass of TriMul (outgoing)
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, D] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights      : dict of torch.Tensor containing all model weights
            - config       : dict with keys
                * "dim"         : D (input channel size)
                * "hidden_dim" : H (intermediate channel size)
                * "nomask"      : bool – if True, mask is ignored
                * "eps"         : optional float for LayerNorm (default 1e-6)

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]                # D
    hidden_dim = config["hidden_dim"]  # H
    eps = config.get("eps", 1e-6)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # convenience handles for weights
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over channel dimension D
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b,
                     eps=eps)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)          # [B,N,N,H]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Optional binary mask (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_ = mask.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core contraction – Triton kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # allocate output buffer for the contraction
    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # strides for the (B,N,N,H) layout (contiguous)
    stride_b, stride_i, stride_k, stride_h = left.stride()   # stride_h should be 1
    # we only need stride_b, stride_i, stride_k inside the kernel
    # (stride_h == 1 and is folded into the base address)

    # block sizes (tuned for H100 – adjust if needed)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # j‑tiles
        B * hidden_dim                         # batch × hidden dimension
    )

    # launch the kernel
    tri_mul_kernel[grid](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        out_ptr=out.data_ptr(),
        B=B,
        N=N,
        H=hidden_dim,
        stride_b=stride_b,
        stride_i=stride_i,
        stride_k=stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,   # typical choice for this shape
    )
    # make sure kernel has finished before we continue
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(out,
                            normalized_shape=(hidden_dim,),
                            weight=to_out_norm_w,
                            bias=to_out_norm_b,
                            eps=eps)

    out_norm = out_norm * out_gate
    output = F.linear(out_norm, to_out_w)    # -> [B,N,N,dim]

    return output
