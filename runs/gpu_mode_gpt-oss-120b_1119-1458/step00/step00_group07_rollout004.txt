    left_ptr, right_ptr, out_ptr,
    B, N, H,

    stride_rr0, stride_rr1, stride_rr2, stride_rr3,

    pid1 = tl.program_id(1)
    pid2 = tl.program_id(2)

    b = pid0 // H
    h = pid0 % H

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lr0, stride_lr1, stride_lr2, stride_lr3,
    stride_rr0, stride_rr1, stride_rr2, stride_rr3,
    stride_or0, stride_or1, stride_or2, stride_or3,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Triton kernel for the core TriMul contraction:
        out[b,i,j,h] = sum_{k=0}^{N-1} left[b,i,k,h] * right[b,j,k,h]

    The kernel processes one (batch, hidden) pair per program-id (axis 0)
    and tiles the (i,j) output plane with BLOCK_M x BLOCK_N tiles.
    The reduction over k is blocked with size BLOCK_K and accumulated in FP32.
    """
    pid0 = tl.program_id(0)   # batch * hidden
    pid1 = tl.program_id(1)   # tile row (i)
    pid2 = tl.program_id(2)   # tile col (j)

    b = pid0 // H
    h = pid0 % H

    # tile start indices
    i0 = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    j0 = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i0 < N
    mask_j = j0 < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for K in range(num_k):
        k_start = K * BLOCK_K
        k_idx = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # left[b, i, k, h]  (M,K)
        left_offset = (
            b * stride_lr0 + h * stride_lr3
            + i0[:, None] * stride_lr1
            + k_idx[None, :] * stride_lr2
        )
        left = tl.load(
            left_ptr + left_offset,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right transposed: right_T[b, k, j, h] = right[b, j, k, h]  (K,N)
        right_offset = (
            b * stride_rr0 + h * stride_rr3
            + k_idx[:, None] * stride_rr1
            + j0[None, :] * stride_rr2
        )
        right = tl.load(
            right_ptr + right_offset,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        acc += tl.dot(left, right)   # (M,N) = (M,K) @ (K,N)

    # write result
    out_offset = (
        b * stride_or0 + h * stride_or3
        + i0[:, None] * stride_or1
        + j0[None, :] * stride_or2
    )
    tl.store(
        out_ptr + out_offset,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the outgoing
    Triangular Multiplicative Update (TriMul) as described in AlphaFold3.
    The heavy 4‑D contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is off‑loaded to a fused Triton kernel. All other linear /
    LayerNorm / gating operations are executed with PyTorch.
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    # ------------------------------------------------------------------
    # LayerNorm on the input (normalize over `dim` channel)
    # ------------------------------------------------------------------
    # weights for LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Apply optional mask
    # ------------------------------------------------------------------
    if mask is not None:
        # mask shape: [B, N, N] (bool or 0/1)
        mask = mask.to(left.dtype).unsqueeze(-1)   # broadcast on channel dim
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to fp16 for the Triton kernel
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Output placeholder (fp16)
    out_fp16 = torch.empty_like(left_fp16)

    # ------------------------------------------------------------------
    # Launch Triton kernel for the O(N³) contraction
    # ------------------------------------------------------------------
    B, N, _, _ = left_fp16.shape
    # pick block sizes (tuned for H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        left_fp16.stride(3),
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        right_fp16.stride(3),
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_stages=3,
        num_warps=8,
    )
    # Ensure kernel finished before proceeding
    torch.cuda.synchronize()

    # Convert back to FP32 for the remaining operations
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # Output layer norm (over hidden_dim)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Apply output gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # shape [B,N,N,dim]

    return out

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,                # pointers
    B, N, H,                                     # batch, seq_len, hidden_dim
    stride_lr0, stride_lr1, stride_lr2, stride_lr3,   # left strides
    stride_rr0, stride_rr1, stride_rr2, stride_rr3,   # right strides
    stride_or0, stride_or1, stride_or2, stride_or3,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Triton kernel implementing the 4‑D einsum:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    One program‑id on axis‑0 processes a (batch, hidden) pair.
    Axes‑1 and ‑2 tile the (i,j) output plane with BLOCK_M × BLOCK_N.
    The reduction over k is blocked with size BLOCK_K.
    """
    pid0 = tl.program_id(0)   # combined batch * hidden index
    pid1 = tl.program_id(1)   # tile row (i)
    pid2 = tl.program_id(2)   # tile col (j)

    # Decode batch and hidden index
    b = pid0 // H
    h = pid0 % H

    # Tile start indices
    i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        k_start = kb * BLOCK_K
        k = k_start + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left[b, i, k, h]  → shape (BLOCK_M, BLOCK_K)
        left_off = (
            b * stride_lr0 + h * stride_lr3
            + i[:, None] * stride_lr1
            + k[None, :] * stride_lr2
        )
        left = tl.load(
            left_ptr + left_off,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right_T[b, k, j, h] = right[b, j, k, h]  → shape (BLOCK_K, BLOCK_N)
        right_off = (
            b * stride_rr0 + h * stride_rr3
            + k[:, None] * stride_rr1
            + j[None, :] * stride_rr2
        )
        right = tl.load(
            right_ptr + right_off,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # Perform the dot product over k
        acc += tl.dot(left, right)          # (BLOCK_M, BLOCK_N) in FP32

    # Store the result (cast back to FP16)
    out_off = (
        b * stride_or0 + h * stride_or3
        + i[:, None] * stride_or1
        + j[None, :] * stride_or2
    )
    tl.store(
        out_ptr + out_off,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]  (float32)
        - mask         : [B, N, N]       (float32/bool) – optional but always provided
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """

    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    # ------------------------------------------------------------------
    # LayerNorm over the feature dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (broadcast on the channel dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to FP16 for the Triton kernel
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Output buffer (FP16)
    out_fp16 = torch.empty_like(left_fp16)

    # ------------------------------------------------------------------
    # Launch the Triton kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left_fp16.shape
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                     # axis 0 : (batch * hidden)
        triton.cdiv(N, BLOCK_M),           # axis 1 : i‑tiles
        triton.cdiv(N, BLOCK_N),           # axis 2 : j‑tiles
    )
    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        left_fp16.stride(3),
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        right_fp16.stride(3),
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        out_fp16.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_stages=3,
        num_warps=8,
    )
    # Synchronise to make sure the kernel has finished
    torch.cuda.synchronize()

    # Cast back to FP32 for the remaining layers
    out = out_fp16.to(torch.float32)          # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Apply output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
