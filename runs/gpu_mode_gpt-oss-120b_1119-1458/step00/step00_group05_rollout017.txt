Define kernel:

```
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # compute batch, block_m, block_n indices
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_batch = pid // (num_pid_m * num_pid_n)
    pid_rem = pid % (num_pid_m * num_pid_n)
    pid_m = pid_rem // num_pid_n
    pid_n = pid_rem % num_pid_n

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for boundary
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # compute pointers for A and B_T
        a_ptrs = a_ptr + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + pid_batch * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        # load with mask
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # convert to float32 to accumulate
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)
    # store result, with mask for m,n
    c_ptrs = c_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    # LN on x
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)  # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)
    # Apply mask if needed
    if not nomask:
        # mask shape [B,N,N]; broadcast
        mask_expand = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_expand
        right = right * mask_expand
    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))
    left = left * left_gate
    right = right * right_gate

    # Cast to half for matmul
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    B, N, _, H = left_f16.shape
    batch_dim = B * H
    # Reshape: (B, N, N, H) -> (B*H, N, N) with hidden as batch
    left_reshape = left_f16.permute(0, 3, 1, 2).contiguous().view(batch_dim, N, N)
    right_reshape = right_f16.permute(0, 3, 1, 2).contiguous().view(batch_dim, N, N)
    # Transpose right for multiplication
    right_T = right_reshape.transpose(-2, -1).contiguous()  # (batch_dim, N, N)

    # Allocate output
    out_f16 = torch.empty((batch_dim, N, N), dtype=torch.float16, device=device)

    # stride values
    stride_a_batch, stride_a_m, stride_a_k = left_reshape.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_T.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_f16.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch_dim * grid_m * grid_n

    # Launch kernel
    batched_matmul_kernel[total_blocks](
        left_reshape, right_T, out_f16,
        batch_dim, N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8
    )

    # Reshape back to (B,N,N,H)
    out = out_f16.view(B, H, N, N).permute(0, 2, 3, 1)  # (B,N,N,hidden_dim)
    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    # Multiply by out_gate
    out = out * out_gate
    # final linear
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)  # shape (B,N,N,dim)
    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched matrix multiplication C = A @ B, where
    A: (batch, M, K)
    B: (batch, K, N)
    C: (batch, M, N)
    Computation is done in FP16 for A/B and accumulated in FP32.
    """
    pid = tl.program_id(0)

    # Decompose the 1‑D program id into batch, M‑tile, N‑tile indices.
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_batch = pid // (num_pid_m * num_pid_n)
    pid_rem   = pid % (num_pid_m * num_pid_n)
    pid_m = pid_rem // num_pid_n
    pid_n = pid_rem % num_pid_n

    # Offsets for the output tile.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension.
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for the current tile of A and B.
        a_ptrs = a_ptr + pid_batch * stride_a_batch \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k

        b_ptrs = b_ptr + pid_batch * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Convert to FP32, do dot product and accumulate.
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # Write back with mask.
    c_ptrs = c_ptr + pid_batch * stride_c_batch \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    The heavy pairwise multiplication over the sequence dimension is
    performed by a custom Triton kernel that computes a batched
    matrix multiplication (A @ B^T) for every hidden channel.
    All other operations (LayerNorm, linear projections, gating)
    are expressed with PyTorch for simplicity.
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    # Unpack arguments.
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------------------------------------------------------------------
    # 1. Input layer‑norm (dim‑wise)
    # ---------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ---------------------------------------------------------------------
    # 2. Linear projections (dim -> hidden_dim)
    # ---------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ---------------------------------------------------------------------
    # 3. Optional mask.
    # ---------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] -> broadcast over hidden dimension.
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left  = left  * mask_exp
        right = right * mask_exp

    # ---------------------------------------------------------------------
    # 4. Gating (sigmoid of another linear projection)
    # ---------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left  * left_gate
    right = right * right_gate

    # ---------------------------------------------------------------------
    # 5. Batched matrix multiplication: out = left @ right^T  (per hidden dim)
    #    We fold the hidden dimension into the batch dimension and run a
    #    custom Triton kernel.
    # ---------------------------------------------------------------------
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    B, N, _, H = left_f16.shape          # B=batch, H=hidden_dim
    batch_dim = B * H

    # (B, N, N, H) -> (B*H, N, N)
    left_reshape  = left_f16.permute(0, 3, 1, 2).contiguous().view(batch_dim, N, N)
    right_reshape = right_f16.permute(0, 3, 1, 2).contiguous().view(batch_dim, N, N)

    # Transpose the second operand so that we can perform a regular matmul.
    right_T = right_reshape.transpose(-2, -1).contiguous()

    # Allocate output tensor.
    out_f16 = torch.empty((batch_dim, N, N), dtype=torch.float16, device=device)

    # Strides for the three tensors.
    stride_a_batch, stride_a_m, stride_a_k = left_reshape.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_T.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_f16.stride()

    # Block sizes – chosen to balance shared‑memory usage and occupancy.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid definition.
    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch_dim * grid_m * grid_n

    # Launch the Triton kernel.
    batched_matmul_kernel[total_blocks](
        left_reshape, right_T, out_f16,
        batch_dim, N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8
    )

    # Bring the result back to the original layout: (B, N, N, hidden_dim)
    out = out_f16.view(B, H, N, N).permute(0, 2, 3, 1)   # float16

    # ---------------------------------------------------------------------
    # 6. Output layer‑norm over hidden_dim.
    # ---------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = out.to(torch.float32)  # LayerNorm works in FP32 for stability.
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # ---------------------------------------------------------------------
    # 7. Apply the output gate and final linear projection (hidden_dim -> dim).
    # ---------------------------------------------------------------------
    out = out * out_gate                     # point‑wise gate (float32)
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)      # final projection

    return out

"""
TriMul (outgoing) forward pass.
The O(N³·hidden_dim) pairwise multiplication is expressed as a
batched matrix multiplication: for each hidden channel d we compute
    out[..., i, j, d] = left[..., i, :, d] @ right[..., j, :, d].T .
By folding the hidden dimension into the batch dimension we obtain a
batch of (B·hidden_dim) independent (N×N) @ (N×N) matmuls.
The heavy computation is off‑loaded to a custom Triton kernel that
operates on FP16 inputs and accumulates in FP32. All other operations
(layer‑norms, linear projections, gating) are performed with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    C = A @ B   with
        A : (batch, M, K)   FP16
        B : (batch, K, N)   FP16 (already transposed)
        C : (batch, M, N)   FP16  (accumulated in FP32)
    """
    pid = tl.program_id(0)

    # 1‑D pid -> (batch_id, tile_m, tile_n)
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_batch = pid // (num_pid_m * num_pid_n)
    pid_rem   = pid % (num_pid_m * num_pid_n)
    pid_m = pid_rem // num_pid_n
    pid_n = pid_rem % num_pid_n

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

                 + offs_k[None, :] * stride_a_k

                 + offs_k[:, None] * stride_b_k \

    # ------------------------------------------------------------------

    right_T = right_mat.transpose(-2, -1).contiguous()

    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_T.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_f16.stride()

    # Tune block sizes – good trade‑off for H100 FP16 tensor cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: one program per output tile (including batch)
    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch_dim * grid_m * grid_n

    # Launch kernel
    batched_matmul_kernel[total_blocks](
        left_mat, right_T, out_f16,
        batch_dim, N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8      # sufficient for H100 tensor cores
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_f16.view(B, H, N, N).permute(0, 2, 3, 1)   # FP16

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = out.to(torch.float32)   # LayerNorm runs in FP32 for stability
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # 7. Apply output gate and final linear projection (hidden_dim -> dim)
    # ------------------------------------------------------------------

    out = F.linear(out, weights["to_out.weight"])   # final projection, FP32
