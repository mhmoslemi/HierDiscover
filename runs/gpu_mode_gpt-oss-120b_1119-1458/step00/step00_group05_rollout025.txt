    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, D,
    stride_proj_b, stride_proj_i, stride_proj_k, stride_proj_d,

    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_i, stride_out_k, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr

    batch = tl.program_id(axis=1) # batch index

"""
TriMul (outgoing) – fused Triton + PyTorch implementation.

The operator works on a 4‑D tensor X ∈ ℝ^[B, N, N, C] and produces an output
with the same shape.  The forward pass consists of:

1. LayerNorm on the last dimension (C).
2. Two linear projections (left/right) from C → H (hidden_dim) and two
   gating projections (same shape) that are passed through a sigmoid.
3. Optional pairwise mask M ∈ ℝ^[B, N, N] that is broadcast over H.
4. Fuse (proj × gate × mask) for the left and right tensors using a
   Triton kernel – this is the only required custom kernel.
5. Compute the “tri‑multiplicative” update:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is a batch of H independent matrix multiplications
   (left as (i,k) and right as (j,k) → left @ rightᵀ).  This is done with
   the highly‑optimized torch.matmul.
6. LayerNorm on the hidden dimension, element‑wise multiplication with an
   output gate (sigmoid of a third linear projection), and a final linear
   projection back to the original dimension C.

Only step 4 is executed on the GPU via Triton; the remaining heavy linear
operations and the batch matrix multiplication are delegated to cuBLAS via
PyTorch, which yields excellent performance on an H100.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def fuse_proj_gate_mask_kernel(
    proj_ptr,            # left_proj or right_proj
    gate_ptr,            # left_gate or right_gate
    mask_ptr,            # pairwise mask (broadcasted over hidden dim)
    out_ptr,             # output tensor (fused result)
    B, N, H,             # batch, seq_len, hidden_dim
    # strides for proj (B,N,N,H)
    stride_proj_b, stride_proj_i, stride_proj_k, stride_proj_h,
    # strides for gate (same layout as proj)
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    # strides for mask (B,N,N) – only three dims
    stride_mask_b, stride_mask_i, stride_mask_k,
    # strides for output (same layout as proj)
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    # compile‑time constants – tile sizes
    BLOCK_I: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    For each tile (i0 … i0+BLOCK_I, k0 … k0+BLOCK_K) this kernel loads
    a sub‑tensor of shape (BLOCK_I, BLOCK_K, BLOCK_H) from the projected
    tensor and its gate, multiplies them element‑wise and also applies the
    binary mask (broadcast over the hidden dimension).  The result is written
    to ``out_ptr``.
    """

    # 2‑D grid: axis‑0 tiles over (i,k), axis‑1 over batch
    pid = tl.program_id(0)
    batch = tl.program_id(1)

    # number of tiles along each spatial axis
    num_i_tiles = tl.cdiv(N, BLOCK_I)
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    # tile coordinates
    i_tile = pid // num_k_tiles
    k_tile = pid %  num_k_tiles

    i0 = i_tile * BLOCK_I
    k0 = k_tile * BLOCK_K

    # -------------------------------------------------------------
    # Offsets for the three dimensions (i, k, hidden)
    # Shape after broadcasting: (BLOCK_I, BLOCK_K, BLOCK_H)
    offs_i = i0 + tl.arange(0, BLOCK_I)[:, None, None]   # (I,1,1)
    offs_k = k0 + tl.arange(0, BLOCK_K)[None, :, None]   # (1,K,1)
    offs_h = tl.arange(0, BLOCK_H)[None, None, :]        # (1,1,H)

    # Boundary masks
    mask_i = offs_i < N
    mask_k = offs_k < N
    mask_h = offs_h < H
    mask_ik = mask_i & mask_k                 # (I,K,1)
    mask_ikh = mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]

    # -------------------------------------------------------------

    mask_ptrs = mask_ptr + batch * stride_mask_b \
                + offs_i * stride_mask_i \
                + offs_k * stride_mask_k                # no hidden stride
    mask = tl.load(mask_ptrs, mask=mask_ik, other=0.0).to(tl.float32)
    mask = mask[:, :, None]                                 # (I,K,1)

    # Load projected values and gates

                + offs_k * stride_proj_k \

                + offs_i * stride_gate_i \
                + offs_k * stride_gate_k \
                + offs_h * stride_gate_h

    proj = tl.load(proj_ptrs, mask=mask_ikh, other=0.0).to(tl.float32)
    gate = tl.load(gate_ptrs, mask=mask_ikh, other=0.0).to(tl.float32)

    # Fused compute
    out = proj * gate * mask

               + offs_k * stride_out_k \

    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] – the result of the TriMul forward operation.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # H (called hidden_dim in the paper)
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Extract weights (all are assumed to be contiguous and on the same device)
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension C
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) and gating projections
    #    Shape after projection: [B, N, N, H]
    left_proj  = torch.nn.functional.linear(x_norm, left_proj_weight)   # (B,N,N,H)
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)

    left_gate  = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate   = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    # ------------------------------------------------------------------
    # 3️⃣ Fuse (proj × gate × mask) with a Triton kernel if a mask is present.
    #    Otherwise simply multiply the projection with its gate.
    if mask is not None and not nomask:
        # Ensure float dtype matches the projections
        mask = mask.to(dtype=left_proj.dtype, device=left_proj.device)

        B, N, _, _ = left_proj.shape

        # Allocate destination tensors
        left  = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # Choose tile sizes.  Hidden dimension is a multiple of 8 (128), so we
        # tile it with BLOCK_H = 8.
        BLOCK_I = 64
        BLOCK_K = 64
        BLOCK_H = 8

        # Compute grid dimensions (tiles over i/k and batch)
        num_i_tiles = (N + BLOCK_I - 1) // BLOCK_I
        num_k_tiles = (N + BLOCK_K - 1) // BLOCK_K
        grid = (num_i_tiles * num_k_tiles, B)   # axis‑0: i/k tiles, axis‑1: batch

        # ----- left side ------------------------------------------------
        (s_pb, s_pi, s_pk, s_ph) = left_proj.stride()
        (s_gb, s_gi, s_gk, s_gh) = left_gate.stride()
        (s_mb, s_mi, s_mk)       = mask.stride()
        (s_ob, s_oi, s_ok, s_oh) = left.stride()

        fuse_proj_gate_mask_kernel[grid](
            left_proj, left_gate, mask, left,
            B, N, hidden_dim,
            s_pb, s_pi, s_pk, s_ph,
            s_gb, s_gi, s_gk, s_gh,
            s_mb, s_mi, s_mk,
            s_ob, s_oi, s_ok, s_oh,
            BLOCK_I=BLOCK_I, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
        )

        # ----- right side -----------------------------------------------
        (s_pb, s_pi, s_pk, s_ph) = right_proj.stride()
        (s_gb, s_gi, s_gk, s_gh) = right_gate.stride()
        (s_ob, s_oi, s_ok, s_oh) = right.stride()

        fuse_proj_gate_mask_kernel[grid](
            right_proj, right_gate, mask, right,
            B, N, hidden_dim,
            s_pb, s_pi, s_pk, s_ph,
            s_gb, s_gi, s_gk, s_gh,
            s_mb, s_mi, s_mk,
            s_ob, s_oi, s_ok, s_oh,
            BLOCK_I=BLOCK_I, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
        )
    else:
        # No mask → simple element‑wise multiplication
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched matrix multiplication:
    #    left  → (B, H, i, k)
    #    right → (B, H, k, j)   (note the transpose on the second N axis)
    left_t  = left.permute(0, 3, 1, 2)   # (B, H, i, k)
    right_t = right.permute(0, 3, 2, 1)  # (B, H, k, j)

    # torch.matmul performs a batched GEMM for each (B, H) pair.
    out = torch.matmul(left_t, right_t)   # (B, H, i, j)
    out = out.permute(0, 2, 3, 1)          # (B, i, j, H)

    # ------------------------------------------------------------------
    # 5️⃣ Output LayerNorm on the hidden dimension, gating, and final projection
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    out = out * out_gate                    # apply output gate (element‑wise)

    # final linear projection back to the original dim C
    output = torch.nn.functional.linear(out, to_out_weight)

    return output
```
