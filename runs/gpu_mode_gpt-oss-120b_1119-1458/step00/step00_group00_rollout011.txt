@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_batch, stride_left_row, stride_left_col,
    stride_right_batch, stride_right_row, stride_right_col,
    stride_out_batch, stride_out_row, stride_out_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)  # batch * hidden_dim
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute mask for OOB
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Compute pointer offsets for left (A) and right (B)
    a_ptrs = left_ptr + pid * stride_left_batch + offs_m[:, None] * stride_left_row + offs_k[None, :] * stride_left_col
    b_ptrs = right_ptr + pid * stride_right_batch + offs_n[:, None] * stride_right_row + offs_k[None, :] * stride_right_col

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        # Load A and B tiles
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < N), other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & (offs_k[None, :] < N), other=0.0).to(tl.float16)

        # Compute dot product
        acc += tl.dot(a, b)

        # Advance pointers
        a_ptrs += BLOCK_K * stride_left_col
        b_ptrs += BLOCK_K * stride_right_col

    # Store result

def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Device & dtype
    device = input_tensor.device
    dtype = input_tensor.dtype

    # LayerNorm on input
    x = torch.nn.functional.layer_norm(input_tensor,
                                       normalized_shape=(dim,),
                                       weight=weights["norm.weight"],
                                       bias=weights["norm.bias"],
                                       eps=eps)

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Mask handling
    if not config.get("nomask", False) and mask_tensor is not None:
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Prepare tensors for Triton kernel
    # permute to [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Cast to fp16 for kernel
    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    # Output buffer
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16)

    B, H, N, _ = left_fp16.shape
    total = B * H

    # Block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (total,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    tri_mul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        N,
        left_fp16.stride(0), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(2), right_fp16.stride(3),
        out_fp16.stride(0), out_fp16.stride(2), out_fp16.stride(3),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Convert back to float32 and restore original layout
    out = out_fp16.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # shape [B,N,N,H]

    # to_out_norm
    out = torch.nn.functional.layer_norm(out,
                                         normalized_shape=(hidden_dim,),
                                         weight=weights["to_out_norm.weight"],
                                         bias=weights["to_out_norm.bias"],
                                         eps=eps)

    # Apply out_gate
    out = out * out_gate

    # Final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
custom_kernel
==============

Implements the “outgoing” TriMul operation from AlphaFold‑3.

The heavy part of the module is the 4‑D einsum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication for every
(batch, channel) pair:

    out_{b,d} = left_{b,d} @ right_{b,d}^T

where left_{b,d}, right_{b,d} have shape (N, N).  This kernel is
implemented in Triton and fused across the channel dimension.
All remaining operations (LayerNorms, linear projections, gates) are
performed with PyTorch for simplicity.

The kernel works with float32 data (the reference implementation uses
float32) but can be switched to float16 by changing the dtype of the
input tensors.  The final output tensor is guaranteed to be float32.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul einsum
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,          # *fp32   left  tensor (B*H, N, N)
    right_ptr,         # *fp32   right tensor (B*H, N, N)
    out_ptr,           # *fp32   output tensor (B*H, N, N)
    N,                 # i32     sequence length
    # strides for the three tensors (in elements, not bytes)
    stride_l_batch, stride_l_row, stride_l_col,
    stride_r_batch, stride_r_row, stride_r_col,
    stride_o_batch, stride_o_row, stride_o_col,
    # ------------------------------------------------------------------
    BLOCK_M: tl.constexpr,  # rows per program block
    BLOCK_N: tl.constexpr,  # cols per program block
    BLOCK_K: tl.constexpr,  # reduction tile size
):
    # pid0 -> (batch * hidden_dim) index
    # pid1 -> block row index
    # pid2 -> block column index
    pid = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets within the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for rows / columns that lie inside the matrix
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to the start of the (pid)-th matrix in left/right
    a_ptrs = (left_ptr  + pid * stride_l_batch
              + offs_m[:, None] * stride_l_row
              + offs_k[None, :] * stride_l_col)
    b_ptrs = (right_ptr + pid * stride_r_batch
              + offs_n[:, None] * stride_r_row
              + offs_k[None, :] * stride_r_col)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K-tiles
    num_k = tl.cdiv(N, BLOCK_K)

    # Main GEMM loop over K
    for k in range(num_k):
        # Global K offset for this iteration
        k_offset = k * BLOCK_K
        cur_k = offs_k + k_offset
        mask_k = cur_k < N                     # shape (BLOCK_K,)

        # Load A and B tiles with correct masking
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_n[:, None] & mask_k[None, :],
                    other=0.0)

        # Compute partial product (dot across K)
        acc += tl.dot(a, b)

        # Move pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_l_col
        b_ptrs += BLOCK_K * stride_r_col

    # Write the result back to global memory
    c_ptrs = (out_ptr + pid * stride_o_batch
              + offs_m[:, None] * stride_o_row
              + offs_n[None, :] * stride_o_col)
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Helper – disable autograd (gradients are not required)
    # ------------------------------------------------------------------
    with torch.no_grad():
        # --------------------------------------------------------------
        # 1) Input LayerNorm
        # --------------------------------------------------------------
        x = torch.nn.functional.layer_norm(
            input_tensor,
            normalized_shape=(dim,),
            weight=weights["norm.weight"],
            bias=weights["norm.bias"],
            eps=eps,
        )  # [B, N, N, dim]

        # --------------------------------------------------------------
        # 2) Linear projections (no bias)
        # --------------------------------------------------------------
        left = torch.nn.functional.linear(x, weights["left_proj.weight"])
        right = torch.nn.functional.linear(x, weights["right_proj.weight"])

        # --------------------------------------------------------------
        # 3) Optional mask (broadcasted on the channel dimension)
        # --------------------------------------------------------------
        if not config.get("nomask", False) and mask_tensor is not None:
            mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
            left = left * mask
            right = right * mask

        # --------------------------------------------------------------
        # 4) Gating (sigmoid after a linear projection)
        # --------------------------------------------------------------
        left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
        right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
        out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

        left = left * left_gate
        right = right * right_gate

        # --------------------------------------------------------------
        # 5) Heavy part – batched matrix multiply via Triton
        #    out_{b,d}[i,j] = Σ_k left_{b,d}[i,k] * right_{b,d}[j,k]
        # --------------------------------------------------------------
        # Rearrange to (batch, hidden_dim, N, N) -> contiguous layout
        left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
        right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

        B, H, N, _ = left_perm.shape
        total_matrices = B * H

        # Output buffer (same dtype as inputs – float32)
        out_perm = torch.empty_like(left_perm, dtype=torch.float32)

        # Block sizes – tuned for modern GPUs (H100)
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64

        # Grid: (batch*hidden, n_blocks_m, n_blocks_n)
        grid = (
            total_matrices,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
        )

        # Launch Triton kernel
        tri_mul_gemm_kernel[grid](
            left_perm,
            right_perm,
            out_perm,
            N,
            left_perm.stride(0), left_perm.stride(2), left_perm.stride(3),
            right_perm.stride(0), right_perm.stride(2), right_perm.stride(3),
            out_perm.stride(0), out_perm.stride(2), out_perm.stride(3),
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )
        # Ensure kernel completion before proceeding
        torch.cuda.synchronize()

        # Restore original layout: [B, N, N, hidden_dim]
        out = out_perm.permute(0, 2, 3, 1).contiguous()   # float32

        # --------------------------------------------------------------
        # 6) Output LayerNorm over the hidden dimension
        # --------------------------------------------------------------
        out = torch.nn.functional.layer_norm(
            out,
            normalized_shape=(hidden_dim,),
            weight=weights["to_out_norm.weight"],
            bias=weights["to_out_norm.bias"],
            eps=eps,
        )  # [B, N, N, hidden_dim]

        # --------------------------------------------------------------
        # 7) Apply the out‑gate
        # --------------------------------------------------------------
        out = out * out_gate

        # --------------------------------------------------------------
        # 8) Final linear projection back to `dim`
        # --------------------------------------------------------------
        out = torch.nn.functional.linear(out, weights["to_out.weight"])

        return out
