@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N,
    stride_lm, stride_li, stride_lk,
    stride_rm, stride_rk, stride_rj,
    stride_om, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):

    batch = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute start offsets
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # create range offsets
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute base pointers for this batch
    left_batch_ptr = left_ptr + batch * stride_lm
    right_batch_ptr = right_ptr + batch * stride_rm
    out_batch_ptr = out_ptr + batch * stride_om

    # create accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, N, BLOCK_K):
        # compute offsets for k dimension
        offs_k = k + tl.arange(0, BLOCK_K)

        # load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_batch_ptr + (offs_i[:, None] * stride_li) + (offs_k[None, :] * stride_lk)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & (offs_k[None, :] < N), other=0.0).to(tl.float32)

        # load right tile as (BLOCK_K, BLOCK_N)
        # we need right[j,k] - treat as row = k, col = j
        right_ptrs = right_batch_ptr + (offs_k[:, None] * stride_rk) + (offs_j[None, :] * stride_rj)
        right = tl.load(right_ptrs, mask=(offs_k[:, None] < N) & mask_j[None, :], other=0.0).to(tl.float32)

        # dot
        # left: (M,K), right: (K,N)
        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_batch_ptr + (offs_i[:, None] * stride_oi) + (offs_j[None, :] * stride_oj)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False) # maybe mask is not needed
    
    # layer_norm over dim
    eps = 1e-5
    # compute mean and var manually using torch? We'll use torch.nn.functional.layer_norm for speed.
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask multiplication if needed
    if not nomask:
        mask_unsq = mask.unsqueeze(-1).to(input_tensor.dtype)  # broadcast
        left = left * mask_unsq
        right = right * mask_unsq

    # gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # convert to fp16 for kernel compute
    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)

    # reshape for kernel
    # Permute to (B, hidden_dim, N, N)
    left_perm = left_half.permute(0, 3, 1, 2).contiguous()
    right_perm = right_half.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape
    M = B * H

    left_flat = left_perm.view(M, N, N)
    right_flat = right_perm.view(M, N, N)

    # allocate output
    out_flat = torch.empty((M, N, N), dtype=torch.float32, device=left.device)

    # compute strides
    stride_lm, stride_li, stride_lj = left_flat.stride()
    stride_rm, stride_ri, stride_rj = right_flat.stride()
    stride_om, stride_oi, stride_oj = out_flat.stride()

    # define block sizes; choose 128 for M? Usually block sizes for matmul: 128/64.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # compute grid dimensions
    grid = (M, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
          left_flat, right_flat, out_flat,
          M, N,
          stride_lm, stride_li, stride_lj,
          stride_rm, stride_ri, stride_rj,
          stride_om, stride_oi, stride_oj,
          BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous() # shape B,N,N,hidden_dim

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    out = out * out_gate # broadcast: out_gate shape B,N,N,hidden_dim

    # final linear to output dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N,
    stride_lm, stride_li, stride_lj,
    stride_rm, stride_ri, stride_rj,
    stride_om, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # coordinates
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # range offsets
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    # bounds mask
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers
    left_batch_ptr = left_ptr + batch * stride_lm
    right_batch_ptr = right_ptr + batch * stride_rm
    out_batch_ptr = out_ptr + batch * stride_om

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)

        # load left tile [i, k]
        left_ptrs = left_batch_ptr + (offs_i[:, None] * stride_li) + (offs_k[None, :] * stride_lj)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & (offs_k[None, :] < N), other=0.0)
        left = left.to(tl.float32)

        # load right tile [k, j] (transposed view)
        right_ptrs = right_batch_ptr + (offs_k[:, None] * stride_ri) + (offs_j[None, :] * stride_rj)
        right = tl.load(right_ptrs, mask=(offs_k[:, None] < N) & mask_j[None, :], other=0.0)
        right = right.to(tl.float32)

        acc += tl.dot(left, right)

    # write output
    out_ptrs = out_batch_ptr + (offs_i[:, None] * stride_oi) + (offs_j[None, :] * stride_oj)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton-accelerated implementation of the outgoing TriMul operation.
    ...
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # proj
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask
    if not config.get("nomask", True):
        # ensure mask provided
        if mask is None:
            raise ValueError("Mask is required when nomask=False")
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Rearrange to (B, hidden_dim, N, N)
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    M = B * H

    # Flatten batch & head dims
    left_flat = left_perm.view(M, N, N)
    right_flat = right_perm.view(M, N, N)

    # Output tensor
    out_flat = torch.empty((M, N, N), dtype=torch.float32, device=device)

    # Strides (must be ints)
    stride_lm, stride_li, stride_lj = left_flat.stride()
    stride_rm, stride_ri, stride_rj = right_flat.stride()
    stride_om, stride_oi, stride_oj = out_flat.stride()

    # block sizes, tuned for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (M, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left_flat, right_flat, out_flat,
        M, N,
        stride_lm, stride_li, stride_lj,
        stride_rm, stride_ri, stride_rj,
        stride_om, stride_oi, stride_oj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Reshape back to (B, H, N, N) -> (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                                         bias=to_out_norm_bias, eps=eps)

    # Apply out gate
    out = out * out_gate

    # Final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The operator works on a tensor x ∈ ℝ^{B×N×N×C} :

1. Layer‑norm over the channel dimension C.
2. Two linear projections (left / right) and three gating projections.
3. Apply an optional pair‑wise mask and the gates.
4. Compute the pairwise multiplication:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication left @ rightᵀ for every
   hidden channel d.  This O(B·C·N²·⌈N/KB⌉) kernel is implemented in
   Triton.
5. Layer‑norm + output gate.
6. Final linear projection back to the original channel dimension C.

The heavy inner loop (step 4) is fully fused in a single Triton kernel,
while the surrounding linear layers, layer‑norms and activations are
performed in PyTorch.  The kernel works in FP16 (input) with FP32
accumulation and returns FP32 results.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the core 4‑D einsum:
#   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# After reshaping (B,hidden,N,N) → (M=N·B·hidden, N, N) the operation becomes
#   out[m,i,j] = Σ_k left[m,i,k] * right[m,k,j]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,      # pointers
    M, N,                             # M = B*hidden_dim, N = seq_len
    stride_lm, stride_li, stride_lj,  # strides for left  (M, i, k)
    stride_rm, stride_ri, stride_rj,  # strides for right (M, k, j)
    stride_om, stride_oi, stride_oj,  # strides for out   (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out = left @ rightᵀ for each “matrix” in the batch.
    left  : (M, N, K)  with K = N
    right : (M, N, K)  (the second operand is accessed as (k, j))
    out   : (M, N, N)
    """
    batch_id = tl.program_id(0)          # which pair of (B, hidden)
    pid_i = tl.program_id(1)             # block row index
    pid_j = tl.program_id(2)             # block column index

    # Offsets for the output tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this batch
    left_batch = left_ptr + batch_id * stride_lm
    right_batch = right_ptr + batch_id * stride_rm
    out_batch = out_ptr + batch_id * stride_om

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k in blocks of size BLOCK_K
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)

        # --- Load left[i, k] -------------------------------------------------
        left_ptrs = left_batch + (offs_i[:, None] * stride_li) + (offs_k[None, :] * stride_lj)
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & (offs_k[None, :] < N),
                            other=0.0).to(tl.float32)

        # --- Load right[k, j] (transposed view) -------------------------------
        right_ptrs = right_batch + (offs_k[:, None] * stride_ri) + (offs_j[None, :] * stride_rj)
        right_tile = tl.load(right_ptrs,
                             mask=(offs_k[:, None] < N) & mask_j[None, :],
                             other=0.0).to(tl.float32)

        # --- Dot product for this k‑block ------------------------------------
        acc += tl.dot(left_tile, right_tile)

    # Store the computed tile
    out_ptrs = out_batch + (offs_i[:, None] * stride_oi) + (offs_j[None, :] * stride_oj)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data (tuple):
            - input_tensor : torch.Tensor of shape [B, N, N, C]
            - mask         : torch.Tensor of shape [B, N, N] (may be ignored)
            - weights      : dict of model parameters (see code)
            - config       : dict containing at least "dim", "hidden_dim",
                             and a boolean "nomask" flag.

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm over the channel dimension (C)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (applied before gating)
    # ------------------------------------------------------------------
    apply_mask = not config.get("nomask", True) and mask is not None
    if apply_mask:
        # broadcast mask to the last dimension
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating (sigmoid on linear projections of the original normalized x)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the Triton kernel
    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy kernel (saves bandwidth)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Rearrange to (B, hidden_dim, N, N) → flatten B·hidden_dim as batch M
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_h.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    M = B * H

    # (M, N, N) tensors that the kernel will consume
    left_flat = left_perm.view(M, N, N)
    right_flat = right_perm.view(M, N, N)

    # Output buffer (FP32 as required by the spec)
    out_flat = torch.empty((M, N, N), dtype=torch.float32, device=device)

    # Strides for the contiguous layout (M, i, j) → (N*N, N, 1)
    stride_lm, stride_li, stride_lj = left_flat.stride()
    stride_rm, stride_ri, stride_rj = right_flat.stride()
    stride_om, stride_oi, stride_oj = out_flat.stride()

    # Tile sizes – tuned for H100 (fits in shared memory, good occupancy)
    BLOCK_M = 64   # rows of the output tile
    BLOCK_N = 64   # cols of the output tile
    BLOCK_K = 64   # reduction tile

    # Grid: (batch M, rows tiles, cols tiles)
    grid = (
        M,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch kernel
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M,
        N,
        stride_lm,
        stride_li,
        stride_lj,
        stride_rm,
        stride_ri,
        stride_rj,
        stride_om,
        stride_oi,
        stride_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to the original layout: [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Output LayerNorm, gate, and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # broadcast over the hidden dimension
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, C]

    return out
