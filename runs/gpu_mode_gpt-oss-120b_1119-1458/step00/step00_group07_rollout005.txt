    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm: mean variance across dim
    # We'll do x_norm = (x - mean)/sqrt(var+eps) * norm_weight + norm_bias
    # Using torch.mean over last dim.
    # Note mask not used in norm.

    # Acquire weights.
    # Using torch.nn.functional.linear: torch.nn.functional.linear(x, weight, bias) expects weight shape (out_features, in_features)
    # Our weight shapes maybe (hidden_dim, dim) for projections.

    # Compute x_norm
    x = input_tensor
    # shape: [bs, N, N, dim]
    # Compute mean and var across dim.
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    # Apply LayerNorm weight and bias
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = x_norm * norm_weight + norm_bias

    # Projections
    left_proj_weight = weights["left_proj.weight"]  # shape (hidden_dim, dim)

    if mask is not None:

"""
TriMul (outgoing) – fast Triton implementation of the triple‐wise multiplicative update.

The kernel computes the core Einstein‑summation
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
where `left` and `right` have shape (B, N, N, H) (H = hidden_dim).
The implementation:
  • LayerNorms, linear projections and gating are done in PyTorch.
  • `left` and `right` are re‑ordered to (B*H, N, N) and cast to FP16.
  • A custom Triton kernel performs a batched matrix multiplication for each
    (B*H) pair:  (N×N) @ (N×N)ᵀ → (N×N).  The kernel works on tiles
    (BLOCK_M × BLOCK_N) and accumulates in FP32.
  • The result is reshaped back, LayerNormed, gated and projected to the
    original channel dimension.

The kernel runs on H100 (torch‑compatible) and uses mixed‑precision
(FP16 inputs → FP32 accumulation).  Only the heavy O(N³) part is in Triton,
the rest stays in PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul with the second operand transposed.
# ----------------------------------------------------------------------
@triton.jit
def tr_mul_out_kernel(
    a_ptr,               # *float16 left  [B*H, N, N]
    b_ptr,               # *float16 right [B*H, N, N]
    c_ptr,               # *float16 out   [B*H, N, N]
    batch_h, M, N, K,    # int64 scalars
    stride_ah, stride_am, stride_ak,   # int64 strides for a
    stride_bh, stride_bn, stride_bk,   # int64 strides for b
    stride_ch, stride_cm, stride_cn,   # int64 strides for c
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    bh_id = tl.program_id(0)        # batch*hidden id
    pid_m = tl.program_id(1)        # tile row
    pid_n = tl.program_id(2)        # tile col

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the beginning of the current batch‐hidden slice
    a_base = a_ptr + bh_id * stride_ah
    b_base = b_ptr + bh_id * stride_bh
    c_base = c_ptr + bh_id * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_am + k_off[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile transposed: we need Bᵀ → shape (BLOCK_K, BLOCK_N)
        # Original B layout: (batch_h, N, K)
        # Access Bᵀ as B[k, n]
        b_ptrs = b_base + k_off[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Dot product (FP16×FP16 → FP32)
        acc += tl.dot(a, b)

    # Store the result back to C (FP16)
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Host entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module using a fused Triton kernel.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim] (float32)
        - mask         : torch.Tensor[bs, N, N]      (float32/bool) or None
        - weights      : dict of model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [bs, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # LayerNorm on the last dimension
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    mu   = input_tensor.mean(dim=-1, keepdim=True)
    var  = ((input_tensor - mu) ** 2).mean(dim=-1, keepdim=True)
    x = (input_tensor - mu) * torch.rsqrt(var + eps)
    x = x * norm_weight + norm_bias                     # [bs, N, N, dim]

    # --------------------------------------------------------------
    # Linear projections (no bias)
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_w  = weights["right_proj.weight"]  # (hidden_dim, dim)
    left  = torch.nn.functional.linear(x, left_proj_w)   # [bs, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)  # [bs, N, N, hidden_dim]

    # --------------------------------------------------------------
    # Optional masking (nomask == True ⇒ skip)
    # --------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask = mask.unsqueeze(-1)                 # [bs, N, N, 1]
        left  = left  * mask
        right = right * mask

    # --------------------------------------------------------------
    # Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # Prepare tensors for the Triton kernel
    # --------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N) → (B*H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()   # [bs, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [bs, H, N, N]

    batch_h = bs * hidden_dim
    left_flat = left.view(batch_h, N, N)            # [B*H, N, N]
    right_flat = right.view(batch_h, N, N)          # [B*H, N, N]

    # Cast to FP16 for the kernel (Tensor‑cores)
    left_fp16  = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # Output buffer (FP16, will be cast back to FP32 later)
    out_fp16 = torch.empty_like(left_fp16, device=device)

    # --------------------------------------------------------------
    # Launch Triton kernel
    # --------------------------------------------------------------
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32  # tuning knobs
    grid = (batch_h,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tr_mul_out_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        batch_h, N, N, N,                               # B*H, M, N, K
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2),
        out_fp16.stride(0), out_fp16.stride(1), out_fp16.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # a reasonable default for H100
    )

    # --------------------------------------------------------------
    # Restore original layout: [bs, N, N, hidden_dim] (FP32)
    # --------------------------------------------------------------
    out = out_fp16.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # --------------------------------------------------------------
    # Output LayerNorm, gating and final linear projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]          # (dim, hidden_dim)

    # LayerNorm over hidden_dim
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # Apply the output gate (same shape as out)
    out = out * out_gate

    # Final linear projection back to `dim`
    out = torch.nn.functional.linear(out, to_out_w)   # [bs, N, N, dim]

    return out
```

This implementation follows the reference PyTorch code, fuses the expensive `einsum` into a high‑throughput Triton kernel, and keeps the rest of the operations in PyTorch for clarity and ease of use. It works for the full range of configurations described in the prompt.
