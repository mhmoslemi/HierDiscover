"""
Custom Triton‑enabled implementation of the “outgoing” TriMul operator
used in AlphaFold‑3. 

Algorithm
---------
1. Layer‑norm the input tensor (shape [B, N, N, C]).
2. Linear projections (left / right) and linear gates (left_gate,
   right_gate, out_gate) – all bias‑free Linear layers.
3. Apply the optional pairwise mask and the gates.
4. Core contraction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply over the index *k* for every
   hidden channel *d*.  The heavy O(N³·H) work is performed by a
   custom Triton kernel that tiles the *i*, *j* and *k* dimensions and
   accumulates a per‑channel result with Tensor‑Core‑accelerated `tl.dot`.
5. Layer‑norm the result (over the hidden dimension), apply the
   out‑gate and the final output Linear projection.

Only the contraction (step 4) is written in Triton – the remaining
operations use standard PyTorch calls.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton kernel for the O(N³·H) contraction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_bi, stride_bk, stride_bd,
    stride_rl, stride_rj, stride_rk, stride_rd,
    stride_ol, stride_oi, stride_oj, stride_od,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
    HIDDEN_DIM: tl.constexpr
):
    """
    Compute   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    Arguments
    ---------
    left/right/out_ptr   : pointers to the tensors (contiguous in memory)
    B,N,H                : batch size, sequence length and hidden dimension
    stride_*             : tensor strides (in elements, not bytes)
    BLOCK_I,J,K         : tiling sizes for i, j, k dimensions
    HIDDEN_DIM           : compile‑time constant = hidden dimension
    """
    pid_b = tl.program_id(2)          # batch index
    pid_i = tl.program_id(0)          # i‑tile index
    pid_j = tl.program_id(1)          # j‑tile index

    # ----- tile offsets -------------------------------------------------
    i_off = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_off = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_mask = i_off < N
    j_mask = j_off < N

    # ----- loop over hidden dimension ------------------------------------
    for d in tl.static_range(0, HIDDEN_DIM):
        # accumulator for the (i,j) tile of a single channel d
        acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

        # ----- loop over k ------------------------------------------------
        for k_start in range(0, N, BLOCK_K):
            k_off = k_start + tl.arange(0, BLOCK_K)
            k_mask = k_off < N

            # left tile  shape (BLOCK_I, BLOCK_K)
            left_ptrs = (
                left_ptr
                + pid_b * stride_bl
                + i_off[:, None] * stride_bi
                + k_off[None, :] * stride_bk
                + d * stride_bd
            )
            left = tl.load(left_ptrs,
                           mask=i_mask[:, None] & k_mask[None, :],
                           other=0.0)

            # right tile shape (BLOCK_K, BLOCK_J)
            # note the order (k, j) so that a dot product can be taken directly
            right_ptrs = (
                right_ptr
                + pid_b * stride_rl
                + k_off[:, None] * stride_rk
                + j_off[None, :] * stride_rj
                + d * stride_rd
            )
            right = tl.load(right_ptrs,
                            mask=k_mask[:, None] & j_mask[None, :],
                            other=0.0)

            # Matrix multiply over the k‑dimension (Tensor‑Core friendly)
            acc += tl.dot(left, right)

        # ----- write the completed tile ----------------------------------
        out_ptrs = (
            out_ptr
            + pid_b * stride_ol
            + i_off[:, None] * stride_oi
            + j_off[None, :] * stride_oj
            + d * stride_od
        )
        tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    # Normalize over the last dimension (C)
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight) # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        # mask may be bool or float; convert to float and broadcast
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gated multiplicative paths
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B,N,N,hidden_dim]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # [B,N,N,hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core contraction – Triton kernel
    # ------------------------------------------------------------------
    # Prepare an empty output buffer
    out = torch.empty((bs, seq_len, seq_len, hidden_dim),
                      dtype=dtype,
                      device=device)

    # Strides (in elements, not bytes) – needed by Triton
    stride_lb = left.stride(0)
    stride_li = left.stride(1)
    stride_lk = left.stride(2)
    stride_ld = left.stride(3)

    stride_rb = right.stride(0)
    stride_rj = right.stride(1)  # j‑dimension
    stride_rk = right.stride(2)  # k‑dimension
    stride_rd = right.stride(3)

    stride_ob = out.stride(0)
    stride_oi = out.stride(1)
    stride_oj = out.stride(2)
    stride_od = out.stride(3)

    # Tile sizes – chosen to fit comfortably in registers/shared memory
    BLOCK_I = 16
    BLOCK_J = 16
    BLOCK_K = 32

    grid = (
        (seq_len + BLOCK_I - 1) // BLOCK_I,   # i‑tiles
        (seq_len + BLOCK_J - 1) // BLOCK_J,   # j‑tiles
        bs                                    # batch dimension
    )

    # Launch the kernel
    tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        bs, seq_len, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        HIDDEN_DIM=hidden_dim,
        num_warps=4,   # reasonable default for H100
    )

    # ------------------------------------------------------------------
    # 6) Output block‑norm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, to_out_weight)    # final shape [B,N,N,dim]

    return out
```
