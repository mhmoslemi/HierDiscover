def matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_batch_A, stride_m_A, stride_k_A,
    stride_batch_B, stride_k_B, stride_n_B,
    stride_batch_C, stride_m_C, stride_n_C,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # compute start row/col
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

@triton.jit
def batched_matmul_kernel(A, B, C,
                          M, N, K,
                          stride_batch_A, stride_m_A, stride_k_A,
                          stride_batch_B, stride_k_B, stride_n_B,
                          stride_batch_C, stride_m_C, stride_n_C,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointers to start of tile of A and B
    a_ptrs = A + pid_batch * stride_batch_A + offs_m[:, None] * stride_m_A + tl.arange(0, BLOCK_K)[None, :] * stride_k_A
    b_ptrs = B + pid_batch * stride_batch_B + tl.arange(0, BLOCK_K)[:, None] * stride_k_B + offs_n[None, :] * stride_n_B

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute mask to avoid out-of-bounds loads
        cur_k = k * BLOCK_K
        k_offset = tl.arange(0, BLOCK_K)
        # Mask for A: rows < M && cur_k + k_offset < K
        mask_a = (offs_m[:, None] < M) & (cur_k + k_offset)[None, :] < K
        # Mask for B: columns < N && cur_k + k_offset < K

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)

        # Advance pointers
        a_ptrs += BLOCK_K * stride_k_A
        b_ptrs += BLOCK_K * stride_k_B

    # write back
    c_ptrs = C + pid_batch * stride_batch_C + offs_m[:, None] * stride_m_C + offs_n[None, :] * stride_n_C

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # Ensure mask is not None to avoid errors.
    if mask is None:
        # create ones mask
        mask = torch.ones_like(input_tensor[...,0], dtype=input_tensor.dtype, device=input_tensor.device)
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm (norm)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # Compute mean and variance over last dimension
    # Use torch.mean/tensor.var for simplicity
    # x shape [B,N,N,dim]
    x = input_tensor
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # Projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if present
    if mask is not None:
        mask_uns = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_uns
        right = right * mask_uns

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for matmul
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, H = left_h.shape

    # Reshape for batched matmul
    left_mat = left_h.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    right_mat = right_h.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    # Transpose right for multiplication
    right_mat_T = right_mat.transpose(2, 1).contiguous()

    # Compute C = left_mat @ right_mat_T using Triton
    out_mat = batched_matmul(left_mat, right_mat_T)  # shape (B*H, N, N), dtype float16

    # Reshape back to (B,N,N,H)
    out = out_mat.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    mean2 = out.mean(-1, keepdim=True)
    var2 = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean2) * torch.rsqrt(var2 + eps)
    out = out * to_out_norm_weight + to_out_norm_bias

    # apply out_gate
    out = out * out_gate

    # final linear
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)  # shape [B,N,N,dim]

    return out

"""
TriMul 'outgoing' implementation.

This kernel fuses the cubic pairwise multiplication step
`out[i,j,:] = sum_k left[i,k,:] * right[j,k,:]` using a
batched matrix‑multiply kernel written in Triton.  The rest
of the TriMul module (LayerNorm, linear projections,
sigmoid gates and the final linear projection) is
implemented with native PyTorch operations.  The heavy
computation is performed in fp16 on the H100 while
accumulating in fp32 for numerical stability.
"""
import torch
import triton
import triton.language as tl

# Batched GEMM kernel used for the cubic update
@triton.jit
def _batched_gemm_kernel(
    A, B, C,
    M, N, K,
    stride_batch_A, stride_m_A, stride_k_A,
    stride_batch_B, stride_k_B, stride_n_B,
    stride_batch_C, stride_m_C, stride_n_C,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start offsets for this tile
    offs_m_0 = pid_m * BLOCK_M + tl.arange(0, BLOCK_M // 2)   # first half of rows
    offs_m_1 = pid_m * BLOCK_M + tl.arange(BLOCK_M // 2, BLOCK_M)  # second half
    offs_n   = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointers to the start of the tiles in A and B
    a_ptrs_0 = A + batch_id * stride_batch_A + offs_m_0[:, None] * stride_m_A + tl.arange(0, BLOCK_K)[None, :] * stride_k_A
    a_ptrs_1 = A + batch_id * stride_batch_A + offs_m_1[:, None] * stride_m_A + tl.arange(0, BLOCK_K)[None, :] * stride_k_A
    b_ptrs   = B + batch_id * stride_batch_B + tl.arange(0, BLOCK_K)[:, None] * stride_k_B + offs_n[None, :] * stride_n_B

    # accumulator in FP32 (more stable)
    acc_0 = tl.zeros((BLOCK_M // 2, BLOCK_N), dtype=tl.float32)
    acc_1 = tl.zeros((BLOCK_M // 2, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # number of remaining K elements in this tile
        k_remaining = K - k * BLOCK_K
        # mask for out‑of‑bounds loads
        mask_k = tl.arange(0, BLOCK_K) < k_remaining

        a0 = tl.load(a_ptrs_0, mask=(offs_m_0[:, None] < M) & mask_k[None, :], other=0.0)
        a1 = tl.load(a_ptrs_1, mask=(offs_m_1[:, None] < M) & mask_k[None, :], other=0.0)
        b  = tl.load(b_ptrs,  mask=mask_k[:, None] & (offs_n[None, :] < N), other=0.0)

        # accumulate
        acc_0 += tl.dot(a0, b)
        acc_1 += tl.dot(a1, b)

        # advance pointers to the next K‑tile
        a_ptrs_0 += BLOCK_K * stride_k_A
        a_ptrs_1 += BLOCK_K * stride_k_A
        b_ptrs   += BLOCK_K * stride_k_B

    # write results back to C
    c_ptrs_0 = C + batch_id * stride_batch_C + offs_m_0[:, None] * stride_m_C + offs_n[None, :] * stride_n_C
    c_ptrs_1 = C + batch_id * stride_batch_C + offs_m_1[:, None] * stride_m_C + offs_n[None, :] * stride_n_C

    mask_c0 = (offs_m_0[:, None] < M) & (offs_n[None, :] < N)
    mask_c1 = (offs_m_1[:, None] < M) & (offs_n[None, :] < N)

    tl.store(c_ptrs_0, acc_0.to(tl.float16), mask=mask_c0)
    tl.store(c_ptrs_1, acc_1.to(tl.float16), mask=mask_c1)


def _batched_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication:  C[b] = A[b] @ B[b]   (A: [B,M,K], B: [B,K,N])
    Both A and B must be ``torch.float16`` and contiguous.
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    # Block sizes – tuned for H100 fp16 tensor‑cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    _batched_gemm_kernel[grid](
        A, B, C,
        M, N, K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M, BLOCK_N, BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] containing the module output.
    """
    # unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Layer‑norm on the input tensor (no bias in the PyTorch model)
    # ------------------------------------------------------------------
    # x shape: [B, N, N, dim]
    x = input_tensor
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # ------------------------------------------------------------------
    # Linear projections (no bias) and optional masking
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    if mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Heavy cubic multiplication – implemented in Triton.
    # ------------------------------------------------------------------
    # Cast to fp16 for the GEMM; the kernel accumulates in fp32.
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, H = left_h.shape

    # reshape to (B*H, N, N) and transpose the right operand
    left_mat = left_h.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    right_mat = right_h.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    right_mat_T = right_mat.transpose(2, 1).contiguous()   # shape (B*H, N, N)

    # C = A @ B  (A: [batch, N, K], B: [batch, K, N])
    out_mat = _batched_gemm(left_mat, right_mat_T)  # fp16 result

    # reshape back to [B, N, N, H]
    out = out_mat.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # Output LayerNorm, gate and final linear projection
    # ------------------------------------------------------------------
    mean2 = out.mean(-1, keepdim=True)
    var2 = out.var(-1, unbiased=False, keepdim=True)
    out = (out - mean2) * torch.rsqrt(var2 + eps)
    out = out * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # Apply the output gate (element‑wise)
    out = out * out_gate

    # Final linear projection back to ``dim``
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul “outgoing” forward pass.

The computationally dominant step is the cubic update

    out[i, j, d] = Σ_k left[i, k, d] * right[j, k, d]

which can be expressed as a batched matrix multiplication
C[b] = A[b] @ B[b]ᵀ with A, B ∈ ℝ^{N×N} and a batch dimension
B·hidden_dim (the hidden channels are independent).  The kernel below
implements this batched GEMM in Triton (fp16 inputs, fp32 accumulation)
while the surrounding LayerNorms, linear projections and gating are
performed with native PyTorch ops.  The result matches the reference
implementation from AlphaFold‑3.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM used for the cubic multiply‑accumulate.
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A, B, C,
    M, N, K,
    stride_batch_A, stride_m_A, stride_k_A,
    stride_batch_B, stride_k_B, stride_n_B,
    stride_batch_C, stride_m_C, stride_n_C,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # --------------------------------------------------------------
    # Program IDs (batch, tile‑row, tile‑col)
    # --------------------------------------------------------------
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Split the M‑dimension into two halves to keep the register count low
    offs_m_0 = pid_m * BLOCK_M + tl.arange(0, BLOCK_M // 2)      # first half
    offs_m_1 = pid_m * BLOCK_M + tl.arange(BLOCK_M // 2, BLOCK_M)  # second half
    offs_n   = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the start of the current tile for A and B
    a_ptrs_0 = A + batch_id * stride_batch_A + offs_m_0[:, None] * stride_m_A + tl.arange(0, BLOCK_K)[None, :] * stride_k_A
    a_ptrs_1 = A + batch_id * stride_batch_A + offs_m_1[:, None] * stride_m_A + tl.arange(0, BLOCK_K)[None, :] * stride_k_A
    b_ptrs   = B + batch_id * stride_batch_B + tl.arange(0, BLOCK_K)[:, None] * stride_k_B + offs_n[None, :] * stride_n_B

    # Accumulators (FP32 for numerical stability)
    acc_0 = tl.zeros((BLOCK_M // 2, BLOCK_N), dtype=tl.float32)
    acc_1 = tl.zeros((BLOCK_M // 2, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the K dimension (blocked)
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_remaining = K - k * BLOCK_K
        mask_k = tl.arange(0, BLOCK_K) < k_remaining

        a0 = tl.load(a_ptrs_0,
                     mask=(offs_m_0[:, None] < M) & mask_k[None, :],
                     other=0.0)
        a1 = tl.load(a_ptrs_1,
                     mask=(offs_m_1[:, None] < M) & mask_k[None, :],
                     other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & (offs_n[None, :] < N),
                    other=0.0)

        # Accumulate the partial products
        acc_0 += tl.dot(a0, b)
        acc_1 += tl.dot(a1, b)

        # Advance the pointers to the next K‑tile
        a_ptrs_0 += BLOCK_K * stride_k_A
        a_ptrs_1 += BLOCK_K * stride_k_A
        b_ptrs   += BLOCK_K * stride_k_B

    # --------------------------------------------------------------
    # Store the result back to C
    # --------------------------------------------------------------
    c_ptrs_0 = C + batch_id * stride_batch_C + offs_m_0[:, None] * stride_m_C + offs_n[None, :] * stride_n_C
    c_ptrs_1 = C + batch_id * stride_batch_C + offs_m_1[:, None] * stride_m_C + offs_n[None, :] * stride_n_C

    mask_c0 = (offs_m_0[:, None] < M) & (offs_n[None, :] < N)
    mask_c1 = (offs_m_1[:, None] < M) & (offs_n[None, :] < N)

    tl.store(c_ptrs_0, acc_0.to(tl.float16), mask=mask_c0)
    tl.store(c_ptrs_1, acc_1.to(tl.float16), mask=mask_c1)


def _batched_gemm(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM: C[b] = A[b] @ B[b]   (A: [B, M, K], B: [B, K, N])
    Both inputs must be contiguous fp16 tensors.
    """
    assert A.is_contiguous() and B.is_contiguous()

    _, K2, N = B.shape
    assert K == K2

    # Tuned block sizes for fp16 Tensor‑Core GPUs (e.g. H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

            (M + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _batched_gemm_kernel[grid](
        A, B, C,
        M, N, K,
        A.stride(0), A.stride(1), A.stride(2),   # batch, row, col strides of A
        B.stride(0), B.stride(1), B.stride(2),   # batch, row, col strides of B
        C.stride(0), C.stride(1), C.stride(2),   # batch, row, col strides of C
        BLOCK_M, BLOCK_N, BLOCK_K,
    )
    return C

    # ------------------------------------------------------------------

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))

    left_h = left.to(torch.float16)

    left_mat = left_h.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()

    # we need rightᵀ in the GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    right_mat_T = right_mat.transpose(2, 1).contiguous()

    out_mat = _batched_gemm(left_mat, right_mat_T)
