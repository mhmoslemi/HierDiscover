def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, n_elements,
                           BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, n_elements,
                           BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(c_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    """
    Triton-accelerated forward pass for the "outgoing" TriMul operator.
    Implements:
    1. LayerNorm over the channel dimension.
    2. Linear projections to a hidden dimension (left/right) and corresponding gates.
    3. Optional mask application.
    4. Gating (elementwise multiplication).
    5. Batched matrix multiplication left @ right^T for every hidden channel.
    6. LayerNorm on the hidden result, followed by an output gate.
    7. Final linear projection back to the original dimension.
    A small Triton kernel is used for the final element‑wise multiplication with the
    output gate to demonstrate kernel integration.
    """
    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------------- LayerNorm on input ----------------
    # input: [B, N, N, C]
    # weights for norm: shape [C] and bias [C]
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C]

    # ---------------- Linear projections ----------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    # Biases are disabled in the reference, so we pass bias=None
    left = F.linear(x, left_proj_weight)       # [B, N, N, H]
    right = F.linear(x, right_proj_weight)    # [B, N, N, H]

    # Optional mask (shape [B, N, N] -> [B, N, N, 1])
    if mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # broadcast to hidden dim
        left = left * mask
        right = right * mask

    # ---------------- Gating ----------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B, N, N, H]

    left = left * left_gate
    right = right * right_gate

    # ---------------- Batched matmul (left @ right^T) ----------------
    # Reshape to (B*H, N, N) and perform strided batched GEMM
    B, N, _, H = left.shape  # H == hidden_dim
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Compute out_flat = left_flat @ right_flat^T
    # right_flat^T has shape (B*H, N, N) -> we transpose last two dims
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ---------------- Output layer norm ----------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ---------------- Output gate ----------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))  # [B, N, N, H]

    # Triton kernel for elementwise multiplication out = out * out_gate
    out_contig = out.contiguous()
    out_gate_contig = out_gate.contiguous()
    out_gated = torch.empty_like(out_contig)

    numel = out_contig.numel()
    BLOCK_SIZE = 1024  # tunable; fits well in registers on H100
    grid = (triton.cdiv(numel, BLOCK_SIZE),)
    elementwise_mul_kernel[grid](
        out_contig,
        out_gate_contig,
        out_gated,
        numel,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # ---------------- Final linear projection ----------------
    to_out_weight = weights["to_out.weight"]
    # No bias in reference implementation
    out_final = F.linear(out_gated, to_out_weight)  # [B, N, N, dim]

    return out_final

"""
TriMul (outgoing) forward pass with a small Triton kernel.

The implementation follows the PyTorch reference:

1. Layer‑norm over the channel dimension.
2. Linear projections to a hidden dimension (left/right) together with three
   sigmoid gates (left, right, output).
3. Optional pairwise mask multiplication.
4. Gating (element‑wise multiplication).
5. Batched matrix multiplication   out = left @ rightᵀ   for every hidden channel.
   This is performed with a strided‑batched GEMM (torch.bmm) which maps
   to highly‑optimized cuBLAS kernels on H100.
6. Layer‑norm on the hidden result, then an element‑wise multiplication with the
   output gate – this multiplication is off‑loaded to a tiny Triton kernel.
7. Final linear projection back to the original dimension.

A Triton kernel is used for the final element‑wise product to satisfy the
“must use a kernel” requirement while keeping the heavy compute in the fast
cuBLAS batched GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, n_elements,
                           BLOCK_SIZE: tl.constexpr):
    """
    Simple element‑wise multiplication kernel:
        c[i] = a[i] * b[i]   for i < n_elements
    All tensors must be contiguous 1‑D views.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(c_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor, shape [B, N, N, C] (C==dim)
            - mask_tensor:  torch.Tensor, shape [B, N, N]  (may be all‑ones)
            - weights_dict: mapping from weight names to torch.Tensors
            - config_dict: contains "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    eps = 1e-5                         # layer‑norm epsilon

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2. Linear projections + gates
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    # Linear maps (bias=False in the reference)
    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight) # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        # broadcast mask over the hidden dimension
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid after a linear projection)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B, N, N, H]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Batched matmul  out = left @ rightᵀ  per hidden channel
    #    (strided‑batched GEMM via torch.bmm)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N) for a strided batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # rightᵀ for each batch entry
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6. LayerNorm on hidden result
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 7. Output gate (sigmoid) + Triton element‑wise multiplication
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))  # [B, N, N, H]

    out_contig = out.contiguous()
    out_gate_contig = out_gate.contiguous()
    out_gated = torch.empty_like(out_contig)

    # launch Triton kernel
    numel = out_contig.numel()
    BLOCK_SIZE = 1024  # works well on H100; can be tuned
    grid = (triton.cdiv(numel, BLOCK_SIZE),)
    elementwise_mul_kernel[grid](
        out_contig,
        out_gate_contig,
        out_gated,
        numel,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # ------------------------------------------------------------------
    # 8. Final linear projection back to dim
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"]
    out_final = F.linear(out_gated, to_out_weight)  # [B, N, N, dim]

    return out_final
