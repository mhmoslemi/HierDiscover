def custom_kernel(data):
    x, mask, weights, config = data
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    # unpack weights as torch tensors.

    # LayerNorm:
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections:
    left = torch.nn.functional.linear(x_norm, left_proj_weight) # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # mask if provided:
    if mask is not None:
        mask_ = mask.unsqueeze(-1) # [B,N,N,1]
        left = left * mask_
        right = right * mask_
    else:
        # If mask is not provided, assume all ones
        pass

    # Gates:
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()   # shape [B,N,N,hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # Move to half precision for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute to bring hidden_dim into batch dimension to launch batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # same

    batch_dim = B * hidden_dim
    left_flat = left_perm.view(batch_dim, N, N)  # [batch_dim, N, N]
    right_flat = right_perm.view(batch_dim, N, N)  # [batch_dim, N, N]

    # Allocate output
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Launch kernel
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch_dim,
    )
    matmul_transposed_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2), # need to confirm order: shape (batch, M, K)

        # M stride = left_flat.stride(1) = K (N)
        # K stride = left_flat.stride(2) = 1

"""
custom_kernel
==============

Implements the “outgoing” TriMul operator (AlphaFold‑3) with a fused
batched GEMM kernel written in Triton.

Algorithm
---------
1. Layer‑norm the input `x` (dim = config["dim"]).
2. Linear projections without bias:
      left  = x @ left_proj_weight                # [B,N,N,H]
      right = x @ right_proj_weight               # [B,N,N,H]
3. Optional pair‑wise mask (broadcast over H).
4. Compute three sigmoid gates and apply them:
      left  = left  * sigmoid(x @ left_gate_weight)
      right = right * sigmoid(x @ right_gate_weight)
5. Core operation (the expensive O(N³) part):
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply of `left` with the *transpose*
   of `right` for each hidden channel `h`.  The kernel flattens the
   `(B,H)` dimensions into a single batch dimension and performs a
   fused GEMM for each batch entry using `tl.dot` (fp16 multiply,
   fp32 accumulation).
6. Layer‑norm over the hidden dimension, multiply by the output gate
   and a final linear projection back to `dim`.

Only the inner GEMM (step 5) is written in Triton; the remaining
operations use PyTorch (high‑level, already highly optimized).  The
kernel works for any batch size ≤ 2, sequence length up to 1024 and
hidden dimension up to 128 / 384 / 768 as required by the tests.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched matmul of A @ Bᵀ
# A : (batch, M, K)   – row‑major (stride_AM = K, stride_AK = 1)
# B : (batch, N, K)   – row‑major but we interpret it as (K,N) by swapping strides
# C : (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ab, stride_am, stride_ak,   # A: batch, row(i), col(k)
    stride_bb, stride_bk, stride_bn,   # B: batch, col(k) (after transpose), row(j)
    stride_cb, stride_cm, stride_cn,   # C: batch, row(i), col(j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    #  Program IDs
    # --------------------------------------------------------------
    pid_b = tl.program_id(2)                 # batch index
    pid_m = tl.program_id(0)                 # tile row of C
    pid_n = tl.program_id(1)                 # tile col of C

    # --------------------------------------------------------------
    #  Offsets in the M and N dimensions
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    #  Accumulator (fp32 for numerical safety)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    #  Loop over K dimension (tile‑wise)
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A block: (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + pid_b * stride_ab \
                       + offs_m[:, None] * stride_am \
                       + cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B block interpreted as (BLOCK_K, BLOCK_N) – note swapped strides
        b_ptrs = b_ptr + pid_b * stride_bb \
                       + cur_k[:, None] * stride_bk \
                       + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Matrix multiply‑accumulate
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # --------------------------------------------------------------
    #  Store the result back to C (cast to fp16)
    # --------------------------------------------------------------
    c_ptrs = c_ptr + pid_b * stride_cb \
                    + offs_m[:, None] * stride_cm \
                    + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask          : torch.Tensor [B, N, N] or ``None``
        - weights       : dict of torch.Tensors (model parameters)
        - config        : dict containing ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper to fetch weight tensors
    # ------------------------------------------------------------------
    def get(name):
        return weights[name].to(device)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = get("norm.weight")
    norm_bias   = get("norm.bias")
    x = F.layer_norm(inp, (dim,), weight=norm_weight, bias=norm_bias)   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, get("left_proj.weight"))   # [B,N,N,hidden_dim]
    right = F.linear(x, get("right_proj.weight"))

    # ------------------------------------------------------------------
    # 3️⃣ Apply optional mask
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)   # [B,N,N,1]
        left  = left  * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, get("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, get("right_gate.weight")))
    out_gate   = torch.sigmoid(F.linear(x, get("out_gate.weight")))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core TriMul (batched GEMM) – move to half precision for speed
    # ------------------------------------------------------------------
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    # Bring hidden_dim into the batch dimension: (B, H, N, N)
    left  = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    batch_dim = B * hidden_dim

    left_flat  = left.view(batch_dim, N, N)   # [B*H, N, N]
    right_flat = right.view(batch_dim, N, N)  # [B*H, N, N]

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Block sizes – tuned for H100 (64 works well for N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles along M
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles along N
        batch_dim,                      # batch dimension
    )

    # Launch the Triton kernel
    _batched_matmul_transpose_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),   # A strides
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),# B (transposed) strides
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),     # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # --------------------------------------------------------------
    # Convert back to float32 and reshape to original layout
    # --------------------------------------------------------------
    out = out_flat.to(torch.float32)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm + output gate + linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=get("to_out_norm.weight"),
                       bias=get("to_out_norm.bias"))
    out = out * out_gate                                 # broadcast over hidden_dim
    out = F.linear(out, get("to_out.weight"))           # [B,N,N,dim]

    return out
