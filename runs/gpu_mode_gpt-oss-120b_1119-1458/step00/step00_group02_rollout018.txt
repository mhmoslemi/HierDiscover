@triton.jit
def batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_lpb, stride_li, stride_lk, stride_ld,
    stride_rpb, stride_rj, stride_rk, stride_rd,
    stride_opb, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bd = tl.program_id(0)  # combined batch*hidden dimension
    pid_m = tl.program_id(1)   # row tile
    pid_n = tl.program_id(2)   # col tile

    # Compute batch and hidden indices
    batch_idx = pid_bd // D
    d_idx = pid_bd % D

    # Compute start offsets
    i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bound i,j
    mask_i = i_offset < N
    mask_j = j_offset < N

    # Pointers offsets
    # base pointers for this batch and channel
    left_base = left_ptr + batch_idx * stride_lpb + d_idx * stride_ld
    right_base = right_ptr + batch_idx * stride_rpb + d_idx * stride_rd
    out_base = out_ptr + batch_idx * stride_opb + d_idx * stride_od

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + i_offset[:, None] * stride_li + k_offset[None, :] * stride_lk
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile (transposed): shape (BLOCK_K, BLOCK_N)

    # Write out

import torch
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_lpb, stride_li, stride_lk, stride_ld,
    stride_rpb, stride_ri, stride_rj, stride_rd,
    stride_opb, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bd = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    batch_idx = pid_bd // D
    d_idx = pid_bd % D

    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = i < N
    mask_j = j < N

    # Base pointers for this batch and channel
    offset_batch = batch_idx * stride_lpb + d_idx * stride_ld
    left_base = left_ptr + offset_batch
    right_base = right_ptr + offset_batch
    out_base = out_ptr + offset_batch

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_idx = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # Load left block (i, k)
        left_ptrs = left_base + i[:, None] * stride_li + k_idx[None, :] * stride_lk
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right block transposed (k, j)
        # right index: (k, j) i.e., original i index is k, original j index is j
        right_ptrs = right_base + k_idx[:, None] * stride_ri + j[None, :] * stride_rj
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(left, right)
    # Store result
    out_ptrs = out_base + i[:, None] * stride_oi + j[None, :] * stride_oj
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) – Triton‑accelerated implementation
----------------------------------------------------
The module computes a “triangular multiplicative update” used in AlphaFold‑3.
Given a 4‑D tensor `x` ∈ [B, N, N, C] it performs:

1. LayerNorm over the channel dimension C.
2. Linear projections to a hidden dimension H (left/right) and three
   gating projections (left‑gate, right‑gate, out‑gate) followed by a sigmoid.
3. Masks the projected tensors (optional) and multiplies by the corresponding gates.
4. Core operation: for each batch‑head (b, h) compute
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   i.e. a batched matrix‑multiply L ⨯ Rᵀ for every hidden channel.
   This O(N³·H) step is the only part implemented in Triton.
5. LayerNorm over the hidden dimension, gated by `out_gate`.
6. Final linear projection back to the original channel dimension.

The Triton kernel is a classic blocked GEMM variant where the batch
dimension is fused with the hidden channel (B·H) so that each program
instance computes one (i, j) tile for one (batch, hidden) pair.
All other operations stay in PyTorch for simplicity and to keep the
implementation concise.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul of the form
#   out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    left_ptr,                # fp16 *
    right_ptr,               # fp16 *
    out_ptr,                 # fp16 *
    B, N, H,                 # int32 scalar parameters
    stride_lpb, stride_li, stride_lk, stride_ld,   # int64 strides of left
    stride_rpb, stride_ri, stride_rj, stride_rd,   # int64 strides of right
    stride_opb, stride_oi, stride_oj, stride_od,   # int64 strides of out
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # --------------------------------------------------------------
    # grid dimensions:
    #   program_id(0) : fused batch * hidden index   (0 ... B*H-1)
    #   program_id(1) : tile row index (i)
    #   program_id(2) : tile col index (j)
    # --------------------------------------------------------------
    pid_bd = tl.program_id(0)
    pid_m  = tl.program_id(1)
    pid_n  = tl.program_id(2)

    # decode batch and hidden ids
    batch_idx = pid_bd // H
    h_idx     = pid_bd %  H

    # offsets for the tile we are computing
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointer for this (batch, hidden) pair
    offset_batch = batch_idx * stride_lpb + h_idx * stride_ld
    left_base  = left_ptr  + offset_batch
    right_base = right_ptr + offset_batch
    out_base   = out_ptr   + offset_batch

    # accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # iterate over the reduction dimension k in blocks of size BLOCK_K
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load left tile (i, k) -------------------------------------------------
        # shape: (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + \
                    offs_i[:, None] * stride_li + \
                    offs_k[None, :] * stride_lk
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)          # fp16 -> promote to fp32 inside dot

        # ---- load right tile transposed (k, j) ------------------------------------
        # we need right_T[k, j] = right[j, k] to perform L @ Rᵀ
        # shape: (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + \
                     offs_k[:, None] * stride_ri + \
                     offs_j[None, :] * stride_rj
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # ---- accumulate ------------------------------------------------------------
        acc += tl.dot(left, right)   # fp16 × fp16 → fp32 accumulation

    # ---- write back -------------------------------------------------------------
    out_ptrs = out_base + \
               offs_i[:, None] * stride_oi + \
               offs_j[None, :] * stride_oj
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Public entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
              - input_tensor : torch.Tensor [B, N, N, C] (float32)
              - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool) or None
              - weights_dict : mapping of all linear/ln weights
              - config_dict  : {"dim": C, "hidden_dim": H}
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections + gating (still in PyTorch)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])            # [B,N,N,H]
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate  = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left_proj.dtype)          # [B,N,N,1]
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 3) Core triangular multiplication (batched GEMM) via Triton
    # ------------------------------------------------------------------
    # Cast to half‑precision for the kernel (memory‑bandwidth friendly)
    left_h  = left.contiguous().to(torch.float16)
    right_h = right.contiguous().to(torch.float16)

    B, N, _, H = left_h.shape
    out_h = torch.empty_like(left_h)   # same shape and dtype (fp16)

    # Gather strides (int64)
    s_lpb, s_li, s_lk, s_ld = left_h.stride()
    s_rpb, s_ri, s_rj, s_rd = right_h.stride()
    s_opb, s_oi, s_oj, s_od = out_h.stride()

    # Block sizes – tuned for H100 (fits in shared memory)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                     # program_id(0) => batch * hidden
        (N + BLOCK_M - 1) // BLOCK_M,              # program_id(1) => i‑tile
        (N + BLOCK_N - 1) // BLOCK_N               # program_id(2) => j‑tile
    )

    batched_matmul_kernel[grid](
        left_h, right_h, out_h,
        B, N, H,
        s_lpb, s_li, s_lk, s_ld,
        s_rpb, s_ri, s_rj, s_rd,
        s_opb, s_oi, s_oj, s_od,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Cast back to float32 for the remaining layers
    out = out_h.to(torch.float32)   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4) Hidden‑dim layer‑norm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])   # back to [B,N,N,dim]

    return out
