import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for batch-wise matrix multiply: C = A * B^T
@triton.jit
def tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak, stride_batch_a,
    stride_bk, stride_bn, stride_batch_b,
    stride_cm, stride_cn, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    C[batch, i, j] = sum_k A[batch, i, k] * B[batch, k, j]
    All tensors are assumed to be row-major and contiguous in the batch dimension.
    A and B are expected in fp16, accumulation in fp32, output in fp32.
    """
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_z = tl.program_id(2)   # batch * head index

    # offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = A_ptr + pid_z * stride_batch_a
    b_base = B_ptr + pid_z * stride_batch_b
    c_base = C_ptr + pid_z * stride_batch_c

    # pointers to the blocks of A and B
    a_ptrs = a_base + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = b_base + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # load tiles (fp16) and cast to fp32 for accumulation
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0).to(tl.float32)
        acc += tl.dot(a, b)

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back result (fp32)
    c_ptrs = c_base + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """

    The heavy triangle multiplicative update `out_{i,j} = Σ_k left_{i,k} * right_{j,k}`

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    # 2) Prepare mask (broadcast to hidden dimension)
    # -----------------------------------------------------------------

        mask = torch.ones(x.shape[:3], dtype=x.dtype, device=device)
    else:

    mask = mask.unsqueeze(-1)            # [B, N, N, 1]

    # -----------------------------------------------------------------

    left = F.linear(x_fp16, left_proj_weight.to(torch.float16))
    right = F.linear(x_fp16, right_proj_weight.to(torch.float16))

    # Apply mask (still in fp16)

    right = right * mask.to(torch.float16)

    # -----------------------------------------------------------------

    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate.to(torch.float16)
    right = right * right_gate.to(torch.float16)

    # 5) Triangle multiplicative update via Triton
    # -----------------------------------------------------------------
    B, N, _, H = left.shape                    # H == hidden_dim
    batch_head = B * H

    # reshape to (batch_head, N, N) for a batched matmul
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(batch_head, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(batch_head, N, N)

    # right must be transposed on the last two axes before the multiplication
    right_t = right_mat.transpose(1, 2).contiguous()

    # Allocate output buffer (fp32)
    out_mat = torch.empty(batch_head, N, N, dtype=torch.float32, device=device)

    BLOCK_N = 64
    BLOCK_K = 64

    # Compute strides (convert to Python integers)
    stride_am = left_mat.stride(1)          # row stride
    stride_ak = left_mat.stride(2)          # col stride
    stride_batch_a = left_mat.stride(0)

    stride_bk = right_t.stride(1)
    stride_bn = right_t.stride(2)
    stride_batch_b = right_t.stride(0)

    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)
    stride_batch_c = out_mat.stride(0)

        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_head,

        left_mat,
        right_t,
        out_mat,
        N, N, N,
        stride_am, stride_ak, stride_batch_a,
        stride_bk, stride_bn, stride_batch_b,
        stride_cm, stride_cn, stride_batch_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to original layout [B, N, N, hidden]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp32

    # -----------------------------------------------------------------
    # 6) Output LayerNorm + gating + final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, to_out_weight)

    return out

"""
TriMul (outgoing) forward pass with a Triton kernel.

The dominant computation is:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

where `left` and `right` are the projected and gated pairwise tensors.
We collapse the batch and hidden dimensions (B·H) and run a batched
matrix‑multiply C = A @ Bᵀ for each (batch, head) pair using a custom
Triton kernel.  All surrounding LayerNorms, gates and linear layers are
implemented in PyTorch.  The kernel works in fp16 for A/B, accumulates in
fp32 and writes out fp32, which matches the rest of the network.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak, stride_batch_a,
    stride_bk, stride_bn, stride_batch_b,
    stride_cm, stride_cn, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute for each batch‑head:
        C[i, j] = Σ_k A[i, k] * B[k, j]

    All tensors are row‑major.  A and B are fp16, accumulation is fp32,
    and C is written as fp32.
    """
    pid_m = tl.program_id(0)            # block row
    pid_n = tl.program_id(1)            # block col
    pid_z = tl.program_id(2)            # batch·head index

    # offsets inside the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # out‑of‑bounds masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch‑head
    a_base = A_ptr + pid_z * stride_batch_a
    b_base = B_ptr + pid_z * stride_batch_b
    c_base = C_ptr + pid_z * stride_batch_c

    # pointers to the current tiles of A and B
    a_ptrs = a_base + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = b_base + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0).to(tl.float32)          # (BM, BK)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0).to(tl.float32)          # (BK, BN)
        acc += tl.dot(a, b)                              # (BM, BN)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write result
    c_ptrs = c_base + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
