def custom_kernel(data):
    # unpack
    inp, mask, weights, config = data

    # casting to appropriate dtype: assume float32.

    # Setup eps
    eps = config.get('eps', 1e-5)

    # Load layernorm parameters
    # norm weight shape (dim,)
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']

    # Use torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(inp, (config['dim'],), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left_proj_weight = weights['left_proj.weight']  # shape (hidden_dim, dim)
    right_proj_weight = weights['right_proj.weight']

    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # shape (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if not nomask
    if mask is not None:
        # mask shape (B,N,N)
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_

    # Compute gates
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Compute out via batched GEMM for each hidden channel
    batch_size, seq_len, _, hidden_dim = left.shape

    # Convert to half for speed
    # optional: we can use full precision for correctness
    # We'll use float16 to accelerate but convert back to float32 before following layers.
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Permute and reshape
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)
    right_perm = right_fp16.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) transposed

    left_flat = left_perm.view(batch_size * hidden_dim, seq_len, seq_len)
    right_flat = right_perm.view(batch_size * hidden_dim, seq_len, seq_len)

    # Use torch.bmm (float16) which utilizes Tensor cores.
    out_flat = torch.bmm(left_flat, right_flat)  # shape (B*H, N, N) float16

    # Reshape back
    out = out_flat.view(batch_size, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,H)

    # Convert to float32 for layernorm and linear
    out = out.to(inp.dtype)

    # to_out_norm
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # final linear to dim
    to_out_weight = weights['to_out.weight']
    out_final = torch.nn.functional.linear(out_norm, to_out_weight)  # shape (B,N,N,dim)

    return out_final

"""
TriMul forward pass (outgoing version) with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B,N,N,dim]).
2. Linear projections to the hidden dimension:
        left  = x @ left_proj_weight   (no bias)
        right = x @ right_proj_weight
3. Apply an optional mask and the per‑position gates.
   The multiplication `left = left * mask * left_gate` (and analog for right)
   is performed by a small Triton kernel that fuses three element‑wise
   multiplications into a single pass over memory.
4. Compute the core contraction
        out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
   which is equivalent to a batched matrix multiplication for each hidden
   channel.  We reshape the tensors to a batch of size B*hidden_dim and
   invoke `torch.bmm` in half precision (FP16) to exploit Tensor‑cores
   on the H100.
5. Layer‑norm over the hidden dimension, apply the output gate, and a
   final linear projection back to `dim`.
"""
import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def fused_mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,  # pointers
    size,                           # total number of elements
    BLOCK_SIZE: tl.constexpr,       # compile‑time constant
):
    """out = a * b * c  (all tensors are 1‑D contiguous)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)

    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """Fused element‑wise product of three tensors using Triton."""
    assert a.shape == b.shape == c.shape
    a = a.contiguous()
    b = b.contiguous()
    c = c.contiguous()
    out = torch.empty_like(a)

    total = a.numel()
    BLOCK = 1024  # tuneable; 1024 works well on H100
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_kernel[grid](
        a, b, c, out, total, BLOCK_SIZE=BLOCK
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim]   (float32, on CUDA)
            - mask_tensor:  [B, N, N]       (float32 or bool, on CUDA)
            - weights_dict: model weights (see skeleton)
            - config_dict:  contains at least "dim", "hidden_dim",
                            optional "eps" and "nomask".

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Layer‑norm on the input
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right_proj = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Gating tensors (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))

    # ------------------------------------------------------------------
    # Apply mask and gates – fused Triton kernel when a mask is present
    # ------------------------------------------------------------------
    if nomask:
        # No mask: simple element‑wise product in PyTorch (still fast)
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # Expand mask to hidden dimension and make it contiguous
        mask_exp = mask.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()
        left = fused_mul3(left_proj, mask_exp, left_gate)
        right = fused_mul3(right_proj, mask_exp, right_gate)

    # ------------------------------------------------------------------
    # Core contraction: out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    # Equivalent to a batched GEMM for each hidden channel.
    # We use half‑precision to exploit Tensor‑cores.
    # ------------------------------------------------------------------
    # Cast to FP16 for the matmul
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # (B, H, N, N) where the third axis is the “k” dimension
    left_mat = left_fp16.permute(0, 3, 1, 2).contiguous()       # [B, H, N, N]
    right_mat = right_fp16.permute(0, 3, 2, 1).contiguous()    # [B, H, N, N] (k,j)

    # Collapse the batch‑and‑hidden dimensions for a single bmm call
    left_flat = left_mat.view(B * hidden_dim, N, N)   # [B*H, M=N, K=N]
    right_flat = right_mat.view(B * hidden_dim, N, N)  # [B*H, K=N, N]

    # Batched matrix multiplication (FP16) → FP16 result
    out_flat = torch.bmm(left_flat, right_flat)       # [B*H, N, N]

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(inp.dtype)   # back to FP32 for the remaining layers

    # ------------------------------------------------------------------
    # Output layer‑norm over hidden dimension
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Output gate
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))  # [B,N,N,hidden_dim]
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_final = F.linear(out_norm, weights["to_out.weight"])   # [B,N,N,dim]

    return out_final
