"""
TriMul (outgoing) implementation in Triton.

The module performs the following steps (see AlphaFold3 TriMul):
1. Layer‑Norm over the last channel.
2. Two linear projections (left/right) and three gating projections.
3. Apply an optional mask, then element‑wise gating.
4. Core cubic update:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is implemented as a batch of independent GEMMs:
   for each (b, d) we compute a matrix multiplication
   (N×N) @ (N×N)^T using a custom Triton kernel.
5. Layer‑Norm over the hidden dimension, gating and final linear
   projection back to the original channel dimension.

Only the cubic update is performed on the GPU via Triton; the remaining
operations are expressed with PyTorch (which also runs on the GPU).
The kernel works for any batch size (≤2), sequence length up to 1024,
and hidden dimensions ≤ 384.  Input/weight tensors are expected to be on
the same CUDA device.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM with the second operand transposed.
#   A : [BH, M, K]   (BH = batch * hidden_dim, M = N, K = N)
#   B : [BH, N, K]   (same K, rows correspond to the "j" index)
#   C : [BH, M, N]   (result)
#   Computes C = A @ B.T  i.e. Σ_k A[i,k] * B[j,k]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr,                     # *mut fp16
    b_ptr,                     # *mut fp16
    c_ptr,                     # *mut fp16
    M, N, K,                   # problem size (int)
    stride_am, stride_ai, stride_ak,   # strides for A (BH, M, K)
    stride_bm, stride_bj, stride_bk,   # strides for B (BH, N, K)
    stride_cm, stride_ci, stride_cj,   # strides for C (BH, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block index over M
    pid_n = tl.program_id(1)          # block index over N
    pid_bh = tl.program_id(2)         # block index over batch*hidden

    # ------------------------------------------------------------------
    # Compute offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for the current (bh, m, n) tile
    a_base = a_ptr + pid_bh * stride_am \
                   + offs_m[:, None] * stride_ai \
                   + 0 * stride_ak          # k offset will be added inside loop
    b_base = b_ptr + pid_bh * stride_bm \
                   + offs_n[:, None] * stride_bj \
                   + 0 * stride_bk

    # accumulator in FP32 (more stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K-loop
    # ------------------------------------------------------------------
    for k_iter in range(0, tl.cdiv(K, BLOCK_K)):
        # Current k coordinates
        cur_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        mask_a = (offs_m < M)[:, None] & (cur_k < K)[None, :]
        mask_b = (offs_n < N)[:, None] & (cur_k < K)[None, :]

        # Load a tile of shape (BLOCK_M, BLOCK_K) and (BLOCK_N, BLOCK_K)
        a = tl.load(a_base + cur_k[None, :] * stride_ak,
                    mask=mask_a, other=0.0)
        b = tl.load(b_base + cur_k[None, :] * stride_bk,
                    mask=mask_b, other=0.0)

        # Convert to FP32 before accumulation
        a_f32 = a.to(tl.float32)
        b_f32 = b.to(tl.float32)

        # Note: tl.dot on a (M×K) and b (N×K) returns (M×N)
        acc += tl.dot(a_f32, b_f32)

    # ------------------------------------------------------------------
    # Write back the result
    # ------------------------------------------------------------------
    c_base = c_ptr + pid_bh * stride_cm \
                    + offs_m[:, None] * stride_ci \
                    + offs_n[None, :] * stride_cj

    mask_c = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    # Store as fp16 (the surrounding code casts back to fp32 when needed)
    tl.store(c_base, acc.to(tl.float16), mask=mask_c)


# ----------------------------------------------------------------------
# Public entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor [B, N, N, D]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]  (optional, bool or float)
        - weights_dict : dict of torch.Tensor containing all model weights
        - config_dict  : dict with at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    if len(data) == 4:
        x, mask, weights, config = data
    else:  # some tests may omit the mask and pass (x, weights, config)
        x, weights, config = data
        mask = None

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = x.device

    # ------------------------------------------------------------------
    # Load weights (they are already on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the channel dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        x, (dim,), weight=norm_weight, bias=norm_bias
    )  # [B, N, N, D], float32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # ensure mask is float32 for multiplication
        mask_f = mask.to(x.dtype).unsqueeze(-1)               # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and element‑wise gating of projections
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Cast to fp16 for the cubic update (saves bandwidth)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Bring the hidden dimension next to the batch dimension:
    #   [B, N, N, H] -> [B, H, N, N] -> flatten to [BH, N, N]
    B, N, _, _ = left_fp16.shape
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()
    BH = B * hidden_dim

    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)

    # ------------------------------------------------------------------
    # 6️⃣ Triton GEMM (left @ right.T) → out_flat [BH, N, N]
    # ------------------------------------------------------------------
    out_flat = torch.empty_like(left_flat)  # fp16 output buffer

    # Block sizes – tuned for H100 fp16 matmul
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks over rows (i)
        triton.cdiv(N, BLOCK_N),   # blocks over cols (j)
        BH,                         # one program per batch‑head
    )

    tri_mul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                         # M
        N,                         # N
        N,                         # K (same as sequence length)
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H] (still fp16)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # back to FP32 for the remaining ops

    # ------------------------------------------------------------------
    # 7️⃣ Final hidden‑dim layer‑norm, gating and projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate  # out_gate already FP32 with shape [B, N, N, H]
    out = torch.nn.functional.linear(out, to_out_weight)  # back to dim channels

    # ------------------------------------------------------------------
    # Return tensor of shape [B, N, N, D] (float32)
    # ------------------------------------------------------------------
    return out
