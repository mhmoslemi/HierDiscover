@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)  # row block idx
    pid_j = tl.program_id(1)  # col block idx
    pid_bh = tl.program_id(2) # combined batch & hidden index

    # compute coordinates:
    b = pid_bh // H
    h = pid_bh % H

    # start coordinates
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K

        # offsets for left: [i,k] slice
        # left with shape [B,N,N,H]
        # i offset: i0 + i_offset; k offset: k0 + k_offset
        # constraint: i < N, k < N
        # compute mask
        i_offset = i0 + tl.arange(0, BLOCK_M)
        k_offset = k0 + tl.arange(0, BLOCK_K)

        # Build ptr offsets: left_ptr + b * stride_left_b + i_offset[:, None]*stride_left_i + k_offset[None,:]*stride_left_k + h*stride_left_h
        left_ptrs = left_ptr + b * stride_left_b + i_offset[:, None] * stride_left_i + k_offset[None, :] * stride_left_k + h * stride_left_h

        left = tl.load(left_ptrs, mask = (i_offset[:, None] < N) & (k_offset[None,:] < N), other=0.0)

        # right_T slice: we need right with shape [B,N,N,H] but transposed k,i? Actually we need right_T shape (K,N) for given h,b.
        # We'll load right_T where rows correspond to k and cols to j.
        j_offset = j0 + tl.arange(0, BLOCK_N)

        # right pointer offset: b*stride_right_b + k_offset[:, None]*stride_right_k + j_offset[None,:]*stride_right_j + h*stride_right_h
        right_ptrs = right_ptr + b * stride_right_b + k_offset[:, None] * stride_right_k + j_offset[None,:] * stride_right_j + h * stride_right_h

"""
tri_mul_outgoing.py – Triton‑accelerated “outgoing” Triangle Multiplicative Update.

The kernel implements the core tensor contraction of AlphaFold3’s TriMul module:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

for every batch `b` and hidden channel `h`.  The surrounding LayerNorm,
linear projections, and sigmoid gates are performed with PyTorch for
convenience; the O(N³·C) contraction (the only heavy part) is computed with a
custom Triton kernel that treats each (b, h) pair as an independent
matrix‑multiply `left[b,:, :,h] @ right[b,:, :,h].T`.  The implementation
supports arbitrary batch size (1 or 2), sequence lengths up to 1024 and hidden
dimensions up to 768, works with optional masks, and returns a float32 tensor
matching the reference PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,          # *float32   left  tensor  [B,N,N,H]
    right_ptr,         # *float32   right tensor  [B,N,N,H]
    out_ptr,           # *float32   output tensor [B,N,N,H]
    B: tl.int32, N: tl.int32, H: tl.int32,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a tiled GEMM where the right‑hand side is loaded in transposed order
    (k as rows, j as columns) so that a single `tl.dot` can be used.
    """
    pid_i = tl.program_id(0)          # tile row index (i)
    pid_j = tl.program_id(1)          # tile col index (j)
    pid_bh = tl.program_id(2)         # combined batch and hidden channel

    # decode batch and hidden channel
    b = pid_bh // H
    h = pid_bh % H

    # start offsets for this tile
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # range vectors for the tile (int32)
    i_range = i_base + tl.arange(0, BLOCK_M)
    j_range = j_base + tl.arange(0, BLOCK_N)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in blocks of size BLOCK_K
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_base = kb * BLOCK_K
        k_range = k_base + tl.arange(0, BLOCK_K)

        # ---------- load left slice [i,k] ----------
        # left[b, i, k, h] -> offsets = b*sb + i*si + k*sk + h*sh
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            i_range[:, None] * stride_left_i + \
            k_range[None, :] * stride_left_k + \
            h * stride_left_h

        mask_left = (i_range[:, None] < N) & (k_range[None, :] < N)
        left = tl.load(left_ptrs, mask=mask_left, other=0.0, dtype=tl.float32)

        # ---------- load right slice transposed [k,j] ----------
        # we want right_T[k, j] = right[b, j, k, h]
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            k_range[:, None] * stride_right_k + \
            j_range[None, :] * stride_right_i + \
            h * stride_right_h

        mask_right = (k_range[:, None] < N) & (j_range[None, :] < N)
        right_T = tl.load(right_ptrs, mask=mask_right, other=0.0, dtype=tl.float32)

        # ---------- accumulate dot product ----------
        acc += tl.dot(left, right_T)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # ---------- store the result ----------
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        i_range[:, None] * stride_out_i + \
        j_range[None, :] * stride_out_j + \
        h * stride_out_h

    mask_out = (i_range[:, None] < N) & (j_range[None, :] < N)
    tl.store(out_ptrs, acc, mask=mask_out)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on the same device).
    """
    # ----------------------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ----------------------------------------------------------------------
    # LayerNorm on the input (norm over the last dimension `dim`)
    # ----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ----------------------------------------------------------------------
    # Linear projections (no bias) and gating
    # ----------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ----------------------------------------------------------------------
    # Apply mask (if any) and gating
    # ----------------------------------------------------------------------
    if mask_tensor is None:
        mask = torch.ones((bs, seq_len, seq_len), dtype=dtype, device=device)
    else:
        mask = mask_tensor

    mask = mask.unsqueeze(-1).to(dtype)               # [B,N,N,1] broadcast over hidden dim
    left  = left * mask * left_gate
    right = right * mask * right_gate

    # ----------------------------------------------------------------------
    # Core contraction via Triton
    # ----------------------------------------------------------------------
    # Prepare output tensor
    out = torch.empty((bs, seq_len, seq_len, hidden_dim), dtype=torch.float32, device=device)

    # Strides in elements (not bytes)
    stride_left  = left.stride()    # (B_stride, i_stride, k_stride, h_stride)
    stride_right = right.stride()
    stride_out   = out.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (seq_len + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        bs * hidden_dim                      # combined (batch, hidden) dimension
    )

    # Launch the Triton kernel
    tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        bs, seq_len, hidden_dim,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----------------------------------------------------------------------
    # Post‑processing: LayerNorm over hidden_dim, gating, final linear proj
    # ----------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    out = out * out_gate  # broadcast gating

    # final projection back to `dim`
    result = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return result
