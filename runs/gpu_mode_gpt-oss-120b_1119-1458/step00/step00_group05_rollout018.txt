    norm_bias = weights["norm.bias"]

    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    eps = 1e-5
    # compute mean and var across last dim
    # input is (B,N,N,dim)
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)

    left = torch.nn.functional.linear(x_norm, left_proj_weight) # shape (B,N,N,hidden_dim)

        right = right * mask_exp

@triton.jit
def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)  # combined batch*H*block_i
    pid1 = tl.program_id(1)  # block_j

    # Compute total blocks per dimension
    num_i_blocks = tl.cdiv(N, BLOCK_M)

    # compute batch_h and i_block
    batch_h_i = pid0
    batch_h = batch_h_i // num_i_blocks
    i_block = batch_h_i % num_i_blocks

    # Decompose batch_h into b and h
    b = batch_h // H
    h = batch_h % H

    # Compute offsets
    i_offset = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid1 * BLOCK_N + tl.arange(0, BLOCK_N)
    k_offset = tl.arange(0, BLOCK_K)

    # Compute base pointers
    # left has shape (B, N, N, H) with axes (b, i, k, h)
    a_ptrs = left_ptr + b * stride_left_b + h * stride_left_h + i_offset[:, None] * stride_left_i + k_offset[None, :] * stride_left_k

    # right has shape (B, N, N, H) with axes (b, j, k, h)
    # we need right transposed: need (k,j) for dot; i.e., we need pointer to right such that we load (k, j) => [k, j].
    # To load that, pointer index: k_offset[:, None] * stride_right_k + j_offset[None, :] * stride_right_j
    b_ptrs = right_ptr + b * stride_right_b + h * stride_right_h + k_offset[:, None] * stride_right_k + j_offset[None, :] * stride_right_j

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Mask for edge cases
    # compute masks for i and j: if i_offset < N and j_offset < N.
    i_mask = i_offset < N
    j_mask = j_offset < N

    for start_k in range(0, tl.cdiv(N, BLOCK_K)):
        # calculate actual K dimension size for this block
        cur_k = start_k * BLOCK_K
        k_valid = (cur_k + k_offset) < N

        # load A block
        a = tl.load(a_ptrs, mask=i_mask[:, None] & k_valid[None, :], other=0.0).to(tl.float32)
        # load B_T block (k,j)
        b = tl.load(b_ptrs, mask=k_valid[:, None] & j_mask[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

        # advance pointers
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # Store result
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + i_offset[:, None] * stride_out_i + j_offset[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])

```
import torch
import triton
import triton.language as tl

@triton.jit
def trmul_outgoing_kernel(left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # compute block indices
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)

    num_i_blocks = tl.cdiv(N, BLOCK_M)

    batch_h = pid0 // num_i_blocks
    i_block = pid0 % num_i_blocks

    b = batch_h // H
    h = batch_h % H

    # Offsets
    i_offset = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid1 * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for edges
    i_mask = i_offset < N
    j_mask = j_offset < N

    # Base pointers
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Prepare pointers for each block of K
    a_ptrs = left_base + i_offset[:, None] * stride_left_i + tl.arange(0, BLOCK_K)[None, :] * stride_left_k
    b_ptrs = right_base + tl.arange(0, BLOCK_K)[:, None] * stride_right_k + j_offset[None, :] * stride_right_i

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        # compute k offset for mask
        k_start = kb * BLOCK_K
        k_offset = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # load tiles, applying masks
        a = tl.load(a_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

        # advance pointers to next K block
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    hidden_dim = config["hidden_dim"]

    # LayerNorm op
    eps = 1e-5
    # x_norm = (input - mean) / sqrt(var + eps) * weight + bias
    # compute mean and var across dim axis
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()

    right = right * right_gate

    right_h = right.to(torch.float16).contiguous()

    BLOCK_N = 64

"""
TriMul “outgoing” kernel (AlphaFold3 style) – fast Triton implementation.

The forward pass consists of:
1. Layer‑norm over the last dimension of the pairwise input.
2. Two linear projections (left/right) + optional binary mask.
3. Three sigmoid‑gates (left, right, out) computed from the normalized input.
4. Element‑wise gating of the left/right projections.
5. A batched matrix‑multiply per hidden channel:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is exactly a per‑channel mat‑mul L @ Rᵀ.
   This O(N³·C) step is performed by a custom Triton kernel.
6. Layer‑norm over the hidden dimension, multiplication by the out‑gate,
   and a final linear projection back to the original feature dimension.

The kernel fuses the inner‑product loops and uses FP16 arithmetic with
FP32 accumulation for speed on H100 GPUs.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# ----------------------------------------------------------------------
# Triton kernel: batched matmul L @ Rᵀ for each (batch, channel) pair
# ----------------------------------------------------------------------
@triton.jit
def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,              # pointers
    B, N, H,                                   # runtime sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    Stored in tensors of shape (B, N, N, H)  (contiguous, row‑major).
    """
    pid0 = tl.program_id(0)          # flattened over (batch*H) × i‑blocks
    pid1 = tl.program_id(1)          # j‑block index

    # ------------------------------------------------------------------
    # 1) Decode the 3‑D index (b, h, i_block) from pid0
    # ------------------------------------------------------------------
    num_i_blocks = tl.cdiv(N, BLOCK_M)
    batch_h_idx = pid0 // num_i_blocks
    i_block = pid0 % num_i_blocks

    b = batch_h_idx // H
    h = batch_h_idx % H

    # ------------------------------------------------------------------
    # 2) Compute absolute offsets for the i and j dimensions
    # ------------------------------------------------------------------
    i_offset = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid1     * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_offset < N
    j_mask = j_offset < N

    # ------------------------------------------------------------------
    # 3) Base pointers for this (b, h) pair
    # ------------------------------------------------------------------
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # ------------------------------------------------------------------
    # 4) Pointers to the start of the K‑dimension tiles
    # ------------------------------------------------------------------
    a_ptrs = left_base + i_offset[:, None] * stride_left_i + tl.arange(0, BLOCK_K)[None, :] * stride_left_k
    b_ptrs = right_base + tl.arange(0, BLOCK_K)[:, None] * stride_right_k + j_offset[None, :] * stride_right_i

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 5) Loop over K‑tiles
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        k_start = kb * BLOCK_K
        k_offset = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # Load a tile of shape (BLOCK_M, BLOCK_K) from left
        a = tl.load(a_ptrs,
                    mask=i_mask[:, None] & k_mask[None, :],
                    other=0.0).to(tl.float32)

        # Load a tile of shape (BLOCK_K, BLOCK_N) from rightᵀ (i.e. right transposed)
        b = tl.load(b_ptrs,
                    mask=k_mask[:, None] & j_mask[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)                # FP32 accumulation

        # Advance K‑pointers to the next tile
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # ------------------------------------------------------------------
    # 6) Write the result tile
    # ------------------------------------------------------------------
    out_ptrs = out_base + i_offset[:, None] * stride_out_i + j_offset[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])


# ----------------------------------------------------------------------
# Public entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operation.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]  (bool / float, may be None)
        - weight_dict  : mapping from string names to torch.Tensor
        - config_dict  : must contain 'dim' and 'hidden_dim';
                         optional 'nomask' (True → ignore mask)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=eps)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])    # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional binary mask (broadcasted on the hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).type_as(left)   # [B,N,N,1]
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid of linear transforms of the normalized input)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Batched mat‑mul L @ Rᵀ using the Triton kernel
    # ------------------------------------------------------------------
    # Cast to FP16 to reduce memory traffic – kernel accumulates in FP32
    left_h  = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    B, N, _, H = left_h.shape

    # Output buffer (FP16)
    out_h = torch.empty((B, N, N, H), dtype=torch.float16, device=left_h.device)

    # Strides (row‑major layout)
    stride_left_b,  stride_left_i, stride_left_k, stride_left_h = left_h.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right_h.stride()
    stride_out_b,   stride_out_i,  stride_out_j,  stride_out_h = out_h.stride()

    # Block sizes – chosen for good occupancy on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (B * H * ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    num_i_blocks = (N + BLOCK_M - 1) // BLOCK_M
    num_j_blocks = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * H * num_i_blocks, num_j_blocks)

    trmul_outgoing_kernel[grid](
        left_h, right_h, out_h,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4  # H100 handles at least 4 warps comfortably
    )

    # Cast back to FP32 for the rest of the pipeline
    out = out_h.to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Post‑multiplication Layer‑norm + out‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=eps)

    out = out * out_gate   # element‑wise gating after norm

    # ------------------------------------------------------------------
    # 7) Final linear projection back to the original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
