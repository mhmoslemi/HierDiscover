def custom_kernel(data):
    input_tensor, mask, weights, config = data

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5 # for layernorm maybe.

    # Norm: using weight and bias
    # compute mean,var across last dim (dim)
    # Use torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Project
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # mask
    if not config.get("nomask", True):
        # mask shape [B,N,N]; unsqueeze -1 to broadcast
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast to half
    left_hp = left.half()
    right_hp = right.half()

    # Prepare output
    out_hp = torch.empty_like(left_hp)  # same shape

    # get strides
    # for left, shape (B,N,N,H), strides: left.stride()
    s_left = left_hp.stride()  # (stride_b, stride_i, stride_k, stride_h)
    s_right = right_hp.stride()
    s_out = out_hp.stride()

    # define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = ( (N + BLOCK_M - 1)//BLOCK_M,
             (N + BLOCK_N - 1)//BLOCK_N,
             bs * hidden_dim )
    # Launch kernel
    trilmul_kernel[grid](
        left_hp,
        right_hp,
        out_hp,
        bs,
        N,
        hidden_dim,
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[1], s_right[2], s_right[3],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert to float32
    out = out_hp.to(torch.float32)

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)
    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

The implementation follows the reference PyTorch TriMul module:

1. Layer‑norm the input tensor (shape [B,N,N,dim]).
2. Linear projections → left, right (shape [B,N,N,hidden_dim]).
3. Optional pairwise mask (broadcasted on the last dim).
4. Gating: sigmoid(linear(x)) for left, right and out.
5. Apply gates to left/right.
6. Core O(N³) operation:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each hidden dimension.
   It is performed by a custom Triton kernel that computes
   a tile‑wise GEMM for every (batch, hidden) pair.
7. Layer‑norm + out‑gate on the hidden dimension, then a final linear
   projection back to the original dim.

Only the heavy matrix‑multiplication (step 6) is written in Triton;
the remaining steps use native PyTorch for simplicity and correctness.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trilmul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # problem size
    B, N, H,
    # strides for left = (B,N,N,H)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right = (B,N,N,H)
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    # strides for out = (B,N,N,H)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM across the ‘k’ dimension.

    Each program instance computes a TILE_M × TILE_N block of the
    output for a specific (batch, hidden) pair.
    """
    pid_m = tl.program_id(0)   # tile row (i)
    pid_n = tl.program_id(1)   # tile col (j)
    pid_bh = tl.program_id(2)  # combined batch * hidden index

    # --------------------  index computation  --------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    bh = pid_bh
    b = bh // H                # batch index
    h = bh % H                 # hidden‑dim index

    # offsets within the tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b,h) pair
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------  tiled reduction over k  --------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_tiles):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, cur_K)
        left_offset = offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_base + left_offset,
                       mask=(mask_i[:, None] & mask_k[None, :]),
                       other=0.0)

        # right block: shape (BLOCK_N, cur_K) – we load as (N, K) and use trans_b=True
        right_offset = offs_j[:, None] * stride_right_i + offs_k[None, :] * stride_right_k
        right = tl.load(right_base + right_offset,
                        mask=(mask_j[:, None] & mask_k[None, :]),
                        other=0.0)

        # GEMM: left (M×K)  •  rightᵀ (K×N) → (M×N)
        # tl.dot returns the same dtype as the inputs (float16), so cast to float32
        acc += tl.dot(left, right, trans_b=True).to(tl.float32)

    # --------------------  write result  --------------------
    out_offset = offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_base + out_offset,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    TriMul forward pass (outgoing version) with a Triton‑accelerated
    hidden‑dim batched matrix multiplication.
    Input:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Load model weights (all are float32 tensors)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(input_tensor,
                          normalized_shape=(dim,),
                          weight=norm_weight,
                          bias=norm_bias,
                          eps=eps)

    # ------------------------------------------------------------------
    # 2) Linear projections & optional mask
    # ------------------------------------------------------------------
    # Linear proj: [B,N,N,dim] → [B,N,N,hidden_dim]
    left  = F.linear(x_norm, left_proj_weight)   # shape (bs,N,N,hidden_dim)
    right = F.linear(x_norm, right_proj_weight)

    if not nomask:
        # mask shape [B,N,N] → broadcast to last dim
        mask_f = mask.to(left.dtype).unsqueeze(-1)   # (B,N,N,1)
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 3) Gating (sigmoid on linear(x_norm))
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4) Cast to half precision for the O(N³) kernel
    # ------------------------------------------------------------------
    left_h  = left.contiguous().half()
    right_h = right.contiguous().half()
    # Prepare output buffer (half)
    out_h = torch.empty_like(left_h)

    # Strides (in number of elements, not bytes)
    s_left  = left_h.stride()   # (b, i, k, h)
    s_right = right_h.stride()
    s_out   = out_h.stride()

    # ------------------------------------------------------------------
    # 5) Triton kernel launch (batched GEMM per hidden dimension)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        bs * hidden_dim,                     # one program per (batch, hidden)
    )

    trilmul_kernel[grid](
        left_h,
        right_h,
        out_h,
        bs,
        N,
        hidden_dim,
        # left strides
        s_left[0], s_left[1], s_left[2], s_left[3],
        # right strides
        s_right[0], s_right[1], s_right[2], s_right[3],
        # out strides
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Back to float32 and post‑processing
    # ------------------------------------------------------------------
    out = out_h.to(torch.float32)

    # Layer‑norm on hidden dimension
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    # Apply out‑gate
    out = out * out_gate

    # Final linear projection back to original dim
    out = F.linear(out, to_out_weight)

    return out
