def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5  # typical

    # LayerNorm over dim (last dimension)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # compute mean and var manually using torch?
    # Use torch.nn.functional.layer_norm: expects normalized_shape = dim
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # mask handling
    if mask is not None:
        mask_expanded = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_expanded
        right = right * mask_expanded

    # gate computations
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # compute contraction with Triton

    # Pre-permute right for k,j order
    right_perm = right.permute(0, 2, 1, 3).contiguous()  # shape [B,N,N,hidden_dim] with dims k,j,h

    # Prepare output tensor
    out = torch.empty_like(left)  # same shape as left

    # stride extraction
    B, N, _, H = left.shape
    stride_left = left.stride()  # (stride_b, stride_i, stride_k, stride_h)
    stride_right = right_perm.stride()
    stride_out = out.stride()

    # Convert to int64
    import triton
    import triton.language as tl

    @triton.jit
    def trimul_kernel(left_ptr, right_ptr, out_ptr,
                      B, N, H,
                      stride_l_b, stride_l_i, stride_l_k, stride_l_h,
                      stride_r_b, stride_r_k, stride_r_j, stride_r_h,
                      stride_o_b, stride_o_i, stride_o_j, stride_o_h,
                      BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr):
        pid0 = tl.program_id(0)  # batch * head index
        pid1 = tl.program_id(1)  # i tile
        pid2 = tl.program_id(2)  # j tile

        b = pid0 // H
        h = pid0 % H

        i_offset = pid1 * BLOCK_I + tl.arange(0, BLOCK_I)
        j_offset = pid2 * BLOCK_J + tl.arange(0, BLOCK_J)

        mask_i = i_offset < N
        mask_j = j_offset < N

        acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

        # Loop over K dimension
        for k in range(0, tl.cdiv(N, BLOCK_K)):
            k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = k_offset < N

            # compute pointers
            left_ptrs = left_ptr + b*stride_l_b + h*stride_l_h + i_offset[:, None]*stride_l_i + k_offset[None, :]*stride_l_k
            right_ptrs = right_ptr + b*stride_r_b + h*stride_r_h + k_offset[:, None]*stride_r_k + j_offset[None, :]*stride_r_j

            left_vals = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
            right_vals = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

            acc += tl.dot(left_vals, right_vals)

        # Store result
        out_ptrs = out_ptr + b*stride_o_b + h*stride_o_h + i_offset[:, None]*stride_o_i + j_offset[None, :]*stride_o_j
        tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    # Block sizes: Choose 64 x 64 tile, K tile 64.
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    grid = (B * H, tl.cdiv(N, BLOCK_I), tl.cdiv(N, BLOCK_J))

    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right_perm,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_l_b=stride_left[0],
        stride_l_i=stride_left[1],
        stride_l_k=stride_left[2],
        stride_l_h=stride_left[3],
        stride_r_b=stride_right[0],
        stride_r_k=stride_right[1],
        stride_r_j=stride_right[2],
        stride_r_h=stride_right[3],
        stride_o_b=stride_out[0],
        stride_o_i=stride_out[1],
        stride_o_j=stride_out[2],
        stride_o_h=stride_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )
    # After kernel, out has shape [B,N,N,H]

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # apply out_gate
    out = out * out_gate

    # final linear
    to_out_weight = weights["to_out.weight"]
    output = torch.nn.functional.linear(out, to_out_weight)  # shape [B,N,N,dim]  

    return output

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm over dim
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask if needed
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Gating
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast to float32 for kernel (if not already)
    left = left.to(torch.float32)
    right = right.to(torch.float32)

    # Permute right to (b,k,j,h) for matmul
    right_perm = right.permute(0, 2, 1, 3).contiguous()

    # Allocate output
    out = torch.empty_like(left, dtype=torch.float32)

    # Strides
    B, N, _, H = left.shape
    stride_l = left.stride()
    stride_r = right_perm.stride()
    stride_o = out.stride()

    # Block sizes
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    # Launch kernel
    grid = (B * H, triton.cdiv(N, BLOCK_I), triton.cdiv(N, BLOCK_J))

        out,

        stride_r[0], stride_r[1], stride_r[2], stride_r[3],

        BLOCK_J=BLOCK_J,

    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_k, stride_r_j, stride_r_h,

    pid1 = tl.program_id(1)
    pid2 = tl.program_id(2)

    h = pid0 % H

    i_offset = pid1 * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = pid2 * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_j = j_offset < N

"""
TriMul “outgoing” forward operator (AlphaFold‑3 style).

The implementation follows the reference PyTorch module:
1. Layer‑norm over the last dimension (dim).
2. Linear projections → left / right tensors of shape [B,N,N,H].
3. Optional pairwise mask (broadcasted on the feature dim).
4. Gating (sigmoid) on left, right and an output gate.
5. Core contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   realized as a batched matrix‑multiply using a custom Triton kernel.
6. Layer‑norm over the hidden dimension H, multiplication by the output gate,
   and a final linear projection back to dim.

Only the O(N³) contraction (step 5) is written in Triton; the remaining
operations are performed with regular PyTorch for simplicity and to keep the
code easy to reason about. The kernel works for any batch size ≤ 2,
any sequence length up to 1024, and any hidden dimension (the tests use
128 or 384).  All tensors are assumed to reside on the CUDA device.

The final output is float32 as required.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_kernel(
    left_ptr,                  # const *float32
    right_ptr,                 # const *float32
    out_ptr,                   # *float32
    B, N, H,                   # int32 scalar sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # int32 strides for left
    stride_r_b, stride_r_k, stride_r_j, stride_r_h,   # int32 strides for right (k,j order)
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # int32 strides for out
    BLOCK_I: tl.constexpr,    # tile size in i dimension
    BLOCK_J: tl.constexpr,    # tile size in j dimension
    BLOCK_K: tl.constexpr,    # tile size in k dimension
):
    """Compute one tile of the TriMul contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,k,j,h]
    """
    pid0 = tl.program_id(0)          # combined batch * head index
    pid1 = tl.program_id(1)          # i‑tile
    pid2 = tl.program_id(2)          # j‑tile

    # decode batch index and hidden‑dim (head) index
    b = pid0 // H
    h = pid0 % H

    # i and j offsets for this tile
    i_offset = pid1 * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = pid2 * BLOCK_J + tl.arange(0, BLOCK_J)

    # bounds masks
    mask_i = i_offset < N
    mask_j = j_offset < N

    # accumulator in FP32 (left/right may be FP16)
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over K dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # pointers for the two tiles:
        #   left  : (i, k)   → shape (BLOCK_I, BLOCK_K)
        #   right : (k, j)   → shape (BLOCK_K, BLOCK_J)
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + h * stride_l_h
            + i_offset[:, None] * stride_l_i
            + k_offset[None, :] * stride_l_k
        )
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + h * stride_r_h
            + k_offset[:, None] * stride_r_k
            + j_offset[None, :] * stride_r_j
        )

        # masked loads (outside‑bounds elements are zero)
        left_tile = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        right_tile = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # matrix‑multiply the two tiles; tl.dot supports 2‑D × 2‑D → 2‑D
        acc += tl.dot(left_tile, right_tile)

    # write the result tile
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + h * stride_o_h
        + i_offset[:, None] * stride_o_i
        + j_offset[None, :] * stride_o_j
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask        : torch.Tensor, shape [B, N, N] (may be all‑ones)
        - weights     : dict of torch.Tensors containing all linear / norm parameters
        - config      : dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------------- #
    # unpack arguments
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # 1. Layer‑norm over the last dimension (dim)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------------- #
    # 2. Linear projections → left / right (shape [B,N,N,hidden_dim])
    # --------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------------- #
    # 3. Apply pairwise mask if provided (mask shape [B,N,N])
    # --------------------------------------------------------------------- #
    # mask could be omitted when config["nomask"] is True – we simply skip the
    # multiplication in that case.
    if mask is not None and not config.get("nomask", False):
        # ensure same dtype as the projected tensors
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------------- #
    # 4. Gating (sigmoid) – left, right and output gate
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Core contraction (Tricky O(N³) part) using Triton
    # --------------------------------------------------------------------- #
    # Convert to float32 for the kernel (kernel expects fp32 loads; converting
    # elsewhere would increase memory bandwidth but keeps the kernel simple).
    left = left.to(torch.float32)
    right = right.to(torch.float32)

    # right must be in (b, k, j, h) order for the matmul
    right_perm = right.permute(0, 2, 1, 3).contiguous()

    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32)

    # strides for the three tensors (as integers)
    stride_l = left.stride()   # (b,i,k,h)
    stride_r = right_perm.stride()  # (b,k,j,h) after permute
    stride_o = out.stride()    # (b,i,j,h)

    # --------------------------------------------------------------------- #
    # Decide tile sizes – 64 works well on H100 for the tested N ≤ 1024.
    # --------------------------------------------------------------------- #
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    # grid dimensions: (batch*head, i‑tiles, j‑tiles)
    grid = (
        B * H,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )

    # launch kernel
    _trimul_kernel[grid](
        left,
        right_perm,
        out,
        B,
        N,
        H,
        stride_l[0],
        stride_l[1],
        stride_l[2],
        stride_l[3],
        stride_r[0],
        stride_r[1],
        stride_r[2],
        stride_r[3],
        stride_o[0],
        stride_o[1],
        stride_o[2],
        stride_o[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=4,   # empirical choice for H100
    )

    # --------------------------------------------------------------------- #
    # 6. Post‑processing: layer‑norm over hidden dimension, output gate, final linear
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    output = F.linear(out, weights["to_out.weight"])  # shape [B,N,N,dim]

    # guarantee float32 output as required
    return output.to(torch.float32)
