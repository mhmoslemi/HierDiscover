@triton.jit
def layernorm_kernel(x_ptr, y_ptr, w_ptr, b_ptr,
                     num_rows, dim,
                     eps,
                     BLOCK_OUT: tl.constexpr, BLOCK_D: tl.constexpr):
    row_idx = tl.program_id(0) * BLOCK_OUT + tl.arange(0, BLOCK_OUT)
    
    mask = row_idx < num_rows
    # Compute base ptr for each row
    x_ptrs = x_ptr + row_idx[:, None] * dim + tl.arange(0, BLOCK_D)[None, :]
    # but we need to support dim > BLOCK_D; so we loop over dim in steps
    # We'll accumulate sum for mean and sumsq.
    sum = tl.zeros([BLOCK_OUT], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_OUT], dtype=tl.float32)
    # iterate over d positions in chunks of BLOCK_D
    for d_offset in range(0, dim, BLOCK_D):
        cur_D = tl.minimum(BLOCK_D, dim - d_offset)
        # load loads: shape (BLOCK_OUT, cur_D)
        cols = tl.arange(0, cur_D)
        x_off = row_idx[:, None] * dim + (d_offset + cols)[None, :]
        x = tl.load(x_ptr + x_off, mask=mask[:, None] & (cols[None, :] < cur_D), other=0.0)
        sum += tl.sum(x, axis=1)  # sum across features
        sum_sq += tl.sum(x * x, axis=1)
    
    # compute mean, var
    mean = sum / dim
    var = sum_sq / dim - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)
    
    # Now compute normalized and write output:
    for d_offset in range(0, dim, BLOCK_D):
        cur_D = tl.minimum(BLOCK_D, dim - d_offset)
        cols = tl.arange(0, cur_D)
        x_off = row_idx[:, None] * dim + (d_offset + cols)[None, :]

    import triton
    import triton.language as tl

    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Ensure contiguous
    x = input_tensor.contiguous()
    mask = mask_tensor.contiguous()

    bs, N, _, _ = x.shape

    # --- layernorm over dim (input) ---
    outer = bs * N * N
    x_norm = torch.empty_like(x)
    BLOCK_ROW = 128
    BLOCK_D = 32
    # compile launch grid
    grid = lambda meta: (triton.cdiv(outer, meta['BLOCK_ROW']),)
    layernorm_kernel[grid](
        x, x_norm, weights["norm.weight"], weights["norm.bias"],
        outer, dim, eps,
        BLOCK_ROW=BLOCK_ROW, BLOCK_D=BLOCK_D, DIM_MAX=dim
    )
    # Ensure result contiguity
    x_norm = x_norm

    # Flatten for linear
    x_flat = x_norm.view(-1, dim)  # (outer, dim)

    # Linear projections
    left = torch.nn.functional.linear(x_flat, weights["left_proj.weight"]).view(bs, N, N, hidden_dim)
    right = torch.nn.functional.linear(x_flat, weights["right_proj.weight"]).view(bs, N, N, hidden_dim)

    # Apply mask if present and not nomask
    if not config.get("nomask", False):
        # mask shape (bs, N, N)
        mask_exp = mask.unsqueeze(-1).to(x.dtype)  # shape (bs,N,N,1)
        left = left * mask_exp
        right = right * mask_exp
    else:
        # no mask, skip multiplication
        pass

    # Gates
    left_gate = torch.nn.functional.linear(x_flat, weights["left_gate.weight"]).view(bs, N, N, hidden_dim).sigmoid()
    right_gate = torch.nn.functional.linear(x_flat, weights["right_gate.weight"]).view(bs, N, N, hidden_dim).sigmoid()
    out_gate = torch.nn.functional.linear(x_flat, weights["out_gate.weight"]).view(bs, N, N, hidden_dim).sigmoid()

    # Apply gate to left/right
    left = left * left_gate
    right = right * right_gate

    # Compute pairwise interaction via batched matmul
    # Permute to (bs, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2)   # b, h, i, k
    right_perm = right.permute(0, 3, 2, 1) # b, h, k, j
    out = torch.matmul(left_perm, right_perm)  # shape (b, h, i, j)
    out = out.permute(0, 2, 3, 1)  # b, i, j, h

    out_gated = out_norm * out_gate
    out_flat = out_gated.view(-1, hidden_dim)
    output = torch.nn.functional.linear(out_flat, weights["to_out.weight"]).view(bs, N, N, dim)

@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    num_rows, dim,
    eps,
    BLOCK_ROW: tl.constexpr,
    BLOCK_D: tl.constexpr,
    DIM_MAX: tl.constexpr
):
    row = tl.program_id(0) * BLOCK_ROW + tl.arange(0, BLOCK_ROW)
    row_mask = row < num_rows

    # Accumulators
    sum_val = tl.zeros([BLOCK_ROW], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_ROW], dtype=tl.float32)

    col = tl.arange(0, BLOCK_D)
    for d_offset in range(0, DIM_MAX, BLOCK_D):
        cur_col = d_offset + col
        col_mask = cur_col < dim

        offset = row[:, None] * dim + cur_col[None, :]
        x = tl.load(x_ptr + offset, mask=row_mask[:, None] & col_mask[None, :], other=0.0)
        x_f = x.to(tl.float32)
        sum_val += tl.sum(x_f, axis=1)
        sum_sq += tl.sum(x_f * x_f, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # Apply affine transform
    for d_offset in range(0, DIM_MAX, BLOCK_D):
        cur_col = d_offset + col
        col_mask = cur_col < dim

        offset = row[:, None] * dim + cur_col[None, :]
        x = tl.load(x_ptr + offset, mask=row_mask[:, None] & col_mask[None, :], other=0.0)
        x_f = x.to(tl.float32)
        normed = (x_f - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + cur_col, mask=col_mask)
        b = tl.load(bias_ptr + cur_col, mask=col_mask)

        out = normed * w + b
        tl.store(y_ptr + offset, out.to(x.dtype), mask=row_mask[:, None] & col_mask[None, :])

"""
TriMul (outgoing) implementation with Triton‑accelerated LayerNorm.

Algorithm
---------
1. Layer‑normalize the input tensor X ∈ ℝ^{B·N·N·C} over the last dimension C
   (the same as nn.LayerNorm).  The reduction (mean/variance) and the affine
   transformation are fused in a Triton kernel.
2. Apply four linear projections (left, right, left‑gate, right‑gate, out‑gate)
   on the normalized data using torch.nn.functional.linear.
3. Mask the left/right tensors if a mask is supplied.
4. Apply the sigmoid gates to the left/right tensors.
5. Compute the core “TriMul” interaction:
       out_{b,i,j,d} = Σ_{k} left_{b,i,k,d} * right_{b,j,k,d}
   This is equivalent to a batched matrix multiplication for each hidden
   channel d: (left[...,i,k,d] @ right[...,j,k,d].T).  Implemented with
   torch.matmul on tensors of shape (B, H, N, N).
6. Layer‑normalize the result over the hidden dimension H using a second
   Triton kernel.
7. Multiply by the out‑gate, then project back to the original dimensionality
   with a final linear layer.

Only the LayerNorm steps are written in Triton; the remaining dense linear
operations and the batched matmul rely on highly‑optimized PyTorch kernels.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _layernorm_kernel(
    x_ptr,                # input   [rows, dim] (flattened)
    y_ptr,                # output  [rows, dim]
    w_ptr,                # weight  [dim]
    b_ptr,                # bias    [dim]
    rows,                 # number of rows (B*N*N)
    dim,                  # actual feature dimension
    eps,                  # epsilon for stability
    BLOCK_ROW: tl.constexpr,
    BLOCK_D: tl.constexpr,
    DIM_MAX: tl.constexpr,   # compile‑time upper bound for dim (e.g. 384)
):
    """
    Per‑row LayerNorm with a fused affine transform.
    The kernel processes BLOCK_ROW rows per program and iterates over the
    feature dimension in steps of BLOCK_D.  DIM_MAX must be >= dim and is used
    for static loop unrolling; excess columns are masked out.
    """
    # ----------------------------------------------------------------------
    # 1) row index & masks
    # ----------------------------------------------------------------------
    row_idx = tl.program_id(0) * BLOCK_ROW + tl.arange(0, BLOCK_ROW)
    row_mask = row_idx < rows

    # ----------------------------------------------------------------------
    # 2) First pass: compute sum and sum of squares for mean/variance
    # ----------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_ROW], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_ROW], dtype=tl.float32)

    col = tl.arange(0, BLOCK_D)               # static column block
    for d_off in range(0, DIM_MAX, BLOCK_D):
        col_idx = d_off + col                 # absolute column indices
        col_mask = col_idx < dim               # ignore columns beyond dim

        # linear offset for loading: row * dim + col
        offset = row_idx[:, None] * dim + col_idx[None, :]
        x = tl.load(
            x_ptr + offset,
            mask=row_mask[:, None] & col_mask[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)
        sum_val += tl.sum(x_f, axis=1)
        sum_sq  += tl.sum(x_f * x_f, axis=1)

    # ----------------------------------------------------------------------
    # 3) Compute mean, variance, rstd
    # ----------------------------------------------------------------------
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)

    # ----------------------------------------------------------------------
    # 4) Second pass: normalize and apply weight/bias
    # ----------------------------------------------------------------------
    for d_off in range(0, DIM_MAX, BLOCK_D):
        col_idx = d_off + col
        col_mask = col_idx < dim

        offset = row_idx[:, None] * dim + col_idx[None, :]
        x = tl.load(
            x_ptr + offset,
            mask=row_mask[:, None] & col_mask[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)
        normed = (x_f - mean[:, None]) * rstd[:, None]

        # weight and bias are 1‑D, broadcast across rows
        w = tl.load(w_ptr + col_idx, mask=col_mask)
        b = tl.load(b_ptr + col_idx, mask=col_mask)

        out = normed * w[None, :] + b[None, :]
        tl.store(
            y_ptr + offset,
            out.to(x.dtype),
            mask=row_mask[:, None] & col_mask[None, :],
        )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seqlen, seqlen, dim] (float32).
    """
    # ----------------------------------------------------------------------
    # unpack inputs
    # ----------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    eps = 1e-5

    # ensure contiguous layout
    x = input_tensor.contiguous()
    mask = mask_tensor.contiguous()

    bs, seqlen, _, _ = x.shape
    outer = bs * seqlen * seqlen                     # number of rows for LN

    # ----------------------------------------------------------------------
    # 1) LayerNorm over the input feature dimension (dim)
    # ----------------------------------------------------------------------
    x_norm = torch.empty_like(x)

    BLOCK_ROW = 128     # rows per program
    BLOCK_D   = 32      # column block size (must divide DIM_MAX)
    grid = lambda meta: (triton.cdiv(outer, meta["BLOCK_ROW"]),)

    _layernorm_kernel[grid](
        x,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        outer,
        dim,
        eps,
        BLOCK_ROW=BLOCK_ROW,
        BLOCK_D=BLOCK_D,
        DIM_MAX=dim,                 # compile‑time bound = actual dim (≤384)
    )

    # ----------------------------------------------------------------------
    # 2) Linear projections & gates (all using torch.nn.functional.linear)
    # ----------------------------------------------------------------------
    x_flat = x_norm.view(-1, dim)                 # (outer, dim)

    left = torch.nn.functional.linear(
        x_flat, weights["left_proj.weight"]
    ).view(bs, seqlen, seqlen, hidden_dim)

    right = torch.nn.functional.linear(
        x_flat, weights["right_proj.weight"]
    ).view(bs, seqlen, seqlen, hidden_dim)

    # ----------------------------------------------------------------------
    # 3) Optional mask (broadcasted on the feature axis)
    # ----------------------------------------------------------------------
    if not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).to(x.dtype)                # (B,N,N,1)
        left  = left * mask_exp
        right = right * mask_exp

    # ----------------------------------------------------------------------
    # 4) Gates
    # ----------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x_flat, weights["left_gate.weight"]
    ).view(bs, seqlen, seqlen, hidden_dim).sigmoid()

    right_gate = torch.nn.functional.linear(
        x_flat, weights["right_gate.weight"]
    ).view(bs, seqlen, seqlen, hidden_dim).sigmoid()

    out_gate = torch.nn.functional.linear(
        x_flat, weights["out_gate.weight"]
    ).view(bs, seqlen, seqlen, hidden_dim).sigmoid()

    # apply multiplicative gates
    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 5) Core TriMul interaction via batched matmul
    # ----------------------------------------------------------------------
    # reshape to (B, H, N, N) and multiply left @ right^T per hidden channel
    left_perm = left.permute(0, 3, 1, 2)   # (B, H, N, K)
    right_perm = right.permute(0, 3, 2, 1) # (B, H, K, N)
    out = torch.matmul(left_perm, right_perm)          # (B, H, N, N)
    out = out.permute(0, 2, 3, 1)                      # (B, N, N, H)

    # ----------------------------------------------------------------------
    # 6) LayerNorm over hidden dimension H (to_out_norm)
    # ----------------------------------------------------------------------
    outer_out = bs * seqlen * seqlen
    out_norm = torch.empty_like(out)

    grid2 = lambda meta: (triton.cdiv(outer_out, meta["BLOCK_ROW"]),)
    _layernorm_kernel[grid2](
        out,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        outer_out,
        hidden_dim,
        eps,
        BLOCK_ROW=BLOCK_ROW,
        BLOCK_D=BLOCK_D,
        DIM_MAX=hidden_dim,          # compile‑time bound = hidden_dim (≤128)
    )

    # ----------------------------------------------------------------------
    # 7) Final gating, linear projection back to dim
    # ----------------------------------------------------------------------
    out = out_norm * out_gate                     # (B,N,N,H)
    out_flat = out.view(-1, hidden_dim)           # (outer, H)

    output = torch.nn.functional.linear(
        out_flat, weights["to_out.weight"]
    ).view(bs, seqlen, seqlen, dim)               # (B,N,N,dim)

    return output
