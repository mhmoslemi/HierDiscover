def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    total, M, N,
    stride_left_m, stride_left_k, stride_left_n,
    stride_right_m, stride_right_n, stride_right_k,
    stride_out_m, stride_out_n, stride_out_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_mat = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute pointer to start of matrix for this mat
    left_matrix_ptr = left_ptr + pid_mat * stride_left_m
    right_matrix_ptr = right_ptr + pid_mat * stride_right_m
    out_matrix_ptr = out_ptr + pid_mat * stride_out_m

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(M, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # load A tile: shape (BLOCK_M, BLOCK_K)

def custom_kernel(data):
    """TriMul outgoing kernel optimized with Triton for the sum over k.

    Computes:
        out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]

    where left/right are linear projections of the normalized input
    with gating and optional mask.
    """
    input_tensor, mask, weights, config = data
    
    device = input_tensor.device
    dtype = input_tensor.dtype
    
    # Gather config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    
    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections (no bias)
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # optional mask
    if not config.get("nomask", False):
        if mask is None:
            raise ValueError("Mask required but not provided")
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    # Gates (sigmoid)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left_proj = left_proj * left_gate
    right_proj = right_proj * right_gate

    # Convert to fp16 for kernel (saving bandwidth)
    left_fp16 = left_proj.to(torch.float16).contiguous()
    right_fp16 = right_proj.to(torch.float16).contiguous()

    B, N, _, _ = left_fp16.shape
    # reshape to (B*hidden_dim, N, N)
    left_flat = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # allocate output
    out_flat = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)

    # strides
    stride_left_mat = left_flat.stride(0)
    stride_left_row = left_flat.stride(1)
    stride_left_col = left_flat.stride(2)
    stride_right_mat = right_flat.stride(0)
    stride_right_row = right_flat.stride(1)
    stride_right_col = right_flat.stride(2)
    stride_out_mat = out_flat.stride(0)
    stride_out_row = out_flat.stride(1)
    stride_out_col = out_flat.stride(2)

    # Block sizes - tuned for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * hidden_dim,
        N,
        stride_left_mat, stride_left_row, stride_left_col,
        stride_right_mat, stride_right_row, stride_right_col,
        stride_out_mat, stride_out_row, stride_out_col,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # LayerNorm and output projection
    out_norm = torch.nn.functional.layer_norm(out,
                                              (hidden_dim,),
                                              weight=weights["to_out_norm.weight"],
                                              bias=weights["to_out_norm.bias"],
                                              eps=eps)
    out_norm = out_norm * out_gate

    # final linear to dim
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])
    return output

"""
TriMul (outgoing) forward pass implemented with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` over the last dimension.
2. Linear projections (no bias) → `left` and `right`.
3. Optional pairwise mask is applied (broadcast on the last dim).
4. Compute three sigmoid gates from the same normalized input:
   `left_gate`, `right_gate`, `out_gate`.
5. Apply gates:   left  = left  * left_gate
                right = right * right_gate
6. Compute the core “tri‑multiply”:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batch of `hidden_dim` matrix multiplications
   (N×N) @ (N×N)^T for each head.  The heavy part is performed
   by a Triton kernel that processes every head (`total = B*hidden_dim`)
   as an independent matrix‑multiply.  The kernel works on fp16
   inputs and accumulates in fp32, storing the result in fp32.
7. Layer‑norm the result over the head dimension, apply the
   `out_gate`, and finally project back to `dim` with a linear layer.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr,                 # fp16   [total, N, N]
    right_ptr,                # fp16   [total, N, N]
    out_ptr,                  # fp32   [total, N, N]
    total,                    # total = B * hidden_dim
    N,                        # sequence length
    stride_l_mat, stride_l_row, stride_l_col,   # strides of left
    stride_r_mat, stride_r_row, stride_r_col,   # strides of right
    stride_o_mat, stride_o_row, stride_o_col,   # strides of out
    BLOCK_M: tl.constexpr,   # tile rows   (i dimension)
    BLOCK_N: tl.constexpr,   # tile cols   (j dimension)
    BLOCK_K: tl.constexpr    # reduction tile (k dimension)
):
    # program IDs
    pid_mat = tl.program_id(0)          # which head (B*hidden_dim)
    pid_m   = tl.program_id(1)          # block row index  (i)
    pid_n   = tl.program_id(2)          # block col index  (j)

    # offsets within the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this matrix
    left_base  = left_ptr  + pid_mat * stride_l_mat
    right_base = right_ptr + pid_mat * stride_r_mat
    out_base   = out_ptr   + pid_mat * stride_o_mat

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        start_k = k * BLOCK_K
        offs_k = start_k + tl.arange(0, BLOCK_K)

        # masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < N) & (offs_k[None, :] < N)
        mask_b = (offs_k[:, None] < N) & (offs_n[None, :] < N)

        # A tile : left[i,k]  (shape BLOCK_M x BLOCK_K)
        a_ptrs = left_base + (offs_m[:, None] * stride_l_row) + (offs_k[None, :] * stride_l_col)
        a = tl.load(a_ptrs, mask=mask_a, other=0.0).to(tl.float32)

        # B tile transposed : right_T[k,j] = right[j,k]
        # shape (BLOCK_K x BLOCK_N)
        b_ptrs = right_base + (offs_n[None, :] * stride_r_row) + (offs_k[:, None] * stride_r_col)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0).to(tl.float32)

        # matrix‑multiply‑accumulate
        acc += tl.dot(a, b)   # (BLOCK_M,BLOCK_K) @ (BLOCK_K,BLOCK_N)

    # store only the valid region
    mask_o = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    out_ptrs = out_base + (offs_m[:, None] * stride_o_row) + (offs_n[None, :] * stride_o_col)
    tl.store(out_ptrs, acc, mask=mask_o)


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, seq_len, seq_len, dim]
        - mask         : torch.Tensor[bs, seq_len, seq_len]  (may be None)
        - weights      : dict of torch.Tensors (see skeleton)
        - config       : dict containing at least
                         {"dim": int, "hidden_dim": int, "nomask": bool}
    Returns
    -------
    torch.Tensor[bs, seq_len, seq_len, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------------------------------
    # 1. Layer‑norm over the last dimension
    # -------------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # -------------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------------------------------
    # 3. Optional mask (broadcast on the last dim)
    # -------------------------------------------------------------------------
    if not nomask:
        if mask is None:
            raise ValueError("mask is required when `nomask` is False")
        mask_ = mask.unsqueeze(-1).to(x.dtype)        # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # -------------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # -------------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # -------------------------------------------------------------------------
    # 5. Apply gates
    # -------------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------------
    # 6. Core Tri‑multiply: batched GEMM across the hidden dimension
    #    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # -------------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Cast to fp16 to reduce memory traffic, keep a copy for the kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # reshape to [total_heads, N, N] where total_heads = B * hidden_dim
    left_flat = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_flat = torch.empty(
        (B * hidden_dim, N, N), dtype=torch.float32, device=device
    )

    # strides in element count (not bytes)
    l_mat, l_row, l_col = left_flat.stride()
    r_mat, r_row, r_col = right_flat.stride()
    o_mat, o_row, o_col = out_flat.stride()

    # Tile sizes – tuned for H100 (shared‑memory friendly)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                 # total matrices (head dimension)
        (N + BLOCK_M - 1) // BLOCK_M,   # row tiles per matrix
        (N + BLOCK_N - 1) // BLOCK_N,   # col tiles per matrix
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * hidden_dim,
        N,
        l_mat, l_row, l_col,
        r_mat, r_row, r_col,
        o_mat, o_row, o_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Re‑assemble to shape [B, N, N, hidden_dim]
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, hidden_dim]

    # -------------------------------------------------------------------------
    # 7. Output LayerNorm + gating + final linear projection
    # -------------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return output
