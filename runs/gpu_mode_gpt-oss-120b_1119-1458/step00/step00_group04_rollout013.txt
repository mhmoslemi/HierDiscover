Now final linear: to_out_weight -> shape [dim, hidden_dim] w/out bias.

"""
custom_kernel for the “outgoing” Triangle Multiplicative Update (TriMul).

Algorithm
---------
1️⃣ Layer‑norm the input tensor `x` (shape [B,N,N,D]).
2️⃣ Fuse projection, gating and optional masking for the left and right
   pairwise tensors using a Triton kernel:
        left  = (x @ left_proj.T) * mask * sigmoid(x @ left_gate.T)
        right = (x @ right_proj.T) * mask * sigmoid(x @ right_gate.T)
   The kernel computes a dense mat‑vec for every (b,i,j) position and every
   hidden channel `h` (hidden_dim = H).  It loops over the feature dimension
   `D` in blocks, accumulates the projection and gate values, applies the
   sigmoid to the gate and finally writes the gated, masked projection.
3️⃣ Compute the “triangle‑multiplicative” product:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply for each hidden channel:
        left_flat  ∈ (B·H,N,N)   = left.permute(0,3,1,2).reshape(B*H,N,N)
        right_flat ∈ (B·H,N,N)   = right.permute(0,3,1,2).reshape(B*H,N,N)
        out_flat   = torch.bmm(left_flat, right_flat.transpose(1,2))
        out        = out_flat.view(B,H,N,N).permute(0,2,3,1)   # [B,N,N,H]
4️⃣ Layer‑norm over the hidden dimension, multiply by a second gate
   (computed from the same normalized input) and project back to the
   original channel dimension:
        out = LayerNorm(out, weight=to_out_norm_weight, bias=to_out_norm_bias)
        out_gate = sigmoid( linear(x_norm, out_gate_weight) )
        out = out * out_gate
        out = linear(out, to_out_weight)                     # → [B,N,N,D]

Only the projection‑gate‑mask step is implemented in Triton; all other
operations use PyTorch’s highly‑optimised kernels.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: fused projection + gate + mask for one side (left/right)
# ----------------------------------------------------------------------
@triton.jit
def _fused_proj_gate_kernel(
    x_ptr,                # *float32   [B,N,N,D]   (normalized input)
    w_proj_ptr,           # *float32   [H,D]       projection weight
    w_gate_ptr,           # *float32   [H,D]       gate weight
    mask_ptr,             # *float32   [B,N,N]    mask (0/1)
    out_ptr,              # *float32   [B,N,N,H]   output

    B, N, D, H,           # scalar problem sizes

    stride_x_b, stride_x_i, stride_x_j, stride_x_d,
    stride_w_proj_h, stride_w_proj_d,
    stride_w_gate_h, stride_w_gate_d,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    BLOCK_H: tl.constexpr,   # hidden‑dim tile size (e.g. 32)
    BLOCK_D: tl.constexpr,   # input‑feature tile size (e.g. 64)
):
    """Compute:
       out = (x @ w_proj.T) * mask * sigmoid(x @ w_gate.T)
    for a single (b,i,j) position across a tile of hidden channels.
    """
    pid = tl.program_id(0)               # index of (b,i,j)
    hid = tl.program_id(1)               # tile index over hidden dimension

    # ------------------------------------------------------------------
    # Decode (b,i,j) from pid
    # ------------------------------------------------------------------
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # ------------------------------------------------------------------
    # Offsets for the hidden dimension tile
    # ------------------------------------------------------------------
    h_offset = hid * BLOCK_H + tl.arange(0, BLOCK_H)
    h_mask = h_offset < H

    # ------------------------------------------------------------------
    # Load the scalar mask value for this (b,i,j)
    # ------------------------------------------------------------------
    mask_val = tl.load(mask_ptr + b * stride_mask_b
                                 + i * stride_mask_i
                                 + j * stride_mask_j,
                       mask=tl.full([1], True, dtype=tl.bool))
    # mask_val is a 1‑element tensor → broadcast later

    # ------------------------------------------------------------------
    # Accumulators (float32 for numerical stability)
    # ------------------------------------------------------------------
    acc_proj = tl.zeros([BLOCK_H], dtype=tl.float32)
    acc_gate = tl.zeros([BLOCK_H], dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the input feature dimension D in blocks
    # ------------------------------------------------------------------
    for d_start in range(0, D, BLOCK_D):
        d_offset = d_start + tl.arange(0, BLOCK_D)
        d_mask = d_offset < D

        # Load input slice x[b,i,j,d_start:d_start+BLOCK_D]
        x = tl.load(
            x_ptr + b * stride_x_b
                     + i * stride_x_i
                     + j * stride_x_j
                     + d_offset * stride_x_d,
            mask=d_mask,
            other=0.0
        )  # shape: [BLOCK_D]

        # Load projection weights w_proj[h, d]
        w_proj = tl.load(
            w_proj_ptr + h_offset[:, None] * stride_w_proj_h
                         + d_offset[None, :] * stride_w_proj_d,
            mask=(h_mask[:, None] & d_mask[None, :]),
            other=0.0
        )  # shape: [BLOCK_H, BLOCK_D]

        # Load gate weights w_gate[h, d]
        w_gate = tl.load(
            w_gate_ptr + h_offset[:, None] * stride_w_gate_h
                         + d_offset[None, :] * stride_w_gate_d,
            mask=(h_mask[:, None] & d_mask[None, :]),
            other=0.0
        )  # shape: [BLOCK_H, BLOCK_D]

        # Partial dot products: (H,D) · (D,) → (H,)
        acc_proj += tl.dot(w_proj, x)   # proj contribution
        acc_gate += tl.dot(w_gate, x)   # gate contribution

    # ------------------------------------------------------------------
    # Apply sigmoid to gate and combine with projection
    # ------------------------------------------------------------------
    gate = 1.0 / (1.0 + tl.exp(-acc_gate))
    out_val = acc_proj * gate * mask_val

    # ------------------------------------------------------------------
    # Store result
    # ------------------------------------------------------------------
    tl.store(
        out_ptr + b * stride_out_b
                 + i * stride_out_i
                 + j * stride_out_j
                 + h_offset * stride_out_h,
        out_val,
        mask=h_mask
    )


def _fused_proj_gate(
    x: torch.Tensor,
    w_proj: torch.Tensor,
    w_gate: torch.Tensor,
    mask: torch.Tensor,
    hidden_dim: int,
    BLOCK_H: int = 32,
    BLOCK_D: int = 64,
) -> torch.Tensor:
    """
    Wrapper for the Triton kernel.
    Returns a tensor of shape [B, N, N, hidden_dim].
    """
    assert x.is_contiguous()
    B, N, _, D = x.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=x.dtype, device=x.device)

    # strides (in elements) for each tensor
    sx = x.stride()          # (B_stride, N_stride_i, N_stride_j, D_stride)
    swp = w_proj.stride()   # (H_stride, D_stride)
    swg = w_gate.stride()
    sm = mask.stride()       # (B_stride, N_stride_i, N_stride_j)
    so = out.stride()

    total_points = B * N * N
    grid = (total_points, triton.cdiv(hidden_dim, BLOCK_H))

    _fused_proj_gate_kernel[grid](
        x,
        w_proj,
        w_gate,
        mask,
        out,
        B, N, D, hidden_dim,
        sx[0], sx[1], sx[2], sx[3],
        swp[0], swp[1],
        swg[0], swg[1],
        sm[0], sm[1], sm[2],
        so[0], so[1], so[2], so[3],
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=4,          # good default for fp32 on H100
    )
    return out


# ----------------------------------------------------------------------
# Entry point as required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator with a fused Triton kernel
    for the left/right projection‑gate‑mask steps.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the last dimension (using the provided weights)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                     # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Prepare mask (if no mask, use all‑ones)
    # ------------------------------------------------------------------
    if config.get("nomask", False) or mask_tensor is None:
        mask = torch.ones(
            (x.shape[0], x.shape[1], x.shape[2]), dtype=x.dtype, device=x.device
        )
    else:
        mask = mask_tensor.to(dtype=x.dtype, device=x.device)

    # make mask 3‑D (B,N,N) – Triton kernel expects a scalar per (b,i,j)
    # (no explicit channel dimension needed)
    mask = mask.contiguous()

    # ------------------------------------------------------------------
    # 3) Left and Right branches (fused projection + gating + mask)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]          # [hidden_dim, dim]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]

    # Ensure all weight tensors are contiguous in memory
    left_proj_w = left_proj_w.contiguous()
    right_proj_w = right_proj_w.contiguous()
    left_gate_w = left_gate_w.contiguous()
    right_gate_w = right_gate_w.contiguous()

    left = _fused_proj_gate(x, left_proj_w, left_gate_w, mask, hidden_dim)
    right = _fused_proj_gate(x, right_proj_w, right_gate_w, mask, hidden_dim)

    # ------------------------------------------------------------------
    # 4) Triangle‑multiplicative product via batched GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for batched matmul
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # out_flat [B*H, N, N] = left_flat @ right_flat^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5) Output post‑processing (layer‑norm, out‑gate, final linear)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                     # [B,N,N,H]

    out_gate_w = weights["out_gate.weight"]
    out_gate = torch.nn.functional.linear(x, out_gate_w)          # [B,N,N,H]
    out_gate = torch.sigmoid(out_gate)

    out = out * out_gate

    to_out_w = weights["to_out.weight"]     # [dim, hidden_dim]
    out = torch.nn.functional.linear(out, to_out_w)   # → [B,N,N,dim]

    return out
