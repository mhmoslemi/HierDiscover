def custom_kernel(data):
    input_tensor, mask, weights, config = data

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # mask handling
    if mask is None or mask.numel() == 0:
        mask = torch.ones((bs, seq_len, seq_len), dtype=input_tensor.dtype, device=input_tensor.device)

    # cast mask to half for kernel
    mask_fp16 = mask.to(torch.float16)

    # LayerNorm on input (x_norm)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # Cast to half for fused kernel
    left_proj_h = left_proj.to(torch.float16)
    right_proj_h = right_proj.to(torch.float16)
    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    # Allocate fused tensors
    left_fused = torch.empty_like(left_proj_h)
    right_fused = torch.empty_like(right_proj_h)

    # Launch Triton fuse kernel
    import triton
    import triton.language as tl

    @triton.jit
    def fuse_proj_gate_kernel(
        lp_ptr, rg_ptr, lg_ptr, rg_gate_ptr,
        mask_ptr,
        lo_ptr, ro_ptr,
        B, N, H,
        stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_h,

        stride_lo_b, stride_lo_i, stride_lo_j, stride_lo_h,
        stride_ro_b, stride_ro_i, stride_ro_j, stride_ro_h,
        BLOCK: tl.constexpr
    ):
        pid = tl.program_id(0)
        total = B * N * N
        if pid >= total:
            return

        # Compute b,i,j
        b = pid // (N * N)
        ij = pid % (N * N)
        i = ij // N
        j = ij % N

        rg_ptr = right_gate_h.data_ptr() + h_start * right_gate_h.element_size()
        lo_ptr = left_fused.data_ptr() + h_start * left_fused.element_size()
        ro_ptr = right_fused.data_ptr() + h_start * right_fused.element_size()

        # Launch

            bs, seq_len, hidden_dim,
            lp_str[0], lp_str[1], lp_str[2], lp_str[3],
            lg_str[0], lg_str[1], lg_str[2], lg_str[3],
            rp_str[0], rp_str[1], rp_str[2], rp_str[3],

            ro_str[0], ro_str[1], ro_str[2], ro_str[3],

    torch.cuda.synchronize()

    # Cast to half is already done
    left_mat = left_fused.permute(0, 3, 1, 2).reshape(bs * hidden_dim, seq_len, seq_len)
    right_mat = right_fused.permute(0, 3, 2, 1).reshape(bs * hidden_dim, seq_len, seq_len)
    out_mat = torch.bmm(left_mat, right_mat)  # result half
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).to(torch.float32)

    # Output LayerNorm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)  # no bias

    return out

import torch
import triton
import triton.language as tl

def custom_kernel(data):
    """
    Triton-accelerated TriMul (outgoing) forward pass.
    Steps:
    1. Layer‑norm the input.
    2. Linear projections for left/right and gating (no bias).
    3. Fuse mask, projection and gate using a small Triton kernel.
    4. Compute the pairwise “tri‑multiplicative” contraction with
       a batched matrix multiplication (fp16, tensor‑core).
    5. Layer‑norm, gating and final linear projection back to the original
       channel dimension.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (float32)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gating (float32)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Prepare mask (float16) – if “nomask” we use a tensor of ones.
    # ------------------------------------------------------------------
    if mask is None or mask.numel() == 0 or config.get("nomask", False):
        mask_fp16 = torch.ones((bs, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16)

    # ------------------------------------------------------------------
    # 5. Cast projections & gates to fp16 – the massive contraction is
    #    executed with tensor‑core friendly fp16.
    # ------------------------------------------------------------------
    left_proj_h = left_proj.to(torch.float16)
    right_proj_h = right_proj.to(torch.float16)
    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    # Allocate fused tensors that will hold (proj * gate * mask)
    left_fused = torch.empty_like(left_proj_h)
    right_fused = torch.empty_like(right_proj_h)

    # ------------------------------------------------------------------
    # 6. Triton kernel – elementwise fusion of (proj * gate * mask)
    # ------------------------------------------------------------------
    @triton.jit
    def fuse_proj_gate_kernel(
            lp_ptr, rg_ptr, lg_ptr, rg_gate_ptr,
            mask_ptr,
            lo_ptr, ro_ptr,
            B, N,
            stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_h,
            stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
            stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_h,
            stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
            stride_mask_b, stride_mask_i, stride_mask_j,
            stride_lo_b, stride_lo_i, stride_lo_j, stride_lo_h,
            stride_ro_b, stride_ro_i, stride_ro_j, stride_ro_h,
            BLOCK: tl.constexpr):
        """
        For every (b,i,j) element:
            left_out  = left_proj[b,i,j,:]  * left_gate[b,i,j,:]  * mask[b,i,j]
            right_out = right_proj[b,i,j,:] * right_gate[b,i,j,:] * mask[b,i,j]
        """
        pid = tl.program_id(0)
        total = B * N * N
        if pid >= total:
            return

        # decode a flat index into (b,i,j)
        b = pid // (N * N)
        tmp = pid % (N * N)
        i = tmp // N
        j = tmp % N

        offs_h = tl.arange(0, BLOCK)

        # mask scalar (broadcasted over hidden)
        mask = tl.load(mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j)

        # left side
        lp = tl.load(lp_ptr + b * stride_lp_b + i * stride_lp_i + j * stride_lp_j
                     + offs_h * stride_lp_h)
        lg = tl.load(lg_ptr + b * stride_lg_b + i * stride_lg_i + j * stride_lg_j
                     + offs_h * stride_lg_h)
        left = lp * lg * mask
        tl.store(lo_ptr + b * stride_lo_b + i * stride_lo_i + j * stride_lo_j
                 + offs_h * stride_lo_h, left)

        # right side
        rp = tl.load(rg_ptr + b * stride_rp_b + i * stride_rp_i + j * stride_rp_j
                     + offs_h * stride_rp_h)
        rg = tl.load(rg_gate_ptr + b * stride_rg_b + i * stride_rg_i + j * stride_rg_j
                     + offs_h * stride_rg_h)
        right = rp * rg * mask
        tl.store(ro_ptr + b * stride_ro_b + i * stride_ro_i + j * stride_ro_j
                 + offs_h * stride_ro_h, right)

    # ------------------------------------------------------------------
    # Launch the fusion kernel in blocks of 128 hidden features.
    # ------------------------------------------------------------------
    BLOCK_SIZE = 128
    total_elements = bs * N * N
    grid = (total_elements,)

    # Strides for all tensors (they are contiguous)
    lp_str = left_proj_h.stride()
    lg_str = left_gate_h.stride()
    rp_str = right_proj_h.stride()
    rg_str = right_gate_h.stride()
    lo_str = left_fused.stride()
    ro_str = right_fused.stride()
    mask_str = mask_fp16.stride()

    for h_start in range(0, hidden_dim, BLOCK_SIZE):
        cur_block = min(BLOCK_SIZE, hidden_dim - h_start)

        # Slice the hidden dimension – this creates a view whose leading
        # hidden‑size is exactly cur_block.
        lp_slice = left_proj_h[..., h_start:h_start + cur_block]
        lg_slice = left_gate_h[..., h_start:h_start + cur_block]
        rp_slice = right_proj_h[..., h_start:h_start + cur_block]
        rg_slice = right_gate_h[..., h_start:h_start + cur_block]
        lo_slice = left_fused[..., h_start:h_start + cur_block]
        ro_slice = right_fused[..., h_start:h_start + cur_block]

        fuse_proj_gate_kernel[grid](
            lp_slice, rp_slice, lg_slice, rg_slice,
            mask_fp16,
            lo_slice, ro_slice,
            bs, N,
            lp_str[0], lp_str[1], lp_str[2], lp_str[3],
            lg_str[0], lg_str[1], lg_str[2], lg_str[3],
            rp_str[0], rp_str[1], rp_str[2], rp_str[3],
            rg_str[0], rg_str[1], rg_str[2], rg_str[3],
            mask_str[0], mask_str[1], mask_str[2],
            lo_str[0], lo_str[1], lo_str[2], lo_str[3],
            ro_str[0], ro_str[1], ro_str[2], ro_str[3],
            BLOCK=cur_block,
            num_warps=4
        )

    # 7. Tri‑multiplicative contraction via batched matrix multiply.
    #    left  : (b,i,k) , right : (b,j,k) → out(b,i,j)
    # ------------------------------------------------------------------

    left_mat = left_fused.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)   # (b,h,i,k)

    # reshape back to [bs, N, N, hidden_dim] and convert to fp32 for the
    # following LayerNorm and final projection.
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, gating and final linear projection.
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

import torch
import triton
import triton.language as tl


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the “outgoing” TriMul operator.

    Pipeline
    --------
    1. Layer‑norm the [B, N, N, C] input.
    2. Linear projections (left/right) and three gating projections (left, right,
       output) – all bias‑free – performed in FP32.
    3. Fuse mask, projection and gating with a tiny Triton kernel.  The fused
       tensors are stored in FP16 to feed a tensor‑core friendly batched GEMM.
    4. Compute the core contraction
           out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
       as a batched matrix‑multiply on the (B·H) matrices (H = hidden_dim).
    5. Convert back to FP32, apply a second LayerNorm, multiply by the output
       gate and finally project back to the original channel dimension.

    This implementation respects the required signature and uses Triton for a
    non‑trivial element‑wise fusion (step 3), while delegating the O(N³) part
    to a highly optimized torch.bmm.
    """

    # ----------------------------------------------------------------------
    # unpack inputs
    # ----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ----------------------------------------------------------------------
    # 1. Input LayerNorm (same eps as nn.LayerNorm)
    # ----------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ----------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ----------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ----------------------------------------------------------------------
    # 3. Gating projections + sigmoid
    # ----------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # ----------------------------------------------------------------------
    # 4. Prepare mask (FP16).  If ``nomask`` we simply use a tensor of ones.
    # ----------------------------------------------------------------------
    if mask is None or mask.numel() == 0 or config.get("nomask", False):
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16)

    # ----------------------------------------------------------------------
    # 5. Cast projections / gates to FP16 – the heavy contraction runs in FP16.
    # ----------------------------------------------------------------------
    left_proj_h = left_proj.to(torch.float16)
    right_proj_h = right_proj.to(torch.float16)
    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    # Allocate fused tensors that will hold (proj * gate * mask)
    left_fused = torch.empty_like(left_proj_h)
    right_fused = torch.empty_like(right_proj_h)

    # ----------------------------------------------------------------------
    # 6. Triton kernel – element‑wise fusion of projection, gate and mask
    # ----------------------------------------------------------------------
    @triton.jit
    def fuse_proj_gate_kernel(
        lp_ptr, rp_ptr, lg_ptr, rg_ptr,
        mask_ptr,
        lo_ptr, ro_ptr,
        B, N,
        stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_h,
        stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
        stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_h,
        stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
        stride_mask_b, stride_mask_i, stride_mask_j,
        stride_lo_b, stride_lo_i, stride_lo_j, stride_lo_h,
        stride_ro_b, stride_ro_i, stride_ro_j, stride_ro_h,
        BLOCK: tl.constexpr):
        """
        For each (b,i,j) element:
            left_out  = left_proj[b,i,j,:]  * left_gate[b,i,j,:]  * mask[b,i,j]
            right_out = right_proj[b,i,j,:] * right_gate[b,i,j,:] * mask[b,i,j]
        """
        pid = tl.program_id(0)
        total = B * N * N
        if pid >= total:
            return

        # decode flat index into (b, i, j)
        b = pid // (N * N)
        tmp = pid % (N * N)
        i = tmp // N
        j = tmp % N

        offs_h = tl.arange(0, BLOCK)

        # mask scalar (broadcasted over hidden)
        mask = tl.load(mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j)

        # left side
        lp = tl.load(lp_ptr + b * stride_lp_b + i * stride_lp_i + j * stride_lp_j
                     + offs_h * stride_lp_h)
        lg = tl.load(lg_ptr + b * stride_lg_b + i * stride_lg_i + j * stride_lg_j
                     + offs_h * stride_lg_h)
        left = lp * lg * mask
        tl.store(lo_ptr + b * stride_lo_b + i * stride_lo_i + j * stride_lo_j
                 + offs_h * stride_lo_h, left)

        # right side
        rp = tl.load(rp_ptr + b * stride_rp_b + i * stride_rp_i + j * stride_rp_j
                     + offs_h * stride_rp_h)
        rg = tl.load(rg_ptr + b * stride_rg_b + i * stride_rg_i + j * stride_rg_j
                     + offs_h * stride_rg_h)
        right = rp * rg * mask
        tl.store(ro_ptr + b * stride_ro_b + i * stride_ro_i + j * stride_ro_j
                 + offs_h * stride_ro_h, right)

    # ----------------------------------------------------------------------
    # Launch the fusion kernel in blocks of 128 hidden features.
    # ----------------------------------------------------------------------
    BLOCK_SIZE = 128
    total_elements = B * N * N
    grid = (total_elements,)

    # strides for all tensors (contiguous layout)
    lp_str = left_proj_h.stride()
    lg_str = left_gate_h.stride()
    rp_str = right_proj_h.stride()
    rg_str = right_gate_h.stride()
    lo_str = left_fused.stride()
    ro_str = right_fused.stride()
    mask_str = mask_fp16.stride()

    for h_start in range(0, hidden_dim, BLOCK_SIZE):
        cur_block = min(BLOCK_SIZE, hidden_dim - h_start)

        # slice the hidden dimension – creates a view with exactly cur_block entries
        lp_slice = left_proj_h[..., h_start:h_start + cur_block]
        lg_slice = left_gate_h[..., h_start:h_start + cur_block]
        rp_slice = right_proj_h[..., h_start:h_start + cur_block]
        rg_slice = right_gate_h[..., h_start:h_start + cur_block]
        lo_slice = left_fused[..., h_start:h_start + cur_block]
        ro_slice = right_fused[..., h_start:h_start + cur_block]

        fuse_proj_gate_kernel[grid](
            lp_slice, rp_slice, lg_slice, rg_slice,
            mask_fp16,
            lo_slice, ro_slice,
            B, N,
            lp_str[0], lp_str[1], lp_str[2], lp_str[3],
            lg_str[0], lg_str[1], lg_str[2], lg_str[3],
            rp_str[0], rp_str[1], rp_str[2], rp_str[3],
            rg_str[0], rg_str[1], rg_str[2], rg_str[3],
            mask_str[0], mask_str[1], mask_str[2],
            lo_str[0], lo_str[1], lo_str[2], lo_str[3],
            ro_str[0], ro_str[1], ro_str[2], ro_str[3],
            BLOCK=cur_block,
            num_warps=4,
        )
    torch.cuda.synchronize()

    # ----------------------------------------------------------------------
    # 7. Tri‑multiplicative contraction via batched matrix‑multiply (FP16)
    # ----------------------------------------------------------------------
    #   left_mat  : (B·H, N, N) with axes (i, k)
    #   right_mat : (B·H, N, N) with axes (k, j)  (transpose of the right side)
    left_mat = (
        left_fused.permute(0, 3, 1, 2)          # (B, H, i, k)
        .contiguous()
        .view(B * hidden_dim, N, N)
    )
    right_mat = (
        right_fused.permute(0, 3, 2, 1)         # (B, H, k, j)
        .contiguous()
        .view(B * hidden_dim, N, N)
    )
    out_mat_h = torch.bmm(left_mat, right_mat)          # FP16 result, shape (B·H, N, N)

    # reshape back to [B, N, N, hidden_dim] and convert to FP32 for the rest
    out = (
        out_mat_h.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)               # (B, N, N, hidden_dim)
        .to(torch.float32)
    )

    # ----------------------------------------------------------------------
    # 8. Output LayerNorm, output gate and final projection
    # ----------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
