    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BATCH_SIZE: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_stages: tl.constexpr = 2, # for prefetch
):

   batch_id = tl.program_id(0)
   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)
   # compute offsets
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

   # create pointers:

   b_ptrs = B_ptr + batch_id * stride_bb + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

"""
TriMul (outgoing) forward implementation with a fused batched
matrix‑multiply kernel in Triton.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Compute four linear projections of the normalized tensor:
   * left_proj, right_proj   – hidden dimension (H)
   * left_gate, right_gate, out_gate – apply sigmoid.
3. Apply optional mask and multiply left/right by their gates.
4. Reshape to a fused batch dimension (B*H) and perform a
   batched matrix multiply:
        out[b·H+i, p, q] = Σₖ left[b·H+i, p, k] * right[b·H+i, q, k]
   This is equivalent to a per‑channel multiplication of left with
   the transpose of right.
   The multiply is executed by a Triton kernel that processes the
   batch, M‑tile and N‑tile in three program‑ID dimensions.
5. Reshape back to [B, N, N, H] and apply a second LayerNorm.
6. Multiply by out_gate and apply the final linear projection back
   to the original dimension.
All heavy arithmetic (step 4) is performed in the Triton kernel,
the remaining ops use high‑performance PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr,                     # left  [B*H, M, K]
    b_ptr,                     # right [B*H, K, N] (interpreted as transposed)
    c_ptr,                     # out   [B*H, M, N]
    M, N, K,                   # matrix sizes
    stride_am, stride_ak, stride_ab,   # strides for A: row, col, batch
    stride_bk, stride_bn, stride_bb,   # strides for B: row (k), col (n), batch
    stride_cm, stride_cn, stride_cb,   # strides for C: row, col, batch
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B  (A shape [batch,M,K], B shape [batch,K,N])"""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers for the start of the tile
    a_ptrs = (
        a_ptr
        + batch_id * stride_ab
        + offs_m[:, None] * stride_am
        + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    )
    b_ptrs = (
        b_ptr
        + batch_id * stride_bb
        + tl.arange(0, BLOCK_K)[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )

    # --------------------------------------------------------------
    # Accumulator
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main loop over K dimension
    # --------------------------------------------------------------
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        cur_k = k * BLOCK_K
        # masks for out‑of‑bounds loads
        mask_a = (offs_m < M)[:, None] & (cur_k + tl.arange(0, BLOCK_K) < K)[None, :]
        mask_b = (cur_k + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n < N)[None, :]

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)                # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # Write back
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + batch_id * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask_c = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc, mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
               - input_tensor: torch.Tensor [B, N, N, D]
               - mask        : torch.Tensor [B, N, N]  (may be ignored)
               - weights     : dict of model parameters
               - config      : dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # convenience aliases for weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) First LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, D]

    # ------------------------------------------------------------------
    # 2) Linear projections and gating pre‑computations
    # ------------------------------------------------------------------
    # Linear maps on the last dimension
    left = F.linear(x, left_proj_weight)          # [B, N, N, H]
    right = F.linear(x, right_proj_weight)        # [B, N, N, H]

    # Optional mask (mask shape [B, N, N])
    if not nomask:
        mask_ = mask.unsqueeze(-1).type_as(left)   # broadcast over H
        left = left * mask_
        right = right * mask_

    # Gates (sigmoid after linear)
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3) Batched matrix multiplication via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B, N, N, H

    # Move hidden dimension into the batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Fuse batch*hidden into a single batch for the kernel
    batch_h = B * hidden_dim
    left_flat = left_perm.view(batch_h, N, N)   # [B*H, M=N, K=N]
    right_flat = right_perm.view(batch_h, N, N) # [B*H, K=N, N=N] (will be read as transposed)

    out_flat = torch.empty_like(left_flat)      # allocate output [B*H, N, N]

    # ------------------------------------------------------------------
    # Triton kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Strides in the flattened tensors (batch, row, col)
    stride_ab, stride_am, stride_ak = left_flat.stride()   # (batch, row, col)
    stride_bb, stride_bk, stride_bn = right_flat.stride()
    stride_cb, stride_cm, stride_cn = out_flat.stride()

    grid = (
        batch_h,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,               # M
        N,               # N
        N,               # K
        stride_am,
        stride_ak,
        stride_ab,
        stride_bk,
        stride_bn,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                     # element‑wise gate
    out = F.linear(out, to_out_weight)       # back to original dimension D

    return out
