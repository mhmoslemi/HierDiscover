"""
TriMul (outgoing) forward implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` over the last channel dimension.
2. Linear projections to the hidden dimension (`left_proj`, `right_proj`) and
   gating linear layers (`left_gate`, `right_gate`, `out_gate`).  Apply a
   sigmoid to the gate values.
3. Optionally mask the projected tensors (mask shape [B, N, N]).
4. Fuse the gates:   left  = left  * left_gate
                     right = right * right_gate
5. Compute the expensive term
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   for every batch `b` and hidden channel `d`.  This is a batch of
   matrix‑multiplications `C = A @ Bᵀ` where each matrix has shape (N, N).
   The operation is carried out by a Triton kernel that treats the
   combined (batch, hidden) dimension as the batch dimension of a GEMM.
   The kernel works in fp16 for the loads, accumulates in fp32 and writes
   the result in fp32.
6. Second layer‑norm over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_einsum_kernel(
    left_ptr,          # *fp16  [B*H, N, N]
    right_ptr,         # *fp16  [B*H, N, N]
    out_ptr,           # *fp32  [B*H, N, N]
    N: tl.constexpr,   # sequence length
    stride_l_bc, stride_l_m, stride_l_k,   # strides for left
    stride_r_bc, stride_r_m, stride_r_k,   # strides for right
    stride_o_bc, stride_o_m, stride_o_k,   # strides for output
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Program IDs
    bc = tl.program_id(0)          # combined batch*hidden index
    pid_m = tl.program_id(1)       # block row index
    pid_n = tl.program_id(2)       # block col index

    # Offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base offset for this (batch, hidden) slice
    bc_offset = bc * stride_l_bc

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension (the summed axis)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- Load A tile (left)  [M, K] -----
        a_ptrs = left_ptr + bc_offset \
                 + offs_m[:, None] * stride_l_m \
                 + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0)   # fp16

        # ----- Load B tile (right) transposed to [K, N] -----
        # We need right[bc, k, n] == right[bc, n, k]ᵀ
        b_ptrs = right_ptr + bc_offset \
                 + offs_k[:, None] * stride_r_m \
                 + offs_n[None, :] * stride_r_k
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0)   # fp16

        # Accumulate A @ B  (A: MxK, B: KxN) → (M,N)
        acc += tl.dot(a, b)   # result is fp32

    # Store the result
    c_ptrs = out_ptr + bc_offset \
             + offs_m[:, None] * stride_o_m \
             + offs_n[None, :] * stride_o_k
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a fused Triton kernel
    for the O(N³) pairwise interaction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim]
            - mask: torch.Tensor, shape [B, N, N] (bool/float)
            - weights: dict of model parameters
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # extract weights (already on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)      # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)    # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Mask (if present)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # broadcast mask to channel dimension
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel for the O(N³) term
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel (the kernel accumulates in fp32)
    left_fp16 = left.to(torch.float16).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_fp16 = right.to(torch.float16).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left_fp16.shape
    # Collapse batch and hidden dimensions => shape (B*H, N, N)
    left_flat = left_fp16.view(B * H, N, N)
    right_flat = right_fp16.view(B * H, N, N)

    # Output buffer in fp32
    out_flat = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # Block sizes – chosen to give good occupancy on H100 (can be tuned)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid: (batch*hidden, num_blocks_i, num_blocks_j)
    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch kernel
    tri_mul_einsum_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        # strides for left (contiguous [BC, N, N])
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # strides for right (same layout)
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        # strides for output (contiguous fp32)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out_perm = out_flat.view(B, H, N, N)                # [B, H, N, N]
    out = out_perm.permute(0, 2, 3, 1).contiguous()    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6. Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # ------------------------------------------------------------------
    # 7. Output gate and final projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
