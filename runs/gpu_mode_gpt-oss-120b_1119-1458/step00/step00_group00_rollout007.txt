```python
@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_az, stride_ai, stride_ak,
    stride_bz, stride_bk, stride_bj,
    stride_cz, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # Batch dimension
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    
    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    
    # Create mask for out-of-range
    mask_m = offs_m < M
    mask_n = offs_n < N
    
    # Pointers for this batch
    a_batch_ptr = a_ptr + batch_idx * stride_az
    b_batch_ptr = b_ptr + batch_idx * stride_bz
    c_batch_ptr = c_ptr + batch_idx * stride_cz
    
    # Load A tile
    a_ptrs = a_batch_ptr + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
    a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0).to(tl.float32)
    
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute actual k offset
        cur_k = k * BLOCK_K
        # Load A tile for this block (reuse with updated offsets)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_ai) + ((cur_k + offs_k)[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & ((cur_k + offs_k) < K), other=0.0).to(tl.float32)
        # Load B tile (B has shape K,N)
        b_ptrs = b_batch_ptr + ((cur_k + offs_k)[:, None] * stride_bk) + (offs_n[None, :] * stride_bj)
        b = tl.load(b_ptrs, mask=((cur_k + offs_k) < K)[:, None] & mask_n[None, :], other=0.0).to(tl.float32)
        
        acc += tl.dot(a, b)
    # Cast to output dtype (float16)
    c = acc.to(tl.float16)  # store as float16 maybe
    # Store results
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if present (mask may be None)
    if mask is not None:
        mask = mask.unsqueeze(-1).type_as(left)
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Convert to half for matmul
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Shape and permute to (B, H, N, N)
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()
    # Transpose right across the two N dims before permuting
    right_T = right.permute(0, 3, 2, 1).contiguous()  # now (B, H, N, N)

    # Flatten batch*H
    left_flat = left.view(-1, N, N)
    right_flat = right_T.view(-1, N, N)

    # output
    out_flat = torch.empty_like(left_flat)  # half dtype

    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    total_batches = left_flat.shape[0]
    grid = (total_batches, (N + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N)
    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape to (B, H, N, N)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # Convert back to float32
    out = out.to(torch.float32)

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_az, stride_ai, stride_ak,
    stride_bz, stride_bk, stride_bj,
    stride_cz, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out of bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to batch
    a_batch_ptr = a_ptr + pid_batch * stride_az
    b_batch_ptr = b_ptr + pid_batch * stride_bz
    c_batch_ptr = c_ptr + pid_batch * stride_cz

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over tiles of K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        # offsets for this k tile
        cur_k_offset = cur_k + tl.arange(0, BLOCK_K)
        # compute mask for k
        mask_k = cur_k_offset < K

        # Load A tile
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_ai + cur_k_offset[None, :] * stride_ak

                    other=0.0)           # A is float32

        # Load B tile (K × N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)           # B is float32

        # Accumulate the outer product
        acc += tl.dot(a, b)               # (BLOCK_M, BLOCK_K) · (BLOCK_K, BLOCK_N)

    # Write the result
    c_ptrs = c_batch + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict containing at least keys “dim” and “hidden_dim”

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (mask is broadcastable to [B,N,N,1])
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) + apply gates
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core multi‑head GEMM (the expensive einsum)
    #    We treat the hidden dimension as an extra batch dimension:
    #       left  → (B·H, N, N)   (i, k)
    #       right → (B·H, N, N)   (k, j)  (transpose on seq dimensions)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # reshape to (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()         # [B, H, N, N]
    right_T = right.permute(0, 3, 2, 1).contiguous()    # [B, H, N, N] (k, j)

    # flatten the batch*hidden dimension
    left_flat = left.view(-1, N, N)          # [B*H, N, N]
    right_flat = right_T.view(-1, N, N)      # [B*H, N, N]

    # allocate output buffer
    out_flat = torch.empty_like(left_flat)   # float32

    # launch parameters (tuned for H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        left_flat.shape[0],                               # batch dimension = B*H
        (N + BLOCK_M - 1) // BLOCK_M,                     # M tiles
        (N + BLOCK_N - 1) // BLOCK_N,                     # N tiles
    )

    _tri_mul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure kernel completion before proceeding (synchronization)
    torch.cuda.synchronize(device)

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # element‑wise gating

    # Final linear projection back to the original dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
