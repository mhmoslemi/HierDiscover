@triton.jit
def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        B, N, D,
        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_i, stride_right_j, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # program ids
    pid_head = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute which batch and hidden dim corresponds to pid_head
    b = pid_head // D  # integer division: batch index
    d = pid_head % D   # hidden dimension index
    # compute row offset
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # check bounds:
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointer offsets for left and right_T with indexing.
    # left: shape (B,N_i,N_k,D) => row dimension i (m), col dimension k (block_k)
    # right_T: shape (B,N_k,N_j,D) => row dimension k, col dimension j (n)
    
    # compute pointer offsets for each k block iteration inside loop.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # masks for k
        mask_k = offs_k < N

        # left load: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + b*stride_left_b + d*stride_left_d \
                    + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right_T load: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + b*stride_right_b + d*stride_right_d \
                     + offs_k[:, None] * stride_right_i + offs_n[None, :] * stride_right_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left, right)  # multiplies (M,K) dot (K,N) => (M,N)
    
    # store result in out
    out_ptrs = out_ptr + b*stride_out_b + d*stride_out_d \
               + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   B, N, D,
                   stride_left_b, stride_left_i, stride_left_k, stride_left_d,
                   stride_right_b, stride_right_i, stride_right_j, stride_right_d,
                   stride_out_b, stride_out_i, stride_out_j, stride_out_d,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_head = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    total_heads = B * D
    # Determine batch and dim index
    b = pid_head // D
    d = pid_head % D

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block (M x K)
        left_ptrs = left_ptr + b * stride_left_b + d * stride_left_d + \
                    offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right_T block (K x N)
        right_ptrs = right_ptr + b * stride_right_b + d * stride_right_d + \
                    offs_k[:, None] * stride_right_i + offs_n[None, :] * stride_right_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left, right)   # (M,K) dot (K,N) -> (M,N)

    # Store result
    out_ptrs = out_ptr + b * stride_out_b + d * stride_out_d + \
               offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward kernel for AlphaFold3‐style pair updates.

Algorithm
---------
1. Layer‑norm the input tensor (shape B×N×N×C).
2. Linear projections → left, right (B×N×N×H) without bias.
3. Optional mask (B×N×N) is applied to left/right.
4. Gated multiplicative update:
   left  = left  * sigmoid(linear(x, left_gate_weight))
   right = right * sigmoid(linear(x, right_gate_weight))
5. Compute the pairwise product over the third dimension (k):
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]   (B×N×N×H)
   This dense N³ contraction is implemented in Triton as a
   batched matrix‑multiply (left @ rightᵀ) for each head h.
6. Layer‑norm over the hidden dimension (H) → to_out_norm.
7. Apply the output gate (sigmoid(linear(x, out_gate_weight))) and
   a final linear projection back to the original channel dimension C.
8. Return the result in float32.

The heavy N³ contraction (step 5) is the only part written in Triton;
all other operations use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the N³ contraction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,            # const float16 *
    right_ptr,           # const float16 *
    out_ptr,             # float16 *
    B, N, H,             # int32 metadata
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication: out = left @ rightᵀ for each (b,h)."""
    pid_head = tl.program_id(0)   # encodes (batch, hidden)
    pid_m = tl.program_id(1)      # block row index (i)
    pid_n = tl.program_id(2)      # block col index (j)

    # --------------------------------------------------------------
    # Decode (batch, hidden) from the flattened head id
    # --------------------------------------------------------------
    total_heads = B * H
    b = pid_head // H
    h = pid_head % H

    # --------------------------------------------------------------
    # Offsets for the current output tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in blocks
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a BLOCK_M×BLOCK_K tile from left (rows i, cols k)
        left_ptrs = left_ptr + \
            b * stride_left_b + h * stride_left_h + \
            offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # Load a BLOCK_K×BLOCK_N tile from rightᵀ (rows k, cols j)
        # right_ptr points to the transposed view (k,j)
        right_ptrs = right_ptr + \
            b * stride_right_b + h * stride_right_h + \
            offs_k[:, None] * stride_right_i + offs_n[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # Accumulate the dot product over K
        acc += tl.dot(left, right)

    # --------------------------------------------------------------
    # Store the result tile (convert back to float16)
    # --------------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_out_b + h * stride_out_h + \
        offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def tri_mul_triton(left: torch.Tensor, right_t: torch.Tensor) -> torch.Tensor:
    """
    Triton‑accelerated implementation of:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right_t[b,k,j,h]
    where `right_t` is the k‑j view (right transposed on the two N axes).
    Both inputs must be of dtype torch.float16 and reside on the same CUDA device.
    """
    assert left.is_cuda and right_t.is_cuda
    B, N, _, H = left.shape

    # Allocate output (same dtype as inputs)
    out = torch.empty_like(left)

    # Strides in element units (as expected by Triton)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rj, s_rh = right_t.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Block sizes – chosen to give good occupancy on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                                     # head dimension (batch × hidden)
        (N + BLOCK_M - 1) // BLOCK_M,              # number of row tiles
        (N + BLOCK_N - 1) // BLOCK_N,              # number of col tiles
    )

    tri_mul_kernel[grid](
        left, right_t, out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rj, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # H100 benefits from 4 warps per block
    )
    return out


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dict with Tensor values for all linear / norm weights
        - config_dict  : dict containing at least 'dim' and 'hidden_dim'

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper: move a weight to the correct device/dtype (half for compute)
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Extract model parameters
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w   = to_half(weights["left_proj.weight"])
    right_proj_w  = to_half(weights["right_proj.weight"])
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(input_tensor,
                          normalized_shape=(dim,),
                          weight=norm_w,
                          bias=norm_b,
                          eps=1e-5)          # [B,N,N,C]  (float32)

    # Cast to half for the heavy lifting
    x_half = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_half, left_proj_w)   # [B,N,N,hidden_dim]
    right = F.linear(x_half, right_proj_w) # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # Ensure mask is float16 and broadcastable
        mask_f = mask.to(dtype=torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_half, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_half, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_half, out_gate_w))  # [B,N,N,hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. N³ contraction via Triton (left @ rightᵀ per hidden dimension)
    # ------------------------------------------------------------------
    # rightᵀ over the two N axes: shape becomes (B, N_k, N_j, hidden_dim)
    right_t = right.permute(0, 2, 1, 3).contiguous()
    out = tri_mul_triton(left, right_t)  # [B,N,N,hidden_dim] (float16)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over hidden_dim (use float32 internally)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(out_fp32,
                                 normalized_shape=(hidden_dim,),
                                 weight=to_out_norm_w.to(torch.float32),
                                 bias=to_out_norm_b.to(torch.float32),
                                 eps=1e-5)
    out_norm = out_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 7. Apply output gate and final linear projection
    # ------------------------------------------------------------------
    gated = out_norm * out_gate
    final = F.linear(gated, to_out_w)    # [B,N,N,dim] (float16)

    # ------------------------------------------------------------------
    # Return in float32 as required by the specification
    # ------------------------------------------------------------------
    return final.to(torch.float32)
