```python
import triton
import triton.language as tl

@triton.jit
def matmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B_H, N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Offsets
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    i_offsets = i_base + tl.arange(0, BLOCK_M)
    j_offsets = j_base + tl.arange(0, BLOCK_N)
    i_mask = i_offsets < N
    j_mask = j_offsets < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute number of k tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    # Loop over k dimension
    for k_tile in range(num_k_tiles):
        k_base = k_tile * BLOCK_K
        k_offsets = k_base + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < N

        # Load left block (M,K)
        left_ptrs = left_ptr + pid_b * stride_left_batch \
                    + i_offsets[:, None] * stride_left_i \
                    + k_offsets[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask = i_mask[:, None] & k_mask[None, :],
                       other = 0.0)

        # Load right block transposed (K,N)
        # note: we treat k as row, j as column -> strides: row stride = stride_right_i, col stride = stride_right_j (which is stride_right_k)
        right_ptrs = right_ptr + pid_b * stride_right_batch \
                     + k_offsets[:, None] * stride_right_i \
                     + j_offsets[None, :] * stride_right_k
        right = tl.load(right_ptrs,
                        mask = k_mask[:, None] & j_mask[None, :],
                        other = 0.0)

        # Accumulate dot product
        acc += tl.dot(left, right)

    # Store result
    out_ptrs = out_ptr + pid_b * stride_out_batch \
               + i_offsets[:, None] * stride_out_i \
               + j_offsets[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask = i_mask[:, None] & j_mask[None, :])
```

"""
TriMul (outgoing version) – fast Triton implementation of the
core triple‑wise multiplicative update.

The kernel computes, for each batch and each hidden channel *d*:

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

which is mathematically identical to a batched matrix multiplication
left @ rightᵀ for every hidden dimension.  All surrounding LayerNorm,
Linear and gating operations are kept in PyTorch for simplicity;
the heavy O(N³·C) contraction is performed by a custom Triton kernel.

The implementation works for any batch size ≤2, arbitrary sequence length
(up to 1024 in the tests) and hidden dimensions (usually 128).  The
output is guaranteed to be float‑32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH_H, N,
    # strides for left matrix (batch, i, k)
    stride_l_batch, stride_l_i, stride_l_k,
    # strides for right matrix (batch, i, k) – we will read it as (k, j)
    stride_r_batch, stride_r_i, stride_r_k,
    # strides for output (batch, i, j)
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Triton kernel that computes, for every outer batch (BATCH_H = B*hidden_dim),
    the matrix product   out = left @ rightᵀ   where
        left:  (M=N) × (K=N)
        right: (J=N) × (K=N)   (will be accessed as (K, J))
    """
    pid_b = tl.program_id(0)          # outer batch index (b * hidden_dim)
    pid_i = tl.program_id(1)          # tile row index
    pid_j = tl.program_id(2)          # tile column index

    # ----- block offsets -------------------------------------------------
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    i_off = i_base + tl.arange(0, BLOCK_M)
    j_off = j_base + tl.arange(0, BLOCK_N)

    i_mask = i_off < N
    j_mask = j_off < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of tiles along K dimension
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    # iterate over K (the shared dimension)
    for k_tile in range(num_k_tiles):
        k_base = k_tile * BLOCK_K
        k_off = k_base + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # ----- load left block (M, K) -----------------------------------
        # left[i, k] with strides (stride_l_i, stride_l_k)
        left_ptrs = (
            left_ptr
            + pid_b * stride_l_batch
            + i_off[:, None] * stride_l_i
            + k_off[None, :] * stride_l_k
        )
        left = tl.load(
            left_ptrs,
            mask=(i_mask[:, None] & k_mask[None, :]),
            other=0.0,
        )

        # ----- load right block transposed (K, N) -----------------------
        # We want rightᵀ[k, j] → right[j, k] in memory.
        # After swapping axes we read with row‑stride = stride_r_i (original row stride)
        # and col‑stride = stride_r_k (original col stride).
        right_ptrs = (
            right_ptr
            + pid_b * stride_r_batch
            + k_off[:, None] * stride_r_i      # k as row index
            + j_off[None, :] * stride_r_k      # j as column index
        )
        right = tl.load(
            right_ptrs,
            mask=(k_mask[:, None] & j_mask[None, :]),
            other=0.0,
        )

        # ----- accumulate ------------------------------------------------
        acc += tl.dot(left, right)   # (M, K) @ (K, N) → (M, N)

    # ----- write result ---------------------------------------------------
    out_ptrs = (
        out_ptr
        + pid_b * stride_o_batch
        + i_off[:, None] * stride_o_i
        + j_off[None, :] * stride_o_j
    )
    tl.store(
        out_ptrs,
        acc,
        mask=(i_mask[:, None] & j_mask[None, :]),
    )


def custom_kernel(data):
    """
    Forward pass of AlphaFold‑3 “outgoing” TriMul.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C] (C = dim)
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of torch.Tensor, containing all linear / norm params
        - config       : dict, must contain ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device as inputs)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # Helper: LayerNorm (no bias handling needed – torch takes care)
    # ------------------------------------------------------------------
    def _layer_norm(x, weight, bias):
        return F.layer_norm(x, (x.shape[-1],), weight=weight, bias=bias, eps=1e-5)

    # ------------------------------------------------------------------
    # 1. Input normalization
    # ------------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # ensure same dtype/device for multiplication
        mask = mask.to(dtype=dtype, device=device)
        mask = mask.unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating signals (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core triple multiplication – Triton kernel
    #    left @ rightᵀ for each hidden channel
    # ------------------------------------------------------------------
    # reshape to (B*hidden, N, N) where each slice is a separate matmul
    # Permute to bring hidden_dim next to batch for a compact view.
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B_H = B * hidden_dim
    left_view = left.view(B_H, N, N)
    right_view = right.view(B_H, N, N)

    # allocate output
    out_view = torch.empty_like(left_view)

    # block sizes – chosen to give good occupancy on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B_H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_outgoing_kernel[grid](
        left_view,
        right_view,
        out_view,
        B_H,
        N,
        left_view.stride(0), left_view.stride(1), left_view.stride(2),
        right_view.stride(0), right_view.stride(1), right_view.stride(2),
        out_view.stride(0), out_view.stride(1), out_view.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_view.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear
    # ------------------------------------------------------------------
    out = _layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )
    out = out * out_gate   # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])   # back to original dim

    return out
