@triton.jit
def layernorm_kernel(
    x_ptr, out_ptr,
    weight_ptr, bias_ptr,
    B, N, D,
    stride_xb, stride_xn1, stride_xn2, stride_xd,
    stride_outb, stride_outn1, stride_outn2, stride_outd,
    BLOCK_D: tl.constexpr,
    EPS: tl.constexpr = 1e-5
):
    pid = tl.program_id(0) # range = B * N * N
    total = B * N * N
    idx = pid
    # compute b,i,j from idx
    i = idx % N
    idx = idx // N
    j = idx % N
    b = idx // N

    # compute pointers base
    x_ptr_offset = b * stride_xb + i * stride_xn1 + j * stride_xn2

    # load block of D
    offs_d = tl.arange(0, BLOCK_D)
    mask = offs_d < D

    x = tl.load(x_ptr + x_ptr_offset + offs_d * stride_xd, mask=mask, other=0.0)
    # compute mean
    x_sum = tl.sum(x, axis=0)
    mean = x_sum / D

    # compute variance
    diff = x - mean
    var_sum = tl.sum(diff * diff, axis=0)
    var = var_sum / D

    # normalize
    inv_std = tl.math.rsqrt(var + EPS)
    x_norm = (x - mean) * inv_std

    # apply weight and bias
    w = tl.load(weight_ptr + offs_d * stride_wd, mask=mask, other=1.0) # weight shape D
    b_ = tl.load(bias_ptr + offs_d * stride_bd, mask=mask, other=0.0)
    y = x_norm * w + b_

    # store
    tl.store(out_ptr + x_ptr_offset + offs_d * stride_outd, y, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    B, N, _, D = input_tensor.shape # D should be dim

    # Get parameters weight pointers for layernorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Ensure they are float32.

    # Use Triton layernorm kernel
    # We'll allocate buffer for normalized x
    x_norm = torch.empty_like(input_tensor)

    # compute grid: num_programs = B*N*N
    # Determine BLOCK_D as next power of 2 >= dim, maybe 128 or 384.
    BLOCK_D = 128 if dim <= 128 else 256 if dim <= 256 else 384 if dim <= 384 else 512

    grid = (B * N * N, )
    layernorm_kernel[grid](
        input_tensor, x_norm,
        norm_weight, norm_bias,
        B, N, dim,
        input_tensor.stride(0), input_tensor.stride(1), input_tensor.stride(2), input_tensor.stride(3),
        x_norm.stride(0), x_norm.stride(1), x_norm.stride(2), x_norm.stride(3),
        BLOCK_D=BLOCK_D
    )

    # linear projections
    # flatten x_norm to [B*N*N, dim]
    x_flat = x_norm.view(-1, dim)
    # left_proj
    left = torch.nn.functional.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = torch.nn.functional.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    # mask
    if mask is not None:
        mask = mask.to(x_norm.dtype)
        mask_expanded = mask.unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_expanded
        right = right * mask_expanded

    # left_gate, right_gate, out_gate
    left_gate = torch.nn.functional.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim)
    left_gate = left_gate.sigmoid()
    right_gate = torch.nn.functional.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim)
    right_gate = right_gate.sigmoid()
    out_gate = torch.nn.functional.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim)
    out_gate = out_gate.sigmoid()

    left = left * left_gate
    right = right * right_gate

    # TriMul step: for each hidden dim compute matmul
    # reshape left/right to [B*hidden_dim, N, N]
    left_t = left.permute(0, 3, 1, 2).contiguous() # [B, hidden, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()
    # flatten batch
    bh = B * hidden_dim
    left_mat = left_t.view(bh, N, N)
    right_mat = right_t.view(bh, N, N)
    # out_mat = left_mat @ right_mat.transpose(1,2)
    # use torch.bmm
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous() # [B,N,N,hidden]

"""
TriMul (outgoing) forward pass implemented with a fused LayerNorm kernel in Triton
and the remaining heavy linear / matmul work delegated to CuBLAS via PyTorch.

Algorithm
---------
1. **LayerNorm (first)** – A custom Triton kernel computes the mean/variance
   across the last `dim` dimension of `x` (shape [B, N, N, dim]) and applies
   the learned weight/bias.  The kernel processes one (b,i,j) element per
   program and works for any `dim` ≤ 384 (BLOCK_D=512 is used).

2. **Fused linear projections** – All five linear layers
   (left_proj, right_proj, left_gate, right_gate, out_gate) share the same
   input `x_norm`.  Their weight matrices (each [hidden, dim]) are stacked
   into a single matrix `[5*hidden, dim]` and a single `torch.nn.functional.linear`
   call produces a tensor of shape `[B, N, N, 5, hidden]`.  The first two slices
   are the projected tensors, the remaining three are gate logits that are
   passed through a sigmoid.

3. **Mask & gating** – If a mask is present (and `nomask` is False) it is
   broadcast and multiplied into `left` and `right`.  Afterwards the gate
   values multiply the projected tensors.

4. **TriMul reduction** – For each hidden channel we need
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   This is exactly a batched matrix multiplication:
   `out_h = left_h @ right_h.T`.  The tensors are reshaped to
   `[B*hidden, N, N]` and a single `torch.bmm` call computes the whole
   reduction efficiently on the GPU.

5. **Output normalisation & projection** – `out` is LayerNorm‑normalised
   with its own learned weight/bias, multiplied element‑wise with the
   previously computed `out_gate`, and finally projected back to `dim`
   with a linear layer.

The final tensor has shape `[B, N, N, dim]` and dtype `torch.float32`.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: fused LayerNorm (no bias for hidden parts)
# --------------------------------------------------------------
@triton.jit
def layernorm_fused_kernel(
    x_ptr,               # *float32  input [B,N,N,dim]
    out_ptr,             # *float32  output normalized
    weight_ptr,          # *float32  LayerNorm weight (dim)
    bias_ptr,            # *float32  LayerNorm bias   (dim)
    B, N, D,             # int64     batch, seq_len, dim
    stride_xb, stride_xn, stride_xm, stride_xd,   # int64 strides of x
    stride_ob, stride_on, stride_om, stride_od,   # int64 strides of out
    stride_w, stride_b,                           # int64 strides of weight/bias (usually 1)
    BLOCK_D: tl.constexpr,                        # compile‑time block size for dim
    EPS: tl.constexpr = 1e-5
):
    pid = tl.program_id(0)                  # one program per (b,i,j)
    total = B * N * N
    if pid >= total:
        return

    # decode (b,i,j)
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # base pointers for the (b,i,j,0) element
    base_x = x_ptr + b * stride_xb + i * stride_xn + j * stride_xm
    base_o = out_ptr + b * stride_ob + i * stride_on + j * stride_om

    # -------------------- load a block of D values --------------------
    offs_d = tl.arange(0, BLOCK_D)
    mask_d = offs_d < D

    x = tl.load(base_x + offs_d * stride_xd, mask=mask_d, other=0.0)

    # -------------------- mean --------------------
    sum_x = tl.sum(x, axis=0)
    mean = sum_x / D

    # -------------------- variance --------------------
    diff = x - mean
    var = tl.sum(diff * diff, axis=0) / D

    # -------------------- normalise --------------------
    inv_std = 1.0 / tl.sqrt(var + EPS)
    x_norm = (x - mean) * inv_std

    # -------------------- apply weight & bias --------------------
    w = tl.load(weight_ptr + offs_d * stride_w, mask=mask_d, other=1.0)
    b_ = tl.load(bias_ptr + offs_d * stride_b, mask=mask_d, other=0.0)
    y = x_norm * w + b_

    # -------------------- store --------------------
    tl.store(base_o + offs_d * stride_od, y, mask=mask_d)


# --------------------------------------------------------------
# Entry point expected by the evaluator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward for the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # unpack
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                # input / output channel dimension
    hidden_dim = config["hidden_dim"]  # inner hidden dimension
    nomask = config.get("nomask", False)

    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    assert D == dim, "input last dimension must match config 'dim'"

    # ----------------------------------------------------------
    # 1) First LayerNorm – Triton
    # ----------------------------------------------------------
    x_norm = torch.empty_like(input_tensor)  # float32

    # Decide a compile‑time block size that covers the maximal dim (≤384)
    # Use 512 (power‑of‑two) – the kernel masks excess elements.
    BLOCK_D = 512

    # Launch one program per (b,i,j)

        input_tensor,
        x_norm,

        weights["norm.bias"],

        input_tensor.stride(0), input_tensor.stride(1),
        input_tensor.stride(2), input_tensor.stride(3),
        x_norm.stride(0), x_norm.stride(1),
        x_norm.stride(2), x_norm.stride(3),
        weights["norm.weight"].stride(0), weights["norm.bias"].stride(0),
        BLOCK_D=BLOCK_D,
        EPS=1e-5
    )

    # ----------------------------------------------------------
    # 2) Fused linear projections (5 * hidden_dim outputs)
    # ----------------------------------------------------------
    # Flatten the normalized tensor for a single matmul.
    x_flat = x_norm.view(-1, dim)                     # [B*N*N, dim]

    # Stack all projection weights: order = left, right, left_gate, right_gate, out_gate
    proj_weights = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0)                                           # [5*hidden, dim]

    # Single linear call – returns [B*N*N, 5*hidden]
    proj = F.linear(x_flat, proj_weights)               # bias is None (all Linear layers have no bias)

    # Reshape & split
    proj = proj.view(B, N, N, 5, hidden_dim)            # [B, N, N, 5, hidden]
    left = proj[..., 0, :]                              # [B, N, N, hidden]
    right = proj[..., 1, :]
    left_gate = proj[..., 2, :].sigmoid()
    right_gate = proj[..., 3, :].sigmoid()
    out_gate = proj[..., 4, :].sigmoid()

    # ----------------------------------------------------------
    # 3) Mask handling + gating
    # ----------------------------------------------------------
    if not nomask:
        # mask_tensor: [B, N, N] – cast to same dtype and broadcast
        mask = mask_tensor.to(x_norm.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------
    # 4) TriMul reduction (batched matmul)
    # ----------------------------------------------------------
    # Rearrange so that hidden dimension becomes batch for bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, hidden, N, N]

    bh = B * hidden_dim
    left_mat = left_perm.view(bh, N, N)                 # [B*hidden, N, N]
    right_mat = right_perm.view(bh, N, N)               # [B*hidden, N, N]

    # out_mat[b*h, i, j] = Σ_k left_mat[b*h, i, k] * right_mat[b*h, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout [B, N, N, hidden]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------
    # 5) Output LayerNorm, gating and final projection
    # ----------------------------------------------------------
    # LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Multiply by out_gate (already sigmoid‑ed)
    out = out * out_gate

    # Final linear projection back to `dim`
    out_flat = out.view(-1, hidden_dim)                 # [B*N*N, hidden]
    out = F.linear(out_flat, weights["to_out.weight"]).view(B, N, N, dim)

    return out
```
