```python
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm: x_norm = (x - mean) / sqrt(var+eps) * weight + bias
    eps = 1e-5
    # Use torch.nn.functional.layer_norm with weight and bias
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5
    )
    # Linear projections: left_proj, right_proj
    # weight shapes: [hidden_dim, dim] for each projection
    # using torch.nn.functional.linear (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if needed
    if not nomask:
        # mask shape [B,N,N]; unsqueeze last dim
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_

    # Gates:
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Convert left/right to fp16 for faster kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape
    # flatten heads
    total_heads = B * hidden_dim  # also called H

    # reshape left/right to (total_heads, N, N)
    left = left.permute(0, 3, 1, 2).reshape(total_heads, N, N).contiguous()
    right = right.permute(0, 3, 1, 2).reshape(total_heads, N, N).contiguous()

    # allocate output
    out = torch.empty(total_heads, N, N, device=left.device, dtype=torch.float32)

    # launch kernel
    _tri_mul_kernel[(total_heads, (N+BLOCK_M-1)//BLOCK_M, (N+BLOCK_N-1)//BLOCK_N)](
        left, right, out,
        N,
        stride_left_head = left.stride(0), stride_left_i = left.stride(1), stride_left_k = left.stride(2),
        stride_right_head = right.stride(0), stride_right_i = right.stride(1), stride_right_k = right.stride(2),
        stride_out_head = out.stride(0), stride_out_i = out.stride(1), stride_out_j = out.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # reshape output back to (B, N, N, hidden_dim)
    out = out.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=1e-5)
    # apply out_gate
    out = out * out_gate

    # final linear to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

import triton
import triton.language as tl

@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_left_head, stride_left_i, stride_left_k,
    stride_right_head, stride_right_i, stride_right_k,
    stride_out_head, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_h = tl.program_id(0)  # head index
    pid_m = tl.program_id(1)  # block row i
    pid_n = tl.program_id(2)  # block col j

    # compute offsets
    i_base = pid_m * BLOCK_M
    j_base = pid_n * BLOCK_N

    # range
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    # mask for boundaries
    mask_i = offs_i < N
    mask_j = offs_j < N

    # offsets for output
    out_ptrs = out_ptr + pid_h * stride_out_head + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    num_blocks_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_blocks_k):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile (M x K)
        left_ptrs = left_ptr + pid_h * stride_left_head + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right tile (K x N) (transposed)
        # right[j,k] we need shape (K,N) where rows correspond to k, cols to j
        right_ptrs = right_ptr + pid_h * stride_right_head + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        # note: stride_right_k = 1 (col), stride_right_i = N (row)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate
        acc += tl.dot(left, right)  # left: (M,K) dot right: (K,N) -> (M,N)

    # store result
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
custom_kernel
==============

Implements the *outgoing* Triangle Multiplicative Update (TriMul) forward pass
described in the AlphaFold‑3 paper.  

The heavy ``einsum('... i k d, ... j k d -> ... i j d')`` is expressed as a
batch of independent matrix‑multiplications ``C = A @ Bᵀ`` where each head
corresponds to one hidden dimension of a batch element.  The GEMM is tiled and
executed with a small Triton kernel, while the surrounding LayerNorms,
linear projections, and gates are handled by PyTorch.  All operations are
performed in mixed precision (projections & gates in fp16, accumulation in
fp32) and the final output is returned as ``float32`` with shape
``[B, N, N, dim]``.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: one head = one (hidden_dim) slice of the tensor.
# For each head we compute C[i, j] = sum_k A[i, k] * B[j, k]
# (equivalent to A @ B.T).  Tiling in M, N and K.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # *fp16   left  = left_proj * left_gate * mask
    right_ptr,         # *fp16   right = right_proj * right_gate * mask
    out_ptr,           # *fp32   result (hidden_dim output)
    N,                 # i32     sequence length
    stride_lh, stride_li, stride_lk,   # i64    strides for left   (head, i, k)
    stride_rh, stride_ri, stride_rk,   # i64    strides for right  (head, j, k)
    stride_oh, stride_oi, stride_oj,   # i64    strides for out    (head, i, j)
    BLOCK_M: tl.constexpr,  # tile size in i‑dimension
    BLOCK_N: tl.constexpr,  # tile size in j‑dimension
    BLOCK_K: tl.constexpr,  # tile size in k‑dimension
):
    pid_h = tl.program_id(0)               # which head (batch*hidden)
    pid_m = tl.program_id(1)               # block row (i)
    pid_n = tl.program_id(2)               # block col (j)

    # --------------------------------------------------------------
    # Offsets for the M and N dimensions, plus out‑of‑bounds masks
    # --------------------------------------------------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Pointers to the beginning of the current head (slice)
    # ------------------------------------------------------------------
    left_head_ptr = left_ptr + pid_h * stride_lh
    right_head_ptr = right_ptr + pid_h * stride_rh
    out_head_ptr = out_ptr + pid_h * stride_oh

    # ------------------------------------------------------------------
    # Accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in blocks
    # ------------------------------------------------------------------
    num_blocks_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_blocks_k):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---------- load left tile [M, K] ----------
        left_ptrs = (
            left_head_ptr
            + offs_i[:, None] * stride_li
            + offs_k[None, :] * stride_lk
        )
        left_mask = mask_i[:, None] & mask_k[None, :]
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # ---------- load right tile transposed to [K, N] ----------
        # right is stored as (head, j, k).  We want rows = k, cols = j.
        right_ptrs = (
            right_head_ptr
            + offs_k[:, None] * stride_rk
            + offs_j[None, :] * stride_ri
        )
        right_mask = mask_k[:, None] & mask_j[None, :]
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # ---------- dot product ----------
        acc += tl.dot(left_tile, right_tile)   # (M,K) @ (K,N) -> (M,N)

    # ------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------
    out_ptrs = out_head_ptr + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)


# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the TriMul module.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"]) # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # Optional masking (mask shape [B, N, N])
    # ------------------------------------------------------------------
    if not nomask:
        # mask is broadcast on the hidden dimension
        mask_f = mask.unsqueeze(-1).to(dtype)                 # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating values (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # keep fp32 for later

    # ------------------------------------------------------------------
    # Apply gates
    # ------------------------------------------------------------------
    left = left.to(torch.float16) * left_gate
    right = right.to(torch.float16) * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the Triton kernel:
    #   reshape to (total_heads, N, N) where total_heads = B * hidden_dim
    #   layout: contiguous (head, i, k) / (head, j, k)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    total_heads = B * hidden_dim

    # (B, N, N, hidden) --> (B, hidden, N, N) --> (total_heads, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # Output buffer (fp32)
    out_flat = torch.empty(
        (total_heads, N, N), dtype=torch.float32, device=device
    )

    # --------------------------------------------------------------
    # Triton launch configuration
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        total_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # Reshape back to (B, N, N, hidden_dim)
    # --------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, hidden]

    # --------------------------------------------------------------
    # Post‑processing: LayerNorm -> gating -> final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # broadcast over hidden dimension

    out = F.linear(out, weights["to_out.weight"])  # final projection to `dim`

    return out
