def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm weight/bias for input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Linear weights
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Ensure contiguity for subsequent flattening
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    if not nomask:
        # Apply mask and gate with Triton kernel
        # Prepare mask as float (0/1 or maybe any value)
        mask = mask_tensor.to(x.dtype)  # Should be broadcastable
        # left_out = left * mask * left_gate
        # right_out = right * mask * right_gate
        left_out = torch.empty_like(left)
        right_out = torch.empty_like(right)

        B, N, _, _ = left.shape  # left.shape = [B,N,N,hidden_dim]; _ is N second; hidden_dim at last

        # Compute total elements

        BLOCK_SIZE = 128  # 128 threads per program; could tune

        stride_mask = mask.stride()
        stride_left_gate = left_gate.stride()
        # Right stride
        stride_right = right.stride()
        stride_right_gate = right_gate.stride()

        # Launch for left

            B, N, hidden_dim,

@triton.jit
def mask_gate_kernel(
    tensor_ptr, mask_ptr, gate_ptr, out_ptr,
    B, N, H,
    stride_tensor_b, stride_tensor_i, stride_tensor_k, stride_tensor_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    total = B * N * N * H
    valid = offs < total

    # compute mask offset
    # integer division
    mask_offs = tl.div(offs, H)  # floor division
    mask_valid = mask_offs < (B * N * N)

    out_val = tensor_val * mask_val * gate_val

    tl.store(out_ptr + offs, out_val, mask=valid)

"""
TriMul (outgoing) forward implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` over the last dimension (dim).
2. Four linear projections (no bias) are applied to the normalized input:
      * left_proj   :  dim → hidden_dim
      * right_proj  :  dim → hidden_dim
      * left_gate   :  dim → hidden_dim (followed by sigmoid)
      * right_gate  :  dim → hidden_dim (followed by sigmoid)
      * out_gate    :  dim → hidden_dim (followed by sigmoid)
3. If a mask is present (`nomask=False`) a small Triton kernel fuses
   three element‑wise operations:
        out = left * mask * left_gate
        out = right * mask * right_gate
   The kernel works on a flattened view of the tensors, so it
   handles any batch size, sequence length and hidden dimension.
   When `nomask=True` the mask is omitted and the same
   element‑wise products are performed with plain PyTorch.
4. The core “TriMul” operation is a batched matrix multiplication:
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   which is exactly a GEMM for each hidden slice:
        (B·hidden_dim, N, N) @ (B·hidden_dim, N, N)ᵀ
   This is performed with `torch.bmm`, which on an H100 uses the
   highly‑optimised cuBLAS kernels.
5. A second LayerNorm over the hidden dimension is applied,
   multiplied by the `out_gate` tensor and finally projected back
   to the original `dim` with a linear layer (no bias).
6. The result has shape `[B, N, N, dim]` and dtype `torch.float32`.

Only the mask & gate fusion is done in Triton; all heavy linear
operations and the N³ accumulation are performed with cuBLAS /
PyTorch for maximum throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused mask * gate * tensor (element‑wise)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    tensor_ptr,          # [B,N,N,H] flattened pointer (input tensor)
    mask_ptr,            # [B,N,N]   flattened pointer (mask)
    gate_ptr,            # [B,N,N,H] flattened pointer (gate)
    out_ptr,             # [B,N,N,H] flattened pointer (output)
    B, N, H,             # static problem sizes
    stride_tensor_b, stride_tensor_i, stride_tensor_k, stride_tensor_h,  # not used (contiguous layout assumed)
    stride_mask_b, stride_mask_i, stride_mask_k,                     # not used
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,     # not used
    BLOCK_SIZE: tl.constexpr,
):
    """
    out = tensor * mask * gate
    All inputs are assumed to be contiguous in memory (C‑order).
    The kernel processes BLOCK_SIZE consecutive elements of the
    flattened tensor.  ``mask`` is broadcast across the hidden
    dimension.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)                     # linear offsets

    total = B * N * N * H
    valid = offs < total

    # mask offset = floor(offs / H)  (broadcast across hidden dim)
    mask_offs = tl.div(offs, H)
    mask_valid = mask_offs < (B * N * N)

    # Load data (mask is a scalar per position, gate and tensor are 1‑to‑1)
    tensor_val = tl.load(tensor_ptr + offs, mask=valid, other=0.0)
    gate_val   = tl.load(gate_ptr   + offs, mask=valid, other=0.0)
    mask_val   = tl.load(mask_ptr   + mask_offs, mask=mask_valid, other=0.0)

    out = tensor_val * mask_val * gate_val
    tl.store(out_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim]
            - mask_tensor  : torch.Tensor, shape [B, N, N] (may be ignored)
            - weights_dict : mapping of weight names -> torch.Tensor
            - config_dict  : contains at least "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on the same device).
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # fetch weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]     # (hidden_dim, dim)

    left_gate_weight   = weights["left_gate.weight"]    # (hidden_dim, dim)
    right_gate_weight  = weights["right_gate.weight"]   # (hidden_dim, dim)
    out_gate_weight    = weights["out_gate.weight"]     # (hidden_dim, dim)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (hidden_dim,)
    to_out_norm_bias   = weights["to_out_norm.bias"]    # (hidden_dim,)

    to_out_weight = weights["to_out.weight"]            # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # 2) Linear projections + sigmoid gates
    # ------------------------------------------------------------------
    left       = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right      = F.linear(x, right_proj_weight)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 3) Mask + gate fusion (Triton) – only when a mask is required
    # ------------------------------------------------------------------
    if not nomask:
        # make sure mask is float (0/1) and on the same device/dtype
        mask = mask_tensor.to(x.dtype)

        # Allocate output tensors
        left_out  = torch.empty_like(left)
        right_out = torch.empty_like(right)

        B, N, _, H = left.shape  # H == hidden_dim
        total_elements = left.numel()
        BLOCK_SIZE = 128  # can be tuned; 128 works well on H100

        # Grid size – one program per BLOCK_SIZE elements
        grid = ((total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE,)

        # Strides (not used inside kernel, but required by signature)
        s_left  = left.stride()
        s_mask  = mask.stride()
        s_lgate = left_gate.stride()
        s_rgate = right_gate.stride()

        # ------------------------------------------------------------------
        # Left side
        # ------------------------------------------------------------------
        mask_gate_kernel[grid](
            left, mask, left_gate, left_out,
            B, N, H,
            s_left[0], s_left[1], s_left[2], s_left[3],
            s_mask[0], s_mask[1], s_mask[2],
            s_lgate[0], s_lgate[1], s_lgate[2], s_lgate[3],
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # ------------------------------------------------------------------
        # Right side
        # ------------------------------------------------------------------
        mask_gate_kernel[grid](
            right, mask, right_gate, right_out,
            B, N, H,
            right.stride()[0], right.stride()[1], right.stride()[2], right.stride()[3],
            s_mask[0], s_mask[1], s_mask[2],
            s_rgate[0], s_rgate[1], s_rgate[2], s_rgate[3],
            BLOCK_SIZE=BLOCK_SIZE,
        )
    else:
        # No mask – simple element‑wise multiplication with gates
        left_out  = left * left_gate
        right_out = right * right_gate
        # mask may still be present but should be ignored when nomask=True
        # (the reference implementation multiplies by an all‑ones mask)
        # If a mask tensor is supplied we keep it for consistency:
        if mask_tensor is not None:
            left_out  = left_out  * mask_tensor.unsqueeze(-1)
            right_out = right_out * mask_tensor.unsqueeze(-1)

    # ------------------------------------------------------------------
    # 4) TriMul core: batched GEMM over the hidden dimension
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # ------------------------------------------------------------------
    B, N, _, H = left_out.shape
    # reshape to (B*H, N, N) for torch.bmm
    left_flat  = left_out.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right_out.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # right needs to be transposed on the last two dimensions (k ↔ j)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (B*H, N, N)

    # reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5) Second LayerNorm, out_gate and final projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(out,
                            normalized_shape=(hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias)

    out_norm = out_norm * out_gate
    output = F.linear(out_norm, to_out_weight)   # back to [B,N,N,dim]

    return output
