    left_ptr, right_ptr, out_ptr,
    B, N, H,

    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    # Determine b, i_block, j_block, h_block
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    num_h = tl.cdiv(H, BLOCK_H)

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # compute indices
    # grid = (B * H * ceil(N/BLOCK_M) * ceil(N/BLOCK_N),)
    # Let's compute per program:
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    # total number of hidden slices = H
    total_per_batch = num_i * num_j * H
    # batch index:
    b = pid // total_per_batch
    pid_in_batch = pid % total_per_batch
    i_block = pid_in_batch // (num_j * H)
    pid_i_j = pid_in_batch % (num_j * H)
    j_block = pid_i_j // H
    h_idx = pid_i_j % H

    # to support B>1
    # i0, j0
    i0 = i_block * BLOCK_M
    j0 = j_block * BLOCK_N
    # range for k
    # offsets
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers for left block: left[b,i,k,h]
    # left base offset: b*stride_left_b + h_idx*stride_left_h
    # then add i*stride_left_i + k*stride_left_k
    a_ptrs = left_ptr + \
        b * stride_left_b + \
        h_idx * stride_left_h + \
        offs_i[:, None] * stride_left_i + \
        offs_k[None, :] * stride_left_k

    # pointers for right block: right[b,j,k,h] but we need layout (k,j) for dot: r[f] shape (k,j)
    # base: b*stride_right_b + h_idx * stride_right_h
    # then add j*stride_right_j + k*stride_right_k
    b_ptrs = right_ptr + \
        b * stride_right_b + \
        h_idx * stride_right_h + \
        offs_k[:, None] * stride_right_k + \
        offs_j[None, :] * stride_right_j

    # accumulator:
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # load tiles
        a = tl.load(a_ptrs, mask=mask_i[:, None] & (offs_k < N), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_j[None, :] & (offs_k < N), other=0.0).to(tl.float32)
        # compute dot
        acc += tl.dot(a, b)
        # advance pointers 
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # store result to out: shape out[b,i,j,h]
    # compute out pointer
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        h_idx * stride_out_h + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j

    # store
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

    num_j = tl.cdiv(N, BLOCK_N)

    # total programs per batch: num_i * num_j * H
    programs_per_batch = num_i * num_j * H
    b = pid // programs_per_batch
    pid_in_batch = pid % programs_per_batch
    i_block = pid_in_batch // (num_j * H)
    pid_i_j = pid_in_batch % (num_j * H)
    j_block = pid_i_j // H
    h_idx = pid_i_j % H

    # compute offsets
    i0 = i_block * BLOCK_M
    j0 = j_block * BLOCK_N

    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    left_base = left_ptr + b * stride_left_b + h_idx * stride_left_h + i0 * stride_left_i
    right_base = right_ptr + b * stride_right_b + h_idx * stride_right_h + j0 * stride_right_j

"""
TriMul (outgoing) – fast Triton implementation
------------------------------------------------
The module applies a LayerNorm → 4 linear projections (left/right + 3 gates)
followed by a dense pairwise reduction:

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where `left` and `right` have shape [B, N, N, H] (H = hidden_dim) and the
summation runs over the sequence dimension `k`.  The reduction is a batched
matrix‑multiply for each hidden slice, i.e. `out[:, :, :, h] = left[..., h] @
right[..., h].T`.  The heavy part is executed in a custom Triton kernel that
computes one (i‑tile, j‑tile) block for a single hidden slice at a time, thus
reusing registers efficiently.

The kernel works on FP16 inputs (left/right) and accumulates in FP32,
producing an FP32 output tensor.  All remaining element‑wise operations
(LayerNorm, gating, final linear projection) are performed with PyTorch
functions.

The implementation supports:
* B ∈ {1,2}, N up to 1024, hidden_dim = 128, dim ∈ {128,384}
* Optional binary mask (applied before the gates)
* Both normal and heavy‑tailed input distributions
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: computes a single (i‑tile, j‑tile) block of the
# pairwise reduction for a *specific* hidden index `h`.
# -------------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # problem sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # -----------------------------------------------------------------
    # Program ID layout:
    #   pid  →  (batch, i_tile, j_tile, hidden_index)
    # -----------------------------------------------------------------
    pid = tl.program_id(0)

    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)

    programs_per_batch = num_i * num_j * H
    b = pid // programs_per_batch
    pid_in_batch = pid % programs_per_batch

    i_tile = pid_in_batch // (num_j * H)
    tmp = pid_in_batch % (num_j * H)
    j_tile = tmp // H
    h_idx = tmp % H

    # ---------------------------------------------------------------
    # Tile offsets
    # ---------------------------------------------------------------
    i0 = i_tile * BLOCK_M
    j0 = j_tile * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ---------------------------------------------------------------
    # Base pointers (pointing at start of the selected hidden slice)
    # ---------------------------------------------------------------
    left_base  = left_ptr  + b * stride_left_b  + h_idx * stride_left_h
    right_base = right_ptr + b * stride_right_b + h_idx * stride_right_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for k_tile in range(0, num_k_tiles):
        k_offset = k_tile * BLOCK_K
        mask_k = (k_offset + offs_k) < N

        # left[i,k]   → shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base \
                 + offs_i[:, None] * stride_left_i \
                 + (k_offset + offs_k)[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[k,j]   (note the transposed layout) → shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base \
                 + (k_offset + offs_k)[:, None] * stride_right_k \
                 + offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs,
                    mask=mask_j[None, :] & mask_k[:, None],
                    other=0.0).to(tl.float32)

        # Accumulate the matrix multiply for this hidden slice
        acc += tl.dot(a, b)

    # ---------------------------------------------------------------
    # Store the result: out[b,i,j,h]
    # ---------------------------------------------------------------
    out_ptrs = out_ptr \
               + b * stride_out_b \
               + h_idx * stride_out_h \
               + offs_i[:, None] * stride_out_i \
               + offs_j[None, :] * stride_out_j

    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------------------
# Public entry point – matches the signature required by the evaluator.
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) block.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of torch.Tensors containing all linear / LN weights
        - config       : dict with keys "dim", "hidden_dim", "nomask", ...

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # LayerNorm on the input (over last dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # Linear projections (no bias)
    # -----------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # Optional mask (broadcast over hidden dimension)
    # -----------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        # ensure same dtype as projected tensors
        mask_f = mask.to(left.dtype).unsqueeze(-1)          # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # Gating values (sigmoid)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Cast to half for the heavy reduction
    # -----------------------------------------------------------------
    left  = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    # -----------------------------------------------------------------
    # Allocate output buffer for the reduction (float32)
    # -----------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # -----------------------------------------------------------------
    # Compute strides (in elements, not bytes)
    # -----------------------------------------------------------------
    ls = left.stride()   # (B, i, k, h)
    rs = right.stride()
    os = out.stride()

    # Triton expects 8‑bit integers for dimensions, 64‑bit for strides.
    # Pass them as plain Python integers.
    # left/right strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h   = ls
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = rs
    # out strides
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = os

    # -----------------------------------------------------------------
    # Tiling parameters (tuned for H100, FP16 matmul)
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = B * num_i * num_j * hidden_dim

    grid = (total_programs,)

    # -----------------------------------------------------------------
    # Launch the Triton kernel
    # -----------------------------------------------------------------
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # Post‑processing on the GPU (still in FP32)
    # -----------------------------------------------------------------
    # LayerNorm on hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply output gate (make sure dtype matches)
    out_gate = out_gate.to(torch.float32)
    out = out * out_gate

    # Final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out
