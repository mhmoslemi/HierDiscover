@triton.jit
def triloc_kernel(left_ptr, right_ptr, out_ptr,
                  B, N, H,
                  stride_b, stride_i, stride_k, stride_d,
                  BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    b_index = pid_z // H
    d = pid_z % H

    # Compute offsets for batch and channel
    # Compute start coordinates
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask out-of-bounds
    mask_i = i_offset < N
    mask_j = j_offset < N

    # Strides
    # left ptr base for given b,d:
    left_base = left_ptr + b_index * stride_b + d * stride_d
    right_base = right_ptr + b_index * stride_b + d * stride_d
    out_base = out_ptr + b_index * stride_b + d * stride_d

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k_offset
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask k within bounds
        mask_k = k_offset < N

        # compute pointers for left tile: left[i, k]
        left_ptrs = left_base + (i_offset[:, None] * stride_i) + (k_offset[None, :] * stride_k)
        # shape (BLOCK_M, BLOCK_K)
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile: we need shape (BLOCK_K, BLOCK_N) with k rows, j cols
        # thus pointer right_base + (k_offset[:, None] * stride_k) + (j_offset[None, :] * stride_i)
        right_ptrs = right_base + (k_offset[:, None] * stride_k) + (j_offset[None, :] * stride_i)
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # compute multiplication and sum
        acc += tl.dot(left_tile, right_tile)

        # increment k offsets automatically by loop increments

    # store result
    out_ptrs = out_base + (i_offset[:, None] * stride_i) + (j_offset[None, :] * stride_k)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

@triton.jit
def tri_mul_contract_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_k, stride_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):

    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    b = pid_z // H
    d = pid_z % H

    # start offsets
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for tile within bounds
    mask_i = i < N
    mask_j = j < N

    # base pointers
    left_base = left_ptr + b * stride_b + d * stride_d
    right_base = right_ptr + b * stride_b + d * stride_d
    out_base = out_ptr + b * stride_b + d * stride_d

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # left tile (i,k)
        left_ptrs = left_base + i[:, None] * stride_i + k_offset[None, :] * stride_k
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile (k,j) - note dimensions swapped
        right_ptrs = right_base + k_offset[:, None] * stride_k + j[None, :] * stride_i
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # multiply-accumulate
        acc += tl.dot(left_tile, right_tile)

    # store result
    out_ptrs = out_base + i[:, None] * stride_i + j[None, :] * stride_k
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """TriMul outgoing implementation. Computes an optimized contraction using Triton.

    The kernel computes out[..., i, j, d] = sum_k left[..., i, k, d] * right[..., j, k, d],
    where left/right are linear projections of the input with gating and optional masking.
    The heavy N^3 reduction is performed by a batched GEMM kernel written in Triton,
    while the surrounding layernorms, linear layers, and sigmoids are done with PyTorch.
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    # unpack
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # Apply mask if present
    if not nomask:
        # mask shape [B,N,N]
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)  # broadcast to hidden_dim
        left = left * mask
        right = right * mask

    # Gating
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous for Triton
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape   # shape: B, N, N, hidden_dim

    # Allocate output tensor of same dtype and device
    out = torch.empty_like(left)

    # Triton kernel launch
    # Get strides (int64)
    stride_b = left.stride(0)
    stride_i = left.stride(1)
    stride_k = left.stride(2)
    stride_d = left.stride(3)

    # block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             B * hidden_dim )

    # Launch kernel
    tri_mul_contract_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_b, stride_i, stride_k, stride_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection to original dim
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)

    return out

"""
TriMul (outgoing) kernel.

The heavy N³ reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented with a custom Triton GEMM kernel.  The surrounding
operations (layer‑norms, linear projections, gating, final projection)
are performed with PyTorch for simplicity.  The kernel computes a
batched matrix multiplication (left @ rightᵀ) for every hidden channel
`d`.  Each program instance is identified by (tile_i, tile_j, channel_id)
where `channel_id = b * hidden_dim + d`.  The tile sizes (64×64×64)
fit comfortably in H100 registers and make good use of tensor‑cores
when the tensors are stored in FP16 (the kernel loads FP16 and
accumulates in FP32).  The final output is returned in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_contract_kernel(
    left_ptr,          # *const float16
    right_ptr,         # *const float16
    out_ptr,           # *float32
    B, N, H,           # int32 scalar sizes
    stride_b, stride_i, stride_k, stride_d,   # int64 strides for the 4‑D tensors
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute for a specific (batch, hidden) pair:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    using tiled GEMM.
    """
    pid_i = tl.program_id(0)      # tile row   (i)
    pid_j = tl.program_id(1)      # tile col   (j)
    pid_z = tl.program_id(2)      # combined (batch, channel)

    # recover batch and hidden‑channel index
    b = pid_z // H
    d = pid_z % H

    # ----------------------------------------------------------------------
    # Tile coordinates
    # ----------------------------------------------------------------------
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_i = i < N
    mask_j = j < N

    # ----------------------------------------------------------------------
    # Base pointers for the current (b,d) slice
    # ----------------------------------------------------------------------
    left_base  = left_ptr  + b * stride_b + d * stride_d
    right_base = right_ptr + b * stride_b + d * stride_d
    out_base   = out_ptr   + b * stride_b + d * stride_d

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ----------------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)   # [BLOCK_K]
        mask_k = k < N

        # left[i,k,d]  -> shape (BLOCK_M, BLOCK_K)
        left_offs = left_base + i[:, None] * stride_i + k[None, :] * stride_k
        left_tile = tl.load(left_offs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0,
                            dtype=tl.float16)

        # right[j,k,d] as (k,j) for the GEMM (transpose on‑the‑fly)
        # shape (BLOCK_K, BLOCK_N)
        right_offs = right_base + k[:, None] * stride_k + j[None, :] * stride_i
        right_tile = tl.load(right_offs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0,
                             dtype=tl.float16)

        # dot (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)
        # accumulation is performed in FP32
        acc += tl.dot(left_tile, right_tile)

    # ----------------------------------------------------------------------
    # Store the result tile
    # ----------------------------------------------------------------------
    out_offs = out_base + i[:, None] * stride_i + j[None, :] * stride_k
    tl.store(out_offs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)                     # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x, left_proj_w)          # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)        # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3) Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [B,N,N,1] → broadcast over hidden_dim
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear projections of the *normed* input)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # [B,N,N,hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to FP16 for the heavy reduction (tensor‑core friendly)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_fp16.shape

    # Allocate output in FP32 (required for subsequent layer‑norm)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left_fp16.device)

    # ------------------------------------------------------------------
    # 6) Launch Triton kernel: batched GEMM per hidden channel
    # ------------------------------------------------------------------
    stride_b = left_fp16.stride(0)   # B‑stride
    stride_i = left_fp16.stride(1)   # i‑stride (first N)
    stride_k = left_fp16.stride(2)   # k‑stride (second N)
    stride_d = left_fp16.stride(3)   # channel‑stride (H)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             B * hidden_dim )   # one program per (batch, hidden)

    _tri_mul_contract_kernel[grid](
        left_fp16, right_fp16, out,
        B, N, hidden_dim,
        stride_b, stride_i, stride_k, stride_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7) Output normalisation, gating and final projection (all fp32)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=eps)

    out = out * out_gate                     # element‑wise gating

    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)            # [B,N,N,dim]

    return out
