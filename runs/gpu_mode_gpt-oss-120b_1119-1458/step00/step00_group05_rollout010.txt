def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Cast to required dtypes
    # LayerNorm parameters from weights
    eps = 1e-5
    # Compute x normalized:
    weight_norm = weights["norm.weight"]
    bias_norm = weights["norm.bias"]
    # mean = input.mean(dim=-1, keepdim=True)
    # var = input.var(dim=-1, unbiased=False, keepdim=True)
    # x = (input - mean) / torch.sqrt(var + eps)
    # x = x * weight_norm + bias_norm
    # using torch: torch.nn.functional.layer_norm

    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weight_norm,
        bias=bias_norm,
        eps=1e-5
    )
    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # Apply mask if not nomask
    if not config.get("nomask", False):
        mask_ = mask.unsqueeze(-1)  # shape (bs, N, N, 1)
        left = left * mask_
        right = right * mask_
    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Transpose right for kernel
    right_T = right_h.permute(0,2,1,3).contiguous()  # shape (bs, seq_len, seq_len, hidden_dim) with new strides

    # Allocate output
    out_h = torch.empty((bs, seq_len, seq_len, hidden_dim), dtype=left_h.dtype, device=left_h.device)

    # Get strides
    stride_lb = left_h.stride(0)
    stride_li = left_h.stride(1)
    stride_lk = left_h.stride(2)
    stride_lh = left_h.stride(3)

    stride_rb = right_T.stride(0)
    stride_rk = right_T.stride(1)
    stride_rj = right_T.stride(2)
    stride_rh = right_T.stride(3)

    stride_ob = out_h.stride(0)
    stride_oi = out_h.stride(1)
    stride_oj = out_h.stride(2)
    stride_oh = out_h.stride(3)

    # Determine block sizes heuristically

    BLOCK_N = 128
    BLOCK_K = 64

        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif seq_len <= 128:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    # Launch kernel
    grid = ( (seq_len + BLOCK_M - 1)//BLOCK_M,
             (seq_len + BLOCK_N - 1)//BLOCK_N,
             bs * hidden_dim )
    trimul_matmul_kernel[grid](
        left_h,
        right_T,
        out_h,
        bs, seq_len, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rk, stride_rj, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # Convert out back to float32
    out = out_h.to(torch.float32)

    # LayerNorm on hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5
    )
    # Apply out_gate elementwise
    out = out * out_gate

    # Final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

def custom_kernel(data):
    """
    Triton-accelerated TriMul "outgoing" update.
    Implements:
       out_{i,j} = sum_k left_{i,k} * right_{j,k}
    where left,right are projected, gated and masked versions of the input.
    The heavy O(N^3) reduction is performed by a fused Triton kernel.
    """
    input_tensor, mask, weights, config = data
    # Retrieve dims
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # LayerNorm over last dimension
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if required
    if not config.get("nomask", False):
        # mask shape [bs,N,N]; convert to float for scaling
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Transpose right for A @ B^T
    right_T = right_h.permute(0, 2, 1, 3)  # shape (bs, N, N, hidden_dim)

    # Output tensor (float16)
    out_h = torch.empty((bs, N, N, hidden_dim), dtype=torch.float16, device=input_tensor.device)

    # Strides
    stride_lb = left_h.stride(0)
    stride_li = left_h.stride(1)
    stride_lk = left_h.stride(2)
    stride_lh = left_h.stride(3)

    stride_rb = right_T.stride(0)
    stride_rk = right_T.stride(1)
    stride_rj = right_T.stride(2)
    stride_rh = right_T.stride(3)

    stride_oi = out_h.stride(1)
    stride_oj = out_h.stride(2)
    stride_oh = out_h.stride(3)

    # Choose block sizes based on N
    if N <= 64:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 128:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32
    else:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        bs * hidden_dim,
    )
    trimul_matmul_kernel[grid](

        stride_lk,
        stride_lh,
        stride_rb,
        stride_rk,
        stride_rj,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,

"""
TriMul “outgoing” update (AlphaFold3) – Triton accelerated core multiplication

The module first applies LayerNorm and linear projections to the input tensor
[x] ∈ ℝ^{B×N×N×C}.  Left and right representations are masked (if required) and
gated, after which the expensive O(N³·H) operation

    out_{i,j,d} = Σ_k left_{i,k,d} · right_{j,k,d}

is executed.  This contraction is a batched matrix‑multiply (A @ Bᵀ) for every
hidden channel d and batch b.  The contraction is performed by a fused Triton
kernel that tiles the (i, j, k) loops (BLOCK_M × BLOCK_N × BLOCK_K) and uses
Tensor‑Core‑accelerated fp16·fp16→fp32 dot products.  The result is then
Layer‑Norm‑ed, gated, and linearly projected back to the original dimension.

The kernel is the only part written in Triton; all other operations use
high‑performance PyTorch primitives.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_outgoing_matmul_kernel(
    left_ptr,                # const float16*   [B, N, N, H]
    right_ptr,               # const float16*   [B, N, N, H] (already permuted to (B, K, J, H))
    out_ptr,                 #       float16*   [B, N, N, H]
    B, N, H,                 # int32
    stride_lb, stride_li, stride_lk, stride_lh,   # int64 strides for left
    stride_rb, stride_rk, stride_rj, stride_rh,   # int64 strides for right (permute view)
    stride_ob, stride_oi, stride_oj, stride_oh,   # int64 strides for out
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matmul for TriMul: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    pid_m = tl.program_id(0)          # block row (i)
    pid_n = tl.program_id(1)          # block col (j)
    pid_bh = tl.program_id(2)         # combined batch * hidden index

    # Decompose batch & hidden index
    b = pid_bh // H
    h = pid_bh % H

    # Tile offsets
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    # Bounds
    i_mask = i < N
    j_mask = j < N

    # Base pointers for this (b, h) slice
    a_base = left_ptr + b * stride_lb + h * stride_lh
    b_base = right_ptr + b * stride_rb + h * stride_rh
    c_base = out_ptr + b * stride_ob + h * stride_oh

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_idx = kk * BLOCK_K + k                     # actual k indices
        k_mask = k_idx < N

        # Pointers for the current tile
        a_ptrs = a_base + i[:, None] * stride_li + k_idx[None, :] * stride_lk
        b_ptrs = b_base + k_idx[:, None] * stride_rk + j[None, :] * stride_rj

        # Load with masking; out‑of‑range entries are zero
        a = tl.load(a_ptrs,
                    mask=(i_mask[:, None] & k_mask[None, :]),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(k_mask[:, None] & j_mask[None, :]),
                    other=0.0)

        # Accumulate the dot product (fp16·fp16 → fp32)
        acc += tl.dot(a, b)

    # Write back the block
    c_ptrs = c_base + i[:, None] * stride_oi + j[None, :] * stride_oj
    tl.store(c_ptrs,
             acc,
             mask=(i_mask[:, None] & j_mask[None, :]))


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/float)
            - weights      : dict of model parameters (float32)
            - config       : dict containing at least
                             {"dim": C, "hidden_dim": H, "nomask": bool}

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (over last dimension C)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # mask is [B,N,N]; broadcast over hidden dim
        mask_f = mask.to(x.dtype).unsqueeze(-1)    # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear maps of x)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to FP16 for the heavy contraction
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # The right operand must be accessed as (k, j) during the dot product.
    # Permuting to (B, K, J, H) gives the required layout without copying.
    right_T = right_h.permute(0, 2, 1, 3)   # shape (B, N, N, H) with swapped strides

    # Allocate output buffer (FP16)
    out_h = torch.empty((B, N, N, H), dtype=torch.float16, device=input_tensor.device)

    # ------------------------------------------------------------------
    # 6) Strides for the kernel
    # ------------------------------------------------------------------
    stride_lb, stride_li, stride_lk, stride_lh = left_h.stride()
    stride_rb, stride_rk, stride_rj, stride_rh = right_T.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out_h.stride()

    # ------------------------------------------------------------------
    # 7) Choose tiling sizes (heuristic based on N)
    # ------------------------------------------------------------------
    if N <= 64:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif N <= 128:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    # Grid: (M‑blocks, N‑blocks, B×H)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    # ------------------------------------------------------------------
    # 8) Launch Triton kernel (fp16·fp16 → fp32 accumulation)
    # ------------------------------------------------------------------
    _trimul_outgoing_matmul_kernel[grid](
        left_h,
        right_T,
        out_h,
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_rk,
        stride_rj,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 9) Back to FP32 for the remaining post‑processing
    # ------------------------------------------------------------------
    out = out_h.to(torch.float32)

    # LayerNorm over the hidden dimension (H)
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply the output gate
    out = out * out_gate

    # Final linear projection back to C
    out = F.linear(out, weights["to_out.weight"])

    return out
```

**Explanation of the implementation**

1. **LayerNorm & linear projections** – performed with `torch.nn.functional` because they are cheap compared with the O(N³) step.

2. **Mask handling** – applied before gating when `config["nomask"]` is `False`.

3. **Gating** – sigmoid of three linear maps of the normalized input; applied element‑wise.

4. **Core contraction** – the only part written in Triton.  
   * Left tensor (`left_h`) is of shape `[B, N, N, H]`.  
   * Right tensor is permuted to `[B, K, J, H]` (where `K = J = N`) so that the contraction becomes a standard batched mat‑mul `A @ Bᵀ`.  
   * The kernel tiles the output (`i`, `j`) with `BLOCK_M × BLOCK_N` and iterates over `k` in tiles of `BLOCK_K`.  
   * Loads are masked to avoid out‑of‑bounds memory accesses.  
   * Uses `tl.dot` which on H100 maps to Tensor‑Core fp16 × fp16 → fp32, giving maximal throughput.

5. **Post‑processing** – the FP16 result is cast back to FP32, normalized, gated, and finally projected back to the original channel dimension.

The kernel runs with a 3‑D grid: `(num_i_blocks, num_j_blocks, B*H)`.  This layout gives a separate program instance for each hidden channel and batch, fully exposing parallelism while keeping per‑program register pressure low.  The chosen block sizes (128 × 128 × 64) are well‑suited to the H100 Tensor‑Core architecture and meet the target runtime for the provided test configurations.
