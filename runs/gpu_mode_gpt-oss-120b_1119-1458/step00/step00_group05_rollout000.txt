"""
TriMul (outgoing) forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor.
2. Project to a hidden dimension with two linear layers (left/right).
3. Compute three sigmoid gates (left/right/out) from the normalized input.
4. Fuse mask, left‑gate and right‑gate application into a single Triton kernel.
   The kernel writes `left_masked` and `right_masked` tensors:
        left_masked  = left  * left_gate  * mask
        right_masked = right * right_gate * mask
5. Perform the core O(N³) contraction:
        out[b,i,j,d] = Σ_k left_masked[b,i,k,d] * right_masked[b,j,k,d]
   which is a batched matrix multiplication per hidden channel.
   We reuse the highly‑optimized cuBLAS via `torch.matmul` after reshaping:
        (B, D, N, N) @ (B, D, N, N).transpose(-2,-1) → (B, D, N, N)
6. Layer‑norm the hidden result, apply the output gate and a final linear
   projection back to the original hidden dimension.

Only step 4 is implemented in Triton; the rest uses PyTorch which already
leverages the H100’s tensor cores.  The Triton kernel is a simple 1‑D
elementwise fusion that works for any batch size, sequence length and
hidden dimension within the given constraints.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _fused_gate_kernel(
    # pointers
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    # shape
    B, N, D,
    # strides for left/right tensors   (B, N, N, D)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    # strides for mask (B, N, N)
    stride_mask_b, stride_mask_i, stride_mask_k,
    # strides for output tensors
    stride_out_b, stride_out_i, stride_out_k, stride_out_d,
    # block size
    BLOCK_SIZE: tl.constexpr,
):
    """Apply mask and sigmoid gates to left/right projected tensors."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    off = start + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * D
    mask = off < total

    # ---- 1‑D → 4‑D index conversion ----
    # b = off // (N * N * D)
    # rem = off % (N * N * D)
    # i = rem // (N * D)
    # rem2 = rem % (N * D)
    # k = rem2 // D
    # d = rem2 % D
    elems_per_batch = N * N * D
    b = off // elems_per_batch
    rem = off - b * elems_per_batch
    elems_per_i = N * D
    i = rem // elems_per_i
    rem2 = rem - i * elems_per_i
    k = rem2 // D
    d = rem2 - k * D

    # ---- load values ----
    left = tl.load(left_ptr
                   + b * stride_left_b
                   + i * stride_left_i
                   + k * stride_left_k
                   + d * stride_left_d, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr
                        + b * stride_gate_b
                        + i * stride_gate_i
                        + k * stride_gate_k
                        + d * stride_gate_d, mask=mask, other=0.0)
    right = tl.load(right_ptr
                    + b * stride_right_b
                    + i * stride_right_i
                    + k * stride_right_k
                    + d * stride_right_d, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr
                         + b * stride_gate_b
                         + i * stride_gate_i
                         + k * stride_gate_k
                         + d * stride_gate_d, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr
                       + b * stride_mask_b
                       + i * stride_mask_i
                       + k * stride_mask_k, mask=mask, other=0.0)

    # ---- fused computation ----
    out_left = left * left_gate * mask_val
    out_right = right * right_gate * mask_val

    # ---- write results ----
    tl.store(out_left_ptr
             + b * stride_out_b
             + i * stride_out_i
             + k * stride_out_k
             + d * stride_out_d, out_left, mask=mask)
    tl.store(out_right_ptr
             + b * stride_out_b
             + i * stride_out_i
             + k * stride_out_k
             + d * stride_out_d, out_right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a fused Triton kernel.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor shape [B, N, N, dim]
        - mask         : torch.Tensor shape [B, N, N] (may be all‑ones)
        - weights      : dict of model weights (see skeleton)
        - config       : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)  # True → ignore mask (use all‑ones)

    # ------------------------------------------------------------------
    # LayerNorm on the input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B,N,N,hidden_dim]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # Prepare mask (float32, shape [B, N, N, 1])
    # ------------------------------------------------------------------
    if mask is None or nomask:
        mask_f = torch.ones_like(input_tensor[..., 0], dtype=torch.float32, device=device)
    else:
        mask_f = mask.to(torch.float32)

    # ------------------------------------------------------------------
    # Fuse mask + gate application via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # Allocate output buffers
    left_masked  = torch.empty_like(left, device=device, dtype=dtype)
    right_masked = torch.empty_like(right, device=device, dtype=dtype)

    # Compute strides (in elements, not bytes)
    # left/right tensors: (B, N, N, D)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_d = left_gate.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_d = right.stride()
    # mask: (B, N, N)
    stride_mask_b, stride_mask_i, stride_mask_k = mask_f.stride()
    # output tensors: same layout as left/right
    stride_out_b, stride_out_i, stride_out_k, stride_out_d = left_masked.stride()

    total_elements = B * N * N * hidden_dim
    BLOCK_SIZE = 1024  # each program processes 1024 elements
    grid = ( (total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE, )

    _fused_gate_kernel[grid](
        left_ptr=left,
        left_gate_ptr=left_gate,
        right_ptr=right,
        right_gate_ptr=right_gate,
        mask_ptr=mask_f,
        out_left_ptr=left_masked,
        out_right_ptr=right_masked,
        B=B,
        N=N,
        D=hidden_dim,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_d=stride_left_d,
        stride_gate_b=stride_gate_b,
        stride_gate_i=stride_gate_i,
        stride_gate_k=stride_gate_k,
        stride_gate_d=stride_gate_d,
        stride_right_b=stride_right_b,
        stride_right_i=stride_right_i,
        stride_right_k=stride_right_k,
        stride_right_d=stride_right_d,
        stride_mask_b=stride_mask_b,
        stride_mask_i=stride_mask_i,
        stride_mask_k=stride_mask_k,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_k=stride_out_k,
        stride_out_d=stride_out_d,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    # Synchronize to make sure kernel finished before next ops
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # Core contraction: batched GEMM over the k dimension
    #   out[b,i,j,d] = Σ_k left_masked[b,i,k,d] * right_masked[b,j,k,d]
    # We reshape to (B*D, N, N) and use torch.matmul (cuBLAS).
    # ------------------------------------------------------------------
    # (B, N, N, D) -> (B, D, N, N)
    left_reshape  = left_masked.permute(0, 3, 1, 2)   # [B, D, N, N]
    right_reshape = right_masked.permute(0, 3, 1, 2)  # [B, D, N, N]

    # Perform batched matmul: (B*D, N, N) @ (B*D, N, N)^T
    # torch.matmul automatically broadcasts the batch dimension.
    out_hidden = torch.matmul(left_reshape, right_reshape.transpose(-2, -1))  # [B, D, N, N]

    # Back to original layout
    out_hidden = out_hidden.permute(0, 2, 3, 1).contiguous()  # [B, N, N, D]

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out_hidden_norm = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # [B, N, N, hidden_dim]

    # Apply output gate
    out_gated = out_hidden_norm * out_gate  # broadcasting over hidden dim

    # Final linear projection back to dim
    to_out_weight = weights["to_out.weight"]  # shape (dim, hidden_dim)
    output = F.linear(out_gated, to_out_weight)  # [B, N, N, dim]

    return output
