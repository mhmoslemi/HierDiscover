"""
TriMul “outgoing” implementation (AlphaFold3 style).

Algorithm
---------
1. Layer‑norm over the last channel dimension (dim).
2. Linear projections → left/right (shape [B,N,N,hidden]).
3. Optional mask multiplication (broadcasted on the hidden dim).
4. Linear gates → left_gate/right_gate/out_gate (sigmoid).
5. Apply gates: left = left * left_gate,  right = right * right_gate.
6. Core N³ operation:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply per hidden channel:
       out_{b,d} = left_{b,d} @ right_{b,d}^T .
   Implemented with a custom Triton kernel that performs
   batched GEMM (half precision, FP32 accumulation) over the
   batch*hidden dimension.
7. Layer‑norm over the hidden dimension, multiply by out_gate.
8. Final linear projection back to the original dim.

The heavy N³ work (step 6) is off‑loaded to Triton; all other
operations use PyTorch for simplicity.  The kernel works for any
batch size (1 – 2), sequence length up to 1024 and hidden_dim up to 384.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM kernel
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ab, stride_am, stride_ak,   # A: (batch, M, K)
    stride_bb, stride_bk, stride_bn,   # B: (batch, K, N)
    stride_cb, stride_cm, stride_cn,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C = A @ B   (A,B half, C float32)."""
    pid_b = tl.program_id(0)   # batch index (B * hidden)
    pid_m = tl.program_id(1)   # block row
    pid_n = tl.program_id(2)   # block col

    # ----- compute offsets -----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers for current tile
    a_base = a_ptr + pid_b * stride_ab
    b_base = b_ptr + pid_b * stride_bb
    a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K tiles
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
                    other=0.0)
        acc += tl.dot(a, b)               # FP32 accumulation

        # Move to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write result
    c_base = c_ptr + pid_b * stride_cb
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc,
             mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])

def _batched_gemm(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM where a,b have shape (B*H, M, K) and (B*H, K, N).
    Returns c of shape (B*H, M, N) in float32.
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((batch, M, N), dtype=torch.float32, device=a.device)

    # Tunable block sizes (work well on H100 for N≤1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _batched_gemm_kernel[grid](
        a, b, c,
        M, N, K,
        a.stride(0), a.stride(1), a.stride(2),   # stride_ab, stride_am, stride_ak
        b.stride(0), b.stride(1), b.stride(2),   # stride_bb, stride_bk, stride_bn
        c.stride(0), c.stride(1), c.stride(2),   # stride_cb, stride_cm, stride_cn
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # reasonable for H100
    )
    return c


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [B, N, N, dim] (float32).
    """
    # Unpack inputs ------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Helper to fetch weight tensors (already on the correct device)
    # ------------------------------------------------------------------
    def _w(name):
        return weights[name].to(device)

    # LayerNorm over the *dim* channel
    norm_w = _w("norm.weight")
    norm_b = _w("norm.bias")

    # Linear projections (no bias)
    left_proj_w  = _w("left_proj.weight")      # (hidden_dim, dim)
    right_proj_w = _w("right_proj.weight")     # (hidden_dim, dim)

    # Gates (no bias)
    left_gate_w  = _w("left_gate.weight")
    right_gate_w = _w("right_gate.weight")
    out_gate_w   = _w("out_gate.weight")

    # Output normalization & projection
    to_out_norm_w = _w("to_out_norm.weight")
    to_out_norm_b = _w("to_out_norm.bias")
    to_out_w      = _w("to_out.weight")       # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # 2) Projections
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask shape: [B,N,N] → [B,N,N,1] to broadcast
        mask = mask.unsqueeze(-1).to(dtype)
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core N³ operation via Triton: batched GEMM per hidden channel
    # ------------------------------------------------------------------
    # Convert to [B, hidden, N, N] layout
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) - (i,k)
    # right needs to be transposed on its last two dims: (j,k) -> (k,j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    B, H, N, _ = left_perm.shape
    batch_total = B * H

    # Flatten batch & hidden into a single leading dimension
    a = left_perm.view(batch_total, N, N).to(torch.float16)   # (batch_total, M, K)
    b = right_perm.view(batch_total, N, N).to(torch.float16)  # (batch_total, K, N)

    # Triton batched matmul (FP16 inputs, FP32 accumulation)
    out_flat = _batched_gemm(a, b)                         # (batch_total, N, N), float32

    # Reshape back to [B, H, N, N]
    out_hidden = out_flat.view(B, H, N, N)

    # Permute back to original layout [B, N, N, hidden]
    out = out_hidden.permute(0, 2, 3, 1).contiguous()      # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 6) Output layer‑norm + out_gate
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7) Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, dim)

    return out
