@triton.jit
def trimul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for boundaries
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch_ptr = a_ptr + batch_id * stride_a_batch
    b_batch_ptr = b_ptr + batch_id * stride_b_batch
    c_batch_ptr = c_ptr + batch_id * stride_c_batch

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load a tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_a_m + cur_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load b tile: shape (BLOCK_K, BLOCK_N) (we treat B as transposed)
        b_ptrs = b_batch_ptr + cur_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # accumulate
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass implemented with Triton for the heavy
pairwise reduction. The algorithm follows the reference PyTorch
implementation:

1. Layer‑norm the input (dim‑wise).
2. Linear projections → left / right tensors and three gating tensors.
3. Optional binary mask applied to left / right before gating.
4. Element‑wise gating (sigmoid) applied to left and right.
5. Core operation: for each hidden channel d,
   out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d].
   This is a batched GEMM (left @ rightᵀ) and is executed by a custom
   Triton kernel for maximal occupancy on H100.
6. Layer‑norm over the hidden dimension, multiply by the out‑gate, and
   a final linear projection back to the original feature dimension.

All operations stay on the GPU; only the O(N³) reduction is off‑loaded to
Triton. The kernel works for arbitrary batch size (1‑2), sequence length
(≤ 1024) and hidden dimension (typically 128). The output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul einsum
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out = a @ bᵀ with a,b,c of shape (batch, M, K), (batch, N, K), (batch, M, N)."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets in the M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # Accumulator in FP32 (regardless of input dtype)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction (K) dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (M×K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + cur_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile transposed (K×N)
        b_ptrs = b_batch + cur_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back the result (store as same dtype as C)
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])


def _trimul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for the TriMul operation.
    Arguments:
        left  – Tensor of shape (B, H, N, N)   (float32)
        right – Tensor of shape (B, H, N, N)   (float32)
    Returns:
        out – Tensor of shape (B, H, N, N)    (float32)
    """
    B, H, N, _ = left.shape
    batch = B * H

    # Flatten the leading two dimensions so that we can treat each channel as a separate batch
    left_flat = left.reshape(batch, N, N).contiguous()
    right_flat = right.reshape(batch, N, N).contiguous()
    out_flat = torch.empty_like(left_flat)

    # Strides (in elements, not bytes)
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    # For B we need stride for the K (column) and N (row) dimensions after the implicit transpose
    stride_b_batch = right_flat.stride(0)
    stride_b_k = right_flat.stride(2)  # original column stride
    stride_b_n = right_flat.stride(1)  # original row stride

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # Block sizes – 64 works well on H100 for FP32 GEMM
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (batch,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _trimul_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,  # tuned for H100
        num_stages=3
    )

    # Reshape back to (B, H, N, N)
    out = out_flat.reshape(B, H, N, N)
    return out


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for the
    O(N³) reduction.

    Args:
        data: Tuple containing
            - input_tensor (torch.Tensor): [B, N, N, dim]
            - mask (torch.Tensor or None): [B, N, N] (binary, optional)
            - weights (dict): model weights
            - config (dict): must contain "dim" and "hidden_dim"

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm over the input feature dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and gating projections
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on the channel dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).type_as(x)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Apply gating
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Rearrange to (B, hidden_dim, N, N) for the Triton kernel
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    # ------------------------------------------------------------------
    # 6. Core TriMul reduction using Triton (left @ rightᵀ per channel)
    # ------------------------------------------------------------------
    out_hidden = _trimul_triton(left, right)        # [B, H, N, N]

    # Back to the original layout
    out_hidden = out_hidden.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7. Post‑processing: second LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # broadcast over hidden dimension
    out = F.linear(out, weights["to_out.weight"])    # [B, N, N, dim]

    return out
