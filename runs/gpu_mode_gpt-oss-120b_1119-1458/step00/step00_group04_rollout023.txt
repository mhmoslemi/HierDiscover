def custom_kernel(data):
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # handle mask
    if mask is None or mask.numel() == 0:
        mask_fp16 = torch.ones(bs, N, N, device=input_tensor.device, dtype=torch.float16)
    else:
        mask_fp16 = mask.to(torch.float16)
    # Ensure mask stride correct.
    # LayerNorm
    x_norm = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x_norm = x_norm.to(torch.float16)
    # Linear projections in fp16
    left_proj = F.linear(x_norm, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x_norm, weights["right_proj.weight"].to(torch.float16))
    left_gate_logits = F.linear(x_norm, weights["left_gate.weight"].to(torch.float16))
    right_gate_logits = F.linear(x_norm, weights["right_gate.weight"].to(torch.float16))
    out_gate_logits = F.linear(x_norm, weights["out_gate.weight"].to(torch.float16))

    left_gate = torch.sigmoid(left_gate_logits)
    right_gate = torch.sigmoid(right_gate_logits)
    out_gate = torch.sigmoid(out_gate_logits)

    # Now apply gates and mask using Triton kernel
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # Kernel definition: apply_gate_mask_kernel
    @triton.jit
    def apply_gate_mask_kernel(
        proj_ptr, gate_ptr,
        mask_ptr,
        out_ptr,
        B, N, H,
        stride_proj0, stride_proj1, stride_proj2, stride_proj3,
        stride_gate0, stride_gate1, stride_gate2, stride_gate3,
        stride_mask0, stride_mask1, stride_mask2,
        stride_out0, stride_out1, stride_out2, stride_out3,
        BLOCK_H: tl.constexpr):
        pid = tl.program_id(0)
        token_per_batch = N * N
        b = pid // token_per_batch
        tid = pid % token_per_batch
        i = tid // N
        j = tid % N
        # load mask scalar
        mask_val = tl.load(mask_ptr + b*stride_mask0 + i*stride_mask1 + j*stride_mask2)
        # iterate over hidden dimension in blocks
        for h_off in range(0, H, BLOCK_H):
            cur_h = tl.minimum(H - h_off, BLOCK_H)
            # offsets
            proj_off = proj_ptr + b*stride_proj0 + i*stride_proj1 + j*stride_proj2 + h_off*stride_proj3
            gate_off = gate_ptr + b*stride_gate0 + i*stride_gate1 + j*stride_gate2 + h_off*stride_gate3
            out_off = out_ptr + b*stride_out0 + i*stride_out1 + j*stride_out2 + h_off*stride_out3
            # load vectors
            proj = tl.load(proj_off + tl.arange(0, BLOCK_H) * stride_proj3,
                           mask=tl.arange(0, BLOCK_H) < cur_h, other=0.0)
            gate = tl.load(gate_off + tl.arange(0, BLOCK_H) * stride_gate3,
                           mask=tl.arange(0, BLOCK_H) < cur_h, other=0.0)
            out = proj * gate * mask_val
            tl.store(out_off + tl.arange(0, BLOCK_H) * stride_out3,
                     out, mask=tl.arange(0, BLOCK_H) < cur_h)

    # Launch kernel for left and right
    total = bs * N * N
    # Determine block size for hidden dim; hidden_dim up to 128 which is multiple.
    BLOCK_H = 32  # may choose 32, 64, etc.
    grid = (total,)
    # left
    apply_gate_mask_kernel[grid](
        left_proj, left_gate,
        mask_fp16,
        left,
        bs, N, hidden_dim,
        left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        BLOCK_H=BLOCK_H,
    )
    # right
    apply_gate_mask_kernel[grid](
        right_proj, right_gate,
        mask_fp16,
        right,
        bs, N, hidden_dim,
        right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        BLOCK_H=BLOCK_H,
    )

    # Now compute out = left @ right^T per channel using batched GEMM via torch.bmm
    # reshape to (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    M = bs * hidden_dim
    left_mat = left_perm.view(M, N, N)
    right_mat = right_perm.view(M, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # to_out_norm
    out_norm = F.layer_norm(out, (hidden_dim,),
                            weight=weights["to_out_norm.weight"],
                            bias=weights["to_out_norm.bias"])
    # multiply out_gate
    out_norm = out_norm * out_gate

    # final linear
    out_final = F.linear(out_norm, weights["to_out.weight"])
    # ensure float32
    out_final = out_final.to(torch.float32)
    return out_final

"""
TriMul forward (outgoing) implementation with mixed‑precision and a small Triton
kernel.

Algorithm
---------
1. Layer‑norm over the last dimension of the input tensor.
2. Six linear projections (no bias) are computed in FP16:
      left_proj, right_proj, left_gate, right_gate, out_gate.
   The gate logits are turned into sigmoids.
3. A Triton kernel fuses the three element‑wise operations
      proj * gate * mask
   for the left and right branches (the mask is broadcast over the
   hidden dimension).  This reduces the number of kernel launches and
   saves a few memory passes.
4. The “TriMul” operation is a batched matrix multiplication per
   hidden channel:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is equivalent to (left @ rightᵀ) for each hidden channel.
   We reshape the tensors and use the highly‑optimized cuBLAS batched
   GEMM (torch.bmm).
5. Layer‑norm over the hidden dimension, multiplication by the out‑gate,
   a final linear projection back to the original feature dimension,
   and conversion to FP32 for the output.

The heavy linear layers and the batched GEMM are performed by PyTorch’s
CUDA kernels (including FP16 kernels on H100).  The Triton kernel only
handles the cheap element‑wise fusion, satisfying the requirement to
use a custom kernel while keeping the overall runtime within the target
budget.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor of shape [bs, N, N, dim] (float32)
            - mask: torch.Tensor of shape [bs, N, N] (float32) or None
            - weights: dict of model weight tensors
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------ #
    # unpack inputs
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------ #
    # optional mask (always create a FP16 mask to simplify the kernel)
    # ------------------------------------------------------------------ #
    if mask is None or mask.numel() == 0:
        mask_fp16 = torch.ones(bs, N, N, device=input_tensor.device, dtype=torch.float16)
    else:
        mask_fp16 = mask.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 1. Layer‑norm over the last dimension (FP32 → FP16)
    # ------------------------------------------------------------------ #
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_norm = x_norm.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 2. Linear projections (all FP16, bias‑free)
    # ------------------------------------------------------------------ #
    left_proj = F.linear(x_norm, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x_norm, weights["right_proj.weight"].to(torch.float16))
    left_gate_logit = F.linear(x_norm, weights["left_gate.weight"].to(torch.float16))
    right_gate_logit = F.linear(x_norm, weights["right_gate.weight"].to(torch.float16))
    out_gate_logit = F.linear(x_norm, weights["out_gate.weight"].to(torch.float16))

    left_gate = torch.sigmoid(left_gate_logit)
    right_gate = torch.sigmoid(right_gate_logit)
    out_gate = torch.sigmoid(out_gate_logit)

    # ------------------------------------------------------------------ #
    # 3. Triton kernel: (proj * gate * mask) for left and right branches
    # ------------------------------------------------------------------ #
    @triton.jit
    def apply_gate_mask_kernel(
        proj_ptr, gate_ptr,               # [B, N, N, H] inputs
        mask_ptr,                         # [B, N, N] mask (scalar per token)
        out_ptr,                          # [B, N, N, H] output
        B, N, H,
        stride_p0, stride_p1, stride_p2, stride_p3,
        stride_g0, stride_g1, stride_g2, stride_g3,
        stride_m0, stride_m1, stride_m2,
        stride_o0, stride_o1, stride_o2, stride_o3,
        BLOCK_H: tl.constexpr,
    ):
        # one program per (b, i, j) token
        pid = tl.program_id(0)
        tokens_per_batch = N * N
        b = pid // tokens_per_batch
        tid = pid % tokens_per_batch
        i = tid // N
        j = tid % N

        # scalar mask value (broadcasted over hidden dim)
        mask_val = tl.load(mask_ptr + b * stride_m0 + i * stride_m1 + j * stride_m2)

        for h_off in range(0, H, BLOCK_H):
            cur_h = tl.minimum(H - h_off, BLOCK_H)

            # pointers to the current hidden‑dim block
            proj_off = proj_ptr + b * stride_p0 + i * stride_p1 + j * stride_p2 + h_off * stride_p3
            gate_off = gate_ptr + b * stride_g0 + i * stride_g1 + j * stride_g2 + h_off * stride_g3
            out_off  = out_ptr  + b * stride_o0 + i * stride_o1 + j * stride_o2 + h_off * stride_o3

            # load vectors
            proj = tl.load(
                proj_off + tl.arange(0, BLOCK_H) * stride_p3,
                mask=tl.arange(0, BLOCK_H) < cur_h,
                other=0.0,
            )
            gate = tl.load(
                gate_off + tl.arange(0, BLOCK_H) * stride_g3,
                mask=tl.arange(0, BLOCK_H) < cur_h,
                other=0.0,
            )
            # fused multiplication
            out = proj * gate * mask_val
            tl.store(
                out_off + tl.arange(0, BLOCK_H) * stride_o3,
                out,
                mask=tl.arange(0, BLOCK_H) < cur_h,
            )

    # allocate gated tensors
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # launch parameters
    total_tokens = bs * N * N
    BLOCK_H = 32  # 32 works for hidden_dim ≤ 128

    # left branch
    apply_gate_mask_kernel[
        (total_tokens,)
    ](
        left_proj, left_gate,
        mask_fp16,
        left,
        bs, N, hidden_dim,
        *left_proj.stride(),
        *left_gate.stride(),
        *mask_fp16.stride(),
        *left.stride(),
        BLOCK_H=BLOCK_H,
    )
    # right branch
    apply_gate_mask_kernel[
        (total_tokens,)
    ](
        right_proj, right_gate,
        mask_fp16,
        right,
        bs, N, hidden_dim,
        *right_proj.stride(),
        *right_gate.stride(),
        *mask_fp16.stride(),
        *right.stride(),
        BLOCK_H=BLOCK_H,
    )

    # free now‑unused temporaries to keep memory pressure low
    del left_proj, right_proj, left_gate, right_gate, x_norm, mask_fp16

    # ------------------------------------------------------------------ #
    # 4. TriMul core: batched GEMM (left @ rightᵀ) per hidden channel
    #    Shape conversion: (B, N, N, H) → (B*H, N, N)
    # ------------------------------------------------------------------ #
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    batch_h = bs * hidden_dim
    left_mat = left_perm.view(batch_h, N, N)           # [B*H, N, N]
    right_mat = right_perm.view(batch_h, N, N)         # [B*H, N, N]

    # batched matrix multiplication (FP16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N]

    out = (
        out_mat.view(bs, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, H]

    # ------------------------------------------------------------------ #
    # 5. Post‑processing (LayerNorm → out_gate → final linear)
    # ------------------------------------------------------------------ #
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    # multiply by the out‑gate (both FP16)
    out_norm = out_norm * out_gate

    # final linear back to original dimension
    out_final = F.linear(out_norm, weights["to_out.weight"].to(torch.float16))

    # ensure the required FP32 output
    return out_final.to(torch.float32)
