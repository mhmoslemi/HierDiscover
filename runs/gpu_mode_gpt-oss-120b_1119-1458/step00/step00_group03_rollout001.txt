def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    eps = 1e-5

    # all tensors on GPU
    x = input_tensor

    # LayerNorm
    # compute mean and var over last dim
    # x_norm = (x - mean) * inv_std * norm_weight + norm_bias

    # Could use torch.nn.functional.layer_norm to avoid manual compute: torch.nn.functional.layer_norm(x, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=eps). This is simple and fast.

    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # compute linear projections
    # cost: use torch.nn.functional.linear
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # shape [bs,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # mask handling
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1).to(left.dtype)  # mask as float maybe 0/1
        left = left * mask_unsq
        right = right * mask_unsq
    else:
        # if no mask, skip
        pass

    # compute gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # cast left/right to float16 for compute
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # allocate out tensor
    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float32, device=x.device)

    # call kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Determine strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # compute grid
    total_h = bs * hidden_dim
    grid = (total_h,
            (N + BLOCK_M - 1)//BLOCK_M,
            (N + BLOCK_N - 1)//BLOCK_N)

    trimul_outgoing_kernel[grid](
        left, right, out,
        bs, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # now out is [bs,N,N,hidden_dim] in fp32

    # Apply to_out_norm (LayerNorm over hidden_dim)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # multiply out_gate (broadcast)
    out = out * out_gate  # out_gate shape [bs,N,N,hidden_dim]

    # final linear
    output = torch.nn.functional.linear(out, to_out_weight)   # shape [bs,N,N,dim]

    return output

"""
custom_kernel:  Outgoing TriMul implementation (AlphaFold3).
---------------------------------------------------------

The kernel computes the core Einstein‑summation
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
for every batch (b) and hidden channel (d).  This is exactly a batch of
matrix multiplications `L @ Rᵀ` where each hidden dimension is an
independent GEMM.  The Triton kernel tiles the output (i, j) plane with
blocks (BLOCK_M, BLOCK_N) and iterates over the reduction dimension `k`
in BLOCK_K steps.  Inputs are stored in FP16 (to exploit Tensor‑Cores);
accumulation is performed in FP32 and the final result is written as FP32.
All surrounding LayerNorms, linear projections, gating and the final
output projection are handled with PyTorch for simplicity.

The kernel launch grid:
    axis 0 : (batch * hidden_dim)   – one program per (b, d)
    axis 1 : tile rows (i) in the N×N output
    axis 2 : tile columns (j) in the N×N output
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # torch tensor [B, N, N, H] (FP16)
    right_ptr,         # torch tensor [B, N, N, H] (FP16)
    out_ptr,           # torch tensor [B, N, N, H] (FP32)
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # -------------------------------------------------------------
    # Program IDs
    # -------------------------------------------------------------
    pid = tl.program_id(0)                     # (b, h) combined
    batch = pid // H                           # batch index
    h = pid % H                                # hidden‑channel index

    pid_m = tl.program_id(1)                   # tile row of output (i)
    pid_n = tl.program_id(2)                   # tile column of output (j)

    # -------------------------------------------------------------
    # Offsets for the current tile
    # -------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Loop over k dimension in tiles of size BLOCK_K
    # -------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k_tiles):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----------- left block (i, k) ---------------------------------
        # layout: (i, k, h) → strides (stride_left_i, stride_left_k, stride_left_h)
        left_ptrs = (
            left_ptr
            + batch * stride_left_b
            + h * stride_left_h
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----------- right block (k, j) ---------------------------------
        # we need the transposed layout (k, j) because the einsum is L @ Rᵀ
        right_ptrs = (
            right_ptr
            + batch * stride_right_b
            + h * stride_right_h
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----------- accumulate -----------------------------------------
        acc += tl.dot(left, right)   # (M,K)·(K,N) → (M,N)

    # -------------------------------------------------------------
    # Write out the tile
    # -------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + batch * stride_out_b
        + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping of layer‑norm and linear weights
            - config_dict  : contains "dim", "hidden_dim" and optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ----------------- unpack -----------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]                # C   (input/output channel)
    hidden_dim = config["hidden_dim"]  # H   (intermediate channel)
    nomask = config.get("nomask", False)

    # ----------------- LayerNorm (x -> x_norm) -----------------
    # torch.nn.functional.layer_norm expects normalized_shape as a tuple
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ----------------- Linear projections -----------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ----------------- Optional masking -----------------
    if (mask is not None) and (not nomask):
        # mask shape [B,N,N] -> [B,N,N,1] and broadcast
        mask_f = mask.unsqueeze(-1).to(left.dtype)  # FP16/FP32 will be cast later
        left = left * mask_f
        right = right * mask_f

    # ----------------- Gating (sigmoid) -----------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))  # used later

    # Cast gates to the same dtype as the projected tensors
    left_gate = left_gate.to(left.dtype)
    right_gate = right_gate.to(right.dtype)

    left = left * left_gate
    right = right * right_gate

    # Cast to FP16 for the heavy matmul kernel (TensorCore friendly)
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    B, N, _, _ = left.shape

    # ----------------- Triton kernel: L @ Rᵀ per hidden channel -----------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=input_tensor.device)

    # Strides (in element counts, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tile size – 64 works well on H100 for both FP16 inputs and FP32 accumulation.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (batch*hidden_dim, ceil(N/BM), ceil(N/BN))
    total_h = B * hidden_dim
    grid = (
        total_h,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_j,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----------------- to_out_norm + out_gate -----------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate  # out_gate is float32; broadcasting over hidden dim

    # ----------------- final linear projection back to dim -----------------
    output = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return output
```
