def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim = config["dim"]; hidden_dim = config["hidden_dim"]; # keys: dim and hidden_dim.

    # Access weights: they are tensors (maybe float16). We'll treat them as torch.tensors.

    # If mask is None (or config's "nomask"): if config.get('nomask', True), then create mask = torch.ones(..., device) else use provided mask.

    B, N, _, _ = input_tensor.shape

    # Apply layer norm on dim dimension:
    # compute mean and var per ...? Use torch.nn.functional.layer_norm
    norm_weight = weights['norm.weight']; norm_bias = weights['norm.bias']
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections:
    # left_proj_weight shape [hidden_dim, dim] (assuming torch Linear weight shape)
    left = torch.nn.functional.linear(x, weights['left_proj.weight'])
    right = torch.nn.functional.linear(x, weights['right_proj.weight'])

    # Apply mask if not nomask:
    if not config.get('nomask', True):
        mask_unsq = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_unsq
        right = right * mask_unsq

    # Gates
    left_gate = torch.nn.functional.linear(x, weights['left_gate.weight']).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights['right_gate.weight']).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights['out_gate.weight']).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Now compute out via Triton kernel:
    # We will allocate out_tensor = torch.empty_like(left) (same type float32)
    # left and right currently dtype of weight maybe float16/float32; we convert to float16 for speed? Retain dtype.

    # Ensure they are contiguous.
    left = left.contiguous()
    right = right.contiguous()

    # Output shape: [B,N,N,hidden_dim]
    out = torch.empty_like(left, dtype=torch.float32)  # maybe wants float32

    # Determine block sizes
    BLOCK_I = 32  # near 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 8  # hidden per block

    # Compute grid dimensions:
    num_i_blocks = (N + BLOCK_I - 1) // BLOCK_I
    num_j_blocks = (N + BLOCK_J - 1) // BLOCK_J
    num_h_blocks = (hidden_dim + BLOCK_H - 1) // BLOCK_H
    total_hb = B * num_h_blocks

    grid = (num_i_blocks, num_j_blocks, total_hb)

    # left.stride(): (stride_b, stride_i, stride_k, stride_h)
    # Typically shape is (B,N,N,H)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Cast pointers to appropriate type:
    left_ptr = left.data_ptr()
    right_ptr = right.data_ptr()
    out_ptr = out.data_ptr()

        B, N, hidden_dim,

        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H

"""
custom_kernel: optimized forward pass of the TriMul (outgoing) operator.

Algorithm
---------
1. Layer‑norm the input tensor X ∈ ℝ^{B×N×N×C}.
2. Compute left/right linear projections (no bias) → L,R ∈ ℝ^{B×N×N×H}.
3. Apply optional pair‑wise mask (broadcast on the hidden dim).
4. Compute three sigmoid gates (left, right, out) from the normalized X.
5. Fuse gates:  L ← L * left_gate ,   R ← R * right_gate.
6. Core “TriMul”:
   out_{b,i,j,:} = Σ_k  L_{b,i,k,:} ⊙ R_{b,j,k,:}
   (⊙ is element‑wise product over the hidden dimension H).
   This is equivalent to a batch of H independent matrix multiplications:
   out_{b,:, :,h} =  L_{b,:, :,h} @ R_{b,:, :,h}.T .
   The heavy reduction over k is implemented in a Triton kernel that
   tiles the i, j and k dimensions (BLOCK_I×BLOCK_J×BLOCK_K) and
   processes a small block of hidden channels (BLOCK_H) per program.
7. Layer‑norm the result over the hidden dimension, multiply by out_gate,
   and project back to the original feature dimension C.

The Triton kernel computes the reduction a tile at a time, accumulates
in float32, and stores the final result in float32 (the surrounding
PyTorch code can cast back to the original dtype if needed).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel for the O(N³·H) reduction.
# ----------------------------------------------------------------------
@triton.jit
def triton_trimul_kernel(
    left_ptr,            # const * left  [B,N,N,H]
    right_ptr,           # const * right [B,N,N,H]
    out_ptr,             # * out          [B,N,N,H]
    B, N, H,             # problem sizes
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    for a block of i, j and a block of hidden channels h.
    """
    pid_i = tl.program_id(0)          # tile index on the i‑dimension
    pid_j = tl.program_id(1)          # tile index on the j‑dimension
    pid_hb = tl.program_id(2)         # combined batch + hidden‑block index

    # ------------------------------------------------------------------
    # Decode batch index and hidden‑block index.
    # ------------------------------------------------------------------
    num_h_blocks = tl.cdiv(H, BLOCK_H)
    b = pid_hb // num_h_blocks                     # batch index 0 ≤ b < B
    h_block = pid_hb % num_h_blocks                 # hidden‑block index

    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J
    h0 = h_block * BLOCK_H

    # offsets for the current tile (int64)
    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_j = j0 + tl.arange(0, BLOCK_J)
    offs_h = h0 + tl.arange(0, BLOCK_H)

    # bounds checking
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # ------------------------------------------------------------------
    # Accumulator for the tile: shape (BLOCK_I, BLOCK_J, BLOCK_H)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k in tiles.
    # ------------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        cur_k = tl.minimum(N - k_start, BLOCK_K)
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # --------------------------------------------------------------
        # Process each hidden channel inside the block.
        # --------------------------------------------------------------
        for dh in range(BLOCK_H):
            # absolute hidden index for this iteration
            h = h0 + dh
            valid_h = h < H
            if not valid_h:
                continue

            # ----------------------------------------------------------
            # Load left[i, k, h]   → shape (BLOCK_I, cur_k)
            # ----------------------------------------------------------
            left_ptrs = left_ptr + \
                        b * stride_lb + \
                        offs_i[:, None] * stride_li + \
                        offs_k[None, :] * stride_lk + \
                        h * stride_lh
            left = tl.load(left_ptrs,
                           mask=mask_i[:, None] & mask_k[None, :],
                           other=0.0,
                           dtype=tl.float16).to(tl.float32)

            # ----------------------------------------------------------
            # Load right[j, k, h]  → shape (BLOCK_J, cur_k)
            # ----------------------------------------------------------
            right_ptrs = right_ptr + \
                         b * stride_rb + \
                         offs_j[:, None] * stride_rj + \
                         offs_k[None, :] * stride_rk + \
                         h * stride_rh
            right = tl.load(right_ptrs,
                            mask=mask_j[:, None] & mask_k[None, :],
                            other=0.0,
                            dtype=tl.float16).to(tl.float32)

            # ----------------------------------------------------------
            # Multiply‑accumulate over k:   (I,K) @ (J,K)^T → (I,J)
            # ----------------------------------------------------------
            acc[:, :, dh] += tl.dot(left, right, trans_b=True)

    # ------------------------------------------------------------------
    # Write the computed tile back to global memory.
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + \
               b * stride_ob + \
               offs_i[:, None, None] * stride_oi + \
               offs_j[None, :, None] * stride_oj + \
               offs_h[None, None, :] * stride_oh
    store_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
    tl.store(out_ptrs, acc, mask=store_mask)


# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor:      torch.Tensor of shape [B, N, N, C]
            - mask_tensor:       torch.Tensor of shape [B, N, N] (or None)
            - weights_dict:      mapping from weight name to torch Tensor
            - config_dict:       includes "dim", "hidden_dim", and "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dim (C)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (bias=False)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # shape [B, N, N, 1] → broadcast on hidden dim
        mask = mask_tensor.to(dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and gate application
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul (batched matrix multiplication for every hidden dim)
    # ------------------------------------------------------------------
    # Ensure contiguous layout for simple stride extraction
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape  # N is the sequence length (same for both dims)
    # Output tensor of shape [B, N, N, hidden_dim], we compute in float32
    out = torch.empty(B, N, N, hidden_dim, dtype=torch.float32, device=device)

    # Triton block sizes (tuned for H100, can be adjusted)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 8   # must divide hidden_dim reasonably; tail handled by masks

    # Grid dimensions: (i_tiles, j_tiles, batch*hidden_tiles)
    i_blocks = (N + BLOCK_I - 1) // BLOCK_I
    j_blocks = (N + BLOCK_J - 1) // BLOCK_J
    h_blocks = (hidden_dim + BLOCK_H - 1) // BLOCK_H
    grid = (i_blocks, j_blocks, B * h_blocks)

    # Strides (int64)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    triton_trimul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B,
        N,
        hidden_dim,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (over hidden_dim), gate and final linear
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate  # element‑wise multiplication (same shape)

    # Final projection back to original dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
