@triton.jit
def matmul_batched_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BATCH,  # number of batches
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)  # linear id
    # compute batch index, block m, block n
    batch_id = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = pid // tl.cdiv(N, BLOCK_N)
    pid_n = pid % tl.cdiv(N, BLOCK_N)
    # offset for this batch
    a_batch_ptr = a_ptr + batch_id * stride_ab
    b_batch_ptr = b_ptr + batch_id * stride_bb
    c_batch_ptr = c_ptr + batch_id * stride_cb

    # offsets within block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # make sure offsets within bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer offsets for A and B
    a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = b_batch_ptr + (tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K - k*BLOCK_K), other=0.0)

        b_ptrs += BLOCK_K * stride_bk

import torch
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(a_ptr, b_ptr, c_ptr,
                          M, N, K,
                          stride_am, stride_ak, stride_ab,
                          stride_bk, stride_bn, stride_bb,
                          stride_cm, stride_cn, stride_cb,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
                          BATCH: tl.constexpr):
    pid = tl.program_id(0)
    # Compute batch index
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch_idx = pid // (num_blocks_m * num_blocks_n)
    pid = pid % (num_blocks_m * num_blocks_n)
    pid_m = pid // num_blocks_n
    pid_n = pid % num_blocks_n

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for this batch
    a_batch = a_ptr + batch_idx * stride_ab
    b_batch = b_ptr + batch_idx * stride_bb
    c_batch = c_ptr + batch_idx * stride_cb

    # offsets for k loop
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Write result
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def triton_matmul_batch(A, B):
    # A,B shape [batch, M, K], [batch, K, N]; dtype float16
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2
    # output
    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)
    # Get strides in elements
    stride_am = A.stride(1)   # M stride
    stride_ak = A.stride(2)   # K stride
    stride_ab = A.stride(0)   # batch stride
    stride_bk = B.stride(1)   # K stride
    stride_bn = B.stride(2)   # N stride
    stride_bb = B.stride(0)   # batch stride
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)
    stride_cb = C.stride(0)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (batch * ((M + BLOCK_M - 1)//BLOCK_M) * ((N + BLOCK_N - 1)//BLOCK_N),)
    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        BATCH=batch,
        num_warps=4,
    )
    return C

    right_flat = right_h.permute(0,3,1,2).contiguous().view(B*hidden_dim, N, N)
    # transpose right for multiplication
    right_T = right_flat.transpose(1,2).contiguous()

    out_flat = triton_matmul_batch(left_flat, right_T)  # shape [B*hidden, N, N] half
    # reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()
    # out is half; cast to float32 for further layernorm and linear
    out = out.to(torch.float32)
    # LayerNorm on out
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out = out * out_gate
    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out
```

"""
TriMul (outgoing) forward pass implemented with Triton.

The operation works on a 4‑D tensor x of shape [B, N, N, C] (C = dim).
1. Layer‑norm over the last dimension.
2. Linear projections (left/right) → [B, N, N, H] (H = hidden_dim).
3. Optional pairwise mask multiplication.
4. Gated elementwise multiplication with sigmoid‑gates.
5. Core reduction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication per hidden channel:
       out_batch = left_batch @ right_batchᵀ
   This N³ reduction is performed by a custom Triton kernel that
   processes many (batch, i‑block, j‑block) tiles in parallel and
   accumulates in FP32 while using FP16 for the inputs.
6. Layer‑norm over H, multiplication by an output gate and a final
   linear projection back to dim.

Only the N³ reduction is Triton‑accelerated; the surrounding LayerNorms
and linear layers are performed with PyTorch for simplicity.
"""

import torch

import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes (all equal to sequence length)
    stride_am, stride_ak, stride_ab,   # strides for A (batch, M, K)
    stride_bk, stride_bn, stride_bb,   # strides for B (batch, K, N)
    stride_cm, stride_cn, stride_cb,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B  (A: [batch,M,K], B: [batch,K,N])"""
    pid = tl.program_id(0)

    # number of tiles per matrix
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # decode batch / tile indices
    batch_idx = pid // (num_blocks_m * num_blocks_n)
    pid = pid % (num_blocks_m * num_blocks_n)
    pid_m = pid // num_blocks_n
    pid_n = pid % num_blocks_n

    # starting coordinates of the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute base pointers for this batch
    a_batch = a_ptr + batch_idx * stride_ab
    b_batch = b_ptr + batch_idx * stride_bb
    c_batch = c_ptr + batch_idx * stride_cb

    # accumulate in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for the current tile
        a_ptrs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        # load with masking
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # dot product (FP16 inputs → FP32 accumulation)
        acc += tl.dot(a, b)

    # write result (store as FP16 for compactness)
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_fp16(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication for FP16 tensors:
    A: [batch, M, K], B: [batch, K, N] (contiguous)
    Returns C: [batch, M, N] (FP16).
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides (in number of elements, not bytes)
    stride_am = A.stride(1)
    stride_ak = A.stride(2)
    stride_ab = A.stride(0)

    stride_bk = B.stride(1)
    stride_bn = B.stride(2)
    stride_bb = B.stride(0)

    stride_cm = C.stride(1)
    stride_cn = C.stride(2)
    stride_cb = C.stride(0)

    # Tile sizes – chosen to give good occupancy on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch * num_blocks_m * num_blocks_n

    grid = (total_blocks,)

    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_ab,
        stride_bk,
        stride_bn,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # enough warps for H100
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        out tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input layer‑norm
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (float32)

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1] → broadcast
        left = left * mask_exp
        right = right * mask_exp

    # -----------------------------------------------------------------
    # 4. Gating
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    left = left * left_gate
    right = right * right_gate
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 5. Core N³ reduction (batched GEMM) using Triton
    # -----------------------------------------------------------------
    # Cast to fp16 for the heavy reduction
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # B, seq_len, seq_len, hidden

    # reshape to [B*H, N, N] and make right transposed (K,N)
    left_flat = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_T = right_flat.transpose(1, 2).contiguous()   # now shape [B*H, K, N] with K==N

    # Batched matmul: out_flat = left_flat @ right_T   (both fp16)
    out_flat_h = _batched_matmul_fp16(left_flat, right_T)   # [B*H, N, N] fp16

    # Reshape back to [B, N, N, H] (float32 for the following steps)
    out = out_flat_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # -----------------------------------------------------------------
    # 6. Output layer‑norm, gating and final projection
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])   # final [B,N,N,dim]

    return out
