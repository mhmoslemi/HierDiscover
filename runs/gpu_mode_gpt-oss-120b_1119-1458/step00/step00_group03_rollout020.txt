```
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_lb, stride_li, stride_lk,
    stride_rb, stride_rj, stride_rk,
    stride_ob, stride_oi, stride_oj,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # Compute i_tile, j_tile, b = ... using grid split.
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    num_blocks = B * num_i * num_j
    pid_b = pid // (num_i * num_j)
    pid_rem = pid % (num_i * num_j)
    pid_i = pid_rem // num_j
    pid_j = pid_rem % num_j

    # Compute offsets
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers base:
    left_ptrs = left_ptr + pid_b * stride_lb + offs_i[:, None] * stride_li + tl.arange(0, BLOCK_K)[None, :] * stride_lk
    right_ptrs = right_ptr + pid_b * stride_rb + offs_j[:, None] * stride_rj + tl.arange(0, BLOCK_K)[None, :] * stride_rk

    # accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        cur_k = k_offset + tl.arange(0, BLOCK_K)
        # mask k
        mask_k = cur_k < N

        # Load left block: shape (BLOCK_I, BLOCK_K)
        left = tl.load(left_ptrs + cur_k[None, :] * stride_lk, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # Load right block: shape (BLOCK_J, BLOCK_K)
        right = tl.load(right_ptrs + cur_k[None, :] * stride_rk, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # Compute dot product: left @ right.T
        # We want sum_k left[i,k] * right[j,k] => dot of left and right over k dimension (right should be transposed)
        # Using tl.dot: out = tl.dot(left, right, trans_b=True) ??? Actually tl.dot expects (M,K) and (K,N). So we need right transposed.
        # We can provide right (BLOCK_J, BLOCK_K); need to transpose to (BLOCK_K, BLOCK_J). Use tl.dot(left, right, trans_b=True)
        acc += tl.dot(left, right, trans_b=True)

    # Store result
    out_ptrs = out_ptr + pid_b * stride_ob + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

import torch
import triton
import triton.language as tl
import math

@triton.jit
def triton_einsum_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_lb, stride_li, stride_lk,
    stride_rb, stride_rj, stride_rk,
    stride_ob, stride_oi, stride_oj,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute batch, i tile, j tile
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    total_tiles_per_batch = num_i * num_j
    b = pid // total_tiles_per_batch
    rem = pid % total_tiles_per_batch
    i_tile = rem // num_j
    j_tile = rem % num_j

    # compute offsets
    offs_i = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)

    # bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers for left and right slices
    left_ptrs = left_ptr + b * stride_lb + offs_i[:, None] * stride_li + tl.arange(0, BLOCK_K)[None, :] * stride_lk
    right_ptrs = right_ptr + b * stride_rb + offs_j[:, None] * stride_rj + tl.arange(0, BLOCK_K)[None, :] * stride_rk

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        # offset k
        cur_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask for k
        mask_k = cur_k < N

        # Load left block
        left = tl.load(left_ptrs + cur_k[None, :] * stride_lk,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # Load right block
        right = tl.load(right_ptrs + cur_k[None, :] * stride_rk,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)

        # dot product across k dimension; right transposed
        acc += tl.dot(left, right, trans_b=True)

    # Write result
    out_ptrs = out_ptr + b * stride_ob + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

Algorithm
---------
1. Layer‐normalize the input tensor `x` on the last dimension (dim).
2. Apply two linear projections (left/right) and two gating projections,
   then multiply by the (optional) pairwise mask.
3. Fuse the gates:   left  = (x @ W_left)   * sigmoid(x @ W_left_gate) * mask
                     right = (x @ W_right)  * sigmoid(x @ W_right_gate)* mask
4. Compute the pairwise multiplicative update:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This 3‑D contraction is implemented as a batched matrix‑multiply per hidden
   channel using a Triton kernel (float16 inputs, float32 accumulation).
5. Layer‑normalize the result over the hidden dimension, multiply by the
   output gate (sigmoid(x @ W_out_gate)) and apply the final linear projection
   back to the original dimension.
All operations are performed on the GPU; only the O(N³) contraction is
offloaded to Triton for speed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # *float16   [B, N, N] (slice for a single hidden dim)
    right_ptr,         # *float16   [B, N, N]
    out_ptr,           # *float32   [B, N, N]
    B, N,              # scalar dimensions
    stride_lb, stride_li, stride_lk,   # strides for left  (batch, i, k)
    stride_rb, stride_rj, stride_rk,   # strides for right (batch, j, k)
    stride_ob, stride_oi, stride_oj,   # strides for out   (batch, i, j)
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute  out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]  for a single hidden channel.
    The kernel tiles the (i, j) output plane and iterates over k in blocks.
    """
    pid = tl.program_id(0)

    # ---- tile indices -------------------------------------------------
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    tiles_per_batch = num_i * num_j

    b = pid // tiles_per_batch                # batch index
    rem = pid % tiles_per_batch
    i_tile = rem // num_j                     # i‑tile index
    j_tile = rem % num_j                      # j‑tile index

    # ---- offsets within the tiles --------------------------------------
    offs_i = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for the current batch
    base_left  = left_ptr  + b * stride_lb
    base_right = right_ptr + b * stride_rb
    base_out   = out_ptr   + b * stride_ob

    # pointers to the beginnings of rows i and columns j
    ptr_i = base_left  + offs_i[:, None] * stride_li   # (BLOCK_I, 1)
    ptr_j = base_right + offs_j[:, None] * stride_rj   # (BLOCK_J, 1)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # ---- loop over k ---------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k):
        cur_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)          # (BLOCK_K)
        mask_k = cur_k < N

        # Load left[i,k]  -> (BLOCK_I, BLOCK_K)
        left = tl.load(
            ptr_i + cur_k[None, :] * stride_lk,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # Load right[j,k] -> (BLOCK_J, BLOCK_K)
        right = tl.load(
            ptr_j + cur_k[None, :] * stride_rk,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # dot(left, right.T)  => (BLOCK_I, BLOCK_J)
        acc += tl.dot(left, right, trans_b=True)

    # ---- write out -----------------------------------------------------
    out_ptrs = base_out + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul (outgoing) module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
              input_tensor: torch.Tensor [B, N, N, C] (float32)
              mask        : torch.Tensor [B, N, N] (bool/float) or None
              weights     : dict of model weights (see keys in skeleton)
              config      : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # layer‑norm on the input (normalizing over the last dimension)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # gates (sigmoid) – also linear, then activation
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0 and not config.get("nomask", False):
        mask_ = mask.to(input_tensor.dtype).unsqueeze(-1)     # [B,N,N,1]
        left  = left  * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # apply element‑wise gates
    # ------------------------------------------------------------------
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy O(N³) contraction
    # ------------------------------------------------------------------
    left_f16  = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()

    # Output of the contraction (float32)
    out_fp32 = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Triton kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    num_i_tiles = (N + BLOCK_I - 1) // BLOCK_I
    num_j_tiles = (N + BLOCK_J - 1) // BLOCK_J
    grid = (B * num_i_tiles * num_j_tiles,)

    # ------------------------------------------------------------------
    # Loop over hidden channels; each iteration launches one kernel that
    # computes the full (i,j) plane for that channel.
    # ------------------------------------------------------------------
    for h in range(hidden_dim):
        # views that expose a single hidden slice but keep original strides
        left_slice  = left_f16[..., h]   # shape (B, N, N)
        right_slice = right_f16[..., h]  # shape (B, N, N)
        out_slice   = out_fp32[..., h]   # shape (B, N, N)

        # stride tuples (elements, not bytes)
        stride_lb, stride_li, stride_lk = left_slice.stride()
        stride_rb, stride_rj, stride_rk = right_slice.stride()
        stride_ob, stride_oi, stride_oj = out_slice.stride()

        # launch the kernel
        trimul_outgoing_kernel[grid](
            left_slice,
            right_slice,
            out_slice,
            B,
            N,
            stride_lb, stride_li, stride_lk,
            stride_rb, stride_rj, stride_rk,
            stride_ob, stride_oi, stride_oj,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
        )
        # synchronize after each hidden channel to avoid race conditions
        # (optional – can be omitted for performance, Triton ensures
        #   completion before the next kernel uses the same memory)
        torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # Post‑processing: layer‑norm over the hidden dimension, gate, final linear
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]          # (dim, hidden_dim)

    # Layer‑norm across hidden_dim for each (b,i,j) element
    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    # Apply the output gate
    out_norm = out_norm * out_gate

    # Final projection back to the original dimension
    output = F.linear(out_norm, to_out_weight)   # [B, N, N, dim]

    return output
