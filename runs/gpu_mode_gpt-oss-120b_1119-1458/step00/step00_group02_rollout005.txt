def custom_kernel(data):
    """
    Triton-accelerated outgoing TriMul forward pass.
    """
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask_tensor, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Prepare mask
    if mask_tensor is None:
        mask_tensor = torch.ones(bs, N, N, device=input_tensor.device, dtype=input_tensor.dtype)
    else:
        # Ensure float type
        mask_tensor = mask_tensor.to(dtype=input_tensor.dtype)

    # LayerNorm on input (dim)
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # Linear projections (float32)
    left_raw = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_raw = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    left_gate_raw = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate_raw = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()  # (bs,N,N,hidden_dim)

    dtype = torch.float16

    right_raw = right_raw.to(dtype)
    left_gate = left_gate_raw.to(dtype)
    right_gate = right_gate_raw.to(dtype)
    mask_fp = mask_tensor.to(dtype)

    # Allocate outputs
    left = torch.empty_like(left_raw)
    right = torch.empty_like(right_raw)

    # Triton kernel to fuse mask and gates
    total_elements = left_raw.numel()
    BLOCK_SIZE = 16384  # tuned for H100

    @triton.jit
    def mask_gate_kernel(
        left_ptr, right_ptr,
        left_gate_ptr, right_gate_ptr,
        mask_ptr,
        out_left_ptr, out_right_ptr,
        total_elems,
        H,
        BLOCK_SIZE: tl.constexpr
    ):
        pid = tl.program_id(0)
        offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
        mask = offs < total_elems

        # Load values (half)
        left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
        right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
        left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

        tl.store(out_left_ptr + offs, out_left, mask=mask)
        tl.store(out_right_ptr + offs, out_right, mask=mask)

    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)
    mask_gate_kernel[grid](
        left_raw, right_raw,
        left_gate, right_gate,
        mask_fp,
        left, right,
        total_elements,
        hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    left = left.to(torch.float32)
    right = right.to(torch.float32)

TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

This implementation follows the reference PyTorch module but moves the
mask‑and‑gate fusion into a tiny Triton kernel.  The heavy linear
projections and the O(N³) “pairwise sum over k” are kept as regular

2. Linear projections for left/right and their element‑wise gates.
3. Fuse ``mask`` and the two gates with a Triton kernel

4. Compute the cubic contraction

   as a batch of matrix multiplications:

def _mask_gate_fusion_kernel(
    left_ptr,               # float16*
    right_ptr,              # float16*
    left_gate_ptr,          # float16*
    right_gate_ptr,         # float16*
    mask_ptr,               # float16*

    total_elements,         # i64  (scalar)
    hidden_dim,             # i64  (scalar)

    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load data (half precision)
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    # mask is 3‑D (B,N,N); broadcast over the hidden dimension.
    # The linear index of the current element is ``offs``.
    # ``offs // hidden_dim`` gives the corresponding mask element.
    mask_index = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_index, mask=mask, other=tl.float16(0.0))

    out_left = left * left_gate * mask_val
    out_right = right * right_gate * mask_val

    tl.store(out_left_ptr + offs, out_left, mask=mask)
    tl.store(out_right_ptr + offs, out_right, mask=mask)


def custom_kernel(data):
    """
    Triton‑backed forward pass for the outgoing version of TriMul.

    Args:
        data (tuple):
            - input_tensor : torch.Tensor of shape [B, N, N, dim]
            - mask_tensor  : torch.Tensor of shape [B, N, N] (or None)
            - weights       : dict of torch.Tensors containing the model weights
            - config        : dict with keys ``dim``, ``hidden_dim`` and ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ----------------------------------------------------------------------
    # unpack arguments
    # ----------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----------------------------------------------------------------------
    # mask handling (if no mask, treat as all‑ones)
    # ----------------------------------------------------------------------
    if mask_tensor is None:
        mask_tensor = torch.ones(B, N, N, device=device, dtype=dtype)
    else:
        mask_tensor = mask_tensor.to(dtype).contiguous()

    # ----------------------------------------------------------------------
    # 1️⃣ Layer‑Norm over the ``dim`` channel
    # ----------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ----------------------------------------------------------------------
    # 2️⃣ Linear projections and gated values (still in FP32)
    # ----------------------------------------------------------------------
    left_raw   = torch.nn.functional.linear(x, weights["left_proj.weight"])          # (B,N,N,hidden_dim)
    right_raw  = torch.nn.functional.linear(x, weights["right_proj.weight"])
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()   # (B,N,N,hidden_dim)

    # ----------------------------------------------------------------------
    # 3️⃣ Fuse mask & gate with a Triton kernel (use FP16 inside the kernel)
    # ----------------------------------------------------------------------
    # Cast the tensors to FP16 for the point‑wise fusion step
    left_raw   = left_raw.to(torch.float16).contiguous()
    right_raw  = right_raw.to(torch.float16).contiguous()
    left_gate  = left_gate.to(torch.float16).contiguous()
    right_gate = right_gate.to(torch.float16).contiguous()
    mask_half  = mask_tensor.to(torch.float16).contiguous()

    # Allocate fused results
    left_fused  = torch.empty_like(left_raw)
    right_fused = torch.empty_like(right_raw)

    total_elems = left_raw.numel()                     # B * N * N * hidden_dim
    BLOCK_SIZE = 1024                                  # ≤ 1024 threads per program

    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)

    _mask_gate_fusion_kernel[grid](
        left_raw,
        right_raw,
        left_gate,
        right_gate,
        mask_half,
        left_fused,
        right_fused,
        total_elems,
        hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # Back to FP32 for the heavy matrix multiplication
    left_fused  = left_fused.to(torch.float32)
    right_fused = right_fused.to(torch.float32)

    # ----------------------------------------------------------------------
    # 4️⃣ Cubic contraction via batched matrix multiply:
    #     out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #     => out_d = left_d @ right_dᵀ for each hidden channel d
    # ----------------------------------------------------------------------
    # reshape to (B, hidden_dim, N, N) – the channel becomes a batch dimension for matmul
    left_perm  = left_fused.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_fused.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # batched matmul (B, H, N, N) @ (B, H, N, N)ᵀ  → (B, H, N, N)
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))

    # restore original layout (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------------------
    # 5️⃣ Final normalisation, output gate and linear projection back to ``dim``
    # ----------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # apply the output gate (out_gate is still FP32)
    out = out * out_gate

    # final linear back to the original feature dimension
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output
