    B, N, H,

    stride_r_b, stride_r_i, stride_r_k, stride_r_h,

import torch
import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where `h` iterates over hidden dimension.
    This kernel fuses the O(N³·H) reduction using tile‑based
    matrix‑multiply style accumulation.  The hidden dimension H
    is flattened into the third program‑id axis, so each kernel
    instance computes a (N × N) matrix for a single (b,h) pair.
    All loads use half precision (fp16) and accumulation is done
    in fp32 to preserve numerical stability.
    """
    pid_m = tl.program_id(axis=0)  # block row (i)
    pid_n = tl.program_id(axis=1)  # block col (j)
    pid_h = tl.program_id(axis=2)  # flattened batch*hidden

    # Decode batch index and hidden index from pid_h
    b = pid_h // H
    h = pid_h % H

    # Compute the range of row/col indices this block works on
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers for left: left[b, i, k, h]
        left_ptrs = left_ptr + \
            b * stride_l_b + \
            offs_m[:, None] * stride_l_i + \
            offs_k[None, :] * stride_l_k + \
            h * stride_l_h
        # Pointers for right (transposed): right[b, j, k, h] -> we need (k, j)
        right_ptrs = right_ptr + \
            b * stride_r_b + \
            offs_k[:, None] * stride_r_k + \
            offs_n[None, :] * stride_r_i + \
            h * stride_r_h

        # Load tiles (fp16)
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(left, right)

    # Write out the result (fp32)
    out_ptrs = out_ptr + \
        b * stride_o_b + \
        offs_m[:, None] * stride_o_i + \
        offs_n[None, :] * stride_o_j + \
        h * stride_o_h
    tl.store(out_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated implementation of the “outgoing” TriMul operator.
    The heavy O(N³·H) contraction is performed by a custom Triton kernel,
    while the surrounding linear projections, gating and layer‑norms are
    executed with PyTorch primitives.
    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # unpack
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -------------------------------------------------------------------------
    # LayerNorm on the input (over the last dimension `dim`)
    # -------------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B,N,N,dim]

    # -------------------------------------------------------------------------
    # Linear projections and gates (remain in float32 for stability)
    # -------------------------------------------------------------------------
    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # -------------------------------------------------------------------------
    # Apply mask (if any) and element‑wise gates
    # -------------------------------------------------------------------------
    if mask_tensor is None:
        mask = torch.ones_like(x[..., :1])
    else:
        mask = mask_tensor.unsqueeze(-1).type_as(x)  # [B,N,N,1]

    left = (left * mask) * left_gate
    right = (right * mask) * right_gate

    # Ensure contiguous storage for Triton (half‑precision)
    left = left.contiguous().to(torch.float16)
    right = right.contiguous().to(torch.float16)

    B, N, _, _ = left.shape

    # -------------------------------------------------------------------------
    # Triton kernel: out = Σ_k left[...,i,k,h] * right[...,j,k,h]
    # -------------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tuning parameters (adjustable)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )
    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # -------------------------------------------------------------------------
    # Post‑processing: LayerNorm over hidden dimension, gating, final linear
    # -------------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm over hidden_dim (last axis)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to `dim`
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

The expensive O(N³·H) contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is performed by a custom Triton kernel.  All surrounding

The kernel tiles the (i, j) output plane with BLOCK_M×BLOCK_N threads.

flattened dimension B·H.  Half‑precision loads are accumulated in FP32

"""

    # strides for left

    # strides for right
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    # strides for out
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where `h` runs over the hidden dimension.  The hidden dimension is
    flattened into the third program‑id axis (pid_h = b*H + h).
    """
    pid_m = tl.program_id(axis=0)      # block row   → i
    pid_n = tl.program_id(axis=1)      # block col   → j
    pid_h = tl.program_id(axis=2)      # flattened batch*hidden

    # decode batch and hidden indices
    b = pid_h // H
    h = pid_h % H

    # row / column coordinates for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, i, k, h]   → shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            b * stride_l_b + \
            offs_m[:, None] * stride_l_i + \
            offs_k[None, :] * stride_l_k + \
            h * stride_l_h

        # right[b, j, k, h] we need (k, j) ordering for the dot
        # → shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + \
            b * stride_r_b + \
            offs_k[:, None] * stride_r_k + \
            offs_n[None, :] * stride_r_i + \
            h * stride_r_h

        # load tiles (half precision) and cast to FP32 for the dot
        left_tile = tl.load(left_ptrs,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_n[None, :],
                             other=0.0).to(tl.float32)

        # accumulate the matrix product for this k‑tile
        acc += tl.dot(left_tile, right_tile)

    # write the result (still FP32)
    out_ptrs = out_ptr + \
        b * stride_o_b + \
        offs_m[:, None] * stride_o_i + \
        offs_n[None, :] * stride_o_j + \
        h * stride_o_h
    tl.store(out_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑based forward pass for the “outgoing” Triangle Multiplicative
    Update (TriMul).  Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Input LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias) and sigmoid gates
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # Apply mask (if provided) and element‑wise gating
    # ------------------------------------------------------------------
    if mask_tensor is None:
        mask = torch.ones_like(x[..., :1])
    else:
        mask = mask_tensor.unsqueeze(-1).type_as(x)   # [B,N,N,1]

    left  = (left  * mask) * left_gate
    right = (right * mask) * right_gate

    # Make tensors contiguous and half‑precision for the kernel
    left  = left.contiguous().to(torch.float16)
    right = right.contiguous().to(torch.float16)

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # Triton kernel: O(N³·H) contraction
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # strides in element units
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tile sizes – these values work well on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )
    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm over hidden_dim, gating, final linear
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # LayerNorm on the hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # Apply output gate (broadcasted)
    out = out * out_gate

    # Final linear projection back to `dim` (no bias)
    out = torch.nn.functional.linear(out, to_out_weight)

    return out
