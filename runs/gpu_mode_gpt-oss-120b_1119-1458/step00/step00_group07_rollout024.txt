@triton.jit
def trmul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b_l, stride_i_l, stride_k_l, stride_h_l,
    stride_b_r, stride_i_r, stride_k_r, stride_h_r,
    stride_b_o, stride_i_o, stride_j_o, stride_h_o,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, D_TILE: tl.constexpr,
):

    pid = tl.program_id(0)
    # total number of blocks = B * (H // D_TILE) * ceil(N/BLOCK_M) * ceil(N/BLOCK_N)
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    num_blocks_h = tl.cdiv(H, D_TILE)

    # compute indices
    batch_idx = pid // (num_blocks_h * num_blocks_m * num_blocks_n)
    rem = pid % (num_blocks_h * num_blocks_m * num_blocks_n)
    h_idx = rem // (num_blocks_m * num_blocks_n)
    rem2 = rem % (num_blocks_m * num_blocks_n)
    block_i = rem2 // num_blocks_n
    block_j = rem2 % num_blocks_n

    # compute offsets
    i_offset = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    h_offset = h_idx * D_TILE + tl.arange(0, D_TILE)

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, D_TILE), dtype=tl.float32)

    # loop over k
    for k0 in range(0, N, BLOCK_K):
        # compute k block offsets
        k_offset = k0 + tl.arange(0, BLOCK_K)
        # compute mask for out of bounds for k
        k_mask = k_offset < N

        # compute pointer offsets for left: (batch, i, k, d)
        left_ptrs = left_ptr + \
            batch_idx * stride_b_l + \
            i_offset[:, None] * stride_i_l + \
            k_offset[None, :] * stride_k_l + \
            h_offset[None, None, :] * stride_h_l
        # same for right: (batch, j, k, d)
        right_ptrs = right_ptr + \
            batch_idx * stride_b_r + \
            j_offset[:, None] * stride_i_r + \
            k_offset[None, :] * stride_k_r + \
            h_offset[None, None, :] * stride_h_r

        # load left and right values, with mask
        left = tl.load(left_ptrs, mask= (i_offset[:, None] < N) & k_mask[None, :] & (h_offset[None, None, :] < H), other=0.0)
        right = tl.load(right_ptrs, mask= (j_offset[:, None] < N) & k_mask[None, :] & (h_offset[None, None, :] < H), other=0.0)

        # shapes: left (BLOCK_M, BLOCK_K, D_TILE), right (BLOCK_N, BLOCK_K, D_TILE) (need to transpose right on k)
        # but right currently shape (BLOCK_N, BLOCK_K, D_TILE); we need to broadcast across i and j: multiplication across k.

        # We need to multiply left[..., d] and right[..., d] across k and sum.
        # Use tl.dot? We can loop over d dimension group inside.
        for d_idx in range(D_TILE):
            left_slice = left[:, :, d_idx]   # (BLOCK_M, BLOCK_K)
            right_slice = right[:, :, d_idx] # (BLOCK_N, BLOCK_K)
            # right_slice transpose to (BLOCK_K, BLOCK_N)
            # Compute dot: left_slice @ right_slice.T
            acc[:, :, d_idx] += tl.dot(left_slice, right_slice)  # dot performs (M,K) @ (K,N) -> (M,N)

    # store accumulator to out
    out_ptrs = out_ptr + \
        batch_idx * stride_b_o + \
        i_offset[:, None] * stride_i_o + \
        j_offset[None, :] * stride_j_o + \
        h_offset[None, None, :] * stride_h_o

    # mask for storing
    mask_i = i_offset[:, None] < N
    mask_j = j_offset[None, :] < N
    mask_h = h_offset[None, None, :] < H
    mask = mask_i & mask_j & mask_h

    tl.store(out_ptrs, acc, mask=mask)

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_b_l, stride_i_l, stride_k_l, stride_h_l,
    stride_b_r, stride_i_r, stride_k_r, stride_h_r,
    stride_b_o, stride_i_o, stride_j_o, stride_h_o,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, D_BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    # calculate total block counts
    num_m_blocks = tl.cdiv(N, BLOCK_M)
    num_n_blocks = tl.cdiv(N, BLOCK_N)
    num_h_blocks = tl.cdiv(D, D_BLOCK)
    # compute batch, h, i, j
    batch = pid // (num_h_blocks * num_m_blocks * num_n_blocks)
    tmp = pid % (num_h_blocks * num_m_blocks * num_n_blocks)
    h_block = tmp // (num_m_blocks * num_n_blocks)
    tmp2 = tmp % (num_m_blocks * num_n_blocks)
    block_i = tmp2 // num_n_blocks
    block_j = tmp2 % num_n_blocks

    # compute index ranges
    i_range = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_range = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k_range_base = tl.arange(0, BLOCK_K)  # will be offset inside loop

    h_range = h_block * D_BLOCK + tl.arange(0, D_BLOCK)

    # mask for out-of-range
    i_mask = i_range < N
    j_mask = j_range < N
    h_mask = h_range < D

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, D_BLOCK), dtype=tl.float32)

    # loop over K dimension
    for k0 in range(0, N, BLOCK_K):
        k_range = k0 + k_range_base
        k_mask = k_range < N

        # left block load: shape (BLOCK_M, BLOCK_K, D_BLOCK)
        # pointer offsets: batch*stride_b_l + i_range[:,None] * stride_i_l + k_range[None,:] * stride_k_l + h_range[None,None,:] * stride_h_l
        left_ptrs = left_ptr + \
            batch * stride_b_l + \
            i_range[:, None] * stride_i_l + \
            k_range[None, :] * stride_k_l + \
            h_range[None, None, :] * stride_h_l

        left = tl.load(left_ptrs, mask= (i_mask[:, None] & k_mask[None,:] & h_mask[None,None,:]), other=0.0)

        # right block load: we need (k,j) orientation: row = k, col = j
        # pointer offsets: batch*stride_b_r + k_range[:, None] * stride_i_r + j_range[None, :] * stride_k_r + h_range[None, None, :] * stride_h_r
        right_ptrs = right_ptr + \
            batch * stride_b_r + \
            k_range[:, None] * stride_i_r + \
            j_range[None, :] * stride_k_r + \
            h_range[None, None, :] * stride_h_r

        right = tl.load(right_ptrs, mask= (k_mask[:, None] & j_mask[None,:] & h_mask[None,None,:]), other=0.0)

        # Compute dot products per hidden channel slice
        # For each d in D_BLOCK:
        for d_idx in range(D_BLOCK):
            # left slice shape (BLOCK_M, BLOCK_K)
            left_slice = left[:, :, d_idx]
            # right slice shape (BLOCK_K, BLOCK_N)
            right_slice = right[:, :, d_idx]  # shape (BLOCK_K, BLOCK_N)
            # dot
            acc[:, :, d_idx] += tl.dot(left_slice, right_slice)

    # store result
    out_ptrs = out_ptr + \
        batch * stride_b_o + \
        i_range[:, None] * stride_i_o + \
        j_range[None, :] * stride_j_o + \
        h_range[None, None, :] * stride_h_o

    tl.store(out_ptrs, acc, mask= (i_mask[:, None] & j_mask[None, :] & h_mask[None, None, :]))

    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_b_l, stride_i_l, stride_k_l, stride_h_l,
    stride_b_r, stride_i_r, stride_k_r, stride_h_r,
    stride_b_o, stride_i_o, stride_j_o, stride_h_o,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, D_BLOCK: tl.constexpr
):
    """Compute out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]"""

    pid = tl.program_id(0)

    # number of blocks along each dimension
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    num_h = tl.cdiv(D, D_BLOCK)

    # decode pid -> (b, h_block, i_block, j_block)
    b = pid // (num_h * num_m * num_n)
    tmp = pid % (num_h * num_m * num_n)
    h_block = tmp // (num_m * num_n)
    tmp = tmp % (num_m * num_n)
    i_block = tmp // num_n
    j_block = tmp % num_n

    # block start indices
    i_range = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j_range = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    h_range = h_block * D_BLOCK + tl.arange(0, D_BLOCK)

    # boundary masks
    i_mask = i_range < N
    j_mask = j_range < N
    h_mask = h_range < D

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, D_BLOCK), dtype=tl.float32)

    # iterate over k dimension in tiles
    NUM_K_TILES = tl.cdiv(N, BLOCK_K)
    for tk in range(0, NUM_K_TILES):
        k_range = tk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # pointers for left: (i,k,d)
        left_ptrs = left_ptr + \
            b * stride_b_l + \
            i_range[:, None] * stride_i_l + \
            k_range[None, :] * stride_k_l + \
            h_range[None, None, :] * stride_h_l

        left = tl.load(left_ptrs,
                       mask=(i_mask[:, None, None] &
                             k_mask[None, :, None] &
                             h_mask[None, None, :]),
                       other=0.0).to(tl.float32)

        # pointers for right: (k,j,d) – note the swapped axes compared to the original tensor
        right_ptrs = right_ptr + \
            b * stride_b_r + \
            k_range[:, None] * stride_i_r + \
            j_range[None, :] * stride_k_r + \
            h_range[None, None, :] * stride_h_r

        right = tl.load(right_ptrs,
                        mask=(k_mask[:, None, None] &
                              j_mask[None, :, None] &
                              h_mask[None, None, :]),
                        other=0.0).to(tl.float32)

        # compute a small GEMM for every hidden‑dim slice
        for d in range(0, D_BLOCK):
            # left: (M, K), right: (K, N)
            acc[:, :, d] += tl.dot(left[:, :, d], right[:, :, d])

    # write result
    out_ptrs = out_ptr + \
        b * stride_b_o + \
        i_range[:, None] * stride_i_o + \
        j_range[None, :] * stride_j_o + \
        h_range[None, None, :] * stride_h_o

    store_mask = (i_mask[:, None, None] &
                  j_mask[None, :, None] &
                  h_mask[None, None, :])
    tl.store(out_ptrs, acc, mask=store_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul operation.
    The heavy tri‑multiplicative update (∑_k left[b,i,k,d] * right[b,j,k,d])
    is performed in a tiled Triton kernel; the surrounding LayerNorms,
    linear projections and gates are handled with PyTorch primitives.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # LayerNorm on the input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias) and gate logits
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # [B,N,N,hidden_dim]
    left = torch.nn.functional.linear(x, left_proj_weight)
    right = torch.nn.functional.linear(x, right_proj_weight)

    if not nomask:
        # mask: [B,N,N] -> [B,N,N,1]
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_

    # gates (shape [B,N,N,hidden_dim])
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton kernel for the cubic update
    # ------------------------------------------------------------------
    # Cast to fp16 to reduce memory bandwidth; kernel accumulates in fp32
    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)

    out = torch.empty_like(left_half, dtype=torch.float32)

    # strides (element strides, not byte strides)
    l_stride_b, l_stride_i, l_stride_k, l_stride_h = left_half.stride()
    r_stride_b, r_stride_i, r_stride_k, r_stride_h = right_half.stride()
    o_stride_b, o_stride_i, o_stride_j, o_stride_h = out.stride()

    # Tile sizes – empirically good for H100 FP16 matrix‑multiply
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    D_BLOCK = 32   # hidden‑dim is a multiple of 32 for the provided configs

    num_h_blocks = (hidden_dim + D_BLOCK - 1) // D_BLOCK
    num_m_blocks = (seq_len + BLOCK_M - 1) // BLOCK_M
    num_n_blocks = (seq_len + BLOCK_N - 1) // BLOCK_N

    grid = (bs * num_h_blocks * num_m_blocks * num_n_blocks,)

    tri_mul_kernel[grid](
        left_half, right_half, out,
        bs, seq_len, hidden_dim,
        l_stride_b, l_stride_i, l_stride_k, l_stride_h,
        r_stride_b, r_stride_i, r_stride_k, r_stride_h,
        o_stride_b, o_stride_i, o_stride_j, o_stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        D_BLOCK=D_BLOCK,
    )

    # ------------------------------------------------------------------
    # Post‑processing: norm, gate, final projection
    # ------------------------------------------------------------------
    # to_out_norm (LayerNorm over hidden_dim)
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias
    )
    # gate
    out = out * out_gate

    # final linear back to original dim
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

import torch
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel that computes
#   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# for a block‑tile of (i,j) and a small slice of the hidden dim.
# --------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_b_l, stride_i_l, stride_k_l, stride_h_l,
    stride_b_r, stride_i_r, stride_k_r, stride_h_r,
    stride_b_o, stride_i_o, stride_j_o, stride_h_o,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, D_BLOCK: tl.constexpr
):
    """Tiled GEMM for the TriMul update.
    left  : [B, i, k, d]
    right : [B, j, k, d]  (we access it as (k, j) to obtain rightᵀ)
    out   : [B, i, j, d]
    """
    pid = tl.program_id(0)

    # ----------------------------------------------------------
    # 1️⃣  Decode the 1‑D program id into (batch, d‑tile, i‑tile, j‑tile)
    # ----------------------------------------------------------
    num_m = tl.cdiv(N, BLOCK_M)            # i‑tiles per sequence
    num_n = tl.cdiv(N, BLOCK_N)            # j‑tiles per sequence
    num_h = tl.cdiv(D, D_BLOCK)            # hidden‑dim tiles

    b = pid // (num_h * num_m * num_n)
    tmp = pid % (num_h * num_m * num_n)
    h_tile = tmp // (num_m * num_n)
    tmp = tmp % (num_m * num_n)
    i_tile = tmp // num_n
    j_tile = tmp % num_n

    # ----------------------------------------------------------
    # 2️⃣  Starting indices of the current tile
    # ----------------------------------------------------------
    i_idx = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    j_idx = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    h_idx = h_tile * D_BLOCK + tl.arange(0, D_BLOCK)

    i_mask = i_idx < N
    j_mask = j_idx < N
    h_mask = h_idx < D

    # ----------------------------------------------------------
    # 3️⃣  Accumulator (FP32)
    # ----------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, D_BLOCK), dtype=tl.float32)

    # ----------------------------------------------------------
    # 4️⃣  Loop over the reduction dimension k in tiles
    # ----------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tk in range(0, num_k_tiles):
        k_idx = tk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_idx < N

        # ----- left block (i, k, d) ---------------------------------
        left_ptrs = left_ptr + \
            b * stride_b_l + \
            i_idx[:, None] * stride_i_l + \
            k_idx[None, :] * stride_k_l + \
            h_idx[None, None, :] * stride_h_l
        left = tl.load(
            left_ptrs,
            mask=(i_mask[:, None, None] &
                  k_mask[None, :, None] &
                  h_mask[None, None, :]),
            other=0.0).to(tl.float32)        # (M, K, D_BLOCK)

        # ----- right block (k, j, d) – note the swapped axes ----------
        right_ptrs = right_ptr + \
            b * stride_b_r + \
            k_idx[:, None] * stride_k_r + \
            j_idx[None, :] * stride_i_r + \
            h_idx[None, None, :] * stride_h_r
        right = tl.load(
            right_ptrs,
            mask=(k_mask[:, None, None] &
                  j_mask[None, :, None] &
                  h_mask[None, None, :]),
            other=0.0).to(tl.float32)        # (K, N, D_BLOCK)

        # ----- small GEMM for every hidden‑dim slice ----------------
        for d in range(0, D_BLOCK):
            # left : (M, K)   right : (K, N)
            acc[:, :, d] += tl.dot(left[:, :, d], right[:, :, d])

    # ----------------------------------------------------------
    # 5️⃣  Write the result back
    # ----------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_b_o + \
        i_idx[:, None] * stride_i_o + \
        j_idx[None, :] * stride_j_o + \
        h_idx[None, None, :] * stride_h_o
    store_mask = (i_mask[:, None, None] &
                  j_mask[None, :, None] &
                  h_mask[None, None, :])
    tl.store(out_ptrs, acc, mask=store_mask)


# --------------------------------------------------------------
# Public entry point required by the evaluation harness
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    The outer‑product update Σₖ left[i,k,d]·right[j,k,d] is performed
    by the Triton kernel above; the surrounding LayerNorms, linear
    projections and multiplicative gates are handled with PyTorch
    primitives.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣  Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections (no bias) and gate logits
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # [B, N, N, hidden_dim]
    left = torch.nn.functional.linear(x, left_proj_weight)
    right = torch.nn.functional.linear(x, right_proj_weight)

    if not nomask:
        mask_ = mask.unsqueeze(-1).to(x.dtype)        # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # gates (sigmoid)
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3️⃣  Triton kernel for the cubic update
    # ------------------------------------------------------------------
    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)

    out = torch.empty_like(left_half, dtype=torch.float32)  # [B,N,N,hidden_dim]

    # element strides (not byte strides)
    lsb, lsi, lsk, lsh = left_half.stride()
    rsb, rsi, rsk, rsh = right_half.stride()
    osb, osi, osj, osh = out.stride()

    # Tile sizes – tuned for H100 fp16 matmul
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    D_BLOCK = 32                         # hidden_dim is a multiple of 32 in all tests

    n_h_blocks = (hidden_dim + D_BLOCK - 1) // D_BLOCK
    n_m_blocks = (seq_len + BLOCK_M - 1) // BLOCK_M
    n_n_blocks = (seq_len + BLOCK_N - 1) // BLOCK_N

    grid = (bs * n_h_blocks * n_m_blocks * n_n_blocks,)

    tri_mul_kernel[grid](
        left_half, right_half, out,
        bs, seq_len, hidden_dim,
        lsb, lsi, lsk, lsh,
        rsb, rsi, rsk, rsh,
        osb, osi, osj, osh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        D_BLOCK=D_BLOCK,
    )

    # ------------------------------------------------------------------
    # 4️⃣  Post‑processing: hidden‑dim LayerNorm, gating and final linear
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # back to [B,N,N,dim]

    return out
