"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B, N, N, C]).
2. Linear projections → left, right (shape [B, N, N, H]).
3. Optional pairwise mask is applied to left/right.
4. Compute three sigmoidal gates (left_gate, right_gate, out_gate) and
   fuse them with the projected tensors.
5. Core update:   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each hidden channel h.
   It is executed by a custom Triton kernel that processes a tile of the
   (i, j) output plane (BLOCK_M×BLOCK_N) for a given (batch, hidden) pair.
6. Apply a second LayerNorm over the hidden dimension, multiply by out_gate,
   and a final linear projection back to the original channel dimension.

The Triton kernel implements step 5, the rest of the operations are
performed with PyTorch‑functional calls for simplicity and numerical
stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the core einsum:
#    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # problem size & hidden dimension
    B, N, H,
    # strides (in elements, not bytes)
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Triton kernel that computes the outgoing triangle multiplicative update.
    Each program instance (pid) processes a tile of size BLOCK_M × BLOCK_N of
    the output (i, j) plane for a specific (batch, hidden) pair.
    """
    pid_m = tl.program_id(0)      # tile row index (i)
    pid_n = tl.program_id(1)      # tile column index (j)
    pid_z = tl.program_id(2)      # combined batch & hidden index

    # ------------------------------------------------------------------
    # Decode batch and hidden channel from pid_z
    # ------------------------------------------------------------------
    b = pid_z // H                # batch index (0 <= b < B)
    h = pid_z % H                 # hidden channel (0 <= h < H)

    # ------------------------------------------------------------------
    # Compute the starting offsets for the tile in the i and j dimensions
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M, dtype=tl.int64)   # [M]
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N, dtype=tl.int64)   # [N]

    # Boundary masks for i‑ and j‑ coordinates
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers that already include batch and hidden offsets
    left_base  = left_ptr  + b * stride_left_batch  + h
    right_base = right_ptr + b * stride_right_batch + h
    out_base   = out_ptr   + b * stride_out_batch   + h

    # Accumulator for the tile output (float32 for numerical safety)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k in blocks of size BLOCK_K
    # ------------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        k_offsets = k_start + tl.arange(0, BLOCK_K, dtype=tl.int64)   # [K]
        mask_k = k_offsets < N

        # ------------------------------------------------------------------
        # Load a tile of left: shape (BLOCK_M, BLOCK_K)
        #   left[b, i, k, h] where i ∈ offs_m, k ∈ k_offsets
        # ------------------------------------------------------------------
        a_ptr = left_base + \
                offs_m[:, None] * stride_left_i + \
                k_offsets[None, :] * stride_left_k
        a = tl.load(a_ptr,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0)                     # FP32 load

        # ------------------------------------------------------------------
        # Load a tile of right transposed: shape (BLOCK_K, BLOCK_N)
        #   right_T[k, j, h] = right[b, j, k, h]
        # ------------------------------------------------------------------
        b_ptr = right_base + \
                k_offsets[:, None] * stride_right_k + \
                offs_n[None, :] * stride_right_i
        b = tl.load(b_ptr,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0)                     # FP32 load

        # ------------------------------------------------------------------
        # Block‑wise dot product and accumulation
        # ------------------------------------------------------------------
        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # Write the computed tile to the output tensor
    # ------------------------------------------------------------------
    c_ptr = out_base + \
            offs_m[:, None] * stride_out_i + \
            offs_n[None, :] * stride_out_j
    tl.store(c_ptr,
             acc,
             mask=(mask_m[:, None] & mask_n[None, :]))


# ----------------------------------------------------------------------
# Entry point used by the autograder
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the Triangle Multiplicative Update (outgoing) module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] containing the updated
        pair representation.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Helper to move weights to the same device/dtype if needed
    # ------------------------------------------------------------------
    def _to_tensor(t):
        return t.to(device=device, dtype=dtype)

    norm_weight = _to_tensor(weights["norm.weight"])
    norm_bias   = _to_tensor(weights["norm.bias"])

    left_proj_weight  = _to_tensor(weights["left_proj.weight"])
    right_proj_weight = _to_tensor(weights["right_proj.weight"])

    left_gate_weight  = _to_tensor(weights["left_gate.weight"])
    right_gate_weight = _to_tensor(weights["right_gate.weight"])
    out_gate_weight   = _to_tensor(weights["out_gate.weight"])

    to_out_norm_weight = _to_tensor(weights["to_out_norm.weight"])
    to_out_norm_bias   = _to_tensor(weights["to_out_norm.bias"])

    to_out_weight = _to_tensor(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Optional pairwise mask (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if mask_tensor is not None and mask_tensor.numel() > 0:
        # ensure mask dtype matches the projected tensors and add channel dim
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B, N, N, H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))   # [B, N, N, H]

    # Apply gates to the projected tensors
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul update via Triton kernel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # batch, seq_len, hidden
    left  = left.contiguous()
    right = right.contiguous()

    # Prepare the output buffer (zero‑initialized to safely handle padding)
    out = torch.zeros_like(left)                 # [B, N, N, H]

    # Strides in element units (torch returns strides in number of elements)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Triton block sizes – chosen to fit comfortably in registers on H100.
    BLOCK_M = 64   # rows (i dimension)
    BLOCK_N = 64   # cols (j dimension)
    BLOCK_K = 64   # reduction dimension (k)

    # Grid definition: (tiles over i, tiles over j, batch×hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    # Launch kernel
    _trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_batch=s_lb,
        stride_left_i=s_li,
        stride_left_k=s_lk,
        stride_right_batch=s_rb,
        stride_right_i=s_ri,
        stride_right_k=s_rk,
        stride_out_batch=s_ob,
        stride_out_i=s_oi,
        stride_out_j=s_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # ------------------------------------------------------------------
    # 7. Apply the output gate and final linear projection back to dim C
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
