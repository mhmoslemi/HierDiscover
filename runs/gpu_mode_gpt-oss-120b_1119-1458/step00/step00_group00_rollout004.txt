"""
custom_kernel: Triton‑accelerated forward of the “outgoing” TriMul module.

Algorithm
---------
1. Layer‑normalize the input tensor `x` (`[B,N,N,C]`) using a custom Triton
   kernel (no TL.mean – we explicitly compute mean/var).
2. Fuse three linear projections:
      left  = x @ W_left   (C → H)   (no bias)
      right = x @ W_right  (C → H)
   (performed with torch.nn.functional.linear for speed).
3. Optionally mask `left`/`right` with `mask` (broadcasted on the last dim).
4. Compute three gate tensors with a sigmoid:
      left_gate  = σ(x @ W_left_gate)   (C → H)
      right_gate = σ(x @ W_right_gate)  (C → H)
      out_gate   = σ(x @ W_out_gate)    (C → H)
   Apply the gates element‑wise to `left` and `right`.
5. Core multiplicative update:
      out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
   This is a batch of H independent matrix‑multiplications
   (`left_d @ right_dᵀ`).  We reshape and call torch.bmm which
   internally uses cuBLAS (highly optimized on H100).
6. Layer‑normalize the intermediate result with a second Triton kernel.
7. Multiply by `out_gate` and apply the final linear projection
   (`out @ W_to_out.T`) to obtain a tensor of shape `[B,N,N,C]`.

Only the two LayerNorms are written in Triton – the remaining
operations rely on highly‑optimized PyTorch kernels.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton LayerNorm (forward only) – operates on NHWC tensors.
# ----------------------------------------------------------------------
@triton.jit
def layer_norm_fwd_kernel(
    X_ptr, Y_ptr,          # input / output pointers
    W_ptr, B_ptr,          # weight & bias (both size C)
    B, N,                  # batch size and sequence length
    stride_xb, stride_xi, stride_xj, stride_xc,
    stride_yb, stride_yi, stride_yj, stride_yc,
    stride_w, stride_b,
    eps,                   # epsilon for numerical stability
    BLOCK_C: tl.constexpr,  # tile size along channel dim
    C: tl.constexpr         # channel dimension (known at compile time)
):
    pid = tl.program_id(0)                     # one program per (b,i,j)
    total = B * N * N
    if pid >= total:
        return

    # ------------- decode (b,i,j) from linear pid --------------------
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # base pointers for this row (the first channel)
    x_ptr = X_ptr + b * stride_xb + i * stride_xi + j * stride_xj
    y_ptr = Y_ptr + b * stride_yb + i * stride_yi + j * stride_yj

    # ---------- first pass: compute mean and variance -----------------
    sum_val = tl.float32(0.0)
    sum_sq  = tl.float32(0.0)

    for off in range(0, C, BLOCK_C):
        c = off + tl.arange(0, BLOCK_C)
        mask = c < C
        # load a vector of size BLOCK_C (float32)
        x = tl.load(x_ptr + c * stride_xc, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x)
        sum_sq  += tl.sum(x * x)

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ---------- second pass: write normalized result -----------------
    for off in range(0, C, BLOCK_C):
        c = off + tl.arange(0, BLOCK_C)
        mask = c < C
        x = tl.load(x_ptr + c * stride_xc, mask=mask).to(tl.float32)
        w = tl.load(W_ptr + c * stride_w, mask=mask).to(tl.float32)
        b_ = tl.load(B_ptr + c * stride_b, mask=mask).to(tl.float32)

        norm = (x - mean) * inv_std          # (x - μ) / σ
        y = norm * w + b_
        tl.store(y_ptr + c * stride_yc, y, mask=mask)


def layer_norm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5) -> torch.Tensor:
    """
    Applies LayerNorm over the last dimension of ``x`` using Triton.
    ``x`` must be contiguous and have shape [B, N, N, C].
    """
    assert x.ndim == 4
    B, N, N2, C = x.shape
    assert N == N2
    total_rows = B * N * N

    # output tensor
    y = torch.empty_like(x)

    BLOCK_C = 32  # fits comfortably in registers (C ≤ 384)
    layer_norm_fwd_kernel[(total_rows,)](
        x, y,
        weight, bias,
        B, N,
        x.stride(0), x.stride(1), x.stride(2), x.stride(3),
        y.stride(0), y.stride(1), y.stride(2), y.stride(3),
        weight.stride(0), bias.stride(0),
        eps,
        BLOCK_C=BLOCK_C,
        C=C,
    )
    return y

# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.
    Input:
        data = (input_tensor, mask, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1) Layer‑norm on the input (C = dim)
    # --------------------------------------------------------------
    x_norm = layer_norm_triton(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2) Linear projections (no bias) – performed with torch's highly
    #    optimized kernels.
    # --------------------------------------------------------------
    left  = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Optional mask (mask shape: [B,N,N])
    # --------------------------------------------------------------
    if not nomask:
        # ensure mask is float32 and broadcastable
        mask_f = mask.to(dtype=x_norm.dtype).unsqueeze(-1)   # [B,N,N,1]
        left  = left  * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4) Gates (sigmoid) and gating of left/right
    # --------------------------------------------------------------
    left_gate  = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Core multiplicative update (einsum)
    #    Equivalent to a batch of hidden_dim independent N×N matmuls.
    # --------------------------------------------------------------
    B, N, _, H = left.shape   # H == hidden_dim

    # Bring the hidden dimension to leading position for an efficient reshape
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Collapse batch & hidden dim -> treat each (b, h) as a separate batch
    left_flat  = left_perm.view(B * H, N, N)             # [B*H, N, N]
    right_flat = right_perm.view(B * H, N, N)

    # Matrix multiply left_flat @ right_flatᵀ   (N×N @ N×Nᵀ)  -> [B*H, N, N]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout: [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6) Post‑multiplicative LayerNorm (C = hidden_dim)
    # --------------------------------------------------------------
    out_norm = layer_norm_triton(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------
    # 7) Apply output gate and final linear projection back to dim
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate
    output = F.linear(out_norm, weights["to_out.weight"])   # [B,N,N,dim]

    return output
```
