    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_k, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    BLOCK_POS: tl.constexpr, BLOCK_H: tl.constexpr
):
    # program ids
    pid_pos = tl.program_id(0)
    pid_h = tl.program_id(1)

    # define offsets for elements
    offset_m = pid_pos * BLOCK_POS + tl.arange(0, BLOCK_POS)
    # compute mask of valid positions
    total = B * N * N
    mask_pos = offset_m < total

    # compute b,i,k for each offset
    b = offset_m // (N * N)
    rem = offset_m % (N * N)
    i = rem // N
    k = rem % N

    # compute offsets for projection and gate (vectors across H)
    h_offset = pid_h * BLOCK_H
    # vector indices
    h_idx = h_offset + tl.arange(0, BLOCK_H)

    mask_h = h_idx < H

"""
TriMul (outgoing) forward pass with a lightweight Triton fusion.

Algorithm
---------
1. Layer‑norm the input tensor `x` using the supplied
   `norm.weight` / `norm.bias`.
2. Linear projections (`left_proj`, `right_proj`) and gate
   projections (`left_gate`, `right_gate`, `out_gate`) are computed
   with `torch.nn.functional.linear`.  The gate tensors are passed
   through a sigmoid.
3. A Triton kernel fuses the three element‑wise operations
        left  = left_proj  * left_gate  * mask
        right = right_proj * right_gate * mask
   The kernel works on a flat view of the `[B, N, N, H]` tensors,
   processing a block of positions (`BLOCK_POS`) and a block of the
   hidden dimension (`BLOCK_H`) per program.
4. The core “multiplicative update’’ is a batched matrix product:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h] .
   This is carried out efficiently by reshaping the tensors to
   `(B*H, N, N)` and using `torch.bmm` (CuBLAS / Tensor‑Cores).
5. The result is normalised with `to_out_norm`, multiplied by the
   `out_gate` tensor and projected back to the original feature
   dimension with the `to_out` linear layer.

Only the element‑wise fusion (step 3) is performed in Triton, satisfying
the requirement that “part of the operations’’ be kernel‑based while
leveraging highly‑optimised CuBLAS for the cubic part of the
computation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse projection, gate and mask
# ----------------------------------------------------------------------
@triton.jit
def _fuse_proj_gate_mask_kernel(
    proj_ptr,                # [B, N, N, H] left/right projection (float16)
    gate_ptr,                # [B, N, N, H] gate tensor (float16)
    mask_ptr,                # [B, N, N]   mask (float16, 0/1)
    out_ptr,                 # [B, N, N, H] output (float16, can be same as proj_ptr)
    B, N, H,                 # compile‑time constants
    stride_proj_b, stride_proj_i, stride_proj_k, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    BLOCK_POS: tl.constexpr,  # number of (b,i,k) positions processed per program
    BLOCK_H: tl.constexpr,    # hidden‑dim chunk size
):
    # ------------------------------------------------------------------
    # 1D grid over (b,i,k) positions, 2D grid over hidden dimension chunks
    # ------------------------------------------------------------------
    pid_pos = tl.program_id(0)
    pid_h   = tl.program_id(1)

    total_pos = B * N * N                     # total number of (b,i,k) triples
    offs_pos = pid_pos * BLOCK_POS + tl.arange(0, BLOCK_POS)
    mask_pos = offs_pos < total_pos

    # --------------------------------------------------------------
    # Decode (b,i,k) from the flat position index
    # --------------------------------------------------------------
    b = offs_pos // (N * N)
    rem = offs_pos % (N * N)
    i = rem // N
    k = rem % N

    # ------------------------------------------------------------------
    # Base offsets (no hidden dimension yet)
    # ------------------------------------------------------------------
    base_proj = b * stride_proj_b + i * stride_proj_i + k * stride_proj_k
    base_gate = b * stride_gate_b + i * stride_gate_i + k * stride_gate_k
    base_mask = b * stride_mask_b + i * stride_mask_i + k * stride_mask_k

    # ------------------------------------------------------------------
    # Hidden‑dimension chunk
    # ------------------------------------------------------------------
    h_off  = pid_h * BLOCK_H
    h_idx  = h_off + tl.arange(0, BLOCK_H)
    mask_h = h_idx < H

    # combined mask for the 2‑D load/store
    mask = mask_pos[:, None] & mask_h[None, :]

    # ------------------------------------------------------------------
    # Pointers for the current block (shape: [BLOCK_POS, BLOCK_H])
    # ------------------------------------------------------------------
    proj_ptrs = proj_ptr + base_proj[:, None] + h_idx[None, :] * stride_proj_h
    gate_ptrs = gate_ptr + base_gate[:, None] + h_idx[None, :] * stride_gate_h
    out_ptrs  = out_ptr  + base_proj[:, None] + h_idx[None, :] * stride_proj_h

    # ------------------------------------------------------------------
    # Load the scalar mask (one per (b,i,k) position) and broadcast
    # ------------------------------------------------------------------
    mask_scalar = tl.load(mask_ptr + base_mask, mask=mask_pos, other=0.0)
    mask_vec = mask_scalar[:, None]               # shape [BLOCK_POS, 1]

    # ------------------------------------------------------------------
    # Load vectors, fuse, and store
    # ------------------------------------------------------------------
    proj = tl.load(proj_ptrs, mask=mask, other=0.0, dtype=tl.float16)
    gate = tl.load(gate_ptrs, mask=mask, other=0.0, dtype=tl.float16)

    out = proj * gate * mask_vec                   # broadcasting mask_vec over H
    tl.store(out_ptrs, out, mask=mask)


# ----------------------------------------------------------------------
# Entry point: custom_kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing’’ Triangle Multiplicative Update (TriMul).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor [B, N, N, dim] (float32)
        - mask        : torch.Tensor [B, N, N] (float32 or bool) or None
        - weights     : dict of weight tensors
        - config      : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1. unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, config = data
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]          # H

    device = x.device
    eps = 1e-5

    # --------------------------------------------------------------
    # 2. Input LayerNorm
    # --------------------------------------------------------------
    # mean / var over the last dimension (dim)
    mean = x.mean(dim=-1, keepdim=True)
    var  = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # --------------------------------------------------------------
    # 3. Linear projections (float32) → reshape to 4‑D tensors
    # --------------------------------------------------------------
    # shape helpers
    B_, N_, _ = B, N, N
    # flatten the first three axes for the linear call
    x_flat = x_norm.reshape(-1, dim)                      # (B*N*N, dim)

    left_proj  = F.linear(x_flat, weights["left_proj.weight"]).reshape(B, N, N, hidden_dim)
    right_proj = F.linear(x_flat, weights["right_proj.weight"]).reshape(B, N, N, hidden_dim)

    left_gate  = torch.sigmoid(
        F.linear(x_flat, weights["left_gate.weight"]).reshape(B, N, N, hidden_dim)
    )
    right_gate = torch.sigmoid(
        F.linear(x_flat, weights["right_gate.weight"]).reshape(B, N, N, hidden_dim)
    )
    out_gate   = torch.sigmoid(
        F.linear(x_flat, weights["out_gate.weight"]).reshape(B, N, N, hidden_dim)
    )

    # --------------------------------------------------------------
    # 4. Mask handling (broadcasted over hidden dimension)
    # --------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        mask = torch.ones((B, N, N), device=device, dtype=x.dtype)
    # ensure same dtype for the kernel (float16)
    mask_f16 = mask.to(torch.float16)

    # --------------------------------------------------------------
    # 5. Cast projections & gates to float16 for the Triton kernel
    # --------------------------------------------------------------
    left_proj_f  = left_proj.to(torch.float16)
    right_proj_f = right_proj.to(torch.float16)
    left_gate_f  = left_gate.to(torch.float16)
    right_gate_f = right_gate.to(torch.float16)

    # --------------------------------------------------------------
    # 6. Triton fusion: proj * gate * mask   (in‑place)
    # --------------------------------------------------------------
    BLOCK_POS = 128          # positions per program
    BLOCK_H   = 32           # hidden‑dim chunk size

    # strides for the 4‑D tensors (contiguous layout)
    sL_b, sL_i, sL_k, sL_h = left_proj_f.stride()
    sG_b, sG_i, sG_k, sG_h = left_gate_f.stride()
    sM_b, sM_i, sM_k        = mask_f16.stride()

    # total number of (b,i,k) triples
    total_pos = B * N * N
    grid_x = (total_pos + BLOCK_POS - 1) // BLOCK_POS
    grid_y = (hidden_dim + BLOCK_H - 1) // BLOCK_H
    # left side
    _fuse_proj_gate_mask_kernel[
        (grid_x, grid_y)
    ](
        left_proj_f, left_gate_f, mask_f16, left_proj_f,    # out can reuse the proj buffer
        B, N, hidden_dim,
        sL_b, sL_i, sL_k, sL_h,
        sG_b, sG_i, sG_k, sG_h,
        sM_b, sM_i, sM_k,
        BLOCK_POS=BLOCK_POS,
        BLOCK_H=BLOCK_H,
    )
    # right side (reuse same kernel)
    _fuse_proj_gate_mask_kernel[
        (grid_x, grid_y)
    ](
        right_proj_f, right_gate_f, mask_f16, right_proj_f,
        B, N, hidden_dim,
        sL_b, sL_i, sL_k, sL_h,
        sG_b, sG_i, sG_k, sG_h,
        sM_b, sM_i, sM_k,
        BLOCK_POS=BLOCK_POS,
        BLOCK_H=BLOCK_H,
    )
    # After this step:
    #   left_proj_f  = left_proj * left_gate * mask
    #   right_proj_f = right_proj * right_gate * mask

    # --------------------------------------------------------------
    # 7. Core multiplicative update (batched GEMM)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) for torch.bmm
    left_batched  = left_proj_f.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # (B*H, N, N)
    right_batched = right_proj_f.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)  # (B*H, N, N)

    # right has to be transposed on the second axis
    out_batched = torch.bmm(left_batched, right_batched.transpose(1, 2))            # (B*H, N, N)

    # reshape back to [B, N, N, H] and cast to float32 for the following layers
    out = out_batched.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 8. Output LayerNorm (over hidden dimension)
    # --------------------------------------------------------------
    out_mean = out.mean(dim=-1, keepdim=True)
    out_var  = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - out_mean) / torch.sqrt(out_var + eps)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # --------------------------------------------------------------
    # 9. Apply the output gate and final linear projection
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                         # broadcasting over H
    # final linear: (B,N,N,H) -> (B,N,N,dim)
    out_final = F.linear(out_norm.reshape(-1, hidden_dim), weights["to_out.weight"])
    out_final = out_final.view(B, N, N, dim)

    return out_final
```
