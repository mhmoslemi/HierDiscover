def custom_kernel(data):
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # Linear projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Apply mask if needed
    if mask is not None and not config.get("nomask", False):
        # mask shape [B,N,N], unsqueeze dim
        mask_f = mask.to(dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    B, N, _, _ = input_tensor.shape

    # Prepare output tensor
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Precompute strides
    left_stride_b = left.stride(0)
    left_stride_i = left.stride(1)
    left_stride_k = left.stride(2)
    left_stride_h = left.stride(3)

    right_stride_b = right.stride(0)
    right_stride_i = right.stride(1)
    right_stride_k = right.stride(2)
    right_stride_h = right.stride(3)

    out_stride_b = out.stride(0)
    out_stride_i = out.stride(1)
    out_stride_j = out.stride(2)
    out_stride_h = out.stride(3)

    # Choose block sizes
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 32

    # Launch kernel for each batch
    for b_idx in range(B):
        left_ptr = left.data_ptr() + b_idx * left_stride_b * left.element_size()
        right_ptr = right.data_ptr() + b_idx * right_stride_b * right.element_size()
        out_ptr = out.data_ptr() + b_idx * out_stride_b * out.element_size()

        # Compute grid
        grid = (
            triton.cdiv(N, BLOCK_I),
            triton.cdiv(N, BLOCK_J),
            triton.cdiv(hidden_dim, BLOCK_H),
        )
        tri_mul_kernel[grid](
            left_ptr,
            right_ptr,
            out_ptr,
            B, N, hidden_dim,
            left_stride_i, left_stride_k, left_stride_h,
            right_stride_i, right_stride_k, right_stride_h,
            out_stride_i, out_stride_j, out_stride_h,
            BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
        )
    # Continue after kernel:
    # Apply to_out_norm (layernorm over hidden dim for each i,j)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Multiply with out_gate
    out_gate = out_gate  # shape (B,N,N,H)
    out = out * out_gate

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul – “outgoing” version

This implementation follows the reference PyTorch module but moves the expensive
pairwise contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

to a custom Triton kernel.  The kernel treats the hidden dimension `d` as an
extra batch dimension, computes a tiled matrix‑multiply for each hidden block,
and uses broadcasting + tl.sum to accumulate over the summed dimension `k`.

All other operations (LayerNorm, linear projections and gates) are performed
with standard PyTorch kernels.  The final output has the same shape as the
reference implementation: `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched contraction across the sequence dimension.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,           # *float32, shape (B, N, N, H) – left side after gating & mask
    right_ptr,          # *float32, shape (B, N, N, H) – right side after gating & mask
    out_ptr,            # *float32, shape (B, N, N, H) – output buffer
    N,                  # int32  – sequence length
    H,                  # int32  – hidden dimension
    stride_left_i,      # int64  – stride for the second axis (i) of left
    stride_left_k,      # int64  – stride for the third axis (k) of left
    stride_left_h,      # int64  – stride for the fourth axis (d) of left
    stride_right_i,     # int64  – stride for the second axis (i) of right (used as j)
    stride_right_k,     # int64  – stride for the third axis (k) of right
    stride_right_h,     # int64  – stride for the fourth axis (d) of right
    stride_out_i,       # int64  – stride for the second axis (i) of output
    stride_out_j,       # int64  – stride for the third axis (j) of output
    stride_out_h,       # int64  – stride for the fourth axis (d) of output
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute
        out[i, j, h] = Σ_k left[i, k, h] * right[j, k, h]
    for a single batch (the batch offset is added by the host before the call).

    The kernel tiles the i, j and hidden dimensions (h) and iterates over k.
    Broadcasting + tl.sum are used to avoid an explicit inner‑loop over h.
    """

    pid_i = tl.program_id(0)    # block row
    pid_j = tl.program_id(1)    # block column
    pid_h = tl.program_id(2)    # hidden‑block

    # ------------------------------------------------------------------
    # 1) Compute coordinate ranges for this block (might exceed N/H)
    # ------------------------------------------------------------------
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    h_start = pid_h * BLOCK_H

    i_range = i_start + tl.arange(0, BLOCK_I)
    j_range = j_start + tl.arange(0, BLOCK_J)
    h_range = h_start + tl.arange(0, BLOCK_H)

    mask_i = i_range < N
    mask_j = j_range < N
    mask_h = h_range < H

    # accumulator for the block: (BLOCK_I, BLOCK_J, BLOCK_H)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 2) Loop over the summed dimension k in tiles
    # ------------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        k_range = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # ---------- left: shape (I, K, H) ----------
        left_offsets = (
            i_range[:, None, None] * stride_left_i
            + k_range[None, :, None] * stride_left_k
            + h_range[None, None, :] * stride_left_h
        )
        left = tl.load(
            left_ptr + left_offsets,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)                     # (I, K, H)

        # ---------- right: shape (K, J, H) ----------
        right_offsets = (
            k_range[:, None, None] * stride_right_k
            + j_range[None, :, None] * stride_right_i
            + h_range[None, None, :] * stride_right_h
        )
        right = tl.load(
            right_ptr + right_offsets,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)                     # (K, J, H)

        # ---------- broadcast & sum over k ----------
        # left  : (I, K, 1, H)
        # right : (1, K, J, H)
        # product: (I, K, J, H)
        prod = left[:, :, None, :] * right[None, :, :, :]   # (I, K, J, H)
        acc += tl.sum(prod, axis=1)                        # sum over K -> (I, J, H)

    # ------------------------------------------------------------------
    # 3) Store the result
    # ------------------------------------------------------------------
    out_offsets = (
        i_range[:, None, None] * stride_out_i
        + j_range[None, :, None] * stride_out_j
        + h_range[None, None, :] * stride_out_h
    )
    tl.store(
        out_ptr + out_offsets,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )


# ----------------------------------------------------------------------
# Entry point called by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.
    `data` is a tuple (input, mask, weights, config).

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on the same CUDA device).
    """
    # unpack arguments -------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]          # input channel dimension
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) + gating
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 3. Optional mask (mask shape = [B, N, N])
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_f = mask.to(dtype).unsqueeze(-1)            # broadcast on hidden dim
        left = left * mask_f
        right = right * mask_f

    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 4. Contract left/right via Triton kernel
    # ------------------------------------------------------------------
    # Output buffer
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides (in *elements*, not bytes)
    # left/right have layout (B, N, N, H) => strides: (N*N*H, N*H, H, 1)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_h = left.stride(3)

    stride_right_i = right.stride(1)   # will be used as “j” stride
    stride_right_k = right.stride(2)
    stride_right_h = right.stride(3)

    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)

    # Tile sizes – chosen to fit comfortably in registers/shared memory on H100
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 32

    # Launch one kernel per batch (B≤2, so the overhead is negligible)
    for b in range(B):
        # base pointers for this batch
        left_ptr = left.data_ptr() + b * left.stride(0) * left.element_size()
        right_ptr = right.data_ptr() + b * right.stride(0) * right.element_size()
        out_ptr = out.data_ptr() + b * out.stride(0) * out.element_size()

        grid = (
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J,
            (hidden_dim + BLOCK_H - 1) // BLOCK_H,
        )
        tri_mul_kernel[grid](
            left_ptr,
            right_ptr,
            out_ptr,
            N,
            hidden_dim,
            stride_left_i,
            stride_left_k,
            stride_left_h,
            stride_right_i,
            stride_right_k,
            stride_right_h,
            stride_out_i,
            stride_out_j,
            stride_out_h,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_H=BLOCK_H,
        )

    # ------------------------------------------------------------------
    # 5. Post‑processing: LayerNorm over hidden dim, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # element‑wise gating

    # final projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out
