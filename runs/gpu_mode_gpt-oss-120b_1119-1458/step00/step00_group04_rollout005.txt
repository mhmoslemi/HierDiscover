def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_size,
    M, N, K,
    stride_batch_a, stride_m_a, stride_k_a,
    stride_batch_b, stride_k_b, stride_n_b,
    stride_batch_c, stride_m_c, stride_n_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    FP16: tl.constexpr,
):
   batch_id = tl.program_id(0)
   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)

   # compute offsets
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
   offs_k = tl.arange(0, BLOCK_K)

   # Create pointers to A and B with offsets
   a_ptrs = a_ptr + batch_id * stride_batch_a + offs_m[:, None] * stride_m_a + offs_k[None, :] * stride_k_a
   b_ptrs = b_ptr + batch_id * stride_batch_b + offs_k[:, None] * stride_k_b + offs_n[None, :] * stride_n_b

   # init accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs)  # shape BLOCK_M x BLOCK_K
        b = tl.load(b_ptrs)  # shape BLOCK_K x BLOCK_N
        # Use dot
        acc += tl.dot(a, b)  # automatically does broadcast inner product of shape (M, K) dot (K, N) -> (M,N)
        # advance pointers
        a_ptrs += BLOCK_K * stride_k_a
        b_ptrs += BLOCK_K * stride_k_b

   # store result
   c_ptrs = c_ptr + batch_id * stride_batch_c + offs_m[:, None] * stride_m_c + offs_n[None, :] * stride_n_c
   tl.store(c_ptrs, acc.to(tl.float32))

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    ...
    # compute norm
    eps = 1e-5
    x = input_tensor
    # compute mean & var on last dim
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias
    # linear proj
    left = torch.nn.functional.linear(x_norm, left_proj_weight) # out feature hidden_dim
    right = torch.nn.functional.linear(x_norm, right_proj_weight)
    if not config.get('nomask', True):
        mask_ = mask.unsqueeze(-1).type_as(left)
        left = left * mask_
        right = right * mask_
    # gating
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()
    left = left * left_gate
    right = right * right_gate
    # now left,right shape [B,N,N,hidden_dim]
    B,N,_ , hidden_dim = left.shape
    # permute to [B, hidden_dim, N, N]
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    # cast to fp16
    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)
    # transpose right for multiplication
    right_T = right_fp16.permute(0,2,1).contiguous()
    # flatten batch*hidden_dim
    batch_flat = B * hidden_dim
    left_flat = left_fp16.view(batch_flat, N, N)
    right_flat_T = right_T.view(batch_flat, N, N)
    # allocate output c
    out_flat = torch.empty((batch_flat, N, N), dtype=torch.float32, device=x.device)
    # compute strides
    stride_batch_a = left_flat.stride(0)
    stride_m_a = left_flat.stride(1)
    stride_k_a = left_flat.stride(2)
    stride_batch_b = right_flat_T.stride(0)
    stride_k_b = right_flat_T.stride(1)
    stride_n_b = right_flat_T.stride(2)
    stride_batch_c = out_flat.stride(0)
    stride_m_c = out_flat.stride(1)
    stride_n_c = out_flat.stride(2)
    # launch kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (batch_flat, (N + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N)
    batched_matmul_kernel[grid](
        left_flat, right_flat_T, out_flat,
        batch_flat, N, N, N,
        stride_batch_a, stride_m_a, stride_k_a,
        stride_batch_b, stride_k_b, stride_n_b,
        stride_batch_c, stride_m_c, stride_n_c,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0,2,3,1)  # shape B,N,N,hidden_dim
    # to_out_norm

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input `x` over the feature dimension (`dim`).
2. Linear projections (no bias) to a hidden dimension `h` (called
   ``left`` and ``right``).  If a mask is supplied, mask both tensors.
3. Compute three sigmoid gates from the normalized input:
   * ``left_gate``  – applied to ``left``
   * ``right_gate`` – applied to ``right``
   * ``out_gate``   – applied after the pairwise product.
4. Fuse the gating: ``left = left * left_gate`` and ``right = right * right_gate``.
5. The heavy inner product:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix‑multiply of ``left_d`` with the transpose of
   ``right_d`` for each hidden channel `d`.  Implemented as a Triton kernel
   that processes ``B * h`` independent ``N×N`` matrix multiplications
   (accumulating in FP32 while loading in FP16 for speed).
6. Layer‑norm the result over the hidden dimension, multiply by ``out_gate``,
   and finally project back to the original dimension with a linear layer.
7. Return a tensor of shape ``[B, N, N, dim]`` (float32).

The Triton kernel is deliberately simple: it loads tiles of size
BLOCK_M×BLOCK_K and BLOCK_K×BLOCK_N, performs a dot‑product, and accumulates
into a FP32 tile.  Masks guard out‑of‑bounds loads for arbitrary sequence
lengths.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ  (A,B : [B,batch, N, N])
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    batch, M, N, K,                     # problem size (batch = B*h)
    stride_batch_a, stride_m_a, stride_k_a,
    stride_batch_b, stride_k_b, stride_n_b,
    stride_batch_c, stride_m_c, stride_n_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for many independent batches.

    A: [batch, M, K]   (row‑major)
    B: [batch, K, N]   (row‑major, already transposed from the original tensor)
    C: [batch, M, N]
    """
    # --------------------------------------------------------------
    # 1) Program IDs
    # --------------------------------------------------------------
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # --------------------------------------------------------------
    # 2) Compute offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for edges in M and N
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch (no offset in K yet)
    a_batch_ptr = a_ptr + batch_id * stride_batch_a
    b_batch_ptr = b_ptr + batch_id * stride_batch_b
    c_batch_ptr = c_ptr + batch_id * stride_batch_c

    # --------------------------------------------------------------
    # 3) Accumulator (FP32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 4) Loop over K dimension
    # --------------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K) – fp16
        a = tl.load(
            a_batch_ptr
            + offs_m[:, None] * stride_m_a
            + offs_k[None, :] * stride_k_a,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # Load B tile (K x N) – fp16
        b = tl.load(
            b_batch_ptr
            + offs_k[:, None] * stride_k_b
            + offs_n[None, :] * stride_n_b,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # Dot product + accumulate (up‑cast to FP32)
        acc += tl.dot(a, b).to(tl.float32)

    # --------------------------------------------------------------
    # 5) Write out the result (FP32)
    # --------------------------------------------------------------
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_m_c + offs_n[None, :] * stride_n_c
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Main entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """TriMul forward (outgoing) with a Triton‑accelerated inner product.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Load weights (all are 2‑D tensors, bias vectors where applicable)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) LayerNorm over the feature dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3) Optional mask (mask shape: [B, N, N])
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # ensure mask dtype matches the projections
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating values (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Batched MatMul: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Equivalent to many independent matmuls: (N×N) @ (N×N)ᵀ per channel.
    # ------------------------------------------------------------------
    # Rearrange to (batch, channel, N, N) → flatten batch*channel
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, hidden_dim, N, N]

    # Cast to fp16 for the kernel (still keep a copy for later)
    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    # Transpose the last two dims of right so we can multiply A @ Bᵀ
    right_fp16_T = right_fp16.transpose(-2, -1).contiguous()  # [B, hidden_dim, N, N]

    # Collapse the first two dimensions (B * hidden_dim) → number of independent matmuls
    batch_flat = bs * hidden_dim
    left_flat = left_fp16.view(batch_flat, seq_len, seq_len)       # [B*h, N, N]
    right_flat_T = right_fp16_T.view(batch_flat, seq_len, seq_len)  # [B*h, N, N]

    # Allocate output (FP32 accumulation)
    out_flat = torch.empty((batch_flat, seq_len, seq_len), dtype=torch.float32, device=input_tensor.device)

    # Strides in elements (not bytes) for Triton
    stride_batch_a = left_flat.stride(0)
    stride_m_a = left_flat.stride(1)
    stride_k_a = left_flat.stride(2)

    stride_batch_b = right_flat_T.stride(0)
    stride_k_b = right_flat_T.stride(1)   # K dimension
    stride_n_b = right_flat_T.stride(2)   # N dimension (after transpose)

    stride_batch_c = out_flat.stride(0)
    stride_m_c = out_flat.stride(1)
    stride_n_c = out_flat.stride(2)

    # Tunable block sizes (chosen for typical N up to 1024 on H100)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_flat,                               # program_id(0) – batch dimension
        (seq_len + BLOCK_M - 1) // BLOCK_M,       # program_id(1) – row tiles
        (seq_len + BLOCK_N - 1) // BLOCK_N,       # program_id(2) – col tiles
    )

    _batched_matmul_transpose_kernel[
        grid
    ](
        left_flat,
        right_flat_T,
        out_flat,
        batch_flat,
        seq_len,
        seq_len,
        seq_len,
        stride_batch_a,
        stride_m_a,
        stride_k_a,
        stride_batch_b,
        stride_k_b,
        stride_n_b,
        stride_batch_c,
        stride_m_c,
        stride_n_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, hidden_dim, N, N] → [B, N, N, hidden_dim]
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6) Output layer‑norm, outer gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)
    out = out * out_gate                     # apply outer gate
    out = F.linear(out, to_out_weight)       # project back to original dim

    return out
