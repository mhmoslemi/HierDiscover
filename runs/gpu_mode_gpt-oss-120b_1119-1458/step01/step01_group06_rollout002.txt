@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N,
    stride_lm, stride_ln, stride_lk,
    stride_rm, stride_rn, stride_rk,
    stride_om, stride_on, stride_ok,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # batch*hidden
    pid_i = tl.program_id(1)  # i tile
    pid_j = tl.program_id(2)  # j tile

    # Offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # K loop
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left: block (i, k)
        left = tl.load(
            left_ptr + pid_m * stride_lm
                     + offs_i[:, None] * stride_ln
                     + offs_k[None, :] * stride_lk,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )  # dtype fp16

        # load right: block (k, j) (transposed)
        right = tl.load(
            right_ptr + pid_m * stride_rm
                      + offs_k[:, None] * stride_rk
                      + offs_j[None, :] * stride_rn,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )
        # accumulate
        acc += tl.dot(left, right).to(tl.float32)

    # Store output
    tl.store(
        out_ptr + pid_m * stride_om
                + offs_i[:, None] * stride_on
                + offs_j[None, :] * stride_ok,
        acc,
        mask=mask_i[:, None] & mask_j[None, :]
    )

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Input LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Gating
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply optional mask

        left = left * mask_exp
        right = right * mask_exp

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Convert to half for kernel

    # Permute to (B, hidden, N, N) to fuse batch*hidden
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    left = left.to(torch.float16)
    right = right.to(torch.float16)

"""
Trion TriMul (outgoing) – fast forward pass.

The core of the operator is the N³ reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which can be expressed as a batched matrix‑multiply
    out[m,i,j] = left[m,i,:] @ right[m,:,j]   (m = b * hidden_dim).

This implementation:
* Performs LayerNorms, linear projections and gating with PyTorch (these ops are cheap).
* Packs the batch and hidden dimensions into a single “M” dimension.
* Executes the massive Gram‑product with a custom Triton kernel that
  – works on FP16 tensors (Tensor‑Core friendly),
  – accumulates in FP32 for numerical stability,
  – uses 128×128 tiles and a 64‑element K‑tile (tuned for H100),
  – employs three‑dimensional launch grid (M, tile‑row, tile‑col).

The kernel loads left‑tiles (i,k) and right‑tiles (k,j) (the right side is read
transposed on‑the‑fly), computes a dot via `tl.dot`, and stores the result.
All other operations (LayerNorm, gating, final linear) are performed in PyTorch.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# -------------------------------------------------------------
# Triton kernel: batched Gram‑matrix (left @ rightᵀ)
# -------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N,
    stride_lm, stride_ln, stride_lk,
    stride_rm, stride_rn, stride_rk,
    stride_om, stride_on, stride_ok,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute, for every m ∈ [0, M):
        out[m, i, j] = Σ_k left[m, i, k] * right[m, k, j]

    left/right/out are 3‑D tensors with layout (M, N, N).
    left  : (i,k)   – we read a (BLOCK_M, BLOCK_K) tile.
    right : (k,j)   – we read a (BLOCK_K, BLOCK_N) tile (transposed on‑fly).
    """
    pid_m = tl.program_id(0)          # batch * hidden index
    pid_i = tl.program_id(1)          # tile‑row index
    pid_j = tl.program_id(2)          # tile‑col index

    # ---- coordinate offsets within the N×N matrix ----
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 (stable & Tensor‑Core friendly)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- reduction over K dimension ----
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k_tiles):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile (i,k)
        left = tl.load(
            left_ptr
            + pid_m * stride_lm
            + offs_i[:, None] * stride_ln
            + offs_k[None, :] * stride_lk,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )   # fp16

        # right tile (k,j) – note the transpose via stride order
        right = tl.load(
            right_ptr
            + pid_m * stride_rm
            + offs_k[:, None] * stride_rk
            + offs_j[None, :] * stride_rn,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )   # fp16

        # dot product of (BLOCK_M, BLOCK_K) × (BLOCK_K, BLOCK_N)
        acc += tl.dot(left, right).to(tl.float32)

    # ---- store the output tile ----
    tl.store(
        out_ptr
        + pid_m * stride_om
        + offs_i[:, None] * stride_on
        + offs_j[None, :] * stride_ok,
        acc,
        mask=mask_i[:, None] & mask_j[:, None]
    )


# -------------------------------------------------------------
# Public entry point used by the test harness
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [batch, seq_len, seq_len, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1. Input LayerNorm
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------
    # 3. Gating (sigmoid) – same shape as left/right
    # -------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # -------------------------------------------------
    # 4. Optional mask (broadcast over hidden dimension)
    # -------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → [B, N, N, 1]
        m = mask.unsqueeze(-1).to(left.dtype)
        left = left * m
        right = right * m

    # -------------------------------------------------
    # 5. Apply element‑wise gates
    # -------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 6. Prepare tensors for the Triton kernel
    #    (B, N, N, H) → (B*H, N, N)  (H = hidden_dim)
    # -------------------------------------------------
    B, N, _, _ = left.shape

    # Bring hidden dimension to second axis for easy flattening
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Cast to FP16 – tensor cores are used inside the kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Merge batch and hidden dimensions
    BH = B * hidden_dim
    left_flat = left.view(BH, N, N)
    right_flat = right.view(BH, N, N)

    # Allocate output buffer (same type as inputs – will be cast later)
    out_flat = torch.empty_like(left_flat)

    # -------------------------------------------------
    # 7. Compute strides (needed by Triton)
    # -------------------------------------------------
    stride_lm, stride_ln, stride_lk = left_flat.stride()
    stride_rm, stride_rn, stride_rk = right_flat.stride()
    stride_om, stride_on, stride_ok = out_flat.stride()

    # -------------------------------------------------
    # 8. Kernel launch parameters (tuned for H100)
    # -------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch the kernel; 8 warps gives good occupancy on H100
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        BH,
        N,
        stride_lm,
        stride_ln,
        stride_lk,
        stride_rm,
        stride_rn,
        stride_rk,
        stride_om,
        stride_on,
        stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # -------------------------------------------------
    # 9. Reshape back to (B, N, N, hidden_dim) and cast to FP32
    # -------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N)      # (B, H, N, N)
    out = out.permute(0, 2, 3, 1).contiguous()    # (B, N, N, H)
    out = out.to(torch.float32)

    # -------------------------------------------------
    # 10. Output LayerNorm → gating → final linear
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
