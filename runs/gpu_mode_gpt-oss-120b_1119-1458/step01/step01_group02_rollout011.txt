"""
Optimized outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1. Layer‑norm the input over the feature dimension `dim`.
2. Two bias‑less linear projections (`left_proj`, `right_proj`) produce tensors of shape
   `[B, N, N, H]`.  An optional binary mask is applied.
3. Three sigmoid gates are computed from the same normalized tensor.
   `left` and `right` are multiplied by their respective gates and then cast to
   FP16 to halve memory traffic for the costly contraction.
4. Heavy contraction (the O(N³) core):
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed by a custom Triton kernel that computes many independent
   matrix products `C = A @ Bᵀ` with `A = left.reshape(B*H,N,N)` and
   `B = right.reshape(B*H,N,N)`.  The kernel works on FP16 data but
   accumulates in FP32 – it tiles the `N×N` output with
   `(BLOCK_M, BLOCK_N) = (128, 128)` and a `BLOCK_K = 64` inner‑tile, which
   gives high Tensor‑Core utilisation on H100.
5. The contraction result (still FP32) is layer‑normed, multiplied by the
   `out_gate` and finally linearly projected back to the original `dim`.

Only the O(N³) contraction is Triton‑implemented; all other operations use
high‑performance PyTorch primitives.  The kernel runs with 8 warps and
exploits FP16 → FP32 accumulation for speed and numerical stability.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B: FP16, C: FP32)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                            # dimensions (M = B*H, N = seq_len, K = seq_len)
    stride_am, stride_ai, stride_ak,    # strides for A (M,N,K)
    stride_bm, stride_bj, stride_bk,    # strides for B (M,N,K)
    stride_cm, stride_ci, stride_cj,    # strides for C (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute many independent matrix products C = A @ Bᵀ.
    A,B are stored as [M, N, K] (row‑major).  The kernel loops over
    K in tiles of length BLOCK_K and accumulates in FP32.
    """
    pid_m = tl.program_id(0)               # tile index along output row  (i)
    pid_n = tl.program_id(1)               # tile index along output col  (j)
    pid_z = tl.program_id(2)               # batch*hidden index

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # --------------------------------------------------------------
        # Load A tile   (M‑tile × K‑tile)   – FP16 -> FP32
        # --------------------------------------------------------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0,
                    dtype=tl.float16)
        a_fp32 = a.to(tl.float32)

        # --------------------------------------------------------------
        # Load B tile as (K‑tile × N‑tile) – note the transposed view
        # --------------------------------------------------------------
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bj
        )
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)
        b_fp32 = b.to(tl.float32)

        # --------------------------------------------------------------
        # Matrix‑multiply the tiles and accumulate
        # --------------------------------------------------------------
        acc += tl.dot(a_fp32, b_fp32)

    # ------------------------------------------------------------------
    # Write back C tile
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper that reshapes tensors and launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute C[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using the Triton kernel.
    Input tensors must be FP16 and contiguous.
    Returns a FP32 tensor with shape [B, N, N, H].
    """
    B, N, _, H = left.shape
    # Collapse batch and hidden dimension for a clean 3‑D view
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # [M, N, K]
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (FP32)
    C = torch.empty_like(A, dtype=torch.float32)

    # Strides in elements
    stride_am, stride_ai, stride_ak = A.stride()        # (N*K, K, 1)
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tile sizes tuned for H100 Tensor‑Core execution (FP16 → FP32)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,                     # M, N, K
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor  [B, N, N, dim]   (float32)
        - mask_tensor   : torch.Tensor [B, N, N]       (float32, binary)
        - weights_dict  : mapping from weight name to torch.Tensor
        - config_dict   : must contain ``dim`` and ``hidden_dim``;
                          optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack arguments
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ Layer‑norm over the feature dimension
    # -------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (FP32)

    # -------------------------------------------------
    # 2️⃣ Linear projections (no bias) – stay in FP32 for now
    # -------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------
    # 3️⃣ Apply mask if present (mask is broadcastable)
    # -------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # -------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # -------------------------------------------------
    # 5️⃣ Fuse gates and cast to FP16 for the heavy contraction
    # -------------------------------------------------
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # -------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton (output is FP32)
    # -------------------------------------------------
    out = _batched_matmul_transpose(left, right)   # [B, N, N, H]   (FP32)

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm over hidden dimension
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # -------------------------------------------------
    # 8️⃣ Apply final gate and linear projection back to `dim`
    # -------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
