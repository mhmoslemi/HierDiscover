def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', True)
    device = input_tensor.device

    # LayerNorm over dim
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights['norm.weight'],
        bias=weights['norm.bias'],
        eps=1e-5,
    )
    # Cast to half
    x_h = x_norm.to(torch.float16)

    # Cast weights to half for half-precision linear ops
    left_proj_w = weights['left_proj.weight'].to(torch.float16)
    right_proj_w = weights['right_proj.weight'].to(torch.float16)
    left_gate_w = weights['left_gate.weight'].to(torch.float16)
    right_gate_w = weights['right_gate.weight'].to(torch.float16)
    out_gate_w = weights['out_gate.weight'].to(torch.float16)

    # Linear projections
    left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden]
    right = F.linear(x_h, right_proj_w)

    # Gating
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w)).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_norm, weights['out_gate.weight']))  # keep fp32

    # Apply gates (and mask if needed)
    left = left * left_gate
    right = right * right_gate
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_f
        right = right * mask_f

    # Reshape for batched GEMM
    B, N, _, _ = left.shape
    total_heads = B * hidden_dim

    left_flat = left.permute(0,3,1,2).contiguous().view(total_heads, N, N)
    right_flat = right.permute(0,3,1,2).contiguous().view(total_heads, N, N)

    # Allocate output buffer (fp16 for compute)
    out_flat = torch.empty((total_heads, N, N), dtype=torch.float16, device=device)

    # Triton kernel launch for batched GEMM
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        total_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert output to fp32
    out = out_flat.to(torch.float32)

    # Reshape back to [B, N, N, hidden]
    out = (
        out.view(B, hidden_dim, N, N)
          .permute(0, 2, 3, 1)
          .contiguous()
    )

    # Post processing
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights['to_out_norm.weight'],
        bias=weights['to_out_norm.bias'],
        eps=1e-5,
    )
    out = out * out_gate  # broadcast
    # final projection
    # cast weight to half if possible
    to_out_w = weights['to_out.weight'].to(torch.float16)
    # convert out to half for efficient matmul
    out = F.linear(out.to(torch.float16), to_out_w).to(torch.float32)

    return out

@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_lh, stride_li, stride_lk,
    stride_rh, stride_ri, stride_rk,
    stride_oh, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_h = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers
    left_head_ptr = left_ptr + pid_h * stride_lh
    right_head_ptr = right_ptr + pid_h * stride_rh
    out_head_ptr = out_ptr + pid_h * stride_oh

    # accumulator fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_blocks_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_blocks_k):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile [M, K]
        left_ptrs = left_head_ptr + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=tl.float16(0.0))

        # right tile [K, N]
        right_ptrs = right_head_ptr + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_ri
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=tl.float16(0.0))

        # dot: (M,K) @ (K,N) -> (M,N) accumulate in fp32
        acc += tl.dot(left_tile, right_tile)

    # store
    out_ptrs = out_head_ptr + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)
```

"""
Optimized outgoing Triangle Multiplicative Update (TriMul) for AlphaFold‑3.

Algorithm
---------
1.  Layer‑Norm over the last dimension (float32).
2.  Cast to FP16 and apply all linear projections (no bias) in half‑precision.
3.  Compute sigmoid gates, multiply them (and an optional mask) – still FP16.
4.  Reshape tensors to (total_heads, N, N) where total_heads = B·hidden_dim.
5.  Batched GEMM C = A @ Bᵀ for every head is performed by a small
    Triton kernel that works on FP16 inputs and accumulates in FP32.
6.  Convert the result back to FP32, restore the original layout and
    apply a second Layer‑Norm, the output gate and the final projection.
The heavy O(N³·hidden_dim) work is carried out by the Triton kernel,
while the surrounding linear & norm ops use PyTorch cuBLAS/cuDNN kernels.
All compute after the kernel is in FP32; the final result is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# -------------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
# A : (head, i, k)  – FP16
# B : (head, j, k)  – FP16 (stored as (head, j, k))
# C : (head, i, j)  – FP32 accumulator, stored as FP32
# -------------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # *fp16  left  = (head, i, k)
    right_ptr,         # *fp16  right = (head, j, k)
    out_ptr,           # *fp32  output = (head, i, j)
    N,                 # i32    sequence length
    # strides for left  (head, i, k)
    stride_lh, stride_li, stride_lk,
    # strides for right (head, j, k)
    stride_rh, stride_ri, stride_rk,
    # strides for out   (head, i, j)
    stride_oh, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(0)          # which head (B·hidden_dim)
    pid_m = tl.program_id(1)          # block row (i)
    pid_n = tl.program_id(2)          # block col (j)

    # -----------------------------------------------------------------
    # Offsets in the i‑ and j‑dimensions
    # -----------------------------------------------------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # -----------------------------------------------------------------
    # Base pointers for this head
    # -----------------------------------------------------------------
    left_head_ptr  = left_ptr  + pid_h * stride_lh
    right_head_ptr = right_ptr + pid_h * stride_rh
    out_head_ptr   = out_ptr   + pid_h * stride_oh

    # -----------------------------------------------------------------
    # Accumulator in FP32
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over the K‑dimension in blocks
    # -----------------------------------------------------------------
    num_blocks_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_blocks_k):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile [M, K] – FP16
        left_ptrs = left_head_ptr + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left_mask = mask_i[:, None] & mask_k[None, :]
        left_tile = tl.load(left_ptrs, mask=left_mask, other=tl.float16(0.0))

        # right tile [K, N] – FP16 (note the transpose via stride ordering)
        right_ptrs = right_head_ptr + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_ri
        right_mask = mask_k[:, None] & mask_j[None, :]
        right_tile = tl.load(right_ptrs, mask=right_mask, other=tl.float16(0.0))

        # Dot product, accumulation in FP32
        acc += tl.dot(left_tile, right_tile)   # (M,K) @ (K,N) -> (M,N)

    # -----------------------------------------------------------------
    # Store the result tile (FP32)
    # -----------------------------------------------------------------
    out_ptrs = out_head_ptr + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)


# -------------------------------------------------------------------------
# Entry point used by the evaluation harness
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim]      (float32)
            - mask:        [B, N, N] (bool / 0‑1) – may be unused.
            - weights: dict containing all LayerNorm, Linear and gate weights.
            - config:   must contain "dim", "hidden_dim" and optional "nomask".

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # -----------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy linear & gate ops
    # -----------------------------------------------------------------
    x_h = x.to(torch.float16)

    # Cast weight tensors once (FP16) – tiny overhead compared to compute
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Linear projections (no bias)
    # -----------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)        # [B, N, N, hidden_dim]  fp16
    right = F.linear(x_h, right_proj_w)       # same shape

    # -----------------------------------------------------------------
    # 4️⃣ Gates (sigmoid)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w)).to(torch.float16)
    # out_gate stays in FP32 – it is used after the second LayerNorm
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # fp32

    # -----------------------------------------------------------------
    # 5️⃣ Apply gates (and optional mask) – all still FP16
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(torch.float16)      # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 6️⃣ Prepare for the batched GEMM  C = A @ Bᵀ
    #     reshape to (total_heads, N, N) where total_heads = B·hidden_dim
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    total_heads = B * hidden_dim

    # (B, N, N, H) -> (B, H, N, N) -> (total_heads, N, N)
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # Output buffer (FP32 accumulation)
    out_flat = torch.empty((total_heads, N, N), dtype=torch.float32, device=device)

    # -----------------------------------------------------------------
    # 7️⃣ Triton kernel launch – batched GEMM
    # -----------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64
    grid = (
        total_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    # -----------------------------------------------------------------
    # 8️⃣ Reshape back to the original layout (still FP32)
    # -----------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)   # (B, H, N, N)
        .permute(0, 2, 3, 1)                # (B, N, N, H)
        .contiguous()
    )  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 9️⃣ Post‑processing – LayerNorm, output gate, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # FP32

    out = out * out_gate  # broadcast over hidden_dim (FP32)

    # Final linear projection to `dim`.  Use half‑precision for the matmul
    # and cast back to FP32 for the returned tensor.
    out = F.linear(out.to(torch.float16), to_out_w).to(torch.float32)

    return out
