"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The heavy N³ reduction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is performed by a custom Triton kernel that batches the independent
matrix‑multiplications over the hidden dimension (d) and the batch.
The kernel works on fp16 tensors (to cut memory traffic) while
accumulating in fp32 for numerical stability.  Block sizes are
chosen at run‑time according to the sequence length to keep the launch
over‑head low for small N and to maximise tensor‑core utilisation for
larger N.  All other operations (LayerNorms, Linear projections,
sigmoids, masking and the final output projection) are kept in PyTorch,
which already uses highly‑optimised cuBLAS/cuDNN kernels.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                     # pointers
    stride_ap, stride_am, stride_ak,          # strides for A: (P, M, K)
    stride_bp, stride_bk, stride_bn,          # strides for B: (P, K, N)
    stride_cp, stride_cm, stride_cn,          # strides for C: (P, M, N)
    M, N, K,                                  # matrix sizes (all = seq_len)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C[p, i, j] = Σ_k A[p, i, k] * B[p, k, j]   (fp16→fp32 accumulation)
    - P = batch * hidden_dim  (each (b, d) pair is an independent matmul)
    - The kernel is launched with a 3‑D grid:
        pid_m: block over M (output row i)
        pid_n: block over N (output column j)
        pid_p: block over P (independent matmul)
    """
    pid_m = tl.program_id(0)      # block row
    pid_n = tl.program_id(1)      # block column
    pid_p = tl.program_id(2)      # which (batch, hidden) pair

    # --------------------------------------------------------------
    # Offsets within the M, N dimensions
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Base pointers for this (batch, hidden) instance
    # --------------------------------------------------------------
    a_base = A_ptr + pid_p * stride_ap
    b_base = B_ptr + pid_p * stride_bp
    c_base = C_ptr + pid_p * stride_cp

    # --------------------------------------------------------------
    # Accumulator (fp32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # K‑loop – tiled dot‑product
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        start_k = k * BLOCK_K
        offs_k = start_k + tl.arange(0, BLOCK_K)

        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)               # fp16

        # Load B tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)               # fp16

        # Dot product (fp16 inputs → fp32 accumulation)
        acc += tl.dot(a, b).to(tl.float32)

    # --------------------------------------------------------------
    # Store result (fp16)
    # --------------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper that runs the Triton kernel
# ----------------------------------------------------------------------
def _triton_matmul(left: torch.Tensor, right: torch.Tensor,
                   seq_len: int, hidden_dim: int,
                   block_m: int = 64, block_n: int = 64, block_k: int = 64):
    """
    Batched matrix multiplication for the TriMul reduction.
    left / right  : [B, N, N, H] in fp16
    Returns out   : [B, N, N, H] in fp16
    """
    B = left.shape[0]

    # ------------------------------------------------------------------
    # Rearrange tensors so that each (b, h) pair becomes an independent
    # matrix multiplication A[p, i, k] @ B[p, k, j]  (C[p, i, j])
    # ------------------------------------------------------------------
    # left → (B, H, N, N)  (i, k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right → (B, H, N, N)  (k, j)  (note the swap of the two N dims)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    # Collapse the (batch, hidden) dimensions into a single dimension P
    P = B * hidden_dim
    left_flat = left_perm.view(P, seq_len, seq_len)
    right_flat = right_perm.view(P, seq_len, seq_len)

    # Allocate output
    out = torch.empty_like(left_flat)

    # --------------------------------------------------------------
    # Strides (contiguous tensors)
    # --------------------------------------------------------------
    stride_ap = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bp = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_cp = out.stride(0)
    stride_cm = out.stride(1)
    stride_cn = out.stride(2)

    # --------------------------------------------------------------
    # Grid launch configuration
    # --------------------------------------------------------------
    grid = (
        (seq_len + block_m - 1) // block_m,   # blocks over M (i)
        (seq_len + block_n - 1) // block_n,   # blocks over N (j)
        P,                                    # one block per (batch, hidden) pair
    )

    # Launch the kernel
    _batched_matmul_kernel[grid](
        left_flat, right_flat, out,
        stride_ap, stride_am, stride_ak,
        stride_bp, stride_bk, stride_bn,
        stride_cp, stride_cm, stride_cn,
        seq_len, seq_len, seq_len,
        BLOCK_M=block_m, BLOCK_N=block_n, BLOCK_K=block_k,
        num_warps=8, num_stages=4
    )

    # Reshape back to the original layout
    out = out.view(B, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator using a Triton‑accelerated
    N³ reduction.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float)
        - weights      : dict of pre‑loaded model weights
        - config       : dict containing at least ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True)
    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]                # input & output channel dimension
    hidden_dim = config["hidden_dim"]  # internal hidden dimension
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (norm)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)                         # [B, N, N, dim]  fp32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]          # (hidden, dim)
    right_proj_weight = weights["right_proj.weight"]        # (hidden, dim)

    # Compute in fp32 – PyTorch's linear uses highly‑optimised cuBLAS kernels
    left = F.linear(x, left_proj_weight)                    # [B, N, N, hidden]
    right = F.linear(x, right_proj_weight)                  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking (applied before gating, exactly as in the reference)
    # ------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)        # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4️⃣ Gating: sigmoid(Linear(x))
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight)).to(left.dtype)   # fp16 if left is fp16 later
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)).to(right.dtype)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))                    # keep in fp32 (used later)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Cast to fp16 for the heavy N³ reduction
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 6️⃣ Choose block sizes dynamically (helps small N and large N)
    # ------------------------------------------------------------------
    seq_len = input_tensor.shape[1]   # N

    # Heuristic block size selection
    if seq_len <= 128:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32
    elif seq_len <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif seq_len <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:  # 768 – 1024 and above
        # 256 works well on H100 for 1024‑wide matmuls while staying inside shared‑mem limits
        BLOCK_M = BLOCK_N = 256 if seq_len % 256 == 0 else 128
        BLOCK_K = 32

    # ------------------------------------------------------------------
    # 7️⃣ Triton kernel: Σ_k left[..., i, k, d] * right[..., j, k, d]
    # ------------------------------------------------------------------
    out_fp16 = _triton_matmul(left_fp16, right_fp16,
                              seq_len=seq_len,
                              hidden_dim=hidden_dim,
                              block_m=BLOCK_M,
                              block_n=BLOCK_N,
                              block_k=BLOCK_K)   # [B, N, N, hidden] fp16

    # Back to fp32 for the rest of the pipeline
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 8️⃣ Output LayerNorm (over the hidden dimension)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    # ------------------------------------------------------------------
    # 9️⃣ Apply the output gate (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 🔟 Final linear projection back to the original dimension
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"]        # (dim, hidden)
    out = F.linear(out, to_out_weight)             # [B, N, N, dim] fp32

    return out
