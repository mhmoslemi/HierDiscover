```python
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # tile row index i
    pid_n = tl.program_id(1)  # tile col index j
    pid_z = tl.program_id(2)  # batch * hidden index

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # A tile (M, K)
        a_ptrs = A_ptr + pid_z * stride_am + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0, dtype=tl.float16)

        # B tile (K, N) (note: we need B transposed)
        b_ptrs = B_ptr + pid_z * stride_bm + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0, dtype=tl.float16)

        # convert to FP32 before dot
        a_fp32 = a.to(tl.float32)
        b_fp32 = b.to(tl.float32)

        acc += tl.dot(a_fp32, b_fp32)

    # store result
    c_ptrs = C_ptr + pid_z * stride_cm + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

Now helper function:

```
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right: [B, N, N, H] half tensors (torch.float16)
    Compute out = Σ_k left[..., i, k, h] * right[..., j, k, h]
    returns half tensor shape [B,N,N,H]
    """
    B,N,_,H = left.shape
    # reshape to (B*H, N, N)
    A = left.permute(0,3,1,2).contiguous().view(B*H, N, N)
    B_ = right.permute(0,3,1,2).contiguous().view(B*H, N, N)
    C = torch.empty_like(A)  # half

    # strides
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Choose block sizes based on N
    # Heuristic: use 128 for N>=512 else 64
    if N >= 512:
        BM, BN, BK = 128, 128, 32
    elif N >= 256:
        BM, BN, BK = 64, 64, 32
    else:
        BM, BN, BK = 64, 64, 16

    grid = ((N + BM - 1)//BM,
            (N + BN - 1)//BN,
            B*H)

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B*H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BM,
        BLOCK_N=BN,
        BLOCK_K=BK,
        num_warps=4,
        # optionally set out data type
    )
    # reshape back
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

Now custom_kernel:

```
def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm (float32)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Cast to fp16 for projections
    x_fp16 = x.to(torch.float16)

    # Projectors (fp16)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left = torch.nn.functional.linear(x_fp16, left_proj_w)   # shape [B,N,N,hidden]
    right = torch.nn.functional.linear(x_fp16, right_proj_w)

    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(torch.float16)
        left = left * mask
        right = right * mask

    # Gating (float32)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Apply gates (convert gates to half)
    left = left * left_gate.to(torch.float16)
    right = right * right_gate.to(torch.float16)

    # Heavy contraction: left/right have shape [B,N,N,hidden]; compute out
    out_fp16 = _batched_matmul_transpose_fp16(left, right)   # half

    # Cast back to float32 for following ops
    out = out_fp16.to(torch.float32)

    # Second LayerNorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply out gate and final projection
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass – Triton‑accelerated heavy contraction.

Algorithm
---------
1️⃣ Layer‑norm over the feature dimension `dim` (FP32).
2️⃣ Linear projections to the hidden dimension (`hidden_dim`) are performed in
   FP16 (no bias).  The result tensors `left` and `right` are therefore
   `torch.float16`.
3️⃣ If a mask is present it is applied (broadcasted) in FP16.
4️⃣ Three sigmoid gates (left, right, out) are computed from the normalized
   tensor in FP32, then cast to FP16 and applied element‑wise.
5️⃣ The O(N³) contraction

        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

   is evaluated by a custom Triton kernel that treats each pair
   (batch × hidden) as an independent matrix multiplication
   `C = A @ Bᵀ`.  FP16 inputs are loaded, up‑converted to FP32,
   multiplied with `tl.dot` (Tensor‑Core friendly) and accumulated in
   FP32.  The kernel uses a tunable tile size (BLOCK_M×BLOCK_N×BLOCK_K)
   and adapts the tile size to the sequence length for better occupancy.
6️⃣ The contraction result is cast back to FP32, followed by a second
   LayerNorm over `hidden_dim`.
7️⃣ The output gate (FP32) is applied and a final FP32 linear projection
   returns the tensor of shape `[B, N, N, dim]`.

Only the O(N³) step is Triton‑implemented; all other operations are
high‑performance PyTorch primitives.  Mixed‑precision (FP16/FP32) reduces
memory traffic and lets H100 Tensor‑Cores accelerate the contraction.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B shape: [M,N,K]  ->  C shape: [M,N,N])
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,            # *float16  [M, N, K]  (M = batch*hidden)
    B_ptr,            # *float16  [M, N, K]
    C_ptr,            # *float16  [M, N, N]
    M, N, K,          # i32 dimensions
    stride_am, stride_ai, stride_ak,   # strides for A
    stride_bm, stride_bj, stride_bk,   # strides for B
    stride_cm, stride_ci, stride_cj,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile row (i)
    pid_n = tl.program_id(1)   # tile col (j)
    pid_z = tl.program_id(2)   # independent matrix index (batch*hidden)

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ----- load A tile (M‑tile × K‑tile) -----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
            dtype=tl.float16,
        )

        # ----- load B tile as (K‑tile × N‑tile) – transposed view -----
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # Convert to FP32 before the dot product (Tensor‑Core friendly)
        a_fp32 = a.to(tl.float32)
        b_fp32 = b.to(tl.float32)

        # Matrix‑multiply‑accumulate
        acc += tl.dot(a_fp32, b_fp32)

    # ------------------------------------------------------------------
    # Write the output tile
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: batched matmul + transpose (FP16 inputs, FP16 output)
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    on tensors of dtype ``torch.float16``.
    Input shape:  [B, N, N, H]
    Output shape: [B, N, N, H]  (dtype torch.float16)
    """
    B, N, _, H = left.shape    # left/right are contiguous after permute

    # Move the hidden dimension next to the batch: [B*H, N, N]
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (FP16)
    C = torch.empty_like(A)

    # Strides in *elements* (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # ------------------------------------------------------------------
    # Tile size selection – simple heuristic to keep occupancy high
    # ------------------------------------------------------------------
    if N >= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    elif N >= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 16

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # rows (i)
        (N + BLOCK_N - 1) // BLOCK_N,   # cols (j)
        B * H,                          # batch * hidden
    )

    # Launch Triton kernel
    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # H100 can handle 8 warps comfortably
    )

    # Reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : Tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor  [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor  [B, N, N]   (bool/float, may be ignored)
        - weights_dict : mapping from weight name to torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``. Optional
                         ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape: [B, N, N, dim]  (FP32)

    # ------------------------------------------------------------------
    # 2️⃣ FP16 linear projections (no bias)
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = torch.nn.functional.linear(x_fp16, left_proj_w)    # [B,N,N,hidden]
    right = torch.nn.functional.linear(x_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted, FP16)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is 0/1 – convert to FP16 and broadcast onto the hidden dim
        mask = mask_tensor.unsqueeze(-1).to(torch.float16)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gating (computed in FP32, then cast)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # Apply gates (cast gates to FP16 for the element‑wise product)
    left = left * left_gate.to(torch.float16)
    right = right * right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via the Triton kernel (FP16)
    # ------------------------------------------------------------------
    out_fp16 = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hidden] fp16

    # Cast back to FP32 for the remainder of the pipeline
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension (float32)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                       # element‑wise gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
