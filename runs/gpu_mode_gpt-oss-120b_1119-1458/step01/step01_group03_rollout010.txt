import torch
import triton
import triton.language as tl

@triton.jit
def trimul_gemm_kernel(
    L_ptr, R_ptr, O_ptr,
    N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    stride_bL, stride_iL, stride_jL, stride_hL,
    stride_bR, stride_iR, stride_jR, stride_hR,
    stride_bO, stride_iO, stride_jO, stride_hO
):
    # program IDs
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_bh = tl.program_id(2)  # combined batch * hidden index

    # decompose combined index into batch and hidden
    b = pid_bh // H
    h = pid_bh % H

    # base pointers for this (b, h) slice
    L_base = L_ptr + b * stride_bL + h * stride_hL
    R_base = R_ptr + b * stride_bR + h * stride_hR
    O_base = O_ptr + b * stride_bO + h * stride_hO

    # offsets within the N×N matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile L[ i, k ]
        L_ptrs = L_base + offs_m[:, None] * stride_iL + offs_k[None, :] * stride_jL
        L = tl.load(L_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right tile transposed: we need R[ k, j ] = right[ b, j, k, h ]
        R_ptrs = R_base + offs_k[:, None] * stride_iR + offs_n[None, :] * stride_jR
        R = tl.load(R_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(L, R)

    # write result
    O_ptrs = O_base + offs_m[:, None] * stride_iO + offs_n[None, :] * stride_jO
    tl.store(O_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.
    -------------------------------------------------------------
    The computation can be written as a batch of independent matrix
    multiplications:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    For each hidden channel `h` we treat ``left`` as an N×N matrix and
    ``right`` as its transpose.  The kernel above implements a classic
    tiled GEMM with FP16 inputs and FP32 accumulation.  Except for the
    heavy N³ contraction everything else (LayerNorm, linear layers,
    gating) stays in PyTorch for simplicity.
    """
    # unpack arguments
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------ 1. Input LayerNorm ------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------ 2. Linear projections ------------------
    # left, right: [B, N, N, hidden_dim]  (float32)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------ 3. Optional mask ------------------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------ 4. Gating ------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Cast gates to half to stay in the same dtype as the projected tensors
    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    left = left * left_gate_h
    right = right * right_gate_h

    # ------------------ 5. Cast to fp16 for the GEMM ------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape   # left_h: [B, N, N, hidden_dim]

    # Allocate output buffer
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left_h.device)

    # ------------------ 6. Triton batched GEMM ------------------
    # Choose tile sizes compatible with Tensor Cores on H100
    # Small N (≤512) benefits from 64‑tile, large N from 128‑tile.
    if N <= 512:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 32
        num_warps = 4
    else:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8

    # Number of tiles per output matrix
    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N

    # Grid: (tiles_m, tiles_n, batch*hidden)
    grid = (num_tiles_m, num_tiles_n, B * hidden_dim)

    # Strides (in element counts)
    stride_bL, stride_iL, stride_jL, stride_hL = left_h.stride()
    stride_bR, stride_iR, stride_jR, stride_hR = right_h.stride()
    stride_bO, stride_iO, stride_jO, stride_hO = out.stride()

    trimul_gemm_kernel[grid](
        left_h,
        right_h,
        out,
        N,
        hidden_dim,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        stride_bL, stride_iL, stride_jL, stride_hL,
        stride_bR, stride_iR, stride_jR, stride_hR,
        stride_bO, stride_iO, stride_jO, stride_hO,
        num_warps=num_warps,
        # Use 3 stages of shared‑memory pipelining (default is 2)
        num_stages=3,
    )

    # ------------------ 7. Output LayerNorm (hidden_dim) ------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------ 8. Output gating ------------------
    out = out * out_gate  # out_gate still float32

    # ------------------ 9. Final projection back to original dim ------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_gemm_kernel(
    L_ptr,               # half *  (B, N, N, H)  – left projection
    R_ptr,               # half *  (B, N, N, H)  – right projection
    O_ptr,               # float * (B, N, N, H)  – output accumulator
    N,                   # sequence length (runtime)
    H,                   # hidden dimension (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    # strides for left
    stride_bL, stride_iL, stride_jL, stride_hL,
    # strides for right
    stride_bR, stride_iR, stride_jR, stride_hR,
    # strides for output
    stride_bO, stride_iO, stride_jO, stride_hO,
):
    """
    Batched GEMM for the TriMul contraction.
    For every (batch, hidden) pair we compute

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    i.e. a matrix‐multiply of size N×N with the transpose of the
    right matrix.  The kernel is tiled with BLOCK_M×BLOCK_N tiles
    and loops over K with tiles of size BLOCK_K, using FP16 inputs
    and FP32 accumulation (Tensor‑Core friendly on H100).
    """
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_bh = tl.program_id(2)         # combined batch*hidden index

    # De‑compose the combined index.
    b = pid_bh // H                    # batch index
    h = pid_bh % H                     # hidden‑channel index

    # Base pointer for the current (b,h) slice.
    L_base = L_ptr + b * stride_bL + h * stride_hL
    R_base = R_ptr + b * stride_bR + h * stride_hR
    O_base = O_ptr + b * stride_bO + h * stride_hO

    # Offsets inside the N×N matrix.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Reduction over k.
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile   L[i,k]
        L_ptrs = L_base + offs_m[:, None] * stride_iL + offs_k[None, :] * stride_jL
        L = tl.load(
            L_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # right tile transposed (k,j) = right[b, j, k, h]
        R_ptrs = R_base + offs_k[:, None] * stride_iR + offs_n[None, :] * stride_jR
        R = tl.load(
            R_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(L, R)   # (M,K) @ (K,N) → (M,N)

    # Store the result.
    O_ptrs = O_base + offs_m[:, None] * stride_iO + offs_n[None, :] * stride_jO
    tl.store(O_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the *outgoing* TriMul module.
    ----------------------------------------------------------------
    The heavy contraction is expressed as a batch of independent GEMMs
    (one per hidden channel) and computed with the tiled kernel above.
    All other pieces – LayerNorms, linear projections, sigmoid gates –
    are performed with PyTorch for simplicity.
    """
    # ------------------------------------------------------------------
    # 1. unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 2. input LayerNorm (over the channel dimension C)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"]) # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 4. Optional pairwise mask
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5. Gating (sigmoid) – keep gates in FP16 to stay in the kernel dtype
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )
    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Apply gates and cast to FP16 for the heavy GEMM
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16) * left_gate_h
    right_h = right.to(torch.float16) * right_gate_h

    B, N, _, _ = left_h.shape               # left_h : [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 7. Allocate output buffer (FP32 accumulator)
    # ------------------------------------------------------------------
    out = torch.empty(
        (B, N, N, hidden_dim), dtype=torch.float32, device=left_h.device
    )

    # ------------------------------------------------------------------
    # 8. Choose tiling parameters (Tensor‑Core friendly)
    # ------------------------------------------------------------------
    if N <= 512:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 32
        num_warps = 4
    else:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8

    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N

    # Grid: (tiles_m, tiles_n, batch*hidden)
    grid = (num_tiles_m, num_tiles_n, B * hidden_dim)

    # Strides (in element counts, not bytes)
    stride_bL, stride_iL, stride_jL, stride_hL = left_h.stride()
    stride_bR, stride_iR, stride_jR, stride_hR = right_h.stride()
    stride_bO, stride_iO, stride_jO, stride_hO = out.stride()

    # ------------------------------------------------------------------
    # 9. Launch the Triton GEMM kernel
    # ------------------------------------------------------------------
    trimul_gemm_kernel[grid](
        left_h,
        right_h,
        out,
        N,
        hidden_dim,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        stride_bL,
        stride_iL,
        stride_jL,
        stride_hL,
        stride_bR,
        stride_iR,
        stride_jR,
        stride_hR,
        stride_bO,
        stride_iO,
        stride_jO,
        stride_hO,
        num_warps=num_warps,
        num_stages=3,               # 3‑stage pipeline for better L2 reuse
    )

    # ------------------------------------------------------------------
    # 10. Output LayerNorm (over hidden dimension)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 11. Output gating
    # ------------------------------------------------------------------
    out = out * out_gate      # out_gate is still FP32, broadcasting over hidden

    # ------------------------------------------------------------------
    # 12. Final linear projection back to the original dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
