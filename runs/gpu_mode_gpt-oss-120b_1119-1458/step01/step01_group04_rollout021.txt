"""
custom_kernel
==============

Implements the outgoing triangle multiplicative update (TriMul) from
AlphaFold‑3. The heavy pairwise “einsum” is expressed as a batch of
independent matrix multiplications C = A @ Bᵀ where each head corresponds
to one hidden dimension.  The GEMM is tiled and executed with a small
Triton kernel that uses FP16 operands and FP32 accumulation, multiple
warps and a two‑stage pipeline.  All other operations (LayerNorm,
linear projections, gating) are performed in PyTorch but in half precision
when possible, so the kernel only sees the already‑gated, masked FP16
matrices.  The final output is returned as float32 with shape
[batch, seq_len, seq_len, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        triton.Config({'BLOCK_M': 64, 'BLOCK_N': 64, 'BLOCK_K': 64, 'num_warps': 4}, num_stages=2),
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 128, 'BLOCK_K': 32, 'num_warps': 8}, num_stages=2),
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 64, 'BLOCK_K': 32, 'num_warps': 8}, num_stages=2),
        triton.Config({'BLOCK_M': 64, 'BLOCK_N': 128, 'BLOCK_K': 32, 'num_warps': 8}, num_stages=2),
    ],
    key=['N']  # choose config based on sequence length
)
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # *fp16   left = left_proj * left_gate * mask
    right_ptr,         # *fp16   right = right_proj * right_gate * mask
    out_ptr,           # *fp32   result (hidden_dim output)
    N,                 # i32    sequence length
    stride_lh, stride_li, stride_lk,   # i64   strides for left   (head, i, k)
    stride_rh, stride_rj, stride_rk,   # i64   strides for right  (head, j, k) – note order of rj,rk
    stride_oh, stride_oi, stride_oj,   # i64   strides for out    (head, i, j)
    BLOCK_M: tl.constexpr,  # tile size in i‑dimension
    BLOCK_N: tl.constexpr,  # tile size in j‑dimension
    BLOCK_K: tl.constexpr,  # tile size in k‑dimension
):
    # program ids
    pid_h = tl.program_id(0)      # head index   (batch * hidden_dim)
    pid_m = tl.program_id(1)      # block row   (i)
    pid_n = tl.program_id(2)      # block col   (j)

    # ------------------------------------------------------------------
    # Offsets for the output tile
    # ------------------------------------------------------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Pointers to the beginning of the current head
    # ------------------------------------------------------------------
    left_head_ptr = left_ptr + pid_h * stride_lh
    right_head_ptr = right_ptr + pid_h * stride_rh
    out_head_ptr = out_ptr + pid_h * stride_oh

    # ------------------------------------------------------------------
    # Accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    num_blocks_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_blocks_k):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: [M, K] (head, i, k)
        left_ptrs = left_head_ptr + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left_mask = mask_i[:, None] & mask_k[None, :]
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # right tile transposed: we need [K, N] = (head, k, j)
        # right is stored as (head, j, k); we swap the axes by swapping strides.
        right_ptrs = right_head_ptr + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj
        right_mask = mask_k[:, None] & mask_j[None, :]
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # dot product: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(left_tile, right_tile)

    # ------------------------------------------------------------------
    # Store tile
    # ------------------------------------------------------------------
    out_ptrs = out_head_ptr + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)

# Triton‑accelerated TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass for the outgoing Triangle Multiplicative Update.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor with shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # Cast to FP16 for the cheap linear layers
    # ------------------------------------------------------------------
    x_fp16 = x.half()

    # ------------------------------------------------------------------
    # Cast model weights to FP16 once (they are small)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    to_out_w      = weights["to_out.weight"].half()

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, hidden_dim]  fp16
    right = F.linear(x_fp16, right_proj_w) # [B, N, N, hidden_dim]  fp16

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # Gating (sigmoid of a linear transform)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # fp16
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))  # fp16
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w)).float()  # fp32 for later

    # ------------------------------------------------------------------
    # Apply gates (remain in fp16)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Reshape to (total_heads, N, N) for the batched GEMM.
    # total_heads = B * hidden_dim
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    total_heads = B * hidden_dim
    # (B, N, N, hidden) -> (B, hidden, N, N) -> (total_heads, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # Output buffer (fp32)
    out_flat = torch.empty((total_heads, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Triton launch parameters (auto‑tuned)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        total_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp32, shape [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Post‑processing: hidden‑dim LayerNorm -> out‑gate -> final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Gating in hidden dim
    out = out * out_gate

    # Final projection back to `dim` (do it in FP16 for speed then cast)
    out = F.linear(out.half(), to_out_w).float()

    return out

    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 – the heavy linear layers are cheap in half‑precision
    # ------------------------------------------------------------------
    x_fp16 = x.half()

    # ------------------------------------------------------------------
    # 3️⃣ Cast model weights to FP16 once (they are tiny compared to activations)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    to_out_w      = weights["to_out.weight"].half()

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x_fp16, left_proj_w)   # [B, N, N, hidden_dim]  fp16
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, hidden_dim]  fp16

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (applied before the GEMM)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left  = left  * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ Gating (sigmoid of a linear map)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # fp16
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))  # fp16
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w)).float()  # fp32

    # ------------------------------------------------------------------
    # 7️⃣ Apply gates (still fp16)
    # ------------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 8️⃣ Prepare data for the Triton kernel:
    #     reshape to (total_heads, N, N) where total_heads = B * hidden_dim
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    total_heads = B * hidden_dim

    # (B, N, N, hidden) -> (B, hidden, N, N) -> (total_heads, N, N)
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # Output buffer (fp32 accumulation result)
    out_flat = torch.empty((total_heads, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 9️⃣ Triton launch configuration (auto‑tuned)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        total_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 10️⃣ Reshape back to (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp32, shape [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 11️⃣ Hidden‑dim LayerNorm → out‑gate → final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                               # gate in hidden dim

    # Final linear (done in fp16 for speed, then cast to fp32)
    out = F.linear(out.half(), to_out_w).float()

    return out
```
