"""
TriMul (outgoing) – highly‑optimized forward pass.

Algorithm
---------
1. Layer‑norm the input (float32).
2. Project to the hidden dimension (`left_proj`, `right_proj`) and compute
   three element‑wise gates (`left_gate`, `right_gate`, `out_gate`) in fp16.
   Optional mask is applied before gating.
3. Fuse hidden & batch dimensions: (B, N, N, H) → (B*H, N, N) and launch a
   custom Triton kernel that computes the batched Gram product
        out[b,i,j] = Σₖ left[b,i,k] * right[b,j,k]
   i.e. a batched matrix multiplication `left @ rightᵀ`.  The kernel loads
   tiles of shape (BLOCK_M, BLOCK_K) and (BLOCK_K, BLOCK_N) in fp16,
   performs the multiplication with Tensor‑Core `tl.dot` (fp16 → fp32
   accumulation) and stores the result in fp16.
4. Reshape the result back to (B, N, N, H), convert to float32 and apply a
   second Layer‑Norm, the output gate, and the final linear projection.
5. Return a float32 tensor of shape (B, N, N, dim).

The kernel is tuned for H100 (BLOCK_M/N = 128, BLOCK_K = 64) and uses
fp16 for all memory traffic while accumulating in fp32 for numerical
stability.
"""

import math
import torch
import triton
import triton.language as tl


@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N,                                 # M = B*H, N = sequence length
    stride_lm, stride_ln, stride_lk,      # left strides
    stride_rm, stride_rn, stride_rk,      # right strides
    stride_om, stride_on, stride_ok,      # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched Gram product: out[m,i,j] = Σₖ left[m,i,k] * right[m,j,k]
    left  : (M, N, N)   (stored as row i, column k)
    right : (M, N, N)   (stored as row j, column k) → we load transposed:
            right_T[m,k,j] = right[m,j,k]

    All tensors are fp16; accumulation is fp32.
    """
    pid_m = tl.program_id(0)  # index over the collapsed batch/hidden dimension
    pid_i = tl.program_id(1)  # tile row index
    pid_j = tl.program_id(2)  # tile col index

    # -----------------------------------------------------------------
    # Tile offsets for the i- and j-dimensions
    # -----------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 (stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Reduction over k
    # -----------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: (BLOCK_M, BLOCK_K)   = left[m, i, k]
        left = tl.load(
            left_ptr
            + pid_m * stride_lm
            + offs_i[:, None] * stride_ln
            + offs_k[None, :] * stride_lk,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right block (transposed): (BLOCK_K, BLOCK_N) = right[m, j, k]ᵀ
        right = tl.load(
            right_ptr
            + pid_m * stride_rm
            + offs_k[:, None] * stride_rk
            + offs_j[None, :] * stride_rn,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # Tensor‑core dot: fp16 * fp16 → fp32 accumulation
        acc += tl.dot(left, right, out_dtype=tl.float32)

    # -----------------------------------------------------------------
    # Write back (cast to fp16)
    # -----------------------------------------------------------------
    tl.store(
        out_ptr
        + pid_m * stride_om
        + offs_i[:, None] * stride_on
        + offs_j[None, :] * stride_ok,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor of shape [B, N, N, dim]
            mask         : torch.Tensor of shape [B, N, N] (or None)
            weights      : dict of model parameters (torch.Tensor)
            config       : dict with keys "dim", "hidden_dim", "nomask"
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    ####################################################################
    # 1. Input LayerNorm (float32)
    ####################################################################
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  fp32

    ####################################################################
    # 2. Cast to fp16 for the cheap linear layers
    ####################################################################
    x_h = x.to(torch.float16)

    # Prepare fp16 weight tensors (one‑time conversion)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    ####################################################################
    # 3. Linear projections + gating (still fp16)
    ####################################################################
    left = torch.nn.functional.linear(x_h, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x_h, right_proj_w)

    # Optional mask (broadcast over hidden dim)
    if not nomask and mask is not None:
        # mask is float32 → convert to fp16, add trailing dim for broadcasting
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    left_gate  = torch.nn.functional.linear(x_h, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_h, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x_h, out_gate_w).sigmoid()

    # Apply gates (element‑wise fp16)
    left  = left * left_gate
    right = right * right_gate

    ####################################################################
    # 4. Prepare tensors for the Triton kernel
    #    (B, N, N, H) → (B*H, N, N)
    ####################################################################
    B, N, _, _ = left.shape
    # (B, H, N, N) contiguous view → flatten batch & hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(B * hidden_dim, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)

    out_flat = torch.empty_like(left_flat)  # fp16 buffer for the kernel

    ####################################################################
    # 5. Kernel launch
    ####################################################################
    # Strides in element units (as returned by .stride())
    stride_lm, stride_ln, stride_lk = left_flat.stride()
    stride_rm, stride_rn, stride_rk = right_flat.stride()
    stride_om, stride_on, stride_ok = out_flat.stride()

    # Choose tile sizes – tuned for H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Grid dimensions: (batch*hidden, ceil(N/BM), ceil(N/BN))
    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    triton_tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * hidden_dim,
        N,
        stride_lm,
        stride_ln,
        stride_lk,
        stride_rm,
        stride_rn,
        stride_rk,
        stride_om,
        stride_on,
        stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,           # H100 benefits from 4 warps per program
        num_stages=2,          # double‑buffering
    )
    # Synchronise to make sure kernel has finished before next ops
    torch.cuda.synchronize()

    ####################################################################
    # 6. Reshape back to (B, N, N, H) and convert to fp32
    ####################################################################
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # fp32 for the next LayerNorm

    ####################################################################
    # 7. Hidden → output LayerNorm, gating and final linear projection
    ####################################################################
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    out = out * out_gate.to(torch.float32)  # broadcast over (B,N,N, hidden)

    out = torch.nn.functional.linear(out, weights["to_out.weight"])  # -> (B,N,N,dim)

    return out
