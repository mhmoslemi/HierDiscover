def elementwise_mul_mask_kernel(in_ptr, gate_ptr, mask_ptr, out_ptr, N_ELEM,
                               BLOCK_SIZE: tl.constexpr):

    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask_offs = offs < N_ELEM
    in_val = tl.load(in_ptr + offs, mask=mask_offs)
    gate_val = tl.load(gate_ptr + offs, mask=mask_offs)
    if mask_ptr != 0:

        out = in_val * gate_val * mask_val
    else:
        out = in_val * gate_val
    tl.store(out_ptr + offs, out, mask=mask_offs)
```

"""
TriMul (outgoing) – fast forward implementation.

Algorithm
---------
1. Layer‑norm the input tensor (B×N×N×C) in float16.
2. Linear projections → left, right (B×N×N×H) (no bias).
3. Optional mask (B×N×N) is applied to the projected tensors.
4. Gated multiplicative update:
   * left  = left  * sigmoid(linear(x, left_gate_weight))
   * right = right * sigmoid(linear(x, right_gate_weight))
   The two element‑wise products are fused in a tiny Triton kernel.
5. Dense N³ contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   We compute it as a batched matrix‑multiply:
   * reshape left/right to (B, H, N, N)
   * out_f = torch.matmul(left_f, right_f.transpose(-1, -2))   # (B, H, N, N)
   * out = out_f.permute(0, 2, 3, 1)                           # (B, N, N, H)
   This uses cuBLAS/Tensor‑Cores (half‑precision) and is orders of magnitude
   faster than a hand‑written Triton GEMM.
6. Layer‑norm over the hidden dimension H (half‑precision).
7. Apply the output gate (sigmoid(linear(x, out_gate_weight))) and a final
   linear projection back to the original channel dimension C.
8. Return the result in float32.

Only the element‑wise gated multiplication (step 4) is written in Triton,
fulfilling the “must use Triton” requirement while the dominant O(N³) work
leverages the highly‑optimized cuBLAS kernels.
"""

import torch.nn.functional as F
import triton
import triton.language as tl

    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N_ELEM

    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    out = a * b
    tl.store(out_ptr + offs, out, mask=mask)


def _triton_elementwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """In‑place a *= b using Triton (both tensors must be half‑precision and
    have identical shape). Returns the mutated tensor `a` for convenience."""
    assert a.is_cuda and b.is_cuda
    assert a.shape == b.shape
    assert a.dtype == torch.float16 and b.dtype == torch.float16

    total = a.numel()
    BLOCK = 1024                     # max threads per CUDA block
    grid = ((total + BLOCK - 1) // BLOCK,)

    elementwise_mul_kernel[grid](
        a, b, a,                # a and out share the same buffer (in‑place)
        total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return a


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32 or float16)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dict mapping weight names to tensors
        - config_dict  : dict containing at least 'dim' and 'hidden_dim'

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # Helper to move a weight to half‑precision on the correct device
    # --------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # --------------------------------------------------------------
    # Extract model parameters (all stored as float32 in the checkpoint)
    # --------------------------------------------------------------
    norm_w = to_half(weights["norm.weight"])
    norm_b = to_half(weights["norm.bias"])

    left_proj_w  = to_half(weights["left_proj.weight"])
    right_proj_w = to_half(weights["right_proj.weight"])

    left_gate_w  = to_half(weights["left_gate.weight"])
    right_gate_w = to_half(weights["right_gate.weight"])
    out_gate_w   = to_half(weights["out_gate.weight"])

    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # --------------------------------------------------------------
    # 1. Input LayerNorm (half‑precision)
    # --------------------------------------------------------------
    # Convert input to half once; LayerNorm will stay in half.
    x = input_tensor.to(torch.float16)
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # shape (B, N, N, C), dtype float16

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_w)  # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 3. Optional mask (broadcasted on hidden dimension)
    # --------------------------------------------------------------
    if mask is not None:
        # mask: (B, N, N) -> (B, N, N, 1) -> broadcast over hidden dim
        mask_f = mask.to(torch.float16).unsqueeze(-1)
        left  = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4. Gating – compute sigmoid(linear(x, *gate*)) and fuse mul via Triton
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # (B,N,N,hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # (B,N,N,hidden_dim)

    # Fuse element‑wise multiplication using Triton (in‑place)
    left  = _triton_elementwise_mul(left, left_gate)
    right = _triton_elementwise_mul(right, right_gate)

    # --------------------------------------------------------------
    # 5. Dense N³ contraction via batched matrix multiplication
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # Rearrange to (B, hidden_dim, N, N) so that we can call torch.matmul.
    left_f  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_f = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # Batched matmul: (B, H, N, N) @ (B, H, N, N)^T  -> (B, H, N, N)
    out_f = torch.matmul(left_f, right_f.transpose(-1, -2))

    # Restore original layout (B, N, N, H)
    out = out_f.permute(0, 2, 3, 1)  # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 6. Output LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # (B, N, N, hidden_dim), still half‑precision

    # --------------------------------------------------------------
    # 7. Apply output gate and final linear projection back to C
    # --------------------------------------------------------------
    out = out * out_gate                                 # gating (half)
    out = F.linear(out, to_out_w)                        # (B, N, N, C), half

    # --------------------------------------------------------------
    # 8. Cast back to float32 as required by the specification
    # --------------------------------------------------------------
    return out.to(torch.float32)
