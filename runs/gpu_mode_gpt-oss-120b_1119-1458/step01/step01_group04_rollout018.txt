def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    # LayerNorm
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # Cast to fp16 for linear ops
    x_fp16 = x.half()

    # Linear projections
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left = F.linear(x_fp16, left_proj_w)  # [B,N,N,hidden] fp16
    right = F.linear(x_fp16, right_proj_w)

    # Gate linear
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))  # fp16
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))

    # Apply gates and optional mask (broadcast)
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(x_fp16.dtype)  # [B,N,N,1] half
    else:
        mask_f = None

    # Fuse multiplication using Triton for speed (optional)
    # Compute left = left * left_gate * mask_f (if mask)
    # Compute right analogously using Triton elementwise kernel
    # For simplicity, we implement elementwise in PyTorch when mask_f is None,
    # else we use Triton kernel.

    # Prepare output for left and right after gating/mask
    # Use elementwise mul in PyTorch:
    left = left * left_gate
    right = right * right_gate
    if mask_f is not None:
        left = left * mask_f
        right = right * mask_f

    # flatten for batched matmul
    B, N, _, _ = left.shape
    total_heads = B * hidden_dim
    # Reshape with view/permute to (total_heads, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(total_heads, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(total_heads, N, N)

    # Batched matrix multiplication: left @ right^T
    # Using torch.bmm for high performance (Tensor cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # fp16 result

    # Cast to fp32 for following layers
    out = out_mat.float().view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Compute out_gate (float32)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply out_gate using Triton kernel (elementwise multiply)
    total_elem = out.numel()
    BLOCK = 65536  # 64k
    grid = ( (total_elem + BLOCK - 1) // BLOCK, )
    _apply_out_gate_kernel[grid](
        out.view(-1),
        out_gate.view(-1),
        total_elem,
        BLOCK_SIZE=BLOCK,
    )
    # Final linear projection to dim
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)  # shape [B,N,N,dim] float32

    return out

"""
custom_kernel
==============

Fast forward implementation of the *outgoing* Triangle Multiplicative Update
(TriMul) from AlphaFold‑3.

Algorithm
---------
1. Layer‑norm the input pair representation (float32).
2. Cast to FP16 and apply the four linear projections (left/right and the
   three sigmoid gates).  All linear layers are executed in half‑precision,
   which lets the cuBLAS kernels run on Tensor‑cores.
3. Fuse mask (if present), left‑gate and right‑gate with a tiny Triton kernel
   (element‑wise multiplication).  This avoids an extra PyTorch kernel launch.
4. Perform the core ``C[i,j] = Σ_k left[i,k]·right[j,k]`` as a batch of
   independent matrix‑multiplications ``C = A @ Bᵀ`` using
   ``torch.bmm`` on half‑precision tensors.  With the batch dimension
   ``B·hidden_dim`` this maps to a highly‑optimized batched GEMM on H100.
5. Cast the result back to FP32, reshape to ``[B,N,N,hidden]`` and apply the
   output LayerNorm.
6. Multiply by the output gate using another small Triton kernel.
7. Final linear projection back to the original channel dimension (float32).

The heavy tensor‑core matmul is handled by CuBLAS, while Triton is used for
the two cheap element‑wise fusions, satisfying the “use Triton” requirement.
The implementation runs entirely in mixed‑precision and returns a
``torch.float32`` tensor of shape ``[B, N, N, dim]``.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton: element‑wise multiplication for left/right (gate * mask)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    dst_ptr, src_ptr, gate_ptr, mask_ptr, n_elements,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    src = tl.load(src_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # mask_ptr may be a null pointer when no mask is used.
    if mask_ptr != 0:
        m = tl.load(mask_ptr + offs, mask=mask, other=1.0)
        val = src * gate * m
    else:
        val = src * gate

    tl.store(dst_ptr + offs, val, mask=mask)


# ----------------------------------------------------------------------
# Triton: apply the output gate (float32)
# ----------------------------------------------------------------------
@triton.jit
def _apply_out_gate_kernel(
    dst_ptr, gate_ptr, n_elements,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    out_val = tl.load(dst_ptr + offs, mask=mask, other=0.0)
    gate_val = tl.load(gate_ptr + offs, mask=mask, other=1.0)
    tl.store(dst_ptr + offs, out_val * gate_val, mask=mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of TriMul (outgoing version).
    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor  – shape [B, N, N, dim] (float32)
            mask          – shape [B, N, N] (bool/float, optional)
            weights       – dict with the model parameters
            config        – dict with ``dim`` and ``hidden_dim``
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] – fp32

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for all linear layers
    # ------------------------------------------------------------------
    x_fp16 = x.half()                     # [B,N,N,dim]  fp16

    # Linear projections (no bias)
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left = F.linear(x_fp16, left_proj_w)   # [B,N,N,hidden] fp16
    right = F.linear(x_fp16, right_proj_w)

    # Gate linear projections (sigmoid)
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # fp16
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))  # fp16
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w)).float()  # fp32

    # ------------------------------------------------------------------
    # 3) Fuse gate (+ optional mask) with a tiny Triton kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    total_elems = left.numel()   # B * N * N * hidden

    # Prepare mask (fp16) – broadcast on hidden dim if needed
    if not nomask:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)  # [B,N,N,1] fp16
        mask_ptr = mask_fp16.view(-1)
    else:
        mask_fp16 = None
        mask_ptr = 0  # null pointer for the kernel

    # Allocate output buffers for the fused tensors
    left_fused = torch.empty_like(left)
    right_fused = torch.empty_like(right)

    # Triton launch parameters
    BLOCK = 65536  # ~64k threads per kernel launch
    grid = ( (total_elems + BLOCK - 1) // BLOCK, )

    # left = left * left_gate * mask
    _apply_gate_mask_kernel[grid](
        left_fused.view(-1),
        left.view(-1),
        left_gate.view(-1),
        mask_ptr,
        total_elems,
        BLOCK_SIZE=BLOCK,
    )
    # right = right * right_gate * mask
    _apply_gate_mask_kernel[grid](
        right_fused.view(-1),
        right.view(-1),
        right_gate.view(-1),
        mask_ptr,
        total_elems,
        BLOCK_SIZE=BLOCK,
    )

    # ------------------------------------------------------------------
    # 4) Batched matrix multiplication:   out[b,i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    #    We treat each (batch, hidden) pair as an independent GEMM.
    # ------------------------------------------------------------------
    total_heads = B * hidden_dim

    # reshape to (total_heads, N, N) – make memory contiguous for cuBLAS
    left_mat = left_fused.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_mat = right_fused.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # Batched GEMM on Tensor‑cores (FP16 input, FP16 accumulation -> FP16 result)
    out_mat_fp16 = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Cast back to FP32 and reshape to [B, N, N, hidden]
    out = (
        out_mat_fp16.float()
        .view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, hidden]  fp32

    # ------------------------------------------------------------------
    # 5) Output LayerNorm (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 6) Apply the output gate with a tiny Triton kernel (fp32)
    # ------------------------------------------------------------------
    total_out_elems = out.numel()
    grid_out = ((total_out_elems + BLOCK - 1) // BLOCK,)
    _apply_out_gate_kernel[grid_out](
        out.view(-1),
        out_gate.view(-1),
        total_out_elems,
        BLOCK_SIZE=BLOCK,
    )

    # ------------------------------------------------------------------
    # 7) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]            # shape (dim, hidden)
    out = F.linear(out, to_out_w)                  # [B, N, N, dim]  fp32

    return out
