@triton.jit
def apply_mask_kernel(x_ptr, mask_ptr, out_ptr,
                      B, N, H,
                      stride_x_b, stride_x_i, stride_x_j, stride_x_h,
                      stride_mask_b, stride_mask_i, stride_mask_j,
                      BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr):
    pid_head = tl.program_id(0)  # encodes (batch, head)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    total_heads = B * H
    b = pid_head // H
    h = pid_head % H

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Load x tile
    x_ptrs = x_ptr + b*stride_x_b + h*stride_x_h + offs_m[:, None]*stride_x_i + offs_n[None, :]*stride_x_j
    x = tl.load(x_ptrs, mask=mask_m[:, None] & mask_n[None, :], other=0.0)

    # Load mask tile (same for all h) -> mask element shape (B,N,N)
    mask_ptrs = mask_ptr + b*stride_mask_b + offs_m[:, None]*stride_mask_i + offs_n[None,:]*stride_mask_j
    mask_val = tl.load(mask_ptrs, mask=mask_m[:, None] & mask_n[None, :], other=0.0)

    # Multiply
    out = x * mask_val

    out_ptrs = out_ptr + b*stride_x_b + h*stride_x_h + offs_m[:, None]*stride_x_i + offs_n[None,:]*stride_x_j
    tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    # extract dims
    B,N,_ ,C = input_tensor.shape
    dim = config['dim']
    hidden_dim = config['hidden_dim']

    # Convert weights to half
    def to_half(t): return t.to(device=input_tensor.device, dtype=torch.float16)

    # LayerNorm
    norm_w = weights['norm.weight']
    norm_b = weights['norm.bias']
    # use float32 computation for stability
    x_norm = torch.nn.functional.layer_norm(input_tensor,
                                            (dim,),
                                            weight=norm_w,
                                            bias=norm_b,
                                            eps=1e-5)
    x_half = x_norm.to(torch.float16)

    # Linear projections
    left_proj_w = to_half(weights['left_proj.weight'])
    right_proj_w = to_half(weights['right_proj.weight'])
    left = torch.nn.functional.linear(x_half, left_proj_w)  # B,N,N,H
    right = torch.nn.functional.linear(x_half, right_proj_w)

    # optional mask using Triton to reduce memory loads
    if mask is not None:
        mask_half = mask.to(dtype=torch.float16)
        # allocate output for masked left/right
        left_masked = torch.empty_like(left)
        right_masked = torch.empty_like(right)
        # kernel grid
        # shape dims for kernel
        total_heads = B * hidden_dim
        BLOCK_M = 32
        BLOCK_N = 32
        grid = (total_heads,
                (N + BLOCK_M - 1) // BLOCK_M,
                (N + BLOCK_N - 1) // BLOCK_N)
        # compute strides
        s_xb, s_xi, s_xj, s_xh = left.stride()
        s_mb, s_mi, s_mj = mask_half.stride()
        apply_mask_kernel[grid](
            left, mask_half, left_masked,
            B, N, hidden_dim,
            s_xb, s_xi, s_xj, s_xh,
            s_mb, s_mi, s_mj,
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N,
            num_warps=4)
        apply_mask_kernel[grid](
            right, mask_half, right_masked,
            B, N, hidden_dim,
            s_xb, s_xi, s_xj, s_xh,
            s_mb, s_mi, s_mj,
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N,
            num_warps=4)
        left = left_masked
        right = right_masked

    # Gates
    left_gate_w = to_half(weights['left_gate.weight'])
    right_gate_w = to_half(weights['right_gate.weight'])
    out_gate_w = to_half(weights['out_gate.weight'])
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_half, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_half, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_half, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # Batched matmul (C++/CUDA)
    # transpose right on N axes
    right_t = right.permute(0,2,1,3).contiguous()
    # reshape batch*head
    left_flat = left.permute(0,3,1,2).contiguous().view(B*hidden_dim, N, N)
    right_flat = right_t.permute(0,3,1,2).contiguous().view(B*hidden_dim, N, N)
    out_flat = torch.bmm(left_flat, right_flat)  # (B*H, N, N) float16
    out = out_flat.view(B, hidden_dim, N, N).permute(0,2,3,1)  # (B,N,N,H)

    # Output normalization
    to_out_norm_w = to_half(weights['to_out_norm.weight'])
    to_out_norm_b = to_half(weights['to_out_norm.bias'])
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = torch.nn.functional.layer_norm(out_fp32,
                                                   (hidden_dim,),
                                                   weight=to_out_norm_w.to(torch.float32),
                                                   bias=to_out_norm_b.to(torch.float32),
                                                   eps=1e-5)
    out_norm = out_norm_fp32.to(torch.float16)

    # Apply output gate and final linear
    gated = out_norm * out_gate
    to_out_w = to_half(weights['to_out.weight'])
    final = torch.nn.functional.linear(gated, to_out_w)

    return final.to(torch.float32)

"""
Highly‑optimized outgoing TriMul (AlphaFold‑3 style) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor (B×N×N×C) in float32.
2. Cast to float16 and apply three linear projections (left, right, gates)
   – all weights are stored in float16 to exploit Tensor‑Cores.
3. Optional mask is applied with a tiny Triton kernel (broadcasted over the
   hidden dimension) to avoid an extra Python launch per head.
4. Gating (sigmoid) is applied element‑wise to `left` and `right`.
5. The N³ contraction is reduced to a batched half‑precision GEMM:
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]    (right is transposed)
   This is performed with `torch.bmm` on the (B·H) batch of N×N matrices,
   which on an H100 is dramatically faster than a hand‑written Triton kernel.
6. Layer‑norm over the hidden dimension, multiplied by an output gate,
   and a final linear projection back to the original channel dimension.
7. Result is returned as float32.

The only Triton code used is the mask‑application kernel; all heavy
computations (linear projections, sigmoid, batched GEMM, layer‑norm) use
highly‑optimised PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: broadcast mask (B,N,N) onto a B×N×N×H tensor.
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_kernel(
    x_ptr,                 # const __half *
    mask_ptr,              # const __half *
    out_ptr,               # __half *
    B, N, H,               # meta‑data
    stride_x_b, stride_x_i, stride_x_j, stride_x_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr
):
    """Apply mask[ b, i, j ] (broadcast on H) to x[ b, i, j, h ]."""
    pid_head = tl.program_id(0)   # encodes (batch, hidden)
    pid_m    = tl.program_id(1)   # block row (i)
    pid_n    = tl.program_id(2)   # block col (j)

    total_heads = B * H
    b = pid_head // H
    h = pid_head % H

    # Tile indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Load a tile of the input tensor
    x_ptrs = x_ptr + b * stride_x_b + h * stride_x_h \
                     + offs_m[:, None] * stride_x_i \
                     + offs_n[None, :] * stride_x_j
    x = tl.load(x_ptrs,
                mask=mask_m[:, None] & mask_n[None, :],
                other=0.0)

    # Load the same mask tile (no hidden dimension)
    mask_ptrs = mask_ptr + b * stride_mask_b \
                         + offs_m[:, None] * stride_mask_i \
                         + offs_n[None, :] * stride_mask_j
    m = tl.load(mask_ptrs,
                mask=mask_m[:, None] & mask_n[None, :],
                other=0.0)

    out = x * m

    out_ptrs = out_ptr + b * stride_x_b + h * stride_x_h \
                        + offs_m[:, None] * stride_x_i \
                        + offs_n[None, :] * stride_x_j
    tl.store(out_ptrs,
             out,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Triton‑accelerated helper to apply a mask (used only when mask!=None).
# ----------------------------------------------------------------------
def _apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """Broadcast mask over the hidden dimension using Triton."""
    B, N, _, H = tensor.shape
    out = torch.empty_like(tensor)

    BLOCK_M = 32
    BLOCK_N = 32
    grid = (
        B * H,                     # (batch, hidden)
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # strides as element offsets
    s_xb, s_xi, s_xj, s_xh = tensor.stride()
    s_mb, s_mi, s_mj = mask.stride()

    apply_mask_kernel[grid](
        tensor, mask, out,
        B, N, H,
        s_xb, s_xi, s_xj, s_xh,
        s_mb, s_mi, s_mj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32, on CUDA)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : dict of torch.Tensors for all Linear / LayerNorm weights
        - config_dict  : {'dim': C, 'hidden_dim': H}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # 0. Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper to move a weight to the correct device/dtype (half)
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x_norm = F.layer_norm(input_tensor,
                          normalized_shape=(dim,),
                          weight=norm_w,
                          bias=norm_b,
                          eps=1e-5)                # still float32
    x = x_norm.to(torch.float16)                     # half for the heavy lifting

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = to_half(weights["left_proj.weight"])
    right_proj_w = to_half(weights["right_proj.weight"])

    left  = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]  half
    right = F.linear(x, right_proj_w)  # [B, N, N, hidden_dim]  half

    # ------------------------------------------------------------------
    # 3. Optional mask – applied with a tiny Triton kernel (broadcast on H)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_h = mask.to(dtype=torch.float16)  # now half
        left  = _apply_mask_triton(left,  mask_h)
        right = _apply_mask_triton(right, mask_h)

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate_w  = to_half(weights["left_gate.weight"])
    right_gate_w = to_half(weights["right_gate.weight"])
    out_gate_w   = to_half(weights["out_gate.weight"])

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # [B,N,N,hidden_dim] half

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. N³ contraction – implement as a batched half‑precision GEMM
    #    (left @ rightᵀ) for each hidden dimension.
    # ------------------------------------------------------------------
    # rightᵀ on the two N axes
    right_t = right.permute(0, 2, 1, 3).contiguous()   # [B, N, N, H]

    # reshape to (B*H, N, N) for torch.bmm
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right_t.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matmul – Tensor‑Core accelerated on H100
    out_flat = torch.bmm(left_flat, right_flat)        # [B*H, N, N] half
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,H] half

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (float32), gate and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # Layer‑norm over hidden_dim in float32
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)

    # Apply output gate
    gated = out_norm * out_gate

    # Final linear back to original channel dimension
    final = F.linear(gated, to_out_w)   # [B,N,N,C] half

    # Return as float32 (as required by the specification)
    return final.to(torch.float32)
