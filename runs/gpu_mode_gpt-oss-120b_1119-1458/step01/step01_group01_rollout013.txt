""" 
Optimized outgoing TriMul forward using fused Triton elementwise kernel and
PyTorch batched matrix multiplication (torch.bmm) for the N³ reduction.
All heavy linear projections and the final N³ reduction are performed with
Tensor‑Core‑enabled fp16 GEMM (torch.bmm), while a lightweight Triton kernel
fuses mask, gating and type‑casting before the reduction.
Result is returned in float32 as required.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fused_mul3_kernel(
    out_ptr, a_ptr, b_ptr, c_ptr,
    size,
    stride_out, stride_a, stride_b, stride_c,
    BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs * stride_a, mask=mask).to(tl.float32)
    b = tl.load(b_ptr + offs * stride_b, mask=mask).to(tl.float32)
    c = tl.load(c_ptr + offs * stride_c, mask=mask).to(tl.float32)

    out = (a * b * c).to(tl.float16)
    tl.store(out_ptr + offs * stride_out, out, mask=mask)

def _apply_mask_gate(proj, mask, gate, dtype_fp16=torch.float16):
    """
    Fuses elementwise multiplication: out = proj * mask * gate
    and casts the result to FP16.
    proj, mask, gate must have identical shape (B, N, N, H) and be contiguous.
    """
    size = proj.numel()
    out = torch.empty_like(proj, dtype=dtype_fp16, device=proj.device)

    # flatten tensors (contiguous guarantees stride == 1)
    proj_f = proj.view(-1)
    mask_f = mask.view(-1)
    gate_f = gate.view(-1)
    out_f = out.view(-1)

    BLOCK_SIZE = 65536  # 64K elements per program
    grid = (triton.cdiv(size, BLOCK_SIZE),)

    fused_mul3_kernel[grid](
        out_f, proj_f, mask_f, gate_f,
        size,
        out_f.stride(0), proj_f.stride(0),
        mask_f.stride(0), gate_f.stride(0),
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )
    return out

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data: tuple (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and gates (sigmoid)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 3. Optional mask – expand to [B, N, N, hidden]
    # ------------------------------------------------------------------
    if config.get("nomask", True):
        # a tensor of ones has the same effect as “no mask”
        mask_exp = torch.ones_like(left_proj, dtype=left_proj.dtype, device=left_proj.device)
    else:
        # mask is [B, N, N]; broadcast over hidden dimension
        mask_exp = mask.unsqueeze(-1).to(left_proj.dtype)               # (B,N,N,1)
        mask_exp = mask_exp.expand(-1, -1, -1, hidden_dim).contiguous() # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4. Fuse mask + gate + cast to fp16 via a tiny Triton kernel
    # ------------------------------------------------------------------
    left_fp16 = _apply_mask_gate(left_proj, mask_exp, left_gate)   # (B,N,N,H), fp16
    right_fp16 = _apply_mask_gate(right_proj, mask_exp, right_gate)# (B,N,N,H), fp16

    # ------------------------------------------------------------------
    # 5. Batched matmul: out = Σ_k left[...,i,k,h] * right[...,j,k,h]
    #    Implemented with torch.bmm (Tensor‑Core‑accelerated)
    # ------------------------------------------------------------------
    B, N, _, H = left_fp16.shape
    # reshape to (B*H, N, N) and transpose right so that the K‑dimension aligns
    left_mat = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_fp16.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)          # fp16, shape (B*H, N, N)

    # reshape back to [B, N, N, H] (fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # for the remaining FP32 ops

    # ------------------------------------------------------------------
    # 6. Post‑tri‑mul LayerNorm + output gate
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)

    out = out * out_gate  # broadcast over hidden dimension

    # ------------------------------------------------------------------
    # 7. Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])  # (B,N,N,dim)

    return out

"""
Optimised outgoing TriMul (AlphaFold‑3) forward pass.

The expensive N³ reduction
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
is performed with a **Tensor‑Core‑accelerated batched GEMM**
(PyTorch ``torch.bmm``) on fp16 data, which is far faster than the
original custom Triton block‑wise mat‑mul.  The only Triton code
required by the problem statement is a tiny element‑wise kernel that
fuses the mask, the per‑channel gates and the cast to fp16 before the
reduction.  All other operations (LayerNorms, linear projections and
the final output projection) are left to PyTorch, which also uses
high‑throughput cuBLAS/Tensor‑Core kernels.

The implementation works for any of the test configurations
(​B∈{1,2}, N∈{128,256,512,1024}, dim∈{128,384},
hidden_dim=128​) and respects the “nomask” flag.
The returned tensor is always ``float32`` as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: element‑wise   out = proj * mask * gate   (cast to fp16)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_kernel(
    out_ptr, proj_ptr, mask_ptr, gate_ptr,
    size,
    stride_out, stride_proj, stride_mask, stride_gate,
    BLOCK_SIZE: tl.constexpr,
):
    """
    A single‑program kernel that multiplies three fp32 tensors
    element‑wise and writes the result as fp16.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < size

    # Load as fp32 for numerical stability
    a = tl.load(proj_ptr + offs * stride_proj, mask=active).to(tl.float32)
    b = tl.load(mask_ptr + offs * stride_mask, mask=active).to(tl.float32)
    c = tl.load(gate_ptr + offs * stride_gate, mask=active).to(tl.float32)

    out = (a * b * c).to(tl.float16)                # cast once at the end
    tl.store(out_ptr + offs * stride_out, out, mask=active)


def _apply_mask_gate(proj: torch.Tensor,
                     mask: torch.Tensor,
                     gate: torch.Tensor) -> torch.Tensor:
    """
    Fuse ``proj * mask * gate`` and cast the result to ``float16``.
    All three tensors must have identical shape and be contiguous.
    """
    size = proj.numel()
    out = torch.empty_like(proj, dtype=torch.float16, device=proj.device)

    # Flatten to 1‑D (contiguous ⇒ stride == 1)
    proj_f = proj.contiguous().view(-1)
    mask_f = mask.contiguous().view(-1)
    gate_f = gate.contiguous().view(-1)
    out_f = out.view(-1)

    BLOCK_SIZE = 65536                           # 64 K elements per program
    grid = (triton.cdiv(size, BLOCK_SIZE),)     # 1‑D grid

    fused_mul3_kernel[grid](
        out_f, proj_f, mask_f, gate_f,
        size,
        out_f.stride(0), proj_f.stride(0),
        mask_f.stride(0), gate_f.stride(0),
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,                             # light‑weight kernel
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, dim]
            - mask         : torch.Tensor of shape [B, N, N]  (may be ignored)
            - weights      : dict of model parameters
            - config       : dict with keys ``dim``, ``hidden_dim`` and ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (dtype = torch.float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # → [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) and sigmoid gates
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if config.get("nomask", True):
        # No masking – we can skip the Triton kernel and multiply directly.
        left_fp16  = (left_proj * left_gate).to(torch.float16)
        right_fp16 = (right_proj * right_gate).to(torch.float16)
    else:
        # Expand mask to [B, N, N, H]
        mask_exp = mask.unsqueeze(-1).to(left_proj.dtype)                  # (B,N,N,1)
        mask_exp = mask_exp.expand(-1, -1, -1, hidden_dim).contiguous()   # (B,N,N,H)

        # Fuse mask + gate + cast with a tiny Triton kernel
        left_fp16  = _apply_mask_gate(left_proj,  mask_exp, left_gate)    # fp16
        right_fp16 = _apply_mask_gate(right_proj, mask_exp, right_gate)   # fp16

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM for the N³ reduction (Tensor‑Core accelerated)
    #      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left_fp16.shape          # H == hidden_dim
    # reshape to (B*H, N, N)   – left: (i,k) ; right: (k,j) after a transpose
    left_mat  = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_fp16.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    # torch.bmm uses cuBLAS/Tensor‑Cores → fast fp16 GEMM
    out_mat = torch.bmm(left_mat, right_mat)                     # (B*H, N, N), fp16

    # back‑to the original layout  →  [B, N, N, H]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )

    # ------------------------------------------------------------------
    # 5️⃣ Post‑tri‑mul LayerNorm + output gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # broadcast over hidden dim

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to ``dim`` (still fp32)
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])    # [B, N, N, dim]

    return out
