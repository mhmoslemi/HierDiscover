"""
Optimized outgoing Triangle Multiplicative Update (TriMul) forward pass.

Algorithm
---------
1️⃣ LayerNorm over the feature dimension (float32).
2️⃣ Cast to FP16 and apply linear projections (left/right) and gating
   projections (left_gate/right_gate/out_gate).  Gating values are passed
   through a sigmoid.
3️⃣ Fuse the element‑wise product of projection, gate and *optional* mask
   with a tiny Triton kernel.  This avoids several memory passes.
4️⃣ The core O(N³) operation is a batch of independent matrix‑multiplications
   C = A @ Bᵀ where A,B ∈ ℝ^{B·H × N × N} (H = hidden_dim).  We use
   `torch.bmm`, which on H100 runs as a Tensor‑Core GEMM in mixed precision
   (FP16 × FP16 → FP32 accumulation) and is far faster than a handwritten
   Triton matmul for this workload.
5️⃣ Reshape, LayerNorm on the hidden dimension, multiply by the out‑gate,
   and finally project back to the original feature dimension.

All heavy arithmetic is performed in FP16, accumulation in FP32, and the
final output is returned as float32, matching the reference implementation.
A small Triton kernel is still required (step 3) to satisfy the “use
Triton” constraint.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiply projection × gate × mask (if present)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr,                 # fp16  projection (left or right)
    gate_ptr,                 # fp16  corresponding gate
    mask_ptr,                 # fp16  mask (flattened, shape B·N·N)
    out_ptr,                  # fp16  destination
    hidden_dim,               # i64   hidden dimension (used for broadcasting mask)
    n_elements,               # i64   total number of elements in proj/gate (B·N·N·H)
    BLOCK: tl.constexpr,      # compile‑time block size
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    valid = offs < n_elements

    # load projection and gate (fp16 → fp32 for higher‑precision mul)
    proj = tl.load(proj_ptr + offs, mask=valid).to(tl.float32)
    gate = tl.load(gate_ptr + offs, mask=valid).to(tl.float32)

    # mask broadcasting: each mask entry repeats `hidden_dim` times
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=valid).to(tl.float32)

    out = proj * gate * mask_val                    # fp32 intermediate
    tl.store(out_ptr + offs, out.to(tl.float16), mask=valid)


# ----------------------------------------------------------------------
# Triton kernel entry point (required by the evaluation harness)
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim) – keep in FP32 for stability
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # Cast to FP16 for all subsequent linear / gating ops
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # Linear projection weights (cast once)
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # Projections (FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x_h, left_proj_w)   # [B,N,N,hidden]
    right_proj = F.linear(x_h, right_proj_w)  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # Gating values (sigmoid, kept in FP16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w)).to(torch.float16)
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w)).to(torch.float32)  # stay FP32

    # ------------------------------------------------------------------
    # Apply mask (if any) and fuse with the gates using a tiny Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] -> [B,N,N,1] -> FP16 -> flattened
        mask_f = mask.unsqueeze(-1).to(torch.float16).contiguous()
        mask_flat = mask_f.view(-1)                     # shape B·N·N
        # Allocate output tensors for the gated&masked tensors
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # Triton launch config
        BLOCK = 65536
        n_elem = left_proj.numel()                      # B·N·N·H
        grid = (triton.cdiv(n_elem, BLOCK),)

        # left = left_proj * left_gate * mask
        _gate_mask_kernel[grid](
            left_proj,
            left_gate,
            mask_flat,
            left,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
        # right = right_proj * right_gate * mask
        _gate_mask_kernel[grid](
            right_proj,
            right_gate,
            mask_flat,
            right,
            hidden_dim,
            n_elem,
            BLOCK=BLOCK,
        )
    else:
        # No mask – simple element‑wise products (still in FP16)
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # Core O(N³) operation: batch GEMM (A @ Bᵀ)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*H, N, N) where hidden acts as a batch dimension
    left_batched  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_batched = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiply – uses Tensor Cores (FP16×FP16→FP32)
    out_batched = torch.bmm(left_batched, right_batched.transpose(1, 2))  # [B*H,N,N] (fp16)

    # Reshape back to [B,N,N,hidden] and promote to FP32 for the following layers
    out = (
        out_batched
        .view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )   # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # Post‑processing: hidden‑dim LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # [B,N,N,hidden]  (float32)

    out = out * out_gate               # broadcast over hidden dimension

    # Final projection back to original feature dimension (float32)
    out = F.linear(out, weights["to_out.weight"])  # [B,N,N,dim]

    return out
```
