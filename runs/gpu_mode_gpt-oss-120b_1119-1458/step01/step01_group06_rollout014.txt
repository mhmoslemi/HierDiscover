def triton_gram_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N,
    stride_lm, stride_li, stride_lk,
    stride_rm, stride_ri, stride_rj,
    stride_om, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute start offsets
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_j = offs_j < N

    # accumulator

    # reduction over k

        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

"""
Outgoing TriMul (AlphaFold3) – fast Triton implementation.

The core of the operator is the N³ reduction

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

which is a batched matrix‑multiply (Gram product) between the
`left` tensor and the *transposed* `right` tensor for every hidden
dimension `d`.  The heavy reduction is performed by a custom Triton
kernel that treats each pair (batch, hidden) as an independent batch.
All other operations (LayerNorms, linear projections, gating, final
Linear) are carried out with PyTorch in FP16 for bandwidth‑limited
parts and converted back to FP32 only for the final output.

Key optimisations
-----------------
* FP16 compute with FP32 accumulation (the `tl.dot` result is cast).
* Tile size 64 × 64 × 128 (M, N, K) – the larger K reduces the number of
  reduction loops by a factor of two compared with the reference.
* The `right` tensor is explicitly transposed before launching the kernel,
  allowing perfectly coalesced loads.
* All linear weights are cast once to FP16; the final output Linear uses
  FP32 weights (as in the reference model).
* Optional mask handling is fused with the projection step (no extra
  tensors are materialised).

The kernel signature follows the Triton convention – three program IDs:
batch‑hidden index, tile‑row, tile‑col.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def triton_gram_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N,                         # M = total batch*hidden, N = seq_len (unused in kernel)
    stride_lm, stride_li, stride_lk,
    stride_rm, stride_ri, stride_rj,
    stride_om, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched Gram matrix:
        out[m, i, j] = Σ_k left[m, i, k] * right[m, k, j]
    All tensors are stored in row‑major (C‑contiguous) order.
    """
    batch = tl.program_id(0)   # index over (batch * hidden_dim)
    pid_i = tl.program_id(1)   # tile row
    pid_j = tl.program_id(2)   # tile column

    # -------------------- tile offsets --------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (more stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- reduction over k --------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_tiles):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_ptr
            + batch * stride_lm
            + offs_i[:, None] * stride_li
            + offs_k[None, :] * stride_lk,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right block (already transposed): shape (BLOCK_K, BLOCK_N)
        right = tl.load(
            right_ptr
            + batch * stride_rm
            + offs_k[:, None] * stride_ri
            + offs_j[None, :] * stride_rj,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # dot → FP16, cast to FP32 for accumulation
        acc += tl.dot(left, right).to(tl.float32)

    # -------------------- store result --------------------
    tl.store(
        out_ptr
        + batch * stride_om
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N]  (may be ignored)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict containing "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # 0. unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1. Cast constant weights once (FP16 where possible)
    # -----------------------------------------------------------------
    # LayerNorm weights stay in FP32 for numerical stability
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]  # final linear stays FP32

    # Linear / gate weights → FP16 (no bias)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 2. Input LayerNorm (FP32) → cast to FP16 for the rest
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )
    x = x.to(torch.float16)  # FP16 for all subsequent cheap ops

    # -----------------------------------------------------------------
    # 3. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_w) # [B, N, N, H]

    # -----------------------------------------------------------------
    # 4. Optional mask (broadcast on the hidden dimension)
    # -----------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → [B, N, N, 1] → broadcast to hidden dim
        m = mask.to(x.dtype).unsqueeze(-1)
        left = left * m
        right = right * m

    # -----------------------------------------------------------------
    # 5. Gating (sigmoid) and gate application
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()  # [B, N, N, H]

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Prepare tensors for the Triton kernel
    #    Shape:  (B, N, N, H) → (B, H, N, N) → (BH, N, N)
    #    The right tensor is transposed on the last two dims so that the
    #    kernel can perform a plain matmul: left @ right_T.
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)
    right_T = right.transpose(-2, -1).contiguous() # (B, H, N, N) with axes (k, j)

    BH = B * hidden_dim
    left_flat = left.view(BH, N, N)          # [BH, N, N], FP16
    right_flat = right_T.view(BH, N, N)      # [BH, N, N], FP16
    out_flat = torch.empty_like(left_flat)   # storage for result (FP16)

    # -----------------------------------------------------------------
    # 7. Launch the Triton kernel – tuned for H100 FP16 tensor cores
    # -----------------------------------------------------------------
    stride_lm, stride_li, stride_lk = left_flat.stride()
    stride_rm, stride_ri, stride_rj = right_flat.stride()
    stride_om, stride_oi, stride_oj = out_flat.stride()

    # Block sizes: M=N=64 for good register usage, K=128 to halve loop count
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 128

    grid = (
        BH,                                 # batch * hidden
        (N + BLOCK_M - 1) // BLOCK_M,       # number of row tiles
        (N + BLOCK_N - 1) // BLOCK_N,       # number of column tiles
    )

    triton_gram_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        BH,
        N,
        stride_lm,
        stride_li,
        stride_lk,
        stride_rm,
        stride_ri,
        stride_rj,
        stride_om,
        stride_oi,
        stride_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 8. Reshape back to the original layout and cast to FP32
    # -----------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # now FP32 for the final layers

    # -----------------------------------------------------------------
    # 9. Hidden → output LayerNorm, output gate, and final linear projection
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate.to(torch.float32)          # apply output gate
    out = torch.nn.functional.linear(out, to_out_w)  # final projection to dim

    return out
