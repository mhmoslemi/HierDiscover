@triton.jit
def trimul_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, N, hidden_dim,
    stride_batch, stride_i, stride_k,
    # for B and C, we can reuse same strides as they have same layout
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)  # batch*hidden index

    # compute batch and hidden ids
    batch_id = pid_z // hidden_dim
    hidden_id = pid_z % hidden_dim
    base_offset = batch_id * stride_batch + hidden_id

    # offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)

        offs_k = k_base + tl.arange(0, BLOCK_K)

        # load B tile (k,j) - note transpose

"""
TriMul (outgoing) forward‑pass accelerated with Triton.

Algorithm
---------
1️⃣ Layer‑norm the input over the feature dimension `dim`.
2️⃣ Project the normalized tensor to the hidden dimension with two bias‑free
   linear layers (left_proj, right_proj).
3️⃣ Optionally mask the projections (mask shape `[B,N,N]` → `[B,N,N,1]`).
4️⃣ Compute three sigmoid gates (left_gate, right_gate, out_gate) from the same
   normalized tensor and apply them to the projections.
5️⃣ Heavy contraction (O(N³)):
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiplication for each (batch, hidden) pair:
        C = A @ Bᵀ   with   A = left,  B = right.
   The contraction is performed by a custom Triton kernel that:
   • works directly on the native `[B,N,N,H]` layout (no permute / copy);
   • loads the two inputs as FP16, accumulates in FP32 (Tensor‑Core friendly);
   • tiles the computation (BLOCK_M×BLOCK_N) and iterates over K in
     BLOCK_K‑sized chunks;
   • launches a grid of (⌈N/BLOCK_M⌉, ⌈N/BLOCK_N⌉, B·H) tiles, each tile
     handling an independent (batch, hidden) mat‑mul.
6️⃣ Layer‑norm over the hidden dimension.
7️⃣ Multiply by the `out_gate` and project back to `dim` with a final linear
   layer.

Only step 5 is Triton‑implemented; all other steps use high‑performance
PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   for many (batch, hidden) pairs
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    A_ptr,                # *fp16   [B, N, N, H] (input left)
    B_ptr,                # *fp16   [B, N, N, H] (input right)
    C_ptr,                # *fp32   [B, N, N, H] (output)
    N,                    # i32 sequence length
    H,                    # i32 hidden dimension
    stride_batch,         # i64 stride for batch dimension (N*N*H)
    stride_i,             # i64 stride for the first N dimension (N*H)
    stride_k,             # i64 stride for the second N dimension (H)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes, for every (b, h) pair:
        C[b, i, j, h] = Σ_k A[b, i, k, h] * B[b, j, k, h]

    The layout of A/B/C is the native PyTorch layout:
    stride_batch = N*N*H, stride_i = N*H, stride_k = H, stride_h = 1.
    """
    pid_i = tl.program_id(0)          # tile row (i)
    pid_j = tl.program_id(1)          # tile col (j)
    pid_z = tl.program_id(2)          # combined batch*hidden index

    # ------------------------------------------------------------------
    # Decode batch and hidden indices from the combined pid_z
    # ------------------------------------------------------------------
    batch_id = pid_z // H
    hidden_id = pid_z % H
    # Base offset that points to the first element of this (b, h) slice
    base = batch_id * stride_batch + hidden_id

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (the reduction axis)
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- Load A tile (i,k) ------------------------------------------------
        a_ptrs = (
            A_ptr
            + base
            + offs_i[:, None] * stride_i        # i‑stride
            + offs_k[None, :] * stride_k        # k‑stride
        )
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)                     # fp16 load

        # ---- Load B tile (k,j) – note the transposed view --------------------
        b_ptrs = (
            B_ptr
            + base
            + offs_k[:, None] * stride_k        # k‑stride (row of Bᵀ)
            + offs_j[None, :] * stride_i        # j‑stride (col of Bᵀ)
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)                     # fp16 load

        # ---- Dot product: fp16 inputs, fp32 accumulation --------------------
        # tl.dot returns fp32 when the accumulator is fp32.
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write the output tile
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + base
        + offs_i[:, None] * stride_i   # i‑stride
        + offs_j[None, :] * stride_k   # j‑stride (same as stride_k = H)
    )
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def _trimul_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication for the TriMul outgoing contraction.
    `left` and `right` are expected to be contiguous tensors of shape
    [B, N, N, H] in FP16. The result is a FP32 tensor of the same shape.
    """
    B, N, _, H = left.shape
    # Strides in *elements* for the native layout
    stride_batch, stride_i, stride_k, stride_h = left.stride()
    assert stride_h == 1  # hidden dim is innermost

    # Output buffer (FP32)
    out = torch.empty_like(left, dtype=torch.float32)

    # Tile sizes – empirically good for H100 with Tensor‑Cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # (batch * hidden) tiles
    )

    _trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        N,
        H,
        stride_batch,
        stride_i,
        stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out

# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask_tensor   : torch.Tensor [B, N, N]   (may be ignored)
        - weights_dict  : mapping from weight name to torch.Tensor
        - config_dict   : contains keys ``dim``, ``hidden_dim`` and optional
                          ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device as input).
    """
    # -------------------------------------------------
    # Unpack arguments
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ LayerNorm over `dim` (the feature dimension)
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # -------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # -------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------
    # 3️⃣ Optional mask (broadcasted over the hidden dimension)
    # -------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)       # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # -------------------------------------------------
    # 4️⃣ Gating (sigmoid) and application
    # -------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via Triton (FP16 work, FP32 accumulate)
    # -------------------------------------------------
    # Cast to FP16 to exploit Tensor‑Cores; the kernel returns FP32.
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out = _trimul_outgoing(left_fp16, right_fp16)   # [B, N, N, hidden_dim] (fp32)

    # -------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # -------------------------------------------------
    # 7️⃣ Gate and project back to `dim`
    # -------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
