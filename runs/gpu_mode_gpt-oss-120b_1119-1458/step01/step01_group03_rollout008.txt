def custom_kernel(data):
   input_tensor, mask_tensor, weights, config = data
   dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)

   EPS = 1e-5

   # 1. LayerNorm
   x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=EPS)

   # 2. Linear projecting + gating
   # Compute left_proj and right_proj (no bias)
   left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
   right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

   # Mask
   if not nomask:
       mask = mask_tensor.unsqueeze(-1).to(x.dtype)
       left_proj = left_proj * mask
       right_proj = right_proj * mask

   # Gating
   left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
   right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
   out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

   left = left_proj * left_gate
   right = right_proj * right_gate

   # Cast to half for GEMM
   left_h = left.to(torch.float16)
   right_h = right.to(torch.float16)

   # Permute and view
   # We want hidden_dim as separate batch dimension
   B, N, _, _ = left_h.shape
   left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
   right_perm = right_h.permute(0, 3, 1, 2).contiguous()
   BH = B * hidden_dim
   left_flat = left_perm.view(BH, N, N)
   right_flat = right_perm.view(BH, N, N)

   # Output buffer
   out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=left_flat.device)

   # Kernel launch
   # Determine number of programs
   BLOCK_M = 64
   BLOCK_N = 64
   BLOCK_K = 64

   num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
   num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
   tiles_per_batch = num_pid_m * num_pid_n
   total_programs = BH * tiles_per_batch

   trimul_gemm_kernel[total_programs](
       left_flat,
       right_flat,
       out_flat,
       N,
       left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
       right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
       out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
       BLOCK_M=BLOCK_M,
       BLOCK_N=BLOCK_N,
       BLOCK_K=BLOCK_K,
       num_stages=2,
       num_warps=4,
   )

   # Reshape back
   out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]

   # Output Layer Norm
   out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=EPS)

   # Output gating
   out = out * out_gate

   # Final linear projection
   out = torch.nn.functional.linear(out, weights["to_out.weight"])

   return out

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The heavy part of the outgoing TriMul update is the 3‑D tensor contraction

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

for every batch `b` and hidden channel `h`.  This can be expressed as a
batched matrix multiplication `left_h @ right_h.t()`.  The kernel below
implements a fused half‑precision GEMM (fp16×fp16 → fp32 accumulation)
using Tensor‑Core friendly tile sizes (64×64×64).  The rest of the
TriMul pipeline (LayerNorm, linear projections, sigmoid gates and the
final projection) is performed with PyTorch for readability.

Key optimisations
-----------------
* **Tensor‐Core utilisation** – inputs stay in `float16`; the
  accumulator is `float32`.  No `to(tl.float32)` conversion is performed,
  allowing the `tl.dot` primitive to map to H100 Tensor Cores.
* **Tuning for H100** – two autotune configurations are provided;
  the kernel automatically picks the best block sizes for a given
  sequence length.
* **Memory layout** – after the linear projections the tensor is
  permuted to `[B, H, N, N]` (contiguous in the hidden dimension) and
  reshaped to `(B*H, N, N)`.  This yields stride‑0 = `N*N`, stride‑1 = `N`,
  stride‑2 = 1, which matches the GEMM kernel and gives fully coalesced
  loads/stores.
* **Optional masking** – applied before the heavy GEMM, but fused with
  the linear projections using standard PyTorch ops (the mask is tiny
  compared to the O(N³) contraction).

The final output tensor has shape `[B, N, N, dim]` and dtype `float32`.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton GEMM kernel for the TriMul contraction
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        # 64×64×64 tiles, good for most N (up to 1024)
        triton.Config(
            {"BLOCK_M": 64, "BLOCK_N": 64, "BLOCK_K": 64,
             "num_stages": 2, "num_warps": 4},
            num_warps=4,
        ),
        # Larger tiles for very large N – still tensor‑core‑friendly
        triton.Config(
            {"BLOCK_M": 128, "BLOCK_N": 128, "BLOCK_K": 64,
             "num_stages": 3, "num_warps": 8},
            num_warps=8,
        ),
    ],
    key=["N"],  # select config based on the runtime sequence length
)
@triton.jit
def _trimul_gemm_kernel(
    L_ptr, R_ptr, O_ptr,          # pointers for left, right and output
    N,                            # sequence length (runtime)
    stride_bL, stride_iL, stride_kL,
    stride_bR, stride_iR, stride_kR,
    stride_bO, stride_iO, stride_jO,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for one hidden channel:
        O = L @ Rᵀ   with L,R ∈ ℝ^{N×N}
    The batch dimension (combined batch * hidden) is folded into the
    program id.
    """
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # 1) 2‑D tiling over the output matrix (M×N)
    # ------------------------------------------------------------------
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_pid_m * num_pid_n

    batch_idx = pid // tiles_per_batch
    tile_idx = pid % tiles_per_batch
    pid_m = tile_idx // num_pid_n
    pid_n = tile_idx % num_pid_n

    # m and n coordinates of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # 2) Pointers to the current batch slice
    # ------------------------------------------------------------------
    L_batch = L_ptr + batch_idx * stride_bL
    R_batch = R_ptr + batch_idx * stride_bR
    O_batch = O_ptr + batch_idx * stride_bO

    # ------------------------------------------------------------------
    # 3) Accumulator – fp32 to keep numeric stability
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 4) Loop over the reduction dimension k (tiling with BLOCK_K)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # L tile: shape (BLOCK_M, BLOCK_K)
        L_ptrs = L_batch + offs_m[:, None] * stride_iL + offs_k[None, :] * stride_kL
        L = tl.load(
            L_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )                           # still fp16

        # R tile (transposed): we need R[k, j] i.e. (BLOCK_K, BLOCK_N)
        R_ptrs = R_batch + offs_k[:, None] * stride_kR + offs_n[None, :] * stride_iR
        R = tl.load(
            R_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                           # still fp16

        # Tensor‑core dot – fp16 × fp16 → fp32 accumulation
        acc += tl.dot(L, R)

    # ------------------------------------------------------------------
    # 5) Write out the tile
    # ------------------------------------------------------------------
    O_ptrs = O_batch + offs_m[:, None] * stride_iO + offs_n[None, :] * stride_jO
    tl.store(O_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point – the Triton‑accelerated TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool or 0/1)
        - weights_dict : dict of model parameters (see key list below)
        - config_dict  : dict containing at least:
            * "dim"         – input channel dimension
            * "hidden_dim"  – hidden dimension (c_z)
            * "nomask"      – when True the mask tensor is ignored

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm on the channel dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of a separate linear projection)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to half‑precision for the massive GEMM
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 6) Reshape / permute so that hidden dimension is the inner most
    #    stride (contiguous memory).  After the view we have shape
    #    (B*H, N, N) with stride(0)=N*N, stride(1)=N, stride(2)=1.
    # ------------------------------------------------------------------
    B, N, _, _ = left_h.shape
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)    # [BH, N, N], dtype=half
    right_flat = right_perm.view(BH, N, N)  # [BH, N, N], dtype=half

    # ------------------------------------------------------------------
    # 7) Output buffer (fp32 accumulation)
    # ------------------------------------------------------------------
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=left_flat.device)

    # ------------------------------------------------------------------
    # 8) Launch Triton GEMM
    # ------------------------------------------------------------------
    # Tile configuration – the autotuner will pick the best one.
    BLOCK_M = 64   # default; the kernel is autotuned anyway
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute total number of programs (tiles * batch*hidden)
    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    tiles_per_batch = num_pid_m * num_pid_n
    total_programs = BH * tiles_per_batch

    _trimul_gemm_kernel[
        total_programs,
        # The following launch arguments control grid size; the rest are
        # passed as kernel arguments.
    ](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        # autotune‑selected launch options
        num_warps=4,          # overridden by the autotune config if different
    )

    # ------------------------------------------------------------------
    # 9) Reshape back to the original layout: [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 10) Output LayerNorm + gating
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11) Final linear projection back to the original dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
