@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk,
    stride_rb, stride_ri, stride_rk,
    stride_ob, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch and hidden indices
    batch_idx = pid_h // H
    head_idx = pid_h % H

    # base ptr for this head
    left_head_ptr = left_ptr + batch_idx * stride_lb + head_idx
    right_head_ptr = right_ptr + batch_idx * stride_rb + head_idx
    out_head_ptr = out_ptr + batch_idx * stride_ob + head_idx

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_blocks_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_blocks_k):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        left_ptrs = left_head_ptr + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left_mask = mask_i[:, None] & mask_k[None, :]
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # right transposed: we need (K,N)
        # right tile: (K,N) = right_head_ptr[offs_k, offs_j]
        right_ptrs = right_head_ptr + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_ri
        right_mask = mask_k[:, None] & mask_j[None, :]
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        acc += tl.dot(left_tile, right_tile)  # (M,K)@(K,N) -> (M,N)

    out_ptrs = out_head_ptr + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    bs, N, _, dim = input_tensor.shape

    hidden_dim = config["hidden_dim"]
    # case where config may have hidden_dim.

    # Norm
    eps = 1e-5
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    # Linear projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Optional mask
    nomask = config.get("nomask", True)
    if not nomask:
        mask_f = mask_tensor.unsqueeze(-1).type_as(left)  # float32 same as left
        left = left * mask_f
        right = right * mask_f

    # Compute gates (float32)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # dtype float32

    # Cast to half for gating and multiplication
    left = left.to(torch.float16) * left_gate.to(torch.float16)
    right = right.to(torch.float16) * right_gate.to(torch.float16)

    # Output buffer
    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float32, device=input_tensor.device)

    # Compute TriMul via Triton
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        bs * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        bs, N, hidden_dim,
        left.stride(0), left.stride(1), left.stride(2),
        right.stride(0), right.stride(1), right.stride(2),
        out.stride(0), out.stride(1), out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    # Post-process
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out

"""
TriMul – outgoing triangle multiplicative update (AlphaFold‑3)

This implementation follows the reference PyTorch module but moves the
costly tensor contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

to a custom Triton kernel.  The kernel processes each (batch, hidden)
pair as an independent matrix‑multiply C = A @ Bᵀ with tiles
BLOCK_M×BLOCK_K and BLOCK_K×BLOCK_N.  Inputs are stored in fp16,
the accumulation is performed in fp32 (Tensor‑Core friendly) and the
result is written as fp32.  By indexing the tensors directly (no
permute / view copy) we avoid an O(B·N²·H) memory move and obtain a
large speed‑up on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: compute one (batch, hidden) slice:
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr,          # fp16  [B,N,N,H]
    right_ptr,         # fp16  [B,N,N,H]
    out_ptr,           # fp32  [B,N,N,H]
    B, N, H,           # i32   batch, sequence length, hidden dim
    stride_lb, stride_li, stride_lk,   # i64   strides for left   (batch,i,k)
    stride_rb, stride_ri, stride_rk,   # i64   strides for right  (batch,j,k)
    stride_ob, stride_oi, stride_oj,   # i64   strides for out    (batch,i,j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(0)            # combined (batch, hidden)
    pid_m = tl.program_id(1)            # block row   (i)
    pid_n = tl.program_id(2)            # block column(j)

    # --------------------------------------------------------------
    # Resolve batch index and hidden (head) index from pid_h
    # --------------------------------------------------------------
    batch_idx = pid_h // H
    head_idx  = pid_h % H

    # Base pointers for the current (batch, hidden) slice
    left_head_ptr  = left_ptr  + batch_idx * stride_lb + head_idx
    right_head_ptr = right_ptr + batch_idx * stride_rb + head_idx
    out_head_ptr   = out_ptr   + batch_idx * stride_ob + head_idx

    # --------------------------------------------------------------
    # Tile offsets for the i and j dimensions
    # --------------------------------------------------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Accumulator in fp32
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k in blocks
    # --------------------------------------------------------------
    num_blocks_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_blocks_k):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile  : shape (BLOCK_M, BLOCK_K)  = left[i,k]
        left_ptrs = left_head_ptr \
            + offs_i[:, None] * stride_li \
            + offs_k[None, :] * stride_lk
        left_mask = mask_i[:, None] & mask_k[None, :]
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # right tile : shape (BLOCK_K, BLOCK_N) = right[k,j]  (transposed)
        right_ptrs = right_head_ptr \
            + offs_k[:, None] * stride_rk \
            + offs_j[None, :] * stride_ri
        right_mask = mask_k[:, None] & mask_j[None, :]
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # Matrix‑multiply‑accumulate using Tensor Cores
        acc += tl.dot(left_tile, right_tile)       # (M,K) @ (K,N) -> (M,N)

    # --------------------------------------------------------------
    # Store the computed block back to global memory
    # --------------------------------------------------------------
    out_ptrs = out_head_ptr \
        + offs_i[:, None] * stride_oi \
        + offs_j[None, :] * stride_oj
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)


# ----------------------------------------------------------------------
# Public entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.
    Args:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor  [B, N, N, dim]   (float32)
            mask         : torch.Tensor  [B, N, N]       (float32/bool)
            weights      : dict of model parameters
            config       : dict containing "dim", "hidden_dim", "nomask" ...
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    inp, mask, weights, config = data
    device = inp.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5

    # --------------------------------------------------------------
    # LayerNorm over the channel dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]   float32

    # --------------------------------------------------------------
    # Linear projections (no bias) → shape (B, N, N, hidden_dim)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # Optional masking (mask shape [B,N,N])
    # --------------------------------------------------------------
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # Gating values (sigmoid of a linear projection)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # keep fp32

    # --------------------------------------------------------------
    # Apply gates (all in fp16)
    # --------------------------------------------------------------
    left = left.to(torch.float16) * left_gate
    right = right.to(torch.float16) * right_gate

    # --------------------------------------------------------------
    # Ensure tensors are contiguous for the Triton kernel
    # --------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape

    # --------------------------------------------------------------
    # Output buffer for the kernel (fp32)
    # --------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Triton launch configuration – tuned for H100 Tensor‑Core cores
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64                       # a power of two → fewer K‑loops

    grid = (
        B * hidden_dim,                                 # program id 0  (batch * hidden)
        (N + BLOCK_M - 1) // BLOCK_M,                   # program id 1  (i‑blocks)
        (N + BLOCK_N - 1) // BLOCK_N,                   # program id 2  (j‑blocks)
    )

    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        left.stride(0), left.stride(1), left.stride(2),
        right.stride(0), right.stride(1), right.stride(2),
        out.stride(0), out.stride(1), out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # H100 can sustain 8 warps per block efficiently
        num_stages=4,         # double‑buffering for latency hiding
    )

    # --------------------------------------------------------------
    # Post‑processing: layer‑norm → out‑gate → final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                      # broadcast over hidden_dim
    out = F.linear(out, weights["to_out.weight"])

    return out
