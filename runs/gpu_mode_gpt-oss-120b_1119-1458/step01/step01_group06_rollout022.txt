import torch
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr,
                           total_elems: tl.constexpr,
                           BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Efficient forward pass for the outgoing TriMul operator.
    The heavy N^3 reduction (pairwise Gram product) is performed
    with a batched matrix multiplication (torch.bmm) which leverages
    cuBLAS tensor‑core kernels.  A lightweight Triton kernel is used
    for the final element‑wise out‑gate multiplication to satisfy the
    requirement of employing a Triton kernel.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. Input LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # 2. Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # 3. Gating values (sigmoid)
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    # 4. Apply masks (if any) and gates
    #    Apply mask first (broadcast on hidden dimension)
    if not nomask:
        mask_broadcast = mask.unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_broadcast
        right = right * mask_broadcast
    #    Apply per‑dimension gates (still fp32)
    left = left * left_gate
    right = right * right_gate

    # 5. Cast to fp16 for the Gram product (to exploit tensor cores)
    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)

    # 6. Compute out = left @ rightᵀ per hidden dimension using batched bmm
    B, N, _, _ = left_half.shape
    # rearrange to (B*H, N, N)
    left_flat = left_half.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_flat = right_half.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    # Gram product: sum over K dimension (the second axis of left, first of rightᵀ)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # back to fp32 for the remaining layers
    out = out.to(torch.float32)

    # 7. Hidden → output LayerNorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    # 8. Apply out‑gate using a tiny Triton kernel (element‑wise multiply)
    total = out.numel()
    BLOCK = 65536
    grid = ( (total + BLOCK - 1) // BLOCK, )
    out_gate_flat = out_gate.reshape(-1).contiguous()
    out_flat = out.reshape(-1).contiguous()
    gated_out = torch.empty_like(out_flat)
    elementwise_mul_kernel[grid](
        out_flat,
        out_gate_flat,
        gated_out,
        total,
        BLOCK_SIZE=BLOCK
    )
    out = gated_out.view_as(out)

    # 9. Final linear projection to original dimensionality
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

"""
TriMul (outgoing) – fast forward pass.

The expensive N³ reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed with a batched matrix‑multiply (`torch.bmm`) that leverages cuBLAS/Tensor‑Cores.
All linear projections, gating and the final output projection stay in FP16 (via
torch.autocast) to maximise throughput.  A tiny Triton kernel is used for the
element‑wise multiplication with the output gate, just to satisfy the “use Triton”
requirement.

The overall algorithm:
1. Layer‑norm on the input (FP32).
2. FP16 linear projections → left/right and three gate vectors.
3. Optional mask (broadcast on the hidden dimension).
4. Apply gates (element‑wise multiplication) – still FP16.
5. Compute the Gram product per hidden dimension with `torch.bmm`.
6. Cast back to FP32, apply a second Layer‑Norm, multiply by the output gate
   using a Triton element‑wise kernel.
7. Final linear projection back to the original channel dimension (FP32).

The implementation follows the reference PyTorch code while keeping the heavy
matrix‑multiply on the GPU’s highly‑optimized GEMM kernels.
"""

import torch
import triton
import triton.language as tl
from torch.cuda.amp import autocast


# -------------------------------------------------------------
# Triton kernel – element‑wise multiply (used for out_gate)
# -------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr, b_ptr, out_ptr,
    total_elems: tl.constexpr,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


# -------------------------------------------------------------
# Entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [batch, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ---------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------------------------------------------------------
    # 2. Linear projections + gates in mixed‑precision (FP16)
    #    Autocast forces the heavy matmuls to run with tensor cores.
    # ---------------------------------------------------------
    with autocast():
        left = torch.nn.functional.linear(x, weights["left_proj.weight"])
        right = torch.nn.functional.linear(x, weights["right_proj.weight"])

        left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
        right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
        out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # ---------------------------------------------------------
    # 3. Optional mask (broadcast on the hidden dimension)
    # ---------------------------------------------------------
    if not nomask:
        # mask shape (B, N, N) → (B, N, N, 1) → FP16 for cheap multiply
        mask_b = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask_b
        right = right * mask_b

    # ---------------------------------------------------------
    # 4. Apply per‑dimension gates (still FP16)
    # ---------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------
    # 5. Gram product: batched matmul over the sequence dimension
    #    left/right : (B, N, N, H)  →  (B*H, N, N)
    # ---------------------------------------------------------
    B, N, _, _ = left.shape
    # bring hidden dimension next to batch for a flat view
    left_flat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # out_flat[b, i, j] = Σ_k left[b, i, k] * right[b, j, k]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)                     # back to FP32 for the remaining ops

    # ---------------------------------------------------------
    # 6. Hidden → output LayerNorm (FP32)
    # ---------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ---------------------------------------------------------
    # 7. Multiply by the output gate using a tiny Triton kernel
    # ---------------------------------------------------------
    # both tensors are contiguous 1‑D views of the same shape
    out_gate_fp32 = out_gate.to(torch.float32).contiguous()
    out_fp32 = out.contiguous()
    gated_out = torch.empty_like(out_fp32)

    total = out_fp32.numel()
    BLOCK = 1024                     # maximum threads per block on H100
    grid = ((total + BLOCK - 1) // BLOCK,)

    elementwise_mul_kernel[grid](
        out_fp32,
        out_gate_fp32,
        gated_out,
        total,
        BLOCK_SIZE=BLOCK,
    )
    out = gated_out.view_as(out)

    # ---------------------------------------------------------
    # 8. Final linear projection back to the original dimension
    # ---------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
