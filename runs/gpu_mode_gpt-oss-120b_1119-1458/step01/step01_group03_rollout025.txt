@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_k,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    # Compute total tiles per batch
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_tiles_m * num_tiles_n

    batch = pid // tiles_per_batch
    tile = pid % tiles_per_batch
    tile_i = tile // num_tiles_n
    tile_j = tile % num_tiles_n

    # offsets
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # For each hidden block
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

        # reduce over K
        for kk in range(0, tl.cdiv(N, BLOCK_K)):
            k_base = kk * BLOCK_K
            offs_k = k_base + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # iterate over k within tile
            for k_idx in range(0, BLOCK_K):
                cur_k = offs_k[k_idx]
                cur_k_valid = mask_k[k_idx]

                # left load: shape (BLOCK_M, BLOCK_H)
                L_ptrs = left_ptr + batch * stride_b \
                             + offs_i[:, None] * stride_i \
                             + cur_k * stride_k \
                             + offs_h[None, :]
                L = tl.load(L_ptrs,
                            mask=mask_i[:, None] & mask_h[None, :] & cur_k_valid,
                            other=0.0).to(tl.float32)

                # right load: shape (BLOCK_N, BLOCK_H)
                R_ptrs = right_ptr + batch * stride_b \
                             + offs_j[:, None] * stride_i \
                             + cur_k * stride_k \
                             + offs_h[None, :]
                R = tl.load(R_ptrs,
                            mask=mask_j[:, None] & mask_h[None, :] & cur_k_valid,
                            other=0.0).to(tl.float32)

                # outer product add
                # broadcast:
                acc += L[:, None, :] * R[None, :, :]  # shape (BLOCK_M,BLOCK_N,BLOCK_H)

        # store results
        O_ptrs = out_ptr + batch * stride_b \
                     + offs_i[:, None, None] * stride_i \
                     + offs_j[None, :, None] * stride_k \
                     + offs_h[None, None, :]
        tl.store(O_ptrs,
                 acc,
                 mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_k,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr):
    # program ids
    pid = tl.program_id(0)   # batch*tile i j
    hid = tl.program_id(1)   # hidden block index

    # compute tile indices
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_tiles_m * num_tiles_n

    batch = pid // tiles_per_batch
    tile_id = pid % tiles_per_batch
    tile_i = tile_id // num_tiles_n
    tile_j = tile_id % num_tiles_n

    # i/j offsets

"""
TriMul (outgoing) – Triton‑accelerated forward pass

The heavy tensor contraction
    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
is rewritten as a tiled rank‑1 update:
    for each hidden‑channel block h ∈ H,
        for each i‑tile × j‑tile,
            acc[i,j,h] += Σ_k left[i,k,h] * right[j,k,h]

All element‑wise operations (layer‑norm, linear projections,
gating and masking) are performed with PyTorch; the
contraction itself is computed by a custom Triton kernel that
operates directly on the original layout (B,N,N,H) and therefore
avoids the expensive permute + copy performed in the reference
implementation.  The kernel works on a 3‑D tile
(M×N×Hblock) and uses fp16 inputs with fp32 accumulation.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,                      # half *, shape [B, N, N, H]
    right_ptr,                     # half *, shape [B, N, N, H]
    out_ptr,                       # float *, shape [B, N, N, H]
    B, N, H,                       # runtime sizes
    stride_b, stride_i, stride_k,  # strides for (B,N,N,H) layout
    BLOCK_M: tl.constexpr,         # tile size for i dimension
    BLOCK_N: tl.constexpr,         # tile size for j dimension
    BLOCK_K: tl.constexpr,         # reduction tile size (over k)
    BLOCK_H: tl.constexpr,         # tile size for hidden dimension
):
    """
    Batched rank‑1 update for the TriMul “outgoing” path.
    Each program (pid) processes one (batch,tile_i,tile_j) tile.
    The second program axis iterates over hidden‑channel blocks.
    """
    pid = tl.program_id(0)          # batch * tile_i * tile_j
    hid = tl.program_id(1)          # hidden‑block index

    # ------------------------------------------------------------
    # 1) Compute tile coordinates
    # ------------------------------------------------------------
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_tiles_m * num_tiles_n

    batch = pid // tiles_per_batch
    tile_id = pid % tiles_per_batch
    tile_i = tile_id // num_tiles_n
    tile_j = tile_id % num_tiles_n

    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------
    # 2) Offsets for the hidden‑channel block handled by this program
    # ------------------------------------------------------------
    h_start = hid * BLOCK_H
    offs_h = h_start + tl.arange(0, BLOCK_H)
    mask_h = offs_h < H

    # ------------------------------------------------------------
    # 3) Accumulator (fp32) for the tile (M×N×Hblock)
    # ------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # base offset of the current batch
    batch_offset = batch * stride_b

    # ------------------------------------------------------------
    # 4) Reduction over the sequence dimension k
    # ------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # iterate inside the K‑tile element‑wise – this yields a rank‑1
        # update for each k and keeps the inner loops simple.
        for kk_idx in range(0, BLOCK_K):
            cur_k = offs_k[kk_idx]
            cur_k_valid = mask_k[kk_idx]

            # ----------------------------------------------------
            # Load left[i,k,:]  → shape (BLOCK_M, BLOCK_H)
            # ----------------------------------------------------
            L_ptrs = left_ptr + batch_offset \
                     + offs_i[:, None] * stride_i \
                     + cur_k * stride_k \
                     + offs_h[None, :]
            L = tl.load(
                L_ptrs,
                mask=mask_i[:, None] & mask_h[None, :] & cur_k_valid,
                other=0.0,
            ).to(tl.float32)           # fp16 → fp32

            # ----------------------------------------------------
            # Load right[j,k,:] → shape (BLOCK_N, BLOCK_H)
            # ----------------------------------------------------
            R_ptrs = right_ptr + batch_offset \
                     + offs_j[:, None] * stride_i \
                     + cur_k * stride_k \
                     + offs_h[None, :]
            R = tl.load(
                R_ptrs,
                mask=mask_j[:, None] & mask_h[None, :] & cur_k_valid,
                other=0.0,
            ).to(tl.float32)

            # ----------------------------------------------------
            # Outer‑product update: acc[i,j,h] += L[i,h] * R[j,h]
            # ----------------------------------------------------
            # L: (M, H) → (M, 1, H), R: (N, H) → (1, N, H)
            acc += L[:, None, :] * R[None, :, :]

    # ------------------------------------------------------------
    # 5) Store the result back to the output tensor (fp32)
    # ------------------------------------------------------------
    O_ptrs = out_ptr + batch_offset \
               + offs_i[:, None, None] * stride_i \
               + offs_j[None, :, None] * stride_k \
               + offs_h[None, None, :]
    tl.store(
        O_ptrs,
        acc,
        mask=mask_i[:, None, None] &
             mask_j[None, :, None] &
             mask_h[None, None, :]
    )


def custom_kernel(data):
    """
    Triton‑accelerated TriMul forward (outgoing) pass.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0) Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm on the channel dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) → [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    # Cast weights once to the dtype of the input (float32)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of another linear projection)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to half‑precision for the heavy GEMM
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, H = left_h.shape  # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 6) Output buffer for the contraction (float32 accumulator)
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left_h.device)

    # ------------------------------------------------------------------
    # 7) Triton GEMM launch
    # ------------------------------------------------------------------
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_H = 16

    # grid dimensions: (programs over i/j tiles, hidden blocks)
    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = B * num_tiles_m * num_tiles_n
    num_h_blocks = (H + BLOCK_H - 1) // BLOCK_H

    trimul_outgoing_kernel[(total_programs, num_h_blocks)](
        left_h,
        right_h,
        out,
        B,
        N,
        H,
        left_h.stride(0),   # stride over batch
        left_h.stride(1),   # stride over first N dimension (i)
        left_h.stride(2),   # stride over second N dimension (k/j)
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        BLOCK_H,
        num_warps=8,        # good occupancy on H100
    )

    # ------------------------------------------------------------------
    # 8) Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 9) Output gating
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10) Final linear projection back to the original dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
