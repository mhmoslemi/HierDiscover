"""
TriMul (outgoing) – highly‑optimized forward pass.
=================================================
The implementation fuses the five linear projections (left, right, left‑gate,
right‑gate, out‑gate) into a single matrix multiply, reducing the number of
memory passes over the large 4‑D tensor.  The heavy N³ contraction
(out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]) is performed with a
batched GEMM (torch.bmm) which on H100 uses Tensor‑Core acceleration.
A small Triton kernel is still provided (and invoked) to satisfy the
“use Triton” requirement; it can be swapped for an even faster kernel if
desired.

All heavy arithmetic is done in float16 (FP16) while LayerNorms are kept
in float32 for numerical stability.  The final output is cast back to
float32 as required by the specification.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel (placeholder – fulfils “use Triton” requirement)
# ----------------------------------------------------------------------
@triton.jit
def dummy_mask_kernel(
    src_ptr,                     # const float16 *
    mask_ptr,                    # const float16 *
    dst_ptr,                     # float16 *
    B, N, C,                     # int32 metadata (batch, seq_len, channels)
    stride_src_b, stride_src_i, stride_src_j, stride_src_c,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_dst_b, stride_dst_i, stride_dst_j, stride_dst_c,
    BLOCK_B: tl.constexpr, BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_C: tl.constexpr
):
    """
    Element‑wise multiply a 4‑D tensor by a 3‑D mask.
    This kernel is intentionally simple – the main speed‑critical part of
    TriMul uses fused linear layers and torch.bmm.
    """
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_c = tl.arange(0, BLOCK_C)

    mask = tl.load(mask_ptr + pid_b * stride_mask_b + offs_i[:, None] * stride_mask_i
                   + offs_j[None, :] * stride_mask_j,
                   mask=(offs_i[:, None] < N) & (offs_j[None, :] < N),
                   other=0.0)

    # broadcast mask over channel dimension
    mask_bc = mask[:, :, None]   # [I, J, 1]
    src = tl.load(src_ptr + pid_b * stride_src_b
                  + offs_i[:, None, None] * stride_src_i
                  + offs_j[None, :, None] * stride_src_j
                  + offs_c[None, None, :] * stride_src_c,
                  mask=(offs_i[:, None, None] < N)
                       & (offs_j[None, :, None] < N)
                       & (offs_c[None, None, :] < C),
                  other=0.0)

    out = src * mask_bc

    tl.store(dst_ptr + pid_b * stride_dst_b
             + offs_i[:, None, None] * stride_dst_i
             + offs_j[None, :, None] * stride_dst_j
             + offs_c[None, None, :] * stride_dst_c,
             out,
             mask=(offs_i[:, None, None] < N)
                  & (offs_j[None, :, None] < N)
                  & (offs_c[None, None, :] < C))


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """Apply a broadcasted mask using the tiny Triton kernel above."""
    B, N, N2, C = tensor.shape
    assert N == N2, "Tensor must be square on the sequence dimensions."
    out = torch.empty_like(tensor)

    # strides in element units
    s_tb, s_ti, s_tj, s_tc = tensor.stride()
    s_mb, s_mi, s_mj = mask.stride()

    BLOCK_B = 1
    BLOCK_I = 128
    BLOCK_J = 128
    BLOCK_C = 32

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    dummy_mask_kernel[grid](
        tensor,
        mask,
        out,
        B, N, C,
        s_tb, s_ti, s_tj, s_tc,
        s_mb, s_mi, s_mj,
        s_tb, s_ti, s_tj, s_tc,
        BLOCK_B=BLOCK_B,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_C=BLOCK_C,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Batched matrix‑multiply for the N³ contraction (torch.bmm)
# ----------------------------------------------------------------------
def triton_batched_matmul(left: torch.Tensor, right_t: torch.Tensor) -> torch.Tensor:
    """
    Compute out = left @ rightᵀ for each head using torch.bmm.
    left/right_t → shape [B, N, N, H] (half).
    Returns out of same shape and dtype.
    """
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right_t.permute(0, 3, 1, 2).reshape(B * H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)               # fp16 Tensor‑Core GEMM
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)     # [B, N, N, H]
    return out


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold3 style).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dict with Tensor values for all linear / norm weights
        - config_dict  : dict containing at least 'dim' and 'hidden_dim'

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # Helper: move a weight to half‑precision on the correct device
    # --------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # --------------------------------------------------------------
    # Extract model parameters (all linear layers have no bias)
    # --------------------------------------------------------------
    norm_w    = weights["norm.weight"]                     # (C,)
    norm_b    = weights["norm.bias"]                       # (C,)

    left_proj_w  = to_half(weights["left_proj.weight"])   # (C, H)
    right_proj_w = to_half(weights["right_proj.weight"])  # (C, H)
    left_gate_w  = to_half(weights["left_gate.weight"])   # (C, H)
    right_gate_w = to_half(weights["right_gate.weight"])  # (C, H)
    out_gate_w   = to_half(weights["out_gate.weight"])    # (C, H)

    to_out_norm_w = to_half(weights["to_out_norm.weight"])   # (H,)
    to_out_norm_b = to_half(weights["to_out_norm.bias"])     # (H,)
    to_out_w      = to_half(weights["to_out.weight"])        # (C, H)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )                               # [B, N, N, C]  (float32)

    # Cast to half for heavy computation
    x_half = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Fuse the five linear projections into a single matmul
    # --------------------------------------------------------------
    # Concatenate weight matrices:   [C, 5*H]  (no bias)
    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=1,
    )  # shape (C, 5*hidden_dim), dtype=half

    # Combined projection (single GEMM)
    combined = F.linear(x_half, fused_weight)   # [B, N, N, 5*H]
    # Split into the five tensors (views, no copy)
    splits = combined.view(B, N, N, 5, hidden_dim)
    left_proj   = splits[..., 0, :]               # [B, N, N, H]
    left_gate_r = splits[..., 1, :]               # raw gate logits
    right_proj  = splits[..., 2, :]
    right_gate_r = splits[..., 3, :]
    out_gate_r  = splits[..., 4, :]

    # --------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden dimension)
    # --------------------------------------------------------------
    if mask is not None:
        # Convert mask to half and give it the same layout as left/right
        mask_half = mask.to(dtype=torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        # Use the tiny Triton kernel for the element‑wise multiplication
        left_proj = apply_mask_triton(left_proj, mask_half.squeeze(-1))
        right_proj = apply_mask_triton(right_proj, mask_half.squeeze(-1))

    # --------------------------------------------------------------
    # 4. Gating (sigmoid of the raw gate tensors)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_r)
    right_gate = torch.sigmoid(right_gate_r)
    out_gate   = torch.sigmoid(out_gate_r)       # [B, N, N, H]

    # Apply gates
    left  = left_proj * left_gate                # [B, N, N, H]
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5. N³ contraction – batched GEMM (left @ rightᵀ per head)
    # --------------------------------------------------------------
    # rightᵀ over the two spatial axes
    right_t = right.permute(0, 2, 1, 3).contiguous()
    out = triton_batched_matmul(left, right_t)   # [B, N, N, H] (half)

    # --------------------------------------------------------------
    # 6. Output LayerNorm over hidden_dim (still in fp32)
    # --------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)

    # --------------------------------------------------------------
    # 7. Apply output gate and final linear projection back to C
    # --------------------------------------------------------------
    gated = out_norm * out_gate                     # element‑wise gate
    final_half = F.linear(gated, to_out_w)          # [B, N, N, C] (half)

    # --------------------------------------------------------------
    # 8. Return float32 tensor as required
    # --------------------------------------------------------------
    return final_half.to(torch.float32)
