"""
TriMul (outgoing) – forward only.
The core N³ contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is implemented as a batched matrix‑multiply (left @ rightᵀ) with a
custom Triton kernel.  All other operations (layer‑norm, linear
projections, gates) run with the highly‑optimised PyTorch kernels in
fp16.  The final tensor is returned in fp32 as required by the API.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the N³ contraction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,                # const half *
    right_ptr,               # const half *
    out_ptr,                 #       half *
    B, N, H,                 # int32 (batch, seq_len, hidden)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched GEMM for each (batch, hidden) pair:
        out = left @ rightᵀ
    left  : [B, N_i, N_k, H]
    right : [B, N_j, N_k, H]   (we implicitly transpose the N_k axis)
    out   : [B, N_i, N_j, H]
    """
    # ------------------------------------------------------------------
    # program IDs
    # ------------------------------------------------------------------
    pid_head = tl.program_id(0)   # encodes (batch, hidden)
    pid_m    = tl.program_id(1)   # block row (i)
    pid_n    = tl.program_id(2)   # block col (j)

    total_heads = B * H
    b = pid_head // H             # batch index
    h = pid_head %  H             # hidden index

    # ------------------------------------------------------------------
    # tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for the current (b, h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over the reduction (k) dimension
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (i, k)
        left_ptrs = left_base + \
            offs_i[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile (transposed on the fly): (k, j)
        # right is stored as (j, k); we index it as (k, j)
        right_ptrs = right_base + \
            offs_k[:, None] * stride_right_k + \
            offs_j[None, :] * stride_right_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # accumulate the dot product (M×K)·(K×N) → (M×N)
        acc += tl.dot(left, right)

    # ------------------------------------------------------------------
    # write back the tile
    # ------------------------------------------------------------------
    out_ptrs = out_base + \
        offs_i[:, None] * stride_out_i + \
        offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton‑accelerated batched matrix multiplication:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Arguments
    ---------
    left  : torch.Tensor of shape [B, N, N, H] (dtype=half, CUDA)
    right : torch.Tensor of same shape (dtype=half, CUDA)

    Returns
    -------
    out   : torch.Tensor of shape [B, N, N, H] (dtype=half, CUDA)
    """
    assert left.is_cuda and right.is_cuda
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides are in element units (as Triton expects)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Blocking – tuned for Hopper (H100)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64      # larger K tile reduces loop overhead

    grid = (
        B * H,                         # (batch * hidden) dimension
        (N + BLOCK_M - 1) // BLOCK_M,  # rows
        (N + BLOCK_N - 1) // BLOCK_N,  # cols
    )

    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,       # Hopper likes 8 warps per block
        num_stages=2,      # double‑buffering of loads
    )
    return out

    left_gate_w   = to_half(weights["left_gate.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(input_tensor,
                          normalized_shape=(dim,),
                          weight=norm_w,
                          bias=norm_b,
                          eps=1e-5)          # [B,N,N,C] float32

    # ------------------------------------------------------------------
    # 2. Cast to half for the heavy part of the computation
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)   # [B,N,N,hidden_dim]  fp16
    right = F.linear(x_h, right_proj_w)  # [B,N,N,hidden_dim]  fp16

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))  # [B,N,N,hidden_dim]
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))   # [B,N,N,hidden_dim]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. N³ contraction: batched matmul (left @ rightᵀ) via Triton
    # ------------------------------------------------------------------
    # The kernel reads `right` in a transposed view on‑the‑fly, so no
    # extra memory copy is required.
    out = tri_mul_triton(left, right)   # [B,N,N,hidden_dim] fp16

    # ------------------------------------------------------------------
    # 7. Output LayerNorm over the hidden dimension (still in fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)

    # ------------------------------------------------------------------
    # 8. Apply output gate and final linear projection back to C
    # ------------------------------------------------------------------
    out = out * out_gate                     # gated hidden representation
    out = F.linear(out, to_out_w)            # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 9. Return fp32 as required by the specification
    # ------------------------------------------------------------------
    return out.to(torch.float32)
