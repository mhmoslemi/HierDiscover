"""
TriMul (outgoing) – fast Triton‑fused implementation
----------------------------------------------------
Algorithm
~~~~~~~~~
1. **Layer‑norm** of the input tensor `x` (shape [B,N,N,D]).
2. **Single linear projection** that concatenates the five weight matrices
   (left_proj, right_proj, left_gate, right_gate, out_gate) into one
   matrix `W_cat` of shape (5*H, D).  This yields a tensor
   `cat = x_norm @ W_cat.T` of shape [B,N,N,5*H].
3. **Fused Triton kernel** extracts the five slices, applies a
   sigmoid to the three gate slices and multiplies the left/right
   slices by their respective gates and an optional mask.  The kernel
   writes three tensors: `left`, `right` and `out_gate` (all
   [B,N,N,H]).
4. **Core multiplicative update**:
        out[i,j,:] = Σ_k left[i,k,:] * right[j,k,:]
   This is a batched GEMM for each hidden channel (implemented with
   `torch.bmm` on FP16 tensors).
5. **Output layer‑norm**, multiplication by the already‑sigmoided
   `out_gate`, and a final linear projection back to the original
   channel dimension `D`.

Only the element‑wise gating + mask step is written in Triton; all
large matrix‑multiplications are delegated to highly‑optimised PyTorch
kernels (which on H100 utilise Tensor‑cores).  This reduces the number
of memory passes compared to the naïve reference implementation and
meets the requirement of “at least part of the operations in a kernel”.

The function `custom_kernel` follows the required signature:
    (input, mask, weights, config) → output.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse sigmoid‑gates, optional mask and final elementwise
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_fused_kernel(
    # input tensors (flattened, shape [total, H])
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr, out_gate_pre_ptr,
    # mask (flattened scalar per element)
    mask_ptr,
    # output tensors (flattened, shape [total, H])
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    # total number of elements = B * N * N
    total: tl.int32,
    # compile‑time constant: hidden dimension
    H: tl.constexpr,
):
    pid = tl.program_id(0)
    if pid >= total:
        return

    # offset of the first hidden channel for this (b,i,j) element
    off = pid * H
    hid = tl.arange(0, H)                     # [0 ... H-1]

    # load the five slices
    left_proj   = tl.load(left_proj_ptr   + off + hid)
    right_proj  = tl.load(right_proj_ptr  + off + hid)
    left_gate_p = tl.load(left_gate_ptr   + off + hid)
    right_gate_p= tl.load(right_gate_ptr + off + hid)
    out_gate_p  = tl.load(out_gate_pre_ptr+ off + hid)

    # mask scalar (0/1) – broadcasted across H
    mask_val = tl.load(mask_ptr + pid)

    # sigmoid for the three gates
    left_gate  = 1.0 / (1.0 + tl.exp(-left_gate_p))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_p))
    out_gate   = 1.0 / (1.0 + tl.exp(-out_gate_p))

    # final left / right after gating and optional masking
    left  = left_proj  * left_gate  * mask_val
    right = right_proj * right_gate * mask_val

    # write results
    tl.store(left_out_ptr   + off + hid, left)
    tl.store(right_out_ptr  + off + hid, right)
    tl.store(out_gate_out_ptr+ off + hid, out_gate)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, D]    (float32, on CUDA)
            - mask_tensor:  [B, N, N]      (float32/bool, on CUDA) – may be None
            - weights_dict: model weights (see skeleton)
            - config_dict:  contains at least "dim", "hidden_dim",
                            optional "eps" and "nomask".

    Returns:
        Tensor of shape [B, N, N, D] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, config = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Layer‑norm on the input (affine)
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, D]

    # --------------------------------------------------------------
    # 2) Concatenated linear projection (produces 5*H channels)
    # --------------------------------------------------------------
    # concat order: left_proj, right_proj, left_gate, right_gate, out_gate
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).contiguous()                                   # (5*H, D)

    concat = F.linear(x_norm, concat_weight)          # [B, N, N, 5*H]

    # split into the five required tensors
    left_proj, right_proj, left_gate_pre, right_gate_pre, out_gate_pre = torch.split(
        concat, hidden_dim, dim=-1
    )   # each [B,N,N,H]

    # --------------------------------------------------------------
    # 3) Fuse sigmoid‑gates, mask and final elementwise multiplication
    # --------------------------------------------------------------
    total_elements = B * N * N

    # flatten tensors to shape [total, H] for the kernel
    left_proj_f = left_proj.contiguous().view(-1, hidden_dim)
    right_proj_f = right_proj.contiguous().view(-1, hidden_dim)
    left_gate_pre_f = left_gate_pre.contiguous().view(-1, hidden_dim)
    right_gate_pre_f = right_gate_pre.contiguous().view(-1, hidden_dim)
    out_gate_pre_f = out_gate_pre.contiguous().view(-1, hidden_dim)

    # prepare mask – a scalar per (b,i,j) element
    if nomask or mask is None:
        mask_f = torch.ones(total_elements, dtype=inp.dtype, device=device)
    else:
        mask_f = mask.contiguous().view(-1).to(inp.dtype)

    # output buffers
    left_f = torch.empty_like(left_proj_f)
    right_f = torch.empty_like(right_proj_f)
    out_gate_f = torch.empty_like(out_gate_pre_f)

    # launch kernel; one program per element, one thread per program
    grid = (triton.cdiv(total_elements, 1),)
    gate_mask_fused_kernel[grid](
        left_proj_f,
        right_proj_f,
        left_gate_pre_f,
        right_gate_pre_f,
        out_gate_pre_f,
        mask_f,
        left_f,
        right_f,
        out_gate_f,
        total_elements,
        H=hidden_dim,
        BLOCK_SIZE=1,
    )

    # reshape back to [B,N,N,H]
    left = left_f.view(B, N, N, hidden_dim)
    right = right_f.view(B, N, N, hidden_dim)
    out_gate = out_gate_f.view(B, N, N, hidden_dim)   # already sigmoid‑ed

    # --------------------------------------------------------------
    # 4) Core contraction: out[i,j,:] = Σ_k left[i,k,:] * right[j,k,:]
    #    (batched GEMM, performed in FP16 for Tensor‑core speed)
    # --------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # reshape to (B*H, N, N) – left: (i,k), right: (j,k) → we need right.T
    left_mat = left_fp16.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_mat = right_fp16.permute(0, 3, 2, 1).contiguous() # [B, H, N, N]

    left_flat = left_mat.view(B * hidden_dim, N, N)
    right_flat = right_mat.view(B * hidden_dim, N, N)

    out_flat = torch.bmm(left_flat, right_flat)            # [B*H, N, N] (FP16)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(inp.dtype)                               # back to FP32

    # --------------------------------------------------------------
    # 5) Output layer‑norm, output‑gate, final linear projection
    # --------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # [B, N, N, H]

    # multiplicative output gate (already sigmoid‑ed)
    out_norm = out_norm * out_gate

    # final linear projection back to dim
    out_final = F.linear(out_norm, weights["to_out.weight"])   # [B,N,N,dim]

    return out_final
