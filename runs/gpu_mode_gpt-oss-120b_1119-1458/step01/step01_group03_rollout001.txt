def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # 1. Norm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weights["norm.weight"], weights["norm.bias"], eps)

    # 2. Proj
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # 3. Mask
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # 4. Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # 5. Cast to half
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Reshape for gemm
    B, N, _, _ = left_h.shape
    left_perm = left_h.permute(0,3,1,2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right_h.permute(0,3,1,2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)   # half
    right_flat = right_perm.view(BH, N, N)  # half

    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=left_flat.device)

    # Determine block sizes (tuned for H100)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (BH, num_blocks_m, num_blocks_n)

    trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The heavy part of the TriMul update is the three‑tensor contraction  

    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]

where `left` and `right` are the gated linear projections of the input.
For each combined batch‑hidden index `bh = b * hidden_dim` this is a
matrix multiplication `L_bh @ R_bhᵀ` of size (N×N)·(N×N).  
The kernel below performs a *batched* GEMM where each program processes
a (BLOCK_M×BLOCK_N) tile of one `(bh, N, N)` multiplication, loads the
tiles in FP16, uses `tl.dot` (tensor‑core friendly) with FP32 accumulation
and writes the result directly to a FP32 output buffer.

All surrounding operations (LayerNorm, linear projections, gating,
output LayerNorm and final projection) are kept in PyTorch – they are
memory‑bound and much cheaper than the O(N³) contraction.

Key optimisations compared to a naïve implementation:
* 2‑D grid layout: (bh, tile_m, tile_n) → no integer division inside the
  kernel.
* FP16 loads + tensor‑core dot (no explicit cast to FP32).
* Minimal masking logic – only the edge tiles are masked.
* All strides are passed explicitly, avoiding Python‑side shape queries.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_gemm_kernel(
    L_ptr,               # half *, (BH, N, N)
    R_ptr,               # half *, (BH, N, N)
    O_ptr,               # float *, (BH, N, N)
    N,                   # seq_len (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    # strides – flat (elements, not bytes)
    stride_bL, stride_iL, stride_kL,
    stride_bR, stride_iR, stride_kR,
    stride_bO, stride_iO, stride_jO,
):
    """
    Batched GEMM for TriMul:
        out[bh] = left[bh] @ right[bh].T
    where each `left/right` slice has shape (N, N) and `bh` runs over
    B * hidden_dim.
    """
    # ---------------  program identifiers  -----------------
    bh_idx = tl.program_id(0)          # combined batch * hidden index
    pid_m = tl.program_id(1)           # tile row
    pid_n = tl.program_id(2)           # tile col

    # ---------------  tile offsets  -----------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # ---------------  pointers to the current slice  -----------------
    L_batch = L_ptr + bh_idx * stride_bL
    R_batch = R_ptr + bh_idx * stride_bR
    O_batch = O_ptr + bh_idx * stride_bO

    # ---------------  accumulation buffer  -----------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------  reduction over k  -----------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_tiles):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (M, K)
        L_ptrs = L_batch + offs_m[:, None] * stride_iL + offs_k[None, :] * stride_kL
        L = tl.load(L_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                     # half → stays half

        # right tile (transposed): we need R[k, j]
        # original layout: R[row, col] with stride_iR = N, stride_kR = 1
        R_ptrs = R_batch + offs_k[:, None] * stride_iR + offs_n[None, :] * stride_kR
        R = tl.load(R_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                     # half

        # tensor‑core friendly dot, FP32 accumulation
        acc += tl.dot(L, R)

    # ---------------  write back the tile  -----------------
    O_ptrs = O_batch + offs_m[:, None] * stride_iO + offs_n[None, :] * stride_jO
    tl.store(O_ptrs, acc, mask=mask_m[:, None] & mask_n[:, None])

    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32.
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the channel dimension `dim`
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) → shape [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] for broadcasting
        mask_exp = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to half precision for the heavy GEMM
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape          # left_h: [B, N, N, hidden_dim]

    # Move `hidden_dim` into the batch dimension so that we have
    # (BH, N, N) tensors, where BH = B * hidden_dim.
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)   # half
    right_flat = right_perm.view(BH, N, N) # half

    # Output buffer – FP32 accumulator as required by the model
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=left_flat.device)

    # ------------------------------------------------------------------
    # 6. Triton batched GEMM launch
    # ------------------------------------------------------------------
    # Tuned tile sizes for H100 (FP16 tensor‑core friendly)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (BH, num_blocks_m, num_blocks_n)

    trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        # strides for L (half)
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        # strides for R (half)
        right_flat.stride(0),
        right_flat.stride(1),
        right_flat.stride(2),
        # strides for O (float32)
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        num_warps=8,          # H100 optimal occupancy
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 8. Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 9. Apply the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10. Final linear projection back to the original dimension `dim`
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```
