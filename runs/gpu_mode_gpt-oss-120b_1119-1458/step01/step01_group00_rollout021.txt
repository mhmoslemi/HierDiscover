"""
TriMul (outgoing) forward pass – optimized for GPU (H100).

Algorithm
---------
1. Layer‑norm the input pair representation (float32 for stability).
2. Half‑precision linear projections → left, right (no bias).
3. Optional mask is applied (broadcast on the hidden dimension).
4. Gated multiplicative update:
   left  = left  * sigmoid(linear(x, left_gate_weight))  * mask
   right = right * sigmoid(linear(x, right_gate_weight)) * mask
5. Dense N³ contraction  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   – for small sequence lengths (N ≤ 256) a hand‑written Triton kernel is
     used (batched matmul per head).  For larger N we fall back to
     PyTorch's highly‑optimized batched GEMM (torch.bmm) which
     exploits Tensor‑cores on H100.
6. Layer‑norm over the hidden dimension (float32 internal accumulation).
7. Output gate (sigmoid(linear(x, out_gate_weight))) applied.
8. Final linear projection back to the original channel dimension.
9. Return in float32.

Only the N³ contraction (Step 5) is performed with Triton for the
“kernel‑required” part; all other ops use native PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the N³ contraction (used for small N)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,               # const float16 *
    right_ptr,              # const float16 *
    out_ptr,                # float16 *
    B, N, H,                # meta‑data
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,k,j]"""
    pid_head = tl.program_id(0)   # encodes (batch, head)
    pid_m    = tl.program_id(1)   # block row index (i)
    pid_n    = tl.program_id(2)   # block col index (j)

    total_heads = B * H
    b = pid_head // H
    h = pid_head % H

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b,h,i,k]
        left_ptrs = left_ptr + \
            b * stride_left_b + h * stride_left_h + \
            offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right[b,h,k,j]  (right is already transposed on the two N axes)
        right_ptrs = right_ptr + \
            b * stride_right_b + h * stride_right_h + \
            offs_k[:, None] * stride_right_i + offs_n[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_ptr + \
        b * stride_out_b + h * stride_out_h + \
        offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def _tri_mul_triton(left: torch.Tensor, right_t: torch.Tensor) -> torch.Tensor:
    """
    Triton‑accelerated batched matmul for the N³ contraction.
    Both tensors must be half‑precision, on the same CUDA device,
    and have shape (B, N, N, H) where ``right_t`` is the transposed view
    of the original right tensor (k ↔ j).
    """
    assert left.is_cuda and right_t.is_cuda, "Triton kernel requires CUDA tensors"
    B, N, _, H = left.shape

    out = torch.empty_like(left)   # same dtype & device

    # strides in element units (not bytes)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right_t.stride()
    s_ob, s_oi, s_ok, s_oh = out.stride()

    # Tuned block sizes – work well for N up to 256 on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                 # batch × hidden
        (N + BLOCK_M - 1) // BLOCK_M,          # row tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # col tiles
    )

    _tri_mul_kernel[grid](
        left, right_t, out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rk, s_rh,
        s_ob, s_oi, s_ok, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,           # H100 benefits from 8 warps per block
    )
    return out

# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor  [B, N, N, C]  (float32)
        - mask_tensor  : torch.Tensor  [B, N, N]   (bool/float) or None
        - weights_dict : dict of model parameters (torch.Tensor)
        - config_dict  : contains at least 'dim' and 'hidden_dim'

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper to move a weight to the right device/dtype (half, for compute)
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Load model weights
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w   = to_half(weights["left_proj.weight"])
    right_proj_w  = to_half(weights["right_proj.weight"])
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32, eps=1e‑5 for stability)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(input_tensor,
                          normalized_shape=(dim,),
                          weight=norm_w,
                          bias=norm_b,
                          eps=1e-5)          # [B, N, N, C]  (float32)

    # Cast to half‑precision for the heavy ops
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_h, left_proj_w)    # [B, N, N, hidden_dim]  (fp16)
    right = F.linear(x_h, right_proj_w)  # same shape

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype=torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))

    # Fuse gating (element‑wise) – in‑place to save memory
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. N³ contraction (batch matmul per hidden head)
    # ------------------------------------------------------------------
    # rightᵀ on the two N axes: shape (B, N, N, H) → (B, N, N, H) where
    # the inner two dimensions are swapped.  This transposed view is
    # required by the batched matmul formulation.
    right_t = right.permute(0, 2, 1, 3).contiguous()   # [B, N, N, H]

    # Choose the fastest implementation:
    #   * For modest sequence lengths (N ≤ 256) we use the custom Triton kernel.
    #   * For larger N we fall back to torch.bmm which leverages Tensor‑cores.
    if N <= 256:
        # Triton version
        out = _tri_mul_triton(left, right_t)           # [B, N, N, H] (fp16)
    else:
        # torch.bmm version – reshape to (B*H, N, N) and multiply
        # left:   (B, N, N, H) → (B, H, N, N)
        left_mat = left.permute(0, 3, 1, 2).contiguous()
        # right_t: (B, N, N, H) → (B, H, N, N)
        right_mat = right_t.permute(0, 3, 1, 2).contiguous()
        B_, H_, N_, _ = left_mat.shape
        left_flat = left_mat.view(B_ * H_, N_, N_)
        right_flat = right_mat.view(B_ * H_, N_, N_)
        out_flat = torch.bmm(left_flat, right_flat)   # (B*H, N, N)
        out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
        # out now has shape [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over the hidden dimension (float32 internal)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(out_fp32,
                                 normalized_shape=(hidden_dim,),
                                 weight=to_out_norm_w.to(torch.float32),
                                 bias=to_out_norm_b.to(torch.float32),
                                 eps=1e-5)
    out_norm = out_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 7. Apply output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    gated = out_norm * out_gate
    final = F.linear(gated, to_out_w)    # [B, N, N, dim]  (fp16)

    # ------------------------------------------------------------------
    # Return the result in float32 as required by the spec
    # ------------------------------------------------------------------
    return final.to(torch.float32)
