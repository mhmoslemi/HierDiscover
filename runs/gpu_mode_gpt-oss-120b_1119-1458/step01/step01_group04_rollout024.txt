```
@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr, N,
    stride_lh, stride_li, stride_lk,
    stride_rh, stride_rj, stride_rk,
    stride_oh, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_stages: tl.constexpr = 2,
):
    pid_h = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers for current head
    left_head = left_ptr + pid_h * stride_lh
    right_head = right_ptr + pid_h * stride_rh
    out_head = out_ptr + pid_h * stride_oh

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of blocks in K dimension
    num_blocks_k = tl.cdiv(N, BLOCK_K)

    # prefetch first tile
    k = 0
    offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
    mask_k = offs_k < N

    left_tile = tl.load(
        left_head + offs_m[:, None] * stride_li + offs_k[None, :] * stride_lk,
        mask=mask_m[:, None] & mask_k[None, :],
        other=0.0,
    )
    right_tile = tl.load(
        right_head + offs_k[:, None] * stride_rk + offs_n[None, :] * stride_rj,
        mask=mask_k[:, None] & mask_n[None, :],
        other=0.0,
    )
    # Compute dot for first tile (W1)
    acc += tl.dot(left_tile, right_tile, dtype=tl.float32)

    # loop over remaining blocks
    for kk in range(1, num_blocks_k):
        # prefetch next tiles
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        left_tile = tl.load(
            left_head + offs_m[:, None] * stride_li + offs_k[None, :] * stride_lk,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        right_tile = tl.load(
            right_head + offs_k[:, None] * stride_rk + offs_n[None, :] * stride_rj,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(left_tile, right_tile, dtype=tl.float32)

    # Store result
    out_ptrs = out_head + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    out_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)
```

"""
TriMul (outgoing) forward pass – Triton‑accelerated core.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B, N, N, C]).
2. Fuse the two linear projections (proj + gate) for the left and right tensors:
   * concatenate the projection weights (C←H) and gate weights (C←H) → (2H, C).
   * a single matmul produces both the raw projection and the raw gate.
   * split the result, apply sigmoid to the gate part and multiply.
   * optional binary mask is applied only to the projection values (as in the
     reference implementation).
3. The expensive “einsum('... i k d, ... j k d -> ... i j d')’’ is a batched
   matrix multiplication C = A @ Bᵀ performed per head (i.e. per hidden
   dimension) with a custom Triton kernel that tiles the M, N and K axes.
   * Input tiles are FP16, the accumulation is FP32 (Tensor‑Core friendly).
   * The kernel uses a simple pre‑fetch loop; the tile size (64‑64‑64) matches
     H100 tensor‑core granularity.
4. Reshape the result back to [B, N, N, H].
5. Apply a second LayerNorm, multiply by the outgoing gate and finally
   project back to the original channel dimension.
All heavy compute (the N³ matmul) runs on the GPU via Triton; the remaining
operations are done with fused PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batch of independent matmuls C = A @ Bᵀ
# A shape: (head, M, K)   – left  tensor (head, i, k)
# B shape: (head, N, K)   – right tensor (head, j, k)
# C shape: (head, M, N)   – output (head, i, j)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr,                # fp16 * (head, M, K)
    right_ptr,               # fp16 * (head, N, K)
    out_ptr,                 # fp32 * (head, M, N)
    N,                       # i32   sequence length
    stride_lh, stride_li, stride_lk,   # i64 strides for left  (head, i, k)
    stride_rh, stride_rj, stride_rk,   # i64 strides for right (head, j, k)
    stride_oh, stride_oi, stride_oj,   # i64 strides for out   (head, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(0)          # head (B * hidden_dim)
    pid_m = tl.program_id(1)          # M‑tile index
    pid_n = tl.program_id(2)          # N‑tile index

    # ------------------------------------------------------------------
    # Offsets for the current M/N tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Base pointers for the current head
    # ------------------------------------------------------------------
    left_head = left_ptr + pid_h * stride_lh
    right_head = right_ptr + pid_h * stride_rh
    out_head = out_ptr + pid_h * stride_oh

    # ------------------------------------------------------------------
    # Accumulator (FP32) for one output tile
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K in blocks
    # ------------------------------------------------------------------
    num_blocks_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_blocks_k):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile [M, K]  (head, i, k)
        left_ptrs = (
            left_head
            + offs_m[:, None] * stride_li
            + offs_k[None, :] * stride_lk
        )
        left_mask = mask_m[:, None] & mask_k[None, :]
        a = tl.load(left_ptrs, mask=left_mask, other=0.0)   # fp16

        # right tile transposed to [K, N]  (head, k, j)
        right_ptrs = (
            right_head
            + offs_k[:, None] * stride_rk
            + offs_n[None, :] * stride_rj
        )
        right_mask = mask_k[:, None] & mask_n[None, :]
        b = tl.load(right_ptrs, mask=right_mask, other=0.0) # fp16

        # FP16×FP16 → FP32 accumulation (Tensor‑Core friendly)
        acc += tl.dot(a, b, dtype=tl.float32)

    # ------------------------------------------------------------------
    # Store the computed tile
    # ------------------------------------------------------------------
    out_ptrs = out_head + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    out_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)


# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        * input_tensor : torch.Tensor [B, N, N, C]  (float32)
        * mask         : torch.Tensor [B, N, N]    (bool/float, may be None)
        * weights      : dict of model parameters (float32)
        * config       : dict, must contain ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (True ⇒ mask ignored).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Layer‑norm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # --------------------------------------------------------------
    # Fuse projection + gate for the left side
    # --------------------------------------------------------------
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, C)
    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, C)

    # Cast normalized tensor to FP16 for the bulk of the matmuls
    x_fp16 = x.to(torch.float16)

    # One large matmul per side yields (proj, gate) together
    left_combined = F.linear(x_fp16, left_combined_w)   # [B, N, N, 2*H]  fp16
    right_combined = F.linear(x_fp16, right_combined_w) # [B, N, N, 2*H]  fp16

    # Split projection / gate part
    left_proj = left_combined[..., :hidden_dim]          # fp16
    left_gate_raw = left_combined[..., hidden_dim:]      # fp16
    right_proj = right_combined[..., :hidden_dim]        # fp16
    right_gate_raw = right_combined[..., hidden_dim:]    # fp16

    # --------------------------------------------------------------
    # Outgoing gate (computed on full‑precision for stability)
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"])
    )  # [B, N, N, H]  float32

    # --------------------------------------------------------------
    # Apply optional mask (mask is broadcast on the hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        # mask may be bool; cast to same dtype as the projections
        mask_f = mask.unsqueeze(-1).to(x.dtype)  # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # --------------------------------------------------------------
    # Apply gating (sigmoid → fp16) and fuse with projection
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw).to(torch.float16)
    right_gate = torch.sigmoid(right_gate_raw).to(torch.float16)

    left = left_proj * left_gate   # [B, N, N, H]  fp16
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # Prepare tensors for the Triton kernel:
    # (head, i, k) and (head, j, k) layout, where head = B * H
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    total_heads = B * hidden_dim

    # (B, N, N, H) -> (B, H, N, N) -> (total_heads, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # Output buffer (FP32 accumulation)
    out_flat = torch.empty((total_heads, N, N), dtype=torch.float32, device=device)

    # --------------------------------------------------------------
    # Triton launch configuration (tile size tuned for H100 Tensor Cores)
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # Reshape back to the original layout: [B, N, N, H]
    # --------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, H]   float32

    # --------------------------------------------------------------
    # Post‑processing: second LayerNorm → outgoing gate → final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # float32

    out = out * out_gate  # broadcast over hidden dimension

    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, dim] float32

    return out
