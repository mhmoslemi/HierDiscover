"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The core of the module is the N³ reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which can be seen as a batched matrix multiplication for every
pair (batch, hidden) of size (seq_len × seq_len) × (seq_len × seq_len)ᵀ.
The kernel below performs this GEMM in fp16 using Tensor‑Core
accelerated `tl.dot`, accumulating in fp32 for numerical stability.
All remaining operations (LayerNorm, Linear projections, sigmoids,
masking, final projection) are carried out with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: one independent GEMM per (batch, hidden) pair
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,                     # pointers
    stride_ap, stride_am, stride_ak,          # strides for A (P, M, K)
    stride_bp, stride_bk, stride_bn,          # strides for B (P, K, N)
    stride_cp, stride_cm, stride_cn,          # strides for C (P, M, N)
    M, N, K,                                  # sizes (all = seq_len)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B   where
        A : (P, M, K)   – row‑major (M rows, K cols)
        B : (P, K, N)   – row‑major (K rows, N cols)
        C : (P, M, N)   – row‑major result
    P = batch * hidden_dim
    """
    pid_m = tl.program_id(0)          # block index over M
    pid_n = tl.program_id(1)          # block index over N
    pid_p = tl.program_id(2)          # independent matmul index (batch*hidden)

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this (batch,hidden) pair
    a_base = A_ptr + pid_p * stride_ap
    b_base = B_ptr + pid_p * stride_bp
    c_base = C_ptr + pid_p * stride_cp

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        start_k = k * BLOCK_K
        offs_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                     # fp16 load

        # B tile (K x N)
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                     # fp16 load

        # Tensor‑core dot (fp16 → fp32 accumulation)
        acc += tl.dot(a, b).to(tl.float32)

    # ------------------------------------------------------------------
    # Write back result (cast back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def _triton_gemm(
    left: torch.Tensor,
    right: torch.Tensor,
    seq_len: int,
    hidden_dim: int,
    *,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
):
    """
    Batched GEMM for the TriMul N³ reduction.
    Input tensors must be fp16, shape: (B, N, N, hidden_dim).
    Returns a fp16 tensor of same shape.
    """
    B = left.shape[0]
    # Bring hidden dimension to the second axis: (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # Right needs K as second axis, J as third (i.e. transpose the last two dims)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    # Collapse batch & hidden into a single matmul dimension
    P = B * hidden_dim
    left_flat = left_perm.view(P, seq_len, seq_len)     # (P, M, K)
    right_flat = right_perm.view(P, seq_len, seq_len)   # (P, K, N)

    out = torch.empty_like(left_flat)                    # (P, M, N)

    # Strides (elements, not bytes – Triton accounts for dtype)
    stride_ap = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bp = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_cp = out.stride(0)
    stride_cm = out.stride(1)
    stride_cn = out.stride(2)

    # Grid configuration
    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,   # blocks over M
        (seq_len + BLOCK_N - 1) // BLOCK_N,   # blocks over N
        P,                                    # one program per (batch, hidden)
    )

    # Heuristic: more warps for larger problem sizes
    num_warps = 8 if seq_len <= 256 else 16

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out,
        stride_ap,
        stride_am,
        stride_ak,
        stride_bp,
        stride_bk,
        stride_bn,
        stride_cp,
        stride_cm,
        stride_cn,
        seq_len,
        seq_len,
        seq_len,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=4,               # aggressive pipelining
    )

    # Reshape back to (B, N, N, hidden)
    out = out.view(B, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()
    return out

        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/int) – may be None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * "dim"         – input & output channel dim
                * "hidden_dim"  – inner hidden dimension
                * "nomask"      – bool, True ⇢ mask is ignored
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )               # [B, N, N, dim] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    # Cast weights once to fp16 – this saves repeated casts inside F.linear
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    # Use fp16 for the heavy path; keep a fp32 copy for subsequent ops
    x_fp16 = x.to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, hidden] – fp16
    right = F.linear(x_fp16, right_proj_w) # [B, N, N, hidden] – fp16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (applied after the projections)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B, N, N] → [B, N, N, 1] → fp16
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4️⃣ Gating values (sigmoid of a linear layer)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    # Gate projections stay in fp32 for better numerical stability,
    # then are cast to the dtype of left/right for the element‑wise product.
    left_gate = torch.sigmoid(F.linear(x, left_gate_w)).to(left.dtype)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)).to(right.dtype)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # keep fp32

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Triton‑accelerated N³ reduction
    # ------------------------------------------------------------------
    seq_len = input_tensor.shape[1]   # N

    # Pick block sizes based on sequence length (empirically good on H100)
    if seq_len <= 256:
        BLOCK_M = BLOCK_N = 64
    elif seq_len <= 512:
        BLOCK_M = BLOCK_N = 128
    else:
        BLOCK_M = BLOCK_N = 128
    BLOCK_K = 64

    out_fp16 = _triton_gemm(
        left, right,
        seq_len=seq_len,
        hidden_dim=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )   # [B, N, N, hidden] – fp16

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing (float32)
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)   # back to fp32 for LayerNorm, etc.

    # LayerNorm over hidden dimension
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )  # [B, N, N, hidden] – fp32

    # Output gate (broadcast over hidden)
    out = out * out_gate

    # Final linear projection back to original dim
    to_out_w = weights["to_out.weight"]          # (dim, hidden)
    out = F.linear(out, to_out_w)                # [B, N, N, dim] – fp32

    return out
