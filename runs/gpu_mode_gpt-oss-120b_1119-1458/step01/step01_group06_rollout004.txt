"""
TriMul (outgoing) forward pass – a Triton‑accelerated implementation.

The heavy O(N³) reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is realized as a batched matrix‑multiply (Gram product) where each
pair (b,d) forms an independent (N×N) matmul.  The kernel loads the
left matrix in FP16, the right matrix (transposed on‑the‑fly) in FP16,
accumulates in FP32 for numerical stability and writes the result in
FP32.  All other operations (LayerNorms, linear projections, gating,
final linear) are performed with PyTorch for simplicity.

The implementation works for any batch size ≤2, any sequence length
up to 1024 and arbitrary hidden‑dimensionality (the hidden dimension
is the width of the intermediate tensors).  The mask is optional and
is applied before the gating steps, exactly as in the reference
PyTorch model.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def triton_tri_mul_kernel(
    left_ptr,               # POINTER TO LEFT  (M,N,N)   FP16
    right_ptr,              # POINTER TO RIGHT (M,N,N)   FP16
    out_ptr,                # POINTER TO OUT   (M,N,N)   FP32
    M,                      # M = batch * hidden_dim
    N,                      # sequence length
    stride_lm, stride_ln, stride_lk,   # left strides: (M, i, k)
    stride_rm, stride_rn, stride_rk,   # right strides: (M, j, k) – note that we load right as (k,j)
    stride_om, stride_on, stride_ok,   # out strides: (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out[m,i,j] = Σ_k left[m,i,k] * right[m,j,k]
    for a batch of M independent (N×N) matrices.
    """

    pid_m = tl.program_id(0)    # index over (batch * hidden_dim)
    pid_i = tl.program_id(1)    # tile row
    pid_j = tl.program_id(2)    # tile column

    # Tile starting offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32 (more stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a (M,i,k) block – FP16
        left = tl.load(
            left_ptr
            + pid_m * stride_lm
            + offs_i[:, None] * stride_ln
            + offs_k[None, :] * stride_lk,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load a (M,k,j) block i.e. transpose of right on‑the‑fly – FP16
        right = tl.load(
            right_ptr
            + pid_m * stride_rm
            + offs_k[:, None] * stride_rk   # k dimension (col of original right)
            + offs_j[None, :] * stride_rn,  # j dimension (row of original right)
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # Dot product (produces FP16) → cast to FP32 before accumulation
        acc += tl.dot(left, right).to(tl.float32)

    # Store the result (FP32) back to memory
    tl.store(
        out_ptr
        + pid_m * stride_om
        + offs_i[:, None] * stride_on
        + offs_j[None, :] * stride_ok,
        acc,
        mask=mask_i[:, None] & mask_j[:, None],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data = (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor, shape [B, N, N, dim] (float32)
            mask        : torch.Tensor, shape [B, N, N] (bool/float) – may be ignored
            weights     : dict of torch.Tensors containing the model weights
            config      : dict with keys:
                "dim"          – input channel dimension (C)
                "hidden_dim"   – projection dimension (H)
                "nomask"       – bool, True if mask should be ignored
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    device = torch.cuda.current_device()
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (dim -> dim)
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (dim -> hidden_dim) – biasless
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # -----------------------------------------------------------------
    if not nomask:
        # Ensure mask is float for multiplication
        mask_f = mask.to(left.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections on the *normalized* x)
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # -----------------------------------------------------------------
    # 5. Apply gating
    # -----------------------------------------------------------------
    left = left * left_gate      # still float32
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Rearrange tensors for the Triton kernel
    #    (B, N, N, H) -> (B*H, N, N)
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape

    # (B, N, N, H) -> (B, H, N, N) and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Cast to FP16 for higher memory bandwidth inside the kernel
    left_half = left_perm.to(torch.float16)
    right_half = right_perm.to(torch.float16)

    BH = B * hidden_dim
    left_flat = left_half.view(BH, N, N)
    right_flat = right_half.view(BH, N, N)

    # Output buffer in FP32 – we accumulate directly in FP32
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # -----------------------------------------------------------------
    # 7. Kernel launch
    # -----------------------------------------------------------------
    stride_lm, stride_ln, stride_lk = left_flat.stride()
    stride_rm, stride_rn, stride_rk = right_flat.stride()
    stride_om, stride_on, stride_ok = out_flat.stride()

    # Tunable block sizes – empirically good for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        BH,                                 # batch * hidden_dim
        (N + BLOCK_M - 1) // BLOCK_M,       # rows
        (N + BLOCK_N - 1) // BLOCK_N,       # columns
    )

    triton_tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        BH,
        N,
        stride_lm,
        stride_ln,
        stride_lk,
        stride_rm,
        stride_rn,
        stride_rk,
        stride_om,
        stride_on,
        stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 8. Reshape back to (B, N, N, hidden_dim)
    # -----------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H] FP32

    # -----------------------------------------------------------------
    # 9. Hidden → output LayerNorm, gating and final linear
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                      # element‑wise gate (float32)
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # -----------------------------------------------------------------
    # 10. Return result in float32
    # -----------------------------------------------------------------
    return out
```
