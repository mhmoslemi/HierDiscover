@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N,
    stride_lm, stride_ln, stride_lk,
    stride_rm, stride_rn, stride_rk,
    stride_om, stride_on, stride_ok,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)   # batch*hidden index
    pid_i = tl.program_id(1)       # tile row
    pid_j = tl.program_id(2)       # tile col

    # -------------------- tile offsets --------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 (more stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- reduction over k --------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_ptr + pid_batch * stride_lm
                     + offs_i[:, None] * stride_ln
                     + offs_k[None, :] * stride_lk,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )  # dtype matches input (fp16)

        # right block transposed: shape (BLOCK_K, BLOCK_N)
        # we need right[k, j] => stride_rk for k, stride_rn for j
        right = tl.load(
            right_ptr + pid_batch * stride_rm
                      + offs_k[:, None] * stride_rk
                      + offs_j[None, :] * stride_rn,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )  # dtype matches input (fp16)

        # dot produces fp16, cast to fp32 before accumulation
        acc += tl.dot(left, right).to(tl.float32)

    # -------------------- store result --------------------
    tl.store(
        out_ptr + pid_batch * stride_om
                + offs_i[:, None] * stride_on
                + offs_j[None, :] * stride_ok,
        acc,
        mask=mask_i[:, None] & mask_j[None, :]

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # config: dim, hidden_dim, maybe nomask, maybe use_bias (none)
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,),
                                        weight=weights["norm.weight"],
                                        bias=weights["norm.bias"])

    # 2. Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # 3. Optional mask
    if not nomask:
        # mask: shape [B,N,N] broadcast on hidden dim
        mask_b = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_b
        right = right * mask_b

    # 4. Gate values
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # 5. Apply gates
    left = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape # shape B,N,N,hidden
    # 6. Permute for kernel: move hidden to second dim then flatten
    left_perm = left.permute(0,3,1,2).contiguous()  # B,H,N,N
    right_perm = right.permute(0,3,1,2).contiguous()  # B,H,N,N

    # cast to half
    left_half = left_perm.to(torch.float16)
    right_half = right_perm.to(torch.float16)

    # flatten batch*hidden
    BH = B * hidden_dim
    left_flat = left_half.view(BH, N, N)
    right_flat = right_half.view(BH, N, N)

    # allocate output buffer float32
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=left_flat.device)

    # compute strides
    stride_lm, stride_ln, stride_lk = left_flat.stride()
    stride_rm, stride_rn, stride_rk = right_flat.stride()
    stride_om, stride_on, stride_ok = out_flat.stride()
    # Launch kernel
    # Block size tuning: choose 128,128,32 often best on H100, guard against memory.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    # Tiling loops
    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    triton_tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        BH,
        N,
        stride_lm, stride_ln, stride_lk,
        stride_rm, stride_rn, stride_rk,
        stride_om, stride_on, stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # 8. reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # out shape B,N,N,H (float32)

    # 9. hidden to output
    out = torch.nn.functional.layer_norm(out,
                                         (hidden_dim,),
                                         weight=weights["to_out_norm.weight"],
                                         bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass – Triton‑accelerated Gram matrix.

The heavy N³ reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix‑multiply (left @ rightᵀ) for every hidden dimension.
All cheap operations (LayerNorm, Linear, gating, final Linear) are done with
PyTorch; the Gram product is computed with a custom Triton kernel that:

* works on a flattened view (B*H, N, N) where H = hidden_dim,
* loads tiles of size BLOCK_M × BLOCK_K from ``left`` and the transposed
  tiles BLOCK_K × BLOCK_N from ``right``,
* accumulates in FP32 for numerical stability,
* stores the result in FP32 and returns a tensor of shape
  (B, N, N, H) (float32).

The kernel is autotuned for a few block‑size configurations that work well
on H100 GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched Gram matrix (left @ rightᵀ) over the sequence dimension.
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        triton.Config({'BLOCK_M': 64, 'BLOCK_N': 64, 'BLOCK_K': 32}, num_warps=4),
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 128, 'BLOCK_K': 64}, num_warps=8),
    ],
    key=['M', 'N'],
)
@triton.jit
def gram_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N,               # M = B*H, N = sequence length
    stride_lm, stride_ln, stride_lk,
    stride_rm, stride_rn, stride_rk,
    stride_om, stride_on, stride_ok,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[m,i,j] = Σ_k left[m,i,k] * right[m,j,k].

    left  : (M, N, N)  – layout (m, i, k)
    right : (M, N, N)  – layout (m, j, k)   (will be accessed as (k, j) to realise the transpose)
    out   : (M, N, N)  – result (m, i, j)
    """
    pid_m = tl.program_id(0)          # batch * hidden index
    pid_i = tl.program_id(1)          # tile row (i)
    pid_j = tl.program_id(2)          # tile col (j)

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (FP32 for stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Reduction over k
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, BLOCK_K)   (i, k)
        left = tl.load(
            left_ptr + pid_m * stride_lm
                     + offs_i[:, None] * stride_ln
                     + offs_k[None, :] * stride_lk,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )   # dtype = fp16 (or bfloat16)

        # right block transposed: shape (BLOCK_K, BLOCK_N)   (k, j)
        # we access right[m, j, k] as (k, j) by swapping the strides
        right = tl.load(
            right_ptr + pid_m * stride_rm
                      + offs_k[:, None] * stride_rk      # stride over k (col in original)
                      + offs_j[None, :] * stride_rn,     # stride over j (row in original)
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )   # dtype = fp16

        # Dot product – result in FP16, cast to FP32 before accumulation
        acc += tl.dot(left, right).to(tl.float32)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    tl.store(
        out_ptr + pid_m * stride_om
                + offs_i[:, None] * stride_on
                + offs_j[None, :] * stride_ok,
        acc,
        mask=mask_i[:, None] & mask_j[None, :]
    )


# ----------------------------------------------------------------------
# Public entry point used by the test harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim]
        - mask         : torch.Tensor[bs, N, N]   (unused if config[\"nomask\"] is True)
        - weights      : dict of torch.Tensors containing all linear‑layer weights
        - config       : dict with keys 'dim', 'hidden_dim', 'nomask' (bool)

    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast on the hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] -> [B, N, N, 1] for broadcast
        m = mask.unsqueeze(-1).to(x.dtype)
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 4. Gating values (sigmoid)
    # --------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # --------------------------------------------------------------
    # 5. Apply gates
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6. Prepare tensors for the Triton kernel
    #    Shape transformation: (B, N, N, H) -> (B*H, N, N)
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    # Move hidden dimension next to batch and make tensors contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Half‑precision for the compute‑intensive GEMM
    left_half = left_perm.to(torch.float16)
    right_half = right_perm.to(torch.float16)

    # Flatten the leading two dimensions: (B*H, N, N)
    BH = B * hidden_dim
    left_flat = left_half.view(BH, N, N)
    right_flat = right_half.view(BH, N, N)

    # Output buffer – FP32 (the kernel accumulates in FP32)
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=left_flat.device)

    # --------------------------------------------------------------
    # 7. Kernel launch configuration
    # --------------------------------------------------------------
    stride_lm, stride_ln, stride_lk = left_flat.stride()
    stride_rm, stride_rn, stride_rk = right_flat.stride()
    stride_om, stride_on, stride_ok = out_flat.stride()

    grid = (
        BH,
        (N + 64 - 1) // 64,   # i‑tiles
        (N + 64 - 1) // 64,   # j‑tiles
    )
    gram_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        BH,
        N,
        stride_lm,
        stride_ln,
        stride_lk,
        stride_rm,
        stride_rn,
        stride_rk,
        stride_om,
        stride_on,
        stride_ok,
        BLOCK_M=64,
        BLOCK_N=64,
        BLOCK_K=32,
    )

    # --------------------------------------------------------------
    # 8. Reshape back to (B, N, N, hidden_dim) – already FP32
    # --------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 9. Hidden → output LayerNorm, gating and final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
