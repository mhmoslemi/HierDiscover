    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)   # float32
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

        right = right * mask

    left = left * left_gate

    left_h = left.to(torch.float16).contiguous()

"""
Highly‑optimized Triton implementation of the “outgoing” TriMul operator
(AlphaFold‑3 style).  The heavy O(N³) reduction is performed by a
batched half‑precision GEMM (A @ Bᵀ) where each hidden channel is fused
into the batch dimension.  All linear layers and sigmoid gates are
executed in FP16 to halve memory traffic, while the LayerNorms and the
final projection stay in FP32 for numerical stability.

Algorithm
---------
1. Layer‑norm the 4‑D input (dim‑last) → FP32.
2. Cast to FP16 and apply three bias‑free linear projections:
   * left  = Linear(dim → hidden)   (FP16)
   * right = Linear(dim → hidden)   (FP16)
   * left_gate, right_gate, out_gate = Linear(dim → hidden) → sigmoid
3. Optional mask (broadcast on hidden) applied in FP16.
4. Fuse gates: left  = left  * left_gate,  right = right * right_gate   (FP16).
5. Core triangle multiplication:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented by a Triton kernel `batch_matmul_transpose_kernel` that
   tiles the GEMM with BLOCK_M = 128, BLOCK_N = 128, BLOCK_K = 32,
   loads FP16 tiles, promotes them to FP32 for accumulation and stores
   the result back in FP16.
6. Cast the result to FP32, apply a second LayerNorm over the hidden
   dimension, multiply by `out_gate`, and finish with the final
   bias‑free linear projection (FP32) to obtain the output tensor
   `[B, N, N, dim]`.

The kernel works for any batch size ≤ 2, sequence length up to 1024,
hidden dimension 128 (or a multiple thereof) and supports both masked
and un‑masked inputs.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def batch_matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_am, stride_ak,
    stride_b_batch, stride_bk, stride_bn,
    stride_c_batch, stride_cm, stride_cn,
    stride_hidden,               # stride of the hidden dimension (==1 for contiguous tensors)
    hidden_dim: tl.constexpr,    # number of hidden channels (Cₕ)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM C = A @ Bᵀ
    A : (B·Cₕ, M, K)   – FP16, row‑major
    B : (B·Cₕ, N, K)   – FP16, row‑major (accessed as transposed)
    C : (B·Cₕ, M, N)   – FP16, row‑major
    """
    # -----------------------------------------------------------------
    # 1️⃣ program identifiers
    # -----------------------------------------------------------------
    batch_idx = tl.program_id(0)      # combined (batch, hidden) index
    pid_m = tl.program_id(1)          # tile‑row index
    pid_n = tl.program_id(2)          # tile‑col index

    # -----------------------------------------------------------------
    # 2️⃣ decode (batch, hidden) from the linearized index
    # -----------------------------------------------------------------
    b = batch_idx // hidden_dim       # original batch (0 … B‑1)
    h = batch_idx % hidden_dim        # hidden channel (0 … Cₕ‑1)

    # Base pointers for this (batch, hidden) pair
    a_batch_ptr = a_ptr + b * stride_a_batch + h * stride_hidden
    b_batch_ptr = b_ptr + b * stride_b_batch + h * stride_hidden
    c_batch_ptr = c_ptr + b * stride_c_batch + h * stride_hidden

    # -----------------------------------------------------------------
    # 3️⃣ tile offsets
    # -----------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # -----------------------------------------------------------------
    # 4️⃣ accumulation buffer (FP32 for numeric stability)
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # 5️⃣ iterate over K‑tiles
    # -----------------------------------------------------------------
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_k):
        k_base = k * BLOCK_K
        cur_k_valid = (k_base + offs_k) < K

        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am) + ((k_base + offs_k)[None, :] * stride_ak)
        a = tl.load(

        b_ptrs = b_batch_ptr + ((k_base + offs_k)[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

            b_ptrs,
            mask=cur_k_valid[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        ).to(tl.float32)

        # ---- dot product (FP32 accumulation) -----------------------------------
        acc += tl.dot(a, b)

    # -----------------------------------------------------------------
    # 6️⃣ write result (cast back to FP16)
    # -----------------------------------------------------------------
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_h, left_gate_weight_h))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, right_gate_weight_h))

    # -----------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden)
    # -----------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # -----------------------------------------------------------------
    # 5️⃣ Fuse gates (still FP16)
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Core triangle multiplication (batched GEMM left @ rightᵀ) via Triton
    # -----------------------------------------------------------------
    # Prepare the output tensor (FP16) – same layout as left/right: [B, N, N, hidden]
    out_h = torch.empty_like(left, dtype=torch.float16)

    # Strides in number of elements (not bytes)
    # left / right / out are contiguous after the operations above
    stride_a_batch = left.stride(0)        # B‑axis stride = N·N·hidden
    stride_am = left.stride(1)            # row (i) stride    = N·hidden
    stride_ak = left.stride(2)            # col (k) stride    = hidden
    stride_a_hidden = left.stride(3)      # hidden stride      = 1

    stride_b_batch = right.stride(0)
    stride_bk = right.stride(2)           # K‑stride (original column)
    stride_bn = right.stride(1)           # N‑stride (original row)
    # note: hidden stride for right is identical to left (1)

    stride_c_batch = out_h.stride(0)
    stride_cm = out_h.stride(1)
    stride_cn = out_h.stride(2)
    # hidden stride for out also 1

    # -----------------------------------------------------------------
    # Launch Triton kernel
    # -----------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    total_batches = bs * hidden_dim
    grid = (
        total_batches,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batch_matmul_transpose_kernel[grid](
        left,
        right,
        out_h,
        N,           # M
        N,           # N
        N,           # K
        stride_a_batch,
        stride_am,
        stride_ak,
        stride_b_batch,
        stride_bk,
        stride_bn,
        stride_c_batch,
        stride_cm,
        stride_cn,
        stride_a_hidden,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Cast back to FP32 for the remaining stages
    # -----------------------------------------------------------------
    out = out_h.to(torch.float32)   # [B, N, N, hidden] (FP32)

    # -----------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension (FP32)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 9️⃣ Output gate (computed in FP32)
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))  # [B, N, N, hidden]
    out = out * out_gate

    # -----------------------------------------------------------------
    # 🔟 Final linear projection hidden → dim (FP32)
    # -----------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
