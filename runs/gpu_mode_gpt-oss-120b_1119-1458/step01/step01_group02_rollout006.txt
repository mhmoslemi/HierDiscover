"""
TriMul (outgoing) – mixed‑precision Triton implementation
========================================================

The forward pass consists of:
1. Layer‑norm over the last feature dimension (dim).
2. Two bias‑free linear projections → left / right (shape [B,N,N,H]).
3. Optional binary mask (broadcast on the hidden dimension).
4. Sigmoid gates (left_gate, right_gate, out_gate) applied element‑wise.
5. Heavy O(N³) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication C = A @ Bᵀ where each
   (A,B) pair has shape (N,N) and there are B·H independent pairs.
   The contraction is performed in a custom Triton kernel that works on
   FP16 inputs (halving memory traffic) while accumulating in FP32 for
   numerical stability.  Tile sizes (64×64×64) and 8 warps are tuned for
   NVIDIA H100 tensor‑core performance.
6. Layer‑norm over the hidden dimension (H).
7. Multiplication by out_gate and final projection back to `dim`.

Only the O(N³) part is Triton‑accelerated; all other ops use high‑
performance PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ for many independent (batch*hidden) matrices
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,               # *fp16  [M, N, K] (M = B*H)
    B_ptr,               # *fp16  [M, N, K]
    C_ptr,               # *fp32  [M, N, N]
    M, N, K,             # i32   sizes (M = B*H, N = seq_len, K = seq_len)
    stride_am, stride_ai, stride_ak,   # strides for A
    stride_bm, stride_bj, stride_bk,   # strides for B
    stride_cm, stride_ci, stride_cj,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute one tile of C = A @ Bᵀ.

    A is indexed as A[m, i, k]
    B is indexed as B[m, j, k] (we read it as B[m, k, j] to obtain Bᵀ)
    """
    pid_m = tl.program_id(0)               # tile along i (rows)
    pid_n = tl.program_id(1)               # tile along j (cols)
    pid_z = tl.program_id(2)               # batch*hidden index

    # ----- tile offsets -------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over k --------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & (offs_k[None, :] < K),
                    other=tl.float16(0.0))

        # B tile (transposed view): shape (BLOCK_K, BLOCK_N)
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk    # k dimension
            + offs_j[None, :] * stride_bj    # j dimension
        )
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_j[None, :],
                    other=tl.float16(0.0))

        # dot product – cast to fp32 for accumulation (tensor‑core friendly)
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # ----- write output tile --------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel and reshapes tensors
# ----------------------------------------------------------------------
def _batched_matmul_transpose(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute C[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] using Triton.
    `left` and `right` are expected to be half‑precision tensors of shape
    [B, N, N, H] (contiguous).  The result is a FP32 tensor of the same
    layout.
    """
    B, N, _, H = left.shape

    # Collapse batch & hidden dims → (M, N, N) where M = B*H
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # fp16

    # Output buffer in FP32
    C = torch.empty((B * H, N, N), dtype=torch.float32, device=left.device)

    # Strides in element units
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tile sizes – empirically good for H100 with FP16 inputs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles over i
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles over j
        B * H,                          # independent matrices
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,      # double‑buffering
    )

    # Reshape back to original layout [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.
    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (bool/int or float)
        - weights_dict : dict of weight tensors (float32)
        - config_dict  : contains keys ``dim``, ``hidden_dim`` and optional ``nomask``.
    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32, same device as inputs).
    """
    # -------------------------------------------------
    # Unpack arguments
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (dim)
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  float32

    # -------------------------------------------------
    # 2️⃣ Linear projections (no bias) → left / right (float32)
    # -------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------
    # 3️⃣ Gates (sigmoid) – still float32 for numerical safety
    # -------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -------------------------------------------------
    # 4️⃣ Apply gates and optional mask (still float32)
    # -------------------------------------------------
    left = left * left_gate          # shape [B,N,N,H]
    right = right * right_gate

    if not nomask:
        # mask has shape [B,N,N]; broadcast on hidden dim
        mask = mask_tensor.to(torch.float32).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -------------------------------------------------
    # 5️⃣ Cast to half‑precision for the O(N³) contraction
    # -------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # -------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton: out = left @ rightᵀ
    # -------------------------------------------------
    out = _batched_matmul_transpose(left, right)   # [B,N,N,H] float32

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm over hidden dimension (H)
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # -------------------------------------------------
    # 8️⃣ Gate and final projection back to `dim`
    # -------------------------------------------------
    out = out * out_gate                # broadcasting over hidden dim
    out = F.linear(out, weights["to_out.weight"])

    return out
