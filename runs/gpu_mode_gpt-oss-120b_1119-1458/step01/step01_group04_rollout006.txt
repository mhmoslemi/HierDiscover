def _apply_gate_and_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr, N_total, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offset = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offset < N_total
    proj = tl.load(proj_ptr + offset, mask=mask).to(tl.float32)
    gate = tl.load(gate_ptr + offset, mask=mask).to(tl.float32)
    mask_val = tl.load(mask_ptr + offset, mask=mask).to(tl.float32) if mask_ptr is not None else 1.0
    # compute product
    out = proj * gate * mask_val
    # perhaps cast back to fp16
    out_fp16 = out.to(tl.float16)
    tl.store(out_ptr + offset, out_fp16, mask=mask)

TriMul (outgoing) forward pass – highly‑optimized variant.

Algorithm
---------
1. Layer‑norm the input (float32).
2. Cast to float16 and run all linear projections/gates in fp16
   (Tensor‑core friendly).
3. Fuse the “proj × gate” element‑wise products with a tiny Triton kernel
   (the only Triton kernel required by the task).
4. Apply the optional mask (broadcasted over the hidden dimension).
5. Rearrange the tensors to (batch, heads, N, N) and perform a

   uses the cuBLAS/H100 Tensor‑core path and dominates the runtime.
6. Cast the result back to float32, apply a second LayerNorm, the output
   gate and the final linear projection.
7. Return a float32 tensor of shape `[B, N, N, dim]`.

The heavy O(N³·hidden) einsum is replaced by a single cuBLAS batched
matrix multiplication, while the cheap element‑wise multiplication is

    a_ptr,            # *fp16   operand A
    b_ptr,            # *fp16   operand B
    c_ptr,            # *fp16   destination (can be same as a_ptr → in‑place)
    size,             # i64    total number of elements
    BLOCK: tl.constexpr,  # compile‑time block size
):
    """c = a * b   (size‑aware, 1‑D launch)."""
    pid = tl.program_id(0)
    offset = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offset < size

    a = tl.load(a_ptr + offset, mask=mask)
    b = tl.load(b_ptr + offset, mask=mask)
    tl.store(c_ptr + offset, a * b, mask=mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor – [B, N, N, dim] (float32)
            mask         – [B, N, N] (bool/int/float) or None
            weights      – dict with all model parameters (float32)
            config       – dict containing at least `dim`, `hidden_dim`,
                           optionally `nomask` (bool)

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for all subsequent linear / sigmoid ops
    # ------------------------------------------------------------------
    x_h = x.half()  # fp16

    # Half‑precision copies of the weights required for the fp16 linear layers
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    # to‑out‑norm and to‑out final projection stay in fp32 (LayerNorm needs fp32)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (fp16)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)   # [B, N, N, hidden]
    right = F.linear(x_h, right_proj_w)  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 4️⃣ Gating values (sigmoid, fp16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))   # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 5️⃣ Fuse proj × gate with a tiny Triton kernel (in‑place)
    # ------------------------------------------------------------------
    # Ensure tensors are contiguous 1‑D views for the kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    total_elements = left.numel()
    BLOCK = 65536  # 2⁶⁵ ≈ 64 K elements per program (fits in registers)

    grid = (math.ceil(total_elements / BLOCK),)

    # left = left * left_gate  (in‑place)
    _elem_mul_kernel[grid](
        left,
        left_gate,
        left,                     # destination = left (in‑place)
        total_elements,
        BLOCK=BLOCK,
    )
    # right = right * right_gate  (in‑place)
    _elem_mul_kernel[grid](
        right,
        right_gate,
        right,
        total_elements,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Optional mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] → fp16
        mask_f = mask.unsqueeze(-1).to(torch.float16)
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 7️⃣ Batched GEMM: out = left @ rightᵀ   (fp16 → fp32 accumulation)
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden, N, N) → contiguous → (B·hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    total_heads = B * H

    left_flat  = left.view(total_heads, N, N)          # [B·H, N, N]
    right_flat = right.view(total_heads, N, N)         # [B·H, N, N]

    # cuBLAS/H100 Tensor‑core batched GEMM (fp16 inputs, fp32 output)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # [B·H, N, N]

    # Back to [B, N, N, hidden] (still fp16)
    out = (
        out_flat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 8️⃣ Cast to float32 for the remaining LayerNorms / gating
    # ------------------------------------------------------------------
    out = out.float()        # now fp32
    out_gate = out_gate.float()  # fp32 broadcasting compatible

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm (hidden dimension)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 10️⃣ Apply the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11️⃣ Final linear projection back to `dim` (float32)
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
