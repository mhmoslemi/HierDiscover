def _tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, SEQ, HIDDEN,
    stride_a_batch, stride_ai, stride_ak, stride_ah,
    stride_b_batch, stride_bj, stride_bk, stride_bh,
    stride_c_batch, stride_ci, stride_cj, stride_ch,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    TILE_H: tl.constexpr,
):
    # program ids:
    pid_b = tl.program_id(2) # batch index
    pid_i = tl.program_id(0) # i tile
    pid_j = tl.program_id(1) # j tile

    # offsets for i, j
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < SEQ
    mask_j = offs_j < SEQ

    # iterate over hidden tiles
    # Number of hidden tiles:
    num_h_tiles = (HIDDEN + TILE_H - 1) // TILE_H
    for h_tile_idx in range(num_h_tiles):
        # hidden offset for this tile
        h_base = h_tile_idx * TILE_H
        offs_h = h_base + tl.arange(0, TILE_H)
        mask_h = offs_h < HIDDEN

        # accumulator for each hidden channel in tile
        # Create a list of accumulators: acc = [tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32) for _ in range(TILE_H)]

        acc = [tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32) for _ in range(TILE_H)]

        # loop over K dimension tiles
        num_k_tiles = tl.cdiv(SEQ, BLOCK_K)
        for k_tile in range(num_k_tiles):
            k_base = k_tile * BLOCK_K
            offs_k = k_base + tl.arange(0, BLOCK_K)
            mask_k = offs_k < SEQ

            # Load A tile: shape (BLOCK_M, BLOCK_K, TILE_H)

                B_ptr
                + pid_b * stride_b_batch

                + offs_k[:, None, None] * stride_bk
                + offs_j[None, :, None] * stride_bj

            # Build pointer for output

                + offs_i[:, None] * stride_ci

"""
TriMul (outgoing) forward pass – Triton‐accelerated O(N³) contraction.

Algorithm
---------
1. Layer‑norm the input over the feature dimension `dim`.
2. Two linear projections (left/right) → hidden dimension `h`.
3. (Optional) mask the projections.
4. Compute three sigmoid gates from the normalized tensor and apply them:
       left  = left  * left_gate
       right = right * right_gate
5. Cast `left/right` to float16 and run a custom Triton kernel that computes,
   for every batch * hidden channel, the matrix product
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
   The kernel processes many hidden channels (TILE_H = 8) together, loading a
   (BLOCK_M × BLOCK_K × TILE_H) tile from `left` and a (BLOCK_K × BLOCK_N ×
   TILE_H) tile from `right` and accumulating with FP32 dot products.
6. Cast the result back to FP32, apply a second LayerNorm over `hidden`,
   multiply by the `out_gate` and project back to `dim` with a final linear
   layer.
The heavy O(N³) work is performed completely on the GPU with minimal launch
overhead (one program per (i‑tile, j‑tile, batch) instead of per hidden
channel).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: C[b,i,j,h] = Σ_k A[b,i,k,h] * B[b,j,k,h]   (A=left, B=right)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,                     # pointers
    BATCH, SEQ, HIDDEN,                      # scalar sizes
    stride_a_batch, stride_ai, stride_ak, stride_ah,
    stride_b_batch, stride_bj, stride_bk, stride_bh,
    stride_c_batch, stride_ci, stride_cj, stride_ch,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    TILE_H: tl.constexpr,
):
    # --------------------------------------------------------------
    # program IDs: i‑tile, j‑tile, batch
    # --------------------------------------------------------------
    pid_i = tl.program_id(0)      # tile along dimension i
    pid_j = tl.program_id(1)      # tile along dimension j
    pid_b = tl.program_id(2)      # batch index

    # --------------------------------------------------------------
    # offsets for the i/j dimensions
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < SEQ
    mask_j = offs_j < SEQ

    # --------------------------------------------------------------
    # Number of hidden‑channel tiles
    # -----------------------------------------------------------------
    h_tiles = (HIDDEN + TILE_H - 1) // TILE_H

    # -----------------------------------------------------------------
    # Loop over hidden tiles
    # -----------------------------------------------------------------
    for t in range(h_tiles):
        h_base = t * TILE_H
        offs_h = h_base + tl.arange(0, TILE_H)
        mask_h = offs_h < HIDDEN

        # one accumulator per hidden channel inside the tile
        acc = [tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32) for _ in range(TILE_H)]

        # --------------------------------------------------------------
        # Loop over the reduction dimension k
        # --------------------------------------------------------------
        num_k_tiles = tl.cdiv(SEQ, BLOCK_K)
        for k_tile in range(num_k_tiles):
            k_base = k_tile * BLOCK_K
            offs_k = k_base + tl.arange(0, BLOCK_K)
            mask_k = offs_k < SEQ

            # ------------------------- load A tile (i,k,h) -----------------
            a_ptrs = (
                A_ptr
                + pid_b * stride_a_batch
                + offs_i[:, None, None] * stride_ai
                + offs_k[None, :, None] * stride_ak
                + offs_h[None, None, :] * stride_ah
            )
            a = tl.load(
                a_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                other=0.0,
            )  # shape (BLOCK_M, BLOCK_K, TILE_H)

            # ------------------------- load B tile (k,j,h) -----------------
            b_ptrs = (
                B_ptr
                + pid_b * stride_b_batch
                + offs_h[None, None, :] * stride_bh
                + offs_k[:, None, None] * stride_bk
                + offs_j[None, :, None] * stride_bj
            )
            b = tl.load(
                b_ptrs,
                mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
                other=0.0,
            )  # shape (BLOCK_K, BLOCK_N, TILE_H)

            # ------------------------- dot per hidden channel -------------
            for hh in range(TILE_H):
                # a[...,hh] : (BLOCK_M, BLOCK_K) , b[...,hh] : (BLOCK_K, BLOCK_N)
                acc[hh] += tl.dot(
                    a[:, :, hh].to(tl.float32), b[:, :, hh].to(tl.float32)
                )

        # --------------------------------------------------------------
        # Store the results for this hidden tile
        # --------------------------------------------------------------
        for hh in range(TILE_H):
            # Only write when the hidden index is valid (mask_h[hh])
            if mask_h[hh]:
                c_ptr = (
                    C_ptr
                    + pid_b * stride_c_batch
                    + offs_i[:, None] * stride_ci
                    + offs_j[None, :] * stride_cj
                    + (h_base + hh) * stride_ch
                )
                tl.store(
                    c_ptr,
                    acc[hh],
                    mask=mask_i[:, None] & mask_j[None, :],
                )


# ----------------------------------------------------------------------
# Helper that prepares strides, casts to FP16 and launches the kernel
# ----------------------------------------------------------------------
def _batched_trimul_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right: [B, N, N, H]  (float32)
    Returns out: [B, N, N, H] (float32)
    """
    B, N, _, H = left.shape
    # ensure contiguous layout after possible mask/gate ops
    left = left.contiguous()
    right = right.contiguous()

    # cast to FP16 for the heavy contraction
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    out_h = torch.empty_like(left_h)  # FP16 output buffer

    # --------------------------------------------------------------
    # strides (in elements)
    # --------------------------------------------------------------
    stride_a_batch, stride_ai, stride_ak, stride_ah = left_h.stride()
    stride_b_batch, stride_bj, stride_bk, stride_bh = right_h.stride()
    stride_c_batch, stride_ci, stride_cj, stride_ch = out_h.stride()

    # --------------------------------------------------------------
    # Tile sizes – tuned for H100 FP16 matmul
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    TILE_H = 8                         # hidden channels per kernel tile

    assert H % TILE_H == 0, "hidden_dim must be a multiple of TILE_H (8) for this kernel"

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j tiles
        B,                              # batch dimension
    )

    _trimul_outgoing_kernel[grid](
        left_h,
        right_h,
        out_h,
        B,
        N,
        H,
        stride_a_batch,
        stride_ai,
        stride_ak,
        stride_ah,
        stride_b_batch,
        stride_bj,
        stride_bk,
        stride_bh,
        stride_c_batch,
        stride_ci,
        stride_cj,
        stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        TILE_H=TILE_H,
        num_warps=8,
    )

    # Cast back to FP32 for the remaining layers
    return out_h.to(torch.float32)


# ----------------------------------------------------------------------
# Top‑level entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data :
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor [B, N, N]  (float32) – may be ignored
        - weights_dict : mapping from weight name to torch.Tensor
        - config_dict  : contains keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (`dim`)
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → hidden dimension
    # --------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted on the hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and application
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via Triton (left/right fp16)
    # --------------------------------------------------------------
    out = _batched_trimul_outgoing(left, right)          # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------
    # 7️⃣ Multiply by out_gate and final projection back to `dim`
    # --------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
