"""
TriMul (outgoing) forward pass – highly optimized Triton implementation.

The expensive part of the module is the N³ reduction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
which is a batched matrix‑multiply (Gram product) for every head `h`.
The kernel below computes this product directly on the [B,N,N,H] tensors
without any costly transposes or copies.  All other operations
(layer‑norms, linear projections, gating, final linear) are performed
with PyTorch in float32/float16, but the core N³ reduction runs in fp16
with fp32 accumulation and Tensor‑Core acceleration.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def triton_tri_mul_outgoing_kernel(
    # tensors
    left_ptr, right_ptr, out_ptr,
    # problem size
    B, N, H,
    # strides for left   (B, N_i, N_j, H)
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    # strides for right  (B, N_i, N_j, H)
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    # strides for out    (B, N_i, N_j, H)
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batch‑wise Gram product: out = left @ rightᵀ over the sequence dimension.
    One program instance handles a tile (BLOCK_M × BLOCK_N) of the output
    for a single (batch, head) pair.
    """
    pid_head = tl.program_id(0)      # combined batch * head index
    pid_i    = tl.program_id(1)      # tile row (i)
    pid_j    = tl.program_id(2)      # tile col (j)

    # derive batch and head from the flattened head index
    b = pid_head // H
    h = pid_head %  H

    # -----------------------------------------------------------------
    # tile offsets
    # -----------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for the selected (b, h)
    left_head_ptr  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_head_ptr = right_ptr + b * stride_r_b + h * stride_r_h
    out_head_ptr   = out_ptr   + b * stride_o_b + h * stride_o_h

    # accumulator in fp32 (more stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # reduction over the k‑dimension
    # -----------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_head_ptr
            + offs_i[:, None] * stride_l_i   # i dimension
            + offs_k[None, :] * stride_l_j,  # k (second N) dimension
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )   # fp16

        # right block (transposed for dot): shape (BLOCK_K, BLOCK_N)
        # we need right[k, j] → rows = k, cols = j
        right = tl.load(
            right_head_ptr
            + offs_k[:, None] * stride_r_j   # k dimension as rows
            + offs_j[None, :] * stride_r_i,  # j dimension as cols
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )   # fp16

        # fp16 -> fp32 accumulation
        acc += tl.dot(left, right).to(tl.float32)

    # -----------------------------------------------------------------
    # write result (still fp16, stored as fp16 tensor)
    # -----------------------------------------------------------------
    tl.store(
        out_head_ptr
        + offs_i[:, None] * stride_o_i   # i
        + offs_j[None, :] * stride_o_j,  # j
        acc,
        mask=mask_i[:, None] & mask_j[None, :]
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [batch, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    nomask       = config.get("nomask", True)

    # -------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (float32 for stability)
    # -------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                   # [B, N, N, dim]  float32

    # -------------------------------------------------------------
    # 2️⃣ Cast to fp16 for all heavy linear work
    # -------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -------------------------------------------------------------
    # 3️⃣ Linear projections (no bias) – fp16
    # -------------------------------------------------------------
    w_left_proj  = weights["left_proj.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)

    left  = torch.nn.functional.linear(x_fp16, w_left_proj)   # [B, N, N, H] fp16
    right = torch.nn.functional.linear(x_fp16, w_right_proj)  # [B, N, N, H] fp16

    # -------------------------------------------------------------
    # 4️⃣ Optional mask (float16)
    # -------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)      # [B, N, N, 1]
        left  = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------------------
    # 5️⃣ Gating values – compute in float32, cast to fp16 afterwards
    # -------------------------------------------------------------
    w_left_gate  = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate   = weights["out_gate.weight"]

    left_gate  = torch.nn.functional.linear(x, w_left_gate).sigmoid().to(torch.float16)
    right_gate = torch.nn.functional.linear(x, w_right_gate).sigmoid().to(torch.float16)
    out_gate   = torch.nn.functional.linear(x, w_out_gate).sigmoid()   # keep float32

    left  = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------
    # 6️⃣ Triton kernel: out = left @ rightᵀ  (batched Gram product)
    # -------------------------------------------------------------
    B, N, _, _ = left.shape                         # N is the sequence length
    out = torch.empty_like(left)                     # fp16 buffer [B, N, N, H]

    # strides for the three tensors (B, N_i, N_j, H)
    stride_l_b, stride_l_i, stride_l_j, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_j, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tunable block sizes – chosen for H100 tensor‑core usage
    BLOCK_M = 128    # rows of the output tile
    BLOCK_N = 64     # cols of the output tile
    BLOCK_K = 32     # reduction tile (must be a multiple of 16 for fp16)

    grid = (
        B * hidden_dim,                               # one program per (batch, head)
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    triton_tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_j, stride_l_h,
        stride_r_b, stride_r_i, stride_r_j, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------------------
    # 7️⃣ Cast back to float32 for the remaining lightweight ops
    # -------------------------------------------------------------
    out = out.to(torch.float32)                     # [B, N, N, H] FP32

    # -------------------------------------------------------------
    # 8️⃣ Hidden → output LayerNorm, gating and final linear
    # -------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
