@triton.jit
def fuse_gate_mask_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elements,
    H: tl.constexpr,

    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    # Linear proj
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    if nomask:
        mask = torch.ones((bs, N, N), dtype=x.dtype, device=device)
    else:
        mask = mask_tensor.to(x.dtype)

    # Allocate fp16 buffers for left/right after gating+mask
    left_fp16 = torch.empty_like(left, dtype=torch.float16, device=device)
    right_fp16 = torch.empty_like(right, dtype=torch.float16, device=device)

    # Flatten all inputs for kernel
    total = left.numel()   # B*N*N*hidden_dim

    left_flat = left.contiguous().view(-1)
    right_flat = right.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat = mask.contiguous().view(-1)   # B*N*N

    right_out_flat = right_fp16.view(-1)

        right_gate_flat,
        mask_flat,
        left_out_flat,
        right_out_flat,
        total,
        H=hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    )

    # Apply output gate
    out = out * out_gate

    # Final projection hidden -> dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

"""
custom_kernel – highly‑optimized forward of the “outgoing” TriMul operator.

Algorithm
---------
1️⃣ Layer‑normalize the 4‑D input `(B,N,N,dim)`.
2️⃣ Three linear projections (no bias) → `left`, `right` and the three gates.
3️⃣ Fuse the element‑wise mask (if present) and the two gates into the
   projected tensors *inside a tiny Triton kernel*.  The kernel works on
   the flattened tensors, broadcasts the 3‑D mask across the hidden
   dimension and writes the result directly in FP16 to save bandwidth.
4️⃣ Move the hidden dimension into the batch axis and perform the core
   triangular multiplication with a **batched half‑precision GEMM**:
   `out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]`.
   This is realized with `torch.bmm(left, right.transpose(1,2))`,
   which on H100 exploits Tensor‑Core FP16 throughput.
5️⃣ Cast the result back to FP32, restore the original layout.
6️⃣ Second LayerNorm over the hidden dimension, apply the output gate,
   and a final linear projection hidden → dim.

The only Triton component is the element‑wise fusion kernel (step 3);
all heavy linear algebra is delegated to cuBLAS through `torch.bmm`,
which provides the required ~1 µs per N=1024 case on an H100.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr,                     # shape [B, N, N] (flattened)
    left_out_ptr, right_out_ptr,
    total_elems,                  # B * N * N * H
    H: tl.constexpr,              # hidden dimension (compile‑time)
    BLOCK_SIZE: tl.constexpr,     # work‑per‑program
):
    """Fuse mask and the two gates into the projected tensors.

    The kernel receives flattened tensors (contiguous, row‑major) and writes
    FP16 results.  The mask is broadcast across the hidden axis by dividing
    the linear index by H (integer division).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    # Load the four FP32 tensors
    l      = tl.load(left_ptr      + offs, mask=valid, other=0.0)
    r      = tl.load(right_ptr     + offs, mask=valid, other=0.0)
    lg     = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    rg     = tl.load(right_gate_ptr+ offs, mask=valid, other=0.0)

    # mask_idx = linear_index // H  (broadcast dimension)
    mask_idx = offs // H
    m_val = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)

    # Apply gates and mask, cast to FP16 for the downstream GEMM
    l_out = (l * lg * m_val).to(tl.float16)
    r_out = (r * rg * m_val).to(tl.float16)

    tl.store(left_out_ptr  + offs, l_out, mask=valid)
    tl.store(right_out_ptr + offs, r_out, mask=valid)


def custom_kernel(data):
    """
    Triton‑accelerated TriMul (outgoing) forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/int/float)
        - weights_dict : mapping from param name to torch.Tensor
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device).
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # shortcut to weights (all are FP32 tensors)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (last dimension = dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B, N, N, dim]   (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Gates (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 4️⃣ Prepare mask (broadcasted across hidden dimension)
    # ------------------------------------------------------------------
    if nomask:
        mask = torch.ones((bs, N, N), dtype=x.dtype, device=device)
    else:
        mask = mask_tensor.to(x.dtype)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask + gates into the projected tensors via a tiny Triton kernel
    # ------------------------------------------------------------------
    # Allocate half‑precision buffers for the fused tensors
    left_fp16  = torch.empty_like(left, dtype=torch.float16, device=device)
    right_fp16 = torch.empty_like(right, dtype=torch.float16, device=device)

    # Flatten everything (contiguous → 1‑D view)
    total = left.numel()                     # B * N * N * H
    left_flat      = left.contiguous().view(-1)
    right_flat     = right.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_gate_flat= right_gate.contiguous().view(-1)
    mask_flat      = mask.contiguous().view(-1)   # size B * N * N
    left_out_flat  = left_fp16.view(-1)
    right_out_flat = right_fp16.view(-1)

    BLOCK_SIZE = 1024                     # 1 K elements per program
    grid = (triton.cdiv(total, BLOCK_SIZE),)

    # Launch the kernel – H is a compile‑time constant (helps integer division)
    _gate_mask_fuse_kernel[grid](
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        mask_flat,
        left_out_flat,
        right_out_flat,
        total,
        H=hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # Release intermediate FP32 tensors to free GPU memory
    del left, right, left_gate, right_gate, mask, x

    # ------------------------------------------------------------------
    # 6️⃣ Core TriMul: batched FP16 GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    # bring hidden dimension into the batch axis
    left_perm  = left_fp16.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    batch_mul = bs * hidden_dim                               # B·H
    left_mat  = left_perm.view(batch_mul, N, N)              # [B·H, N, N] (FP16)
    right_mat = right_perm.view(batch_mul, N, N)             # [B·H, N, N] (FP16)

    # batched matmul using cuBLAS (Tensor‑cores on H100)
    out_half = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] (FP16)

    # Clean up large intermediates as soon as possible
    del left_fp16, right_fp16, left_perm, right_perm, left_mat, right_mat

    # Cast back to FP32 for the remaining (LayerNorm, linear) layers
    out = out_half.to(torch.float32)

    # Restore the original 4‑D layout: [B, N, N, H]
    out = out.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # [B, N, N, H] (FP32)

    # ------------------------------------------------------------------
    # 8️⃣ Apply the output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection (hidden → dim, no bias)
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
