def matmul_kernel(a_ptr, b_ptr, c_ptr,
                  M, N, K,
                  stride_am, stride_ak,
                  stride_bk, stride_bn,
                  stride_cm, stride_cn,
                  batch_stride_a, batch_stride_b, batch_stride_c,
                  BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)  # value over (batch * num_blocks_m)
    pid_n = tl.program_id(1)

    # compute batch and block_m
    batch = pid // tl.cdiv(M, BLOCK_M)
    pid_m = pid % tl.cdiv(M, BLOCK_M)

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute offsets for a and b
        a_ptrs = a_ptr + batch * batch_stride_a + offs_m[:, None] * stride_am + (k * BLOCK_K + offs_k)[None, :] * stride_ak
        b_ptrs = b_ptr + batch * batch_stride_b + (k * BLOCK_K + offs_k)[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=offs_m < M, other=0.0)
        b = tl.load(b_ptrs, mask=offs_n < N, other=0.0)

        # dot product accumulation
        acc += tl.dot(a, b)

    # write out C
    c_ptrs = c_ptr + batch * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)
```

"""
TriMul (outgoing) forward pass – Triton‑accelerated version
================================================================
Algorithm
---------
1. Layer‑norm the 4‑D input tensor `x` (shape [B, N, N, D]).
2. Two linear projections (no bias) produce `left` and `right` tensors of shape
   [B, N, N, H] (`H = hidden_dim`).
3. Optional mask and three per‑position gates (sigmoid) are fused into a single
   Triton kernel that computes `tensor * mask * gate`.
4. Core contraction
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   is a batched matrix multiplication for each hidden channel.  We
   permute to shapes (B*H, N, N) and (B*H, N, N) (the second matrix
   already transposed on the two N axes) and run a custom Triton GEMM
   kernel that works on FP16 inputs and accumulates in FP32.
5. Layer‑norm over the hidden dimension, apply the output gate (sigmoid)
   and a final linear projection back to the original feature dimension.
All heavy compute (step 3‑4) is done on the GPU with Triton; the remaining
operations use highly‑optimized Torch primitives.

The implementation targets NVIDIA H100 (Tensor‑cores) and aims to run the
full forward pass in ≈1 ms for the provided benchmark sizes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Element‑wise fused multiplication: a * b * c
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,  # pointers
    size,                           # total number of elements
    BLOCK_SIZE: tl.constexpr,       # compile‑time constant
):
    """out = a * b * c (contiguous 1‑D tensors)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)

    tl.store(out_ptr + offs, a * b * c, mask=mask)


def fused_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """Fused element‑wise product of three tensors using Triton."""
    assert a.shape == b.shape == c.shape
    a, b, c = a.contiguous(), b.contiguous(), c.contiguous()
    out = torch.empty_like(a)

    total = a.numel()
    BLOCK = 1024
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_kernel[grid](
        a, b, c, out, total, BLOCK_SIZE=BLOCK
    )
    return out


# ----------------------------------------------------------------------
# 2️⃣  Batched FP16 GEMM (A @ B) where A: (B, M, K), B: (B, K, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # strides for A (batch, row, col)
    stride_bk, stride_bn,          # strides for B (batch, row(K), col(N))
    stride_cm, stride_cn,          # strides for C (batch, row, col)
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C[b] = A[b] @ B[b]   (FP16 → FP32 accumulation)."""
    pid = tl.program_id(0)      # combined (batch, block_m)
    pid_n = tl.program_id(1)    # block_n

    # --------------------------------------------------------------
    # De‑compose pid into batch index and block‑row index
    # --------------------------------------------------------------
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    batch = pid // num_blocks_m
    pid_m = pid % num_blocks_m

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # --------------------------------------------------------------
    # Accumulator in FP32
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K‑tiles
    # --------------------------------------------------------------
    num_tiles_k = tl.cdiv(K, BLOCK_K)
    for tile_k in range(num_tiles_k):
        k_start = tile_k * BLOCK_K

        # A block: (BLOCK_M, BLOCK_K)
        a_offset = (
            batch * batch_stride_a
            + offs_m[:, None] * stride_am
            + (k_start + offs_k)[None, :] * stride_ak
        )
        a_mask = (offs_m[:, None] < M) & ((k_start + offs_k)[None, :] < K)
        a = tl.load(a_ptr + a_offset, mask=a_mask, other=0.0)

        # B block: (BLOCK_K, BLOCK_N)
        b_offset = (
            batch * batch_stride_b
            + (k_start + offs_k)[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b_mask = ((k_start + offs_k)[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptr + b_offset, mask=b_mask, other=0.0)

        # Dot product (+ accumulation)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write back C
    # --------------------------------------------------------------
    c_offset = (
        batch * batch_stride_c
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptr + c_offset, acc.to(tl.float16), mask=c_mask)


def batched_gemm_fp16(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for FP16 tensors:
        C = A @ B
    A: (B, M, K)  FP16, row‑major
    B: (B, K, N)  FP16, row‑major
    Returns C of shape (B, M, N) in FP16.
    """
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, Kb, N = B.shape
    assert K == Kb

    # Output allocation
    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Block sizes – tuned for H100 Tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Strides for contiguous row‑major tensors
    stride_am = K                     # A row stride (M → K)
    stride_ak = 1                     # A col stride
    stride_bk = N                     # B row stride (K → N)
    stride_bn = 1                     # B col stride
    stride_cm = N                     # C row stride (M → N)
    stride_cn = 1                     # C col stride

    batch_stride_a = M * K
    batch_stride_b = K * N
    batch_stride_c = M * N

    # Grid: (batch * num_blocks_m, num_blocks_n)
    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch * num_blocks_m, num_blocks_n)

    batched_gemm_kernel[grid](
        A, B, C,
        M, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                # good occupancy on H100
    )
    return C


# ----------------------------------------------------------------------
# 3️⃣  Entrypoint – the custom TriMul kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: Tensor [B, N, N, D]            (float32, CUDA)
            - mask_tensor:  Tensor [B, N, N]               (float32 or bool, CUDA)
            - weights_dict: model weights (see skeleton)
            - config_dict: contains at least "dim", "hidden_dim",
                           optional "eps" and "nomask".

    Returns:
        Tensor of shape [B, N, N, D] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Gating tensors (sigmoid of a linear projection of x_norm)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask + gates – fused element‑wise kernel when mask present
    # ------------------------------------------------------------------
    if nomask:
        left = left * left_gate
        right = right * right_gate
    else:
        # mask: (B, N, N) → (B, N, N, 1) → broadcast over hidden dim
        mask_exp = mask.unsqueeze(-1).type_as(left)
        left = fused_mul3(left, mask_exp, left_gate)
        right = fused_mul3(right, mask_exp, right_gate)

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction (batched GEMM) – all in FP16 for speed
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Permute to (B, H, M, K) and (B, H, K, N) respectively
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (M=i, K=k)
    right_perm = right_fp16.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (K=k, N=j)

    # Collapse batch+hidden into a single GEMM batch dimension
    batch_hidden = B * hidden_dim
    left_flat = left_perm.view(batch_hidden, N, N)   # (B*H, M=N, K=N)
    right_flat = right_perm.view(batch_hidden, N, N)  # (B*H, K=N, N=N)

    # Batched matrix multiplication using our Triton kernel
    out_flat = batched_gemm_fp16(left_flat, right_flat)  # (B*H, N, N) FP16

    # Reshape back to [B, N, N, hidden_dim] and cast to FP32 for the following layers
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(inp.dtype)

    # ------------------------------------------------------------------
    # 6️⃣ Output layer‑norm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 7️⃣ Output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
