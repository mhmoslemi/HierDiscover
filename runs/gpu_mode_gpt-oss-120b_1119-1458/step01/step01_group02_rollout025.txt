def _trimul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, H, N, K,
    stride_a_batch, stride_a_h, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_h, stride_b_j, stride_b_k,
    stride_c_batch, stride_c_h, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0) # i tile
    pid_j = tl.program_id(1) # j tile
    pid_z = tl.program_id(2) # combined batch_hidden index
    # compute batch and hidden indices
    hidden = pid_z % H
    batch = pid_z // H

    # compute offsets in i and j axes
    i_offsets = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offsets = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_offsets < N
    mask_j = j_offsets < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)

        # load A tile
        a_ptrs = (A_ptr
                  + batch * stride_a_batch
                  + hidden * stride_a_h
                  + i_offsets[:, None] * stride_a_i
                  + k_offset[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & (k_offset[None, :] < K), other=0.0).to(tl.float32)

        # load B tile (transposed)
        b_ptrs = (B_ptr
                  + batch * stride_b_batch
                  + hidden * stride_b_h
                  + k_offset[:, None] * stride_b_k
                  + j_offsets[None, :] * stride_b_j)
        b = tl.load(b_ptrs, mask=(k_offset[:, None] < K) & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # store
    c_ptrs = (C_ptr
              + batch * stride_c_batch
              + hidden * stride_c_h
              + i_offsets[:, None] * stride_c_i
              + j_offsets[None, :] * stride_c_j)
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
Optimized outgoing TriMul forward implementation.

Algorithm:
1. Layer‑norm over the feature dimension.
2. Fuse the five linear projections (left_proj, right_proj, left_gate,
   right_gate, out_gate) into a single matmul with a concatenated weight
   matrix.  The result is reshaped to obtain each projection/gate.
3. Apply the optional mask, sigmoid gates and cast the gated projections
   to float16.
4. Compute the heavy O(N³) contraction
        C[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   with a custom Triton kernel that works directly on the original
   [B,N,N,H] layout (no extra permute/copy) and accumulates in FP32.
5. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate and project back to the original feature dimension.

Only the O(N³) contraction is Triton‑implemented; all other ops use
high‑performance PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_outgoing_kernel(
    A_ptr,                     # *half    [B, N, N, H] (left)
    B_ptr,                     # *half    [B, N, N, H] (right)
    C_ptr,                     # *float32 [B, N, N, H] (output)
    N, K,                      # i32 sizes (both = seq_len)
    stride_a_batch, stride_a_h, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_h, stride_b_j, stride_b_k,
    stride_c_batch, stride_c_h, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    HIDDEN: tl.constexpr,
):
    # program IDs
    pid_i = tl.program_id(0)          # i‑tile index
    pid_j = tl.program_id(1)          # j‑tile index
    pid_z = tl.program_id(2)          # combined batch‑hidden index

    # ------------------------------------------------------------------
    # Decode batch and hidden channel from pid_z
    # ------------------------------------------------------------------
    hid = pid_z % HIDDEN
    batch = pid_z // HIDDEN

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ------------------------------------------------------------------
        # Load A tile: shape (BLOCK_M, BLOCK_K)  (i, k)
        # ------------------------------------------------------------------
        a_ptrs = (
            A_ptr
            + batch * stride_a_batch
            + hid * stride_a_h
            + offs_i[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & (offs_k[None, :] < K),
                    other=0.0)                # half

        # ------------------------------------------------------------------
        # Load B tile transposed: shape (BLOCK_K, BLOCK_N)  (k, j)
        # ------------------------------------------------------------------
        b_ptrs = (
            B_ptr
            + batch * stride_b_batch
            + hid * stride_b_h
            + offs_k[:, None] * stride_b_k
            + offs_j[None, :] * stride_b_j
        )
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_j[None, :],
                    other=0.0)                # half

        # convert to FP32 for accumulation (dot returns half, we cast)
        acc += tl.dot(a, b).to(tl.float32)

    # ------------------------------------------------------------------
    # Write back the result tile
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + batch * stride_c_batch
        + hid * stride_c_h
        + offs_i[:, None] * stride_c_i
        + offs_j[None, :] * stride_c_j
    )
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
    where `left` and `right` are half‑precision tensors of shape
    [B, N, N, H].  Returns a FP32 tensor of the same shape.
    """
    B, N, _, H = left.shape
    # output buffer (FP32)
    out = torch.empty((B, N, N, H), device=left.device, dtype=torch.float32)

    # strides (in elements)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h = right.stride()
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h = out.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_outgoing_kernel[grid](
        left,
        right,
        out,
        N,
        N,
        stride_a_batch, stride_a_h, stride_a_i, stride_a_k,
        stride_b_batch, stride_b_h, stride_b_j, stride_b_k,
        stride_c_batch, stride_c_h, stride_c_i, stride_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN=H,
        num_warps=8,
    )
    return out

        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask_tensor   : torch.Tensor [B, N, N] (may be bool or float)
        - weights_dict  : mapping from weight name to torch.Tensor
        - config_dict   : contains keys ``dim``, ``hidden_dim`` and optional ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device as ``input_tensor``).
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # If we are on a CPU fallback to the reference implementation
    # ------------------------------------------------------------------
    if not input_tensor.is_cuda:
        # reference implementation (no Triton)
        x = torch.nn.functional.layer_norm(
            input_tensor,
            (dim,),
            weight=weights["norm.weight"],
            bias=weights["norm.bias"],
        )
        # five linear projections fused manually without Triton
        left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
        right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
        left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
        right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
        out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

        if not nomask:
            mask = mask_tensor.unsqueeze(-1).type_as(x)
            left_proj = left_proj * mask
            right_proj = right_proj * mask

        left = left_proj * left_gate
        right = right_proj * right_gate
        out = torch.einsum('... i k d, ... j k d -> ... i j d', left, right)
        out = torch.nn.functional.layer_norm(
            out,
            (hidden_dim,),
            weight=weights["to_out_norm.weight"],
            bias=weights["to_out_norm.bias"],
        )
        out = out * out_gate
        out = torch.nn.functional.linear(out, weights["to_out.weight"])
        return out

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension ``dim``
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear transforms into one matmul
    # ------------------------------------------------------------------
    # order: left_proj, right_proj, left_gate, right_gate, out_gate
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    )  # shape [(5*hidden_dim), dim]

    # combined linear (still FP32)
    combined = torch.nn.functional.linear(x, fused_weight)  # [B,N,N,5*hidden_dim]
    B, N, _, _ = combined.shape  # keep original B,N
    combined = combined.view(B, N, N, 5, hidden_dim)

    left_proj = combined[..., 0, :]            # [B,N,N,hidden]
    right_proj = combined[..., 1, :]
    left_gate = torch.sigmoid(combined[..., 2, :])
    right_gate = torch.sigmoid(combined[..., 3, :])
    out_gate = torch.sigmoid(combined[..., 4, :])

    # ------------------------------------------------------------------
    # 3️⃣ Apply optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).type_as(left_proj)  # [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gate the projections and cast to half‑precision for the matmul
    # ------------------------------------------------------------------
    left = (left_proj * left_gate).to(torch.float16)   # [B,N,N,hidden]
    right = (right_proj * right_gate).to(torch.float16)

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via Triton
    # ------------------------------------------------------------------
    out = _batched_matmul_outgoing(left, right)      # [B,N,N,hidden] (FP32)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 7️⃣ Gate and final projection back to ``dim``
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
Optimized outgoing TriMul forward implementation.

Algorithm
---------
1️⃣ Layer‑norm over the feature dimension ``dim``.
2️⃣ Fuse the five linear transforms (left_proj, right_proj, left_gate,
   right_gate, out_gate) into a single matrix multiplication with a
   concatenated weight matrix.  After reshaping we obtain the
   individual projections/gates.
3️⃣ Apply the optional mask, sigmoid gates and cast the gated
   projections to ``float16`` (half‑precision) to reduce memory traffic.
4️⃣ Compute the heavy O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   with a custom Triton kernel that works directly on the original
   ``[B,N,N,H]`` layout (no extra permute/copy).  The kernel accumulates
   in FP32 and writes the result as FP32.
5️⃣ Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate and project back to the original ``dim`` with a final
   linear layer.

Only the O(N³) contraction is Triton‑implemented; all other ops use
high‑performance PyTorch primitives.  The implementation avoids the
expensive ``permute(...).contiguous()`` copies from the reference code
and uses half‑precision for the matmul to substantially reduce runtime.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_outgoing_kernel(
    A_ptr,                     # *half    [B, N, N, H]   (left)
    B_ptr,                     # *half    [B, N, N, H]   (right)
    C_ptr,                     # *float32 [B, N, N, H]   (output)
    N, K,                      # i32 sizes (both = seq_len)
    stride_a_batch, stride_a_h, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_h, stride_b_j, stride_b_k,
    stride_c_batch, stride_c_h, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    HIDDEN: tl.constexpr,
):
    pid_i = tl.program_id(0)          # i‑tile index
    pid_j = tl.program_id(1)          # j‑tile index
    pid_z = tl.program_id(2)          # combined batch‑hidden index

    # --------------------------------------------------------------
    # Decode batch and hidden channel from pid_z
    # --------------------------------------------------------------
    hid = pid_z % HIDDEN
    batch = pid_z // HIDDEN

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ----------------------------------------------------------
        # Load A tile: (i, k)
        # ----------------------------------------------------------
        a_ptrs = (
            A_ptr
            + batch * stride_a_batch
            + hid * stride_a_h
            + offs_i[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )  # half

        # ----------------------------------------------------------
        # Load B tile transposed: (k, j)
        # ----------------------------------------------------------
        b_ptrs = (
            B_ptr
            + batch * stride_b_batch
            + hid * stride_b_h
            + offs_k[:, None] * stride_b_k
            + offs_j[None, :] * stride_b_j
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=0.0,
        )  # half

        # dot on half inputs, cast result to FP32 for accumulation
        acc += tl.dot(a, b).to(tl.float32)

    # --------------------------------------------------------------
    # Write back the result tile
    # --------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + batch * stride_c_batch
        + hid * stride_c_h
        + offs_i[:, None] * stride_c_i
        + offs_j[None, :] * stride_c_j
    )
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
    where `left` and `right` are half‑precision tensors of shape
    [B, N, N, H].  Returns a FP32 tensor of the same shape.
    """
    B, N, _, H = left.shape
    out = torch.empty((B, N, N, H), device=left.device, dtype=torch.float32)

    # strides (in elements)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h = right.stride()
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h = out.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_outgoing_kernel[grid](
        left,
        right,
        out,
        N,
        N,
        stride_a_batch,
        stride_a_h,
        stride_a_i,
        stride_a_k,
        stride_b_batch,
        stride_b_h,
        stride_b_j,
        stride_b_k,
        stride_c_batch,
        stride_c_h,
        stride_c_i,
        stride_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN=H,
        num_warps=8,
    )
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[
        torch.Tensor,                     # input tensor [B, N, N, dim]
        torch.Tensor,                     # mask tensor  [B, N, N]
        Dict[str, torch.Tensor],          # model weights
        Dict[str, Any],                   # configuration
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask_tensor   : torch.Tensor [B, N, N] (may be bool or float)
        - weights_dict   : mapping from weight name to torch.Tensor
        - config_dict    : contains ``dim``, ``hidden_dim`` and optional
                           ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device as
        ``input_tensor``).
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # CPU fallback (no Triton)
    # ------------------------------------------------------------------
    if not input_tensor.is_cuda:
        x = torch.nn.functional.layer_norm(
            input_tensor,
            (dim,),
            weight=weights["norm.weight"],
            bias=weights["norm.bias"],
        )
        left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
        right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
        left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
        right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
        out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

        if not nomask:
            mask = mask_tensor.unsqueeze(-1).type_as(x)
            left_proj = left_proj * mask
            right_proj = right_proj * mask

        left = left_proj * left_gate
        right = right_proj * right_gate
        out = torch.einsum('... i k d, ... j k d -> ... i j d', left, right)
        out = torch.nn.functional.layer_norm(
            out,
            (hidden_dim,),
            weight=weights["to_out_norm.weight"],
            bias=weights["to_out_norm.bias"],
        )
        out = out * out_gate
        out = torch.nn.functional.linear(out, weights["to_out.weight"])
        return out

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear transforms into a single matmul
    # ------------------------------------------------------------------
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    )  # [(5*hidden_dim), dim]

    # combined output: [B, N, N, 5*hidden_dim]
    combined = torch.nn.functional.linear(x, fused_weight)
    B, N, _, _ = combined.shape
    combined = combined.view(B, N, N, 5, hidden_dim)

    left_proj = combined[..., 0, :]          # [B, N, N, H]
    right_proj = combined[..., 1, :]         # [B, N, N, H]
    left_gate = torch.sigmoid(combined[..., 2, :])
    right_gate = torch.sigmoid(combined[..., 3, :])
    out_gate = torch.sigmoid(combined[..., 4, :])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).type_as(left_proj)  # [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4️⃣ Apply gates and cast to half‑precision for the matmul
    # ------------------------------------------------------------------
    left = (left_proj * left_gate).to(torch.float16)   # [B,N,N,hidden]
    right = (right_proj * right_gate).to(torch.float16)

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction via Triton
    # ------------------------------------------------------------------
    out = _batched_matmul_outgoing(left, right)      # FP32, shape [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 7️⃣ Gate and final projection back to ``dim``
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
