Pseudo:

@triton.jit
def tri_mul_kernel( left_ptr, right_ptr, out_ptr, N, stride_lh, stride_li, stride_lk, stride_rh, stride_rj, stride_rk, stride_oh, stride_oi, stride_oj, BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
   pid_h = tl.program_id(0)
   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)

   # compute offsets in i, j as before
   off_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   off_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

   # masks
   mask_i = off_i < N
   mask_j = off_j < N

   # pointer base to head
   left_base = left_ptr + pid_h * stride_lh
   right_base = right_ptr + pid_h * stride_rh
   out_base = out_ptr + pid_h * stride_oh

   # allocate accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # prefetch first K tile
   # maybe load first tile into registers
   num_k_blocks = tl.cdiv(N, BLOCK_K)
   k = 0
   # sets of pointers for current tile
   # store tile loops: double buffering
   # Use static for loop over k blocks
   for kb in range(num_k_blocks):
       k0 = kb * BLOCK_K
       off_k = k0 + tl.arange(0, BLOCK_K)
       mask_k = off_k < N

       # left tile
       left_ptrs = left_base + off_i[:, None] * stride_li + off_k[None, :] * stride_lk
       left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

       # right tile (need transpose)
       right_ptrs = right_base + off_k[:, None] * stride_rk + off_j[None, :] * stride_rj
       right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

       acc += tl.dot(left, right)

   # store
   out_ptrs = out_base + off_i[:, None] * stride_oi + off_j[None, :] * stride_oj
   tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def _tri_mul_kernel(

    N, 
    stride_lh, stride_li, stride_lk,
    stride_rh, stride_rj, stride_rk,
    stride_oh, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_h = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start offsets
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N

"""
TriMul – “outgoing” update (AlphaFold‑3) implemented with a custom Triton GEMM.
The heavy `einsum('... i k d, ... j k d -> ... i j d')` is expressed as a batch
of independent matrix‑multiplications C = A @ Bᵀ where each head corresponds to
one hidden dimension.  The kernel is a classic tiled fp16 GEMM with fp32
accumulation (Tensor‑Core friendly).  All surrounding LayerNorms, linear
projections and gating are performed with PyTorch; only the O(N³) matmul is
off‑loaded to Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel:  out[head, i, j] = Σ_k left[head, i, k] * right[head, j, k]
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16 *, shape (head, i, k)
    right_ptr,         # fp16 *, shape (head, j, k)
    out_ptr,           # fp32 *, shape (head, i, j)
    N,                 # i32  – sequence length
    stride_lh, stride_li, stride_lk,   # i64 strides for left
    stride_rh, stride_rj, stride_rk,   # i64 strides for right
    stride_oh, stride_oi, stride_oj,   # i64 strides for out
    BLOCK_M: tl.constexpr,  # tile size in i‑dimension
    BLOCK_N: tl.constexpr,  # tile size in j‑dimension
    BLOCK_K: tl.constexpr,  # tile size in k‑dimension
):
    pid_h = tl.program_id(0)               # which head  (B * hidden_dim)
    pid_m = tl.program_id(1)               # block row   (i)
    pid_n = tl.program_id(2)               # block col   (j)

    # ------------------------------------------------------------------
    # i‑ and j‑tile offsets + boundary masks
    # ------------------------------------------------------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Pointers to the beginning of the head slice
    # ------------------------------------------------------------------
    left_head  = left_ptr  + pid_h * stride_lh
    right_head = right_ptr + pid_h * stride_rh
    out_head   = out_ptr   + pid_h * stride_oh

    # ------------------------------------------------------------------
    # Accumulator (fp32) – one tile per program
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K in blocks
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        k0 = kb * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile:   [i, k]
        left_ptrs = (left_head
                     + offs_i[:, None] * stride_li
                     + offs_k[None, :] * stride_lk)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0)                     # fp16, dot will up‑cast

        # right tile (transposed on‑the‑fly): [k, j]
        right_ptrs = (right_head
                      + offs_k[:, None] * stride_rk
                      + offs_j[None, :] * stride_rj)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0)                    # fp16

        # matrix‑multiply‑accumulate
        acc += tl.dot(left, right)            # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ------------------------------------------------------------------
    # Store the computed tile
    # ------------------------------------------------------------------
    out_ptrs = (out_head
                + offs_i[:, None] * stride_oi
                + offs_j[None, :] * stride_oj)
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entrypoint used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    nomask       = config.get("nomask", True)
    eps          = 1e-5
    device       = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (last dimension = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → hidden_dim
    # ------------------------------------------------------------------
    # Cast weights to the same dtype as the activation for faster matmuls
    left  = F.linear(x,  weights["left_proj.weight"].to(x.dtype))   # [B,N,N,hidden]
    right = F.linear(x,  weights["right_proj.weight"].to(x.dtype))

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(x.dtype)         # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating values (sigmoid of a linear layer)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(x.dtype))).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(x.dtype))).to(torch.float16)
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(x.dtype)))   # keep fp32

    # ------------------------------------------------------------------
    # 5️⃣ Apply gates (keep the tensors in fp16 for the GEMM)
    # ------------------------------------------------------------------
    left  = left.to(torch.float16)  * left_gate   # [B,N,N,hidden] fp16
    right = right.to(torch.float16) * right_gate  # [B,N,N,hidden] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Reshape to (total_heads, N, N) where total_heads = B * hidden_dim
    #     Layout after permute: (head, i, k) / (head, j, k) – good for
    #     contiguous loads in the kernel.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    total_heads = B * hidden_dim

    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # ------------------------------------------------------------------
    # 7️⃣ Allocate output buffer (fp32, as required by the subsequent
    #     LayerNorm → out_gate → final linear)
    # ------------------------------------------------------------------
    out_flat = torch.empty(
        (total_heads, N, N), dtype=torch.float32, device=device
    )

    # ------------------------------------------------------------------
    # 8️⃣ Triton launch configuration – tiled GEMM
    # ------------------------------------------------------------------
    BLOCK_M = 128          # rows   per program
    BLOCK_N = 128          # cols   per program
    BLOCK_K = 64           # reduction dim per iteration (powers of two)

    grid = (
        total_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Reshape back to the original 4‑D layout
    # ------------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)          # [B, N, N, hidden]
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 10️⃣ Post‑processing:
    #      • LayerNorm over hidden_dim
    #      • Multiply by out_gate (broadcast over hidden_dim)
    #      • Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
