def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    B,N,_,C = input_tensor.shape
    dim = config['dim']
    hidden = config['hidden_dim']

    # extract weights
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    left_proj_weight = weights['left_proj.weight']
    right_proj_weight = weights['right_proj.weight']
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    to_out_weight = weights['to_out.weight']

    # LayerNorm (float32)
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5
    )  # float32

    # Cast to half
    x = x_norm.half()

    # Projections
    left = torch.nn.functional.linear(x, left_proj_weight)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Gating values
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight)).half()
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight)).half()
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight)).half()

    # Apply mask if any (mask shape [B,N,N] boolean or float)
    if mask is not None:
        mask_half = mask.to(dtype=torch.float16).unsqueeze(-1)  # [B,N,N,1]
    else:
        mask_half = None

    # Fuse gating and (optional) mask using Triton kernel (elementwise)
    left, right = fuse_gate_mask(left, left_gate, right, right_gate, mask_half)

    # Contraction
    # permute right to transpose N axes
    right_t = right.permute(0, 2, 1, 3).contiguous()

    # reshape for batched GEMM
    # left: [B,N,N,H] -> [B*H,N,N]
    # right_t similarly
    left_batched = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_batched = right_t.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    out_batched = torch.bmm(left_batched, right_batched)  # (B*H, N, N)

    out_fp32 = out.to(torch.float32)
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        normalized_shape=(hidden,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5
    ).half()

    # Apply output gate and final linear
    gated = out_norm * out_gate
    final = torch.nn.functional.linear(gated, to_out_weight)   # (B,N,N,dim) half

    return final.to(torch.float32)

"""
TriMul (outgoing) – fast forward implementation.
-------------------------------------------------
The forward pass consists of:
1. Layer‑norm over the channel dimension C.
2. Two linear projections (left / right) from C → H (no bias).
3. Optional pairwise mask (broadcast on H).
4. Per‑channel gating (sigmoid over a linear projection of the normed tensor).
5. Dense N³ contraction:   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   This is a batched matrix‑multiply (left @ rightᵀ) for every head h.
   A single Triton kernel is used to fuse the (optional) mask with the two
   gating multiplications – a cheap element‑wise kernel that saves two
   memory passes.
6. Layer‑norm over the hidden dimension H.
7. Output gate (sigmoid) and final linear projection back to C.
All heavy numeric work (projections, N³ contraction, final linear) is done
in FP16 using Tensor‑Cores; the two layer‑norms run in FP32 for stability.
The result is returned as FP32.

The N³ contraction is implemented with a cuBLAS batched GEMM
(`torch.bmm`) which is far faster than a pure‑Triton implementation
on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse left/right gating and (optional) mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # const fp16 *
    left_gate_ptr,          # const fp16 *
    right_ptr,              # const fp16 *
    right_gate_ptr,         # const fp16 *
    mask_ptr,               # const fp16 *   (only used when HAVE_MASK==1)
    left_out_ptr,           # fp16 *
    right_out_ptr,          # fp16 *
    B, N, H,                # meta‑data
    # strides (in elements)
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_m_b, stride_m_i, stride_m_j,
    stride_lo_b, stride_lo_i, stride_lo_j, stride_lo_h,
    stride_ro_b, stride_ro_i, stride_ro_j, stride_ro_h,
    BLOCK_SIZE: tl.constexpr,
    HAVE_MASK: tl.constexpr
):
    """Fuse element‑wise mask + gating:
       left_out  = left  * left_gate  * mask
       right_out = right * right_gate * mask
       mask is optional (HAVE_MASK = 0 → mask = 1.0)."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    valid = offs < total

    # ----- decode linear index into (b,i,j,h) -----
    nnh = N * N * H          # size of one batch slice
    nh  = N * H
    b = offs // nnh
    rem = offs - b * nnh
    i = rem // nh
    rem2 = rem - i * nh
    j = rem2 // H
    h = rem2 - j * H

    # ----- compute flat offsets for every tensor -----
    # left
    l_off = (b * stride_l_b + i * stride_l_i + j * stride_l_j +
             h * stride_l_h)
    # left gate
    lg_off = (b * stride_lg_b + i * stride_lg_i + j * stride_lg_j +
              h * stride_lg_h)
    # right
    r_off = (b * stride_r_b + i * stride_r_i + j * stride_r_j +
             h * stride_r_h)
    # right gate
    rg_off = (b * stride_rg_b + i * stride_rg_i + j * stride_rg_j +
              h * stride_rg_h)

    # mask (no h dimension)
    if HAVE_MASK:
        m_off = b * stride_m_b + i * stride_m_i + j * stride_m_j
        mask = tl.load(mask_ptr + m_off, mask=valid, other=0.0)
    else:
        mask = 1.0

    # ----- load values -----
    left_val      = tl.load(left_ptr      + l_off,  mask=valid, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + lg_off, mask=valid, other=0.0)
    right_val      = tl.load(right_ptr      + r_off,  mask=valid, other=0.0)
    right_gate_val = tl.load(right_gate_ptr + rg_off, mask=valid, other=0.0)

    # ----- compute fused results -----
    left_out_val  = left_val      * left_gate_val  * mask
    right_out_val = right_val     * right_gate_val * mask

    # ----- store -----
    lo_off = (b * stride_lo_b + i * stride_lo_i + j * stride_lo_j +
              h * stride_lo_h)
    ro_off = (b * stride_ro_b + i * stride_ro_i + j * stride_ro_j +
              h * stride_ro_h)

    tl.store(left_out_ptr  + lo_off, left_out_val,  mask=valid)
    tl.store(right_out_ptr + ro_off, right_out_val, mask=valid)


def _fuse_gate_and_mask(left, left_gate, right, right_gate, mask):
    """
    Fuse (left * left_gate) and (right * right_gate) together with an
    optional mask using a Triton kernel.  When ``mask`` is ``None`` the
    operation collapses to a simple element‑wise product performed by
    PyTorch (the cost of the kernel would outweigh its benefit).
    """
    if mask is None:
        # No mask – simple element‑wise multiplication is fast enough.
        left_out = left * left_gate
        right_out = right * right_gate
        return left_out, right_out

    # mask shape: [B, N, N, 1]  (already broadcast‑compatible)
    B, N, _, H = left.shape
    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    # strides (in element units)
    s_l_b, s_l_i, s_l_j, s_l_h = left.stride()
    s_lg_b, s_lg_i, s_lg_j, s_lg_h = left_gate.stride()
    s_r_b, s_r_i, s_r_j, s_r_h = right.stride()
    s_rg_b, s_rg_i, s_rg_j, s_rg_h = right_gate.stride()
    s_m_b, s_m_i, s_m_j = mask.stride()
    s_lo_b, s_lo_i, s_lo_j, s_lo_h = left_out.stride()
    s_ro_b, s_ro_i, s_ro_j, s_ro_h = right_out.stride()

    total = B * N * N * H
    BLOCK_SIZE = 1024                     # 1 K threads → good occupancy
    grid = (triton.cdiv(total, BLOCK_SIZE),)

    _gate_mask_fuse_kernel[grid](
        left, left_gate,
        right, right_gate,
        mask,
        left_out, right_out,
        B, N, H,
        s_l_b, s_l_i, s_l_j, s_l_h,
        s_lg_b, s_lg_i, s_lg_j, s_lg_h,
        s_r_b, s_r_i, s_r_j, s_r_h,
        s_rg_b, s_rg_i, s_rg_j, s_rg_h,
        s_m_b, s_m_i, s_m_j,
        s_lo_b, s_lo_i, s_lo_j, s_lo_h,
        s_ro_b, s_ro_i, s_ro_j, s_ro_h,
        BLOCK_SIZE=BLOCK_SIZE,
        HAVE_MASK=1,
        num_warps=4,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold‑3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dict with all model weight tensors
        - config_dict  : contains at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper to move model parameters to the correct device/dtype
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w   = to_half(weights["left_proj.weight"])
    right_proj_w  = to_half(weights["right_proj.weight"])
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numeric stability)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )                     # [B, N, N, C]  (float32)

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for all heavy work
    # ------------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias) – FP16 matmuls (cuBLAS)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4. Gating values (sigmoid of a linear projection)
    #    They are kept in FP16 – the error is negligible for the target
    #    tolerance and yields a huge speed‑up.
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden_dim] (fp16)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5. Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask → float16 and add fake hidden dim for broadcasting
        mask_fp16 = mask.to(dtype=torch.float16).unsqueeze(-1)  # [B,N,N,1]
    else:
        mask_fp16 = None

    # ------------------------------------------------------------------
    # 6. Fuse mask + gating + projection with a tiny Triton kernel
    # ------------------------------------------------------------------
    left, right = _fuse_gate_and_mask(left, left_gate, right, right_gate, mask_fp16)

    # ------------------------------------------------------------------
    # 7. N³ contraction: batched GEMM (left @ rightᵀ) per head
    #    We use torch.bmm which maps to cuBLAS Tensor‑Core kernels.
    # ------------------------------------------------------------------
    # Transpose the two N axes of ``right`` (k ↔ j)
    right_t = right.permute(0, 2, 1, 3).contiguous()          # [B, N, N, hidden_dim]

    # Reshape to (B*H, N, N) for a single batched GEMM
    left_b = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_b = right_t.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matmul (FP16 → Tensor‑Core)
    out_batched = torch.bmm(left_b, right_b)                 # (B*H, N, N)

    # Restore the original 4‑D layout
    out = out_batched.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 8. Output LayerNorm over hidden dimension (float32 for stability)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 9. Apply output gate and final linear projection back to C
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                      # still FP16
    final = F.linear(gated, to_out_w)                # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 10. Return FP32 as required by the specification
    # ------------------------------------------------------------------
    return final.to(torch.float32)
