```python
import torch
import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_gemm_kernel(
    L_ptr, R_ptr, O_ptr,
    N,
    HIDDEN_DIM,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    stride_bL, stride_iL, stride_kL, stride_dL,
    stride_bR, stride_iR, stride_kR, stride_dR,
    stride_bO, stride_iO, stride_jO, stride_dO,
):
    """
    Batched GEMM for outgoing Triangle Multiplication.
    Computes:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    Input tensors are in layout [B, N, N, D] (contiguous).
    The kernel iterates over tiles (BLOCK_M, BLOCK_N) of the (i,j) plane
    for each combined batch index (b * D + d).
    """
    # 1) Identify the tile (i,j) and the combined batch/hid index.
    pid_tile = tl.program_id(0)          # tile identifier within a batch/hidden slice
    batch_hidden = tl.program_id(1)      # combined batch*hidden index

    # Derive batch and hidden ids.
    b = batch_hidden // HIDDEN_DIM
    d = batch_hidden % HIDDEN_DIM

    # Offsets for the (i, j) tile.
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid_tile // num_pid_n
    pid_n = pid_tile % num_pid_n

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for the current (b,d) slice.
    L_base = L_ptr + b * stride_bL + d * stride_dL
    R_base = R_ptr + b * stride_bR + d * stride_dR
    O_base = O_ptr + b * stride_bO + d * stride_dO

    # Accumulator in FP32.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension k.
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        L_ptrs = L_base + offs_i[:, None] * stride_iL + offs_k[None, :] * stride_kL
        L = tl.load(L_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load right tile transposed (K,N) => right[b,j,k,d] accessed as (k,j)
        R_ptrs = R_base + offs_k[:, None] * stride_kR + offs_j[None, :] * stride_iR
        R = tl.load(R_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(L, R)

    # Store the result tile into the output tensor.
    O_ptrs = O_base + offs_i[:, None] * stride_iO + offs_j[None, :] * stride_jO
    tl.store(O_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.
    The routine follows the reference PyTorch implementation but
    replaces the O(N³) contraction with a custom batched GEMM kernel
    that works directly on the original `[B, N, N, H]` layout,
    thus eliminating the costly permute+view copy.
    """
    input_tensor, mask_tensor, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # shape [B, N, N, 1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4) Gating + elementwise multiplication
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Heavy contraction using Triton (keep FP16 inputs, FP32 accumulation)
    # -----------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape   # left_h : [B, N, N, hidden_dim]

    # Output buffer (float32 for numerical stability)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left_h.device)

    # -----------------------------------------------------------------
    # 6) Kernel launch configuration
    # -----------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    num_tiles = num_tiles_m * num_tiles_n      # tiles per (b, d) slice
    batch_hidden = B * hidden_dim               # total (b, d) count

    # Strides for the three tensors (as integers)
    stride_bL, stride_iL, stride_kL, stride_dL = left_h.stride()
    stride_bR, stride_iR, stride_kR, stride_dR = right_h.stride()
    stride_bO, stride_iO, stride_jO, stride_dO = out.stride()

    # Launch the kernel: axis0 = tile, axis1 = batch*hidden
    trimul_outgoing_gemm_kernel[(num_tiles, batch_hidden)](
        left_h,
        right_h,
        out,
        N,
        hidden_dim,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        stride_bL, stride_iL, stride_kL, stride_dL,
        stride_bR, stride_iR, stride_kR, stride_dR,
        stride_bO, stride_iO, stride_jO, stride_dO,
        num_warps=8,
    )

    # -----------------------------------------------------------------
    # 7) Output LayerNorm, gating and final projection back to `dim`
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

"""
TriMul (outgoing) forward pass – Triton‑accelerated version.

The heavy O(N³·H) contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is rewritten as a batched GEMM (left @ rightᵀ) where the hidden
dimension H is treated as a batch axis.  The kernel works directly on the
original `[B, N, N, H]` layout, thus removing the costly
`permute(...).contiguous()` copy that dominates the reference Triton
implementation.  All other operations (LayerNorm, linear projections,
gating, final projection) are kept in PyTorch for simplicity.

The kernel receives the half‑precision left/right tensors and writes the
result in float32 to preserve numerical stability.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_gemm_kernel(
    L_ptr,                     # half *  (B, N, N, H)
    R_ptr,                     # half *  (B, N, N, H)
    O_ptr,                     # float * (B, N, N, H)
    N,                         # seq_len
    HIDDEN_DIM: tl.constexpr,  # hidden dimension (H)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    # strides for L
    stride_bL, stride_iL, stride_kL, stride_dL,
    # strides for R
    stride_bR, stride_iR, stride_kR, stride_dR,
    # strides for O
    stride_bO, stride_iO, stride_jO, stride_dO,
):
    """
    Batched GEMM for outgoing TriMul:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    The kernel is tiled over the (i,j) plane (BLOCK_M×BLOCK_N) and
    iterates over k in steps of BLOCK_K.  Each program processes one
    (i‑tile, j‑tile) for a specific (b,h) slice.
    """
    # --------------------------------------------------------------
    # Identify tile (i,j) and the combined batch/hidden index.
    # --------------------------------------------------------------
    pid_tile   = tl.program_id(0)               # tiles per (b,h)
    batch_hid  = tl.program_id(1)               # b*H + h

    # ----------- decode batch & hidden ids ----------
    b = batch_hid // HIDDEN_DIM
    h = batch_hid %  HIDDEN_DIM

    # ----------- tile indices ----------
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid_tile // num_pid_n
    pid_n = pid_tile %  num_pid_n

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ----------- base pointers for the selected (b,h) ----------
    L_base = L_ptr + b * stride_bL + h * stride_dL
    R_base = R_ptr + b * stride_bR + h * stride_dR
    O_base = O_ptr + b * stride_bO + h * stride_dO

    # --------------------------------------------------------------
    # Accumulator (FP32) for the tile.
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k.
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i,k]  -> shape (BLOCK_M, BLOCK_K)
        L_ptrs = L_base + offs_i[:, None] * stride_iL + offs_k[None, :] * stride_kL
        L = tl.load(L_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[j,k] (transposed) -> shape (BLOCK_K, BLOCK_N)
        # We need right[b, j, k, h]; treat it as (k, j) for the dot.
        R_ptrs = R_base + offs_k[:, None] * stride_kR + offs_j[None, :] * stride_iR
        R = tl.load(R_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(L, R)

    # --------------------------------------------------------------
    # Store the computed tile.
    # --------------------------------------------------------------
    O_ptrs = O_base + offs_i[:, None] * stride_iO + offs_j[None, :] * stride_jO
    tl.store(O_ptrs, acc, mask=mask_i[:, None] & mask_j[:, None])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)
    eps        = 1e-5

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left  = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4) Gating
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Heavy contraction via the custom Triton GEMM kernel
    # -----------------------------------------------------------------
    left_h  = left.to(torch.float16)   # [B, N, N, H]  half
    right_h = right.to(torch.float16)  # [B, N, N, H]  half

    B, N, _, _ = left_h.shape

    # Output buffer (float32 accumulator)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left_h.device)

    # -----------------------------------------------------------------
    # 6) Kernel launch parameters
    # -----------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    tiles_per_bn = tiles_m * tiles_n                      # tiles per (b,h) slice
    batch_hidden = B * hidden_dim                        # total (b,h) combinations

    # Strides for the three tensors (as plain Python ints)
    stride_bL, stride_iL, stride_kL, stride_dL = left_h.stride()
    stride_bR, stride_iR, stride_kR, stride_dR = right_h.stride()
    stride_bO, stride_iO, stride_jO, stride_dO = out.stride()

    # Launch: axis0 – tile id, axis1 – combined batch*hidden id
    trimul_outgoing_gemm_kernel[(tiles_per_bn, batch_hidden)](
        left_h,
        right_h,
        out,
        N,
        hidden_dim,          # HIDDEN_DIM (constexpr for the kernel)
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        stride_bL, stride_iL, stride_kL, stride_dL,
        stride_bR, stride_iR, stride_kR, stride_dR,
        stride_bO, stride_iO, stride_jO, stride_dO,
        num_warps=8,
    )

    # -----------------------------------------------------------------
    # 7) Output processing: LayerNorm, gating and final projection
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
