```python
"""
Optimized Triton kernel for the outgoing triangle multiplicative update (TriMul).

Algorithm:
- Apply LayerNorm and linear projections (PyTorch, highly optimized).
- Apply optional mask and per‑position sigmoid gates.
- Pack the per‑head tensors into a (total_heads, N, N) layout.
- Compute the heavy N³ “einsum” as a batched matrix multiplication C = A @ Bᵀ using a
  custom Triton kernel that works on fp16 inputs and accumulates in fp32, exploiting
  tensor‑core MMA (BLOCK_M/N multiple of 16, BLOCK_K = 64).
- Reshape back, apply the output LayerNorm and output gate, and a final linear projection.
All heavy compute is performed in FP16 with FP32 accumulation to preserve numerical
stability while achieving near‑peak H100 throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def _tri_mul_kernel(
    left_ptr,      # fp16 pointer, shape (head, i, k)
    right_ptr,     # fp16 pointer, shape (head, j, k)
    out_ptr,       # fp32 pointer, shape (head, i, j)
    N,             # i32 scalar: sequence length
    stride_lh, stride_li, stride_lk,   # i64 strides for left tensor
    stride_rh, stride_rj, stride_rk,   # i64 strides for right tensor (note: rj = stride over j)
    stride_oh, stride_oi, stride_oj,   # i64 strides for out tensor
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # head index (batch * hidden_dim)
    pid_h = tl.program_id(0)
    # block indices for output matrix tiles
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute global row/col offsets
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    # Masks for boundaries
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers to the start of the current head
    left_head  = left_ptr  + pid_h * stride_lh
    right_head = right_ptr + pid_h * stride_rh
    out_head   = out_ptr   + pid_h * stride_oh

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_blocks):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -----------------------------------------------------------------
        # Load left tile [M, K] (fp16)
        # -----------------------------------------------------------------
        left_offsets = left_head + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left_mask    = mask_i[:, None] & mask_k[None, :]
        left_tile    = tl.load(left_offsets, mask=left_mask, other=0.0)  # fp16

        # -----------------------------------------------------------------
        # Load right tile transposed to [K, N] (fp16)
        # right stored as (head, j, k); we need rows = k, cols = j
        # -----------------------------------------------------------------
        right_offsets = right_head + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj
        right_mask    = mask_k[:, None] & mask_j[None, :]
        right_tile    = tl.load(right_offsets, mask=right_mask, other=0.0)  # fp16

        # -----------------------------------------------------------------
        # Matrix multiply (M,K) * (K,N) -> (M,N).  Accumulate in fp32.
        # tl.dot automatically promotes fp16 inputs to fp32 accumulation.
        # -----------------------------------------------------------------
        acc += tl.dot(left_tile, right_tile)

    # -----------------------------------------------------------------
    # Write out tile
    # -----------------------------------------------------------------
    out_offsets = out_head + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    out_mask    = mask_i[:, None] & mask_j[None, :]
    tl.store(out_offsets, acc, mask=out_mask)


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, dim]
        - mask       : torch.Tensor of shape [B, N, N] (ignored if config['nomask'] is True)
        - weights    : dict of model parameters
        - config     : dict containing 'dim', 'hidden_dim', and optional 'nomask'

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------
    # 2. Linear projections (float32) -> later converted to fp16
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4. Gate values (sigmoid) – keep in fp32 for stability, then cast
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # fp32

    # --------------------------------------------------------------
    # 5. Apply gates and cast to fp16 (the heavy kernel works in fp16)
    # --------------------------------------------------------------
    left = left.to(torch.float16) * left_gate
    right = right.to(torch.float16) * right_gate

    # --------------------------------------------------------------
    # 6. Prepare tensors for the batched matmul kernel
    #    layout: (total_heads, N, N) where total_heads = B * hidden_dim
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    total_heads = B * hidden_dim

    #   (B, N, N, hidden) -> (B, hidden, N, N) -> (total_heads, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # Output buffer (fp32 accumulator)
    out_flat = torch.empty((total_heads, N, N), dtype=torch.float32, device=device)

    # --------------------------------------------------------------
    # 7. Launch Triton kernel
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64   # 64 is a good trade‑off for H100 tensor cores

    grid = (
        total_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),  # note: stride_rj = stride over j (dim 2 after permute)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # enable multiple warps per block
    )

    # --------------------------------------------------------------
    # 8. Reshape back to (B, N, N, hidden_dim)
    # --------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden]

    # --------------------------------------------------------------
    # 9. Output LayerNorm, gating and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # broadcast over hidden dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
```

"""
Optimized Triton implementation of the *outgoing* Triangle Multiplicative Update
(TriMul) used in AlphaFold‑3 and related protein‑structure models.

The heavy O(N³) einsum
    out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]
is expressed as a batched matrix multiplication C = A @ Bᵀ where A and B have
shape (head, N, N) and “head” = batch * hidden_dim.  The custom Triton kernel
operates on fp16 inputs and accumulates in fp32, fully exploiting H100 tensor‑core
MMA (BLOCK_M/N are multiples of 16, BLOCK_K = 64).  All remaining ops (LayerNorm,
linear projections, sigmoid gates, final projection) are performed with PyTorch
kernels, which are already highly optimized.

The kernel loads tiles of A (M×K) and Bᵀ (K×N), performs a tiled dot product
using `tl.dot`, and stores the result tile (M×N).  Boundary masks handle arbitrary
sequence lengths.  The overall algorithm follows the reference PyTorch code
exactly up to numerical precision (final output in float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16 * left  = (head, i, k)
    right_ptr,         # fp16 * right = (head, j, k)
    out_ptr,           # fp32 * out   = (head, i, j)
    N,                 # i32 sequence length
    # strides for left tensor
    stride_lh, stride_li, stride_lk,
    # strides for right tensor (ri = stride over j, rk = stride over k)
    stride_rh, stride_ri, stride_rk,
    # strides for output tensor
    stride_oh, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(0)          # which head (batch * hidden_dim)
    pid_m = tl.program_id(1)          # block row (i)
    pid_n = tl.program_id(2)          # block col (j)

    # -----------------------------------------------------------------
    # Global offsets for the output tile (i, j)
    # -----------------------------------------------------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # -----------------------------------------------------------------
    # Pointers to the start of the current head slice
    # -----------------------------------------------------------------
    left_head  = left_ptr  + pid_h * stride_lh
    right_head = right_ptr + pid_h * stride_rh
    out_head   = out_ptr   + pid_h * stride_oh

    # -----------------------------------------------------------------
    # Accumulator (fp32)
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over the reduction dimension k in blocks
    # -----------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_blocks):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------- left tile [M, K] -----------------------
        left_ptrs = (
            left_head
            + offs_i[:, None] * stride_li
            + offs_k[None, :] * stride_lk
        )
        left_mask = mask_i[:, None] & mask_k[None, :]
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0)   # fp16

        # ------------------- right tile transposed [K, N] ----------
        # right is stored as (head, j, k); we need rows = k, cols = j
        right_ptrs = (
            right_head
            + offs_k[:, None] * stride_rk
            + offs_j[None, :] * stride_ri
        )
        right_mask = mask_k[:, None] & mask_j[None, :]
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0)  # fp16

        # ------------------- matrix multiplication ------------------
        # tl.dot promotes fp16 inputs to fp32 accumulation automatically
        acc += tl.dot(left_tile, right_tile)

    # -----------------------------------------------------------------
    # Store the computed output tile
    # -----------------------------------------------------------------
    out_ptrs = out_head + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    out_mask = mask_i[:, None] & mask_j[:, None]
    tl.store(out_ptrs, acc, mask=out_mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (ignored if config["nomask"] is True)
        - weights      : dict of model parameters
        - config       : dict containing at least ``dim`` and ``hidden_dim``.
                         Optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the processed TriMul output.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # -----------------------------------------------------------------
    # 2. Linear projections (still float32)
    # -----------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])    # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Compute gating values (sigmoid) – keep them in fp32 then cast
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # fp32

    # -----------------------------------------------------------------
    # 5. Apply gates and cast to fp16 (the heavy kernel works in fp16)
    # -----------------------------------------------------------------
    left = left.to(torch.float16) * left_gate
    right = right.to(torch.float16) * right_gate

    # -----------------------------------------------------------------
    # 6. Rearrange tensors for the Triton kernel:
    #    (B, N, N, hidden) -> (B, hidden, N, N) -> (total_heads, N, N)
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    total_heads = B * hidden_dim

    left_flat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # Output buffer (fp32 accumulation)
    out_flat = torch.empty((total_heads, N, N), dtype=torch.float32, device=device)

    # -----------------------------------------------------------------
    # 7. Triton kernel launch configuration
    # -----------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64          # 64 works well with H100 WMMA (multiple of 16)

    grid = (
        total_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),   # left strides
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),# right strides
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),     # output strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                # multiple warps per block for high occupancy
    )

    # -----------------------------------------------------------------
    # 8. Reshape back to (B, N, N, hidden_dim)
    # -----------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
               .permute(0, 2, 3, 1)
               .contiguous()
    )   # [B, N, N, hidden_dim] (float32)

    # -----------------------------------------------------------------
    # 9. Output LayerNorm, gating and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # broadcast over hidden dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
