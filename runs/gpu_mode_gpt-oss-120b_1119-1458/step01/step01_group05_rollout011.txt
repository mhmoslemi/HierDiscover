def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,  # dims
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_j, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
   block_i = tl.program_id(0)  # row tile index
   block_j = tl.program_id(1)  # col tile index
   batch_hidden = tl.program_id(2)  # combined BH index, decode to b, h.

   b = batch_hidden // H
   h = batch_hidden % H

   // compute offsets based on strides

   // initialize accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   // loop over K tiles
   for k in range(0, tl.cdiv(N, BLOCK_K)):
       // load tile from left: shape (BLOCK_M, BLOCK_K)
       left_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
       left_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
       a_ptrs = left_ptr + b*stride_left_b + h*stride_left_h + left_i*stride_left_i + left_k*stride_left_k

       // same for right: shape (BLOCK_K, BLOCK_N), but note right dims: j dimension for rows and k dimension for columns (since we need right[j,k]).
       right_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
       // For right, we need row index j and column index k. Since we permute dims before multiplication, we get shape (B, H, N, N) where axes are (b,h,j,k). We'll treat that.
       b_ptrs = right_ptr + b*stride_right_b + h*stride_right_h + right_j*stride_right_j + left_k*stride_right_k

       a = tl.load(a_ptrs, mask = a_i < N and a_k < N)
       b = tl.load(b_ptrs,...)

       // compute dot
       acc += tl.dot(a, b) // shape (BLOCK_M, BLOCK_N)

   // store result
   c_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
   c_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
   c_ptrs = out_ptr + b*stride_out_b + h*stride_out_h + c_i*stride_out_i + c_j*stride_out_j
   tl.store(c_ptrs, acc.to(tl.float16), mask=... )

@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_stride_left, hidden_stride_left, i_stride_left, k_stride_left,
    batch_stride_right, hidden_stride_right, j_stride_right, k_stride_right,
    batch_stride_out, hidden_stride_out, i_stride_out, j_stride_out,
    B, H, N,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # decode block indexes
    pid_m = tl.program_id(0)  # row block index
    pid_n = tl.program_id(1)  # col block index
    pid_hb = tl.program_id(2) # combined batch*hidden idx

    # Compute b, h indexes
    b = pid_hb // H
    h = pid_hb % H

    # Compute start offsets for i, j
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for output bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for kb in range(0, tl.cdiv(N, BLOCK_K)):
        # Compute offset for k elements
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)

        # Mask for k
        mask_k = offs_k < N

        # Load left tile (M x K)
        # left: [b, i, k, h]
        # compute addresses:

    left_ptr, right_ptr, out_ptr,
    # left strides
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    # right strides
    stride_right_b, stride_right_h, stride_right_j, stride_right_k,
    # out strides
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    B, H, N,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid_n = tl.program_id(1)
    pid_hb = tl.program_id(2)

    h = pid_hb % H

            + (offs_j[None, :] * stride_out_j)

def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.
    The kernel fuses the costly N³ contraction into a single
    batched half‑precision GEMM implemented in Triton, while the
    surrounding LayerNorm, linear projections and gating are kept in
    PyTorch. This reduces memory traffic (no extra permutes / copies)
    and leverages Tensor‑cores on H100 for high throughput.
    """
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden]
    right_proj = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Apply optional mask and element‑wise gates
    #    (mask * linear * gate).  Fuse three element‑wise products with a
    #    tiny Triton kernel when a mask is present; otherwise use plain
    #    PyTorch multiplication.
    # ------------------------------------------------------------------
    if nomask:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # expand mask to hidden dimension
        mask_exp = mask.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()
        left = fused_mul3(left_proj, mask_exp, left_gate)
        right = fused_mul3(right_proj, mask_exp, right_gate)

    # ------------------------------------------------------------------
    # 5. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    implemented as a batched GEMM in Triton.
    # ------------------------------------------------------------------
    # Cast to half‑precision before the GEMM
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Allocate output buffer (half‑precision)
    out_h = torch.empty_like(left_h)

    # Strides (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_h.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right_h.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_h.stride()

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    trimul_gemm_kernel[grid](
        left_h,
        right_h,
        out_h,
        # left strides
        stride_left_b,
        stride_left_h,
        stride_left_i,
        stride_left_k,
        # right strides
        stride_right_b,
        stride_right_h,
        stride_right_i,
        stride_right_k,
        # out strides
        stride_out_b,
        stride_out_h,
        stride_out_i,
        stride_out_j,
        B,
        hidden_dim,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Back to FP32 for the remaining layers
    out = out_h.to(inp.dtype)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 7. Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate
    out_final = F.linear(out_norm, weights["to_out.weight"])

    return out_final

"""
TriMul (outgoing) forward pass – Triton‑accelerated.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B,N,N,dim]).
2. Linear projections to the hidden dimension (`left_proj`, `right_proj`).
3. Compute gating vectors (`left_gate`, `right_gate`, `out_gate`) with a sigmoid.
4. Apply optional mask and the three‑way element‑wise product
   `x = proj * mask * gate`.  The three‑way product is fused in a tiny Triton
   kernel when a mask is present.
5. Core N³ contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed by a custom batched GEMM written in Triton (half‑precision
   compute, float32 accumulation).  This avoids the costly permute /
   view steps of the reference implementation.
6. Layer‑norm over the hidden dimension, apply the output gate, and a final
   linear projection back to `dim`.

All heavy compute (steps 4 & 5) are fused in Triton; the remaining
operations stay in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3‑way elementwise product (a * b * c).  Used only when a mask is present.
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,  # pointers
    size,                           # total number of elements
    BLOCK_SIZE: tl.constexpr,       # compile‑time constant
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)

    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """Fused element‑wise product of three tensors using Triton."""
    assert a.shape == b.shape == c.shape
    a = a.contiguous()
    b = b.contiguous()
    c = c.contiguous()
    out = torch.empty_like(a)

    total = a.numel()
    BLOCK = 1024
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_kernel[grid](
        a, b, c, out, total, BLOCK_SIZE=BLOCK
    )
    return out


# ----------------------------------------------------------------------
# Batched GEMM for the core contraction.
# out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,                     # pointers
    # strides for the left tensor (B,N_i,N_k,H)
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    # strides for the right tensor (B,N_j,N_k,H)
    stride_right_b, stride_right_h, stride_right_j, stride_right_k,
    # strides for the output tensor (B,N_i,N_j,H)
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    B, H, N,                                          # runtime sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row (i)
    pid_n = tl.program_id(1)          # tile col (j)
    pid_hb = tl.program_id(2)         # combined batch * hidden index

    # decode batch and hidden indices
    b = pid_hb // H
    h = pid_hb % H

    # offsets for the output tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k_tiles):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load a tile from left: shape (BLOCK_M, BLOCK_K)
        # left layout: [b, i, k, h]
        # ------------------------------------------------------------------
        a_ptrs = left_ptr \
                 + b * stride_left_b \
                 + h * stride_left_h \
                 + (offs_i[:, None] * stride_left_i) \
                 + (offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)                     # FP16 load, cast to FP32 inside dot

        # ------------------------------------------------------------------
        # Load a tile from right: shape (BLOCK_K, BLOCK_N)
        # right layout: [b, j, k, h] – we need (k, j) order for the GEMM.
        # ------------------------------------------------------------------
        b_ptrs = right_ptr \
                 + b * stride_right_b \
                 + h * stride_right_h \
                 + (offs_k[:, None] * stride_right_k) \
                 + (offs_j[None, :] * stride_right_j)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # ------------------------------------------------------------------
        # Matrix multiply (M×K)·(K×N) → (M×N)
        # tl.dot automatically promotes the half inputs to FP32 accumulation.
        # ------------------------------------------------------------------
        acc += tl.dot(a, b)

    # ----------------------------------------------------------------------
    # Store the result (cast back to FP16)
    # ----------------------------------------------------------------------
    c_ptrs = out_ptr \
             + b * stride_out_b \
             + h * stride_out_h \
             + (offs_i[:, None] * stride_out_i) \
             + (offs_j[None, :] * stride_out_j)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

# Public entry point
# ----------------------------------------------------------------------

    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]   (float32, CUDA)
        - mask_tensor  : [B, N, N]       (float32/bool, CUDA)
        - weights_dict : model weights (see skeleton)
        - config_dict  : contains at least "dim", "hidden_dim",
                         optional "eps" and "nomask".

    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right_proj = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gating tensors (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Apply mask and gates
    # ------------------------------------------------------------------
    if nomask:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # mask is broadcast over the hidden dimension
        mask_f = mask.to(left_proj.dtype)                     # float32
        mask_exp = mask_f.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()
        left = fused_mul3(left_proj, mask_exp, left_gate)
        right = fused_mul3(right_proj, mask_exp, right_gate)

    # ------------------------------------------------------------------
    # 5. Core contraction (batched GEMM) – Triton kernel
    # ------------------------------------------------------------------
    # Cast to FP16 to exploit Tensor‑cores
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Allocate output buffer (also FP16)
    out_h = torch.empty_like(left_h)

    # Strides (in elements) for the three tensors
    sL_b, sL_i, sL_k, sL_h = left_h.stride()      # (B,N_i,N_k,H)
    sR_b, sR_i, sR_k, sR_h = right_h.stride()     # (B,N_j,N_k,H)
    sO_b, sO_i, sO_j, sO_h = out_h.stride()       # (B,N_i,N_j,H)

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # rows (i)
        (N + BLOCK_N - 1) // BLOCK_N,   # cols (j)
        B * hidden_dim,                 # combined batch * hidden index
    )

    trimul_gemm_kernel[grid](
        left_h,
        right_h,
        out_h,
        # left strides
        sL_b,
        sL_h,
        sL_i,
        sL_k,
        # right strides
        sR_b,
        sR_h,
        sR_i,
        sR_k,
        # output strides
        sO_b,
        sO_h,
        sO_i,
        sO_j,
        B,
        hidden_dim,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to FP32 for the remaining (norm + linear) layers
    out = out_h.to(inp.dtype)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 7. Output gate & final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate
    out_final = F.linear(out_norm, weights["to_out.weight"])

    return out_final
```
