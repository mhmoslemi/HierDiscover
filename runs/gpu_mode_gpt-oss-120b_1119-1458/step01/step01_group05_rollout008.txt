@triton.jit
def batch_matmul_kernel(A_ptr, B_ptr, C_ptr,
                        M, N, K,
                        stride_am, stride_ak, stride_ab,   # stride for batch, row (M), col (K) for A
                        stride_bm, stride_bn, stride_bk, stride_bb,   # maybe b batch stride, row (N), col (K)
                        stride_cm, stride_cn, stride_cb,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offset for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointers to start of each block
    a_ptrs = A_ptr + batch * stride_ab + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = B_ptr + batch * stride_bb + offs_n[None, :] * stride_bn + tl.arange(0, BLOCK_K)[:, None] * stride_bk

    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=offs_m < M & (k * BLOCK_K + tl.arange(0,BLOCK_K)) < K, other=0.0)
        b = tl.load(b_ptrs, mask=offs_n < N & (k * BLOCK_K + tl.arange(0, BLOCK_K)) < K, other=0.0)
        # Dot product
        acc += tl.dot(a, b)
        # stride increments
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result
    c_ptrs = C_ptr + batch * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=... ) # but we accumulate float32 for numeric stability.

@triton.jit
def batch_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bn, stride_Bk,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # 3D grid
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start offsets
    a_batch_ptr = A + batch_id * stride_Ab
    b_batch_ptr = B + batch_id * stride_Bb
    c_batch_ptr = C + batch_id * stride_Cb

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute K offsets
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        b_ptrs = b_batch_ptr + offs_n[None, :] * stride_Bn + offs_k[:, None] * stride_Bk

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_hidden = tl.program_id(0)  # total B*H
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # derive original batch and hidden index
    b = batch_hidden // H
    h = batch_hidden % H

    # calculate offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers for left: left[b, i, k, h]
        left_ptrs = left_ptr + b * stride_left_b + h * stride_left_h \
                    + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        # pointers for right: right[b, j, k, h] -> need transpose (K,N) order: right[b, j, k, h] where we need right as (j,k) but dot expects (k,N). we can load as (k,j). We'll load with shape (BLOCK_K, BLOCK_N) where rows = k, cols=j.
        right_ptrs = right_ptr + b * stride_right_b + h * stride_right_h \
                    + offs_j[None, :] * stride_right_j + offs_k[:, None] * stride_right_k

        # Load
        a = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b_mat = tl.load(right_ptrs, mask=mask_j[None, :] & mask_k[:, None], other=0.0)

        # Compute dot
        acc += tl.dot(a, b_mat)
        # increment pointer in loops? Already using index based on offs_k, not need to increment.

    # store
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h \
              + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul – “outgoing” version (AlphaFold‑3)
----------------------------------------

The forward pass consists of

1️⃣ Layer‑norm on the 4‑D input `x`  (float‑32).
2️⃣ Linear projections from `dim → hidden_dim` for the left and right
   tensors (no bias).  The results are immediately cast to FP16.
3️⃣ Three‑way element‑wise product   left = left_proj * mask * left_gate
   (and analog for right).  The three multiplies are fused in a small
   Triton kernel `fused_mul3` to avoid extra passes over memory.
4️⃣ Core contraction (the O(N³) part)

      out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]

   This is a batched matrix multiplication for every hidden channel.
   A custom Triton kernel `trimul_outgoing_kernel` computes all
   B·hidden_dim independent GEMMs directly on the original (strided)
   layout, thus eliminating the expensive `permute(...).contiguous()`
   copies that dominate the reference implementation.
   The kernel works on FP16 data and accumulates in FP32, using
   `tl.dot` (Tensor‑core friendly).

5️⃣ Layer‑norm over the hidden dimension, an output gate, and a final
   linear projection back to `dim` (float‑32).

The whole pipeline is fully differentiable in PyTorch – we only provide
the forward pass, as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3‑way element‑wise product (a * b * c) – fused Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,  # pointers
    size,                           # total elements
    BLOCK_SIZE: tl.constexpr,
):
    """out = a * b * c  (all 1‑D contiguous tensors)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)

    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """Fused 3‑way element‑wise product (Triton).  All tensors must have
    identical shape, be CUDA‑resident and contiguous."""
    assert a.shape == b.shape == c.shape
    a = a.contiguous()
    b = b.contiguous()
    c = c.contiguous()
    out = torch.empty_like(a)

    total = a.numel()
    BLOCK = 1024                     # good default on H100
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_kernel[grid](
        a, b, c, out, total,
        BLOCK_SIZE=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# Core contraction – batched “left @ rightᵀ”, one batch per (batch,head)
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides for the left tensor  (B,N,K,H)   →  indices (b,i,k,h)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for the right tensor (B,N,K,H)   →  indices (b,j,k,h)
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    # strides for the output tensor (B,N,N,H) →  indices (b,i,j,h)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the TriMul contraction.
    Each program instance computes a (BLOCK_M × BLOCK_N) tile of the
    output matrix for a particular (batch, head) pair."""
    batch_head = tl.program_id(0)                 # 0 … B*H-1
    pid_m = tl.program_id(1)                      # row block index
    pid_n = tl.program_id(2)                      # col block index

    # recover the original batch index b and hidden dimension h
    b = batch_head // H
    h = batch_head % H

    # tile offsets in the output matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32 (for numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction axis k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load a tile of left  : shape (BLOCK_M, BLOCK_K)
        #   left[b, i, k, h]
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + offs_m[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        a = tl.load(
            left_ptrs,
            mask=(mask_m[:, None] & mask_k[None, :]),
            other=0.0,
        )

        # ------------------------------------------------------------------
        # Load a tile of right (transposed) : shape (BLOCK_K, BLOCK_N)
        #   we need rightᵀ   →   right[b, j, k, h]  as (k, j)
        # ------------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + offs_n[None, :] * stride_right_j
            + offs_k[:, None] * stride_right_k
        )
        b_tile = tl.load(
            right_ptrs,
            mask=(mask_n[None, :] & mask_k[:, None]),
            other=0.0,
        )

        # ------------------------------------------------------------------
        # Tile‑wise dot product (uses Tensor‑cores for FP16)
        # ------------------------------------------------------------------
        acc += tl.dot(a, b_tile)

    # ------------------------------------------------------------------
    # Write the result tile
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_m[:, None] * stride_out_i
        + offs_n[None, :] * stride_out_j
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=(mask_m[:, None] & mask_n[None, :]),
    )


# ----------------------------------------------------------------------
# Entry point – the custom kernel required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : [B, N, N, dim]   (float32, on CUDA)
            - mask_tensor  : [B, N, N]       (float32 / bool, on CUDA)
            - weights_dict : all linear / LN parameters
            - config_dict  : contains at least "dim", "hidden_dim",
                             optional "eps" and "nomask"

    Returns:
        Tensor : [B, N, N, dim]  (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2. Linear projections (to hidden_dim) – keep FP16 for speed
    # ------------------------------------------------------------------
    left_proj = F.linear(x_norm, weights["left_proj.weight"]).to(torch.float16)
    right_proj = F.linear(x_norm, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid) – still FP16 after casting
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"])).to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Apply mask and gates (fused Triton kernel when a mask exists)
    # ------------------------------------------------------------------
    if nomask:
        # no mask → simple element‑wise product
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # mask is broadcast to the hidden dimension
        mask_exp = mask.unsqueeze(-1).to(torch.float16)          # [B,N,N,1]
        mask_exp = mask_exp.expand(-1, -1, -1, hidden_dim)      # [B,N,N,H]
        left = fused_mul3(left_proj, mask_exp, left_gate)
        right = fused_mul3(right_proj, mask_exp, right_gate)

    # ------------------------------------------------------------------
    # 5. Core contraction – custom batched GEMM via Triton
    # ------------------------------------------------------------------
    # Strides for the three tensors (all are contiguous in the last dim)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    # Output tensor (same layout as left/right)
    out_fp16 = torch.empty_like(left)   # [B,N,N,H] FP16

    # Tile sizes – empirically good for H100 FP16 Tensor‑cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (batch*head, num_blocks_M, num_blocks_N)
    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out_fp16,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_j,
        stride_r_k,
        stride_r_h,
        stride_l_b,          # output strides are identical to left/right strides
        stride_l_i,
        stride_l_k,
        stride_l_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to FP32 for the remaining layers
    out = out_fp16.to(inp.dtype)   # [B,N,N,H] float32

    # ------------------------------------------------------------------
    # 6. Output layer‑norm over the hidden dimension
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B,N,N,H] float32

    # ------------------------------------------------------------------
    # 7. Output gate (sigmoid) – same linear as in the reference model
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))  # [B,N,N,H]
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_final = F.linear(out_norm, weights["to_out.weight"])  # [B,N,N,dim] float32

    return out_final
