"""
TriMul “outgoing” forward pass – mixed‑precision Triton for the heavy
contraction.  

Algorithm
---------
1. Layer‑norm the input 4‑D tensor (B,N,N,D).
2. Linear projections to the hidden dimension (no bias) for the left and
   right branches, followed by sigmoid gates.
3. Fuse mask, left‑gate and right‑gate with a tiny Triton kernel
   (`fused_mul3`).
4. Perform the core contraction
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   using a custom batched‑GEMM Triton kernel that computes a matrix
   multiplication for each (batch * hidden) slice in FP16 with FP32
   accumulation.
5. Layer‑norm over the hidden dimension, apply the output gate and a final
   linear projection back to the original channel size.
All kernels run on the GPU; the final tensor is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,    # pointers (1‑D views)
    size,                           # total number of elements
    BLOCK_SIZE: tl.constexpr,       # compile‑time constant
):
    """out = a * b * c  (all tensors are 1‑D contiguous)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)

    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """Fused element‑wise product of three tensors using Triton."""
    assert a.shape == b.shape == c.shape
    a = a.contiguous()
    b = b.contiguous()
    c = c.contiguous()
    out = torch.empty_like(a)

    total = a.numel()
    BLOCK = 1024
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_kernel[grid](
        a, b, c, out, total, BLOCK_SIZE=BLOCK
    )
    return out


@triton.jit
def batched_gemm_kernel(
    A, B, C,                     # pointers to (batch, M, K), (batch, K, N), (batch, M, N)
    M, N, K,                     # matrix dimensions
    stride_Ab, stride_Am, stride_Ak,   # strides for A
    stride_Bb, stride_Bk, stride_Bn,   # strides for B
    stride_Cb, stride_Cm, stride_Cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM kernel: C[b] = A[b] @ B[b]   (FP16 inputs, FP16 output,
    FP32 accumulation).  Grid shape = (ceil(M/BM), ceil(N/BN), batch).
    """
    pid_m = tl.program_id(0)        # block row
    pid_n = tl.program_id(1)        # block column
    pid_b = tl.program_id(2)        # batch id

    # ---- tile offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k_idx in range(num_k_blocks):
        k_offset = k_idx * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # pointers to tiles
        a_ptrs = A + pid_b * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        b_ptrs = B + pid_b * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

        # load tiles (FP16) and cast to FP32 for accumulation
        a = tl.load(a_ptrs, mask=mask_a, other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(a, b)

    # ---- write back ----------------------------------------------------
    c_ptrs = C + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def batched_gemm_fp16(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Wrapper around the Triton GEMM kernel.
    A: (batch, M, K)  FP16
    B: (batch, K, N)  FP16
    Returns C: (batch, M, N) FP16
    """
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    assert A.shape[0] == B.shape[0]
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Block sizes (tuned for H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N), batch)

    batched_gemm_kernel[grid](
        A, B, C,
        M, N, K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    return C


# ----------------------------------------------------------------------
# TriMul forward (outgoing) – entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass for the “outgoing” TriMul operation.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    Tensor of shape [B, N, N, D] (float32) – the processed pairwise features.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣  Layer‑norm over the last dimension
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right_proj = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣  Gating (sigmoid of a linear layer)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))

    # ------------------------------------------------------------------
    # 4️⃣  Apply mask & gates (fused element‑wise three‑way product)
    # ------------------------------------------------------------------
    if nomask:
        # No mask – just two‑way multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # Expand mask to hidden dimension and fuse three multiplies
        mask_exp = mask.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()
        left = fused_mul3(left_proj, mask_exp, left_gate)
        right = fused_mul3(right_proj, mask_exp, right_gate)

    # ------------------------------------------------------------------
    # 5️⃣  Prepare tensors for the batched GEMM (FP16 for Tensor‑cores)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Rearrange to (batch, hidden, N, N) & make contiguous
    left_mat = left_fp16.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_mat = right_fp16.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] (K,J)

    # Collapse the (B, H) dimensions → a single batch dimension for the kernel
    batch_cnt = B * hidden_dim
    left_flat = left_mat.view(batch_cnt, N, N)   # [batch, M=N, K=N]
    right_flat = right_mat.view(batch_cnt, N, N) # [batch, K=N, N]

    # ------------------------------------------------------------------
    # 6️⃣  Core contraction via Triton batched GEMM
    # ------------------------------------------------------------------
    out_flat = batched_gemm_fp16(left_flat, right_flat)   # [batch, N, N] in FP16

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(inp.dtype)   # back to FP32 for the subsequent layers

    # ------------------------------------------------------------------
    # 7️⃣  Output layer‑norm over hidden dimension
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 8️⃣  Output gate
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))   # [B,N,N,hidden_dim]
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 9️⃣  Final linear projection back to the original channel size
    # ------------------------------------------------------------------
    out_final = F.linear(out_norm, weights["to_out.weight"])   # [B, N, N, dim]

    return out_final
