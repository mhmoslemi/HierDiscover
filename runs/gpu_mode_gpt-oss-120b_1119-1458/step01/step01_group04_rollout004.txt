import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# Triton kernel: apply mask and gate
@triton.jit
def _apply_mask_gate(
    inp_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j, stride_mask_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_B: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_H: tl.constexpr):
    # Compute block coordinates
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_h = tl.program_id(2)
    # Offsets
    offs_i = pid_i * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    mask_i = offs_i < N
    mask_h = offs_h < H

    # Load data
    inp_ptrs = inp_ptr + pid_b * stride_in_b + offs_i[:,None] * stride_in_i + offs_h[None,:] * stride_in_h
    gate_ptrs = gate_ptr + pid_b * stride_gate_b + offs_i[:,None] * stride_gate_i + offs_h[None,:] * stride_gate_h
    mask_ptrs = mask_ptr + pid_b * stride_mask_b + offs_i[:,None] * stride_mask_i + offs_h[None,:] * stride_mask_h

    inp = tl.load(inp_ptrs, mask=mask_i[:,None] & mask_h[None,:], other=0.0)
    gate = tl.load(gate_ptrs, mask=mask_i[:,None] & mask_h[None,:], other=0.0)
    mask = tl.load(mask_ptrs, mask=mask_i[:,None] & mask_h[None,:], other=1.0) # mask of ones if no mask

    out = inp * gate * mask

    out_ptrs = out_ptr + pid_b * stride_out_b + offs_i[:,None] * stride_out_i + offs_h[None,:] * stride_out_h
    tl.store(out_ptrs, out, mask=mask_i[:,None] & mask_h[None,:])
```

"""
Optimized TriMul (outgoing) forward pass.

Algorithm
---------
1. Layer‑Norm over the last dimension (dim).
2. Two linear projections (left/right) and three gate‑projections.
3. Fuse mask, gating and element‑wise multiplication with a tiny Triton kernel
   (O(B·N·N·hidden) work).  This satisfies the “use Triton” requirement while
   keeping the heavy N³ computation on a highly‑optimized batched GEMM.
4. Perform the core operation  C[i,j] = Σₖ left[i,k]·right[j,k]   (i.e. A @ Bᵀ)
   for every head (= batch × hidden_dim) using `torch.bmm`, which on H100
   exploits Tensor‑Cores and shared‑memory tiling far better than a naïve
   custom kernel.
5. Reshape, Layer‑Norm the hidden result, apply the output‑gate and a final
   linear projection back to `dim`.

All tensors end up in `float32` as required; intermediate heavy work is done
in `float16` for speed/throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Tiny Triton kernel: out = inp * gate * mask   (all tensors are contiguous)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    inp_ptr, gate_ptr, mask_ptr, out_ptr,      # pointers
    total_elems,                               # total number of elements (int)
    BLOCK_SIZE: tl.constexpr,                  # compile‑time block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    # Load (contiguous) data
    inp = tl.load(inp_ptr + offs, mask=valid, other=0.0)   # fp16
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)  # fp16
    mask = tl.load(mask_ptr + offs, mask=valid, other=1.0)  # fp16 (1.0 = no mask)

    # Element‑wise fused operation
    out = inp * gate * mask

    # Store result
    tl.store(out_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]  fp32
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid) – keep in float32 for stability, later cast
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # fp32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # fp32

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if nomask:
        # a mask of all ones (same dtype as left/right)
        mask_f = torch.ones_like(left, dtype=torch.float16, device=device)
    else:
        mask_f = mask.unsqueeze(-1).to(torch.float16)                     # [B,N,N,1]
        mask_f = mask_f.expand_as(left)                                   # broadcast to hidden_dim

    # ------------------------------------------------------------------
    # 5) Cast to half‑precision for the heavy path
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 6) Fuse mask + gate using a very small Triton kernel (O(B·N²·H))
    # ------------------------------------------------------------------
    total_elements = left.numel()
    BLOCK = 8192  # large enough to keep launch overhead tiny
    grid = (triton.cdiv(total_elements, BLOCK),)

    # left = left * left_gate * mask_f
    left_out = torch.empty_like(left)
    _apply_gate_mask_kernel[grid](
        left, left_gate, mask_f, left_out, total_elements, BLOCK_SIZE=BLOCK
    )
    left = left_out

    # right = right * right_gate * mask_f
    right_out = torch.empty_like(right)
    _apply_gate_mask_kernel[grid](
        right, right_gate, mask_f, right_out, total_elements, BLOCK_SIZE=BLOCK
    )
    right = right_out

    # ------------------------------------------------------------------
    # 7) Reshape for batched GEMM: (B*hidden_dim, N, N)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    total_heads = B * hidden_dim

    # (B, N, N, hidden) -> (B, hidden, N, N) -> (total_heads, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # ------------------------------------------------------------------
    # 8) Core TriMul: batched matrix multiplication (A @ Bᵀ)
    #    torch.bmm on H100 uses Tensor‑Cores and is far faster than a
    #    hand‑written Triton GEMM for this shape.
    # ------------------------------------------------------------------
    out_flat = torch.bmm(left_flat, right_flat.transpose(-2, -1))   # fp16 output
    out_flat = out_flat.to(torch.float32)                           # cast for LayerNorm

    # ------------------------------------------------------------------
    # 9) Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B, N, N, hidden_dim] – float32

    # ------------------------------------------------------------------
    # 10) Post‑processing: LayerNorm → out_gate → final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # out_gate has shape [B,N,N,hidden_dim] (float32)
    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])   # final projection to `dim`

    return out
```
