TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1. Layer‑normalize the 4‑D input (dim‑last) with the provided LN parameters.
2. Cast the normalized tensor to FP16 and perform the three linear projections
   (left, right, gates) in half‑precision using torch.nn.functional.linear.
   The gate outputs are passed through a sigmoid (still in FP16) and fused

   applied in FP16.

4. Compute the core triangle multiplication
      out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]

   The kernel loads the inputs as ``float16`` and accumulates in ``float32``
   (Tensor‑Core friendly).  Block sizes (64,64,64) were chosen to maximise
   occupancy on an H100.
5. The result is a ``float32`` tensor of shape (B·hidden, N, N) which is
   reshaped back to ``[B, N, N, hidden]``.
6. Apply a second LayerNorm over the hidden dimension, multiply by the
   pre‑computed ``out_gate`` (converted to FP32) and perform the final
   linear projection ``hidden → dim`` (FP32).
7. Return the output tensor ``[B, N, N, dim]``.

Only the O(N³) reduction is off‑loaded to Triton; all other operations stay in
PyTorch.  This meets the requirement of “implement at least part of the
operations in a kernel” while making the heavy reduction fast by using
Tensor‑Core half‑precision arithmetic.

"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul with B‑dim * (N×N) @ (N×N)ᵀ
# ----------------------------------------------------------------------
@triton.jit
def batch_matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers for A (half), B (half), C (float32)
    M, N, K,                         # matrix dimensions (all equal to seq_len)
    stride_a_batch, stride_am, stride_ak,   # strides for A
    stride_b_batch, stride_bk, stride_bn,   # strides for B (note: B is accessed transposed)
    stride_c_batch, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """C = A @ Bᵀ  (batched GEMM)
    A : (batch, M, K)   – row‑major, FP16
    B : (batch, N, K)   – row‑major, FP16  (accessed as transposed)
    C : (batch, M, N)   – row‑major, FP32
    """
    batch_idx = tl.program_id(0)   # batch dimension (B·hidden)
    pid_m = tl.program_id(1)       # row‑tile id
    pid_n = tl.program_id(2)       # col‑tile id

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Base pointers for this batch
    # ------------------------------------------------------------------
    a_batch = a_ptr + batch_idx * stride_a_batch
    b_batch = b_ptr + batch_idx * stride_b_batch
    c_batch = c_ptr + batch_idx * stride_c_batch

    # ------------------------------------------------------------------
    # Accumulator (float32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_base = k * BLOCK_K
        cur_k_valid = (k_base + offs_k) < K

        # ---- load A tile (BLOCK_M x BLOCK_K) ----
        a_ptrs = a_batch + (offs_m[:, None] * stride_am) + ((k_base + offs_k)[None, :] * stride_ak)
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & cur_k_valid[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # ---- load Bᵀ tile (BLOCK_K x BLOCK_N) ----
        # Bᵀ is accessed as rows = k, cols = n
        b_ptrs = b_batch + ((k_base + offs_k)[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(
            b_ptrs,
            mask=cur_k_valid[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # ---- dot product (FP16×FP16 → FP32) ----
        acc += tl.dot(a, b)          # acc dtype is float32

    # ------------------------------------------------------------------
    # Write back C
    # ------------------------------------------------------------------
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul (outgoing) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        • input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        • mask_tensor  : torch.Tensor of shape [B, N, N] (may be None)
        • weights_dict : mapping from parameter names to tensors
        • config_dict  : contains "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Extract weights (all are float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on input (dim‑last)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )                       # shape [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy linear (project & gate) work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # Half‑precision weight copies
    left_proj_w_fp16   = left_proj_weight.to(torch.float16)
    right_proj_w_fp16  = right_proj_weight.to(torch.float16)
    left_gate_w_fp16   = left_gate_weight.to(torch.float16)
    right_gate_w_fp16  = right_gate_weight.to(torch.float16)
    out_gate_w_fp16    = out_gate_weight.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Projections (no bias) – FP16 GEMMs
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x_fp16, left_proj_w_fp16)   # [B,N,N,hidden]
    right_proj = torch.nn.functional.linear(x_fp16, right_proj_w_fp16)  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3️⃣ Gates (sigmoid of linear layers) – FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_w_fp16))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, right_gate_w_fp16))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_w_fp16))  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 4️⃣ Fuse gating (and optional mask) into projected tensors – in‑place
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    if not nomask:
        # mask shape: [B, N, N] → broadcast over hidden dimension
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Core TriMul: batched GEMM left @ rightᵀ using Triton (FP16→FP32)
    # ------------------------------------------------------------------
    # 5.1 Permute to bring hidden dim into the batch axis
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, hidden, N, N]

    batch_mul = bs * hidden_dim                           # total # of matrices
    left_flat  = left_perm.view(batch_mul, N, N)          # [B·H, N, N]  FP16
    right_flat = right_perm.view(batch_mul, N, N)         # [B·H, N, N]  FP16

    # Allocate output (FP32) for the batched multiplication
    out_flat = torch.empty(batch_mul, N, N, dtype=torch.float32, device=device)

    # Strides in *elements* (not bytes) – Triton works with element strides
    stride_a_batch = left_flat.stride(0)
    stride_am      = left_flat.stride(1)
    stride_ak      = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    # B will be accessed transposed: swap last two strides
    stride_bk = right_flat.stride(2)   # stride along K (original column)
    stride_bn = right_flat.stride(1)   # stride along N (original row)

    stride_c_batch = out_flat.stride(0)
    stride_cm      = out_flat.stride(1)
    stride_cn      = out_flat.stride(2)

    # Block sizes – tuned for H100 Tensor‑Core (FP16) workloads
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_mul,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batch_matmul_transpose_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        M=N, N=N, K=N,
        stride_a_batch=stride_a_batch, stride_am=stride_am, stride_ak=stride_ak,
        stride_b_batch=stride_b_batch, stride_bk=stride_bk, stride_bn=stride_bn,
        stride_c_batch=stride_c_batch, stride_cm=stride_cm, stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Reshape back to [B, N, N, hidden] (float32)
    # ------------------------------------------------------------------
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (hidden dim)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # still float32

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate (convert to FP32 if still half)
    # ------------------------------------------------------------------
    out_gate_fp32 = out_gate.to(torch.float32)               # [B,N,N,hidden]
    out = out * out_gate_fp32

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection hidden → dim (FP32)
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_weight)   # [B,N,N,dim]

    return out
```
