    M, N, K,

    stride_cm, stride_cn,

):

    ...

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

"""
TriMul (outgoing) – fast Triton implementation.
------------------------------------------------
The heavy part of the TriMul forward pass is the tensor contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]      (h ∈ hidden)

which is exactly a batched matrix multiplication C = A @ Bᵀ for every
(b, h) pair.  The kernel below tiles the M, N and K dimensions,
loads A and the transposed B in fp16, accumulates in fp32 (Tensor‑Core
friendly) and finally stores the result in fp16.  The surrounding
LayerNorms, linear projections and gates are executed in PyTorch,
but all intermediate tensors are kept in half‑precision to cut memory
traffic in half.  The final output is cast back to float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: one “head’’ = one (batch * hidden) slice.
# Computes C = A @ Bᵀ  where A,B have shape (head, N, N) and dtype=fp16.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16 * (head, i, k)
    right_ptr,         # fp16 * (head, j, k)   (k is the inner dimension)
    out_ptr,           # fp16 * (head, i, j)
    N,                 # i32  sequence length
    stride_lh, stride_li, stride_lk,   # i64 strides for left   (head,i,k)
    stride_rh, stride_rj, stride_rk,   # i64 strides for right  (head,j,k)
    stride_oh, stride_oi, stride_oj,   # i64 strides for out    (head,i,j)
    BLOCK_M: tl.constexpr,  # tile size in i‑dim
    BLOCK_N: tl.constexpr,  # tile size in j‑dim
    BLOCK_K: tl.constexpr,  # tile size in k‑dim
):
    pid_h = tl.program_id(0)   # which head (batch * hidden)
    pid_m = tl.program_id(1)   # block row   (i)
    pid_n = tl.program_id(2)   # block col   (j)

    # ------------------------------------------------------------------
    # Starting indices of the current tile.
    # ------------------------------------------------------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Pointers to the beginning of the current head.
    # ------------------------------------------------------------------
    left_head  = left_ptr  + pid_h * stride_lh
    right_head = right_ptr + pid_h * stride_rh
    out_head   = out_ptr   + pid_h * stride_oh

    # ------------------------------------------------------------------
    # Accumulator (fp32) – start from zero.
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Number of full K‑tiles (no out‑of‑bounds masking required).
    # ------------------------------------------------------------------
    num_full_k = N // BLOCK_K
    for kk in range(num_full_k):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)          # always < N

        # left tile  [M, K]
        left_ptrs = (left_head
                     + offs_i[:, None] * stride_li
                     + offs_k[None, :] * stride_lk)
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None], other=0.0).to(tl.float32)

        # right tile transposed to [K, N]
        # right is stored as (head, j, k); we want rows=k, cols=j
        right_ptrs = (right_head
                      + offs_k[:, None] * stride_rk
                      + offs_j[None, :] * stride_rj)
        right_tile = tl.load(right_ptrs, mask=mask_j[None, :], other=0.0).to(tl.float32)

        # matrix multiply‑accumulate
        acc += tl.dot(left_tile, right_tile)   # (M,K) @ (K,N) -> (M,N)

    # ------------------------------------------------------------------
    # Handle the remainder K‑block (if any) with masking.
    # ------------------------------------------------------------------
    rem = N % BLOCK_K
    if rem > 0:
        k0 = num_full_k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        left_ptrs = (left_head
                     + offs_i[:, None] * stride_li
                     + offs_k[None, :] * stride_lk)
        left_mask = mask_i[:, None] & mask_k[None, :]
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        right_ptrs = (right_head
                      + offs_k[:, None] * stride_rk
                      + offs_j[None, :] * stride_rj)
        right_mask = mask_k[:, None] & mask_j[None, :]
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        acc += tl.dot(left_tile, right_tile)

    # ------------------------------------------------------------------
    # Convert accumulator to fp16 and store.
    # ------------------------------------------------------------------
    out_fp16 = acc.to(tl.float16)
    out_ptrs = (out_head
                + offs_i[:, None] * stride_oi
                + offs_j[None, :] * stride_oj)
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, out_fp16, mask=out_mask)


# ----------------------------------------------------------------------
# Entry point expected by the test harness.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs.
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # Cast model weights to the precision we will work with (fp16).
    # ------------------------------------------------------------------
    def _to_fp16(t: torch.Tensor) -> torch.Tensor:
        return t.to(device=device, dtype=torch.float16)

    norm_w = _to_fp16(weights["norm.weight"])
    norm_b = _to_fp16(weights["norm.bias"])

    left_proj_w  = _to_fp16(weights["left_proj.weight"])
    right_proj_w = _to_fp16(weights["right_proj.weight"])

    left_gate_w  = _to_fp16(weights["left_gate.weight"])
    right_gate_w = _to_fp16(weights["right_gate.weight"])
    out_gate_w   = _to_fp16(weights["out_gate.weight"])

    to_out_norm_w = _to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = _to_fp16(weights["to_out_norm.bias"])

    to_out_w = _to_fp16(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim) – keep result in fp32
    # because the input is still float32 at this point.
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )  # shape [B, N, N, dim], dtype float32

    # ------------------------------------------------------------------
    # Switch to fp16 for the heavy linear/gate work.
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # Linear projections (no bias).
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # [B, N, N, hidden]
    right = F.linear(x, right_proj_w)  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # Apply optional mask (broadcast on hidden dimension).
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating values (sigmoid of linear transforms).
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)).to(torch.float16)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w)).to(torch.float16)  # same shape as left/right

    # ------------------------------------------------------------------
    # Apply gates.
    # ------------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the Triton kernel.
    #   (total_heads, N, N) where total_heads = B * hidden_dim.
    #   Layout: (head, i, k) and (head, j, k) – both contiguous in k.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    total_heads = B * hidden_dim

    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    out_flat = torch.empty(
        (total_heads, N, N), dtype=torch.float16, device=device
    )

    # ------------------------------------------------------------------
    # Triton launch configuration.
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64  # 64 works well on Hopper with Tensor‑Cores

    grid = (
        total_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to (B, N, N, hidden_dim) – still fp16.
    # ------------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)          # [B, N, N, hidden]
        .contiguous()
    )

    # ------------------------------------------------------------------
    # Post‑processing on the hidden dimension.
    #   1) LayerNorm (still in fp16, accumulates in fp32 internally)
    #   2) Multiplication by the out‑gate
    #   3) Final linear projection to `dim`
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate

    out = F.linear(out, to_out_w)          # now shape [B, N, N, dim] in fp16

    # ------------------------------------------------------------------
    # Cast back to float32 as required by the API.
    # ------------------------------------------------------------------
    return out.to(torch.float32)
