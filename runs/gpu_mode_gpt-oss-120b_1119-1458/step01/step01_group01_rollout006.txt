"""
TriMul (outgoing) – fast Triton‑accelerated implementation.

The heavy N³ reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix‑multiply (L @ Rᵀ) for each hidden channel d.
We reshape the tensors to (B·H, N, N) and launch a 3‑D Triton kernel that
computes many independent GEMMs in parallel:
    * each program‑id along axis‑2 selects one (batch, hidden) pair,
    * axes‑0/1 tile the M×N output,
    * the K‑loop accumulates with fp32 precision while reading fp16.
All other operations (LayerNorms, linear projections, gates, mask) are
performed with PyTorch in fp16 (except the final LayerNorms which stay
fp32).  The final result is returned as float32, matching the reference
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched half‑precision GEMM (L @ Rᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr,                 # left  : (P, M, K)  fp16
    B_ptr,                 # right : (P, K, N)  fp16  (already transposed)
    C_ptr,                 # out   : (P, M, N) fp16
    stride_ap, stride_am, stride_ak,   # strides of A
    stride_bp, stride_bk, stride_bn,   # strides of B
    stride_cp, stride_cm, stride_cn,   # strides of C
    M, N, K,               # matrix sizes (all = seq_len)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)               # block row
    pid_n = tl.program_id(1)               # block col
    pid_p = tl.program_id(2)               # (batch, hidden) index

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this (batch, hidden) GEMM
    a_base = A_ptr + pid_p * stride_ap
    b_base = B_ptr + pid_p * stride_bp
    c_base = C_ptr + pid_p * stride_cp

    # Accumulator in fp32 (more numerically stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # K‑loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        start_k = k * BLOCK_K
        offs_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                     # fp16

        # Load B tile (K x N)
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                     # fp16

        # Multiply‑accumulate (fp16→fp32)
        acc += tl.dot(a, b).to(tl.float32)

    # ------------------------------------------------------------------
    # Store result
    # ------------------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that invokes the Triton kernel
# ----------------------------------------------------------------------
def _triton_matmul(left: torch.Tensor,
                   right: torch.Tensor,
                   seq_len: int,
                   hidden_dim: int,
                   *,
                   BLOCK_M: int = 64,
                   BLOCK_N: int = 64,
                   BLOCK_K: int = 64):
    """
    Perform   out = left @ rightᵀ   for every hidden channel.
    Expected shapes (fp16):
        left  : [B, N, N, H]
        right : [B, N, N, H]   (will be transposed on the last two axes)
    Returns:
        out   : [B, N, N, H]   (fp16)
    """
    B = left.shape[0]

    # --------------------------------------------------------------
    # Re‑order to (B, H, N, N) – we keep a contiguous copy because the
    # GEMM kernel expects the hidden dimension to be the leading one.
    # --------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()          # (B,H,N,N)
    right = right.permute(0, 3, 2, 1).contiguous()        # (B,H,K,J) -> (B,H,N,N) with (k,j)

    P = B * hidden_dim                                      # total independent GEMMs

    # Flatten the leading dimensions so that each GEMM sees a (N,N) matrix
    left_flat  = left.view(P, seq_len, seq_len)
    right_flat = right.view(P, seq_len, seq_len)

    out = torch.empty_like(left_flat)                       # fp16 output

    # ------------------------------------------------------------------
    # Strides (contiguous tensors → straightforward strides)
    # ------------------------------------------------------------------
    stride_ap = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bp = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_cp = out.stride(0)
    stride_cm = out.stride(1)
    stride_cn = out.stride(2)

    # Grid: (blocks over M, blocks over N, P)
    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,
        (seq_len + BLOCK_N - 1) // BLOCK_N,
        P,
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out,
        stride_ap,
        stride_am,
        stride_ak,
        stride_bp,
        stride_bk,
        stride_bn,
        stride_cp,
        stride_cm,
        stride_cn,
        seq_len,
        seq_len,
        seq_len,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # Reshape back to the original layout (B, N, N, H)
    out = out.view(B, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim]   (float32)
        - mask         : torch.Tensor[bs, N, N]        (bool/float32) or None
        - weights      : dict of torch.Tensors (all float32)
        - config       : dict with keys:
            * "dim"          – input / output channel dimension
            * "hidden_dim"   – hidden dimension inside TriMul
            * "nomask"       – bool, if True the mask is ignored

    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    use_mask = not config.get("nomask", True) and mask is not None

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (float32, kept for numerical stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)                       # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy part
    # ------------------------------------------------------------------
    x_fp16 = x.half()

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (no bias) – keep everything in fp16
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].half()   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].half()

    left  = F.linear(x_fp16, left_proj_w)               # [B, N, N, hidden]  fp16
    right = F.linear(x_fp16, right_proj_w)              # [B, N, N, hidden]  fp16

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (multiply after projection, before gating)
    # ------------------------------------------------------------------
    if use_mask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid(linear(x))) – compute in fp16 for speed,
    #    but keep the output gate in fp32 because it multiplies later
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w   = weights["out_gate.weight"].half()

    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # fp16
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w)) # fp16
    # out_gate stays fp32 (used after to_out_norm)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # fp32

    # Apply element‑wise gates
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ N³ reduction using the Triton kernel (L @ Rᵀ)
    # ------------------------------------------------------------------
    seq_len = input_tensor.shape[1]          # N
    out_fp16 = _triton_matmul(left, right,
                              seq_len=seq_len,
                              hidden_dim=hidden_dim,
                              BLOCK_M=64, BLOCK_N=64, BLOCK_K=64)

    # Back to fp32 for the remaining parts
    out = out_fp16.to(torch.float32)         # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 7️⃣ LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=eps)

    # ------------------------------------------------------------------
    # 8️⃣ Apply the output gate (broadcast over hidden dim)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim` (float32)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]      # (dim, hidden)
    out = F.linear(out, to_out_w)            # [B, N, N, dim]  float32

    return out
