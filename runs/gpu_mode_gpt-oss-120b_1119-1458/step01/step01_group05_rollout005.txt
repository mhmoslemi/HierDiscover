def custom_kernel(data):
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)
    
    # Cast to fp16 for speed
    inp_fp16 = inp.half()
    
    # layernorm on input (dim)
    # flatten
    B, N, _, _ = inp.shape
    M = B * N * N
    x_flat = inp_fp16.view(M, dim).contiguous()
    weight_norm = weights["norm.weight"].half().contiguous()
    bias_norm = weights["norm.bias"].half().contiguous()
    x_norm_flat = layernorm_triton(x_flat, weight_norm, bias_norm, eps)
    x_norm = x_norm_flat.view(B, N, N, dim)
    
    # Linear projections (no bias)
    # weights half
    left_proj_weight = weights["left_proj.weight"].half()
    right_proj_weight = weights["right_proj.weight"].half()
    # Using F.linear: input: [..., dim] weight: [hidden_dim, dim]
    left = F.linear(x_norm, left_proj_weight)   # shape B,N,N,hidden_dim
    right = F.linear(x_norm, right_proj_weight)
    
    # Gating tensors (sigmoid) on x_norm
    left_gate_weight = weights["left_gate.weight"].half()
    right_gate_weight = weights["right_gate.weight"].half()
    out_gate_weight = weights["out_gate.weight"].half()
    
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))
    
    # Apply mask and gates
    if nomask:
        left = left * left_gate
        right = right * right_gate
    else:

        mask_exp = mask.unsqueeze(-1).to(torch.float16).expand(-1, -1, -1, hidden_dim).contiguous()
        left = fused_mul3(left, mask_exp, left_gate)

Algorithm
---------
1.  Cast the input to FP16 and apply a custom LayerNorm (Triton) over the
    last dimension (dim).  This avoids the PyTorch reduction and saves a
    memory pass.
2.  Linear projections to the hidden dimension (no bias).  We keep everything
    in FP16 to exploit Tensor‑cores.
3.  Compute per‑position gates (sigmoid of a linear projection) also in FP16.
4.  Fuse the optional mask, left‑gate and right‑gate with a single Triton
    element‑wise kernel (out = a * b * c).  When `nomask=True` we fall back
    to a simple PyTorch multiplication.
5.  Core contraction:
        out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    This is a batched matrix multiplication for each hidden channel.
    We reorder the tensors to (B * hidden, N, N) and use `torch.bmm`
    on FP16 tensors – a highly‑optimised cuBLAS/Tensor‑core kernel.
6.  Apply a second custom LayerNorm (over `hidden_dim`), multiply by the
    output gate and a final linear projection back to `dim`.
7.  Cast the result back to FP32 for downstream consumption.

All heavy element‑wise work (LayerNorm and mask×gate fusion) is performed
with Triton kernels, while the cubic contraction is handled by cuBLAS
via `torch.bmm`.  Mixed‑precision (FP16) is used throughout to achieve
the target ~1 ms latency on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,      # pointers
    size,                               # total number of elements
    BLOCK_SIZE: tl.constexpr,           # compile‑time constant
):
    """Fused element‑wise product out = a * b * c (1‑D contiguous tensors)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)

    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """Python wrapper that launches `fused_mul3_kernel`."""
    assert a.shape == b.shape == c.shape
    a = a.contiguous()
    b = b.contiguous()
    c = c.contiguous()
    out = torch.empty_like(a)

    total = a.numel()
    BLOCK = 1024  # good default for 1‑D loads on H100
    grid = (triton.cdiv(total, BLOCK),)

    fused_mul3_kernel[grid](
        a, b, c, out, total,
        BLOCK_SIZE=BLOCK,
    )
    return out


@triton.jit
def layernorm_kernel(
    x_ptr, out_ptr, weight_ptr, bias_ptr,
    M, D, eps,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm.
    Input `x_ptr` points to a flattened (M, D) tensor (contiguous).
    Output written to `out_ptr` with the same layout.
    """
    row = tl.program_id(0)                     # 0 ≤ row < M
    col = tl.arange(0, BLOCK_D)                # 0 ≤ col < D  (thread index)
    mask = col < D

    # linear index into the flattened tensor
    idx = row * D + col
    x = tl.load(x_ptr + idx, mask=mask, other=0.0)

    # promote to FP32 for numerics
    x_fp32 = x.to(tl.float32)

    # ---- mean ---------------------------------------------------------
    sum_val = tl.sum(x_fp32, axis=0)           # scalar reduction across BLOCK_D threads
    mean = sum_val / D

    # ---- variance -----------------------------------------------------
    x_centered = x_fp32 - mean
    var_sum = tl.sum(x_centered * x_centered, axis=0)
    var = var_sum / D

    # ---- normalize ----------------------------------------------------
    inv_std = tl.rsqrt(var + eps)               # scalar, FP32
    normed_fp32 = x_centered * inv_std          # still FP32
    normed = normed_fp32.to(x.dtype)           # back to original dtype (FP16/FP32)

    # ---- scale + bias -------------------------------------------------
    w = tl.load(weight_ptr + col, mask=mask, other=0.0)
    b = tl.load(bias_ptr + col, mask=mask, other=0.0)

    out = normed * w + b
    tl.store(out_ptr + idx, out, mask=mask)


def layernorm_triton(x: torch.Tensor, weight: torch.Tensor,
                     bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    """
    Wrapper for `layernorm_kernel`.
    Arguments:
        x      – Tensor of shape (M, D), contiguous.
        weight – 1‑D tensor of length D.
        bias   – 1‑D tensor of length D.
    Returns a tensor of the same shape and dtype as `x`.
    """
    assert x.is_contiguous()
    M, D = x.shape
    out = torch.empty_like(x)

    # Choose a block size that covers the dimension D (max 384 in our configs)
    if D <= 32:
        BLOCK = 32
    elif D <= 64:
        BLOCK = 64
    elif D <= 128:
        BLOCK = 128
    else:
        BLOCK = 256  # safe upper bound; will be trimmed by mask

    grid = (M,)  # one program instance per row
    layernorm_kernel[grid](
        x, out, weight, bias,
        M, D, eps,
        BLOCK_D=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Input tuple `data`:
        - input_tensor : [B, N, N, dim]  (float32, on CUDA)
        - mask_tensor  : [B, N, N]       (float32 or bool, on CUDA)
        - weights_dict : model weights (see skeleton)
        - config_dict  : {"dim", "hidden_dim", "eps", "nomask"}

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Cast everything to FP16 for speed (except final output)
    # ------------------------------------------------------------------
    inp_fp16 = inp.half()  # [B,N,N,dim] in FP16

    # ------------------------------------------------------------------
    # Layer‑norm on the input (dim)
    # ------------------------------------------------------------------
    x_flat = inp_fp16.view(B * N * N, dim).contiguous()
    norm_w = weights["norm.weight"].half().contiguous()
    norm_b = weights["norm.bias"].half().contiguous()
    x_norm_flat = layernorm_triton(x_flat, norm_w, norm_b, eps)
    x_norm = x_norm_flat.view(B, N, N, dim)  # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # Linear projections (no bias) to hidden dimension
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left = F.linear(x_norm, left_proj_w)   # [B,N,N,hidden_dim] FP16
    right = F.linear(x_norm, right_proj_w)

    # ------------------------------------------------------------------
    # Gating tensors (sigmoid of linear layers) – also FP16
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))

    # ------------------------------------------------------------------
    # Apply mask + gates (fused Triton kernel when a mask is present)
    # ------------------------------------------------------------------
    if nomask:
        left = left * left_gate
        right = right * right_gate
    else:
        # mask: [B,N,N] -> [B,N,N,1] -> expand to hidden_dim
        mask_exp = mask.unsqueeze(-1).to(torch.float16).expand(-1, -1, -1, hidden_dim).contiguous()
        left = fused_mul3(left, mask_exp, left_gate)
        right = fused_mul3(right, mask_exp, right_gate)

    # ------------------------------------------------------------------
    # Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Rearrange tensors to (B * hidden, N, N) for batched GEMM.
    # left as (i,k) , right as (k,j)  → torch.bmm does the sum over k.
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N] (i,k)
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # [B, hidden, N, N] (k,j)

    left_flat = left_mat.view(B * hidden_dim, N, N)    # [B*H, N, N]
    right_flat = right_mat.view(B * hidden_dim, N, N)  # [B*H, N, N]

    # Batched matrix multiply in FP16 → exploits Tensor‑cores
    out_flat = torch.bmm(left_flat, right_flat)       # [B*H, N, N]

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out_flat2 = out.view(B * N * N, hidden_dim).contiguous()
    to_out_norm_w = weights["to_out_norm.weight"].half().contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].half().contiguous()
    out_norm_flat = layernorm_triton(out_flat2, to_out_norm_w, to_out_norm_b, eps)
    out_norm = out_norm_flat.view(B, N, N, hidden_dim)  # FP16

    # ------------------------------------------------------------------
    # Apply output gate (already in FP16) and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                     # element‑wise gating
    to_out_w = weights["to_out.weight"].half()
    out_fp16 = F.linear(out_norm, to_out_w)            # [B,N,N,dim] FP16

    # Cast to FP32 as required by the caller
    return out_fp16.float()
