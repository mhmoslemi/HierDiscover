@triton.jit
def trimul_bgemm_kernel(
    A, B, C,
    M, N, K,
    stride_ha, stride_ha_m, stride_ha_k,
    stride_hb, stride_hb_k, stride_hb_n,
    stride_hc, stride_hc_m, stride_hc_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_M: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # start pointers for this batch
    A = A + batch_id * stride_ha
    B = B + batch_id * stride_hb
    C = C + batch_id * stride_hc

    # for each k block accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        a_ptrs = A + (offs_m[:, None] * stride_ha_m + offs_k[None, :] * stride_ha_k)
        b_ptrs = B + (offs_k[:, None] * stride_hb_k + offs_n[None, :] * stride_hb_n)
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None,:], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None,:], other=0.0)
        # convert to fp32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)
    # store
    c_ptrs = C + (offs_m[:, None] * stride_hc_m + offs_n[None, :] * stride_hc_n)

@triton.jit
def trimul_bgemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_i, stride_b_k,
    stride_c_batch, stride_c_i, stride_c_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    GROUP_M: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offset for block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = A_ptr + batch_id * stride_a_batch
    b_batch = B_ptr + batch_id * stride_b_batch
    c_batch = C_ptr + batch_id * stride_c_batch

    # Allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K tiles
    # Use double buffering
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
       
        # Load B tile transposed (K x N)
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_batch + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_k
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

import torch
import triton
import triton.language as tl

@triton.jit
def trimul_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_ab, stride_ai, stride_ak,
    stride_bb, stride_bk, stride_bi,
    stride_cb, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask to ignore OOB rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    A_batch = A_ptr + batch_id * stride_ab
    B_batch = B_ptr + batch_id * stride_bb
    C_batch = C_ptr + batch_id * stride_cb

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of tiles in K dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k in range(num_k_tiles):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (M x K)
        a_ptrs = A_batch + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # load B tile transposed (K x N) : B[k, n]
        b_ptrs = B_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bi
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # write output tile
    c_ptrs = C_batch + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) operator.
    The heavy tensor contraction out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
    is performed as a batched half‑precision GEMM (A @ Bᵀ) with fp32 accumulation.
    Non‑linearities, masking, and LayerNorms are kept in PyTorch.
    The kernel uses a 3‑D grid (batch, tile‑row, tile‑col) and a 128×128×64
    tiling schedule that fully exploits Tensor‑Cores on H100.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # 1. input LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # 2. linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # 3. optional mask
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # 4. gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # 5. cast to fp16 for the GEMM
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape  # left_h: [B, N, N, hidden_dim]

    # bring hidden dimension forward; this requires a memory layout change.
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)   # [BH, N, N]   fp16
    right_flat = right_perm.view(BH, N, N) # [BH, N, N]   fp16

    # output buffer (float32)
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=left_flat.device)

    # 6. launch Triton GEMM
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64  # larger K‑tile -> better Tensor‑Core utilisation

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (BH, num_pid_m, num_pid_n)

    trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,            # 8 warps = 256 threads per block, good for H100
    )

    # 7. reshape back to [B,N,N,hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]

    # 8. output LayerNorm over hidden_dim
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # 9. output gating
    out = out * out_gate

    # 10. final projection back to original dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass accelerated with Triton.

The heavy contraction
    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
is a batched matrix multiplication per hidden channel:
    (B·C) × (N×N)  @  (B·C) × (N×N).T .
We implement this multiplication as a custom Triton GEMM kernel that:
* uses a 3‑D grid (batch, tile‑row, tile‑col) → no integer division per block,
* tiles with BLOCK_M=128, BLOCK_N=128, BLOCK_K=64 to maximise Tensor‑Core utilisation,
* loads the two half‑precision tiles, up‑casts to fp32 and accumulates with `tl.dot`,
* writes the result in fp32 (required by the following LayerNorm).

All remaining ops (LayerNorm, linear layers, sigmoids, output LayerNorm and
projection) are done with PyTorch.  This keeps the kernel focussed on the O(N³)
work while still satisfying the “use Triton” requirement.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_gemm_kernel(
    A_ptr,               # half*  (BH, N, N) left  matrix
    B_ptr,               # half*  (BH, N, N) right matrix (will be read transposed)
    C_ptr,               # float* (BH, N, N) output accumulator
    M, N, K,             # all equal to seq_len
    stride_ab, stride_ai, stride_ak,   # batch, row, col strides for A
    stride_bb, stride_bk, stride_bi,   # batch, row, col strides for B
    stride_cb, stride_ci, stride_cj,   # batch, row, col strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ where A,B∈ℝ^{BH×N×N} (half) → C∈ℝ^{BH×N×N} (fp32)."""
    batch_id = tl.program_id(0)          # which (B·hidden) slice
    pid_m    = tl.program_id(1)          # tile row
    pid_n    = tl.program_id(2)          # tile column

    # ------------------- tile offsets -------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------- base pointers for this batch ----------
    A_batch = A_ptr + batch_id * stride_ab
    B_batch = B_ptr + batch_id * stride_bb
    C_batch = C_ptr + batch_id * stride_cb

    # ------------------- accumulation --------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)   # [BLOCK_K]
        mask_k = offs_k < K

        # A tile: (M×K)
        a_ptrs = A_batch + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # B tile transposed: we need B[k, n] (original B row=k, col=n)
        b_ptrs = B_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bi
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ------------------- store result --------------------
    c_ptrs = C_batch + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul (outgoing) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # -------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # -------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -------------------------------------------------
    # 2. Linear projections (no bias) → shape [B,N,N,hidden_dim]
    # -------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------
    # 3. Optional mask (broadcasted on the channel axis)
    # -------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)       # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # -------------------------------------------------
    # 4. Gating (sigmoid of an additional linear projection)
    # -------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5. Cast to fp16 – the GEMM will run in half‑precision
    # -------------------------------------------------
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape            # left_h : [B, N, N, hidden_dim]

    # Bring the hidden dimension forward so we can treat (B·hidden) as a
    # single batch dimension for the GEMM.
    left_perm  = left_h.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim                                 # combined batch size
    left_flat  = left_perm.view(BH, N, N)               # [BH, N, N] (fp16)
    right_flat = right_perm.view(BH, N, N)              # [BH, N, N] (fp16)

    # -------------------------------------------------
    # 6. Output buffer for GEMM (fp32 accumulator)
    # -------------------------------------------------
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=left_flat.device)

    # ------------------ Triton GEMM launch ------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64                      # larger K‑tile → better Tensor‑Core utilisation

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (BH, num_pid_m, num_pid_n)  # (batch, row‑tile, col‑tile)

    trimul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                                   # M,N,K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),   # A strides
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),# B strides
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),     # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,        # 8 warps = 256 threads per block – good occupancy on H100
    )
    # -------------------------------------------------
    # 7. Restore original layout: [B,N,N,hidden_dim]
    # -------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]

    # -------------------------------------------------
    # 8. Output LayerNorm over hidden_dim
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # -------------------------------------------------
    # 9. Output gating
    # -------------------------------------------------
    out = out * out_gate

    # -------------------------------------------------
    # 10. Final projection back to original dim
    # -------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
