4. **Batched N³ contraction**:
   For each head *h* (hidden_dim) we need
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,k,j,h] .
   This is exactly a batched matrix‑multiply `left @ rightᵀ` for every
   (b,h) pair.  The heavy work is done in a custom Triton kernel that
   computes the product tile‑wise (BLOCK_M × BLOCK_N) with a BLOCK_K loop.
   No explicit transpose of `right` is performed – the kernel reads
   `right` with swapped index ordering on‑the‑fly, saving a costly
   `permute(...).contiguous()` copy.
5. Layer‑norm over the hidden dimension (still in FP32) and an output gate.
6. Final linear projection back to the original channel dimension C.
7. Return the result in FP32.

The kernel uses FP16 inputs, accumulates in FP32 (Tensor‑Core friendly),
runs with 8 warps per block and block sizes (128,128,64) that give high
occupancy on an H100.  All other operations stay in PyTorch (highly
optimized torch kernels).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the N³ contraction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,            # const float16 *
    right_ptr,           # const float16 *
    out_ptr,             # float16 *
    B, N, H,             # int32 metadata
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,k,j,h]   (batched GEMM)
    The kernel works on a flattened (batch × head) dimension.
    """
    pid_head = tl.program_id(0)   # encodes (batch, head)
    pid_m = tl.program_id(1)      # block row   → i
    pid_n = tl.program_id(2)      # block col   → j

    # --------------------------------------------------------------
    # Decode batch and head from flattened head id
    # --------------------------------------------------------------
    total_heads = B * H
    b = pid_head // H
    h = pid_head % H

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the K dimension in blocks
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile  (i,k)
        left_ptrs = left_ptr + \
            b * stride_left_b + h * stride_left_h + \
            offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # Load right tile (k,j) – note the swapped index order on‑the‑fly
        right_ptrs = right_ptr + \
            b * stride_right_b + h * stride_right_h + \
            offs_k[:, None] * stride_right_i + offs_n[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # Accumulate the dot product over K
        acc += tl.dot(left, right)

    # --------------------------------------------------------------
    # Write the result tile (convert back to float16)
    # --------------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_out_b + h * stride_out_h + \
        offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton‑accelerated contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,k,j,h]
    Both tensors must be FP16, CUDA‑resident and have shape (B,N,N,H).
    Returns a FP16 tensor of the same shape.
    """
    assert left.is_cuda and right.is_cuda
    B, N, _, H = left.shape

    # Allocate output (same dtype as inputs)
    out = torch.empty_like(left)

    # Strides in element units (as expected by Triton)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rj, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Block sizes – tuned for H100 Tensor‑Core performance
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        B * H,                                     # (batch * head)
        (N + BLOCK_M - 1) // BLOCK_M,              # tiles along M (i)
        (N + BLOCK_N - 1) // BLOCK_N,              # tiles along N (j)
    )

    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rj, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,   # H100 benefits from 8 warps per block
    )
    return out


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradient required).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C]  (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (bool/float) or None
        - weights_dict : dict of torch.Tensor containing all Linear/Norm weights
        - config_dict  : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper to move a Tensor to the target device/dtype (half‑precision)
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Load model parameters
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w   = to_half(weights["left_proj.weight"])
    right_proj_w  = to_half(weights["right_proj.weight"])
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numeric stability)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(input_tensor,
                          normalized_shape=(dim,),
                          weight=norm_w,
                          bias=norm_b,
                          eps=1e-5)                 # [B,N,N,C]  fp32

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for the heavy computation
    # ------------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # [B,N,N,hidden_dim]  fp16
    right = F.linear(x, right_proj_w) # same shape

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask is float‑like or bool → convert to half and add a trailing dim
        mask_f = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # [B,N,N,1]
        # Fuse mask with the gate factors later
    else:
        mask_f = None

    # ------------------------------------------------------------------
    # 5. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # same
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # same

    # Fuse mask if present (mask * gate)
    if mask_f is not None:
        left_gate = left_gate * mask_f
        right_gate = right_gate * mask_f

    # Apply gating
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. N³ contraction – Batched matmul via Triton
    # ------------------------------------------------------------------
    # The kernel reads `right` as (b,k,j,h) on‑the‑fly, therefore no transpose.
    out = tri_mul_triton(left, right)        # [B,N,N,hidden_dim] fp16

    # ------------------------------------------------------------------
    # 7. Output LayerNorm (float32 internally)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(out_fp32,
                                 normalized_shape=(hidden_dim,),
                                 weight=to_out_norm_w.to(torch.float32),
                                 bias=to_out_norm_b.to(torch.float32),
                                 eps=1e-5)
    out_norm = out_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 8. Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate
    final = F.linear(gated, to_out_w)    # [B,N,N,dim]  fp16

    # ------------------------------------------------------------------
    # 9. Return in float32 as required by the spec
    # ------------------------------------------------------------------
    return final.to(torch.float32)
