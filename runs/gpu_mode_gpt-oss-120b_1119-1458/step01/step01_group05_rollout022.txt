@triton.jit
def proj_gate_kernel(
    x_ptr,
    mask_ptr,
    w_proj_ptr,
    w_gate_ptr,
    out_ptr,
    M, K, H,
    stride_xm, stride_xk,
    stride_maskm,
    stride_wkp, stride_whp,
    stride_outm, stride_outh,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)
    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    # Determine masks for bound check
    mask_m = offs_m < M
    mask_h = offs_h < H

    # Initialize accum
    acc_proj = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_gate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Load x block (M,K)
        x = tl.load(x_ptr + (offs_m[:, None] * stride_xm + k_offset[None, :] * stride_xk),
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)  # shape BLOCK_M, BLOCK_K

        # Load weight blocks
        w_proj = tl.load(w_proj_ptr + (k_offset[:, None] * stride_wkp + offs_h[None, :] * stride_whp),
                         mask=mask_k[:, None] & mask_h[None, :],
                         other=0.0)  # shape BLOCK_K, BLOCK_H

@triton.jit
def proj_gate_kernel(
    x_ptr, mask_ptr,
    w_proj_ptr, w_gate_ptr,
    out_ptr,
    M, K, H,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    # bounds
    m_mask = offs_m < M
    h_mask = offs_h < H

    # Initialize accumulators
    acc_proj = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_gate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    # Compute starting points for pointers: x stride: K, w stride: H
    for k_ptr in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k_ptr * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < K

        # Load x block (M,K)
        x = tl.load(x_ptr + (offs_m[:, None] * K + k_offset[None, :]), mask=m_mask[:, None] & k_mask[None, :], other=0.0)

        # Load w_proj, w_gate (K,H)
        w_proj = tl.load(w_proj_ptr + (k_offset[:, None] * H + offs_h[None, :]), mask=k_mask[:, None] & h_mask[None, :], other=0.0)
        w_gate = tl.load(w_gate_ptr + (k_offset[:, None] * H + offs_h[None, :]), mask=k_mask[:, None] & h_mask[None, :], other=0.0)

        # Accumulate
        acc_proj += tl.dot(x, w_proj)   # shape (BLOCK_M,BLOCK_H)
        acc_gate += tl.dot(x, w_gate)

    # Apply sigmoid
    gate_sig = 1.0 / (1.0 + tl.exp(-acc_gate))

    out = acc_proj * gate_sig

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_m, mask=m_mask, other=1.0)  # broadcast scalar per row
        out = out * mask_val[:, None]

    # Store
    tl.store(out_ptr + (offs_m[:, None] * H + offs_h[None, :]), out, mask=m_mask[:, None] & h_mask[None, :])

def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    batch_stride_a, row_stride_a, col_stride_a,
    batch_stride_b, row_stride_b, col_stride_b,
    batch_stride_c, row_stride_c, col_stride_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets for block tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute pointer base for this batch
    a_batch_ptr = a_ptr + pid_batch * batch_stride_a
    b_batch_ptr = b_ptr + pid_batch * batch_stride_b
    c_batch_ptr = c_ptr + pid_batch * batch_stride_c

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # Load A tile
        a = tl.load(a_batch_ptr + (offs_m[:, None] * row_stride_a + k_offset[None, :] * col_stride_a),
                    mask=(offs_m[:, None] < M) & (k_offset[None, :] < K), other=0.0)
        # Load B tile
        b = tl.load(b_batch_ptr + (k_offset[:, None] * row_stride_b + offs_n[None, :] * col_stride_b),
                    mask=(k_offset[:, None] < K) & (offs_n[None, :] < N), other=0.0)

        # Dot product
        acc += tl.dot(a, b)

    # Store to C
    tl.store(c_batch_ptr + (offs_m[:, None] * row_stride_c + offs_n[None, :] * col_stride_c),
             acc.to(tl.float16), mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_aba, stride_aaK, stride_aKc,   # Actually what to name? Use generic:
    stride_ah, stride_ak, stride_ac,
    stride_bk, stride_bn, stride_bc,
    stride_ch, stride_cn, stride_cc,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Compute base pointers for this batch
    a_batch_ptr = a_ptr + batch_id * stride_a_batch
    b_batch_ptr = b_ptr + batch_id * stride_b_batch
    c_batch_ptr = c_ptr + batch_id * stride_c_batch

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile
        a = tl.load(a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None,:] * stride_ak),
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0, dtype=tl.float16)

        # Load B tile (K,N)
        b = tl.load(b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None,:] * stride_bn),
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0, dtype=tl.float16)

        acc += tl.dot(a, b)

    # Store result
    tl.store(c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
             acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul “outgoing” forward pass – Triton‑accelerated version
========================================================

Algorithm
---------
1. Layer‑norm the 4‑D input tensor `x`  (shape [B,N,N,D]).
2. Fuse the linear projection, per‑position gating and optional mask for the
   *left* and *right* branches:
        left  = (x @ W_left_proj) * sigmoid(x @ W_left_gate) * mask
        right = (x @ W_right_proj) * sigmoid(x @ W_right_gate) * mask
   The four mat‑muls are performed in a single Triton kernel that reads each
   row of `x` only once and writes the gated result directly in FP16.
3. Compute the core contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication
   `left[b,h] @ right[b,h].T`.  This is implemented with a custom
   Triton GEMM kernel that works on the (B·H) independent small matrices
   (N×N) in FP16 while accumulating in FP32.
4. Apply a hidden‑dimensional layer‑norm, an output gate
   (sigmoid(linear(x))) and a final linear projection back to the original
   channel dimension `dim`.
5. All heavy linear work is done in half‑precision; the final output
   is returned as float32.

The implementation follows the reference PyTorch code but replaces the
projection‑+‑gate step and the N³ contraction with fused Triton kernels,
leading to a large reduction in memory traffic and runtime.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Fusion of projection + gate (+ optional mask)
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_kernel(
    x_ptr,               # pointer to X_flat   [M, K] (FP16)
    mask_ptr,            # pointer to mask     [M]   (FP16) – ignored if HAS_MASK==0
    w_proj_ptr,          # weight for projection  [K, H] (FP16) = W.T
    w_gate_ptr,          # weight for gate       [K, H] (FP16) = W.T
    out_ptr,             # output [M, H] (FP16)
    M, K, H,             # integer sizes
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid_m = tl.program_id(0)   # row block
    pid_h = tl.program_id(1)   # hidden‑channel block

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_m = offs_m < M
    mask_h = offs_h < H

    # accumulator in FP32 (Tensor‑core friendly)
    acc_proj = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
    acc_gate = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

    # loop over K (the input dimension)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # X block  (M,K)
        x = tl.load(
            x_ptr + (offs_m[:, None] * K + offs_k[None, :]),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # projection weight block (K,H)
        w_proj = tl.load(
            w_proj_ptr + (offs_k[:, None] * H + offs_h[None, :]),
            mask=mask_k[:, None] & mask_h[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # gate weight block (K,H)
        w_gate = tl.load(
            w_gate_ptr + (offs_k[:, None] * H + offs_h[None, :]),
            mask=mask_k[:, None] & mask_h[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # matrix‑multiply‑accumulate
        acc_proj += tl.dot(x, w_proj)   # (BLOCK_M,BLOCK_H) in FP32
        acc_gate += tl.dot(x, w_gate)

    # sigmoid on the gate (still FP32)
    gate_sig = 1.0 / (1.0 + tl.exp(-acc_gate))
    out = acc_proj * gate_sig

    # optional mask (broadcasted over the hidden dimension)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_m, mask=mask_m, other=1.0, dtype=tl.float16)
        out = out * mask_val[:, None]

    # store result (convert back to FP16)
    tl.store(
        out_ptr + (offs_m[:, None] * H + offs_h[None, :]),
        out.to(tl.float16),
        mask=mask_m[:, None] & mask_h[None, :],
    )


def fused_projection(x_flat, mask_flat, w_proj, w_gate, hidden_dim, has_mask):
    """
    Compute (x @ w_proj) * sigmoid(x @ w_gate) * mask in a single Triton kernel.
    All tensors are FP16 and contiguous.
    Returns a FP16 tensor of shape (M, hidden_dim).
    """
    M, K = x_flat.shape
    H = hidden_dim
    out = torch.empty((M, H), dtype=torch.float16, device=x_flat.device)

    # Tunable block sizes – work well on H100 for the given dimensions
    BLOCK_M = 128
    BLOCK_K = 32
    BLOCK_H = 64

    grid = (triton.cdiv(M, BLOCK_M), triton.cdiv(H, BLOCK_H))
    proj_gate_kernel[grid](
        x_flat,
        mask_flat,
        w_proj,
        w_gate,
        out,
        M,
        K,
        H,
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        HAS_MASK=1 if has_mask else 0,
    )
    return out

def batched_gemm_kernel(

    c_ptr,               # [batch, M, N] (FP16)   – output

    BLOCK_K: tl.constexpr,

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the current batch slice
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # K‑loop
    for k in range(0, tl.cdiv(K, BLOCK_K)):

                                  hidden_dim, use_mask)  # [M, hidden_dim] (FP16)

    # ------------------------------------------------------------------

    right_t   = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k → rows)

    # compute strides (in number of elements)
    stride_a_batch, stride_a_row, stride_a_col = a.stride()
    stride_b_batch, stride_b_row, stride_b_col = b.stride()
    stride_c_batch, stride_c_row, stride_c_col = c.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_gemm_kernel[grid](
        a,
        b,
        c,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_row,
        stride_a_col,
        stride_b_batch,
        stride_b_row,
        stride_b_col,
        stride_c_batch,
        stride_c_row,
        stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # back‑to 4‑D shape  [B, N, N, hidden_dim]
    out = c.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_fp32 = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 3) Output layer‑norm, gating and final linear projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, hidden_dim]

    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))  # [B,N,N,hidden_dim]
    out_norm = out_norm * out_gate

    out_final = F.linear(out_norm, weights["to_out.weight"])  # back to [B,N,N,dim] (float32)

    return out_final
