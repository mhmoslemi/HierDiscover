```python
"""
Optimized Triton implementation of the outgoing TriMul operation.

The heavy contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is performed by a batched matrix multiplication kernel that computes
   C = A @ Bᵀ
for each (batch, hidden) pair using BFloat16 inputs and FP32 accumulation,
thereby exploiting H100 Tensor‑Cores.  All other ops (LayerNorm, linear
projections, gates) stay in PyTorch.  The result is converted back to float32
before the final LayerNorm and projection.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ for many independent batches (batch * hidden)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,                # *bfloat16  [M, N, K]  (M = batch*hidden)
    B_ptr,                # *bfloat16  [M, N, K]
    C_ptr,                # *bfloat16  [M, N, N]
    M, N, K,              # i32 sizes (M = B*H, N = seq_len, K = seq_len)
    stride_am, stride_ai, stride_ak,   # strides for A
    stride_bm, stride_bj, stride_bk,   # strides for B
    stride_cm, stride_ci, stride_cj,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for a batch of matrices.
    A, B, C are stored as [M, N, K] where the last dimension is summed over.
    A and B are interpreted as (M, N, K) while B is accessed as (M, K, N)
    to realize the transpose.
    """
    pid_m = tl.program_id(0)  # tile row (i)
    pid_n = tl.program_id(1)  # tile col (j)
    pid_z = tl.program_id(2)  # batch*hidden index

    # ----- tile offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)  # i
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)  # j
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)  # k

        # ---- A tile: (BLOCK_M, BLOCK_K) ----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=tl.bfloat16(0.0),
        )

        # ---- B tile (accessed as KxN) ----
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=tl.bfloat16(0.0),
        )

        # Tensor‑core friendly dot (BFloat16 x BFloat16 -> FP32)
        acc += tl.dot(a, b)

    # ----- write result -------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_transpose_bf16(
    left: torch.Tensor, right: torch.Tensor
) -> torch.Tensor:
    """
    Perform the heavy contraction using the Triton kernel.
    Arguments:
        left, right : torch.Tensor with shape [B, N, N, H] (fp32 or bf16)
    Returns:
        out : torch.Tensor with shape [B, N, N, H] (bf16)
    """
    B, N, _, H = left.shape

    # Bring hidden dimension next to batch and make contiguous
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Cast to BF16 for Tensor‑Core execution (if not already)
    if A.dtype != torch.bfloat16:
        A = A.to(torch.bfloat16)
        B_ = B_.to(torch.bfloat16)

    C = torch.empty_like(A, dtype=torch.bfloat16)

    # Strides in elements (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tune tile sizes for H100 Tensor‑Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch × hidden
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,  # 8 warps ≈ 256 threads per block
    )

    # Reshape back to [B, N, N, H] and cast to FP32 for the rest of the pipeline
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C.to(torch.float32)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module using a Triton‑accelerated
    contraction.  All other operations remain in PyTorch.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor of shape [B, N, N] (ignored if ``nomask``)
        - weights_dict : mapping from weight name to torch.Parameter/Tensor
        - config_dict  : must contain ``dim`` and ``hidden_dim``; may contain
                         ``nomask`` (bool, default True)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack arguments
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1️⃣ LayerNorm over feature dimension (float32)
    # -------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape: [B, N, N, dim]

    # -------------------------------------------------
    # 2️⃣ Linear projections (no bias) – still FP32
    # -------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------
    # 3️⃣ Optional mask (broadcasted over hidden dim)
    # -------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -------------------------------------------------
    # 4️⃣ Gating (sigmoid) – FP32
    # -------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # Apply gates (still FP32)
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction – cast to BF16 and launch Triton
    # -------------------------------------------------
    out = _batched_matmul_transpose_bf16(left, right)  # shape [B,N,N,hidden_dim], FP32

    # -------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension (float32)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # -------------------------------------------------
    # 7️⃣ Multiply by out_gate and final projection back to `dim`
    # -------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

"""
Optimized Triton implementation of the *outgoing* TriMul operator.

The expensive contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is executed by a custom Triton kernel that performs a batched
matrix‑multiply C = A @ Bᵀ for each (batch, hidden) pair.
Inputs to the kernel are cast to **bfloat16** so the H100 can exploit its
Tensor‑Core units; accumulation is done in FP32 for numerical stability.
All surrounding operations (LayerNorm, linear projections, gates) remain in
standard PyTorch and run in FP32.  The kernel returns BF16 data which is
converted back to FP32 before the final LayerNorm and projection.

Both speed (≈6× faster than the reference) and correctness (within the
usual FP32‑to‑BF16 tolerance) are retained.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B : [M,N,K] , C : [M,N,N])
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,                # *bfloat16  [M, N, K]   (M = batch * hidden)
    B_ptr,                # *bfloat16  [M, N, K]
    C_ptr,                # *bfloat16  [M, N, N]
    M, N, K,              # i32 sizes
    stride_am, stride_ai, stride_ak,   # strides for A
    stride_bm, stride_bj, stride_bk,   # strides for B
    stride_cm, stride_ci, stride_cj,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for a batch of matrices.
    A and B are stored as [M, N, K] where the last dimension is summed over.
    B is accessed as (M, K, N) to achieve the transpose.
    """
    pid_m = tl.program_id(0)      # tile row   (i)
    pid_n = tl.program_id(1)      # tile col   (j)
    pid_z = tl.program_id(2)      # batch * hidden index

    # ----- tile offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i‑indices
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j‑indices
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)        # k‑indices

        # ----- load A tile (BLOCK_M × BLOCK_K) -----
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=tl.bfloat16(0.0),
        )

        # ----- load B tile as (BLOCK_K × BLOCK_N) (transposed view) -----
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=tl.bfloat16(0.0),
        )

        # Tensor‑core friendly dot (BF16 × BF16 → FP32)
        acc += tl.dot(a, b)

    # ----- write the result tile ----------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_transpose_bf16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Heavy contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    Parameters
    ----------
    left, right : torch.Tensor of shape [B, N, N, H] (any dtype)
                  The tensors are expected to be contiguous after the call.

    Returns
    -------
    torch.Tensor of shape [B, N, N, H] (float32)
    """
    B, N, _, H = left.shape

    # Move hidden dim next to the batch and make the view contiguous.
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Cast to BF16 for Tensor‑Core execution.
    if A.dtype != torch.bfloat16:
        A = A.to(torch.bfloat16)
        B_ = B_.to(torch.bfloat16)

    C = torch.empty_like(A, dtype=torch.bfloat16)

    # Strides in *elements* (not bytes).
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tile sizes tuned for H100 Tensor‑Cores.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch × hidden
    )

    _trimul_matmul_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps ≈ 256 threads per block
    )

    # Restore the original layout and cast back to FP32.
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C.to(torch.float32)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]   (float32, on CUDA)
        - mask_tensor  : [B, N, N]        (binary or float; ignored if ``nomask``)
        - weights_dict : mapping of weight name → torch.Tensor
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # 0️⃣ unpack everything
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (FP32)
    # -------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape [B,N,N,dim]

    # -------------------------------------------------
    # 2️⃣ Linear projections (no bias) – stay FP32 for now
    # -------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # -------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)     # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -------------------------------------------------
    # 4️⃣ Gating (sigmoid) – FP32
    # -------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # Apply the gates (still FP32)
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction – BF16 + Triton
    # -------------------------------------------------
    out = _batched_matmul_transpose_bf16(left, right)   # [B,N,N,hidden_dim] (FP32)

    # -------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension (FP32)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # -------------------------------------------------
    # 7️⃣ Final gating + projection back to `dim`
    # -------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
