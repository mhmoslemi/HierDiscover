"""
TriMul (outgoing) forward pass – highly‑optimized Triton implementation.

Algorithm
---------
1. Layer‑norm the input tensor (float32).
2. Cast to fp16 and apply three bias‑free linear projections:
   * left  = x @ left_proj_weight   (H dimensions)
   * right = x @ right_proj_weight  (H dimensions)
   * out_gate = x @ out_gate_weight (H dimensions, later used as a sigmoid gate)
   The two “gate” projections (left_gate, right_gate) are fused with the
   corresponding projections in a single matmul each (2*H outputs → split).
3. Optional mask is broadcast‑multiplied (fp16) onto left/right.
4. Apply sigmoid gates to left/right.
5. N³ contraction: for every head h,
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This dense contraction is a batched matrix‑multiply (left @ rightᵀ)
   implemented in Triton (fp16 × fp16 → fp32 accumulation → fp16 store).
   No extra tensor copy of the transposed right is performed – the kernel
   swaps the stride indices on‑the‑fly.
6. Layer‑norm over the hidden dimension (fp32), multiply by the sigmoid‑gated
   out_gate and project back to the original channel dimension with a final
   linear layer.
7. Output is returned in float32.

The Triton kernel uses large 256×256 tiles, a K‑tile of 32, 8 warps per block,
and 3 pipeline stages, which gives high occupancy on an H100 and fully
exploits Tensor‑Core fp16 matrix‑multiply.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the N³ contraction (left @ rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,            # const fp16 *
    right_ptr,           # const fp16 *
    out_ptr,             #      fp16 *
    B, N, H,             # int32 metadata
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
       (right is accessed as the transposed view (k, j)).
    """
    pid_head = tl.program_id(0)   # encodes (batch, head)
    pid_m    = tl.program_id(1)   # block row (i)
    pid_n    = tl.program_id(2)   # block col (j)

    total_heads = B * H
    b = pid_head // H
    h = pid_head % H

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension in blocks
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile:  (i, k)
        left_ptrs = left_ptr + \
            b * stride_left_b + h * stride_left_h + \
            offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # rightᵀ tile: (k, j) – we scramble the two N‑strides on‑the‑fly
        right_ptrs = right_ptr + \
            b * stride_right_b + h * stride_right_h + \
            offs_k[:, None] * stride_right_i + offs_n[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # accumulate the dot product over K
        acc += tl.dot(left, right)

    # ------------------------------------------------------------------
    # Store the tile (cast back to fp16)
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_out_b + h * stride_out_h + \
        offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton‑accelerated batched matmul:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    Arguments
    ---------
    left  : [B, N, N, H]  fp16 tensor
    right : [B, N, N, H]  fp16 tensor (will be accessed as (k, j) on‑the‑fly)

    Returns
    -------
    out   : [B, N, N, H]  fp16 tensor
    """
    assert left.is_cuda and right.is_cuda
    assert left.dtype == torch.float16 and right.dtype == torch.float16

    B, N, _, H = left.shape
    out = torch.empty_like(left)      # same dtype / device

    # strides in element units (not bytes)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rj, s_rh = right.stride()
    # right will be viewed as (k, j) → swap the i‑ and j‑strides
    s_rt_i, s_rt_j = s_rj, s_ri

    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Heuristic block sizes that give high occupancy on H100
    BLOCK_M = 256
    BLOCK_N = 256
    BLOCK_K = 32

    grid = (
        B * H,                                 # (batch × head) dimension
        (N + BLOCK_M - 1) // BLOCK_M,          # rows
        (N + BLOCK_N - 1) // BLOCK_N,          # columns
    )

    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_rt_i, s_rt_j, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps → 256 threads per block
        num_stages=3,         # pipeline for overlapping loads/computation
    )
    return out


# ----------------------------------------------------------------------
# Public entry point – the forward pass of the outgoing TriMul module
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : dict with all Linear/LayerNorm weights
        - config_dict  : must contain ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack and move to correct device/dtype
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper converting model parameters to the working dtype/device
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # Layer‑norm parameters stay in float32 for numerical stability
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]

    # Linear / gate weights → fp16
    left_proj_w   = to_half(weights["left_proj.weight"])
    right_proj_w  = to_half(weights["right_proj.weight"])
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(input_tensor,
                          normalized_shape=(dim,),
                          weight=norm_w,
                          bias=norm_b,
                          eps=1e-5)               # [B,N,N,C]  (fp32)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy computation
    # ------------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections + corresponding gates (fused 2*H → split)
    # ------------------------------------------------------------------
    # left projection + left gate
    left_and_lgate = F.linear(x, torch.cat([left_proj_w, left_gate_w], dim=0))
    left  = left_and_lgate[..., :hidden_dim]
    left_gate = torch.sigmoid(left_and_lgate[..., hidden_dim:])

    # right projection + right gate
    right_and_rgate = F.linear(x, torch.cat([right_proj_w, right_gate_w], dim=0))
    right = right_and_rgate[..., :hidden_dim]
    right_gate = torch.sigmoid(right_and_rgate[..., hidden_dim:])

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask may be bool – convert to fp16 and broadcast
        mask_fp16 = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5. Apply the sigmoid gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. N³ contraction via the Triton kernel (left @ rightᵀ per head)
    # ------------------------------------------------------------------
    # The kernel accesses `right` as a (k, j) view internally – no copy required.
    out = tri_mul_triton(left, right)                     # [B,N,N,hidden_dim] fp16

    # ------------------------------------------------------------------
    # 7. Output gate (computed on the original normalized input)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # [B,N,N,hidden_dim] fp16

    # ------------------------------------------------------------------
    # 8. Layer‑norm over the hidden dimension (float32 for stability)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(out_fp32,
                                 normalized_shape=(hidden_dim,),
                                 weight=to_out_norm_w.to(torch.float32),
                                 bias=to_out_norm_b.to(torch.float32),
                                 eps=1e-5)
    out_norm = out_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 9. Gated scaling then final projection back to original dim
    # ------------------------------------------------------------------
    gated = out_norm * out_gate
    final = F.linear(gated, to_out_w)                    # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 10. Return in float32 as required by the specification
    # ------------------------------------------------------------------
    return final.to(torch.float32)
