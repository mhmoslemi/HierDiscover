"""
custom_kernel
==============

Implements the *outgoing* Triangle Multiplicative Update (TriMul) from the
AlphaFold‑3 paper.

Algorithm
---------
1. **LayerNorm** over the last channel dimension of the input tensor.
2. **Half‑precision projection + gating**
   - Project the normalized tensor to the hidden dimension (`left` / `right`).
   - Apply the optional pair‑wise mask.
   - Compute sigmoid gates (`left_gate`, `right_gate`, `out_gate`) in fp16.
   - Fuse the gates with the projections (still fp16).
3. **Batched matrix‑multiplication** (`C = A @ Bᵀ`) for every hidden head.
   The 4‑D operation `einsum('... i k d, ... j k d -> ... i j d')` is
   translated into `B*hidden_dim` independent GEMMs of shape `N×N`.  Each GEMM
   is executed by a small Triton kernel that tiles the computation in M,
   N and K (BLOCK_M×BLOCK_N×BLOCK_K) and uses Tensor‑Core `tl.dot`
   (fp16 ➜ fp32 accumulation).  The kernel is launched with
   `num_warps=8` and `num_stages=2` to keep the H100 fully occupied.
4. **Post‑processing**
   - Layer‑norm over the hidden dimension (`to_out_norm`).
   - Apply the output gate.
   - Final linear projection back to the original channel dimension.

All heavy arithmetic (the N³ “einsum”) runs on the GPU in mixed precision,
while the surrounding LayerNorm / linear layers run in PyTorch (already
highly optimized).  The final output is `float32` with shape
`[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# -------------------------------------------------------------
# Triton kernel: computes C = A @ Bᵀ for many independent heads.
# A, B : fp16 tensors of shape (head, N, N) interpreted as (head, i, k) /
#        (head, j, k).  The result has shape (head, N, N) (fp32).
# -------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16 * left  (head, i, k)
    right_ptr,         # fp16 * right (head, j, k)
    out_ptr,           # fp32 * output (head, i, j)
    N,                 # i32   sequence length
    stride_lh, stride_li, stride_lk,   # i64 strides for left   (head, i, k)
    stride_rh, stride_ri, stride_rk,   # i64 strides for right  (head, j, k)
    stride_oh, stride_oi, stride_oj,   # i64 strides for out    (head, i, j)
    BLOCK_M: tl.constexpr,  # tile rows
    BLOCK_N: tl.constexpr,  # tile cols
    BLOCK_K: tl.constexpr,  # tile reduction dim
):
    # ---------------------------------------------------------
    # Identify which head (batch * hidden) and which tile we are on
    # ---------------------------------------------------------
    pid_h = tl.program_id(0)   # head index
    pid_m = tl.program_id(1)   # tile row
    pid_n = tl.program_id(2)   # tile col

    # ---------------------------------------------------------
    # Compute absolute offsets for the current M/N tiles
    # ---------------------------------------------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)   # (BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)   # (BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ---------------------------------------------------------
    # Base pointers for the current head
    # ---------------------------------------------------------
    left_head_ptr  = left_ptr  + pid_h * stride_lh
    right_head_ptr = right_ptr + pid_h * stride_rh
    out_head_ptr   = out_ptr   + pid_h * stride_oh

    # ---------------------------------------------------------
    # Accumulator (fp32) for the tile C[i,j]
    # ---------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------
    # Loop over K dimension in blocks
    # ---------------------------------------------------------
    num_blocks_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_blocks_k):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- Load left tile A[i,k] (shape BLOCK_M x BLOCK_K) ----
        # left is stored as (head, i, k)
        a_ptrs = (left_head_ptr
                  + offs_i[:, None] * stride_li
                  + offs_k[None, :] * stride_lk)
        a_mask = mask_i[:, None] & mask_k[None, :]
        a_tile = tl.load(a_ptrs, mask=a_mask, other=0.0)   # fp16

        # ---- Load right tile B[j,k] and transpose on‑the‑fly ----
        # right is stored as (head, j, k); we need (k, j)
        b_ptrs = (right_head_ptr
                  + offs_k[:, None] * stride_rk
                  + offs_j[None, :] * stride_ri)
        b_mask = mask_k[:, None] & mask_j[None, :]
        b_tile = tl.load(b_ptrs, mask=b_mask, other=0.0)   # fp16

        # ---- Tensor‑core dot product (fp16 * fp16 -> fp32) ----
        acc += tl.dot(a_tile, b_tile)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ---------------------------------------------------------
    # Write the tile back to global memory
    # ---------------------------------------------------------
    out_ptrs = (out_head_ptr
                + offs_i[:, None] * stride_oi
                + offs_j[None, :] * stride_oj)
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)


# -------------------------------------------------------------
# Entry point expected by the evaluation harness
# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float)
        - weights      : dict of model weights (all torch.Tensors)
        - config       : dict containing "dim", "hidden_dim", "nomask" ...

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ---------------------------------------------------------
    # Unpack inputs
    # ---------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # ---------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension (dim)
    # ---------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]  float32

    # ---------------------------------------------------------
    # 2️⃣ Cast to fp16 once – all subsequent linear / gating ops stay in half‑precision
    # ---------------------------------------------------------
    x_fp16 = x.half()                     # [B, N, N, dim]  fp16
    mask_fp16 = mask.unsqueeze(-1).to(torch.float16) if not nomask else None

    # Cast projection / gate weights to fp16 (one‑time conversion)
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    # ---------------------------------------------------------
    # 3️⃣ Linear projections (no bias)
    # ---------------------------------------------------------
    left  = F.linear(x_fp16, left_proj_w)   # [B,N,N,hidden] fp16
    right = F.linear(x_fp16, right_proj_w)  # [B,N,N,hidden] fp16

    # ---------------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dim)
    # ---------------------------------------------------------
    if not nomask:
        left  = left * mask_fp16
        right = right * mask_fp16

    # ---------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – keep in fp16
    # ---------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # fp16
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))  # fp16
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w))   # fp16

    # ---------------------------------------------------------
    # 6️⃣ Fuse gates into the projected tensors
    # ---------------------------------------------------------
    left  = left * left_gate   # fp16
    right = right * right_gate # fp16

    # ---------------------------------------------------------
    # 7️⃣ Prepare contiguous view (head, i, k) / (head, j, k)
    # ---------------------------------------------------------
    B, N, _, _ = left.shape          # B, N, N, hidden
    total_heads = B * hidden_dim

    # Layout required by the Triton kernel: (head, N, N)
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # ---------------------------------------------------------
    # 8️⃣ Allocate output buffer (fp32)
    # ---------------------------------------------------------
    out_flat = torch.empty((total_heads, N, N), dtype=torch.float32, device=device)

    # ---------------------------------------------------------
    # 9️⃣ Triton launch configuration
    # ---------------------------------------------------------
    # Tile sizes tuned for H100 Tensor‑Core performance.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        total_heads,                          # head dimension
        (N + BLOCK_M - 1) // BLOCK_M,         # number of row tiles
        (N + BLOCK_N - 1) // BLOCK_N,         # number of col tiles
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps → 256 threads per block
        num_stages=2,         # double‑buffering for the K‑loop
    )

    # ---------------------------------------------------------
    # 10️⃣ Reshape back to [B, N, N, hidden_dim]
    # ---------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)   # (B, hidden, N, N)
        .permute(0, 2, 3, 1)                # (B, N, N, hidden)
        .contiguous()
    )  # float32

    # ---------------------------------------------------------
    # 11️⃣ Post‑processing: hidden‑dim LayerNorm, output gate, final projection
    # ---------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # out_gate is fp16 → auto‑casts to fp32 for the multiplication
    out = out * out_gate

    # Final linear back to the original dimensionality (float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
