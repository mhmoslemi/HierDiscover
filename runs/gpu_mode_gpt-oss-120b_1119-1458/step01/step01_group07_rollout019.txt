"""
TriMul (outgoing) – Triton‑accelerated core multiplication.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32).
2. Three linear projections and three sigmoid gates (float32).
3. Apply optional pair‑wise mask and fuse the 2 “left/right” gates
   (still float32).  The resulting tensors have shape
   [B, N, N, H] where H = hidden_dim.
4. Cast the gated tensors to fp16 and reshape them to a
   batched matrix‑multiply view    (B·H, N, N).  The heavy O(N³) reduction
   is performed by a custom Triton kernel that computes
   C = A @ Bᵀ with fp16 inputs, fp32 accumulation and fp16 output.
5. Convert the kernel output back to fp32, apply a second LayerNorm
   (hidden dim) and the output gate.
6. Final linear projection back to the original channel dimension.

Only the O(N³) reduction is off‑loaded to Triton; all other
operations use PyTorch (which is already well‑optimised).  Using
fp16 for the large matmul halves memory traffic and lets the
kernel leverage Tensor‑Core mma, giving a several‑fold speed‑up
over the pure‑float32 reference implementation.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def batch_matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # matrix sizes (all == seq_len)
    stride_a_batch, stride_am, stride_ak,   # A strides (batch, M, K)
    stride_b_batch, stride_bk, stride_bn,   # B strides (batch, K, N) – note transposed view
    stride_c_batch, stride_cm, stride_cn,   # C strides (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for a batch of matrices.
    A : (batch, M, K)   (fp16)
    B : (batch, N, K)   (fp16) – accessed as transposed
    C : (batch, M, N)   (fp16)
    """
    batch_idx = tl.program_id(0)          # which matrix in the batch
    pid_m = tl.program_id(1)              # block row
    pid_n = tl.program_id(2)              # block column

    # ------------------- block offsets -------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------- batch base pointers -------------
    a_batch = a_ptr + batch_idx * stride_a_batch
    b_batch = b_ptr + batch_idx * stride_b_batch
    c_batch = c_ptr + batch_idx * stride_c_batch

    # ------------------- accumulator ----------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k = tl.cdiv(K, BLOCK_K)

    for k in range(0, num_k):
        k_base = k * BLOCK_K
        cur_k = k_base + offs_k
        mask_k = cur_k < K

        # A block: (BLOCK_M, BLOCK_K) – fp16
        a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (cur_k[None, :] * stride_ak)
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # Bᵀ block: (BLOCK_K, BLOCK_N) – fp16
        # we treat B as (N, K) and access its transpose
        b_ptrs = b_batch + (cur_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # dot‑product (Tensor‑Core) – accumulation in fp32
        acc += tl.dot(a, b)

    # ------------------- write back -----------------------
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        • input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        • mask_tensor  : torch.Tensor, shape [B, N, N] (bool/int, may be None)
        • weights_dict : mapping from name → torch.Tensor (all on same device)
        • config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device).
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # retrieve weight tensors
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (float32)
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim] – float32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (float32, no bias)
    # --------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)    # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B, N, N, H]

    # --------------------------------------------------------------
    # 3️⃣ Gating (sigmoid, float32)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 5️⃣ Fuse gates (still float32)
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Core TriMul – batched GEMM left @ rightᵀ (fp16 Triton kernel)
    # --------------------------------------------------------------
    # Cast the gated tensors to fp16 (memory‑bandwidth reduction)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Move hidden dimension to the batch axis: [B, H, N, N] → (B·H, N, N)
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    batch_mul = bs * hidden_dim
    left_flat = left_perm.view(batch_mul, N, N)   # [B·H, N, N]
    right_flat = right_perm.view(batch_mul, N, N) # [B·H, N, N]

    # Allocate output tensor for the batched matmul (fp16)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # -----------------------------------------------------------------
    # stride information (in element units, not bytes)
    # -----------------------------------------------------------------
    stride_a_batch = left_flat.stride(0)
    stride_am = left_flat.stride(1)   # stride along M (rows)
    stride_ak = left_flat.stride(2)   # stride along K (cols)

    stride_b_batch = right_flat.stride(0)
    # for the transposed view we swap the last two strides
    stride_bk = right_flat.stride(2)   # original K stride
    stride_bn = right_flat.stride(1)   # original N stride

    stride_c_batch = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # --------------------------------------------------------------
    # launch configuration (tuned for H100 Tensor‑Core)
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # multiple of 16 → Tensor‑Core friendly

    grid = (
        batch_mul,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batch_matmul_transpose_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_b_batch=stride_b_batch,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_c_batch=stride_c_batch,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # Restore 4‑D layout and cast back to fp32
    # --------------------------------------------------------------
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.float()   # fp32 for the remaining FP operations

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm over hidden dimension (float32)
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 8️⃣ Apply the output gate (float32)
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 9️⃣ Final linear projection hidden → dim (float32, no bias)
    # --------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
