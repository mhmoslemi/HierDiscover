Triton‑accelerated forward pass for the “outgoing” TriMul operator.

Algorithm
---------
1. Layer‑norm the input tensor (float32 → float16 after the norm).

        out[i, j, d] = Σ_k left[i, k, d] * right[j, k, d]
   which is a batched GEMM for each hidden channel.  The tensors are
   transposed to (B, H, N, N), cast to FP16 and multiplied with a single
   `torch.bmm` call.
5. Layer‑norm over the hidden dimension, a fused element‑wise multiply with
   the output gate (implemented as a tiny Triton kernel), and a final
   linear projection back to the model dimension.
6. The final output is returned in the original dtype (float32).

Only the element‑wise product of the output norm and its gate is written
as a Triton kernel (`fused_mul2`).  All heavy linear algebra is performed
with the highly‑optimised Torch/CuBLAS kernels, while the number of
memory‑bound passes is reduced by combining the four initial linear layers
into a single GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def fused_mul2_kernel(
    a_ptr, b_ptr, out_ptr,                     # pointers
    size,                                       # total number of elements
    BLOCK_SIZE: tl.constexpr,                  # compile‑time constant
):
    """Element‑wise product out = a * b (all tensors 1‑D contiguous)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)

    out = a * b
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul2(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Fused element‑wise product of two tensors using Triton."""
    assert a.shape == b.shape
    a = a.contiguous()
    b = b.contiguous()
    out = torch.empty_like(a)

    total = a.numel()
    BLOCK = 1024  # good default for H100
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul2_kernel[grid](
        a, b, out, total, BLOCK_SIZE=BLOCK
    )
    return out


# ----------------------------------------------------------------------
# Main kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor   shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor   shape [B, N, N] (bool or float)
        - weights_dict : dict of model weights
        - config_dict  : contains "dim", "hidden_dim", optional "eps",
                         optional "nomask"

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    H = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input Layer‑Norm (float32 → keep FP32 for numerical stability)
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] (float32)

    # --------------------------------------------------------------
    # 2. Fused linear projections (proj + gate for left/right)
    # --------------------------------------------------------------
    # Concatenate the four weight matrices into a single (4*H, dim) matrix
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],   # (H, dim)
            weights["left_gate.weight"],   # (H, dim)
            weights["right_proj.weight"],  # (H, dim)
            weights["right_gate.weight"],  # (H, dim)
        ],
        dim=0,
    ).to(torch.float16)                       # (4*H, dim) in FP16

    # Cast the normalized tensor to FP16 for the large GEMM
    x_norm_fp16 = x_norm.to(torch.float16)

    # Single linear pass: [B, N, N, 4*H]
    combined = F.linear(x_norm_fp16, combined_weight)  # bias = None

    # Split back to the four constituent tensors
    left_proj, left_gate_pre, right_proj, right_gate_pre = torch.split(
        combined, H, dim=-1
    )  # each: [B, N, N, H] (FP16)

    # Apply sigmoid to obtain gates
    left_gate = torch.sigmoid(left_gate_pre)   # [B, N, N, H] (FP16)
    right_gate = torch.sigmoid(right_gate_pre)  # [B, N, N, H] (FP16)

    # --------------------------------------------------------------
    # 3. Apply mask (if any) and combine with gates
    # --------------------------------------------------------------
    if nomask:
        left = left_proj * left_gate                      # FP16
        right = right_proj * right_gate                   # FP16
    else:
        # mask may be bool or float – convert to FP16 and broadcast
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left_proj * left_gate * mask_fp16
        right = right_proj * right_gate * mask_fp16

    # --------------------------------------------------------------
    # 4. Core contraction  out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    # --------------------------------------------------------------
    # Reshape for batched GEMM: move hidden dimension to batch axis
    # left : (B, N, N, H) → (B, H, N, N) where rows = i, cols = k
    left_perm = left.transpose(1, 3).contiguous()            # (B, H, N, N)
    # right: need rows = j, cols = k → transpose then swap last two dims
    right_perm = right.transpose(1, 3).contiguous()          # (B, H, N, N)
    right_t = right_perm.transpose(2, 3).contiguous()       # (B, H, N, N) with (k, j)

    # Collapse batch and hidden dimensions for a single bmm call
    left_flat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_t.view(B * H, N, N)    # (B*H, N, N)

    # Batched matrix multiply in FP16
    out_flat = torch.bmm(left_flat, right_flat)  # (B*H, N, N)

    # Restore original shape [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 5. Output Layer‑Norm over hidden dimension (keep FP16)
    # --------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )  # (B, N, N, H) FP16

    # Output gate (sigmoid) – also in FP16
    out_gate = torch.sigmoid(
        F.linear(x_norm_fp16, weights["out_gate.weight"].to(torch.float16))
    )  # (B, N, N, H)

    # Fuse the final element‑wise product with a tiny Triton kernel
    out_gated = fused_mul2(out_norm, out_gate)  # (B, N, N, H) FP16

    # --------------------------------------------------------------
    # 6. Final linear projection back to `dim`
    # --------------------------------------------------------------
    out_final = F.linear(
        out_gated, weights["to_out.weight"].to(torch.float16)
    )  # (B, N, N, dim) FP16

    # Cast back to the original dtype (float32)
    return out_final.to(inp.dtype)
```
