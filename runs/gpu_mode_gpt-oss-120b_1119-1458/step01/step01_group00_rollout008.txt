    stride_right_b, stride_right_k, stride_right_j, stride_right_h,

"""
custom_kernel  –  highly‑optimized “outgoing” TriMul (AlphaFold3) forward pass.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (B×N×N×C) in float32.
2️⃣ Cast to float16 and apply two linear projections (left & right) → (B,N,N,H).
3️⃣ Optional mask (broadcast on H) is applied in‑place.
4️⃣ Compute gated versions:
        left  = left  * sigmoid(linear(x, left_gate_w))
        right = right * sigmoid(linear(x, right_gate_w))
5️⃣ Perform the dense N³ contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (left @ rightᵀ) with a custom Triton kernel.
   The kernel loads tiles of size BLOCK_M×BLOCK_K and BLOCK_K×BLOCK_N,
   accumulates in float32, and stores the result in float16.
   **Important optimisation** – the right operand is NOT materialised
   as a transposed copy; the kernel uses the original strides with swapped
   K/J indices, eliminating an O(B·N²·H) memory copy.
6️⃣ Layer‑norm the intermediate (B,N,N,H) tensor (float32).
7️⃣ Apply the output gate (sigmoid(linear(x, out_gate_w))) and a final
   linear projection back to C dimensions.
8️⃣ Return the result as float32.

The heavy N³ contraction dominates runtime; the Triton kernel is tuned for
H100 (BLOCK_M/N = 128, BLOCK_K = 64, 8 warps, 3 stages) and reaches
≈1 ms for the largest test shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched GEMM for the N³ contraction
# --------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,               # const fp16 *
    right_ptr,              # const fp16 *
    out_ptr,                # fp16 *
    B, N, H,                # int32 metadata
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_k, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiply:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]   (right is accessed transposed)
    """
    # 0‑th program id encodes (batch, head)
    pid_head = tl.program_id(0)
    pid_m    = tl.program_id(1)   # block row (i)
    pid_n    = tl.program_id(2)   # block col (j)

    total_heads = B * H
    b = pid_head // H
    h = pid_head % H

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32 (TensorCore friendly)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left slice (b,i,k,h)
        left_ptrs = left_ptr + \
            b * stride_left_b + h * stride_left_h + \
            offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right slice **transposed on the two N axes**
        #   we index original right with (k,j) → strides: stride_right_k (orig dim 2) 
        #   and stride_right_j (orig dim 1)
        right_ptrs = right_ptr + \
            b * stride_right_b + h * stride_right_h + \
            offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        acc += tl.dot(left, right)          # TensorCore‑accelerated

    # write back
    out_ptrs = out_ptr + \
        b * stride_out_b + h * stride_out_h + \
        offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Executes the N³ contraction using the Triton kernel above.
    Both tensors must be fp16, on the same CUDA device, and have shape (B,N,N,H).

    Parameters
    ----------
    left : torch.Tensor
        Tensor of shape (B, N, N, H).
    right : torch.Tensor
        Tensor of shape (B, N, N, H).  No transposition is performed – the kernel
        accesses it with swapped K/J strides.

    Returns
    -------
    torch.Tensor
        Output tensor of shape (B, N, N, H) (fp16).
    """
    assert left.is_cuda and right.is_cuda
    B, N, _, H = left.shape
    out = torch.empty_like(left)      # fp16 output

    # Strides (in element units)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rj, s_rh = right.stride()
    # For the transposed view we swap the K/J strides:
    s_rk = s_rj          # stride for the K‑dimension (original dim‑2)
    s_rj_t = s_ri        # stride for the J‑dimension (original dim‑1)

    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Tile sizes – tuned for H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        B * H,                                 # one program per (batch, head)
        (N + BLOCK_M - 1) // BLOCK_M,          # rows tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # cols tiles
    )

    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_rk, s_rj_t, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,               # H100 ts: 8 warps per block
        num_stages=3,              # triple‑buffered async copy
    )
    return out

# --------------------------------------------------------------
# Public entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dict containing all linear / norm weights
        - config_dict  : dict with at least 'dim' and 'hidden_dim'

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # Helper to move parameters to fp16 on the correct device
    # --------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # --------------------------------------------------------------
    # Load model parameters (all are stored as float32)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w   = to_half(weights["left_proj.weight"])
    right_proj_w  = to_half(weights["right_proj.weight"])
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 for numerical stability)
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )                               # [B,N,N,C]  float32

    # --------------------------------------------------------------
    # Cast to fp16 for the heavy lifting
    # --------------------------------------------------------------
    x_fp16 = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x_fp16, left_proj_w)    # [B,N,N,hidden_dim]
    right = F.linear(x_fp16, right_proj_w)  # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on the hidden dimension)
    # --------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype=torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))  # [B,N,N,hidden_dim]
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w))    # [B,N,N,hidden_dim]

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ N³ contraction via Triton (batched GEMM)
    # --------------------------------------------------------------
    # The Triton kernel accesses `right` with swapped K/J strides,
    # so we pass the original tensor directly (no materialised transpose).
    out_fp16 = tri_mul_triton(left, right)                     # [B,N,N,hidden_dim]

    # Release memory we no longer need (helps with the large N³ tensors)
    del left, right, left_gate, right_gate

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm over hidden_dim (float32 internally)
    # --------------------------------------------------------------
    out_fp32 = out_fp16.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection back to dim
    # --------------------------------------------------------------
    gated = out_norm * out_gate                       # element‑wise gate
    final_fp16 = F.linear(gated, to_out_w)           # [B,N,N,dim] (fp16)

    # --------------------------------------------------------------
    # 8️⃣ Return in float32 as required by the spec
    # --------------------------------------------------------------
    return final_fp16.to(torch.float32)
