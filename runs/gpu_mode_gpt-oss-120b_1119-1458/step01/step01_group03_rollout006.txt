"""
TriMul (outgoing) forward pass accelerated with Triton.

The heavy contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is a batched GEMM for every hidden channel h.
We treat (batch, hidden) as a fused batch dimension and compute
a GEMM per (b,h) using a custom Triton kernel that reads half‑precision
inputs, performs fp16×fp16→fp32 dot‐products (Tensor‑Core), and writes the
accumulated float32 result.  This avoids the costly permutation/copy that
the naïve implementation performs (left.permute(...).contiguous()) and
therefore yields a large speed‑up on H100.  All other TriMul steps
(LayerNorm, linear projections, gating, output LayerNorm and projection)
are kept in PyTorch.

The kernel receives the strides of the 4‑D tensors so that no extra
memory layout transformations are required.
"""
import torch
import triton
import triton.language as tl


@triton.jit
def trimul_gemm_kernel(
    L_ptr,  # half *, shape (B, N, N, H)
    R_ptr,  # half *, shape (B, N, N, H)
    O_ptr,  # float *, shape (B, N, N, H)
    N,                # sequence length (runtime int)
    H,                # hidden_dim (runtime int)
    stride_bL, stride_iL, stride_kL, stride_hL,
    stride_bR, stride_iR, stride_kR, stride_hR,
    stride_bO, stride_iO, stride_jO, stride_hO,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    The third program dimension packs (batch, hidden) → batch_h.
    """
    pid_m = tl.program_id(0)                # tile row index (i)
    pid_n = tl.program_id(1)                # tile col index (j)
    pid_h = tl.program_id(2)                # combined (batch, hidden)

    # Decode batch and hidden indices
    batch_idx = pid_h // H
    hidden_idx = pid_h % H

    # Offsets for the current tile in i, j dimensions
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for the current (batch, hidden) slice
    L_base = L_ptr + batch_idx * stride_bL + hidden_idx * stride_hL
    R_base = R_ptr + batch_idx * stride_bR + hidden_idx * stride_hR
    O_base = O_ptr + batch_idx * stride_bO + hidden_idx * stride_hO

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a tile of L: shape (BLOCK_M, BLOCK_K)
        L_ptrs = L_base + offs_i[:, None] * stride_iL + offs_k[None, :] * stride_kL
        L = tl.load(L_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load a transposed tile of R (K, N)    R has layout (b, j, k, h)
        R_ptrs = R_base + offs_k[:, None] * stride_kR + offs_j[None, :] * stride_iR
        R = tl.load(R_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(L, R)          # FP32 dot ⇒ Tensor‑Core when inputs are half

    # Store the result
    O_ptrs = O_base + offs_i[:, None] * stride_iO + offs_j[None, :] * stride_jO
    tl.store(O_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated TriMul (outgoing) forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # Unpack arguments ---------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    device = input_tensor.device
    dtype = input_tensor.dtype

    # 1. Input LayerNorm -------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim]  float32

    # 2. Linear projections (no bias) -----------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # 3. Optional mask (broadcast over hidden) --------------------------------
    if not nomask:
        # mask shape [B,N,N] -> [B,N,N,1] for broadcasting
        mask_f = mask_tensor.to(dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # 4. Gates -----------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # 5. Cast to half for the heavy GEMM --------------------------------------
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    B, N, _, _ = left_h.shape   # left_h : [B,N,N,hidden_dim]

    # 6. Allocate output buffer (float32 accumulator) -------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # 7. Launch Triton GEMM ----------------------------------------------------
    # Block sizes can be tuned; values below work well on H100.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid: (tiles_m, tiles_n, batch*hidden)
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    # Strides (convert to python int for Triton)
    stride_bL, stride_iL, stride_kL, stride_hL = left_h.stride()
    stride_bR, stride_iR, stride_kR, stride_hR = right_h.stride()
    stride_bO, stride_iO, stride_jO, stride_hO = out.stride()

    trimul_gemm_kernel[grid](
        left_h,
        right_h,
        out,
        N,
        hidden_dim,
        stride_bL,
        stride_iL,
        stride_kL,
        stride_hL,
        stride_bR,
        stride_iR,
        stride_kR,
        stride_hR,
        stride_bO,
        stride_iO,
        stride_jO,
        stride_hO,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        num_warps=8,          # H100 prefers 8 warps per block
    )

    # 8. Output LayerNorm ------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # 9. Output gating ---------------------------------------------------------
    out = out * out_gate

    # 10. Final linear projection back to original dim -------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
