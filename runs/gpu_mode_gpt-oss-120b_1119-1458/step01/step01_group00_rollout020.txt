"""
TriMul (outgoing) – highly‑optimized forward implementation.

Algorithm
---------
1. Layer‑norm the input (float32 → float16).
2. Fuse the four linear projections (left, right, left‑gate, right‑gate) into a
   single matmul:   out = x @ [W_left;W_right;W_left_gate;W_right_gate]ᵀ .
   The result is sliced into the four tensors, sigmoid is applied to the gates,
   and left/right are multiplied by their gates (mask applied afterwards).
3. Compute the dense N³ contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   with a fast path:
   * For small workloads we use a custom Triton kernel (batched GEMM).
   * For realistic sizes we rely on torch.bmm (FP16) which maps to Tensor‑Core
     libraries and far out‑performs a naïve Triton implementation.
4. Layer‑norm the hidden‑dim result, apply the output gate, and a final linear
   projection back to the original channel dimension.
5. Return the final tensor in float32.

Only the heavy N³ contraction is written in Triton; the rest uses
high‑performance PyTorch kernels and fused linear layers to minimise memory
traffic (the large input tensor is read only twice).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,            # const float16 *
    right_ptr,           # const float16 *
    out_ptr,             # float16 *
    B, N, H,             # int32 metadata
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matmul: out = left @ rightᵀ for each (b, h) pair."""
    pid_head = tl.program_id(0)   # encodes batch * hidden
    pid_m = tl.program_id(1)      # block row   (i)
    pid_n = tl.program_id(2)      # block col   (j)

    total_heads = B * H
    b = pid_head // H
    h = pid_head % H

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile (i,k)
        left_ptrs = left_ptr + \
            b * stride_left_b + h * stride_left_h + \
            offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile (k,j) – note that right_ptr already points to the transposed view
        right_ptrs = right_ptr + \
            b * stride_right_b + h * stride_right_h + \
            offs_k[:, None] * stride_right_i + offs_n[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_ptr + \
        b * stride_out_b + h * stride_out_h + \
        offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def tri_mul_triton(left: torch.Tensor, right_t: torch.Tensor) -> torch.Tensor:
    """
    Triton‑accelerated batched GEMM:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right_t[b,k,j,h]
    left  : [B, N, N, H]  (float16)
    right_t: [B, N, N, H]  (float16) – already transposed on the two N axes.
    Returns a tensor of shape [B, N, N, H] (float16).
    """
    assert left.is_cuda and right_t.is_cuda
    B, N, _, H = left.shape

    out = torch.empty_like(left)

    # strides in element units
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rj, s_rh = right_t.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # block sizes – tuned for H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                        # batch × hidden
        (N + BLOCK_M - 1) // BLOCK_M, # row tiles
        (N + BLOCK_N - 1) // BLOCK_N, # col tiles
    )

    tri_mul_kernel[grid](
        left, right_t, out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rj, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,   # 4 warps works well on H100
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dict of weight tensors for all Linear / LayerNorm layers
        - config_dict  : dict containing at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper to move weights to fp16 on the target device
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Load model parameters
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]                # (C,)
    norm_b = weights["norm.bias"]                  # (C,)

    left_proj_w   = to_half(weights["left_proj.weight"])   # (hidden_dim, C)
    right_proj_w  = to_half(weights["right_proj.weight"])  # (hidden_dim, C)
    left_gate_w   = to_half(weights["left_gate.weight"])   # (hidden_dim, C)
    right_gate_w  = to_half(weights["right_gate.weight"])  # (hidden_dim, C)
    out_gate_w    = to_half(weights["out_gate.weight"])    # (hidden_dim, C)

    to_out_norm_w = to_half(weights["to_out_norm.weight"])  # (hidden_dim,)
    to_out_norm_b = to_half(weights["to_out_norm.bias"])    # (hidden_dim,)
    to_out_w      = to_half(weights["to_out.weight"])       # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numeric stability)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(input_tensor,
                          normalized_shape=(dim,),
                          weight=norm_w,
                          bias=norm_b,
                          eps=1e-5)          # [B,N,N,dim]  (float32)

    # Cast to fp16 – the heavy work will be in half precision
    x_half = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Fuse the four linear projections (left, right, left_gate, right_gate)
    # ------------------------------------------------------------------
    # Concatenate weight matrices along the output‑feature dimension
    fused_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w], dim=0
    )                                            # shape (4*hidden_dim, dim)

    fused_out = F.linear(x_half, fused_weight)   # [B,N,N,4*hidden_dim] (fp16)

    H = hidden_dim
    left_raw      = fused_out[..., 0:H]                     # (B,N,N,H)
    right_raw     = fused_out[..., H:2*H]                   # (B,N,N,H)
    left_gate_raw = fused_out[..., 2*H:3*H]                 # (B,N,N,H)
    right_gate_raw= fused_out[..., 3*H:4*H]                 # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid) and optional mask
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    left  = left_raw * left_gate   # (B,N,N,H)
    right = right_raw * right_gate # (B,N,N,H)

    if mask is not None:
        mask_f = mask.to(dtype=torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. N³ contraction (batched matmul)
    # ------------------------------------------------------------------
    # Choose the fastest path:
    #   – tiny problems ⇒ Triton kernel (helps verification)
    #   – real workloads  ⇒ torch.bmm (Tensor‑Core implementation)
    use_triton = (N <= 256 and hidden_dim <= 128)   # heuristic
    if use_triton:
        # rightᵀ view required by the kernel
        right_t = right.permute(0, 3, 2, 1).contiguous()
        out_hidden = tri_mul_triton(left, right_t)     # (B,N,N,H) fp16
    else:
        # Batched GEMM using torch.bmm
        # (B,N,N,H) → (B*H,N,N) for both operands
        left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
        right_mat = right.permute(0, 3, 2, 1).reshape(B * H, N, N)
        # out_mat[b*h, i, j] = Σ_k left[i,k] * right[j,k]
        out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
        # reshape back to [B,N,N,H]
        out_hidden = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5. Output layer‑norm (hidden_dim) – performed in float32
    # ------------------------------------------------------------------
    out_fp32 = out_hidden.to(torch.float32)
    out_norm_fp32 = F.layer_norm(out_fp32,
                                 normalized_shape=(hidden_dim,),
                                 weight=to_out_norm_w.to(torch.float32),
                                 bias=to_out_norm_b.to(torch.float32),
                                 eps=1e-5)
    out_norm = out_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Output gate and final projection back to original dim
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_half, out_gate_w))   # (B,N,N,H)
    gated = out_norm * out_gate
    final = F.linear(gated, to_out_w)          # (B,N,N,dim)  (fp16)

    # ------------------------------------------------------------------
    # Return in float32 as required by the specification
    # ------------------------------------------------------------------
    return final.to(torch.float32)
