                        stride_om, stride_on, stride_ok,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):

    num_pid_m = tl.cdiv(N, BLOCK_M)

import triton
import triton.language as tl

@triton.jit
def triton_tri_mul_kernel_fused(
    left_ptr, right_ptr, out_ptr,
    M, N,
    stride_lm, stride_ln, stride_lk,
    stride_rm, stride_rk, stride_rn,
    stride_om, stride_on, stride_ok,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # each program handles a 2D tile (BLOCK_M, BLOCK_N) of a matrix in the batch.
    pid = tl.program_id(0)   # batch dimension (B*H)
    pid_m = tl.program_id(1) # tile row
    pid_n = tl.program_id(2) # tile col

    # Compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out-of-range rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M x K)
        left = tl.load(
            left_ptr + pid * stride_lm
                     + offs_m[:, None] * stride_ln
                     + offs_k[None, :] * stride_lk,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )
        # Load right tile transposed (K x N)
        right = tl.load(
            right_ptr + pid * stride_rm
                      + offs_k[:, None] * stride_rk
                      + offs_n[None, :] * stride_rn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0

        acc += tl.dot(left, right).to(tl.float32)
    # Store with mask
    tl.store(
        out_ptr + pid * stride_om
                + offs_m[:, None] * stride_on
                + offs_n[None, :] * stride_ok,
        acc,
        mask=mask_m[:, None] & mask_n[:, None]  # careful: mask_n need be broadcast across rows
)

"""
TriMul (outgoing) forward pass – high‑performance Triton implementation.

Algorithm
---------
1. Layer‑norm the input tensor (float32) using the provided weights.
2. Project the normalized tensor to the hidden dimension with four linear
   layers (left/right projection and three gates).  All linear ops are
   performed in float16 to improve bandwidth.
3. Optional pairwise mask is applied before gating.
4. Gates (sigmoid) are multiplied into the projected tensors.
5. The heavy N³ reduction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix‑multiply (left @ rightᵀ) for each hidden channel.
   We fuse this computation in a single Triton kernel that uses
   `tl.dot` (tensor‑core friendly) with FP16 inputs and FP32 accumulation.
6. The result is reshaped back to (B, N, N, hidden_dim) and cast to FP32.
7. A second layer‑norm, an output gate, and a final linear projection
   return a tensor of shape (B, N, N, dim).

The Triton kernel is tuned for H100 GPUs:
  * Block sizes 128 × 128 × 32 (M, N, K) give good occupancy and exploit
    Tensor‑cores.
  * All memory loads are guarded with simple range masks; the kernel works
    for any sequence length (N) without padding.
  * Accumulation is done in FP32 for numerical stability, then cast back to
    FP16 for storage.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# -------------------------------------------------------------------------
# Triton kernel: batched Gram matrix (left @ rightᵀ) over the sequence dim.
# -------------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,               # *fp16
    right_ptr,              # *fp16
    out_ptr,                # *fp16
    M,                      # total batches = B * hidden_dim
    N,                      # sequence length
    stride_lm, stride_ln, stride_lk,
    stride_rm, stride_rk, stride_rn,
    stride_om, stride_on, stride_ok,
    BLOCK_M: tl.constexpr,  # tile rows
    BLOCK_N: tl.constexpr,  # tile cols
    BLOCK_K: tl.constexpr,  # reduction tile
):
    """
    Compute for each batch `m`:
        out[m, i, j] = Σ_k left[m, i, k] * right[m, j, k]
    which is exactly left @ rightᵀ.
    """
    pid = tl.program_id(0)   # batch*hidden index
    pid_i = tl.program_id(1) # tile row
    pid_j = tl.program_id(2) # tile col

    # -----------------------------------------------------------------
    # Tile start offsets
    # -----------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 (more stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over K dimension in blocks
    # -----------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_ptr
            + pid * stride_lm
            + offs_i[:, None] * stride_ln
            + offs_k[None, :] * stride_lk,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right block transposed: we need right[k, j] (K × N)
        right = tl.load(
            right_ptr
            + pid * stride_rm
            + offs_k[:, None] * stride_rk
            + offs_j[None, :] * stride_rn,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # Perform FP16 dot, accumulate in FP32
        acc += tl.dot(left, right).to(tl.float32)

    # -----------------------------------------------------------------
    # Write back – cast accumulator to FP16
    # -----------------------------------------------------------------
    tl.store(
        out_ptr
        + pid * stride_om
        + offs_i[:, None] * stride_on
        + offs_j[None, :] * stride_ok,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# -------------------------------------------------------------------------
# Public entry point
# -------------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments:
        data = (input_tensor, mask, weights, config)

    Returns:
        Tensor of shape [batch, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # -----------------------------------------------------------------
    # Configuration
    # -----------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numeric stability)
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (bias=False).  We cast to fp16 after LN.
    # -----------------------------------------------------------------
    x_fp16 = x.half()

    left = torch.nn.functional.linear(x_fp16, weights["left_proj.weight"].half())
    right = torch.nn.functional.linear(x_fp16, weights["right_proj.weight"].half())

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dimension)
    # -----------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → [B, N, N, 1]
        mask_b = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_b
        right = right * mask_b

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid) – performed in float32 then cast to fp16.
    # -----------------------------------------------------------------
    # Note: gating uses the *float32* LN output for better accuracy.
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Cast gates to fp16 so that multiplication stays in fp16.
    left_gate_fp16 = left_gate.half()
    right_gate_fp16 = right_gate.half()

    # -----------------------------------------------------------------
    # 5. Apply gates
    # -----------------------------------------------------------------
    left = left * left_gate_fp16
    right = right * right_gate_fp16

    # -----------------------------------------------------------------
    # 6. Prepare tensors for the Triton kernel:
    #    (B, N, N, H) → (B*H, N, N)   where H = hidden_dim
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    # Rearrange to (B, H, N, N) then flatten batch*hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)   # (BH, N, N)
    right_flat = right_perm.view(BH, N, N)

    # Output buffer (fp16)
    out_flat = torch.empty_like(left_flat)

    # -----------------------------------------------------------------
    # 7. Triton kernel launch
    # -----------------------------------------------------------------
    stride_lm, stride_ln, stride_lk = left_flat.stride()
    stride_rm, stride_rk, stride_rn = right_flat.stride()
    stride_om, stride_on, stride_ok = out_flat.stride()

    # Tuned block sizes for H100 – they are constexpr in the kernel.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        BH,                                     # batch*hidden
        (N + BLOCK_M - 1) // BLOCK_M,           # tile rows
        (N + BLOCK_N - 1) // BLOCK_N,           # tile cols
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        BH,
        N,
        stride_lm,
        stride_ln,
        stride_lk,
        stride_rm,
        stride_rk,
        stride_rn,
        stride_om,
        stride_on,
        stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 8. Reshape back to (B, N, N, H) and cast to float32 for the rest
    # -----------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.float()  # from fp16 to fp32

    # -----------------------------------------------------------------
    # 9. Hidden → output LayerNorm, gating, final linear
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate  # out_gate is float32
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
