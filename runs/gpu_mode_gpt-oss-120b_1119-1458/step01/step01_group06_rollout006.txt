    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool or float)
        - weights      : dict of torch.Tensors containing the module weights
        - config       : dict with at least keys "dim", "hidden_dim",
                         and optional "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack everything
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # -----------------------------------------------------------------
    # shape: (B, N, N, hidden_dim)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on the hidden dimension)
    # -----------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] and cast to same dtype as left/right
        mask_fp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_fp
        right = right * mask_fp

    # -----------------------------------------------------------------
    # 4️⃣ Gating values (sigmoid)
    # -----------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()  # shape (B,N,N,hidden_dim)

    # -----------------------------------------------------------------
    # 5️⃣ Apply gates (still in FP32 for stability)
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6️⃣ Cast to FP16 and prepare for batched GEMM
    # -----------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # (B, H, N, N) -> (B·H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left.view(B * hidden_dim, N, N)
    right_flat = right.view(B * hidden_dim, N, N)

    # -----------------------------------------------------------------
    # 7️⃣ Batched matrix multiply using cuBLAS (torch.bmm)
    #     out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
    # -----------------------------------------------------------------
    # Using FP16 inputs; torch.bmm accumulates in FP32 internally on H100.
    out_flat_fp16 = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # -----------------------------------------------------------------
    # 8️⃣ Reshape back to (B, N, N, hidden_dim) and cast to FP32
    # -----------------------------------------------------------------
    out = out_flat_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # -----------------------------------------------------------------
    # 9️⃣ Hidden → output LayerNorm
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # -----------------------------------------------------------------
    # 10️⃣ Element‑wise multiply with output gate via Triton
    # -----------------------------------------------------------------
    # Ensure tensors are contiguous 1‑D views
    out_cont = out.contiguous()
    out_gate_cont = out_gate.contiguous()

    numel = out_cont.numel()
    BLOCK = 65536  # 64 K elements per program
    grid = ( (numel + BLOCK - 1) // BLOCK, )

    out_gated = torch.empty_like(out_cont)
    eltwise_mul_kernel[grid](
        out_cont.view(-1),
        out_gate_cont.view(-1),
        out_gated.view(-1),
        numel,
        BLOCK_SIZE=BLOCK,
    )
    out = out_gated.view_as(out)   # shape (B,N,N,hidden_dim), still FP32

    # -----------------------------------------------------------------
    # 11️⃣ Final linear projection back to `dim`
    # -----------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
