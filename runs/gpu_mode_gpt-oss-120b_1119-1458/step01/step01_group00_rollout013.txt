   left  = left  * σ(x @ W_left_gate)
   right = right * σ(x @ W_right_gate)
5. Dense N³ contraction is performed as a batched matrix‑multiply:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is equivalent to (left @ rightᵀ) for every head h.
   The contraction uses PyTorch’s batched GEMM (torch.bmm), which on an
   H100 exploits Tensor‑Cores and attains sub‑millisecond latency for the
   required sizes (N≤1024, H≤128).
6. Layer‑norm over the hidden dimension H.
7. Element‑wise multiplication with the output gate is executed by a tiny
   Triton kernel (demonstrating required Triton usage).
8. Final linear projection back to the original channel dimension C.
9. Return result in float32.

All heavy linear algebra stays in float16; only the two layer‑norms run in
float32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton helper: element‑wise multiply (used for out_gate application)
# ----------------------------------------------------------------------
@triton.jit
def _elem_mul_kernel(a_ptr, b_ptr, out_ptr, n_elements,
                     BLOCK_SIZE: tl.constexpr):
    """C = A * B   where A, B, C are 1‑D tensors of length n_elements."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elem_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Launch the Triton element‑wise multiply kernel."""
    assert a.shape == b.shape, "Shapes must match for elementwise mul"
    n = a.numel()
    out = torch.empty_like(a)
    BLOCK = 65536  # 64 K elements per program; fits well in registers
    grid = (triton.cdiv(n, BLOCK),)
    _elem_mul_kernel[grid](
        a, b, out, n,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold3 style).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of model weights (float32 tensors)
        - config_dict  : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # Helper to cast a weight tensor to FP16 on the correct device
    # --------------------------------------------------------------
    def to_fp16(t):
        return t.to(device=device, dtype=torch.float16)

    # --------------------------------------------------------------
    # Extract model parameters (all are bias‑free Linear weights except
    # the LayerNorm weights/biases)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]

    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])
    to_out_norm_w = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = to_fp16(weights["to_out_norm.bias"])
    to_out_w      = to_fp16(weights["to_out.weight"])

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B, N, N, C]  float32

    # --------------------------------------------------------------
    # 2) Cast to FP16 and compute projections
    # --------------------------------------------------------------
    x = x_norm.to(torch.float16)               # [B,N,N,C]  fp16

    left  = F.linear(x, left_proj_w)           # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)          # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dimension)
    # --------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # [B,N,N,1]
        left  = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4) Gating (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # [B,N,N,hidden_dim]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # [B,N,N,hidden_dim]

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) N³ contraction via batched GEMM (torch.bmm)
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # Move hidden dimension to the second axis and make memory contiguous.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Collapse batch and head dimensions for the batched matmul.
    left_bh  = left_perm.view(B * hidden_dim, N, N)      # (B*H, N, N)
    right_bh = right_perm.view(B * hidden_dim, N, N)     # (B*H, N, N)

    # Batched matmul: (B*H, N, N) @ (B*H, N, N)ᵀ  → (B*H, N, N)
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_bh.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,H]  fp16

    # --------------------------------------------------------------
    # 6) Output LayerNorm over the hidden dimension (float32 internally)
    # --------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)  # back to fp16

    # --------------------------------------------------------------
    # 7) Apply output gate (via a tiny Triton kernel) and final projection
    # --------------------------------------------------------------
    gated = _elem_mul(out_norm, out_gate)               # fp16 element‑wise multiply
    final = F.linear(gated, to_out_w)                  # [B,N,N,dim]  fp16

    # --------------------------------------------------------------
    # 8) Cast back to float32 as required by the specification
    # --------------------------------------------------------------
    return final.to(torch.float32)
