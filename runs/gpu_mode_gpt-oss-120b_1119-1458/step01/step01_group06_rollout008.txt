    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_j = offs_j < N

                     + offs_k[None, :] * stride_lk,

"""
TriMul (outgoing) forward pass – Triton accelerated N³ reduction.

The heavy part of the operator is a batched Gram‐product:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden channel `d` (and each batch) we compute a matrix
multiplication `left[d] @ right[d].T`.  The kernel below implements a
batched fp16 GEMM with fp32 accumulation using Tensor‑cores.  The
remaining LayerNorms, linear projections and gating are kept in PyTorch.
The kernel follows a classic 3‑D tiling (i‑tile, j‑tile, k‑tile) and
uses compile‑time block sizes that are well‑suited for the H100.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched Gram matrix (left @ rightᵀ) over the sequence dim.
# Input tensors have shape (M, N, K) where
#   M = batch * hidden_dim,
#   N = seq_len (rows),
#   K = seq_len (cols, summed over).
# The kernel computes
#   out[m, i, j] = Σ_k left[m,i,k] * right[m,k,j]
# ----------------------------------------------------------------------
@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    M, N,                                      # M = BH, N = seq_len
    stride_lm, stride_ln, stride_lk,           # left strides (M, i, k)
    stride_rm, stride_rk, stride_rn,           # right strides (M, k, j) – note the swap
    stride_om, stride_on, stride_ok,           # out strides (M, i, j)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)   # index over (batch * hidden_dim)
    pid_i     = tl.program_id(1)   # tile row index
    pid_j     = tl.program_id(2)   # tile col index

    # --------------------------------------------------------------
    # Tile offsets and masks
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 (stable, Tensor‑core friendly)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension K
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_ptr
            + pid_batch * stride_lm
            + offs_i[:, None] * stride_ln
            + offs_k[None, :] * stride_lk,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )   # dtype = fp16

        # right block (already logically transposed via swapped strides):
        # shape (BLOCK_K, BLOCK_N)
        right = tl.load(
            right_ptr
            + pid_batch * stride_rm
            + offs_k[:, None] * stride_rk
            + offs_j[None, :] * stride_rn,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )   # dtype = fp16

        # Tensor‑core GEMM: fp16 × fp16 → fp32 accumulation
        acc += tl.dot(left, right, out_dtype=tl.float32)

    # --------------------------------------------------------------
    # Write back the result tile
    # --------------------------------------------------------------
    tl.store(
        out_ptr
        + pid_batch * stride_om
        + offs_i[:, None] * stride_on
        + offs_j[None, :] * stride_ok,
        acc,
        mask=mask_i[:, None] & mask_j[None, :]
    )


# ----------------------------------------------------------------------
# Public entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [batch, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] -> [B,N,N,1] -> broadcast to hidden
        m = mask.unsqueeze(-1).to(left.dtype)
        left  = left  * m
        right = right * m

    # --------------------------------------------------------------
    # 4. Gating values (sigmoid)
    # --------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # --------------------------------------------------------------
    # 5. Apply gates
    # --------------------------------------------------------------
    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6. Cast to fp16 and reshape for the Triton kernel
    # --------------------------------------------------------------
    B, N, _, _ = left.shape                        # original batch and seq_len
    left_half  = left.to(torch.float16)
    right_half = right.to(torch.float16)

    # (B, N, N, hidden) -> (B, hidden, N, N)
    left_perm  = left_half.permute(0, 3, 1, 2).contiguous()
    right_perm = right_half.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim                           # batch * hidden dimension
    left_flat  = left_perm.view(BH, N, N)         # (BH, N, N)
    right_flat = right_perm.view(BH, N, N)        # (BH, N, N)

    # Output buffer (fp16, will be cast back to fp32 later)
    out_flat = torch.empty_like(left_flat)

    # --------------------------------------------------------------
    # 7. Compute block sizes dynamically (compile‑time constants)
    # --------------------------------------------------------------
    # Choose block sizes that fit the problem size while keeping
    # occupancy high on H100.  All three are compile‑time constexprs.
    BLOCK_M = 128 if N >= 128 else N
    BLOCK_N = 128 if N >= 128 else N
    BLOCK_K = 64  if N >= 64  else max(1, N // 2)

    # --------------------------------------------------------------
    # 8. Launch the Triton kernel
    # --------------------------------------------------------------
    stride_lm, stride_ln, stride_lk = left_flat.stride()
    stride_rm, stride_rn, stride_rk = right_flat.stride()
    # NOTE: right is accessed as (M, K, N) → swap K and N strides
    stride_rm = stride_rm          # M stride stays the same
    stride_rk = stride_rk          # original K stride (last axis) = 1
    stride_rn = stride_rn          # original N stride (middle axis) = N

    stride_om, stride_on, stride_ok = out_flat.stride()

    grid = (
        BH,                              # batch * hidden
        (N + BLOCK_M - 1) // BLOCK_M,    # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,    # j‑tiles
    )

    triton_tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        BH,
        N,
        stride_lm,
        stride_ln,
        stride_lk,
        stride_rm,
        stride_rk,
        stride_rn,
        stride_om,
        stride_on,
        stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        # Enable aggressive compilation options for H100:
        num_warps=8,          # 8 warps => 256 threads per block (good for fp16)
        num_stages=3,         # double‑buffering depth
    )

    # --------------------------------------------------------------
    # 9. Reshape back to (B, N, N, hidden) and cast to fp32
    # --------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # LayerNorm expects fp32

    # --------------------------------------------------------------
    # 10. Hidden → output LayerNorm, gating and final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
```

**Explanation of key optimisations**

1. **Larger tile sizes** – `BLOCK_M/N = 128` (instead of 64) reduces the total number of program‑ids by a factor ≈ 4, cutting kernel launch overhead.
2. **Tensor‑core GEMM** – The kernel uses `tl.dot(..., out_dtype=tl.float32)`, letting the H100 execute fp16 × fp16 → fp32 Tensor‑core operations.
3. **Minimal memory traffic** – All heavy data (`left`, `right`) are stored and accessed in `float16`. Only the accumulation buffer is `float32`.
4. **Dynamic block sizing** – Tile dimensions are chosen based on the actual sequence length, keeping them ≤ N to avoid unnecessary padding.
5. **Fusion of mask and gating before the kernel** – The mask and gating are applied once in PyTorch, so the kernel only sees the compact tensors it needs to multiply.

The rest of the TriMul pipeline (LayerNorms, linear layers, sigmoid gates) stays in PyTorch, which is cheap compared to the `N³` reduction. This implementation satisfies the required API and should meet the aggressive runtime target on the H100.
