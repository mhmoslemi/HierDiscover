@triton.jit
def tri_mul_outgoing_kernel(left_ptr, right_ptr, out_ptr,
                            N,
                            stride_lh, stride_li, stride_lk,
                            stride_rh, stride_rk, stride_rj,
                            stride_oh, stride_oi, stride_oj,
                            BLOCK_M: tl.constexpr,
                            BLOCK_N: tl.constexpr,
                            BLOCK_K: tl.constexpr,
                            MASK: tl.constexpr):
   pid_head = tl.program_id(0)
   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)

   # Compute start offsets
   i_start = pid_m * BLOCK_M
   j_start = pid_n * BLOCK_N

   offs_i = i_start + tl.arange(0, BLOCK_M)
   offs_j = j_start + tl.arange(0, BLOCK_N)
   mask_i = offs_i < N
   mask_j = offs_j < N

   # Compute pointers for head offset
   left_head = left_ptr + pid_head * stride_lh
   right_head = right_ptr + pid_head * stride_rh
   out_head = out_ptr + pid_head * stride_oh

   # accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   for k in range(0, tl.cdiv(N, BLOCK_K)):
       k_start = k * BLOCK_K
       offs_k = k_start + tl.arange(0, BLOCK_K)
       mask_k = offs_k < N

       # load left tile [M,K]
       left_offsets = left_head + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
       if MASK:
           left_mask = mask_i[:, None] & mask_k[None, :]
           left = tl.load(left_offsets, mask=left_mask, other=0.0).to(tl.float32)
       else:
           left = tl.load(left_offsets).to(tl.float32)

       # load right tile [K,N] (right already transposed)
       right_offsets = right_head + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj
       if MASK:
           right_mask = mask_k[:, None] & mask_j[None, :]
           right = tl.load(right_offsets, mask=right_mask, other=0.0).to(tl.float32)
       else:
           right = tl.load(right_offsets).to(tl.float32)

       acc += tl.dot(left, right)

   # store result
   out_offsets = out_head + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
   if MASK:
       store_mask = mask_i[:, None] & mask_j[None, :]
       tl.store(out_offsets, acc, mask=store_mask)
   else:
       tl.store(out_offsets, acc)
```

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The heavy part of the TriMul module is the tensor contraction

    out[b,i,j,d] = Σₖ left[b,i,k,d] · right[b,j,k,d]

which is a batch of independent matrix multiplications
C = A @ Bᵀ   (A : N×N, B : N×N) for every hidden channel *d*.
For a tensor of shape [B, N, N, hidden] we therefore need *B·hidden*
independent GEMMs of size N×N.

The implementation below

*   casts the normalised input to **FP16** and performs all projections,
    gates and the element‑wise mask in FP16 (the final linear layers stay in FP32).
*   rearranges the right operand so that it is stored as (head, k, j) – i.e.
    the inner dimension *k* is contiguous – which lets the Triton kernel load
    a left tile of shape (M, K) and a right tile of shape (K, N) without an
    extra transpose.
*   launches a custom Triton kernel that tiles the GEMM with
    `BLOCK_M = BLOCK_N = BLOCK_K = 64`.  The kernel keeps the accumulator in
    FP32 while feeding FP16 tiles to `tl.dot(..., out_dtype=tl.float32)`,
    thereby exploiting the H100 tensor‑cores.
*   uses a per‑program (per‑tile) accumulator of only 16 KB (64 × 64 × 4 B) –
    far smaller than the 64 KB used in the reference implementation – which
    raises occupancy and reduces run‑time.
*   handles out‑of‑bounds loads with a simple bounds mask (required for the
    trailing tiles) and relies on the outer mask (if present) being applied
    beforehand in PyTorch.

The rest of the module – LayerNorms, gating and the final projection – is
performed with PyTorch’s highly‑optimised kernels.

The function `custom_kernel` matches the required entry point signature.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_kernel(
    left_ptr,          # fp16   left  = (head, i, k)
    right_ptr,         # fp16   right = (head, k, j)
    out_ptr,           # fp32   output = (head, i, j)
    N,                 # i32    sequence length
    stride_lh, stride_li, stride_lk,   # i64 strides for left   (head, i, k)
    stride_rh, stride_rk, stride_rj,   # i64 strides for right  (head, k, j)
    stride_oh, stride_oi, stride_oj,   # i64 strides for out    (head, i, j)
    BLOCK_M: tl.constexpr,  # tile size in i (rows)
    BLOCK_N: tl.constexpr,  # tile size in j (cols)
    BLOCK_K: tl.constexpr,  # tile size in k (reduction)
):
    """
    Computes for a single head:
        out[i, j] = Σₖ left[i, k] * right[k, j]
    using FP16 tiles and FP32 accumulation.
    """
    pid_h = tl.program_id(0)          # which head  (B * hidden)
    pid_m = tl.program_id(1)          # block row (i)
    pid_n = tl.program_id(2)          # block col (j)

    # ------------------------------------------------------------------
    # Compute the absolute i/j offsets of the current tile
    # ------------------------------------------------------------------
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Pointers to the start of the current head
    # ------------------------------------------------------------------
    left_head_ptr  = left_ptr  + pid_h * stride_lh
    right_head_ptr = right_ptr + pid_h * stride_rh
    out_head_ptr   = out_ptr   + pid_h * stride_oh

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension
    # ------------------------------------------------------------------
    num_blocks_k = tl.cdiv(N, BLOCK_K)
    for kb in range(num_blocks_k):
        k_start = kb * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile [M, K] -------------------------------------------------
        left_offsets = (
            left_head_ptr
            + offs_i[:, None] * stride_li          # i dimension
            + offs_k[None, :] * stride_lk          # k dimension
        )
        left_mask = mask_i[:, None] & mask_k[None, :]
        left_tile = tl.load(left_offsets, mask=left_mask, other=0.0)   # fp16

        # ----- load right tile [K, N] (already transposed) ---------------------------
        right_offsets = (
            right_head_ptr
            + offs_k[:, None] * stride_rk          # k dimension
            + offs_j[None, :] * stride_rj          # j dimension
        )
        right_mask = mask_k[:, None] & mask_j[None, :]
        right_tile = tl.load(right_offsets, mask=right_mask, other=0.0) # fp16

        # ----- mixed‑precision dot (FP16 inputs, FP32 accumulator) -------------------
        acc += tl.dot(left_tile, right_tile, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------
    out_offsets = (
        out_head_ptr
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj
    )
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_offsets, acc, mask=out_mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]    (float32)
        - mask         : [B, N, N]        (broadcastable, float or bool)
        - weights      : dict of model parameters (float32)
        - config       : dict with at least `dim` and `hidden_dim`.

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (float32)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Gate that will be applied later (keep in fp32 for higher precision)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))  # fp32, shape [B,N,N,hidden]

    # ------------------------------------------------------------------
    # Cast the normalised tensor to FP16 for the heavy linear work
    # ------------------------------------------------------------------
    x_half = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (fp16, no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_half, left_proj_w)   # [B,N,N,hidden]  fp16
    right = F.linear(x_half, right_proj_w) # [B,N,N,hidden]  fp16

    # ------------------------------------------------------------------
    # 4️⃣ Optional masking (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣ Gating values (sigmoid, keep fp16)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_half, left_gate_w)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x_half, right_gate_w)).to(torch.float16)

    # ------------------------------------------------------------------
    # 6️⃣ Apply gates (still fp16)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Prepare data for the Triton kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape               # B, seq_len, seq_len, hidden_dim
    total_heads = B * hidden_dim           # each hidden channel is an independent GEMM

    #   left  : (head, i, k)   = permute(0,3,1,2)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    #   right : (head, k, j)   = permute(0,3,2,1)  (k and j swapped)
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(total_heads, N, N)

    #   Output buffer (fp32)
    out_flat = torch.empty((total_heads, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 8️⃣ Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        # strides for left  (head, i, k)
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # strides for right (head, k, j)
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        # strides for out   (head, i, j)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Reshape back to [B, N, N, hidden]
    # ------------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)   # (B, hidden, N, N)
        .permute(0, 2, 3, 1)                # (B, N, N, hidden)
        .contiguous()
    )  # fp32

    # ------------------------------------------------------------------
    # 10️⃣ Post‑processing: LayerNorm → out‑gate → final linear (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                         # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])  # final projection to `dim`

    return out
```
