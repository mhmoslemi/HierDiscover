"""
Optimized Triton‑accelerated forward pass for the “outgoing” TriMul operator.

Algorithm
---------
1. Cast the input to FP16 and apply a fused LayerNorm (torch.nn.functional.layer_norm)
   using FP16 weights/bias.
2. Linear projections (left/right) and three gating MLPs are computed in FP16.
3. When a mask is present we fuse the three element‑wise multiplies
   (proj * mask * gate) with a tiny Triton kernel (`fused_mul3`), thus
   reducing memory traffic and kernel launch overhead.
4. Core contraction:
        out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
   This is exactly a batched matrix multiplication for every hidden channel.
   We reshape the tensors to shape (B*H, N, N) and invoke `torch.bmm`
   (FP16) which is highly tuned on H100 (Tensor‑cores).
5. Apply the output LayerNorm (over hidden_dim), multiply by the output gate,
   and a final FP16 linear projection back to the original dimension.
6. Cast the result back to float32.

All heavy arithmetic stays in FP16, only the final output is upcast.
A single Triton kernel is used to fuse the mask‑gate‑proj multiplication.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused element‑wise product of three tensors (a * b * c)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,  # pointers to contiguous 1‑D tensors
    size,                          # total number of elements (scalar)
    BLOCK_SIZE: tl.constexpr,      # compile‑time constant
):
    """Compute out = a * b * c for three half‑precision tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0).to(tl.float16)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0).to(tl.float16)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0).to(tl.float16)

    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """Convenience wrapper around `fused_mul3_kernel` for 4‑D tensors."""
    assert a.shape == b.shape == c.shape, "All inputs must share the same shape"
    # Ensure contiguity and half‑precision
    a = a.contiguous().to(torch.float16)
    b = b.contiguous().to(torch.float16)
    c = c.contiguous().to(torch.float16)

    out = torch.empty_like(a, dtype=torch.float16)

    total = a.numel()
    BLOCK = 131072  # 128 KB per block -> 2 ⁴ elements, good for H100
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_kernel[grid](
        a, b, c, out, total, BLOCK_SIZE=BLOCK
    )
    return out


# ----------------------------------------------------------------------
# Entry‑point: custom Triton‑accelerated TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float32, CUDA)
        - weights_dict : model weights with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
        - config_dict  : dict containing at least:
                "dim" (int), "hidden_dim" (int),
                optional "eps" (float) and "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # 1. Unpack inputs & convert everything to FP16 (half)
    # --------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # Cast the input to half‑precision once
    x = inp.to(torch.float16)

    # Cast all weight tensors to half‑precision (no bias terms in linear layers)
    w_norm = weights["norm.weight"].to(torch.float16)
    b_norm = weights["norm.bias"].to(torch.float16)

    w_left_proj = weights["left_proj.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)

    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)

    w_to_out = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 2. Layer‑norm on the input (still half‑precision)
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w_norm,
        bias=b_norm,
        eps=eps,
    )  # [B, N, N, dim] (fp16)

    # --------------------------------------------------------------
    # 3. Linear projections (no bias) – result is [B,N,N,hidden]
    # --------------------------------------------------------------
    left = F.linear(x_norm, w_left_proj)   # [B, N, N, hidden]
    right = F.linear(x_norm, w_right_proj) # [B, N, N, hidden]

    # --------------------------------------------------------------
    # 4. Gating tensors (sigmoid of linear layers)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_norm, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x_norm, w_out_gate))

    # --------------------------------------------------------------
    # 5. Apply mask and gates – fused Triton kernel if mask present
    # --------------------------------------------------------------
    if nomask:
        # No mask: simple element‑wise product (still FP16)
        left = left * left_gate
        right = right * right_gate
    else:
        # Broadcast mask to hidden dimension and fuse three multiplies
        mask_h = mask.unsqueeze(-1).to(torch.float16)                     # [B,N,N,1]
        # left = left * mask_h * left_gate   (fused)
        left = fused_mul3(left, mask_h, left_gate)
        # right = right * mask_h * right_gate (fused)
        right = fused_mul3(right, mask_h, right_gate)

    # --------------------------------------------------------------
    # 6. Core contraction: out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    #    Implemented as a batched GEMM (FP16) over the (i,k) and (j,k)
    #    dimensions for each hidden channel.
    # --------------------------------------------------------------
    # Rearrange tensors so that the hidden dimension becomes the batch
    # dimension for torch.bmm.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, hidden, N, N] (k,j order)

    # Collapse batch & hidden into a single batch dimension.
    left_flat = left_perm.view(B * hidden, N, N)    # [B*hidden, N, N]
    right_flat = right_perm.view(B * hidden, N, N) # [B*hidden, N, N]

    # Batched matrix multiplication (uses Tensor‑cores on H100)
    out_flat = torch.bmm(left_flat, right_flat)    # [B*hidden, N, N] (FP16)

    # Reshape back to (B, N, N, hidden)
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7. Output Layer‑norm over hidden dimension (FP16)
    # --------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # [B, N, N, hidden] (fp16)

    # --------------------------------------------------------------
    # 8. Apply the output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                     # element‑wise gate
    out_final = F.linear(out_norm, w_to_out)           # [B, N, N, dim] (fp16)

    # --------------------------------------------------------------
    # 9. Cast back to float32 for downstream consumption
    # --------------------------------------------------------------
    return out_final.to(torch.float32)
