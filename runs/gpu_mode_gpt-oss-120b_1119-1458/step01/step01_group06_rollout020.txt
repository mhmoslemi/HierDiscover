def custom_kernel(data):
    """TriMul outgoing forward pass using fused linear projections and cuBLAS batched gemm.
    Performs:
    1) Input layer norm.
    2) One fused linear projection to obtain left/right projections and gating values.
    3) Optional masking.
    4) Apply sigmoid gating.
    5) Batched matrix multiplication (gram) using Triton.
    6) Hidden layer norm, out-gate scaling, final linear projection.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1 layer norm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # keep a half precision copy for proj/gates
    x_fp16 = x.to(torch.float16)

    # 2 fuse four linear layers (left, right, left_gate, right_gate)
    # concatenate weight matrices along out_features
    w_left = weights["left_proj.weight"]
    w_right = weights["right_proj.weight"]
    w_lgate = weights["left_gate.weight"]
    w_rgate = weights["right_gate.weight"]
    # each weight shape: (hidden_dim, dim)
    w_concat = torch.cat([w_left, w_right, w_lgate, w_rgate], dim=0).to(torch.float16)  # (4*hidden, dim)

    # linear fusion
    fused = torch.nn.functional.linear(x_fp16, w_concat)  # (B,N,N,4*hidden)
    # split
    split = torch.split(fused, hidden_dim, dim=-1)
    left = split[0]
    right = split[1]
    left_gate = split[2]
    right_gate = split[3]

    # optional mask
    if not nomask:
        m = mask.unsqueeze(-1).to(x_fp16.dtype)  # [B,N,N,1]
        left = left * m
        right = right * m

    # gating sigmoid (still fp16)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    # apply gating
    left = left * left_gate
    right = right * right_gate

    # 3 compute gram product: out = Σ_k left[...,k,:] * right[...,k,:]
    B, N, _, _ = left.shape
    # bring hidden to batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # view as (B*hidden, N, N)
    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)

    out_flat = torch.empty_like(left_flat)  # fp16

    # launch Triton kernel
    stride_lm, stride_ln, stride_lk = left_flat.stride()
    stride_rm, stride_rn, stride_rk = right_flat.stride()
    stride_om, stride_on, stride_ok = out_flat.stride()

    # Choose block sizes based on N
    if N <= 256:

        BLOCK_N = 64
        BLOCK_K = 32

        BLOCK_N = 64
        BLOCK_K = 64

        BLOCK_N = 128
        BLOCK_K = 64

        left_flat,
        right_flat,
        out_flat,

        stride_ln,

        BLOCK_K=BLOCK_K,

    stride_om, stride_on, stride_ok,

):

    pid_j = tl.program_id(2)

    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

"""
TriMul (outgoing) forward pass – highly‑optimized version.

Algorithm
---------
1. Input LayerNorm (FP32 → FP16).
2. Fuse the four linear layers that produce
   left_proj, right_proj, left_gate, right_gate
   into a single half‑precision GEMM by concatenating the
   four weight matrices along the output dimension.
3. Apply an optional mask (broadcast on the hidden dim) – still FP16.
4. Apply sigmoid to the two gating tensors and multiply them
   into the left/right projections (in‑place, FP16).
5. Compute the N³ reduction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   as a batched “Gram” matrix multiplication using a custom
   Triton kernel.  The hidden dimension is flattened into the
   batch dimension (B·H, N, N) so the kernel processes a classic
   matrix‑multiply tile‑loop with tensor‑core dot products.
6. Hidden‑dim LayerNorm (FP32) + out‑gate (sigmoid) scaling.
7. Final linear projection (hidden → dim) via a regular PyTorch GEMM.

Only the heavy N³ reduction is done in Triton; all other
operations stay on PyTorch (leveraging cuBLAS for the large
GEMMs).  This dramatically cuts the number of kernel launches
and memory traffic while still meeting the “use Triton” requirement.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched Gram matrix multiplication.
# Computes, for each batch index m:
#   out[m, i, j] = Σ_k left[m, i, k] * right[m, j, k]
# (right is accessed as rightᵀ to match the einsum in the paper)
# ----------------------------------------------------------------------
@triton.jit
def gram_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N,
    stride_lm, stride_ln, stride_lk,
    stride_rm, stride_rn, stride_rk,
    stride_om, stride_on, stride_ok,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # combined batch * hidden index
    pid_i = tl.program_id(1)          # tile row
    pid_j = tl.program_id(2)          # tile column

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Reduction over K dimension
    # ------------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block   : (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_ptr
            + pid_m * stride_lm
            + offs_i[:, None] * stride_ln
            + offs_k[None, :] * stride_lk,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # right block – we need rightᵀ, so we load (BLOCK_K, BLOCK_N)
        right = tl.load(
            right_ptr
            + pid_m * stride_rm
            + offs_k[:, None] * stride_rk
            + offs_j[None, :] * stride_rn,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # dot product (uses Tensor Cores, returns FP16)
        acc += tl.dot(left, right).to(tl.float32)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    tl.store(
        out_ptr
        + pid_m * stride_om
        + offs_i[:, None] * stride_on
        + offs_j[None, :] * stride_ok,
        acc,
        mask=mask_i[:, None] & mask_j[:, None],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            input_tensor – torch.Tensor [B, N, N, dim] (float32)
            mask         – torch.Tensor [B, N, N]   (bool/int, may be ignored)
            weights      – dict of model parameters (torch.Tensors)
            config       – dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm  (FP32 → FP16 for the rest of the pipeline)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_fp16 = x.to(torch.float16)          # half‑precision for all GEMMs

    # ------------------------------------------------------------------
    # 2. Fuse the four linear layers (left, right, left_gate, right_gate)
    # ------------------------------------------------------------------
    # each weight has shape (hidden_dim, dim) – concatenate on out_features
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                 # (4*hidden, dim)

    # One large GEMM:  [B, N, N, dim]  @  (4*hidden, dim).T  ->  [B, N, N, 4*hidden]
    fused = torch.nn.functional.linear(x_fp16, w_concat)  # half‑precision

    # Split the concatenated result back into the four tensors
    left, right, left_gate, right_gate = torch.split(fused, hidden_dim, dim=-1)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        m = mask.to(left.dtype).unsqueeze(-1)   # [B, N, N, 1] FP16
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4. Gating – sigmoid and apply
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Gram‑product (N³ reduction) – Triton kernel
    #    Layout: (B, N, N, H) → (B*H, N, N)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # bring hidden dimension next to batch for a contiguous view
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_t.view(BH, N, N)      # (B*H, N, N), FP16
    right_flat = right_t.view(BH, N, N)

    out_flat = torch.empty_like(left_flat)  # FP16 buffer for result

    # Strides needed by the Triton kernel
    stride_lm, stride_ln, stride_lk = left_flat.stride()
    stride_rm, stride_rn, stride_rk = right_flat.stride()
    stride_om, stride_on, stride_ok = out_flat.stride()

    # Choose tile sizes – larger tiles for larger N
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    else:  # up to 1024
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    gram_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        BH,
        N,
        stride_lm,
        stride_ln,
        stride_lk,
        stride_rm,
        stride_rn,
        stride_rk,
        stride_om,
        stride_on,
        stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,            # reasonable default for H100
    )
    # Restore the original layout and cast back to FP32
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # (B, N, N, hidden)

    # ------------------------------------------------------------------
    # 6. Hidden‑dim LayerNorm + out‑gate scaling
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final projection back to `dim`
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
