def _batched_matmul_transpose(left, right):
    # left, right: [B, N, N, H] dtype bf16
    B,N,_,H = left.shape
    # reshape
    A = left.permute(0,3,1,2).contiguous().view(B*H, N, N)
    B_ = right.permute(0,3,1,2).contiguous().view(B*H, N, N)
    # allocate output in fp32
    C = torch.empty((B*H, N, N), dtype=torch.float32, device=left.device)
    # strides in elements
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # block sizes
    BLOCK_M = 64
    BLOCK_N = 64

"""
TriMul (outgoing) – highly‑optimized forward pass.

Algorithm
---------
1. Layer‑norm over the feature dimension `dim` (FP32 for numerical stability).
2. Cast the normalized tensor to BF16.
3. Four bias‑free linear projections (left, right, left_gate, right_gate,
   out_gate) are performed in BF16 using the model weights (also cast to BF16).
4. Optional mask (broadcasted on the hidden dimension) is applied in BF16.
5. Apply sigmoid to the three gate tensors and multiply element‑wise with the
   projected tensors (`left` and `right`).  All tensors are now BF16.
6. Heavy contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication `C = A @ Bᵀ` for
   `batch*hidden = B*H` independent `N×N` matrices.  The custom Triton
   kernel tiles the matrices (BLOCK_M×BLOCK_N) and accumulates in FP32
   while loading BF16 → Tensor‑Core matmul.
7. Second LayerNorm over the hidden dimension (FP32), multiply by `out_gate`
   (converted to FP32) and linearly project back to `dim`.  The final linear
   is also performed in BF16 and cast back to FP32 for the returned tensor.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   (A,B: BF16, C: FP32)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,  # *bf16  [M, N, K]   (M = batch*hidden)
    B_ptr,  # *bf16  [M, N, K]
    C_ptr,  # *fp32  [M, N, N]
    M, N, K,
    stride_am, stride_ai, stride_ak,   # strides for A
    stride_bm, stride_bj, stride_bk,   # strides for B
    stride_cm, stride_ci, stride_cj,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile along i (first N)
    pid_n = tl.program_id(1)          # tile along j (second N)
    pid_z = tl.program_id(2)          # batch*hidden index

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ----------- load A tile (M‑tile × K‑tile) ---------------
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
            dtype=tl.bfloat16,
        )

        # ----------- load B tile as (K‑tile × N‑tile) ----------
        # note: we load B in transposed order to realise Bᵀ
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
            dtype=tl.bfloat16,
        )

        # Tensor‑core dot (BF16·BF16 → FP32)
        acc += tl.dot(a, b)   # result is FP32

    # --------------------------------------------------------------
    # Store the output tile (FP32)
    # --------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel on many independent matmuls
# ----------------------------------------------------------------------
def _batched_matmul_transpose(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute C[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    left/right: BF16 tensor of shape [B, N, N, H]
    Returns: FP32 tensor of shape [B, N, N, H]
    """
    B, N, _, H = left.shape
    # reshape to [B*H, N, N] (contiguous)
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # output buffer in FP32
    C = torch.empty((B * H, N, N), dtype=torch.float32, device=left.device)

    # strides in elements (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tuned tile sizes for Hopper (H100) – BF16 + Tensor Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                           # batch*hidden
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (int/float, may be ignored)
        - weights_dict : mapping from weight name to torch.Tensor (float32)
        - config_dict  : contains ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device as input)
    """
    # -------------------------------------------------
    # Unpack arguments
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (FP32)
    # -------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim]  (FP32)

    # -------------------------------------------------
    # 2️⃣ Cast to BF16 for the heavy work
    # -------------------------------------------------
    x_bf16 = x.to(torch.bfloat16)   # [B, N, N, dim]

    # -------------------------------------------------
    # 3️⃣ Linear projections (bias‑free) in BF16
    # -------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.bfloat16)   # [hidden, dim]
    right_proj_w = weights["right_proj.weight"].to(torch.bfloat16)

    left = torch.nn.functional.linear(x_bf16, left_proj_w)   # [B, N, N, hidden]
    right = torch.nn.functional.linear(x_bf16, right_proj_w)

    # -------------------------------------------------
    # 4️⃣ Optional mask (broadcasted on hidden dim) – BF16
    # -------------------------------------------------
    if not nomask:
        # mask is 0/1 – convert to BF16 and broadcast
        mask = mask_tensor.unsqueeze(-1).to(torch.bfloat16)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # -------------------------------------------------
    # 5️⃣ Gates (BF16) – sigmoid + element‑wise multiplication
    # -------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.bfloat16)
    right_gate_w = weights["right_gate.weight"].to(torch.bfloat16)
    out_gate_w = weights["out_gate.weight"].to(torch.bfloat16)

    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_bf16, left_gate_w)
    )   # [B, N, N, hidden]  (BF16)
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_bf16, right_gate_w)
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_bf16, out_gate_w)
    )   # [B, N, N, hidden]  (BF16)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton (BF16→FP32)
    # -------------------------------------------------
    # left/right are BF16, the kernel returns FP32
    out = _batched_matmul_transpose(left, right)       # [B, N, N, hidden] (FP32)

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm over hidden dimension (FP32)
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # [B, N, N, hidden] (FP32)

    # -------------------------------------------------
    # 8️⃣ Multiply by out_gate (cast to FP32)
    # -------------------------------------------------
    out = out * out_gate.to(torch.float32)

    # -------------------------------------------------
    # 9️⃣ Final projection back to `dim` – BF16 matmul, cast to FP32
    # -------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.bfloat16)   # [dim, hidden]
    # Cast the tensor to BF16 for the final matmul
    out_bf16 = out.to(torch.bfloat16)
    out = torch.nn.functional.linear(out_bf16, to_out_w)    # BF16 result
    out = out.to(torch.float32)    # final output in FP32

    return out
