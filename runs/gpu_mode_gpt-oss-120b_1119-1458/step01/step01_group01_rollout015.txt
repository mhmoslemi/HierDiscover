    A_ptr, B_ptr, C_ptr,

    stride_bp, stride_bk, stride_bn,

    stride_mask_b, stride_mask_i, stride_mask_j,
    M, N, K,
    hidden_dim, BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Layernorm input
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Proj linear
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)    # B,N,N,H
    right = F.linear(x, right_proj_weight) # B,N,N,H

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))  # keep float32

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # If mask present: apply mask later or incorporate in kernel
    if not config.get("nomask", True):
        # mask: B,N,N bool or 0/1
        mask = mask.to(left.dtype)
        left = left * mask.unsqueeze(-1)
        right = right * mask.unsqueeze(-1)

    # Cast to half
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Triton matmul
    out_fp16 = _triton_matmul(left_fp16, right_fp16,
                seq_len=input_tensor.shape[1],
                hidden_dim=hidden_dim,
                mask=mask if not config.get("nomask", True) else None,
                BLOCK_M=128 if input_tensor.shape[1] >= 512 else 64,
                BLOCK_N=128 if input_tensor.shape[1] >= 512 else 64,
                BLOCK_K=64)

    # Convert back to float32
    out = out_fp16.to(torch.float32)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)

    out = out * out_gate

    # final projection
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)  # uses linear: input H -> output dim

    return out

def _triton_matmul(left: torch.Tensor, right: torch.Tensor,
                   seq_len: int, hidden_dim: int,
                   mask: Optional[torch.Tensor],
                   BLOCK_M=64, BLOCK_N=64, BLOCK_K=64):
    B = left.shape[0]
    # left: [B, N, N, H], right: same

    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()
    P = B * hidden_dim

    left_flat = left_perm.view(P, seq_len, seq_len)
    right_flat = right_perm.view(P, seq_len, seq_len)

    out = torch.empty_like(left_flat)

    # get strides
    stride_ap = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bp = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_cp = out.stride(0)
    stride_cm = out.stride(1)
    stride_cn = out.stride(2)

    if mask is not None:
        # permute mask to (B, N, N) contiguous
        mask_perm = mask.contiguous()
        # mask stride
        stride_mask_b = mask_perm.stride(0)

    # compute left and right proj
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)

    right = right * right_gate

    # mask handling

    if use_mask:
        mask_tensor = mask
        # ensure mask dtype matches left (float16 after gating)
        mask_tensor = mask_tensor.to(torch.float16)
    else:

    right_fp16 = right.to(torch.float16)

"""
Triton‑accelerated outgoing Triangle‑Multiplicative update (TriMul).

The heavy N³ reduction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is evaluated with a custom batched‑matmul kernel that folds the batch
and the hidden dimension (d) into a single program‑id axis.  The kernel
operates on FP16 inputs while accumulating in FP32 for numerical stability.
When a mask is present the kernel also fuses the mask multiplication,
removing two extra memory passes that would otherwise be needed in the
PyTorch reference implementation.

All other linear layers, LayerNorms, sigmoids and the final projection are
performed with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Optional, Tuple, Dict


# ----------------------------------------------------------------------
# Kernel without a mask (full‑precision flow)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_ap, stride_am, stride_ak,
    stride_bp, stride_bk, stride_bn,
    stride_cp, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_p = tl.program_id(2)          # (batch * hidden)
    pid_m = tl.program_id(0)          # block over M (i dimension)
    pid_n = tl.program_id(1)          # block over N (j dimension)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this (batch, hidden) slice
    a_base = A_ptr + pid_p * stride_ap
    b_base = B_ptr + pid_p * stride_bp
    c_base = C_ptr + pid_p * stride_cp

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A tile: (M, K)
        a_ptrs = a_base + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)          # fp16 load

        # B tile: (K, N)
        b_ptrs = b_base + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)          # fp16 load

        # accumulate (fp16 → fp32)
        acc += tl.dot(a, b).to(tl.float32)

    # write back
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# Kernel with a mask (mask is fused into the reduction)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_mask_kernel(
    A_ptr, B_ptr, C_ptr, mask_ptr,
    stride_ap, stride_am, stride_ak,
    stride_bp, stride_bk, stride_bn,
    stride_cp, stride_cm, stride_cn,
    stride_mask_b, stride_mask_i, stride_mask_j,
    M, N, K,
    hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_p = tl.program_id(2)          # (batch * hidden)
    pid_m = tl.program_id(0)          # block over M (i)
    pid_n = tl.program_id(1)          # block over N (j)

    # ---- batch index -------------------------------------------------
    b = pid_p // hidden_dim            # integer division, works on device

    # ---- offsets -----------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---- base pointers ------------------------------------------------
    a_base = A_ptr + pid_p * stride_ap
    b_base = B_ptr + pid_p * stride_bp
    c_base = C_ptr + pid_p * stride_cp
    mask_base = mask_ptr + b * stride_mask_b

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----- left tile (M, K) --------------------------------------
        a_ptrs = a_base + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # left‑mask tile (M, K) → mask[b, i, k]
        mask_a_ptrs = mask_base + offs_m[:, None] * stride_mask_i + cur_k[None, :] * stride_mask_j
        mask_a = tl.load(mask_a_ptrs,
                         mask=mask_m[:, None] & mask_k[None, :],
                         other=0.0)
        a = a * mask_a

        # ----- right tile (K, N) -------------------------------------
        b_ptrs = b_base + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # right‑mask tile (K, N) → mask[b, j, k] (note the swap)
        mask_b_ptrs = mask_base + cur_k[:, None] * stride_mask_j + offs_n[None, :] * stride_mask_i
        mask_b = tl.load(mask_b_ptrs,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)
        b = b * mask_b

        acc += tl.dot(a, b).to(tl.float32)

    # ---- store -------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the appropriate Triton kernel
# ----------------------------------------------------------------------
def _triton_matmul(
    left: torch.Tensor,
    right: torch.Tensor,
    seq_len: int,
    hidden_dim: int,
    mask: Optional[torch.Tensor] = None,
    BLOCK_M: int = 64,
    BLOCK_N: int = 64,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Batched matrix‑multiply reduction for the outgoing TriMul.
    left/right : [B, N, N, H] (fp16)
    mask       : optional [B, N, N] (fp16) – broadcasted on the hidden dim.
    Returns out : [B, N, N, H] (fp16)
    """
    B = left.shape[0]

    # (B, N, N, H) → (B, H, N, N)  (left)  and (B, H, N, N) (right, transposed)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    P = B * hidden_dim                     # total independent matmuls
    left_flat = left_perm.view(P, seq_len, seq_len)
    right_flat = right_perm.view(P, seq_len, seq_len)

    out = torch.empty_like(left_flat)

    # strides for the three tensors (all are contiguous)
    stride_ap = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bp = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_cp = out.stride(0)
    stride_cm = out.stride(1)
    stride_cn = out.stride(2)

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,     # blocks over M
        (seq_len + BLOCK_N - 1) // BLOCK_N,     # blocks over N
        P,                                      # one program per (batch, hidden)
    )

    if mask is None:
        _batched_matmul_kernel[grid](
            left_flat,
            right_flat,
            out,
            stride_ap,
            stride_am,
            stride_ak,
            stride_bp,
            stride_bk,
            stride_bn,
            stride_cp,
            stride_cm,
            stride_cn,
            seq_len,
            seq_len,
            seq_len,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=8,
            num_stages=3,
        )
    else:
        # mask must be contiguous and on the same device
        mask_contig = mask.contiguous()
        stride_mask_b = mask_contig.stride(0)
        stride_mask_i = mask_contig.stride(1)
        stride_mask_j = mask_contig.stride(2)
        _batched_matmul_mask_kernel[grid](
            left_flat,
            right_flat,
            out,
            mask_contig,
            stride_ap,
            stride_am,
            stride_ak,
            stride_bp,
            stride_bk,
            stride_bn,
            stride_cp,
            stride_cm,
            stride_cn,
            stride_mask_b,
            stride_mask_i,
            stride_mask_j,
            seq_len,
            seq_len,
            seq_len,
            hidden_dim,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=8,
            num_stages=3,
        )

    # reshape back to original layout (B, N, N, H)
    out = out.view(B, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation with Triton‑accelerated
    N³ reduction.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim]  (float32)
            - mask:          [B, N, N]       (bool / 0‑1) – may be None when `nomask` is True
            - weights: dict of parameters (see code below)
            - config:  {"dim": int, "hidden_dim": int, "nomask": bool, ...}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]                # model input & output dimension
    hidden_dim = config["hidden_dim"]  # internal hidden size
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid of a linear transformation)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # float32

    # Apply hidden‑dim gates before the heavy reduction
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4) Optional mask handling – we *do not* multiply here if a mask is
    #    present, because the mask will be fused inside the Triton kernel.
    # ------------------------------------------------------------------
    use_mask = not config.get("nomask", True)
    mask_tensor = None
    if use_mask:
        # Ensure mask is on the same device and cast to fp16 for the kernel.
        mask_tensor = mask.to(torch.float16)

    # ------------------------------------------------------------------
    # 5) Cast to FP16 and run the Triton kernel
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Pick block sizes that work well for the typical sequence lengths.
    # The H100 prefers multiples of 32; we use 128 for larger N.
    seq_len = input_tensor.shape[1]
    BLOCK = 128 if seq_len > 256 else 64

    out_fp16 = _triton_matmul(
        left_fp16,
        right_fp16,
        seq_len=seq_len,
        hidden_dim=hidden_dim,
        mask=mask_tensor,
        BLOCK_M=BLOCK,
        BLOCK_N=BLOCK,
        BLOCK_K=64,
    )  # [B, N, N, hidden] (fp16)

    # ------------------------------------------------------------------
    # 6) Convert back to FP32 for the remaining (lighter) ops
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 7) LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 8) Output gate (broadcasted over the hidden dim)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
