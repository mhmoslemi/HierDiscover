"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The heavy N³ reduction
    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
is realized as a batched matrix multiplication where, for every
(batch, hidden) pair, we compute  left @ rightᵀ  (both N×N matrices).
The reduction is performed in a custom Triton kernel that works in
fp16 (to lower memory traffic) while accumulating in fp32 for
numerical stability.  All surrounding operations (LayerNorms,
Linear projections, sigmoid gates and optional masking) are executed
with PyTorch.  The final output is returned in float32.

Supported configurations:
    B ∈ {1,2}, N ∈ {128,256,512,768,1024},
    dim ∈ {128,384}, hidden_dim = 128 (or any multiple of 8)
Two mask modes are available (nomask=True ⇒ ignore mask).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched per‑channel matrix multiplication
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                     # pointers
    stride_ap, stride_am, stride_ak,          # strides for A: (P, M, K)
    stride_bp, stride_bk, stride_bn,          # strides for B: (P, K, N)
    stride_cp, stride_cm, stride_cn,          # strides for C: (P, M, N)
    M, N, K,                                  # matrix sizes (all = seq_len)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute, for each p in 0..P-1:
        C[p, :, :] = A[p, :, :] @ B[p, :, :].T
    where A, B, C are (M,N) matrices stored in fp16 and the
    accumulation is performed in fp32.
    """
    pid_m = tl.program_id(0)      # block row index (i)
    pid_n = tl.program_id(1)      # block col index (j)
    pid_p = tl.program_id(2)      # batch * hidden index

    # --------------------------------------------------------------
    # Offsets for the current block
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Base pointers for this (batch, hidden) slice
    # ------------------------------------------------------------------
    a_base = A_ptr + pid_p * stride_ap
    b_base = B_ptr + pid_p * stride_bp
    c_base = C_ptr + pid_p * stride_cp

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the K dimension in tiles
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        start_k = k * BLOCK_K
        offs_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (M, K)
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)          # fp16 load

        # B tile: (K, N)
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)          # fp16 load

        # Multiply‑accumulate (fp16 → fp32)
        acc += tl.dot(a, b).to(tl.float32)

    # --------------------------------------------------------------
    # Store the result (convert back to fp16 for compactness)
    # --------------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel and reshapes tensors
# ----------------------------------------------------------------------
def _triton_matmul(left: torch.Tensor,
                   right: torch.Tensor,
                   seq_len: int,
                   hidden_dim: int,
                   BLOCK_M: int = 64,
                   BLOCK_N: int = 64,
                   BLOCK_K: int = 64) -> torch.Tensor:
    """
    Batched matrix multiplication for the TriMul reduction.

    Args:
        left:  [B, N, N, H]  (fp16)
        right: [B, N, N, H]  (fp16)
        seq_len: N
        hidden_dim: H
    Returns:
        out: [B, N, N, H] (fp16)
    """
    B = left.shape[0]

    # (B, N, N, H) -> (B, H, N, N)  (contiguous)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right needs a transpose on the two N axes before flattening
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # (B, H, K, J)

    P = B * hidden_dim                     # total independent matmuls
    left_flat = left_perm.view(P, seq_len, seq_len)     # (P, M, K)
    right_flat = right_perm.view(P, seq_len, seq_len)   # (P, K, N)

    out = torch.empty_like(left_flat)                     # (P, M, N)

    # Strides for the flattened tensors (contiguous layout)
    stride_ap = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bp = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_cp = out.stride(0)
    stride_cm = out.stride(1)
    stride_cn = out.stride(2)

    # Grid: (blocks over M, blocks over N, P)
    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,
        (seq_len + BLOCK_N - 1) // BLOCK_N,
        P,
    )

    _batched_matmul_kernel[grid](
        left_flat, right_flat, out,
        stride_ap, stride_am, stride_ak,
        stride_bp, stride_bk, stride_bn,
        stride_cp, stride_cm, stride_cn,
        seq_len, seq_len, seq_len,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # Reshape back to (B, N, N, H)
    out = out.view(B, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------

    """
    Forward pass of the outgoing TriMul operator with a Triton‑accelerated
    N³ reduction.

    Args:
        data (tuple):
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (bool or float)
            - weights      : dict of model parameters (see docstring)
            - config       : dict containing:
                * dim          (int)          – input/output channel dim
                * hidden_dim   (int)          – internal hidden dim
                * nomask       (bool)         – True ⇒ ignore mask
                * eps          (float, opt.) – LayerNorm epsilon

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → left / right (float32)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking (applied after projection, before gating)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B, N, N] → [B, N, N, 1]
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4️⃣ Gates: sigmoid(Linear(x))
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(left.dtype)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(right.dtype)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # keep fp32

    # Apply gates (still float32)
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Cast to fp16 for the N³ reduction
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 6️⃣ Triton kernel: out = Σₖ left[..., i, k, h] * right[..., j, k, h]
    # ------------------------------------------------------------------
    seq_len = input_tensor.shape[1]  # N
    out_fp16 = _triton_matmul(
        left_fp16,
        right_fp16,
        seq_len=seq_len,
        hidden_dim=hidden_dim,
        BLOCK_M=64,
        BLOCK_N=64,
        BLOCK_K=64,
    )   # [B, N, N, hidden] (fp16)

    # Back to float32 for the remaining ops
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 7️⃣ to_out_norm (LayerNorm over hidden dimension)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate (broadcast over the hidden dim)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to dim (no bias)
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
```
