import triton
import triton.language as tl
import torch
from torch import nn
from typing import Tuple, Dict, Any

@triton.jit
def _trimul_matmul_fp16_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # i tile
    pid_n = tl.program_id(1)   # j tile
    pid_z = tl.program_id(2)   # batch*hidden

    # offset in rows and cols
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(K, BLOCK_K)

    for k_iter in range(num_k):
        cur_k = k_iter * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # load A tile (M_i, K)
        a_ptrs = A_ptr + pid_z * stride_am + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=tl.float16(0.0))

        # load B tile (K, N) - note B is stored as (batch*hidden, N, K)
        b_ptrs = B_ptr + pid_z * stride_bm + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=tl.float16(0.0))

        # dot product (fp16 inputs, fp32 accumulation)
        acc += tl.dot(a, b)

    # store result
    c_ptrs = C_ptr + pid_z * stride_cm + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h] for fp16 left/right.
    left/right shape: [B,N,N,H] (fp16).
    Returns: torch.float32 tensor of same shape.

    stride_bm, stride_bj, stride_bk = B_.stride()

"""
Outgoing TriMul (AlphaFold3) – Triton‐accelerated implementation.

Algorithm (FP32 ↔ FP16)
-----------------------
1. Layer‑norm the input tensor over the feature dimension (`dim`).
2. Linear projections (`left_proj`, `right_proj`) → shape [B,N,N,hidden].
3. Optional mask (broadcasted on the hidden dimension) is applied.
4. Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`) from the
   normalized tensor and multiply them elementwise onto the projections.
5. Cast the gated projections to **FP16** to halve memory traffic.
6. Custom Triton kernel performs the heavy O(N³) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication `C = A @ Bᵀ` where each batch‑hidden
   pair is a separate GEMM of size N×N with **Tensor‑core** (FP16 → FP32)
   accumulation.
7. Second Layer‑norm over `hidden_dim`, multiply by `out_gate`,
   and a final linear projection back to `dim`.

The kernel processes tiles of 128×128 with a K‑tile of 64, uses 8 warps,
and accumulates in FP32 for numerical safety.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any
import torch.nn.functional as F


@triton.jit
def _trimul_matmul_fp16_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # sizes (M = batch*hidden, N = seq_len, K = seq_len)
    stride_am, stride_ai, stride_ak,   # A strides
    stride_bm, stride_bj, stride_bk,   # B strides (B stored as (batch*hidden, N, K))
    stride_cm, stride_ci, stride_cj,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """C[pid_z, i, j] = Σ_k A[pid_z, i, k] * B[pid_z, j, k]   (FP16 inputs, FP32 acc)."""
    pid_i = tl.program_id(0)  # tile over i (first N dimension)
    pid_j = tl.program_id(1)  # tile over j (second N dimension)
    pid_z = tl.program_id(2)  # batch * hidden index

    # ----- coordinate offsets within the current tile -----
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over the reduction axis k -----
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for kk in range(num_k_tiles):
        cur_k = kk * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & (offs_k[None, :] < K),
            other=tl.float16(0.0),
        )

        # Load B tile as (BLOCK_K, BLOCK_N) – note B is logically transposed
        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_j[None, :],
            other=tl.float16(0.0),
        )

        # Tensor‑core dot: FP16 × FP16 → FP32 accumulation
        acc += tl.dot(a, b)

    # ----- write back the result tile -----
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def _batched_matmul_transpose_fp16(
    left: torch.Tensor, right: torch.Tensor
) -> torch.Tensor:
    """
    Batched GEMM for TriMul:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    Arguments
    ---------
    left, right : torch.Tensor
        FP16 tensors of shape [B, N, N, H].

    Returns
    -------
    torch.Tensor
        FP32 tensor of shape [B, N, N, H].
    """
    B, N, _, H = left.shape
    # Merge batch and hidden dimensions for a clean 3‑D view
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)       # [B*H, N, N]
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)      # [B*H, N, N]

    # Output buffer (FP32)
    C = torch.empty_like(A, dtype=torch.float32)

    # Strides in elements (not bytes)
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tile sizes – tuned for H100 Tensor‑core performance
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden
    )

    _trimul_matmul_fp16_kernel[grid](
        A,
        B_,
        C,
        B * H,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,   # enough warps to keep the tensor‑cores busy
    )

    # Reshape back to the original layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool), may be ignored
        - weights_dict : mapping from weight name to torch.Parameter / tensor
        - config_dict  : contains keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack arguments
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (dim)
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape [B, N, N, dim], dtype float32

    # -------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # -------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # -------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -------------------------------------------------
    # 4️⃣ Gating (sigmoid) and application
    # -------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5️⃣ Cast to FP16 before the heavy contraction
    # -------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # -------------------------------------------------
    # 6️⃣ Heavy O(N³) contraction via Triton (batched GEMM)
    # -------------------------------------------------
    out = _batched_matmul_transpose_fp16(left_fp16, right_fp16)   # [B,N,N,hidden], FP32

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm over hidden dimension
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # -------------------------------------------------
    # 8️⃣ Gate and final projection back to `dim`
    # -------------------------------------------------
    out = out * out_gate                     # elementwise gating
    out = F.linear(out, weights["to_out.weight"])   # final linear, no bias

    return out
