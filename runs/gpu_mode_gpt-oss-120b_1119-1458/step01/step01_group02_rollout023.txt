"""
TriMul (outgoing) Triton‑accelerated forward pass.

1. Layer‑norm over the feature dimension (float32).
2. Half‑precision linear projections + optional mask.
3. Sigmoid gates (float32 → half for gating).
4. Batched half‑precision matrix multiplication left @ rightᵀ
   with a custom Triton kernel (accumulate in FP32).
5. Float32 LayerNorm over the hidden dimension.
6. Apply output gate and final half‑precision projection back to `dim`.
   The final tensor is returned as float32.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ for many (batch*hidden) matrices
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,                # *fp16  [M, N, K]
    B_ptr,                # *fp16  [M, N, K]
    C_ptr,                # *fp32  [M, N, N]
    M, N, K,              # i32 sizes
    stride_am, stride_ai, stride_ak,   # strides for A
    stride_bm, stride_bj, stride_bk,   # strides for B
    stride_cm, stride_ci, stride_cj,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)            # tile row (i)
    pid_n = tl.program_id(1)            # tile col (j)
    pid_z = tl.program_id(2)            # index over batch*hidden

    # -------------------------------------------------
    # Offsets for the i‑ and j‑tiles
    # -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # Pointers to the start of the current batch*hidden slice
    # -------------------------------------------------
    a_ptrs = A_ptr + pid_z * stride_am + offs_m[:, None] * stride_ai
    b_ptrs = B_ptr + pid_z * stride_bm + offs_n[None, :] * stride_bj

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        mask_k = offs_k < K

        # load A tile (BLOCK_M x BLOCK_K)
        a = tl.load(a_ptrs + offs_k[None, :] * stride_ak,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # load B tile (BLOCK_K x BLOCK_N) – note the transposed view
        b = tl.load(b_ptrs + offs_k[:, None] * stride_bk,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # half‑precision matmul, accumulation in FP32
        acc += tl.dot(a, b)

    # -------------------------------------------------
    # Write the result tile
    # -------------------------------------------------
    c_ptrs = C_ptr + pid_z * stride_cm + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper that reshapes tensors and launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute C[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using the Triton kernel defined above.
    Input tensors are expected to be half‑precision tensors of shape
    [B, N, N, H] (contiguous). Output tensor is float32 with the same shape.
    """
    B, N, _, H = left.shape

    # Move hidden dimension next to batch and make contiguous
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Allocate output (float32)
    C = torch.empty((B * H, N, N), dtype=torch.float32, device=left.device)

    # Strides in elements
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tile sizes – tuned for H100 and fp16 arithmetic
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=4,
    )

    # Reshape back to [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C

# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------

                               Dict[str, torch.Tensor],

    """

    """

    # -------------------------------------------------

        input_tensor,

        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # -------------------------------------------------
    # 2️⃣ Cast to half‑precision for the heavy linear work
    # -------------------------------------------------
    x_fp16 = x.to(torch.float16)

    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    # -------------------------------------------------
    left = torch.nn.functional.linear(x_fp16, left_proj_w)   # [B,N,N,hidden] fp16

    # 4️⃣ Optional mask (broadcasted on hidden dimension)
    # -------------------------------------------------

    # -------------------------------------------------

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The operation works on a 4‑D tensor X ∈ ℝ^{B×N×N×C} and proceeds as follows:

1️⃣ Layer‑norm over the feature dimension C (float32).
2️⃣ Half‑precision linear projections (left, right) with optional mask.
3️⃣ Three sigmoid gates (left, right, out) computed in float32,
   the first two are cast back to fp16 and applied element‑wise.
4️⃣ Heavy O(N³) contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched matrix multiplication left @ rightᵀ
   using a custom Triton kernel (fp16 operands, FP32 accumulator).
5️⃣ Second LayerNorm over the hidden dimension H (float32).
6️⃣ Multiply by the out‑gate (float32) and a final half‑precision linear
   projection back to the original feature dimension C.
   The result is returned as float32.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched C = A @ Bᵀ  (A,B: fp16, C: fp32)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    A_ptr,                # *fp16  [M, N, K]
    B_ptr,                # *fp16  [M, N, K]
    C_ptr,                # *fp32  [M, N, N]
    M, N, K,              # i32 sizes
    stride_am, stride_ai, stride_ak,   # strides for A
    stride_bm, stride_bj, stride_bk,   # strides for B
    stride_cm, stride_ci, stride_cj,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)            # tile row (i)
    pid_n = tl.program_id(1)            # tile col (j)
    pid_z = tl.program_id(2)            # batch*hidden index

    # -------------------------------------------------
    # Offsets for the current i‑/j‑tile
    # -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # -------------------------------------------------
    # Accumulator (FP32) for the tile
    # -------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # Base pointers for the current batch*hidden slice
    # -------------------------------------------------
    a_base = A_ptr + pid_z * stride_am + offs_m[:, None] * stride_ai
    b_base = B_ptr + pid_z * stride_bm + offs_n[None, :] * stride_bj

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (BLOCK_M × BLOCK_K)
        a = tl.load(
            a_base + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load B tile as (BLOCK_K × BLOCK_N) – transposed view
        b = tl.load(
            b_base + offs_k[:, None] * stride_bk,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # half‑precision matmul, accumulation in FP32
        acc += tl.dot(a, b)

    # -------------------------------------------------
    # Write back the result tile
    # -------------------------------------------------
    c_ptrs = C_ptr + pid_z * stride_cm + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that reshapes tensors and launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using the Triton kernel above.
    Input tensors must be half‑precision tensors of shape [B, N, N, H]
    (contiguous). Output is a float‑32 tensor of the same shape.
    """
    B, N, _, H = left.shape

    # (B, H, N, N) → (B*H, N, N) ; keep contiguous for efficient loads
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (FP32)
    C = torch.empty((B * H, N, N), dtype=torch.float32, device=left.device)

    # Strides in elements
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tuned tile sizes for H100 fp16 matmul
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch*hidden
    )

    _trimul_matmul_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=4,
    )

    # Reshape back to the original layout
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradient computation).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (float32, may be ignored)
        - weights_dict : mapping from weight name to torch.Parameter (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``;
                         optional ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # -------------------------------------------------
    # Unpack
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (float32)
    # -------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]   float32

    # -------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # -------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # Cast only the weights used in half‑precision
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Linear projections (no bias) – fp16
    # -------------------------------------------------
    left  = torch.nn.functional.linear(x_fp16, left_proj_w)   # [B,N,N,hidden] fp16
    right = torch.nn.functional.linear(x_fp16, right_proj_w)

    # -------------------------------------------------
    # 4️⃣ Optional mask (broadcasted on hidden dimension)
    # -------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(torch.float16)   # [B,N,N,1] fp16
        left  = left * mask
        right = right * mask

    # -------------------------------------------------
    # 5️⃣ Gating (sigmoid) – computed in float32 then cast to fp16
    # -------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate.to(torch.float16)
    right = right * right_gate.to(torch.float16)

    # -------------------------------------------------
    # 6️⃣ O(N³) contraction via Triton (fp16 → fp32)
    # -------------------------------------------------
    out = _batched_matmul_transpose(left, right)          # [B,N,N,hidden] fp32

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm over hidden dimension (float32)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # -------------------------------------------------
    # 8️⃣ Output gating and final projection back to `dim`
    # -------------------------------------------------
    out = out * out_gate                                   # float32 * float32

    # Cast to fp16 for the final linear layer
    out_fp16 = out.to(torch.float16)
    out_fp16 = torch.nn.functional.linear(out_fp16, to_out_w)  # fp16 result
    out = out_fp16.to(torch.float32)                       # back to float32

    return out
