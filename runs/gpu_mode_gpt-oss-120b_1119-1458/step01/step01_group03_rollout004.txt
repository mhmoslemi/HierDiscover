"""
Triton‑accelerated outgoing TriMul (AlphaFold3).

The expensive part of TriMul is the three‑dimensional contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which can be rewritten as a batched matrix multiplication for each
hidden channel d:
    out_h = L_h @ R_hᵀ        with L_h,R_h ∈ ℝ^{N×N}.

We treat the hidden dimension as an extra batch dimension (BH = B·hidden)
and launch a 3‑D Triton grid:
    program_id(0) → BH (batch × hidden)
    program_id(1) → tile along the M‑axis (i)
    program_id(2) → tile along the N‑axis (j)

Each program loads a (BLOCK_M×BLOCK_K) tile of L and a (BLOCK_K×BLOCK_N)
tile of Rᵀ, performs an FP16 × FP16 → FP32 dot‑product using
Tensor‑Core‑friendly `tl.dot`, accumulates in FP32, and finally stores the
result.  The surrounding LayerNorms, linear projections and gating are
performed with PyTorch (they are cheap compared to the O(N³) GEMM).

The kernel works for any batch size (1 or 2), sequence length up to 1024,
and hidden dimension 128.  Input/weights are expected on the same CUDA
device as the kernel launch.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_gemm_kernel(
    L_ptr,               # half * (BH, N, N)
    R_ptr,               # half * (BH, N, N)
    O_ptr,               # float * (BH, N, N)
    N,                   # sequence length (runtime)
    # strides for L
    stride_bl, stride_rl, stride_cl,
    # strides for R
    stride_br, stride_rr, stride_cr,
    # strides for O
    stride_bo, stride_ro, stride_co,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: out = L @ Rᵀ  for each hidden slice.
    L,R are (BH, N, N) in row‑major order (i, k) and (j, k) respectively.
    """
    batch_idx = tl.program_id(0)          # BH = B * hidden_dim
    pid_m = tl.program_id(1)              # tile row index (i)
    pid_n = tl.program_id(2)              # tile col index (j)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to the current batch slice
    L_batch = L_ptr + batch_idx * stride_bl
    R_batch = R_ptr + batch_idx * stride_br
    O_batch = O_ptr + batch_idx * stride_bo

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- Load L tile (M x K) -----
        # L layout: (i, k) → row stride = stride_rl, col stride = stride_cl
        L_ptrs = L_batch + offs_m[:, None] * stride_rl + offs_k[None, :] * stride_cl
        L = tl.load(
            L_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- Load R tile transposed (K x N) -----
        # Original R layout: (j, k). We need Rᵀ[k, j] → row = k, col = j
        # row stride = stride_rr, col stride = stride_cr
        R_ptrs = R_batch + offs_k[:, None] * stride_rr + offs_n[None, :] * stride_cr
        R = tl.load(
            R_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- Compute partial dot product -----
        acc += tl.dot(L, R)   # (BLOCK_M, BLOCK_K) • (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # ----- Write out the result -----
    O_ptrs = O_batch + offs_m[:, None] * stride_ro + offs_n[None, :] * stride_co
    tl.store(O_ptrs, acc, mask=mask_m[:, None] & mask_n[:, None])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton‑fused GEMM.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) LayerNorm on the channel dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid on linear projections of x)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to FP16 for the O(N³) GEMM
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape                     # [B, N, N, hidden]

    # Bring hidden dimension forward so we can treat (B * hidden) as a batch axis
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)                # [BH, N, N] (fp16)
    right_flat = right_perm.view(BH, N, N)              # [BH, N, N] (fp16)

    # Output buffer: accumulation in FP32 (as required by the reference)
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 6) Triton batched GEMM launch
    # ------------------------------------------------------------------
    # Tile sizes – chosen to align with Tensor‑Core 16×16×16 folds.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (BH, num_tiles_m, num_tiles_n)

    # Strides for the flattened tensors (row‑major)
    stride_bl, stride_rl, stride_cl = left_flat.stride()
    stride_br, stride_rr, stride_cr = right_flat.stride()
    stride_bo, stride_ro, stride_co = out_flat.stride()

    trimul_outgoing_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        # strides for L
        stride_bl,
        stride_rl,
        stride_cl,
        # strides for R
        stride_br,
        stride_rr,
        stride_cr,
        # strides for O
        stride_bo,
        stride_ro,
        stride_co,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        num_warps=8,                     # H100 benefits from 8‑warp blocks
    )

    # ------------------------------------------------------------------
    # 7) Reshape back to [B, N, N, hidden]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 8) Output LayerNorm (hidden dimension)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 9) Output gating
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10) Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
