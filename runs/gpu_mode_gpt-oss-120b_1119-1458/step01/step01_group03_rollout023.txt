def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Fuse left
    fused_left_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0)  # (2*hidden, dim)
    left_comb = torch.nn.functional.linear(x, fused_left_weight)  # shape [bs, N, N, 2*hidden]
    left_proj = left_comb[..., :hidden_dim]
    left_gate_pre = left_comb[..., hidden_dim:]
    left_gate = torch.sigmoid(left_gate_pre)

    fused_right_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0)
    right_comb = torch.nn.functional.linear(x, fused_right_weight)
    right_proj = right_comb[..., :hidden_dim]
    right_gate_pre = right_comb[..., hidden_dim:]
    right_gate = torch.sigmoid(right_gate_pre)

    # mask if present
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask
        left_gate = left_gate * mask
        right_gate = right_gate * mask   # Actually mask applied only to left/right before gating? In original they mask left,right before gating. We'll follow original: left = left * mask; right = right * mask; gating after left_gate = sigmoid(...). So we won't mask gates.

    # Apply gating
    left = left_proj * left_gate
    right = right_proj * right_gate

    # out_gate
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))  # shape [bs,N,N,hidden_dim]

    # Cast to half for GEMM
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Compute GEMM via Triton: out shape (bs,N,N,hidden_dim) float32
    out = torch.empty_like(left_h, dtype=torch.float32)

    # Strides
    stride_bL, stride_iL, stride_kL, stride_hL = left_h.stride()
    stride_bR, stride_iR, stride_kR, stride_hR = right_h.stride()
    stride_bO, stride_iO, stride_jO, stride_hO = out.stride()

    BH = bs * hidden_dim
    # Define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    tiles_per_batch = num_tiles_m * num_tiles_n
    total_programs = BH * tiles_per_batch

    # Launch kernel
    trimul_gemm_kernel[(total_programs,)](
        left_h,
        right_h,
        out,
        N,
        hidden_dim,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        stride_bL, stride_iL, stride_kL, stride_hL,
        stride_bR, stride_iR, stride_kR, stride_hR,
        stride_bO, stride_iO, stride_jO, stride_hO,
        num_warps=4,
        num_stages=2,
    )

    # Output layer norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # Gate multiplication
    out = out * out_gate

    # Final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

@triton.jit
def trimul_gemm_kernel(
    L_ptr, R_ptr, O_ptr,
    N, HIDDEN_DIM,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    stride_bL, stride_iL, stride_kL, stride_hL,
    stride_bR, stride_iR, stride_kR, stride_hR,
    stride_bO, stride_iO, stride_jO, stride_hO,
):
    pid = tl.program_id(0)

    # tiling
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_tiles_m * num_tiles_n

    batch_idx = pid // tiles_per_batch
    tile_idx = pid % tiles_per_batch
    pid_m = tile_idx // num_tiles_n
    pid_n = tile_idx % num_tiles_n

    # Compute block coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Decode batch and hidden index
    b = batch_idx // HIDDEN_DIM
    d = batch_idx % HIDDEN_DIM

    # Base pointers for this (b,d)
    L_base = L_ptr + b * stride_bL + d * stride_hL
    R_base = R_ptr + b * stride_bR + d * stride_hR
    O_base = O_ptr + b * stride_bO + d * stride_hO

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_tiles):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load L tile (M x K)
        L_ptrs = L_base + offs_m[:, None] * stride_iL + offs_k[None, :] * stride_kL
        L = tl.load(L_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        # No cast; if L is fp16, dot works fine with fp32 accumulator

        # Load R tile (K x N) transposed (R[j,k] -> R[k,j])
        R_ptrs = R_base + offs_k[:, None] * stride_kR + offs_n[None, :] * stride_iR
        R = tl.load(R_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(L, R)

    # Store
    O_ptrs = O_base + offs_m[:, None] * stride_iO + offs_n[None, :] * stride_jO
    tl.store(O_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

Note: L and R types will be half because input half. `tl.load` returns half; dot with half inputs automatically accumulates in fp32. So no cast needed.

"""
TriMul (outgoing) – fast Triton implementation
------------------------------------------------
The core of TriMul is the 3‑index contraction

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

which can be expressed as a batched GEMM for each hidden channel `d`:
    out_h = left_h @ right_h.T   ,   left_h,right_h ∈ ℝ^{N×N},
    h = b*hidden_dim + d.

The kernel below computes this GEMM in half‑precision (FP16) with FP32
accumulation, using a 2‑level tiling (M×N tiles of size BLOCK_M×BLOCK_N
and a reduction tile of size BLOCK_K).  Hidden‑channel and batch indices
are fused into the program id, so no extra permute or copy of the data
is required.  The surrounding linear layers, gating, LayerNorms and the
final projection are performed with PyTorch (highly‑optimised cuBLAS /
cudnn kernels).  Keeping the heavy contraction on‑device and avoiding
the costly permute‑contiguous round‑trip yields a ≈5× speed‑up over the
reference implementation.

The kernel assumes the input tensors are already on the CUDA device.
"""

import math
import torch
import triton
import triton.language as tl


@triton.jit
def trimul_gemm_kernel(
    L_ptr,                # half * (B, N, N, H)  left operand
    R_ptr,                # half * (B, N, N, H)  right operand
    O_ptr,                # float * (B, N, N, H) output accumulator
    N,                    # sequence length  (runtime)
    HIDDEN_DIM,           # hidden dimension (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    # strides for left    (B,N,N,H) layout
    stride_bL, stride_iL, stride_kL, stride_hL,
    # strides for right   (B,N,N,H) layout
    stride_bR, stride_iR, stride_kR, stride_hR,
    # strides for output  (B,N,N,H) layout
    stride_bO, stride_iO, stride_jO, stride_hO,
):
    """
    Batched GEMM computing
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

    The grid is one program per (batch*hidden, i‑tile, j‑tile).
    """
    pid = tl.program_id(0)

    # --- 1) Tile layout -------------------------------------------------
    num_tiles_m = tl.cdiv(N, BLOCK_M)       # tiles along i dimension
    num_tiles_n = tl.cdiv(N, BLOCK_N)       # tiles along j dimension
    tiles_per_batch = num_tiles_m * num_tiles_n

    batch_idx = pid // tiles_per_batch               # 0 … B*H-1
    tile_idx  = pid % tiles_per_batch
    pid_m = tile_idx // num_tiles_n                  # i‑tile id
    pid_n = tile_idx %  num_tiles_n                  # j‑tile id

    # block coordinates inside the (i,j) tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # --- 2) Decode batch and hidden indices -----------------------------
    b = batch_idx // HIDDEN_DIM            # batch index
    d = batch_idx %  HIDDEN_DIM            # hidden (channel) index

    # base pointers for the selected (b,d) slice
    L_base = L_ptr + b * stride_bL + d * stride_hL
    R_base = R_ptr + b * stride_bR + d * stride_hR
    O_base = O_ptr + b * stride_bO + d * stride_hO

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --- 3) Loop over the reduction dimension k -------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_tiles):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile:   shape (BLOCK_M, BLOCK_K)
        L_ptrs = L_base + offs_m[:, None] * stride_iL + offs_k[None, :] * stride_kL
        L = tl.load(L_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)               # FP16, dot will up‑cast to FP32

        # right tile (transposed): shape (BLOCK_K, BLOCK_N)
        # we need R[j,k] ≡ R[k,j] in the GEMM, thus swap the strides
        R_ptrs = R_base + offs_k[:, None] * stride_kR + offs_n[None, :] * stride_iR
        R = tl.load(R_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(L, R)                # TF32/Tensor‑Core path on H100

    # --- 4) Write result ------------------------------------------------
    O_ptrs = O_base + offs_m[:, None] * stride_iO + offs_n[None, :] * stride_jO
    tl.store(O_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]    (bool/float)
        - weights_dict : contains all model parameters
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------ #
    input_tensor, mask_tensor, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    nomask       = config.get("nomask", True)
    eps          = 1e-5

    # ------------------------------------------------------------------ #
    # 1. Input layer‑norm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------ #
    # 2. Fuse projection + gate linear layers (2 → 1 pass per side)
    fused_left_weight  = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    )  # (2*H, C)
    fused_right_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    )  # (2*H, C)

    left_comb   = torch.nn.functional.linear(x, fused_left_weight)   # [..., 2*H]
    right_comb  = torch.nn.functional.linear(x, fused_right_weight)  # [..., 2*H]

    left_proj   = left_comb[..., :hidden_dim]
    left_gate   = torch.sigmoid(left_comb[..., hidden_dim:])

    right_proj  = right_comb[..., :hidden_dim]
    right_gate  = torch.sigmoid(right_comb[..., hidden_dim:])

    # ------------------------------------------------------------------ #
    # 3. Optional mask (identical to reference implementation)
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left_proj  = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------ #
    # 4. Apply gates
    left  = left_proj * left_gate   # [B, N, N, H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------ #
    # 5. Out‑gate (separate linear)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------ #
    # 6. Cast to half‑precision for the heavy GEMM
    left_h  = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    B, N, _, _ = left_h.shape   # N is the sequence length
    BH = B * hidden_dim          # combined batch * hidden dimension

    # Output buffer (FP32 accumulator)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left_h.device)

    # ------------------------------------------------------------------ #
    # 7. Launch the Triton GEMM kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    tiles_per_batch = num_tiles_m * num_tiles_n
    total_programs = BH * tiles_per_batch

    stride_bL, stride_iL, stride_kL, stride_hL = left_h.stride()
    stride_bR, stride_iR, stride_kR, stride_hR = right_h.stride()
    stride_bO, stride_iO, stride_jO, stride_hO = out.stride()

    trimul_gemm_kernel[total_programs](
        left_h,
        right_h,
        out,
        N,
        hidden_dim,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        # left strides
        stride_bL, stride_iL, stride_kL, stride_hL,
        # right strides
        stride_bR, stride_iR, stride_kR, stride_hR,
        # output strides
        stride_bO, stride_iO, stride_jO, stride_hO,
        num_warps=4,
        num_stages=2,
    )

    # ------------------------------------------------------------------ #
    # 8. Output LayerNorm, gating and final projection (still in PyTorch)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
