    j0 = pid_n * BLOCK_N

    offs_j = j0 + tl.arange(0, BLOCK_N)

"""
TriMul (outgoing) – Triton‑accelerated implementation.

The heavy tensor contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is expressed as a batch of independent GEMMs (A @ Bᵀ) where each
(given batch b and hidden dimension h) is a matrix multiplication
of size N×N.  The kernel loads tiles of size (BLOCK_M, BLOCK_K) from
`left` and (BLOCK_K, BLOCK_N) from `right`, accumulates in FP32
(using Tensor‑cores via `tl.dot`) and writes the result in FP32.
All surrounding operations (LayerNorms, Linear projections,
gates, final linear) are performed in PyTorch with half‑precision
where possible to reduce memory traffic.  No extra copies of the
large tensors are performed – the kernel works directly on the
original layout (B, N, N, hidden) by computing the correct offsets
from the program id.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: computes for each (batch, hidden) pair the matrix
# product C = A @ Bᵀ  with  A[i,k] = left[b,i,k,h]  and  B[j,k] = right[b,j,k,h].
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,               # *fp16  [B, N, N, hidden]
    right_ptr,              # *fp16  [B, N, N, hidden]
    out_ptr,                # *fp32  [B, N, N, hidden]
    B, N, hidden_dim,       # scalar parameters
    stride_batch,           # i64 stride of the batch dimension
    stride_i0, stride_i1,   # i64 strides for the first and second N axes
    BLOCK_M: tl.constexpr,  # tile size in the i‑dimension
    BLOCK_N: tl.constexpr,  # tile size in the j‑dimension
    BLOCK_K: tl.constexpr,  # tile size in the k‑dimension
):
    pid_head = tl.program_id(0)          # = b * hidden_dim + h
    pid_m    = tl.program_id(1)          # block row (i)
    pid_n    = tl.program_id(2)          # block col (j)

    # ------------------------------------------------------------------
    # Decode (batch, hidden) from the flattened head id
    # ------------------------------------------------------------------
    b = pid_head // hidden_dim
    h = pid_head % hidden_dim

    # Base offset for this (b,h) slice.
    # hidden dimension is the innermost one → stride 1.
    head_offset = b * stride_batch + h

    # ------------------------------------------------------------------
    # Compute tile coordinates and masks
    # ------------------------------------------------------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the k‑dimension in blocks
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_blocks):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i,k]  → left_ptr + head_offset + i*stride_i0 + k*stride_i1
        left_ptrs = (
            left_ptr
            + head_offset
            + offs_i[:, None] * stride_i0
            + offs_k[None, :] * stride_i1
        )
        left_mask = mask_i[:, None] & mask_k[None, :]
        a = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # right[j,k]  (note that the first N axis plays the role of j here)
        right_ptrs = (
            right_ptr
            + head_offset
            + offs_j[:, None] * stride_i0
            + offs_k[None, :] * stride_i1
        )
        right_mask = mask_j[:, None] & mask_k[None, :]
        b = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # Accumulate the product (Tensor‑core friendly)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result tile (FP32)
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + head_offset
        + offs_i[:, None] * stride_i0
        + offs_j[None, :] * stride_i1
    )
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)


# ----------------------------------------------------------------------
# Triton‑backed forward pass of the outgoing Triangle‑Multiplicative
# update (TriMul) as described in the AlphaFold‑3 paper.
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Arguments
    ---------
    data = (input_tensor, mask, weights, config)

    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]                 # input feature dimension
    hidden_dim = config["hidden_dim"]   # hidden dimension for the TriMul gates
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the input feature dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 and fuse the five linear projections (proj + gates + out_gate)
    # ------------------------------------------------------------------
    # Convert input to half‑precision for the heavy linear maps
    x_h = x.to(torch.float16)

    # Gather weight matrices and cast them once to FP16
    w_left_proj = weights["left_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    # Concatenate along the output‑feature axis: shape ((5*hidden_dim), dim)
    w_concat = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate],
        dim=0,
    )

    # Single fused linear: result shape [B, N, N, 5*hidden_dim] (FP16)
    fused = F.linear(x_h, w_concat)  # bias is None for all these layers

    # Split into the individual tensors
    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        fused, hidden_dim, dim=-1
    )

    # ------------------------------------------------------------------
    # 3️⃣ Activate gates (sigmoid) – still FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)  # used later after to_out_norm

    # ------------------------------------------------------------------
    # 4️⃣ Optional masking (mask broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(torch.float16)  # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 5️⃣ Apply gates to the projections (still FP16)
    # ------------------------------------------------------------------
    left = left_proj * left_gate          # [B, N, N, hidden]  FP16
    right = right_proj * right_gate       # [B, N, N, hidden]  FP16

    # ------------------------------------------------------------------
    # 6️⃣ Tri‑multiplicative update – the only part executed in Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Strides of the (B, N, N, hidden) layout (in element units)
    stride_batch = left.stride(0)   # B‑dimension stride = N*N*hidden
    stride_i0 = left.stride(1)      # first N (i) stride = N*hidden
    stride_i1 = left.stride(2)      # second N (j/k) stride = hidden

    # Output buffer (FP32) – same layout as left/right
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Tile sizes – tuned for H100 Tensor Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                     # flattened (batch, hidden) dimension
        (N + BLOCK_M - 1) // BLOCK_M,       # number of i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,       # number of j‑tiles
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_batch,
        stride_i0,
        stride_i1,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,            # leverage H100’s large thread‑block capacity
    )

    # ------------------------------------------------------------------
    # 7️⃣ Post‑processing: LayerNorm → out‑gate → final linear
    # ------------------------------------------------------------------
    # Normalize over the hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # FP32

    # Apply the output gate (converted to FP32 for the multiplication)
    out = out * out_gate.to(torch.float32)

    # Final linear projection back to the original `dim` (FP32)
    out = F.linear(out, weights["to_out.weight"])

    return out
