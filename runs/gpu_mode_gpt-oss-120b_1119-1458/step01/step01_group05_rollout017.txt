def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Determine batch (b*h index)
    batch = tl.program_id(0)  # 0 <= batch < B*H
    b = batch // H
    h = batch % H
    # Determine block coordinates for i, j
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    left_base = left_ptr + b * stride_left_b + h * stride_left_h

    nomask = config.get("nomask", False)

    # Cast weights to half for speed
    # We'll make a dict of half versions
    for name in weights:
        if weights[name].dtype != torch.float16:
            weights[name] = weights[name].half()

    # Layernorm
    x_norm = F.layer_norm(inp, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Cast to half
    x_norm_h = x_norm.half()

    # Linear projections with half
    left_proj = F.linear(x_norm_h, weights["left_proj.weight"])
    right_proj = F.linear(x_norm_h, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x_norm_h, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm_h, weights["right_gate.weight"]))

    if nomask:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        mask_h = mask.unsqueeze(-1).to(torch.float16) # broadcast
        left = fused_mul3(left_proj, mask_h, left_gate)
        right = fused_mul3(right_proj, mask_h, right_gate)

    # Outgoing contraction using Triton GEMM
    # out shape [B,N,N,H]
    out = torch.empty_like(left, dtype=torch.float16)  # allocate output

    # Compute stride values
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Choose block sizes heuristically based on N
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
    else:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64

    grid = (B * hidden_dim, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)
    trimul_outgoing_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert to float32 for remaining
    out_fp32 = out.to(torch.float32)

    # Output layernorm
    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    out_gate = torch.sigmoid(F.linear(x_norm_h, weights["out_gate.weight"]))
    out_norm = out_norm * out_gate

    out_final = F.linear(out_norm, weights["to_out.weight"])

    return out_final
```

```
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # batch index
    batch_idx = tl.program_id(0)
    # compute batch and hidden idx
    b = batch_idx // H
    h = batch_idx % H

    # tile indices
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets for M and N
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Compute pointers offsets for left and right per batch/h
    left_batch_ptr = left_ptr + b * stride_left_b + h * stride_left_h
    right_batch_ptr = right_ptr + b * stride_right_b + h * stride_right_h
    out_batch_ptr = out_ptr + b * stride_out_b + h * stride_out_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile (M x K)
        left_ptrs = left_batch_ptr + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load right tile (N x K)
        right_ptrs = right_batch_ptr + offs_n[:, None] * stride_right_j + offs_k[None, :] * stride_right_k
        right = tl.load(right_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # dot (M x K) * (N x K)^T -> (M,N)
        # we need right transposed: (K,N) -> using trans_b=True
        acc += tl.dot(left, right, trans_b=True)

    # Store output
    out_ptrs = out_batch_ptr + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# fused multiplication kernel
@triton.jit
def fused_mul3_kernel(a_ptr, b_ptr, c_ptr, out_ptr, size, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    c = tl.load(c_ptr + offs, mask=mask)
    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)

def fused_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    assert a.shape == b.shape == c.shape
    a = a.contiguous()
    b = b.contiguous()
    c = c.contiguous()
    out = torch.empty_like(a)
    total = a.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)
    fused_mul3_kernel[grid](
        a, b, c, out, total, BLOCK_SIZE=BLOCK
    )
    return out

# main contraction kernel
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # batch (b, h)
    batch_idx = tl.program_id(0)
    b = batch_idx // H
    h = batch_idx % H

    # tile indices (i, j)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # ranges within N
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute base pointers for this (b, h)
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile MxK
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right tile KxN (transposed version of right[j,k])
        right_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(left, right)   # (M,N) result in float32

    # write result
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Steps:
    1. Layer‑norm the input.
    2. Two linear projections (left/right) and three gating projections (left_gate,
       right_gate, out_gate). All linear ops are performed in FP16.
    3. Optional mask‑gate fusion using a tiny Triton kernel (fused_mul3).
    4. Core contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
       implemented as a custom batched GEMM in Triton, working directly on the
       [B,N,N,H] layout to avoid extra transposes.
    5. Layer‑norm over the hidden dimension, output gate, and final linear
       projection back to `dim`.
    Returns a float‑32 tensor of shape [B, N, N, dim].
    """
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # -------- Cast all weights to half (if not already) --------
    # List of required weight keys
    weight_keys = [
        "norm.weight", "norm.bias",
        "left_proj.weight", "right_proj.weight",
        "left_gate.weight", "right_gate.weight", "out_gate.weight",
        "to_out_norm.weight", "to_out_norm.bias",
        "to_out.weight"
    ]
    for k in weight_keys:
        w = weights[k]
        if w.dtype != torch.float16:
            weights[k] = w.half()

    # -------- Layer‑norm (half) --------
    # torch.layer_norm works with half tensors
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).half()   # [B,N,N,dim] half

    # -------- Linear projections (half) --------
    left_proj = F.linear(x_norm, weights["left_proj.weight"])    # [B,N,N,hidden_dim]
    right_proj = F.linear(x_norm, weights["right_proj.weight"])

    left_gate_pre = F.linear(x_norm, weights["left_gate.weight"])
    right_gate_pre = F.linear(x_norm, weights["right_gate.weight"])
    out_gate_pre = F.linear(x_norm, weights["out_gate.weight"])

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)   # [B,N,N,hidden_dim]

    # -------- Apply mask & gates (fused) --------
    if nomask:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # mask is [B,N,N]; expand to hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16)   # [B,N,N,1] -> broadcast
        left = fused_mul3(left_proj, mask_h, left_gate)
        right = fused_mul3(right_proj, mask_h, right_gate)

    # -------- Core contraction (custom Triton GEMM) --------
    # Allocate output tensor in half
    out_h = torch.empty_like(left)   # same shape [B,N,N,hidden_dim], half

    # Compute strides (int64)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_h.stride()

    # Choose block sizes; tuned for H100
    # For N up to 1024 we keep blocks 128x128 and inner k 32
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    trimul_outgoing_kernel[grid](
        left, right, out_h,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------- Output processing (half → float32) --------
    # Layer‑norm over hidden dim (still in half for speed)
    out_norm = F.layer_norm(
        out_h,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate

    # Final projection back to dim (half)
    out_final = F.linear(out_norm, weights["to_out.weight"])

    # Cast back to float32 as required by the spec
    return out_final.float()

   is implemented as a custom batched GEMM that works directly on the
   original [B,N,N,H] layout, thus avoiding the permute / view overhead
   of the naïve torch.bmm implementation.
5. Layer‑norm over the hidden dimension, application of the output gate,
   and a final linear projection back to the original channel dimension.
   The final result is cast back to float32 as required by the API.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# tiny element‑wise kernel: out = a * b * c
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr, size,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    c = tl.load(c_ptr + offs, mask=mask)

    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """Fuse three element‑wise products with a Triton kernel."""
    assert a.shape == b.shape == c.shape
    a = a.contiguous()
    b = b.contiguous()
    c = c.contiguous()
    out = torch.empty_like(a)

    total = a.numel()
    BLOCK = 1024  # works well on H100
    grid = (triton.cdiv(total, BLOCK),)

    fused_mul3_kernel[grid](
        a, b, c, out, total, BLOCK_SIZE=BLOCK
    )
    return out


# ----------------------------------------------------------------------
# batched GEMM for the TriMul contraction (outgoing version)
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ----- batch / hidden‑dim indices (b, h) -----
    batch_idx = tl.program_id(0)
    b = batch_idx // H
    h = batch_idx % H

    # ----- tile indices for the i‑ and j‑axes -----
    pid_m = tl.program_id(1)   # i‑tile
    pid_n = tl.program_id(2)   # j‑tile

    # ----- i / j coordinates inside the tile -----
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ----- base pointers for this (b, h) pair -----
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- iterate over the reduction axis k -----
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left:   shape (BLOCK_M, BLOCK_K)   (i, k)
        left_ptrs = left_base + \
                    offs_i[:, None] * stride_left_i + \
                    offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right: we need the transposed view (k, j) of right[j, k]
        # shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + \
                     offs_k[:, None] * stride_right_k + \
                     offs_j[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # matrix multiplication for this k‑tile
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N) in FP32

    # ----- write the result back (store as FP16) -----
    out_ptrs = out_base + \
               offs_i[:, None] * stride_out_i + \
               offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# entry point required by the challenge
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : [B, N, N, dim]  (float32, CUDA)
        mask_tensor  : [B, N, N]      (bool/float32, CUDA)
        weights_dict : model weights with keys as in the reference module
        config_dict  : contains at least "dim" and "hidden_dim",
                       optional "eps" and "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # cast all model parameters to FP16 once (weights are small)
    # ------------------------------------------------------------------
    for k in [
        "norm.weight", "norm.bias",
        "left_proj.weight", "right_proj.weight",
        "left_gate.weight", "right_gate.weight", "out_gate.weight",
        "to_out_norm.weight", "to_out_norm.bias",
        "to_out.weight",
    ]:
        w = weights[k]
        if w.dtype != torch.float16:
            weights[k] = w.half()

    # ------------------------------------------------------------------
    # 1. Layer‑norm (stay in FP16)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).half()   # [B, N, N, dim]  (fp16)

    # ------------------------------------------------------------------
    # 2. Linear projections (FP16, no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right_proj = F.linear(x_norm, weights["right_proj.weight"])

    left_gate_pre = F.linear(x_norm, weights["left_gate.weight"])
    right_gate_pre = F.linear(x_norm, weights["right_gate.weight"])
    out_gate_pre = F.linear(x_norm, weights["out_gate.weight"])

    left_gate = torch.sigmoid(left_gate_pre)      # FP16
    right_gate = torch.sigmoid(right_gate_pre)    # FP16
    out_gate = torch.sigmoid(out_gate_pre)        # FP16 ⟶ later used after contraction

    # ------------------------------------------------------------------
    # 3. Apply optional mask and fuse with gates
    # ------------------------------------------------------------------
    if nomask:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # mask: [B, N, N] → [B, N, N, 1] → broadcast to hidden_dim
        mask_h = mask.unsqueeze(-1).to(torch.float16)   # (fp16)
        left = fused_mul3(left_proj, mask_h, left_gate)
        right = fused_mul3(right_proj, mask_h, right_gate)

    # ------------------------------------------------------------------
    # 4. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented with a custom Triton GEMM working directly on the
    #    [B,N,N,H] layout – no transposes, no extra copies.
    # ------------------------------------------------------------------
    out_h = torch.empty_like(left)   # allocate FP16 output tensor

    # strides for the three tensors (int64)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_rj, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out_h.stride()

    # tile sizes – these values give good occupancy on an H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                          # batch * hidden
        (N + BLOCK_M - 1) // BLOCK_M,            # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,            # j‑tiles
    )
    trimul_outgoing_kernel[grid](
        left, right, out_h,
        B, N, hidden_dim,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_rj, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 5. Output layer‑norm, gating and final projection (still FP16)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out_h,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate                       # apply output gate

    out_final = F.linear(out_norm, weights["to_out.weight"])   # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # Cast back to FP32 as required by the external API
    # ------------------------------------------------------------------
    return out_final.float()
```
