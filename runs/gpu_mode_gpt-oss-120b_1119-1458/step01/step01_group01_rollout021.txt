"""
Highly‑optimized outgoing Triangle Multiplicative update (TriMul).

The kernel computes the cubic reduction
    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
for every batch b and hidden channel h.  The reduction is expressed as a
batched matrix multiplication (B·H independent matmuls of size N×N) and
is carried out with a custom Triton kernel that:
  * works in fp16 (Tensor‑Core friendly) and accumulates in fp32,
  * uses a 128×128×64 tile (BLOCK_M/BLOCK_N = 128, BLOCK_K = 64),
  * launches 16 warps per block and a 4‑stage pipeline for maximal
    overlap of loads and compute on an H100,
  * avoids any Python loops – the whole N³ reduction is performed on‑GPU.

All cheap operations (LayerNorms, Linear projections, sigmoids,
masking, final projection) are done with PyTorch in half‑precision when
possible, keeping the final result in float32 for downstream layers.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the N³ reduction
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                     # tensor pointers
    stride_ap, stride_am, stride_ak,          # A: (P, M, K) strides
    stride_bp, stride_bk, stride_bn,          # B: (P, K, N) strides
    stride_cp, stride_cm, stride_cn,          # C: (P, M, N) strides
    M, N, K,                                  # matrix sizes (all = seq_len)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes C[p, m, n] = Σₖ A[p, m, k] • B[p, k, n]   (fp16 → fp32 accumulation)
    """
    pid_m = tl.program_id(0)        # block index over M
    pid_n = tl.program_id(1)        # block index over N
    pid_p = tl.program_id(2)        # block index over (batch × hidden)

    # ----------- coordinate offsets ------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this (batch,hidden) slice
    a_base = A_ptr + pid_p * stride_ap
    b_base = B_ptr + pid_p * stride_bp
    c_base = C_ptr + pid_p * stride_cp

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------- K‑loop (tiling over the reduction dimension) ---------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        start_k = k * BLOCK_K
        offs_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (M × K)
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                         # fp16 → fp32 on the fly

        # B tile (K × N)
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # fused fp16 dot product, result accumulated in fp32
        acc += tl.dot(a, b).to(tl.float32)

    # ----------- write back -------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def _triton_matmul(left: torch.Tensor, right: torch.Tensor,
                   seq_len: int, hidden_dim: int,
                   BLOCK_M=128, BLOCK_N=128, BLOCK_K=64):
    """
    Batched matrix multiplication for the outgoing TriMul.
    left/right : [B, N, N, H]  (fp16)
    returns    : [B, N, N, H]  (fp16)
    """
    B = left.shape[0]

    # Rearrange to (B, H, N, N) – contiguous in memory – and flatten
    # (no extra copy beyond the required contiguous view)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # note the transpose on the two N axes

    P = B * hidden_dim                          # total independent matmuls
    left_flat = left_perm.view(P, seq_len, seq_len)
    right_flat = right_perm.view(P, seq_len, seq_len)

    out = torch.empty_like(left_flat)

    # Strides of the flattened tensors (they are contiguous)
    stride_ap = left_flat.stride(0)   # = N * N
    stride_am = left_flat.stride(1)   # = N
    stride_ak = left_flat.stride(2)   # = 1

    stride_bp = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_cp = out.stride(0)
    stride_cm = out.stride(1)
    stride_cn = out.stride(2)

    # launch grid: (M‑tiles, N‑tiles, P)
    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,
        (seq_len + BLOCK_N - 1) // BLOCK_N,
        P,
    )

    _batched_matmul_kernel[grid](
        left_flat, right_flat, out,
        stride_ap, stride_am, stride_ak,
        stride_bp, stride_bk, stride_bn,
        stride_cp, stride_cm, stride_cn,
        seq_len, seq_len, seq_len,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=16,          # H100 can sustain many warps per block
        num_stages=4,          # deeper pipeline to hide memory latency
    )

    # reshape back to the original 4‑D layout
    out = out.view(B, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()
    return out

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator with a Triton‑accelerated
    N³ reduction.  All tensors are assumed to be on the correct CUDA device.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]                # input & output channel dimension
    hidden_dim = config["hidden_dim"]  # internal hidden dimension
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 → fp16 after)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)                     # [B, N, N, dim]  float32

    # convert to half‑precision for the heavy path
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) – keep them in fp16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)      # [B, N, N, hidden]  fp16
    right = F.linear(x_fp16, right_proj_w)    # [B, N, N, hidden]  fp16

    # ------------------------------------------------------------------
    # 3) Optional mask (applied in fp16 after the projections)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B, N, N] → [B, N, N, 1] and broadcast to hidden dim
        mask_fp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid(Linear(x))) – compute in fp16
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))    # [B,N,N,hidden] fp16
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))    # kept in fp16 for later broadcast

    # apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Heavy N³ reduction via the Triton kernel
    # ------------------------------------------------------------------
    seq_len = input_tensor.shape[1]   # N
    out_fp16 = _triton_matmul(left, right,
                              seq_len=seq_len,
                              hidden_dim=hidden_dim,
                              BLOCK_M=128,
                              BLOCK_N=128,
                              BLOCK_K=64)

    # ------------------------------------------------------------------
    # 6) Bring back to float32 for the final LayerNorm / projection
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)   # [B, N, N, hidden]

    # to_out_norm (LayerNorm over hidden dimension)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=eps)

    # apply the output gate (broadcast over hidden)
    out = out * out_gate.to(torch.float32)

    # final linear back to original `dim`
    to_out_w = weights["to_out.weight"].to(torch.float32)   # (dim, hidden)
    out = F.linear(out, to_out_w)                         # [B, N, N, dim] float32

    return out
```
