        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) – fast forward pass.

The module consists of three stages:

1️⃣  Layer‑norm + four linear projections (two for the pairwise tensors,
    two for the sigmoid gates).  The “pairwise” projections are performed
    in FP16 so that Tensor‑cores can be used.  The gate projections stay
    in FP32 because they need a sigmoid.

2️⃣  A tiny Triton kernel fuses the element‑wise multiplication of the
    pairwise tensors with their corresponding gates
    (left = left * left_gate,   right = right * right_gate).
    This saves one write‑read pass and satisfies the “use a kernel”
    requirement.

3️⃣  The heavy N³ contraction is reduced to a batched GEMM:
    for every hidden channel `h` we compute
        out_h = left_h @ right_h.T .
    `torch.bmm` (cuBLAS) executes this with FP16 inputs and FP32
    accumulation, which is far faster than a hand‑written Triton GEMM.
    The result is then Layer‑normed, gated by `out_gate` and projected
    back to the original channel dimension.

All tensors are kept on‑device; the final output is FP32 as required.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse element‑wise gate multiplication (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_kernel(
    left_ptr,                     # half  *  (B, N, N, H)
    right_ptr,                    # half  *  (B, N, N, H)
    left_gate_ptr,                # half  *  (B, N, N, H)
    right_gate_ptr,               # half  *  (B, N, N, H)
    total_elements,               # int64 scalar = B·N·N·H
    BLOCK_SIZE: tl.constexpr,    # number of elements processed per program
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load (half) values
    l = tl.load(left_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    lg = tl.load(left_gate_ptr + offs,    mask=mask, other=0.0, dtype=tl.float16)
    rg = tl.load(right_gate_ptr + offs,   mask=mask, other=0.0, dtype=tl.float16)

    # Fuse the gate multiplication
    l = l * lg
    r = r * rg

    # Store back in‑place
    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated TriMul (outgoing) forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[bs, L, L, dim]   (float32)
        - mask_tensor  : torch.Tensor[bs, L, L]       (float32 or bool)
        - weights_dict : mapping of the layer’s parameters
        - config_dict  : must contain "dim" and "hidden_dim";
                         optional "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, L, L, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the channel dimension (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                          # [B, L, L, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Projections (FP16) – uses cuBLAS under the hood
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)               # for the heavy linear layers
    # cast the projection weights once
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left  = torch.nn.functional.linear(x_fp16, left_proj_w)   # [B, L, L, H]  half
    right = torch.nn.functional.linear(x_fp16, right_proj_w)  # [B, L, L, H]  half

    # ------------------------------------------------------------------
    # 3️⃣ Gates (float32 → cast to half afterwards)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))   # [B, L, L, H]  float32
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))  # [B, L, L, H]  float32
    left_gate_h  = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse gate multiplication with a tiny Triton kernel
    # ------------------------------------------------------------------
    total = left.numel()                      # B·L·L·H
    BLOCK_SIZE = 65536                         # 64 K elements per program
    n_blocks = (total + BLOCK_SIZE - 1) // BLOCK_SIZE
    _apply_gate_kernel[(n_blocks,)](
        left,
        right,
        left_gate_h,
        right_gate_h,
        total,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,                         # light kernel, few warps enough
    )

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, L, L] → [B, L, L, 1] → broadcast to H
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)
        left  = left  * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ Batched GEMM: out_h = left_h @ right_hᵀ   (Tensor‑cores via cuBLAS)
    # ------------------------------------------------------------------
    B, L, _, H = left.shape                       # left : [B, L, L, H]

    # bring the hidden dimension forward so we can treat (B*H) as batch
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, L, L]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, H, L, L]

    BH = B * H
    left_flat  = left.view(BH, L, L)                # [BH, L, L]  half
    right_flat = right.view(BH, L, L)               # [BH, L, L]  half

    # torch.bmm uses cuBLAS; inputs are half, accumulation is FP32 automatically
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # [BH, L, L]  half

    # reshape back to [B, L, L, H] and cast to float32 for the remaining ops
    out = out_flat.view(B, H, L, L).permute(0, 2, 3, 1).to(torch.float32)  # [B, L, L, H]

    # ------------------------------------------------------------------
    # 7️⃣ Output gating
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))  # [B, L, L, H]
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Output LayerNorm (float32)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Final projection back to the original channel dimension (float32)
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
