"""
TriMul (outgoing) – fast Triton‑accelerated forward pass.

Algorithm
---------
1. Layer‑norm the input (`x`) in FP16.
2. Fuse *all* linear projections (left/right projections + gates + out‑gate)
   into a *single* large GEMM:  x @ W_all   where
   W_all = [left_proj; left_gate; right_proj; right_gate; out_gate]
   (shape (5*H, C)).  This eliminates four separate GEMM launches and
   reduces memory traffic on the huge [B,N,N,C] tensor.
3. Split the result into the five tensors, apply the sigmoid gates,
   optionally multiply by a mask, and obtain the gated tensors
   `left` and `right`.
4. Core contraction:
        out[i,j,h] = Σ_k left[i,k,h] * right[j,k,h]
   This is a batched GEMM for each hidden channel.
   We reshape to (B*H, N, N) and call `torch.bmm` in FP16,
   which uses Tensor‑cores on H100.
5. Layer‑norm over the hidden dimension, apply the output gate,
   and a final linear projection back to `dim`.
6. Cast the final result back to FP32 (the model expects float32).

Only the element‑wise product of three tensors (mask × proj × gate) is
implemented as a Triton kernel (`fused_mul3`).  All heavy arithmetic
is performed in FP16 to maximise throughput on the H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused element‑wise product of three same‑shaped tensors
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,                # device pointers
    size,                                         # total number of elements
    BLOCK_SIZE: tl.constexpr,                     # compile‑time constant
):
    """out = a * b * c   (all tensors are 1‑D contiguous, FP16)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    # Load with default dtype (derived from the pointer type)
    a = tl.load(a_ptr + offs, mask=mask, other=tl.float16(0.0))
    b = tl.load(b_ptr + offs, mask=mask, other=tl.float16(0.0))
    c = tl.load(c_ptr + offs, mask=mask, other=tl.float16(0.0))

    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """
    Fuse three element‑wise multiplications using Triton.
    All inputs must have the same shape and be on CUDA.
    Returns a tensor of the same shape in FP16.
    """
    assert a.shape == b.shape == c.shape
    a = a.contiguous()
    b = b.contiguous()
    c = c.contiguous()
    out = torch.empty_like(a)

    total = a.numel()
    BLOCK = 1024                     # good default for H100
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_kernel[grid](
        a, b, c, out,
        total,
        BLOCK_SIZE=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the *outgoing* TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : [B, N, N, C]   (float32, CUDA)
            - mask_tensor  : [B, N, N]      (float32/bool, CUDA) – may be unused
            - weights_dict : model parameters (already on CUDA)
            - config_dict  : contains at least
                * "dim"        – C
                * "hidden_dim" – H
                * optional "eps"      (LayerNorm epsilon, default 1e‑5)
                * optional "nomask"   (True → ignore mask)

    Returns:
        Tensor of shape [B, N, N, C] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    H = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Cast everything to FP16 (half) for maximal throughput
    # ------------------------------------------------------------------
    x = inp.to(torch.float16)                     # [B, N, N, C]

    # Cast weights once – they are small, the cost is negligible
    norm_w   = weights["norm.weight"].to(torch.float16)
    norm_b   = weights["norm.bias"].to(torch.float16)

    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, C)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, C)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)  # (H, C)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)  # (H, C)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (H, C)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16) # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)   # (H,)
    to_out_w      = weights["to_out.weight"].to(torch.float16)      # (C, H)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the channel dimension (C)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )   # [B, N, N, C] (FP16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse *all* linear maps into a single GEMM
    #    weight matrix:   (5*H, C)
    #    output shape:    [B, N, N, 5*H]
    # ------------------------------------------------------------------
    all_weight = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,                         # concatenate on the “out_features” axis
    )                                   # (5*H, C)

    # Combined linear – huge GEMM but executed once
    combined = F.linear(x_norm, all_weight)        # [B, N, N, 5*H]

    # ------------------------------------------------------------------
    # 3️⃣ Split the combined result and apply sigmoids
    # ------------------------------------------------------------------
    # each tensor has shape [B, N, N, H] (FP16)
    left_proj, left_gate_pre, right_proj, right_gate_pre, out_gate_pre = torch.split(
        combined, H, dim=-1
    )

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # ------------------------------------------------------------------
    # 4️⃣ Apply gates (and optional mask) – fused element‑wise product
    # ------------------------------------------------------------------
    if nomask:
        left  = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # mask → FP16 and broadcast to hidden dimension
        mask_h = mask.to(torch.float16).unsqueeze(-1).expand(-1, -1, -1, H).contiguous()
        left  = fused_mul3(left_proj, mask_h, left_gate)
        right = fused_mul3(right_proj, mask_h, right_gate)

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction  out[i,j,h] = Σ_k left[i,k,h] * right[j,k,h]
    #    Implemented as a batched GEMM (FP16) → Tensor‑cores
    # ------------------------------------------------------------------
    # Bring the hidden dimension to the batch axis
    left_mat  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]   (i‑k)
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]   (j‑k)

    # Collapse batch × hidden for a single bmm call
    left_flat  = left_mat.view(B * H, N, N)   # [B*H, N, N]
    right_flat = right_mat.view(B * H, N, N)  # [B*H, N, N]

    # Batched matrix multiplication (FP16)
    out_flat = torch.bmm(left_flat, right_flat)   # [B*H, N, N]

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Output layer‑norm over the hidden dimension (H)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )   # [B, N, N, H]  (FP16)

    # 7️⃣ Apply the output gate
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to `dim` and cast to FP32
    # ------------------------------------------------------------------
    out_fp32 = F.linear(out_norm, to_out_w).float()   # [B, N, N, C] (float32)

    return out_fp32
