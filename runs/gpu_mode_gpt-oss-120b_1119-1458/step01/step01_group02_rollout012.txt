    half‑precision (fp16) to halve memory traffic while keeping the final
    result in fp32.

2️⃣  Linear projections (no bias) → `left` and `right` (shape: [B,N,N,H]) and
    three sigmoid gates (`left_gate`, `right_gate`, `out_gate`).

3️⃣  Apply optional mask (broadcasted on the hidden dimension) and the
    element‑wise gates.

4️⃣  Heavy contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    This is a batched matrix multiplication C = A @ Bᵀ for each
    (batch, hidden) pair.  The Triton kernel works on the tensors in their
    original layout ((B,N,N,H)) – **no permute / copy** – and keeps the data
    in fp16 while accumulating in fp32.

5️⃣  Layer‑norm over the hidden dimension (`H`), multiply by `out_gate`,
    final linear projection back to `dim`, and cast the result to fp32.

The kernel is tiled (BLOCK_M × BLOCK_N) with a K‑tile of size BLOCK_K.
All loads are coalesced; accumulation uses FP32 for numerical stability.
"""
import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: C[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,               # *fp16 pointers
    B, N, H,                           # i32 sizes
    stride_a_b, stride_a_i, stride_a_k, stride_a_h,   # A strides
    stride_b_b, stride_b_j, stride_b_k, stride_b_h,   # B strides
    stride_c_b, stride_c_i, stride_c_j, stride_c_h,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)            # tile along i (first N dim)
    pid_j = tl.program_id(1)            # tile along j (second N dim)
    pid_z = tl.program_id(2)            # combined batch * hidden index

    # ------------------------------------------------------------------
    # Decode batch and hidden indices from pid_z
    # ------------------------------------------------------------------
    b = pid_z // H                      # batch index
    h = pid_z % H                       # hidden index

    # ------------------------------------------------------------------
    # Offsets for the output tile (i and j)
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (to use tensor cores internally)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K tiles
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # ----- Load left tile: shape (BLOCK_M, BLOCK_K) -----
        a_ptrs = (
            A_ptr
            + b * stride_a_b
            + h * stride_a_h
            + offs_i[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & (offs_k[None, :] < N),
                    other=0.0,
                    dtype=tl.float16)

        # ----- Load right tile (transposed): shape (BLOCK_K, BLOCK_N) -----
        b_ptrs = (
            B_ptr
            + b * stride_b_b
            + h * stride_b_h
            + offs_k[:, None] * stride_b_k
            + offs_j[None, :] * stride_b_j
        )
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < N) & mask_j[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # ----- Accumulate (fp32) -----
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # ------------------------------------------------------------------
    # Write back the result (store as fp16, later layers work in fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + b * stride_c_b
        + h * stride_c_h
        + offs_i[:, None] * stride_c_i
        + offs_j[None, :] * stride_c_j
    )
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel on tensors in layout [B,N,N,H]
# ----------------------------------------------------------------------
def _trimul_outgoing_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the outgoing TriMul contraction in fp16 using Triton.
    Input tensors must have shape [B, N, N, H] and be contiguous.
    Returns a tensor of the same shape and dtype (fp16).
    """
    B, N, _, H = left.shape
    assert left.is_contiguous() and right.is_contiguous(), "inputs must be contiguous"

    # Output buffer (fp16)
    out = torch.empty_like(left)

    # Strides (in elements, not bytes)
    stride_a_b, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_b, stride_b_j, stride_b_k, stride_b_h = right.stride()
    stride_c_b, stride_c_i, stride_c_j, stride_c_h = out.stride()

    # Tunable tile sizes – good for H100 FP16 Tensor Cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64   # larger K‑tile reduces loop count

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden
    )

    _trimul_outgoing_kernel[grid](
        left, right, out,
        B, N, H,
        stride_a_b, stride_a_i, stride_a_k, stride_a_h,
        stride_b_b, stride_b_j, stride_b_k, stride_b_h,
        stride_c_b, stride_c_i, stride_c_j, stride_c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # use 8 warps per block on H100
    )
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradient).
    Returns a tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Cast everything to fp16 (except the final output)
    # ------------------------------------------------------------------
    device = input_tensor.device
    dtype_fp16 = torch.float16

    # layer‑norm parameters
    norm_w = weights["norm.weight"].to(dtype_fp16)
    norm_b = weights["norm.bias"].to(dtype_fp16)

    # linear weights (no bias)
    left_proj_w   = weights["left_proj.weight"].to(dtype_fp16)
    right_proj_w  = weights["right_proj.weight"].to(dtype_fp16)
    left_gate_w   = weights["left_gate.weight"].to(dtype_fp16)
    right_gate_w  = weights["right_gate.weight"].to(dtype_fp16)
    out_gate_w    = weights["out_gate.weight"].to(dtype_fp16)

    # output‑norm + final projection
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype_fp16)
    to_out_w      = weights["to_out.weight"].to(dtype_fp16)

    # ------------------------------------------------------------------
    # 1️⃣  Layer‑norm over the feature dimension (keep fp16)
    # ------------------------------------------------------------------
    x = input_tensor.to(dtype_fp16)
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim]  (fp16)

    # ------------------------------------------------------------------
    # 2️⃣  Projections (fp16, bias‑free)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣  Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(dtype_fp16)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣  Gates (sigmoid, fp16) and application
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣  Heavy O(N³) contraction via Triton (still fp16)
    # ------------------------------------------------------------------
    # Ensure tensors are contiguous – linear returns contiguous tensors.
    out = _trimul_outgoing_matmul(left, right)   # [B,N,N,hidden] fp16

    # ------------------------------------------------------------------
    # 6️⃣  Second LayerNorm over hidden dimension (fp16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣  Gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] fp16

    # Cast back to fp32 for the final output as required by the spec
    return out.to(torch.float32)
```
