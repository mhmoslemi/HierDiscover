import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

The heavy N³ reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched matrix multiply (Gram product) using Triton.
All other operations (layer‑norms, linear projections, gating, final linear)
are performed with PyTorch for simplicity.
"""

import math
import torch
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton kernel: batched Gram matrix (left @ rightᵀ) over the
# sequence dimension.  Input tensors have shape (M, N, N) where
# M = batch_size * hidden_dim.  The kernel computes
#   out[m, i, j] = Σ_k left[m,i,k] * right[m,k,j]
# -------------------------------------------------------------
@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N,
    stride_lm, stride_ln, stride_lk,
    stride_rm, stride_rn, stride_rk,
    stride_om, stride_on, stride_ok,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)   # batch*hidden index
    pid_i = tl.program_id(1)       # tile row
    pid_j = tl.program_id(2)       # tile col

    # -------------------- tile offsets --------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 (more stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- reduction over k --------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_ptr + pid_batch * stride_lm
                     + offs_i[:, None] * stride_ln
                     + offs_k[None, :] * stride_lk,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )  # dtype matches input (fp16)

        # right block transposed: shape (BLOCK_K, BLOCK_N)
        # we need right[k, j] => stride_rk for k, stride_rn for j
        right = tl.load(
            right_ptr + pid_batch * stride_rm
                      + offs_k[:, None] * stride_rk
                      + offs_j[None, :] * stride_rn,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )  # dtype matches input (fp16)

        # dot produces fp16, cast to fp32 before accumulation
        acc += tl.dot(left, right).to(tl.float32)

    # -------------------- store result --------------------
    tl.store(
        out_ptr + pid_batch * stride_om
                + offs_i[:, None] * stride_on
                + offs_j[None, :] * stride_ok,
        acc,
        mask=mask_i[:, None] & mask_j[None, :]
    )


# -------------------------------------------------------------
# Public entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [batch, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast on the hidden dimension)
    # -----------------------------------------------------------------
    if not nomask:
        m = mask.unsqueeze(-1)               # [B, N, N, 1]
        left = left * m
        right = right * m

    # -----------------------------------------------------------------
    # 4. Gating values (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # -----------------------------------------------------------------
    # 5. Apply gates
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Prepare tensors for the Triton kernel
    #    Shape transformation: (B, N, N, H) -> (B*H, N, N)
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    left = left.contiguous()
    right = right.contiguous()

    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Cast to fp16 for higher throughput; Triton kernel will accumulate in fp32
    left_half = left_perm.to(torch.float16)
    right_half = right_perm.to(torch.float16)

    BH = B * hidden_dim
    left_flat = left_half.view(BH, N, N)
    right_flat = right_half.view(BH, N, N)

    out_flat = torch.empty_like(left_flat)             # fp16 buffer

    # -----------------------------------------------------------------
    # 7. Launch Triton kernel
    # -----------------------------------------------------------------
    stride_lm, stride_ln, stride_lk = left_flat.stride()
    stride_rm, stride_rn, stride_rk = right_flat.stride()
    stride_om, stride_on, stride_ok = out_flat.stride()

    # Tuned block sizes for H100 (feel free to modify)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    triton_tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        BH,
        N,
        stride_lm,
        stride_ln,
        stride_lk,
        stride_rm,
        stride_rn,
        stride_rk,
        stride_om,
        stride_on,
        stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 8. Reshape back to (B, N, N, hidden_dim) and cast to fp32
    # -----------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # -----------------------------------------------------------------
    # 9. Hidden → output LayerNorm, gating and final linear
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
