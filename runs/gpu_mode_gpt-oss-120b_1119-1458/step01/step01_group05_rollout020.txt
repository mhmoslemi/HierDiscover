@triton.jit
def batched_gemm_fp16_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_ab, stride_ai, stride_ak, stride_ah,
    stride_bb, stride_bk, stride_bj, stride_bh,
    stride_cb, stride_ci, stride_cj, stride_ch,
    M, N, K,
    hidden_dim: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # batch id
    pid = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    # compute batch and hidden indices
    batch_id = pid // hidden_dim
    d_id = pid % hidden_dim

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out of range
    mask_i = offs_i < M
    mask_j = offs_j < N

    # base pointers for this (batch,d)
    a_base = a_ptr + batch_id * stride_ab + d_id * stride_ah
    b_base = b_ptr + batch_id * stride_bb + d_id * stride_bh
    c_base = c_ptr + batch_id * stride_cb + d_id * stride_ch

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a_ptrs = a_base + offs_i[:, None] * stride_ai + cur_k[None, :] * stride_ak
        b_ptrs = b_base + cur_k[:, None] * stride_bk + offs_j[None, :] * stride_bj

        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)

    # write out
    c_ptrs = c_base + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) – fast Triton implementation
------------------------------------------------
Algorithm
1) Layer‑norm the input tensor `x` (float32).
2) Cast to FP16 and compute the four linear maps (proj + gate) in *two*
   fused matmuls:  `[proj|gate] = x @ [W_proj;W_gate]`.  This halves the
   number of cuBLAS calls.
3) Split the result, apply the sigmoid to the gate part and (optionally)
   fuse the mask with the three element‑wise multiplications using a tiny
   Triton kernel.
4) Core contraction:
        out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
   This is a batched GEMM with batch = B·hidden_dim.  A custom Triton
   kernel computes the batched GEMM directly on the original memory layout,
   avoiding the expensive permute‑and‑contiguous copies used in the
   reference implementation.
5) Cast back to FP32, layer‑norm over the hidden dimension, apply the output
   gate, and a final FP16 linear projection back to `dim`.
All heavy arithmetic is performed in FP16 with FP32 accumulation,
giving the best H100 Tensor‑core performance while the final output
remains FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Element‑wise 3‑way product (mask * projection * gate)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,  # 1‑D contiguous pointers
    size,                           # total number of elements
    BLOCK_SIZE: tl.constexpr,       # compile‑time block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)

    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """Fused a * b * c using Triton (all tensors 1‑D contiguous)."""
    assert a.shape == b.shape == c.shape
    a, b, c = a.contiguous(), b.contiguous(), c.contiguous()
    out = torch.empty_like(a)
    total = a.numel()
    BLOCK = 1024
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)
    fused_mul3_kernel[grid](
        a, b, c, out, total, BLOCK_SIZE=BLOCK
    )
    return out


# ----------------------------------------------------------------------
# 2️⃣  Batched GEMM kernel (FP16, FP32 accumulation)
#    Computes for each hidden channel d:
#        C[b,i,j,d] = Σ_k A[b,i,k,d] * B[b,k,j,d]
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_fp16_kernel(
    a_ptr, b_ptr, c_ptr,
    # strides for A (left)
    stride_ab, stride_ai, stride_ak, stride_ah,
    # strides for B (right, already transposed on the fly)
    stride_bb, stride_bk, stride_bj, stride_bh,
    # strides for C (output)
    stride_cb, stride_ci, stride_cj, stride_ch,
    M, N, K,
    hidden_dim: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_batch = tl.program_id(0)   # flattened (batch, hidden)
    pid_i = tl.program_id(1)       # output‑row block
    pid_j = tl.program_id(2)       # output‑col block

    # de‑flatten batch and hidden indices
    batch_id = pid_batch // hidden_dim
    d_id = pid_batch % hidden_dim

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # base pointers for this (batch, hidden) pair
    a_base = a_ptr + batch_id * stride_ab + d_id * stride_ah
    b_base = b_ptr + batch_id * stride_bb + d_id * stride_bh
    c_base = c_ptr + batch_id * stride_cb + d_id * stride_ch

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- k‑reduction loop -----
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A[i,k]  : shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_i[:, None] * stride_ai + cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # B[k,j] : shape (BLOCK_K, BLOCK_N)  (right is accessed as (k,j))
        b_ptrs = b_base + cur_k[:, None] * stride_bk + offs_j[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # write back
    c_ptrs = c_base + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs,
              acc.to(tl.float16),
              mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# 3️⃣  Entry point – the forward pass of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: shape [B, N, N, dim] (float32, CUDA)
            - mask_tensor:  shape [B, N, N] (float32 or bool, CUDA)
            - weights_dict: model weights (see skeleton)
            - config_dict: contains "dim", "hidden_dim", optional "eps", "nomask"

    Returns:
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 0️⃣ Layer‑norm on the input (FP32 for stability)
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] FP32

    # Cast to FP16 for the heavy linear work
    xh = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 1️⃣  Fuse the two projections (proj + gate) for left and right
    # --------------------------------------------------------------
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                       # (2*hidden, dim)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                      # (2*hidden, dim)

    # Combined linear: [proj | gate]
    left_comb = F.linear(xh, left_comb_w)    # [B,N,N,2*hidden] FP16
    right_comb = F.linear(xh, right_comb_w)

    # Split
    left_proj = left_comb[..., :hidden]               # FP16
    left_gate_pre = left_comb[..., hidden:]           # FP16
    right_proj = right_comb[..., :hidden]
    right_gate_pre = right_comb[..., hidden:]

    left_gate = torch.sigmoid(left_gate_pre)          # FP16
    right_gate = torch.sigmoid(right_gate_pre)        # FP16

    # --------------------------------------------------------------
    # 2️⃣  Apply mask and gates (fused Triton kernel when mask present)
    # --------------------------------------------------------------
    if nomask:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # mask → FP16, broadcast to hidden dim
        mask_fp16 = mask.to(torch.float16)
        mask_exp = mask_fp16.unsqueeze(-1).expand(-1, -1, -1, hidden).contiguous()
        left = fused_mul3(left_proj, mask_exp, left_gate)
        right = fused_mul3(right_proj, mask_exp, right_gate)

    # --------------------------------------------------------------
    # 3️⃣  Core contraction – batched GEMM over the hidden dimension
    # --------------------------------------------------------------
    # Allocate output tensor (FP16)
    out = torch.empty_like(left, dtype=torch.float16)

    # Strides for the three tensors (they are contiguous in the
    # [B,N,N,hidden] layout)
    a_b, a_i, a_k, a_h = left.stride()
    b_b, b_i, b_k, b_h = right.stride()
    c_b, c_i, c_j, c_h = out.stride()   # note: j is the second N in output

    # Total number of GEMM batches = B * hidden
    total_batches = B * hidden

    # Tile sizes – tuned for H100 tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        total_batches,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_fp16_kernel[grid](
        left,                               # a_ptr
        right,                              # b_ptr
        out,                                # c_ptr
        a_b, a_i, a_k, a_h,                 # strides for A
        b_b, b_k, b_i, b_h,                 # strides for B (note: k‑stride then j‑stride)
        c_b, c_i, c_j, c_h,                 # strides for C
        N, N, N,                            # M, N, K
        hidden_dim=hidden,                  # constexpr for batch ↔ hidden mapping
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 4️⃣  Post‑processing (still in FP32)
    # --------------------------------------------------------------
    out_fp32 = out.to(torch.float32)               # [B,N,N,hidden]

    # Output Layer‑norm over hidden dim
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B,N,N,hidden]

    # Output gate (half precision, then back to FP32)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(xh, out_gate_w)).to(torch.float32)   # [B,N,N,hidden]

    out_norm = out_norm * out_gate

    # --------------------------------------------------------------
    # 5️⃣  Final linear projection back to `dim`
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    # Use FP16 for the final matmul
    out_final = F.linear(out_norm.to(torch.float16), to_out_w).to(torch.float32)

    return out_final
