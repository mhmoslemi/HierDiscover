stores the result back in FP16.  The rest of the TriMul pipeline
(layer‑norms, linear projections, sigmoids and the final output projection)
remains in PyTorch – this keeps the implementation simple while still
meeting the requirement of having a Triton‑based component.

Significant speed‑ups over the reference implementation are achieved by:
  * Larger tile sizes (128×128) which improve compute/traffic ratio.
  * Removing the unnecessary cast after `tl.dot` (the dot already returns
    FP32 when inputs are FP16).
  * Using FP16 for the heavy reduction (half the memory bandwidth).
  * A single kernel launch for all (batch,head) pairs (no Python‐level loop).

The final output is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched per‑head matrix multiplication
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                     # pointers
    stride_ap, stride_am, stride_ak,          # strides for A: (P, M, K)
    stride_bp, stride_bk, stride_bn,          # strides for B: (P, K, N)
    stride_cp, stride_cm, stride_cn,          # strides for C: (P, M, N)
    M, N, K,                                  # matrix sizes (all = seq_len)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute a batch of independent matrix multiplications:
        C[p, i, j] = Σ_k A[p, i, k] * B[p, k, j]   (fp16 → fp32 accumulation)
    where p = batch_id * hidden_dim + head_id.
    """
    pid_m = tl.program_id(0)        # block row index
    pid_n = tl.program_id(1)        # block col index
    pid_p = tl.program_id(2)        # batch * head index (P)

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this (p) slice
    a_base = A_ptr + pid_p * stride_ap
    b_base = B_ptr + pid_p * stride_bp
    c_base = C_ptr + pid_p * stride_cp

    # Accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in tiles
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        start_k = k * BLOCK_K
        offs_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A[i, k] tile   (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )   # fp16 load

        # Load B[k, j] tile   (K x N)
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )   # fp16 load

        # Core multiply‑accumulate (tensor‑core dot)
        acc += tl.dot(a, b)  # result already fp32, no cast needed

    # --------------------------------------------------------------
    # Write back result (fp16)
    # --------------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def _triton_matmul(left: torch.Tensor, right: torch.Tensor,
                   seq_len: int, hidden_dim: int,
                   BLOCK_M: int = 128, BLOCK_N: int = 128, BLOCK_K: int = 32):
    """
    Batched matrix multiplication using the Triton kernel above.

    Args:
        left  : [B, N, N, H]  (fp16)
        right : [B, N, N, H]  (fp16)  (the second N dimension is already
                transposed, i.e. right[...,k,j])
        seq_len   : N
        hidden_dim: H
    Returns:
        out : [B, N, N, H] (fp16)
    """
    B = left.shape[0]
    # (B, N, N, H) -> (B, H, N, N) and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right needs a transpose on the two N axes before flattening
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # (B, H, K, J)

    P = B * hidden_dim                                 # total independent matmuls
    left_flat = left_perm.view(P, seq_len, seq_len)    # (P, M, K)
    right_flat = right_perm.view(P, seq_len, seq_len)  # (P, K, N)

    out = torch.empty_like(left_flat)                  # (P, M, N)

    # strides (contiguous tensors)
    stride_ap = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bp = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_cp = out.stride(0)
    stride_cm = out.stride(1)
    stride_cn = out.stride(2)

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,   # blocks over M
        (seq_len + BLOCK_N - 1) // BLOCK_N,   # blocks over N
        P,                                    # one program per (batch, head)
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out,
        stride_ap,
        stride_am,
        stride_ak,
        stride_bp,
        stride_bk,
        stride_bn,
        stride_cp,
        stride_cm,
        stride_cn,
        seq_len,
        seq_len,
        seq_len,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # reshape back to (B, N, N, H)
    out = out.view(B, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point expected by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    This follows the reference PyTorch implementation but replaces the
    O(N³·H) reduction with a highly‑optimized Triton kernel.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input & output channel dimension
    hidden_dim = config["hidden_dim"]  # internal hidden dimension
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # LayerNorm on the input (norm)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]          # (hidden, dim)
    right_proj_weight = weights["right_proj.weight"]        # (hidden, dim)

    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # Optional masking (mask is broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B, N, N] -> [B, N, N, 1]  and cast
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # Gating (sigmoid(Linear(x)))
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # fp32 → fp32
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, N, N, hidden]

    # Apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to fp16 for the expensive N³ reduction
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ------------------------------------------------------------------
    # Triton‑accelerated batched matmul:
    #   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    seq_len = input_tensor.shape[1]         # N
    out_fp16 = _triton_matmul(
        left_fp16,
        right_fp16,
        seq_len=seq_len,
        hidden_dim=hidden_dim,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=32,
    )  # [B, N, N, hidden] (fp16)

    # Convert back to fp32 for the remaining pipeline
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # Output layer‑norm over the hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Apply the output gate (broadcast over hidden dim)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # Final linear projection back to the original dimension (no bias)
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"]   # (dim, hidden)
    out = F.linear(out, to_out_weight)        # [B, N, N, dim]  (float32)

    return out
