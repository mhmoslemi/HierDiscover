"""
TriMul (outgoing) – fast Triton implementation

The core of TriMul is the N‑by‑N‑by‑N contraction
    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
which can be rewritten as a batched matrix multiply
    out_bc = left_bc @ right_bc.T   (left/right ∈ ℝ^{N×N})

Each (b,c) pair is independent, so we treat the combined
batch dimension B·C as a single batch for a GEMM kernel.

Optimisations compared to the reference Triton kernel:
* Loads are kept in fp16 (half) – the previous version forced
  a conversion to fp32 before the dot, disabling Tensor‑Core
  utilization.
* The reduction loop uses a larger K‑tile (64) to cut the number
  of iterations.
* The accumulator stays in fp32 for numeric stability while
  the dot product itself executes on Tensor‑Cores (fp16×fp16→fp16).
* Minimal masking work – any optional mask is applied in the
  PyTorch part before the GEMM.
* The remaining TriMul pipeline (LayerNorm, linear layers,
  gating, output LayerNorm/projection) stays in PyTorch – these
  ops are far cheaper than the N³ GEMM.

The kernel launches one program for each (M‑tile, N‑tile) per
combined batch (B·C).  For N=1024 and C=128 this yields 8192
programs, each handling an 128×128 tile and looping over K‑tiles
of size 64.

The final tensor is returned as float32, matching the reference
implementation.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_gemm_kernel(
    L_ptr,                     # half * (BH, N, N)
    R_ptr,                     # half * (BH, N, N)
    O_ptr,                     # float * (BH, N, N)
    N,                         # seq_len (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    stride_bL, stride_iL, stride_kL,
    stride_bR, stride_iR, stride_kR,
    stride_bO, stride_iO, stride_jO,
):
    """
    Batched GEMM for TriMul:
        out_batch = left_batch @ right_batch.T
    One program handles an (M,N) tile for a specific batch index.
    """
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Tiling layout
    # ------------------------------------------------------------------
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_pid_m * num_pid_n

    batch_idx = pid // tiles_per_batch         # which (b,c) pair
    tile_idx  = pid % tiles_per_batch
    pid_m = tile_idx // num_pid_n
    pid_n = tile_idx % num_pid_n

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to the start of the batch slice
    L_batch = L_ptr + batch_idx * stride_bL
    R_batch = R_ptr + batch_idx * stride_bR
    O_batch = O_ptr + batch_idx * stride_bO

    # ------------------------------------------------------------------
    # Accumulator (float32 for stability)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Reduction over k
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile  (M x K)  – fp16
        L_ptrs = L_batch + offs_m[:, None] * stride_iL + offs_k[None, :] * stride_kL
        L = tl.load(L_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                     # stays fp16

        # Load right tile transposed (K x N)  – we need R[k, j]
        # Original layout of R is (row=j, col=k); we load (k, j)
        R_ptrs = R_batch + offs_k[:, None] * stride_kR + offs_n[None, :] * stride_iR
        R = tl.load(R_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                     # stays fp16

        # Matrix‑multiply the tiles (fp16×fp16 → fp16) and accumulate into fp32
        acc += tl.dot(L, R).to(tl.float32)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    O_ptrs = O_batch + offs_m[:, None] * stride_iO + offs_n[None, :] * stride_jO
    tl.store(O_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (bool or 0/1)
        - weights_dict : mapping str → torch.Tensor
        - config_dict  : must contain "dim", "hidden_dim" and optionally "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) → [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear projections of the normalized input)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to fp16 for the GEMM
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape                     # left_h : [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6) Rearrange to treat (B·hidden_dim) as a single batch dimension:
    #    (B, hidden_dim, N, N) → (BH, N, N)
    # ------------------------------------------------------------------
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()    # [B, hidden_dim, N, N]
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)   # [BH, N, N]   fp16
    right_flat = right_perm.view(BH, N, N) # [BH, N, N]   fp16

    # ------------------------------------------------------------------
    # 7) Allocate output buffer (float32 accumulator)
    # ------------------------------------------------------------------
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=left_flat.device)

    # ------------------------------------------------------------------
    # 8) Triton GEMM launch
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64                         # larger K‑tile → fewer loops
    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    tiles_per_batch = num_pid_m * num_pid_n
    total_programs = BH * tiles_per_batch

    trimul_gemm_kernel[
        (total_programs,),
        num_warps=8,          # good occupancy on H100
        num_stages=2,         # double‑buffered shared memory
    ](
        left_flat,
        right_flat,
        out_flat,
        N,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        # strides for left
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        # strides for right
        right_flat.stride(0),
        right_flat.stride(1),
        right_flat.stride(2),
        # strides for output
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
    )

    # ------------------------------------------------------------------
    # 9) Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 10) Output LayerNorm (hidden_dim)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 11) Output gating
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 12) Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
