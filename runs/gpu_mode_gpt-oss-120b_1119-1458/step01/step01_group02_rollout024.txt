@triton.jit
def _trimul_matmul_kernel(
    A_ptr,
    B_ptr,
    C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0) # i tile
    pid_n = tl.program_id(1) # j tile
    pid_z = tl.program_id(2) # batch*hidden

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        a_ptrs = (
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=tl.float16(0.0))

        b_ptrs = (
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bj
        )
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=tl.float16(0.0))

        acc += tl.dot(a, b) # accumulate into fp32

    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):

    pid_n = tl.program_id(1)
    pid_z = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_n = offs_n < N

    num_k = tl.cdiv(K, BLOCK_K)

                    other=tl.float16(0.0))

"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation
-----------------------------------------------------------------

Algorithm
---------
1. Layer‑Norm over the feature dimension `dim` (half precision).
2. Two linear projections (`left_proj`, `right_proj`) to the hidden
   dimension `hidden_dim` (no bias, half precision).
3. Optional mask multiplication (broadcasted on the hidden axis).
4. Three sigmoid gates are computed from the normalized tensor and
   applied element‑wise to the projections.
5. Heavy contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   i.e. a batched matrix multiplication `C = A @ Bᵀ` for each
   (batch, hidden) pair.  This O(N³) step is performed by a custom
   Triton kernel that works in FP16 (inputs) and accumulates in FP32.
6. Second Layer‑Norm over `hidden_dim`.
7. Multiply by `out_gate` and a final linear projection back to `dim`.
8. Cast the result to float32 (the API expects FP32 output).

The kernel is deliberately tiled (BLOCK_M × BLOCK_N) with a modest
BLOCK_K, uses FP16 loads/stores and Tensor‑Core‑friendly dot products,
and runs with 8 warps per block to fully exploit the H100.
All other operations use native PyTorch kernels, which are already
highly optimized.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ  for many independent batches (batch*hidden)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_fp16_kernel(
    A_ptr, B_ptr, C_ptr,              # pointers
    M, N, K,                          # dimensions (M = batch*hidden, N = seq_len, K = seq_len)
    stride_am, stride_ai, stride_ak,  # strides for A (M,N,K)
    stride_bm, stride_bj, stride_bk,  # strides for B (M,N,K) – note bj = stride of N, bk = stride of K
    stride_cm, stride_ci, stride_cj,  # strides for C (M,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # tile along first N dimension (i)
    pid_n = tl.program_id(1)  # tile along second N dimension (j)
    pid_z = tl.program_id(2)  # batch*hidden index

    # -------------------------------------------------
    # Tile offsets
    # -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32 (Tensor‑Core accumulation)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # -------------------------------------------------
        # Load A tile: shape (BLOCK_M, BLOCK_K) – FP16
        # -------------------------------------------------
        a = tl.load(
            A_ptr
            + pid_z * stride_am
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=tl.float16(0.0),
        )

        # -------------------------------------------------
        # Load B tile as (BLOCK_K, BLOCK_N) – note the transpose
        # -------------------------------------------------
        b = tl.load(
            B_ptr
            + pid_z * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bj,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=tl.float16(0.0),
        )

        # FP16 dot product, accumulation in FP32
        acc += tl.dot(a, b)

    # -------------------------------------------------
    # Write back the FP16 result
    # -------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_z * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that reshapes tensors and launches the Triton kernel
# ----------------------------------------------------------------------
def _batched_matmul_transpose_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using the FP16 Triton kernel above.
    Input tensors shape: [B, N, N, H] (half, contiguous).
    Output tensor shape:   [B, N, N, H] (half, contiguous).
    """
    B, N, _, H = left.shape

    # reshape to [B*H, N, N] (M, N, K) layout expected by the kernel
    A = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    B_ = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Allocate output buffer (half)
    C = torch.empty_like(A, dtype=torch.float16)

    # Strides in elements
    stride_am, stride_ai, stride_ak = A.stride()
    stride_bm, stride_bj, stride_bk = B_.stride()
    stride_cm, stride_ci, stride_cj = C.stride()

    # Tile sizes – tuned for H100 FP16 TensorCores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_matmul_fp16_kernel[grid](
        A, B_, C,
        B * H, N, N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Reshape back to original layout [B, N, N, H]
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return C


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no grads).
    Arguments
    ---------
    data : (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 / bool) – may be ignored
        - weights_dict  : mapping from parameter name → torch.Tensor (float32)
        - config_dict   : contains ``dim`` and ``hidden_dim``; optional ``nomask``.
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # Cast everything to FP16 for speed (except final output)
    # -------------------------------------------------
    x = input_tensor.to(torch.float16)

    # -------------------------------------------------
    # 1️⃣ LayerNorm over `dim` (half precision)
    # -------------------------------------------------
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"].to(torch.float16),
        bias=weights["norm.bias"].to(torch.float16),
    )  # [B, N, N, dim] (half)

    # -------------------------------------------------
    # 2️⃣ Linear projections (no bias) – FP16
    # -------------------------------------------------
    left = torch.nn.functional.linear(
        x, weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(
        x, weights["right_proj.weight"].to(torch.float16)
    )

    # -------------------------------------------------
    # 3️⃣ Optional mask (broadcasted on hidden dimension)
    # -------------------------------------------------
    if not nomask:
        # mask is broadcasted to shape [B,N,N,1]
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # -------------------------------------------------
    # 4️⃣ Gating – compute three sigmoid gates (half precision)
    # -------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16))
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16))
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16))
    )

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction (FP16 Triton kernel)
    # -------------------------------------------------
    out = _batched_matmul_transpose_fp16(left, right)   # [B,N,N,hidden_dim] (half)

    # -------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden_dim (half)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # -------------------------------------------------
    # 7️⃣ Gate and final projection back to `dim`
    # -------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(
        out, weights["to_out.weight"].to(torch.float16)
    )  # [B,N,N,dim] (half)

    # -------------------------------------------------
    # Cast back to FP32 for the public API
    # -------------------------------------------------
    return out.to(torch.float32)
