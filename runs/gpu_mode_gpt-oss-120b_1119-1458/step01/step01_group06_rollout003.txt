def fuse_gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elems,
    stride_left, stride_right, stride_left_gate, stride_right_gate, stride_mask,
    BLOCK_SIZE: tl.constexpr
):

    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    left_val = tl.load(left_ptr + offs * stride_left, mask=mask, other=0.0)

"""
TriMul (outgoing) – fast forward implementation.

Algorithm
---------
1. Layer‐norm the 4‑D input tensor (B,N,N,C).
2. Compute the 5 linear projections (left/right, three gates) in FP32 and
   immediately cast them to FP16.
3. Fuse the three element‑wise operations
        left = left * mask * left_gate
        right = right * mask * right_gate
   with a tiny Triton kernel (1‑D grid).  The mask is broadcasted over the
   hidden dimension by indexing ``mask_flat[idx // H]`` where ``H`` is the
   hidden size.
4. Perform the heavy N³ reduction as a batched matrix multiplication:
        out(b,d,i,j) = Σ_k left(b,d,i,k) * right(b,d,j,k)
   which is exactly ``left @ rightᵀ``.  This is delegated to ``torch.bmm``
   (cuBLAS / Tensor‑cores) after reshaping ``(B,H,N,N) → (B*H,N,N)``.
5. Convert the result back to (B,N,N,H) and apply the output‑layer‑norm,
   the output gate, and the final linear projection.
6. Return the result in FP32.

Only step‑3 (the element‑wise fusion) is written in Triton; all
large‑scale matrix work leverages the highly‑optimised cuBLAS paths.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if any) and the two gating tensors.
# The tensors are all contiguous, 1‑D views of shape
#    [B * N * N * H]    (left, right, left_gate, right_gate)
# and a 1‑D view of the mask [B * N * N].
# ----------------------------------------------------------------------
@triton.jit
def _fuse_gate_mask_kernel(
    left_ptr, right_ptr,          # [total]
    left_gate_ptr, right_gate_ptr,# [total]
    mask_ptr,                     # [mask_len] (ignored when NOMASK=1)
    total_elems: tl.constexpr,   # B*N*N*H
    hidden_dim: tl.constexpr,    # H
    BLOCK_SIZE: tl.constexpr,    # number of elements per program
    NOMASK: tl.constexpr,        # compile‑time flag
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    # load the four main tensors
    left_val      = tl.load(left_ptr      + offs, mask=valid, other=0.0)
    right_val     = tl.load(right_ptr     + offs, mask=valid, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate_val= tl.load(right_gate_ptr+ offs, mask=valid, other=0.0)

    # apply the per‑channel gates
    left_res  = left_val  * left_gate_val
    right_res = right_val * right_gate_val

    # optional mask (broadcasted over the hidden dimension)
    if not NOMASK:
        # compute the index of the corresponding mask element:
        #   mask_idx = flat_index // hidden_dim
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)
        left_res  = left_res  * m
        right_res = right_res * m

    # store the fused results back in‑place
    tl.store(left_ptr  + offs, left_res,  mask=valid)
    tl.store(right_ptr + offs, right_res, mask=valid)


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C]   (float32)
        - mask         : torch.Tensor [B, N, N]    (bool/float)
        - weights      : dict of torch.Tensors containing all linear
                         and layer‑norm parameters (float32)
        - config       : dict with keys
                         * "dim" (int) – input channel dimension C
                         * "hidden_dim" (int) – hidden dimension H
                         * "nomask" (bool, optional)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim        = config["dim"]          # C
    hidden_dim = config["hidden_dim"]   # H
    nomask     = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B,N,N,C]  float32

    # ------------------------------------------------------------------
    # 2) Linear projections & gates (float32 → cast to fp16)
    # ------------------------------------------------------------------
    # left / right projections
    left   = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B,N,N,H]
    right  = F.linear(x, weights["right_proj.weight"]).to(torch.float16)  # [B,N,N,H]

    # gating tensors
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"])).to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Fuse mask + gates with a tiny Triton kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    total_elements = left.numel()                     # B * N * N * H
    # 1‑D contiguous views (no extra memory copy for mask – we use idx//H)
    left_flat       = left.view(-1)
    right_flat      = right.view(-1)
    left_gate_flat  = left_gate.view(-1)
    right_gate_flat = right_gate.view(-1)

    if nomask:
        # dummy mask (will never be read)
        mask_flat = left_flat   # any tensor of same dtype works
        NOMASK = 1
    else:
        # mask is [B,N,N]; we keep it FP16 to match the kernel
        mask_fp16 = mask.to(torch.float16)
        mask_flat = mask_fp16.reshape(-1)   # shape B*N*N
        NOMASK = 0

    BLOCK_SIZE = 8192    # ~8k threads per program (tuned for H100)
    num_blocks  = (total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE
    grid = (num_blocks,)

    _fuse_gate_mask_kernel[grid](
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        mask_flat,
        total_elements,
        hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
        NOMASK=NOMASK,
    )

    # ------------------------------------------------------------------
    # 4) Heavy N³ reduction via batched GEMM (cuBLAS / Tensor‑cores)
    # ------------------------------------------------------------------
    # Reshape to (B*H, N, N) for torch.bmm
    left_batched  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_batched = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # out_batched = left_batched @ right_batchedᵀ  → shape (B*H, N, N)
    out_batched = torch.bmm(left_batched, right_batched.transpose(1, 2))

    # Bring back to (B, N, N, H) and convert to float32 for the remaining ops
    out = out_batched.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)     # [B,N,N,H]  float32

    # ------------------------------------------------------------------
    # 5) Hidden → output LayerNorm, output gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # out_gate: shape [B,N,N,H] – already FP16, promote to FP32 for the multiply
    out_gate_fp32 = out_gate.to(torch.float32)
    out = out * out_gate_fp32

    # final linear projection back to original channel dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
