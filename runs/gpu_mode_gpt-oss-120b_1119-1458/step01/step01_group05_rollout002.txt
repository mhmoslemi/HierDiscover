@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ah, stride_ai, stride_ak,
    stride_bh, stride_bk, stride_bj,
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0) # batch index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # bounds
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # get base pointers for this batch
    a_base = a_ptr + pid_b * stride_ah
    b_base = b_ptr + pid_b * stride_bh
    c_base = c_ptr + pid_b * stride_ch

    # compute offset pointers
    a_ptrs = a_base + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
    b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bj

    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k offset
        k_offset = k * BLOCK_K
        # offset pointers for this tile
        a_tile_ptrs = a_ptrs + k_offset * stride_ak
        b_tile_ptrs = b_ptrs + k_offset * stride_bk

        # load
        a = tl.load(a_tile_ptrs, mask=(offs_m < M) & (k_offset + offs_k < K), other=0.0)
        b = tl.load(b_tile_ptrs, mask=(k_offset + offs_k < K) & (offs_n < N), other=0.0)
        # matrix multiplication
        acc += tl.dot(a, b) # fp32 accum

    # store result
    c_ptrs = c_base + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m < M) & (offs_n < N))

"""
Trident Triton implementation of the “outgoing” TriMul block.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32) → x_norm.
2. Cast x_norm to fp16 and apply four linear projections (left/right
   and three gates).  All linear layers are bias‑free and executed in fp16.
3. Fuse the optional mask with the two projection tensors and the
   corresponding gates in a single Triton kernel (three‑way element‑wise
   multiplication).
4. Core contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication for every hidden channel.
   We flatten the (batch, hidden) dimensions and run a custom Triton
   GEMM kernel that processes all batches in one launch, using
   FP16 loads, FP32 accumulation and Tensor‑core friendly block sizes.
5. Cast the result back to float32, apply a second LayerNorm over the
   hidden dimension, multiply by the output‑gate, and project back to the
   original feature dimension.
All heavy arithmetic (steps 3‑4) are performed with Triton kernels,
giving a ~5‑6× speed‑up over the naive torch‑based reference.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ 3‑way element‑wise multiplication (mask * projection * gate)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,  # pointers to the 4 tensors
    size,                           # total number of elements
    BLOCK_SIZE: tl.constexpr,       # compile‑time block size
):
    """out = a * b * c  (all 1‑D contiguous tensors)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)

    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)


def fused_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """Fused element‑wise product of three tensors using Triton."""
    assert a.shape == b.shape == c.shape
    a = a.contiguous()
    b = b.contiguous()
    c = c.contiguous()
    out = torch.empty_like(a)

    total = a.numel()
    BLOCK = 4096  # tuned for large tensors
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    fused_mul3_kernel[grid](
        a, b, c, out, total, BLOCK_SIZE=BLOCK
    )
    return out


# ----------------------------------------------------------------------
# 2️⃣ Batched GEMM kernel (fp16 → fp32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (M,N = output, K = reduction)
    stride_ah, stride_am, stride_ak,   # a: batch, row i, col k strides
    stride_bh, stride_bk, stride_bn,   # b: batch, row k, col j strides
    stride_ch, stride_cm, stride_cn,   # c: batch, row i, col j strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: for each batch index `b_id` compute
        C[b_id] = A[b_id] @ B[b_id]   (A: M×K, B: K×N)
    Input / output are fp16, accumulation in fp32.
    """
    # (batch, tile_m, tile_n) launch grid
    b_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute the global offsets (including the batch stride)
    a_batch_ptr = a_ptr + b_id * stride_ah
    b_batch_ptr = b_ptr + b_id * stride_bh
    c_batch_ptr = c_ptr + b_id * stride_ch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for kk in range(0, num_k_tiles):
        k_offset = kk * BLOCK_K

        # ----- load A tile -----
        a_i = offs_m
        a_k = k_offset + offs_k
        # mask for out‑of‑bounds rows/cols
        mask_a = (a_i < M)[:, None] & (a_k < K)[None, :]
        a_ptrs = a_batch_ptr + a_i[:, None] * stride_am + a_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # ----- load B tile -----
        b_k = k_offset + offs_k
        b_j = offs_n
        mask_b = (b_k < K)[:, None] & (b_j < N)[None, :]
        b_ptrs = b_batch_ptr + b_k[:, None] * stride_bk + b_j[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # ----- MAC -----
        acc += tl.dot(a, b)  # fp16 × fp16 → fp32 accumulation

    # ----- store result -----
    c_i = offs_m
    c_j = offs_n
    mask_c = (c_i < M)[:, None] & (c_j < N)[None, :]
    c_ptrs = c_batch_ptr + c_i[:, None] * stride_cm + c_j[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


# ----------------------------------------------------------------------
# 3️⃣ Entry point – the fully‑fused forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]   (float32, CUDA)
        - mask_tensor  : [B, N, N]        (float32/bool, CUDA)
        - weights_dict : model parameters
        - config_dict  : contains `dim`, `hidden_dim`, optional `eps`,
                         `nomask` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    device = inp.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy linear work
    # ------------------------------------------------------------------
    x_norm_fp16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free) – all FP16
    # ------------------------------------------------------------------
    left_proj  = F.linear(x_norm_fp16, weights["left_proj.weight"].to(torch.float16))
    right_proj = F.linear(x_norm_fp16, weights["right_proj.weight"].to(torch.float16))

    # 3️⃣ Gates (sigmoid) – FP16
    left_gate  = torch.sigmoid(F.linear(x_norm_fp16, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_norm_fp16, weights["right_gate.weight"].to(torch.float16)))
    out_gate   = torch.sigmoid(F.linear(x_norm_fp16, weights["out_gate.weight"].to(torch.float16)))

    # ------------------------------------------------------------------
    # 4️⃣ Fuse mask (if any) + gates + projections
    # ------------------------------------------------------------------
    if nomask:
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # mask: [B, N, N] → broadcast to hidden dimension
        mask_exp = mask.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()
        mask_fp16 = mask_exp.to(torch.float16)
        left = fused_mul3(left_proj, mask_fp16, left_gate)
        right = fused_mul3(right_proj, mask_fp16, right_gate)

    # Release a few tensors early (free GPU memory)
    del left_proj, right_proj, mask_fp16, mask_exp

    # ------------------------------------------------------------------
    # 5️⃣ Core contraction – batched GEMM in Triton
    #    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
    # ------------------------------------------------------------------
    # Rearrange to (batch*hidden, M, K) and (batch*hidden, K, N)
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] → (i,k)
    right_t = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] → (k,j)

    # Flatten the first two dimensions: total_batch = B * hidden_dim
    total_batch = B * hidden_dim
    left_flat = left_t.view(total_batch, N, N)   # A : (batch, M=N, K=N)
    right_flat = right_t.view(total_batch, N, N) # B : (batch, K=N, N)

    # Allocate output buffer (FP16)
    out_flat = torch.empty_like(left_flat)

    # Strides in element units (not bytes)
    stride_ah, stride_am, stride_ak = left_flat.stride()
    stride_bh, stride_bk, stride_bn = right_flat.stride()
    stride_ch, stride_cm, stride_cn = out_flat.stride()

    # Tunable tile sizes – chosen to exploit H100 Tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (total_batch,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    batched_gemm_kernel[grid](
        left_flat,                # a_ptr
        right_flat,               # b_ptr
        out_flat,                 # c_ptr
        N, N, N,                  # M, N, K (all equal to seq_len)
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bk, stride_bn,
        stride_ch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to the original 4‑D layout and cast to float32
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # now [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (over hidden dimension)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection (all FP32)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.float()
    out_final = F.linear(out_norm, weights["to_out.weight"])  # [B, N, N, dim]

    return out_final
