Pseudo:

```
def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps=1e-5

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps)

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # shape [B, N, N, 1]
        left = left * mask
        right = right * mask

    left = left * left_gate
    right = right * right_gate

    # Cast to half for GEMM
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Reshape for batched GEMM (B * hidden_dim, N, N)
    B = left_h.shape[0]
    N = left_h.shape[1]   # same seq_len
    # left_h shape (B, N, N, hidden_dim)
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()
    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)

    # GEMM using torch.bmm
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # to float32
    out_flat = out_flat.to(torch.float32)

    # Reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # Output layernorm
    out = torch.nn.functional.layer_norm(out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps)

    # Output gating
    out = out * out_gate

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

"""
TriMul (outgoing) forward pass for AlphaFold‑3 – optimized implementation.

Algorithm
---------
1. Layer‑norm the input on the channel dimension ``dim``.
2. Linear projections (no bias) produce the left/right tensors of shape
   ``[B, N, N, hidden_dim]``.
3. Apply optional mask and Sigmoid gates (left_gate, right_gate, out_gate).
4. Cast the gated tensors to FP16 and reshape them to a batched‐matrix
   layout ``(B*hidden_dim, N, N)``.
5. Compute the expensive contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   as a batched GEMM using a custom Triton kernel.  The kernel works on
   half‑precision inputs and accumulates in FP32 (``tl.dot(...,out_dtype=tl.float32)``),
   which lets the H100 Tensor‑cores run at peak throughput.
6. Reshape the result back to ``[B, N, N, hidden_dim]`` and apply a second
   LayerNorm.
7. Multiply by the output‑gate and finish with a linear projection back to
   the original dimension ``dim``.

The Triton kernel is autotuned over several block sizes to obtain the
fastest configuration for a given sequence length.  All other lightweight
operations are performed with PyTorch (which already uses highly‑optimized
CUDA kernels).  The final output is FP32.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM (left @ right.T) used by TriMul
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        # Small matrices – higher occupancy with many warps
        triton.Config({"BLOCK_M": 64, "BLOCK_N": 64, "BLOCK_K": 32},
                      num_stages=3, num_warps=4),
        # Typical case – 128×128 tiles map well onto Tensor‑cores
        triton.Config({"BLOCK_M": 128, "BLOCK_N": 128, "BLOCK_K": 32},
                      num_stages=3, num_warps=8),
        # Very large N – bigger tiles reduce launch overhead
        triton.Config({"BLOCK_M": 256, "BLOCK_N": 256, "BLOCK_K": 32},
                      num_stages=2, num_warps=8),
    ],
    key=["N"],  # choose config based on the sequence length
)
@triton.jit
def trimul_gemm_kernel(
    L_ptr,            # half  * (BH, N, N)  – left  matrix
    R_ptr,            # half  * (BH, N, N)  – right matrix
    O_ptr,            # float * (BH, N, N)  – output accumulator
    N,                # sequence length (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    stride_bL, stride_iL, stride_kL,
    stride_bR, stride_iR, stride_kR,
    stride_bO, stride_iO, stride_jO,
):
    """Batched GEMM ``out = left @ right.T`` for the TriMul contraction.

    Parameters
    ----------
    L_ptr / R_ptr : half pointers to the input matrices.
    O_ptr          : float pointer to the output buffer (FP32 accumulation).
    N              : sequence length.
    stride_*       : corresponding strides of the three tensors.
    BLOCK_M/N/K    : compile‑time tile sizes.
    """
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Compute the 2‑D tile indices for the current batch.
    # --------------------------------------------------------------
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_pid_m * num_pid_n

    batch_idx = pid // tiles_per_batch
    tile_idx = pid % tiles_per_batch
    pid_m = tile_idx // num_pid_n
    pid_n = tile_idx % num_pid_n

    # Offsets inside the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers to the start of the batch slice
    L_batch = L_ptr + batch_idx * stride_bL
    R_batch = R_ptr + batch_idx * stride_bR
    O_batch = O_ptr + batch_idx * stride_bO

    # --------------------------------------------------------------
    # Compute the dot‑product over the reduction dimension k
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile (M × K) in FP16
        L_ptrs = L_batch + offs_m[:, None] * stride_iL + offs_k[None, :] * stride_kL
        L = tl.load(
            L_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=tl.float16(0),
        )  # half

        # right tile (K × N) – we load the transpose on‑the‑fly
        # Original layout: (row=j, col=k).  We need (k, j) for the dot.
        R_ptrs = R_batch + offs_k[:, None] * stride_kR + offs_n[None, :] * stride_iR
        R = tl.load(
            R_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=tl.float16(0),
        )  # half

        # Mixed‑precision dot (FP16 inputs, FP32 accumulation)
        acc += tl.dot(L, R, out_dtype=tl.float32)

    # --------------------------------------------------------------
    # Store the FP32 result tile
    # --------------------------------------------------------------
    O_ptrs = O_batch + offs_m[:, None] * stride_iO + offs_n[None, :] * stride_jO
    tl.store(O_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------

    """
    Triton‑accelerated TriMul (outgoing) forward pass.

    Parameters
    ----------
    data : tuple

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input channel dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → [B, N, N, 1] for broadcasting
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Cast to half‑precision for the heavy contraction
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape                     # [B, N, N, hidden]

    # Bring the hidden dimension into the batch axis so that we have
    # a simple (BH, N, N) tensor for the GEMM.
    # Permute → (B, hidden, N, N) → view → (BH, N, N)
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)   # half
    right_flat = right_perm.view(BH, N, N) # half

    # ------------------------------------------------------------------
    # 6️⃣ Batched GEMM: out = left @ right.T  (FP16 inputs, FP32 acc)
    # ------------------------------------------------------------------
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=left_flat.device)

    # Compute launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    tiles_per_batch = num_pid_m * num_pid_n
    total_programs = BH * tiles_per_batch

    trimul_gemm_kernel[(total_programs,)](
        left_flat,
        right_flat,
        out_flat,
        N,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        # strides for left
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        # strides for right
        right_flat.stride(0),
        right_flat.stride(1),
        right_flat.stride(2),
        # strides for output
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        # autotuned launch parameters
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Reshape back to the original 4‑D layout
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 8️⃣ Output LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Apply the output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 🔟 Final linear projection back to the model dimension ``dim``
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
