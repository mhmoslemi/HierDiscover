@triton.jit
def elementwise_gate_kernel(
    A_ptr, mask_ptr, gate_ptr,
    B_ptr, out_ptr,
    B, N, H,
    stride_A_b, stride_A_i, stride_A_k, stride_A_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK
    offs = block_start + tl.arange(0, BLOCK)
    mask = offs < B * N * N * H
    # compute indices
    b = (offs // (N*N*H))
    rem = offs % (N*N*H)
    i = rem // (N*H)
    rem2 = rem % (N*H)
    k = rem2 // H
    h = rem2 % H

    # compute linear offsets
    a_ptr = A_ptr + b*stride_A_b + i*stride_A_i + k*stride_A_k + h*stride_A_h
    gate_ptr_off = gate_ptr + b*stride_gate_b + i*stride_gate_i + k*stride_gate_k + h*stride_gate_h
    mask_ptr_off = mask_ptr + b*stride_mask_b + i*stride_mask_i + k*stride_mask_k

    a = tl.load(a_ptr, mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr_off, mask=mask, other=0.0)
    gate_val = tl.load(gate_ptr_off, mask=mask, other=0.0)
    out = a * mask_val * gate_val
    tl.store(out_ptr + a_ptr - A_ptr, out, mask=mask)
```

def custom_kernel(data):
    x, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', True)
    device = x.device

    dtype = torch.bfloat16 # compute dtype

    # convert x to dtype
    x = x.to(dtype)

    # layernorm 1
    norm_weight = weights['norm.weight'].to(dtype)
    norm_bias = weights['norm.bias'].to(dtype)
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # linear projections (no bias)
    left_proj_weight = weights['left_proj.weight'].to(dtype) # shape hidden_dim x dim?
    right_proj_weight = weights['right_proj.weight'].to(dtype)
    left = torch.nn.functional.linear(x, left_proj_weight) # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    if not nomask:
        mask_ = mask.to(dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_
        right = right * mask_
    
    # gates
    left_gate_weight = weights['left_gate.weight'].to(dtype)
    right_gate_weight = weights['right_gate.weight'].to(dtype)
    out_gate_weight = weights['out_gate.weight'].to(dtype)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # fuse gating using Triton
    left = _gate_kernel(left, left_gate)  # returns left * left_gate
    right = _gate_kernel(right, right_gate)
    # maybe integrate mask early: left = left * mask (if mask) before gating
    # but we already did mask multiplication earlier.

    # heavy contraction via batched GEMM
    B, N, _, H = left.shape
    # reshape to (B*H,N,N)
    left_view = left.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)
    right_view = right.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)
    out = torch.bmm(left_view, right_view.transpose(1,2))
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # second layernorm
    to_out_norm_weight = weights['to_out_norm.weight'].to(dtype)
    to_out_norm_bias = weights['to_out_norm.bias'].to(dtype)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # out_gate multiplication
    out = out * out_gate

    # final linear projection
    to_out_weight = weights['to_out.weight'].to(dtype)
    out = torch.nn.functional.linear(out, to_out_weight)

    # cast to float32
    out = out.to(torch.float32)

    return out

@triton.jit
def _gate_kernel(
    in_ptr, gate_ptr, out_ptr,
    B, N, H,
    stride_in_b, stride_in_i, stride_in_k, stride_in_h,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # compute indices
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    k = rem2 // H
    h = rem2 % H

    # compute linear offsets
    in_offset = b * stride_in_b + i * stride_in_i + k * stride_in_k + h * stride_in_h
    gate_offset = b * stride_gate_b + i * stride_gate_i + k * stride_gate_k + h * stride_gate_h
    out_offset = b * stride_out_b + i * stride_out_i + k * stride_out_k + h * stride_out_h

    a = tl.load(in_ptr + in_offset, mask=mask, other=0.0)
    g = tl.load(gate_ptr + gate_offset, mask=mask, other=0.0)
    res = a * g
    tl.store(out_ptr + out_offset, res, mask=mask)

"""
TriMul (outgoing) forward pass – fast Triton‑fused gating + batched BF16 GEMM.

Algorithm
---------
1. Layer‑norm the input tensor (float32) → x_norm.
2. Cast x_norm to BF16 (or FP16 on older GPUs) for the heavy work.
3. Linear projections (left/right) to `hidden_dim` without bias.
4. Optional pairwise mask multiplication (broadcasted over the hidden dim).
5. Compute three sigmoid gates from the *float32* normalized tensor.
   The gates are then cast to the compute dtype.
6. Fuse the element‑wise multiplication `proj * gate` for `left` and `right`
   with a tiny Triton kernel that works in‑place (no extra memory).
7. Heavy contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication `C = A @ Bᵀ` where
   `A = left.permute(0,3,1,2).reshape(B*H,N,N)` and
   `B = right.permute(0,3,1,2).reshape(B*H,N,N)`.
   The batched GEMM is performed with `torch.bmm` on BF16/TensorCore
   which is far faster than a naïve Triton matmul.
8. Second LayerNorm over the hidden dimension (still in BF16).
9. Multiply by `out_gate` and a final linear projection back to `dim`.
10. Cast the result to float32 for the caller.

Only step 6 is implemented as a custom Triton kernel (the in‑place gating);
all other ops rely on highly‑optimised PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

# ----------------------------------------------------------------------
# Triton kernel: in‑place elementwise multiply a tensor with its gate.
# ----------------------------------------------------------------------
@triton.jit
def _gate_inplace_kernel(
    a_ptr,                # *TYPE   [B, N, N, H]
    g_ptr,                # *TYPE   same shape as a_ptr
    B, N, H,              # i32 sizes
    stride_a_b, stride_a_i, stride_a_k, stride_a_h,   # strides for a
    stride_g_b, stride_g_i, stride_g_k, stride_g_h,   # strides for g
    BLOCK: tl.constexpr,  # number of elements processed by one program
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # ---- decode linear index into (b,i,k,h) ----
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    k = rem2 // H
    h = rem2 % H

    # ---- compute the flat offsets into the tensors ----
    a_offset = (b * stride_a_b + i * stride_a_i +
                k * stride_a_k + h * stride_a_h)
    g_offset = (b * stride_g_b + i * stride_g_i +
                k * stride_g_k + h * stride_g_h)

    a_val = tl.load(a_ptr + a_offset, mask=mask, other=0.0)
    g_val = tl.load(g_ptr + g_offset, mask=mask, other=0.0)

    tl.store(a_ptr + a_offset, a_val * g_val, mask=mask)


def _apply_gate_inplace(tensor: torch.Tensor, gate: torch.Tensor) -> None:
    """
    In‑place elementwise multiplication ``tensor *= gate`` using Triton.
    Both tensors must have the same shape and dtype (BF16/FP16).
    """
    B, N, _, H = tensor.shape
    stride_a = tensor.stride()
    stride_g = gate.stride()
    total = B * N * N * H
    BLOCK = 1024  # arbitrary, works well for typical sizes
    grid = ( (total + BLOCK - 1) // BLOCK, )
    _gate_inplace_kernel[grid](
        tensor,
        gate,
        B, N, H,
        stride_a[0], stride_a[1], stride_a[2], stride_a[3],
        stride_g[0], stride_g[1], stride_g[2], stride_g[3],
        BLOCK=BLOCK,
        num_warps=4,
    )


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor   : torch.Tensor [B, N, N] (float32, 0/1)
        - weights_dict  : mapping name → torch.Tensor (float32)
        - config_dict   : ``dim``, ``hidden_dim``, optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32, same device as input).
    """
    # -------------------------------------------------
    # Unpack arguments
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # -------------------------------------------------
    # Choose compute dtype (BF16 on H100, fallback to FP16)
    # -------------------------------------------------
    if torch.cuda.is_available() and torch.cuda.get_device_capability(device) >= (9, 0):
        compute_dtype = torch.bfloat16
    else:
        compute_dtype = torch.float16

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → keep high‑precision)
    # -------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32

    # -------------------------------------------------
    # Cast to compute dtype for all heavy work
    # -------------------------------------------------
    x = x_norm.to(compute_dtype)

    # -------------------------------------------------
    # 2️⃣ Linear projections (no bias) in compute dtype
    # -------------------------------------------------
    left = torch.nn.functional.linear(
        x,
        weights["left_proj.weight"].to(compute_dtype)
    )  # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(
        x,
        weights["right_proj.weight"].to(compute_dtype)
    )

    # -------------------------------------------------
    # 3️⃣ Optional pairwise mask (broadcast over hidden dim)
    # -------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(compute_dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -------------------------------------------------
    # 4️⃣ Gating (sigmoid) – computed in float32 then cast
    # -------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["left_gate.weight"])
    ).to(compute_dtype)  # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["right_gate.weight"])
    ).to(compute_dtype)
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["out_gate.weight"])
    ).to(compute_dtype)  # [B,N,N,hidden_dim]

    # -------------------------------------------------
    # 5️⃣ Fuse projection × gate using Triton (in‑place)
    # -------------------------------------------------
    _apply_gate_inplace(left, left_gate)   # left ← left * left_gate
    _apply_gate_inplace(right, right_gate) # right ← right * right_gate

    # -------------------------------------------------
    # 6️⃣ Heavy contraction (batched GEMM) – BF16/FP16 TensorCores
    # -------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_view = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_view = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication: C = A @ Bᵀ
    out_b = torch.bmm(left_view, right_view.transpose(1, 2))
    # reshape back to [B, N, N, H]
    out = out_b.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm over hidden_dim (still in compute dtype)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].to(compute_dtype),
        bias=weights["to_out_norm.bias"].to(compute_dtype),
    )

    # -------------------------------------------------
    # 8️⃣ Apply out_gate (element‑wise)
    # -------------------------------------------------
    out = out * out_gate

    # -------------------------------------------------
    # 9️⃣ Final projection back to `dim`
    # -------------------------------------------------
    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].to(compute_dtype)
    )  # [B, N, N, dim] in compute dtype

    # -------------------------------------------------
    # 10️⃣ Cast back to float32 and return
    # -------------------------------------------------
    return out.to(torch.float32)
