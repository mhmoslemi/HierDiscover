def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    # no transpose
    assert left.is_cuda and right.is_cuda
    B, N, _, H = left.shape

    out = torch.empty_like(left)   # same dtype

    # strides (in elements)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rj, s_rh = right.stride()

    # For right we treat K dimension (rows) as third dimension (original j)
    stride_right_i = s_rj   # row stride (k)
    stride_right_j = s_ri   # col stride (j)

    s_ob, s_oi, s_oj, s_oh = out.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, stride_right_i, stride_right_j, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )
    return out

"""
TriMul (outgoing) forward pass – highly optimized Triton implementation.

Algorithm
---------
1️⃣ Layer‑norm the input (float32 → float16 for the heavy lifting).
2️⃣ Linear projections (no bias) → `left` and `right` (half‑precision).
3️⃣ Optional mask is broadcast and applied in‑place.
4️⃣ Gating: sigmoid of linear projections, applied element‑wise.
5️⃣ Core N³ contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM `left @ rightᵀ` per (batch, head) pair.
   **No materialised transposition of `right`** – the kernel swaps the
   stride arguments on‑the‑fly, eliminating an O(B·N²·H) memory copy.
6️⃣ Output LayerNorm (still in half‑precision) and gate.
7️⃣ Final linear projection back to `dim` and cast to float32 for the result.

All heavy numeric work (step 5) runs on Tensor‑cores via a custom Triton
kernel; the remaining ops use the highly‑optimised PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the N³ contraction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,           # const float16 *
    right_ptr,          # const float16 *
    out_ptr,            # float16 *
    B, N, H,            # int32 metadata
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched matrix multiplication:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    The right tensor is accessed with swapped strides so that the
    implicit transpose (over the two N axes) is performed on‑the‑fly.
    """
    pid_head = tl.program_id(0)   # encodes (batch, hidden)
    pid_m    = tl.program_id(1)   # block row (i)
    pid_n    = tl.program_id(2)   # block col (j)

    # ------------------------------------------------------------------
    # Decode batch (b) and head (h) from the flattened head dimension
    # ------------------------------------------------------------------
    total_heads = B * H
    b = pid_head // H
    h = pid_head % H

    # ------------------------------------------------------------------
    # Offsets for the output tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K in blocks
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a BLOCK_M×BLOCK_K tile from `left` (rows i, cols k)
        left_ptrs = left_ptr + \
            b * stride_left_b + h * stride_left_h + \
            offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # Load a BLOCK_K×BLOCK_N tile from `rightᵀ` (rows k, cols j)
        # NOTE: `stride_right_i` is the stride for the K‑dimension (row)
        # and `stride_right_j` is the stride for the J‑dimension (col).
        right_ptrs = right_ptr + \
            b * stride_right_b + h * stride_right_h + \
            offs_k[:, None] * stride_right_i + offs_n[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # Accumulate dot product over K
        acc += tl.dot(left, right)

    # ------------------------------------------------------------------
    # Store the output tile (convert back to float16)
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_out_b + h * stride_out_h + \
        offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton‑accelerated batched GEMM for the outgoing TriMul.
    Arguments
    ---------
    left  : torch.Tensor of shape [B, N, N, H], dtype torch.float16
    right : torch.Tensor of shape [B, N, N, H], dtype torch.float16
            (no explicit transposition – the kernel swaps strides internally)

    Returns
    -------
    torch.Tensor of shape [B, N, N, H], dtype torch.float16
    """
    assert left.is_cuda and right.is_cuda, "Tensors must be on CUDA"
    B, N, _, H = left.shape

    # Allocate the output tensor (same dtype/device/shape as left)
    out = torch.empty_like(left)

    # Strides (in elements) for left, right and output
    s_lb, s_li, s_lk, s_lh = left.stride()          # (B, i, k, h)
    s_rb, s_ri, s_rj, s_rh = right.stride()         # (B, i, j, h)
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # For the implicit transpose we need:
    #   * row stride = original stride of the *third* dimension (j)
    #   * col stride = original stride of the *second* dimension (i)
    stride_right_i = s_rj   # row stride → K‑dimension after transpose
    stride_right_j = s_ri   # column stride → J‑dimension after transpose

    # Tunable tile sizes – work well on Hopper/H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                                     # head dimension (batch × hidden)
        (N + BLOCK_M - 1) // BLOCK_M,              # number of row tiles
        (N + BLOCK_N - 1) // BLOCK_N,              # number of col tiles
    )

    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, stride_right_i, stride_right_j, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,          # more warps => better H100 utilisation
        num_stages=4,         # double‑buffering to hide latency
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : dict of model parameters (torch.Tensor)
        - config_dict  : contains at least 'dim' and 'hidden_dim'

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper: cast a weight to half‑precision on the correct device
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Load and cast weights
    # ------------------------------------------------------------------
    norm_w   = to_half(weights["norm.weight"])
    norm_b   = to_half(weights["norm.bias"])
    left_proj_w   = to_half(weights["left_proj.weight"])
    right_proj_w  = to_half(weights["right_proj.weight"])
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 → half for the subsequent heavy ops)
    # ------------------------------------------------------------------
    x_norm_fp32 = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w.to(torch.float32),
        bias=norm_b.to(torch.float32),
        eps=1e-5,
    )                                   # [B, N, N, dim]  (float32)
    x = x_norm_fp32.to(torch.float16)   # half‑precision view for the rest

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)    # [B, N, N, hidden_dim] (half)
    right = F.linear(x, right_proj_w)   # [B, N, N, hidden_dim] (half)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask may be bool or float; convert to half and broadcast
        mask_f = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # [B,N,N,hidden_dim]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # [B,N,N,hidden_dim]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ N³ contraction via Triton (batched GEMM: left @ rightᵀ)
    # ------------------------------------------------------------------
    out = tri_mul_triton(left, right)   # [B, N, N, hidden_dim] (half)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (still in half‑precision)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )                                   # [B, N, N, hidden_dim] (half)

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate               # element‑wise gating (half)
    out = F.linear(out, to_out_w)      # final linear (half) → [B,N,N,dim]

    # ------------------------------------------------------------------
    # Return as float32 (as required by the specification)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
