"""
TriMul (outgoing) – highly‑optimized forward pass.

Algorithm
---------
1. Layer‑norm the input (float32) → x_norm.
2. Cast to float16 and apply two bias‑free linear projections (left/right).
3. Optional mask (broadcast on the hidden dimension) is applied to left/right.
4. Compute three gating scalars (sigmoid(linear(x))) and multiply them into left/right/out.
5. Heavy N³ contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matrix‑multiply (left @ rightᵀ) for each (b,h) using Triton.
   The transposition of the second and third axes of `right` is performed *in‑place*
   via strided loads – no extra memory copy is required.
6. Apply a second Layer‑norm (hidden_dim) → out_norm.
7. Multiply by the output‑gate and a final linear projection back to dim.
8. Return float32 tensor.

The Triton kernel works on half‑precision data, accumulates in float32,
and supports arbitrary batch size, sequence length and hidden dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the N³ contraction (left @ rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # const float16 *
    right_ptr,         # const float16 *
    out_ptr,           # float16 *
    B, N, H,           # int32 metadata
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication: out = left @ rightᵀ for each (b, h)."""
    pid_head = tl.program_id(0)   # encodes (batch, hidden)
    pid_m = tl.program_id(1)      # block row index (i)
    pid_n = tl.program_id(2)      # block col index (j)

    total_heads = B * H
    b = pid_head // H            # batch index
    h = pid_head % H             # hidden index

    # Offsets for the output tile (i, j)
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # --------------------------------------------------------------
        # Load left tile: rows i, cols k
        # --------------------------------------------------------------
        left_ptrs = left_ptr + \
            b * stride_left_b + h * stride_left_h + \
            offs_m[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # --------------------------------------------------------------
        # Load right tile transposed on the two N axes:
        # we need rows = k, cols = j → use stride_right_k for rows,
        # stride_right_i for columns (no explicit transpose).
        # --------------------------------------------------------------
        right_ptrs = right_ptr + \
            b * stride_right_b + h * stride_right_h + \
            offs_k[:, None] * stride_right_k + \
            offs_n[None, :] * stride_right_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # --------------------------------------------------------------
        # Accumulate dot product over K
        # --------------------------------------------------------------
        acc += tl.dot(left, right)

    # --------------------------------------------------------------
    # Store output tile (cast back to fp16)
    # --------------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_out_b + h * stride_out_h + \
        offs_m[:, None] * stride_out_i + \
        offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton‑accelerated contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Parameters
    ----------
    left : torch.Tensor   (B, N, N, H)  float16
        Tensor after left projection and gating.
    right : torch.Tensor  (B, N, N, H)  float16
        Tensor after right projection and gating.

    Returns
    -------
    out : torch.Tensor    (B, N, N, H)  float16
        Result of the N³ contraction.
    """
    assert left.is_cuda and right.is_cuda
    B, N, _, H = left.shape

    # Output allocation (same dtype and device)
    out = torch.empty_like(left)

    # Strides in element units (required by Triton)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Tuned block sizes for H100 – adjust if needed.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                                   # head dimension (batch × hidden)
        (N + BLOCK_M - 1) // BLOCK_M,            # row tiles
        (N + BLOCK_N - 1) // BLOCK_N,            # col tiles
    )

    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,               # H100 benefits from 8 warps per block
        num_stages=3,
    )
    return out


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : dict of torch.Tensors with all linear / norm weights
        - config_dict  : dict containing at least 'dim' and 'hidden_dim'

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # Helper to move weights to the correct device/dtype (half for compute)
    # --------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # --------------------------------------------------------------
    # Extract model parameters
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w   = to_half(weights["left_proj.weight"])
    right_proj_w  = to_half(weights["right_proj.weight"])
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )   # [B, N, N, C] (float32)

    # --------------------------------------------------------------
    # Cast to half for the heavy operations
    # --------------------------------------------------------------
    x_half = x_norm.to(torch.float16)

    # --------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # --------------------------------------------------------------
    left = F.linear(x_half, left_proj_w)    # [B, N, N, hidden_dim]
    right = F.linear(x_half, right_proj_w)

    # --------------------------------------------------------------
    # 3. Apply optional mask (broadcast on hidden_dim)
    # --------------------------------------------------------------
    if mask is not None:
        # Ensure mask has dtype half and shape [B, N, N, 1]
        mask_f = mask.to(dtype=torch.float16).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4. Gating (sigmoid of linear layers)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_half, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_half, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_half, out_gate_w))   # [B,N,N,hidden_dim]

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. N³ contraction via Triton (batched matmul left @ rightᵀ)
    #    No explicit transpose – the kernel performs strided loads.
    # --------------------------------------------------------------
    left  = left.contiguous()
    right = right.contiguous()
    out = tri_mul_triton(left, right)   # [B, N, N, hidden_dim] (float16)

    # --------------------------------------------------------------
    # 6. Output LayerNorm over hidden_dim (float32 internally)
    # --------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )
    out_norm = out_norm_fp32.to(torch.float16)

    # --------------------------------------------------------------
    # 7. Apply output gate and final projection back to dim
    # --------------------------------------------------------------
    gated = out_norm * out_gate
    final = F.linear(gated, to_out_w)   # [B, N, N, dim] (float16)

    # --------------------------------------------------------------
    # 8. Return in float32 as required
    # --------------------------------------------------------------
    return final.to(torch.float32)
```
