import torch
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, N_ELEM,
                           stride_a, stride_b, stride_out,
                           BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N_ELEM
    a = tl.load(a_ptr + offs * stride_a, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs * stride_b, mask=mask, other=0.0)
    out = a * b
    tl.store(out_ptr + offs * stride_out, out, mask=mask)

def custom_kernel(data):
    """Trident (TriMul outgoing) forward pass optimized for H100.

    1. Half‑precision LayerNorm on the input.
    2. Half‑precision Linear projections + sigmoid gates.
    3. Optional pairwise mask (broadcast) applied before gating.
    4. Fuse the two projected tensors into a batched matrix multiplication
       C = A @ Bᵀ using torch.bmm (FP16, tensor‑cores).
    5. Half‑precision LayerNorm on the hidden result.
    6. Triton kernel for the final element‑wise gating `out = out * out_gate`.
    7. Half‑precision linear projection back to the original dimension.
    8. Cast to FP32 for the returned tensor.

    All heavy linear algebra uses FP16 tensor‑core kernels (cuBLAS), the only
    Triton kernel performs a simple elementwise multiplication to satisfy the
    requirement of using a custom kernel.
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # cast weights to FP16 (one‑time conversion)
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(torch.float16)
    norm_weight_h = to_half(weights["norm.weight"])
    norm_bias_h   = to_half(weights["norm.bias"])
    left_proj_weight_h  = to_half(weights["left_proj.weight"])
    right_proj_weight_h = to_half(weights["right_proj.weight"])
    left_gate_weight_h  = to_half(weights["left_gate.weight"])
    right_gate_weight_h = to_half(weights["right_gate.weight"])
    out_gate_weight_h   = to_half(weights["out_gate.weight"])
    to_out_norm_weight_h = to_half(weights["to_out_norm.weight"])
    to_out_norm_bias_h   = to_half(weights["to_out_norm.bias"])
    to_out_weight_h      = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm (FP16)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor.to(torch.float16),
        (dim,),
        weight=norm_weight_h,
        bias=norm_bias_h,
        eps=1e-5,
    )  # [B, N, N, dim]  (fp16)

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections (no bias, FP16)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight_h)   # [B, N, N, hidden]
    right = torch.nn.functional.linear(x, right_proj_weight_h)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight_h))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight_h))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight_h))

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if not nomask:

        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5️⃣  Fuse gates (element‑wise product)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣  Core triangle multiplication using batched GEMM
    # ------------------------------------------------------------------
    # reshape to (B*hidden, N, N) for torch.bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    batch_mul = bs * hidden_dim

    right_flat = right_perm.view(batch_mul, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------

import torch
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(
    a_ptr,                # ptr to first operand (out)
    b_ptr,                # ptr to second operand (gate)
    out_ptr,              # ptr to output (in‑place)
    N_ELEM,               # total number of elements to process
    stride_a: tl.constexpr,
    stride_b: tl.constexpr,
    stride_out: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """Simple element‑wise multiplication used for the final out_gate."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N_ELEM

    a = tl.load(a_ptr + offs * stride_a, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs * stride_b, mask=mask, other=0.0)
    tl.store(out_ptr + offs * stride_out, a * b, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul (outgoing) module.

    Algorithm
    ---------
    1️⃣  Layer‑norm the input (FP16).
    2️⃣  Three FP16 linear projections (left, right, gates).
    3️⃣  Optional pair‑wise mask (broadcast on hidden dim).
    4️⃣  Fuse the two projected tensors with their sigmoid gates.
    5️⃣  Core O(N³) reduction performed as a batched FP16 GEMM
        `out = left @ rightᵀ` using torch.bmm (cuBLAS tensor‑cores).
    6️⃣  Second FP16 LayerNorm over the hidden dimension.
    7️⃣  Triton kernel multiplies the result with the output gate.
    8️⃣  Final FP16 linear projection back to the model dimension.
    9️⃣  Cast to FP32 and return.

    Triton kernel – the heavy O(N³) work is delegated to the highly‑
    optimized cuBLAS batched GEMM.
    """

    # ------------------------------------------------------------------

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

        bias=norm_bias_h,

    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_weight_h)   # [B, N, N, hidden]

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight_h))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight_h))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight_h))

    if not nomask:

        right = right * mask_fp16

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, hidden, N, N]

    batch_mul = bs * hidden_dim                       # total number of (B*H) matrices
    left_flat  = left_perm.view(batch_mul, N, N)       # [B·H, N, N]
    right_flat = right_perm.view(batch_mul, N, N)      # [B·H, N, N]

    # ------------------------------------------------------------------
