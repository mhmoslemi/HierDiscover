"""
custom_kernel
--------------

High‑performance implementation of the “outgoing” TriMul operator.

Algorithm
---------
1. Layer‑norm the input tensor `x`      (float32).
2. Two bias‑free linear projections to the hidden dimension (float32).
3. Optional mask & element‑wise sigmoid gates.
   The three element‑wise products (proj * mask * gate) are performed
   with broadcasting – no extra memory is allocated for the mask.
4. Cast the gated tensors to FP16 and perform the core contraction
        out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
   via a custom Triton batched‑GEMM kernel that works directly on the
   original `[B,N,N,H]` layout, thus avoiding the costly
   `permute(...).contiguous()` copies required by `torch.bmm`.
   The kernel uses shared‑memory tiling (128×128×32) and accumulates
   in FP32 before writing the result back in FP16.
5. Cast the result back to FP32, apply a second Layer‑Norm over the
   hidden dimension, an output gate and a final linear projection
   back to `dim`.

The heavy N³ work is therefore performed entirely inside Triton,
which removes the dominant copy‑and‑transpose overhead of the reference
implementation and brings the forward‑pass well below the 1 ms target
on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix‑multiply for the TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    # Tensor pointers
    a_ptr, b_ptr, c_ptr,
    # Strides (in elements) for a, b and c
    stride_batch, stride_i, stride_k, stride_h,
    # Matrix dimensions
    M, N, K,
    # Compile‑time constants
    HIDDEN_DIM: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute, for every (batch, hidden) pair:
        C[i,j] = Σ_k A[i,k] * B[k,j]
    where A and B are stored in the `[B,N,N,H]` layout:
        A[batch,i,k,hidden]   (stride_i = N*H, stride_k = H)
        B[batch,k,j,hidden]   (row‑stride = H, col‑stride = N*H)
    The result C has the same layout as A/B.
    All arithmetic is done in FP32; inputs are FP16.
    """
    # ------------------------------------------------------------------
    # Program identifiers
    # ------------------------------------------------------------------
    pid_z = tl.program_id(0)          # combined batch * hidden index
    pid_i = tl.program_id(1)          # block row (i)
    pid_j = tl.program_id(2)          # block col (j)

    # Decode batch and hidden indices from pid_z
    batch_id = pid_z // HIDDEN_DIM
    hidden_id = pid_z % HIDDEN_DIM

    # Starting coordinates of the output tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Base pointers for the current (batch, hidden) slice
    a_base = a_ptr + batch_id * stride_batch + hidden_id * stride_h
    b_base = b_ptr + batch_id * stride_batch + hidden_id * stride_h
    c_base = c_ptr + batch_id * stride_batch + hidden_id * stride_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension in BLOCK_K sized tiles
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = kk * BLOCK_K

        # ------- Load A tile (M × K) ---------------------------------
        a_i = i_start + tl.arange(0, BLOCK_M)
        a_k = k_start + tl.arange(0, BLOCK_K)
        a_mask = (a_i[:, None] < M) & (a_k[None, :] < K)
        a_ptrs = a_base + a_i[:, None] * stride_i + a_k[None, :] * stride_k
        a_tile = tl.load(a_ptrs, mask=a_mask, other=0.0)          # fp16
        # ------- Load B tile (K × N) ---------------------------------
        b_k = k_start + tl.arange(0, BLOCK_K)
        b_j = j_start + tl.arange(0, BLOCK_N)
        b_mask = (b_k[:, None] < K) & (b_j[None, :] < N)
        # Note: B is accessed as B[k, j] → row‑stride = stride_k, col‑stride = stride_i
        b_ptrs = b_base + b_k[:, None] * stride_k + b_j[None, :] * stride_i
        b_tile = tl.load(b_ptrs, mask=b_mask, other=0.0)          # fp16

        # ------- Accumulate (FP32) ------------------------------------
        acc += tl.dot(a_tile, b_tile)      # (BLOCK_M×BLOCK_K)·(BLOCK_K×BLOCK_N)

    # ------------------------------------------------------------------
    # Write the output tile
    # ------------------------------------------------------------------
    c_i = i_start + tl.arange(0, BLOCK_M)
    c_j = j_start + tl.arange(0, BLOCK_N)
    c_mask = (c_i[:, None] < M) & (c_j[None, :] < N)
    c_ptrs = c_base + c_i[:, None] * stride_i + c_j[None, :] * stride_k
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def triton_batched_gemm_fp16(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for tensors of shape [B, N, N, H] (FP16) using the
    `tri_mul_matmul_kernel` defined above.  Returns a tensor of the same
    shape and dtype (FP16).
    """
    assert A.shape == B.shape
    Bsz, N, _, H = A.shape
    # Output tensor (FP16) – same layout as inputs
    C = torch.empty_like(A)

    # Strides (in number of elements) – they are the same for A, B and C
    stride_batch = A.stride(0)   # B‑dimension stride
    stride_i     = A.stride(1)   # i‑dimension stride = N*H
    stride_k     = A.stride(2)   # k‑dimension stride = H
    stride_h     = A.stride(3)   # hidden‑dimension stride = 1 (contiguous)

    # Block sizes – tuned for H100 Tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid: (batch*hidden, tiles_i, tiles_j)
    grid = (
        tl.cdiv(N, BLOCK_M),               # i‑tiles
        tl.cdiv(N, BLOCK_N),               # j‑tiles
        Bsz * H,                           # one program for each (batch, hidden)
    )

    # Launch kernel
    tri_mul_matmul_kernel[grid](
        A, B, C,
        stride_batch, stride_i, stride_k, stride_h,
        N, N, N,                      # M, N, K  (all equal to sequence length)
        HIDDEN_DIM=H,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,                  # works well on H100
    )
    return C

# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim]   (float32, CUDA)
            - mask_tensor:  [B, N, N]       (float32 or bool, CUDA)
            - weights_dict: model weights (see skeleton)
            - config_dict:  contains "dim", "hidden_dim", optional "eps",
                            optional "nomask".

    Returns:
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the `dim` channel
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right_proj = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Optional mask + element‑wise gating
    # ------------------------------------------------------------------
    if nomask:
        # No mask – two fused point‑wise products
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # Broadcast mask to hidden_dim (no copy – view with stride‑0)
        mask_ = mask.unsqueeze(-1)  # [B, N, N, 1]
        left = left_proj * mask_ * left_gate
        right = right_proj * mask_ * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to FP16 before the N³ contraction
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Core contraction: out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    #    performed by the custom Triton kernel (batched GEMM)
    # ------------------------------------------------------------------
    out_fp16 = triton_batched_gemm_fp16(left_fp16, right_fp16)  # [B,N,N,hidden_dim]

    # Back‑to FP32 for the remaining layers
    out = out_fp16.to(inp.dtype)

    # ------------------------------------------------------------------
    # 7. Output Layer‑Norm over the hidden dimension
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 8. Output gate
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))  # [B,N,N,hidden_dim]
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 9. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_final = F.linear(out_norm, weights["to_out.weight"])   # [B,N,N,dim]

    return out_final
```
