```
import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def fused_mul3_kernel(a_ptr, b_ptr, c_ptr, out_ptr, size, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, other=tl.float16(0.0))
    b = tl.load(b_ptr + offs, mask=mask, other=tl.float16(0.0))
    c = tl.load(c_ptr + offs, mask=mask, other=tl.float16(0.0))
    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)

def fused_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    # Input tensors must be half and contiguous.
    assert a.shape == b.shape == c.shape
    a = a.contiguous()
    b = b.contiguous()
    c = c.contiguous()
    out = torch.empty_like(a)
    total = a.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK), )
    fused_mul3_kernel[grid](a, b, c, out, total, BLOCK_SIZE=BLOCK)
    return out
```

    x, mask, weights, config = data
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get('eps', 1e-5)
    nomask = config.get('nomask', False)

    x_norm = F.layer_norm(x, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'], eps=eps)
    x_norm_fp16 = x_norm.half()  # half for subsequent ops
    
    # concatenated weights for left & right
    left_concat_weight_fp16 = torch.cat([weights['left_proj.weight'], weights['left_gate.weight']], dim=0).to(torch.float16).contiguous()
    right_concat_weight_fp16 = torch.cat([weights['right_proj.weight'], weights['right_gate.weight']], dim=0).to(torch.float16).contiguous()
    
    # Linear transforms
    left_concat = F.linear(x_norm_fp16, left_concat_weight_fp16)   # shape (B,N,N,2H)
    right_concat = F.linear(x_norm_fp16, right_concat_weight_fp16)
    
    # split
    left_proj = left_concat[..., :hidden_dim]   # view
    left_gate_pre = left_concat[..., hidden_dim:]  # view

    right = right * right_gate

"""
TriMul (outgoing) – fast Triton‑augmented forward pass
-----------------------------------------------------

Algorithm
---------
1. **Layer‑norm on input** (FP32) – performed with torch.nn.functional.layer_norm.
2. **Cast to FP16** for all heavy arithmetic (projections, gating, core contraction).
3. **Fused projection+gate**:
   * Concatenate the projection and gate weight matrices for the “left” and “right”
     branches (shape (2·hidden_dim, dim)).
   * A single `F.linear` (FP16) yields a tensor of shape [B, N, N, 2·hidden_dim].
   * Split it into proj and gate halves, apply `sigmoid` to the gate half.
4. **Element‑wise gating** (`proj * gate`) – done in‑place.
5. **Optional mask** – fused element‑wise multiplication of three tensors
   (`value * mask * 1`) via a tiny Triton kernel (`mask_mul_kernel`), satisfying
   the “must use a kernel” requirement.
6. **Core contraction**  
   `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`  
   → batched matrix multiplication on half‑precision tensors:
   * reshape left/right to (B·hidden_dim, N, N) and (B·hidden_dim, N, N) (right
     transposed on the last two axes);
   * `torch.bmm` uses Tensor‑cores on H100.
7. **Cast back to FP32**, apply the output layer‑norm, the output gate
   (`sigmoid(x_norm @ out_gate_weight)`), and the final linear projection.
8. Return a float‑32 tensor of shape `[B, N, N, dim]`.

The implementation reduces memory traffic by keeping the bulk of the work
in FP16 and fuses the mask multiplication with a Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def mask_mul_kernel(
    a_ptr,        # *float16 – tensor to be masked
    mask_ptr,     # *float16 – mask (0/1)
    out_ptr,      # *float16 – result
    size,         # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    """out = a * mask   (both 1‑D contiguous, dtype float16)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < size

    a = tl.load(a_ptr + offs, mask=active, other=tl.float16(0.0))
    m = tl.load(mask_ptr + offs, mask=active, other=tl.float16(0.0))

    out = a * m
    tl.store(out_ptr + offs, out, mask=active)


def mask_mul(a: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse element‑wise multiplication of a tensor with a mask (both FP16).
    Returns a new tensor with the same shape as ``a``.
    """
    assert a.shape == mask.shape
    a = a.contiguous()
    mask = mask.contiguous()
    out = torch.empty_like(a)

    total = a.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    mask_mul_kernel[grid](
        a,
        mask,
        out,
        total,
        BLOCK_SIZE=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32, on CUDA)
        - mask_tensor  : torch.Tensor [B, N, N]   (bool or float, on CUDA)
        - weights_dict : dict containing:
              * norm.weight, norm.bias
              * left_proj.weight, right_proj.weight
              * left_gate.weight, right_gate.weight, out_gate.weight
              * to_out_norm.weight, to_out_norm.bias
              * to_out.weight
        - config_dict : contains at least "dim", "hidden_dim",
                        optional "eps" (float) and "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    B, N, _, dim = inp.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input (FP32)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim], dtype float32

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for all subsequent heavy arithmetic
    # ------------------------------------------------------------------
    x_norm_h = x_norm.half()  # [B, N, N, dim], dtype float16

    # ------------------------------------------------------------------
    # 3. Concatenated projection + gate weights (FP16)
    # ------------------------------------------------------------------
    # left: (proj, gate) → (2*H, dim)
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16).contiguous()
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # 4. Linear transforms (single GEMM per side)
    # ------------------------------------------------------------------
    # Result shape: [B, N, N, 2*hidden_dim] (FP16)
    left_concat = F.linear(x_norm_h, left_concat_w)
    right_concat = F.linear(x_norm_h, right_concat_w)

    # ------------------------------------------------------------------
    # 5. Split into proj / gate and apply sigmoid to gate tensors
    # ------------------------------------------------------------------
    # Views – no extra copy
    left_proj = left_concat[..., :hidden_dim]          # [B,N,N,H]
    left_gate_pre = left_concat[..., hidden_dim:]     # [B,N,N,H]
    right_proj = right_concat[..., :hidden_dim]
    right_gate_pre = right_concat[..., hidden_dim:]

    # Gate activations (FP16)
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # ---------------------------------------------------------------
    # 6. Apply gates (in‑place to keep memory low)
    # ---------------------------------------------------------------
    left = left_proj * left_gate    # FP16, shape [B,N,N,H]
    right = right_proj * right_gate

    # ---------------------------------------------------------------
    # 7. Optional mask – fused via a tiny Triton kernel
    # ---------------------------------------------------------------
    if not nomask:
        # Convert mask to half and broadcast
        mask_h = mask.to(torch.float16).unsqueeze(-1)               # [B,N,N,1]
        mask_h = mask_h.expand(-1, -1, -1, hidden_dim).contiguous()  # [B,N,N,H]

        left = mask_mul(left, mask_h)
        right = mask_mul(right, mask_h)

    # ------------------------------------------------------------------
    # 8. Core contraction  (batched GEMM over the 'k' dimension)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Reshape to (B, H, N, N) for batched GEMM
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    # right needs the k‑dimension as the second axis => transpose last two dims
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]

    # Collapse batch & hidden, yielding (B*H, N, N) matrices
    left_flat = left_mat.view(B * hidden_dim, N, N)   # [B*H, N, N]
    right_flat = right_mat.view(B * hidden_dim, N, N)  # [B*H, N, N]

    # Batch matrix multiplication in FP16 (Tensor‑cores)
    out_flat = torch.bmm(left_flat, right_flat)      # [B*H, N, N]

    # Reshape back to [B, N, N, hidden_dim]
    out_h = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, H] – still FP16

    # ------------------------------------------------------------------
    # 9. Cast back to FP32 for the remaining (norm‑, gate‑, linear) ops
    # ------------------------------------------------------------------
    out = out_h.float()  # [B, N, N, H] FP32

    # ------------------------------------------------------------------
    # 10. Output layer‑norm (over hidden_dim)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, H] (FP32)

    # ------------------------------------------------------------------
    # 11. Output gate (sigmoid) – keep it in FP16 for speed, then cast
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16).contiguous()
    out_gate_h = torch.sigmoid(F.linear(x_norm_h, out_gate_w))  # FP16
    out_gate = out_gate_h.float()  # back to FP32

    out_norm = out_norm * out_gate  # FP32 element‑wise

    # ------------------------------------------------------------------
    # 12. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_final = F.linear(out_norm, weights["to_out.weight"])  # FP32, shape [B,N,N,dim]

    return out_final
