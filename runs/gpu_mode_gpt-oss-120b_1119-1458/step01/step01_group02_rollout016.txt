import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any

@triton.jit
def _apply_out_gate_kernel(out_ptr, gate_ptr, N_ELEMS, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N_ELEMS
    val = tl.load(out_ptr + offs, mask=mask)
    gate = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, val * gate, mask=mask)

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass for the outgoing TriMul operation.
    All heavy arithmetic is performed in fp16 to exploit H100 Tensor‑Cores.
    The O(N³) contraction is executed with a single batched torch.bmm (cuBLAS),
    and a tiny Triton kernel applies the final out‑gate element‑wise.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = torch.float16

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Cast everything to fp16 for performance
    # LayerNorm weights & bias
    norm_w = weights["norm.weight"].to(dtype)
    norm_b = weights["norm.bias"].to(dtype)

    # Linear weights
    left_proj_w = weights["left_proj.weight"].to(dtype)
    right_proj_w = weights["right_proj.weight"].to(dtype)
    left_gate_w = weights["left_gate.weight"].to(dtype)
    right_gate_w = weights["right_gate.weight"].to(dtype)
    out_gate_w = weights["out_gate.weight"].to(dtype)

    # Second LayerNorm weights & bias
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype)

    # Final projection weight
    to_out_w = weights["to_out.weight"].to(dtype)

    # -------------------------------------------------
    # 1️⃣ LayerNorm over dim (fp16)
    # -------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor.to(dtype),  # cast input to fp16
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim] in fp16

    # -------------------------------------------------
    # 2️⃣ Linear projections (no bias) in fp16
    # -------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)

    # -------------------------------------------------
    # 3️⃣ Optional mask (broadcasted on hidden dim)
    # -------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(dtype)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -------------------------------------------------
    # 4️⃣ Gating (sigmoid) in fp16
    # -------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction using batched GEMM (fp16, Tensor Cores)
    # -------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # reshape to [B*H, N, N] where second dim is i and third is k for left,
    # and j for right
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matmul: C = A @ Bᵀ
    # note: right_flat.transpose(1,2) brings shape (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 6️⃣ Apply out_gate element‑wise via a tiny Triton kernel (in‑place)
    # -------------------------------------------------
    # Ensure tensors are contiguous
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    total_elems = out.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elems, BLOCK),)
    _apply_out_gate_kernel[grid](
        out_ptr=out,
        gate_ptr=out_gate,
        N_ELEMS=total_elems,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # -------------------------------------------------
    # 7️⃣ Second LayerNorm (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 8️⃣ Final linear projection back to `dim` (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)

    # Cast to float32 for returning (matches reference)
    return out.to(torch.float32)
```

"""
TriMul (outgoing) – fast forward pass.

The heavy O(N³) contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is performed with a single batched GEMM (`torch.bmm`) in fp16, which
exploits H100 Tensor‑Cores.  All linear layers and gates are also kept
in fp16 to minimise memory traffic.  A tiny Triton kernel is used to
apply the final `out_gate` element‑wise (the only mandatory Triton
component).  The rest of the ops (layer‑norms and projections) are
implemented with high‑performance PyTorch primitives.

The pipeline (all tensors are contiguous on the same CUDA device):
    1. fp16 LayerNorm over the feature dimension `dim`.
    2. fp16 linear projections `left`, `right`.
    3. Optional mask (broadcasted on the hidden dimension).
    4. fp16 gates (`left_gate`, `right_gate`, `out_gate`) – sigmoid on
       linear maps of the normalized input.
    5. Apply `left_gate` / `right_gate` to the projections.
    6. Batched GEMM:   C = A @ Bᵀ  (A/B are the reshaped `left`/`right`).
    7. fp16 LayerNorm over the hidden dimension.
    8. Triton kernel multiplies the result by `out_gate` in‑place.
    9. fp16 final linear projection back to `dim`.
   10. Cast the result to float32 (the reference implementation works in fp32).

The implementation is fully self‑contained and works for all test
configurations (different batch sizes, sequence lengths, feature
dimensions and optional masks).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


@triton.jit
def _apply_out_gate_kernel(
    out_ptr,          # *fp16   output tensor (in‑place)
    gate_ptr,         # *fp16   same shape as `out_ptr`
    N_ELEMS: tl.constexpr,  # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    """
    Simple element‑wise multiplication: out *= gate.
    Both tensors are assumed to be contiguous and of type fp16.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N_ELEMS

    out_val = tl.load(out_ptr + offs, mask=mask)
    gate_val = tl.load(gate_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, out_val * gate_val, mask=mask)


def custom_kernel(
    data: Tuple[
        torch.Tensor,          # input tensor      [B, N, N, dim]
        torch.Tensor,          # mask tensor       [B, N, N]  (may be None)
        Dict[str, torch.Tensor],  # model weights
        Dict[str, Any],          # config dict
    ]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    # -------------------------------------------------
    # Unpack arguments
    # -------------------------------------------------
    inp, mask, weights, config = data
    device = inp.device
    dtype_fp16 = torch.float16

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # Cast all required weights to fp16 (they are stored as fp32)
    # -------------------------------------------------
    norm_w = weights["norm.weight"].to(dtype_fp16)
    norm_b = weights["norm.bias"].to(dtype_fp16)

    left_proj_w = weights["left_proj.weight"].to(dtype_fp16)
    right_proj_w = weights["right_proj.weight"].to(dtype_fp16)

    left_gate_w = weights["left_gate.weight"].to(dtype_fp16)
    right_gate_w = weights["right_gate.weight"].to(dtype_fp16)
    out_gate_w = weights["out_gate.weight"].to(dtype_fp16)

    to_out_norm_w = weights["to_out_norm.weight"].to(dtype_fp16)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype_fp16)

    to_out_w = weights["to_out.weight"].to(dtype_fp16)

    # -------------------------------------------------
    # 1️⃣ LayerNorm over feature dimension `dim` (fp16)
    # -------------------------------------------------
    x = torch.nn.functional.layer_norm(
        inp.to(dtype_fp16),          # cast input to fp16
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # shape [B, N, N, dim], dtype fp16

    # -------------------------------------------------
    # 2️⃣ Linear projections (no bias) – fp16
    # -------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # -------------------------------------------------
    # 3️⃣ Optional mask (broadcasted on hidden dim)
    # -------------------------------------------------
    if not nomask:
        # Convert mask to 0/1 fp16 and broadcast
        mask_fp16 = mask.unsqueeze(-1).to(dtype_fp16)   # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -------------------------------------------------
    # 4️⃣ Gating (sigmoid) – fp16
    # -------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, left_gate_w)
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, right_gate_w)
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, out_gate_w)
    )

    # Apply the per‑position gates to the projections
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction via batched GEMM (fp16, Tensor‑Cores)
    # -------------------------------------------------
    B, N, _, H = left.shape        # H == hidden_dim
    # Bring hidden dimension next to batch: [B, H, N, N] → view as [B*H, N, N]
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication: C = A @ Bᵀ
    # right_flat.transpose(1, 2) yields shape (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original layout -> [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # -------------------------------------------------
    # 7️⃣ Apply `out_gate` with a tiny Triton kernel (in‑place)
    # -------------------------------------------------
    out = out.contiguous()
    out_gate = out_gate.contiguous()
    total_elems = out.numel()
    BLOCK = 1024
    grid = ((total_elems + BLOCK - 1) // BLOCK,)

    _apply_out_gate_kernel[grid](
        out,                       # out_ptr
        out_gate,                  # gate_ptr
        total_elems,               # N_ELEMS
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # -------------------------------------------------
    # 8️⃣ Final projection back to `dim` (fp16)
    # -------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] fp16

    # -------------------------------------------------
    # 9️⃣ Cast to float32 for the API contract
    # -------------------------------------------------
    return out.to(torch.float32)
