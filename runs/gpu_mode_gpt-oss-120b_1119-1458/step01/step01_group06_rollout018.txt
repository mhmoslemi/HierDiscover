    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Step 1: layernorm
    x = torch.nn.functional.layer_norm(input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Step 2: projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if not nomask:
        mask_f = mask.unsqueeze(-1).to(left.dtype)   # broadcast to dim
        left = left * mask_f
        right = right * mask_f

    # Step 3: gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Cast everything to half for compute intensive ops
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Prepare for batched GEMM
    B, N, _, _ = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(B * hidden_dim, N, N)   # (BH, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)

    # Batched matrix multiplication on GPU (fp16 inputs, fp32 accumulation internally)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (BH, N, N)

    # Reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # Hidden -> output layernorm
    out = torch.nn.functional.layer_norm(out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],

    )

"""
TriMul (outgoing) forward – fast implementation.

Algorithm
---------
1. Input LayerNorm (torch, float32).
2. Linear projections → left / right (float32) and three gating projections.
3. Cast proj. & gate tensors to float16, apply sigmoid gates and optional mask.
4. Reshape to (B*H, N, N) and compute the Gram product
       out[b*h,i,j] = Σ_k left[b*h,i,k] * right[b*h,j,k]
   with a single batched cuBLAS GEMM (`torch.bmm`) on fp16 inputs.
   This matches the `einsum('...i k d, ...j k d -> ...i j d')` in the reference.
5. Reshape back to (B, N, N, H) and cast to float32.
6. Hidden‑to‑output LayerNorm (torch, float32).
7. Multiply by the outer gate (`out_gate`) using a tiny Triton element‑wise kernel.
8. Final linear projection to the original channel dimension (torch).

Only the element‑wise gate application is performed with Triton – the heavy
N³ reduction is done by a highly‑optimized cuBLAS batched GEMM, which on an
H100 easily meets the target latency.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Tiny Triton kernel: out = a * b   (in‑place on `c`)
# ----------------------------------------------------------------------
@triton.jit
def mul_kernel(a_ptr, b_ptr, c_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(c_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor: torch.Tensor [B, N, N, dim] (float32)
            mask:        torch.Tensor [B, N, N]       (bool / int / float)
            weights:     dict of torch.Tensors with the model parameters
            config:      dict containing at least "dim", "hidden_dim" and optionally "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B, N, N]; expand to [B, N, N, 1] and cast to match proj dtype
        mask_exp = mask.unsqueeze(-1).to(left.dtype)  # float32 → will be cast later
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating values (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 5. Cast to fp16 for the heavy reduction and apply gates
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)

    left = left * left_gate   # element‑wise
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Prepare tensors for the batched GEMM:
    #    (B, N, N, H) → (B*H, N, N)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left.view(B * hidden_dim, N, N)    # (BH, N, N)
    right_flat = right.view(B * hidden_dim, N, N)

    # ------------------------------------------------------------------
    # 7. Batched Gram matrix using cuBLAS (torch.bmm) – fp16 inputs,
    #    fp32 accumulation internally, result in fp16.
    # ------------------------------------------------------------------
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (BH, N, N)

    # ------------------------------------------------------------------
    # 8. Reshape back to (B, N, N, H) and promote to float32 for the rest
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 9. Hidden → output LayerNorm (float32)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 10. Apply the outer gate using a tiny Triton kernel (in‑place)
    # ------------------------------------------------------------------
    # flatten tensors to 1‑D
    out_flat = out.view(-1)
    out_gate_flat = out_gate.contiguous().view(-1).to(out_flat.dtype)  # cast to float32
    total_elements = out_flat.numel()
    BLOCK = 1024
    grid = ((total_elements + BLOCK - 1) // BLOCK,)

    mul_kernel[grid](out_flat, out_gate_flat, out_flat, total_elements, BLOCK=BLOCK)

    # ------------------------------------------------------------------
    # 11. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
