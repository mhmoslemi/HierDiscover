"""
TriMul (outgoing) – fast forward pass.

Key ideas
---------
1. Reduce the number of PyTorch Linear calls:
   * Concatenate the left/right projections → a single Linear producing
     a tensor of shape (..., 2·hidden_dim).
   * Concatenate the three gate projections → a single Linear producing
     (..., 3·hidden_dim).

2. Perform all cheap ops (masking, sigmoids, gate‑multiplication) in fp16.

3. The N³ reduction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is carried out by a custom Triton kernel that works directly on the
   original layout (B,N,N,hidden_dim).  No permute/contiguous copies are
   required – the kernel receives the four strides of the inputs and
   computes `batch` and `hidden` indices from the first program id.

4. The kernel uses fp16 inputs, fp32 accumulation (tensor‑core friendly)
   and is auto‑tuned over a few block‑size configurations.

5. Hidden‑dim LayerNorm and the final Linear are executed in float32
   (the cost is negligible compared with the N³ work).

The entry point `custom_kernel` follows the signature described in the
prompt and returns a float32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched Gram product   out[b,i,j,h] = Σ_k L[b,i,k,h] * R[b,j,k,h]
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        triton.Config({'BLOCK_M': 64,  'BLOCK_N': 64,  'BLOCK_K': 32}, num_warps=4),
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 128, 'BLOCK_K': 32}, num_warps=8),
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 128, 'BLOCK_K': 64}, num_warps=8),
    ],
    key=['N']  # pick the config based on the sequence length
)
@triton.jit
def tri_mul_gram_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,                                 # batch, seq_len, hidden_dim
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides (b,i,k,h)
    stride_rb, stride_rj, stride_rk, stride_rh,   # right strides (b,j,k,h) – note rj is the row‑stride (original dim‑2)
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides  (b,i,j,h)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)   # combined batch·hidden index
    pid_i = tl.program_id(1)   # tile row (i)
    pid_j = tl.program_id(2)   # tile col (j)

    # -------------------- decode batch and hidden index --------------------
    batch_id = pid0 // H
    hidden_id = pid0 % H

    # -------------------- tile offsets --------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 (more stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- reduction over k --------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block:   (i,k) slice for a fixed (batch, hidden)
        left = tl.load(
            left_ptr
            + batch_id * stride_lb
            + hidden_id * stride_lh
            + offs_i[:, None] * stride_li
            + offs_k[None, :] * stride_lk,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )   # fp16

        # right block:  (j,k) slice (note the row‑stride is rj, col‑stride is rk)
        right = tl.load(
            right_ptr
            + batch_id * stride_rb
            + hidden_id * stride_rh
            + offs_j[:, None] * stride_rj
            + offs_k[None, :] * stride_rk,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0
        )   # fp16

        # matrix‑multiply the tile and accumulate in fp32
        acc += tl.dot(left, right).to(tl.float32)

    # -------------------- write back --------------------
    tl.store(
        out_ptr
        + batch_id * stride_ob
        + hidden_id * stride_oh
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :]
    )


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Arguments:
        data = (input_tensor, mask, weights, config)

    Returns:
        Tensor of shape [batch, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32, keeps numerical stability)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for the heavy linear work
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Combined projection (left and right) – one matmul instead of two
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    proj_weight = torch.cat([left_proj_w, right_proj_w], dim=0).to(torch.float16)  # (2*H, dim)
    proj = F.linear(x_fp16, proj_weight)            # shape [B, N, N, 2*H]

    left = proj[..., :hidden_dim]                   # (B, N, N, H)
    right = proj[..., hidden_dim:]                  # (B, N, N, H)

    # -----------------------------------------------------------------
    # 4. Optional mask (broadcast on the hidden dimension)
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_fp16
        right = right * mask_fp16

    # -----------------------------------------------------------------
    # 5. Combined gating (left_gate, right_gate, out_gate)
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    gate_weight = torch.cat([left_gate_w, right_gate_w, out_gate_w], dim=0).to(torch.float16)  # (3*H, dim)
    gate = F.linear(x_fp16, gate_weight)          # (B, N, N, 3*H)

    left_gate  = torch.sigmoid(gate[..., :hidden_dim])
    right_gate = torch.sigmoid(gate[..., hidden_dim:2 * hidden_dim])
    out_gate   = torch.sigmoid(gate[..., 2 * hidden_dim:])

    # -----------------------------------------------------------------
    # 6. Apply gating (still fp16)
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 7. Heavy N³ reduction – Triton batched Gram product
    # -----------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)   # (B, N, N, H), fp16

    # strides for the original layout (B, N_i, N_k, H)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_ri, stride_rj, stride_rh = right.stride()
    # For the kernel we need:
    #   left: (b,i,k,h) -> strides (lb, li, lk, lh)
    #   right row = j (original dim‑2), col = k (original dim‑1)
    stride_rj = stride_rj                      # row‑stride (j)
    stride_rk = stride_ri                      # col‑stride (k)

    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Block sizes – the autotune decorator will pick the best one at compile time
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                         # combined batch·hidden dimension
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_gram_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # -----------------------------------------------------------------
    # 8. Hidden‑dim layer‑norm, output gating and final linear
    # -----------------------------------------------------------------
    # Convert to fp32 for the remaining cheap ops
    out_fp32 = out.to(torch.float32)

    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    out_norm = out_norm * out_gate.to(torch.float32)

    # final projection back to `dim`
    out_final = F.linear(out_norm, weights["to_out.weight"])

    return out_final
