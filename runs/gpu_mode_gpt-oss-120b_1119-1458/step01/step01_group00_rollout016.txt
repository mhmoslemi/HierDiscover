def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    assert left.is_cuda and right.is_cuda
    B,N,_,H = left.shape
    out = torch.empty_like(left)
    # Strides
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()
    # tile sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (B*H, (N+BLOCK_M-1)//BLOCK_M, (N+BLOCK_N-1)//BLOCK_N)
    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out

@triton.jit
def tri_mul_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_head = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    total_heads = B * H
    b = pid_head // H
    h = pid_head % H

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Left load: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            b * stride_left_b + h * stride_left_h + \
            offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Right load: shape (BLOCK_K, BLOCK_N) with k as row, j as column
        right_ptrs = right_ptr + \
            b * stride_right_b + h * stride_right_h + \
            offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_i
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    out_ptrs = out_ptr + \
        b * stride_out_b + h * stride_out_h + \
        offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def matmul_kernel(A, B, C,

                  stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,

"""
TriMul (outgoing) forward pass – Triton‐accelerated N³ contraction.

Algorithm
---------
1. Layer‑norm the input tensor (B×N×N×C) in float32.
2. Linear projections → left, right (B×N×N×H) in float32, then cast to float16.
3. (Optional) mask is applied (broadcast on H) in float16.
4. Compute three gates (left, right, out) from the normalized input in float32,
   cast to float16 and apply element‑wise.
5. Perform the dense N³ contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   using a custom Triton kernel that treats the hidden dimension H as an
   extra batch dimension, i.e. a batched matrix multiplication for each (b,h).
   The kernel works on the original right tensor (no explicit transpose) by
   swapping the stride indices for the K and J dimensions.  Inputs are
   float16, accumulation is float32, and the result is stored as float16.
6. Layer‑norm over the hidden dimension (float32) followed by an output gate
   (float16) and a final linear projection back to C (float32).

The kernel is tuned for NVIDIA H100: BLOCK_M=N‑BLOCK_N=128, BLOCK_K=32,
8 warps per block, and uses Tensor‑Core‑friendly fp16 arithmetic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (left @ rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,                     # const fp16 *
    right_ptr,                    # const fp16 *
    out_ptr,                      # fp16 *
    B, N, H,                      # int32 scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: out = left @ rightᵀ for each (batch, hidden)."""
    pid_head = tl.program_id(0)   # encodes (batch, hidden)
    pid_m = tl.program_id(1)      # block row index (i)
    pid_n = tl.program_id(2)      # block col index (j)

    total_heads = B * H
    b = pid_head // H
    h = pid_head % H

    # ------------------------------------------------------------------
    # Offsets for the output tile (i‑j plane)
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i indices
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j indices
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)   # k indices
        mask_k = offs_k < N

        # Load left tile   [BLOCK_M, BLOCK_K]  (i, k)
        left_ptrs = left_ptr + \
            b * stride_left_b + h * stride_left_h + \
            offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # Load right tile  [BLOCK_K, BLOCK_N]  (k, j) – note swapped strides
        right_ptrs = right_ptr + \
            b * stride_right_b + h * stride_right_h + \
            offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # Accumulate the dot product over k
        acc += tl.dot(left, right)   # shape (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # Store the output tile (convert back to fp16)
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_out_b + h * stride_out_h + \
        offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton‑accelerated batched matrix multiplication for the TriMul contraction.
    Arguments
    ----------
    left  : torch.Tensor of shape [B, N, N, H] (fp16)
    right : torch.Tensor of shape [B, N, N, H] (fp16) – the “right” operand.
            The kernel accesses it as rightᵀ via stride manipulation (no copy).

    Returns
    -------
    out : torch.Tensor of shape [B, N, N, H] (fp16)
    """
    assert left.is_cuda and right.is_cuda, "Tensors must be on CUDA"
    B, N, _, H = left.shape

    out = torch.empty_like(left)   # fp16, same shape

    # Strides in element units (required by Triton)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Tile size – good occupancy on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                                   # (batch * hidden) dimension
        (N + BLOCK_M - 1) // BLOCK_M,            # rows tiles
        (N + BLOCK_N - 1) // BLOCK_N,            # cols tiles
    )

    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,        # 8 warps → 256 threads per block
    )
    return out


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping of all Linear / LayerNorm weights
        - config_dict  : must contain \"dim\" and \"hidden_dim\"

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Load model parameters (they are stored as float32 tensors)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w   = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]        # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(input_tensor,
                          normalized_shape=(dim,),
                          weight=norm_w,
                          bias=norm_b,
                          eps=1e-5)          # [B, N, N, C]  (float32)

    # ------------------------------------------------------------------
    # 2. Linear projections (float32 -> cast to fp16 for the heavy GEMM)
    # ------------------------------------------------------------------
    left_fp32 = F.linear(x_norm, left_proj_w)   # [B, N, N, hidden_dim]
    right_fp32 = F.linear(x_norm, right_proj_w)

    left = left_fp32.to(torch.float16)
    right = right_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype=torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections, computed in float32)
    # ------------------------------------------------------------------
    left_gate_fp32  = torch.sigmoid(F.linear(x_norm, left_gate_w))
    right_gate_fp32 = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate_fp32   = torch.sigmoid(F.linear(x_norm, out_gate_w))

    # Cast gates to half for the element‑wise multiplications
    left_gate  = left_gate_fp32.to(torch.float16)
    right_gate = right_gate_fp32.to(torch.float16)
    out_gate   = out_gate_fp32.to(torch.float16)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. N³ contraction via Triton (batched GEMM per hidden channel)
    # ------------------------------------------------------------------
    out_fp16 = tri_mul_triton(left, right)      # [B, N, N, hidden_dim] (fp16)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over hidden_dim (float32 for stability)
    # ------------------------------------------------------------------
    out_fp32 = out_fp16.to(torch.float32)
    out_norm_fp32 = F.layer_norm(out_fp32,
                                 normalized_shape=(hidden_dim,),
                                 weight=to_out_norm_w,
                                 bias=to_out_norm_b,
                                 eps=1e-5)

    out_norm = out_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 7. Apply output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                         # half * half → half
    final = F.linear(gated.to(torch.float32), to_out_w) # back to float32

    # ------------------------------------------------------------------
    # Return result
    # ------------------------------------------------------------------
    return final
