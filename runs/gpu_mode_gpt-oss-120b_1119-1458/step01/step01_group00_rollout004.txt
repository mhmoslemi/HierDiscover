"""
TriMul (outgoing) forward pass – highly optimized implementation.

Algorithm
---------
1. Layer‑norm the input tensor (B×N×N×C) in FP32.
2. Cast to FP16 and apply three linear projections (left, right, gates) – no bias.
3. Optional 2‑D mask is broadcast on the hidden dimension.
4. Apply sigmoid gates and element‑wise masking.
5. Compute the dense N³ contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply (`left @ rightᵀ`) for each (b,h).
   Implemented in Triton with a tiled‑matrix‑multiply kernel that:
     * works on FP16 inputs,
     * accumulates in FP32,
     * uses double‑buffering (num_stages=2) and 8 warps per block,
     * chooses block sizes (128×128×32) tuned for H100.
6. Layer‑norm the result over the hidden dimension (FP32 for accuracy).
7. Apply the output gate (sigmoid) and a final linear projection back to C.
8. Return the output in FP32.

All heavy arithmetic (step 5) runs in Triton; the surrounding
operations use PyTorch's highly‑optimized kernels.
"""

):
    """
    Batched matrix multiplication:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, k, j, h]
    left  : (B, N, N, H)   – stored as (b,i,k,h)
    right : (B, N, N, H)   – transposed on the two N axes, i.e. (b,k,j,h)
    out   : (B, N, N, H)   – same layout as left
    """
    pid_head = tl.program_id(0)   # encodes (batch, hidden) -> (b, h)
    pid_m    = tl.program_id(1)   # block row index (i)
    pid_n    = tl.program_id(2)   # block col index (j)

    # --------------------------------------------------------------
    # Decode batch (b) and head (h) from flattened head id
    # --------------------------------------------------------------
    total_heads = B * H
    b = pid_head // H
    h = pid_head % H

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the K dimension in blocks
    # --------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- left tile (i,k) -------------------------------------------------
        left_ptrs = left_ptr + \
            b * stride_left_b + h * stride_left_h + \
            offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ---- rightᵀ tile (k,j) -----------------------------------------------
        # right_ptr is already the transposed view (k,j)
        right_ptrs = right_ptr + \
            b * stride_right_b + h * stride_right_h + \
            offs_k[:, None] * stride_right_i + offs_n[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # ---- accumulate --------------------------------------------------------
        acc += tl.dot(left, right)

        # Prefetch next K‑block (if any) – improves L2 hit rate
        if k + 1 < num_k_blocks:
            next_offs_k = (k + 1) * BLOCK_K + tl.arange(0, BLOCK_K)
            _ = tl.load(left_ptrs, mask=mask_m[:, None] & (next_offs_k < N),
                        other=0.0, cache_hint=tl.constant(tl.CacheHint.PREFETCH))
            _ = tl.load(right_ptrs, mask=(next_offs_k < N)[:, None] & mask_n[None, :],
                        other=0.0, cache_hint=tl.constant(tl.CacheHint.PREFETCH))

    # --------------------------------------------------------------
    # Write out the tile (cast back to FP16)
    # --------------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_out_b + h * stride_out_h + \
        offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def tri_mul_triton(left: torch.Tensor, right_t: torch.Tensor) -> torch.Tensor:
    """
    Triton‑accelerated batched GEMM:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right_t[b,k,j,h]

    Arguments
    ---------
    left      – torch.float16 tensor, shape (B,N,N,H)
    right_t   – torch.float16 tensor, shape (B,N,N,H) (right transposed on
                the two N axes). Must be contiguous.

    Returns
    -------
    out – torch.float16 tensor, shape (B,N,N,H)
    """
    assert left.is_cuda and right_t.is_cuda, "Tensors must be on CUDA"
    B, N, _, H = left.shape

    # Allocate output tensor
    out = torch.empty_like(left)

    # Strides in element counts (required by Triton)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rj, s_rh = right_t.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # --------- Block size selection (tuned for H100) --------------------
    # For very small N we shrink the tile to avoid over‑allocation.
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32

    grid = (
        B * H,                                 # (batch, head) dimension
        (N + BLOCK_M - 1) // BLOCK_M,          # rows
        (N + BLOCK_N - 1) // BLOCK_N,          # cols
    )

    tri_mul_kernel[grid](
        left,
        right_t,
        out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rj, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps = 256 threads per block (fits H100 well)
        num_stages=2,         # double‑buffering of tiles
    )
    return out


# ----------------------------------------------------------------------
# Public entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul module (AlphaFold‑3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : dict mapping string names to torch.Tensor weights
        - config_dict  : dict containing at least 'dim' and 'hidden_dim'

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper to move weights to the correct device/dtype (half‑precision)
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Load model parameters
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]          # (C,)
    norm_b = weights["norm.bias"]            # (C,)

    left_proj_w   = to_half(weights["left_proj.weight"])   # (hidden_dim, C)
    right_proj_w  = to_half(weights["right_proj.weight"])  # (hidden_dim, C)

    left_gate_w   = to_half(weights["left_gate.weight"])   # (hidden_dim, C)
    right_gate_w  = to_half(weights["right_gate.weight"])  # (hidden_dim, C)
    out_gate_w    = to_half(weights["out_gate.weight"])    # (hidden_dim, C)

    to_out_norm_w = to_half(weights["to_out_norm.weight"]) # (hidden_dim,)
    to_out_norm_b = to_half(weights["to_out_norm.bias"])   # (hidden_dim,)

    to_out_w      = to_half(weights["to_out.weight"])      # (C, hidden_dim)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32 for numeric stability)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B,N,N,C] float32

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy lifting
    # ------------------------------------------------------------------
    x = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias, FP16)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask may be bool or float – convert to FP16 and add a trailing dim
        mask_f = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # [B,N,N,hidden_dim]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # [B,N,N,hidden_dim]

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. N³ contraction (batched GEMM) via Triton
    # ------------------------------------------------------------------
    # rightᵀ on the two N axes – layout becomes (B, K, J, H)
    right_t = right.permute(0, 2, 1, 3).contiguous()
    out = tri_mul_triton(left, right_t)               # [B,N,N,hidden_dim] (FP16)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over hidden_dim (FP32 for accuracy)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=1e-5,
    )  # [B,N,N,hidden_dim] FP32
    out_norm = out_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 7. Apply output gate and final linear projection back to C
    # ------------------------------------------------------------------
    gated = out_norm * out_gate
    final = F.linear(gated, to_out_w)               # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # Return as FP32 (as required by the spec)
    # ------------------------------------------------------------------
    return final.to(torch.float32)
