def custom_kernel(data):
    x, mask, weights, config = data
    bs, N, _, dim = x.shape
    hidden = config["hidden_dim"]
    eps = 1e-5

    # input LayerNorm
    x = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # combine left_proj and left_gate weights into a single weight matrix.
    left_combined_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0)   # (2*hidden, dim)
    right_combined_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0)

    # linear pass for left and left_gate together
    left_all = F.linear(x, left_combined_weight)   # shape [B, N, N, 2*hidden]
    right_all = F.linear(x, right_combined_weight)

    # split and apply gating
    left = left_all[..., :hidden]    # fp32
    left_gate = torch.sigmoid(left_all[..., hidden:])   # fp32
    right = right_all[..., :hidden]
    right_gate = torch.sigmoid(right_all[..., hidden:])

    # optional mask
    if not config.get("nomask", True):
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # cast to fp16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Compute out = left * right.T across K dimension using Triton kernel
    out_fp16 = _triton_matmul_optimized(left_fp16, right_fp16, N, hidden)

    # back to fp32
    out = out_fp16.to(torch.float32)

    # to_out_norm
    out = F.layer_norm(out, (hidden,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # out gate
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # fp32
    out = out * out_gate

    # final linear
    out = F.linear(out, weights["to_out.weight"])

    return out

    A_ptr, B_ptr, C_ptr,
    stride_b0, stride_i, stride_k, stride_h,
    stride_B0, stride_kb, stride_jb,

    a_base = A_ptr + b * stride_b0 + h * stride_h
    b_base = B_ptr + b * stride_B0 + h * stride_h

"""
TriMul – “outgoing” triangle multiplicative update (AlphaFold‑3).

The heavy N³ reduction
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
is performed by a single Triton kernel.  The kernel works on the original
layout of the tensors [B, N, N, H] (no costly permute/contiguous copies),
processes every (batch, hidden) pair as an independent matrix‑multiply,
and accumulates in fp32 while loading in fp16.  All other
operations (LayerNorm, Linear‑projections, sigmoid‑gates, masking,
output‑norm and final projection) are done with PyTorch in fp32
and the result is cast back to fp32 at the end.

The implementation also fuses the two Linear‑projections (proj+gate)
per side into a single matmul to cut the number of passes over the
large input tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the outgoing update
# ----------------------------------------------------------------------
@triton.jit
def _outgoing_tri_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                                        # pointers
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,         # left strides
    stride_b_batch, stride_b_k, stride_b_j, stride_b_h,         # right strides
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,         # out strides
    M, N, K, H,                                                # matrix sizes, hidden dim
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes, for every (batch, hidden) pair:
        C[b,h,i,j] = Σ_k A[b,i,k,h] * B[b,k,j,h]

    A : (B, N, N, H)  (left  projection, fp16)
    B : (B, N, N, H)  (right projection, fp16, *already transposed* on K‑J)
    C : (B, N, N, H)  (output, fp16)
    """
    pid_m = tl.program_id(0)          # block index over output rows (i)
    pid_n = tl.program_id(1)          # block index over output cols (j)
    pid_ph = tl.program_id(2)         # combined (batch, hidden) index

    # ------------------------------------------------------------------
    # decode (batch, hidden) from the linear pid
    # ------------------------------------------------------------------
    h = pid_ph % H                     # hidden index
    b = pid_ph // H                    # batch index

    # ------------------------------------------------------------------
    # base pointers for this (b, h) pair
    # ------------------------------------------------------------------
    a_base = A_ptr + b * stride_a_batch + h * stride_a_h
    b_base = B_ptr + b * stride_b_batch + h * stride_b_h
    c_base = C_ptr + b * stride_c_batch + h * stride_c_h

    # ------------------------------------------------------------------
    # tile offsets for the current block
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # K‑loop (reduction over the third dimension)
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        start_k = kk * BLOCK_K
        offs_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)                     # fp16 load

        # B tile: shape (BLOCK_K, BLOCK_N)  (right is already K‑J transposed)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)                     # fp16 load

        # matrix‑multiply (fp16 → fp32 accumulation)
        acc += tl.dot(a, b).to(tl.float32)

    # ------------------------------------------------------------------
    # write result back (fp16)
    # ------------------------------------------------------------------
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def _triton_outgoing_matmul(left_fp16: torch.Tensor,
                            right_fp16: torch.Tensor,
                            seq_len: int,
                            hidden_dim: int,
                            BLOCK_M: int = 64,
                            BLOCK_N: int = 64,
                            BLOCK_K: int = 32) -> torch.Tensor:
    """
    left/right: [B, N, N, H] (fp16)
    returns out:   [B, N, N, H] (fp16)
    """
    B = left_fp16.shape[0]
    N = seq_len
    H = hidden_dim

    # allocate output tensor
    out = torch.empty_like(left_fp16)

    # strides of the original tensors (they are contiguous in dim order)
    stride_a_batch = left_fp16.stride(0)
    stride_a_i     = left_fp16.stride(1)
    stride_a_k     = left_fp16.stride(2)
    stride_a_h     = left_fp16.stride(3)          # = 1

    # right needs the K‑J transposed view; we do NOT materialise it,
    # we just reinterpret the strides.
    stride_b_batch = right_fp16.stride(0)
    # right[b, j, k, h] -> we need K as dim‑2 and J as dim‑1
    stride_b_k = right_fp16.stride(2)            # stride over k (original 2nd N dim)
    stride_b_j = right_fp16.stride(1)            # stride over j (original 1st N dim)
    stride_b_h = right_fp16.stride(3)            # = 1

    # output strides
    stride_c_batch = out.stride(0)
    stride_c_i     = out.stride(1)
    stride_c_j     = out.stride(2)
    stride_c_h     = out.stride(3)                # = 1

    # launch configuration
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks over i  (M)
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks over j  (N)
        B * H,                          # one program per (batch, hidden)
    )

    _outgoing_tri_matmul_kernel[grid](
        left_fp16, right_fp16, out,
        stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
        stride_b_batch, stride_b_k, stride_b_j, stride_b_h,
        stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
        N, N, N, H,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8, num_stages=3
    )
    return out


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3 style)
    using a custom Triton kernel for the N³ reduction.
    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
            - mask         : torch.Tensor [B, N, N]     (bool / float)
            - weights      : dict of torch.Tensors containing all model weights
            - config       : dict with keys
                * dim          – input / output channel dimension
                * hidden_dim   – inner hidden dimension
                * nomask (bool) – if True, the mask is ignored
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]                # input & output channel dimension C
    hidden = config["hidden_dim"]      # hidden dimension H
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  fp32

    # ------------------------------------------------------------------
    # 2. Fused linear + gate for the “left” side
    #    (proj  + gate are both  Linear(dim → hidden) → we concat the weights)
    # ------------------------------------------------------------------
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    )  # (2*hidden, dim)
    left_all = F.linear(x, left_combined_w)  # [B, N, N, 2*hidden]  fp32
    left = left_all[..., :hidden]            # projection
    left_gate = torch.sigmoid(left_all[..., hidden:])  # gate (fp32)

    # ------------------------------------------------------------------
    # 3. Fused linear + gate for the “right” side
    # ------------------------------------------------------------------
    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    )
    right_all = F.linear(x, right_combined_w)
    right = right_all[..., :hidden]
    right_gate = torch.sigmoid(right_all[..., hidden:])

    # ------------------------------------------------------------------
    # 4. Optional mask (applied after projection & gating)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask_exp = mask.unsqueeze(-1).to(left.dtype)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 5. Apply the element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Cast to fp16 and invoke the Triton kernel
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    seq_len = input_tensor.shape[1]   # N
    out_fp16 = _triton_outgoing_matmul(
        left_fp16,
        right_fp16,
        seq_len,
        hidden,
        BLOCK_M=64,
        BLOCK_N=64,
        BLOCK_K=32,
    )  # [B, N, N, H]  fp16

    # back to fp32 for the remaining pipeline
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 8. Output gate (sigmoid of a linear over the *original* x)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # fp32
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9. Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, dim]

    return out
