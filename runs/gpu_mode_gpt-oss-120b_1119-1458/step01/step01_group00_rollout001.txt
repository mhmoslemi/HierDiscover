"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1. Layer‑norm the input tensor `x` (B×N×N×C) in fp32.
2. Linear projections (no bias) → `left` and `right` (B×N×N×H) in fp16.
3. Optional mask (B×N×N) is applied to both tensors (broadcast on H).
4. Gated multiplicative update:
   left  = left  * sigmoid(Linear(x, left_gate_w))
   right = right * sigmoid(Linear(x, right_gate_w))
5. Heavy N³ contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
   Implemented as a batched matrix‑multiply `left @ rightᵀ` for each
   head h using a custom Triton kernel (the only part that must be in
   Triton).
6. Layer‑norm over the hidden dimension H.
7. Apply the output gate (sigmoid(Linear(x, out_gate_w))) and a final
   linear projection back to the original channel dimension C.
8. Return the result in fp32.

The Triton kernel works on a per‑head (batch×head) tile and multiplies
tiles of size BLOCK_M × BLOCK_N with a K‑blocking of BLOCK_K.  The
grid is `(batch*H, ceil(N/BLOCK_M), ceil(N/BLOCK_N))`.  All heavy
computations use fp16 inputs with fp32 accumulation to keep numerical
stability while exploiting Tensor‑cores on the H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the N³ contraction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,            # const half *
    right_ptr,           # const half *
    out_ptr,             # half *
    B, N, H,             # int32 metadata
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_k, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right_t[b,k,j,h]
    where ``right_t`` is the (k,j) view of ``right`` (i.e. ``right.permute(0,2,1,3)``).
    """
    pid_head = tl.program_id(0)   # encodes (batch, head)
    pid_m    = tl.program_id(1)   # tile row index (i)
    pid_n    = tl.program_id(2)   # tile col index (j)

    # ------------------------------------------------------------------
    # Decode batch (b) and head (h) from a flattened head id
    # ------------------------------------------------------------------
    total_heads = B * H
    b = pid_head // H
    h = pid_head % H

    # ------------------------------------------------------------------
    # Offsets for the current output tile (i‑tile, j‑tile)
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in BLOCK_K chunks
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            b * stride_left_b + h * stride_left_h + \
            offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # rightᵀ tile (k,j): shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + \
            b * stride_right_b + h * stride_right_h + \
            offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(left, right)

    # ------------------------------------------------------------------
    # Store result tile (convert back to fp16)
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_out_b + h * stride_out_h + \
        offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def tri_mul_triton(left: torch.Tensor, right_t: torch.Tensor) -> torch.Tensor:
    """
    Triton‑accelerated contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right_t[b,k,j,h]

    Parameters
    ----------
    left : torch.Tensor
        Tensor of shape (B, N, N, H) in fp16.
    right_t : torch.Tensor
        Tensor of shape (B, N, N, H) in fp16 (already transposed on the
        two N‑axes, i.e. ``right.permute(0,2,1,3)``).

    Returns
    -------
    torch.Tensor
        Tensor of shape (B, N, N, H) in fp16.
    """
    assert left.is_cuda and right_t.is_cuda
    B, N, _, H = left.shape

    out = torch.empty_like(left)  # fp16 output

    # Strides in element units (required by Triton)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_rk, s_rj, s_rh = right_t.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Tile sizes – tuned for H100 Tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32      # a multiple of 8 works best with fp16 Tensor‑cores

    grid = (
        B * H,                               # one program per (batch, head)
        (N + BLOCK_M - 1) // BLOCK_M,        # rows
        (N + BLOCK_N - 1) // BLOCK_N,        # cols
    )

    tri_mul_kernel[grid](
        left, right_t, out,
        B, N, H,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_rk, s_rj, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # H100 reaches high occupancy with 8 warps
        num_stages=2,         # double‑buffering of loads
    )
    return out

    # Unpack inputs
    # ------------------------------------------------------------------

    device = x.device
    B, N, _, C = x.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # Helper to move weight tensors to the right dtype/device
    # ------------------------------------------------------------------

        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w   = to_half(weights["left_proj.weight"])
    right_proj_w  = to_half(weights["right_proj.weight"])
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])
    to_out_norm_w = to_half(weights["to_out_norm.weight"])
    to_out_norm_b = to_half(weights["to_out_norm.bias"])
    to_out_w      = to_half(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(x,
                          normalized_shape=(dim,),
                          weight=norm_w,
                          bias=norm_b,
                          eps=1e-5)                    # [B,N,N,C] fp32

    # Cast to half for the heavy portion
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) → left, right  [B,N,N,H] fp16
    # ------------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)     # (B,N,N,hidden_dim)
    right = F.linear(x_h, right_proj_w)    # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask may be bool or float; convert to fp16 and broadcast
        mask_f = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))   # [B,N,N,H] fp16

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. N³ contraction via Triton (batched matmul per head)
    # ------------------------------------------------------------------
    # Transpose the inner N axes of right for the matmul:
    # right_t[b, k, j, h] = right[b, j, k, h]
    right_t = right.permute(0, 2, 1, 3).contiguous()
    out = tri_mul_triton(left, right_t)                # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over the hidden dimension (float32 internally)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm_fp32 = F.layer_norm(out_fp32,
                                 normalized_shape=(hidden_dim,),
                                 weight=to_out_norm_w.to(torch.float32),
                                 bias=to_out_norm_b.to(torch.float32),
                                 eps=1e-5)
    out_norm = out_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 7. Apply output gate and final projection back to dim
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                      # fp16
    final = F.linear(gated, to_out_w)                # fp16, shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # Return in float32 as required by the specification
    # ------------------------------------------------------------------
    return final.to(torch.float32)
