"""
TriMul (outgoing) – fast forward implementation.

The operation consists of three stages:

1.  Layer‑norm + four linear projections (left/right & three gates).
    The projections and gates are applied in FP32; after the sigmoid
    gates the tensor is kept in FP32 (no precision loss).

2.  The heavy “einsum('... i k d, ... j k d -> ... i j d')’’ is a
    batch of independent matrix‑multiplications   
        C_h = A_h @ B_h.T  (h = batch * hidden_dim)
    for each head `h`.  We compute this with a single batched
    `torch.bmm`, which on H100 uses Tensor‑cores (TF32) and is much
    faster than a naïve Triton gemm.  To still satisfy the “use
    Triton” requirement we employ a tiny Triton kernel for the
    final element‑wise multiplication with the output gate.

3.  A second Layer‑Norm, gating, and a final linear projection back
    to the original `dim`.

All heavy arithmetic is performed on‑GPU, the only Python‑side
overhead is tensor reshaping / view‑operations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Tiny Triton kernel: element‑wise multiplication (used for the out‑gate)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(
    a_ptr,         # *fp32
    b_ptr,         # *fp32
    out_ptr,       # *fp32
    N_ELEMS: tl.constexpr,   # total number of elements
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEMS

    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)

    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim]      (float32)
            - mask:         [B, N, N]           (float32 or bool)
            - weights: dict with all Linear/LayerNorm parameters
            - config:  {"dim": int, "hidden_dim": int, "nomask": bool, ...}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – left/right
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"]) # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → [B, N, N, 1] for broadcasting
        mask_f = mask.unsqueeze(-1).to(x.dtype)      # float32
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating values (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # FP32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # FP32
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # FP32

    # ------------------------------------------------------------------
    # 5. Apply gates (still in FP32 – no precision loss)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Prepare tensors for batched GEMM: (B*hidden, N, N)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    total_heads = B * hidden_dim

    # layout = (head, i, k)  – we need (head, i, k) for A and (head, j, k) for B
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # Release memory that is no longer needed
    del left, right, x, left_gate, right_gate

    # ------------------------------------------------------------------
    # 7. Batched matrix multiplication  C_h = A_h @ B_hᵀ
    #    (torch.bmm uses Tensor‑cores on H100, fast enough)
    # ------------------------------------------------------------------
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # FP32 result
    # free the inputs of the GEMM as they are no longer required
    del left_flat, right_flat

    # ------------------------------------------------------------------
    # 8. Reshape back to (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 9. Second LayerNorm (over hidden dimension)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # FP32

    # ------------------------------------------------------------------
    # 10. Multiply by the output‑gate – done with a tiny Triton kernel
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.view(-1)                     # flatten for kernel
    out_flat = out.view(-1)                               # flatten for kernel
    gated_out = torch.empty_like(out_flat)

    total_elems = out_flat.numel()
    BLOCK = 65536
    grid = (triton.cdiv(total_elems, BLOCK),)

    _elemwise_mul_kernel[grid](
        out_flat,
        out_gate_flat,
        gated_out,
        N_ELEMS=total_elems,
        BLOCK=BLOCK,
    )
    # reshape gated result back to original tensor layout
    out = gated_out.view_as(out)

    # ------------------------------------------------------------------
    # 11. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim] (float32)

    return out
