def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    device = input_tensor.device
    # LayerNorm over dim
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=eps)
    # Cast x to fp16 for following matmuls
    x_fp16 = x.to(torch.float16)
    B, N, _, _ = x_fp16.shape

    # Concatenate weights for linear and gates
    # All weight tensors shape (hidden_dim, dim)
    w_left = weights["left_proj.weight"]
    w_right = weights["right_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]
    weight_concat = torch.cat([w_left, w_right,
                               w_left_gate, w_right_gate,
                               w_out_gate], dim=0).to(torch.float16)

    # Combine linear operations in one matmul
    # Reshape to (B*N*N, dim)
    x_flat = x_fp16.view(-1, dim)
    combined = F.linear(x_flat, weight_concat)  # (B*N*N, 5*hidden_dim) fp16

    # Split combined
    splits = torch.split(combined, hidden_dim, dim=-1)  # tuple of 5 tensors
    left_proj = splits[0].view(B, N, N, hidden_dim)
    right_proj = splits[1].view(B, N, N, hidden_dim)
    left_gate_raw = splits[2].view(B, N, N, hidden_dim)
    right_gate_raw = splits[3].view(B, N, N, hidden_dim)
    out_gate_raw = splits[4].view(B, N, N, hidden_dim)

    # Gates
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)  # will stay fp16 for now

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    if not nomask:
        mask_f = mask.unsqueeze(-1).to(torch.float16)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Prepare for batched GEMM (head dimension = B * hidden_dim)
    total_heads = B * hidden_dim
    # Permute to (B, hidden, N, N) then view
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # Output buffer (fp32)
    out_flat = torch.empty((total_heads, N, N), dtype=torch.float32, device=device)

    # Launch Triton kernel for matrix multiplication
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        total_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # shape [B,N,N,hidden]

    # LayerNorm over hidden_dim
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Apply out_gate (cast to float32)
    out = out * out_gate.to(out.dtype)  # out_gate fp16 broadcast

    # Final projection to dim
    # Use fp16 matmul for speed
    out = F.linear(out.to(torch.float16), weights["to_out.weight"].to(torch.float16))
    # Cast back to float32
    out = out.to(torch.float32)

    return out

"""
TriMul (outgoing) – fast Triton‑accelerated implementation
==========================================================

The heavy ``einsum('... i k d, ... j k d -> ... i j d')`` is rewritten as a
batch of independent matrix‑multiplications ``C = A @ Bᵀ`` where each head
corresponds to one hidden dimension of a batch element.  All five linear
projections (two value projections and three gating projections) are fused into
a single FP16 matmul to avoid repeated loads of the large input tensor.
The batched GEMM is executed by a small Triton kernel that works on 2‑D tiles
(128×128) with a K‑tile of 64 and accumulates in FP32, thereby exploiting the
tensor‑core units of an H100.  The remaining LayerNorms, element‑wise gates and
the final projection are performed with native PyTorch kernels (also in FP16
where possible).  The function returns a ``torch.float32`` tensor of shape
``[B, N, N, dim]``.

The implementation satisfies the required API:
    custom_kernel(data) -> torch.Tensor
where ``data`` is a tuple ``(input, mask, weights, config)``.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the ``einsum`` part.
# For each head h = b * hidden_dim + d we compute
#   out[h, i, j] = Σ_k left[h, i, k] * right[h, j, k]
# i.e. a matmul of shape (N, N) @ (N, N)ᵀ.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # *fp16  left  = (head, i, k)
    right_ptr,         # *fp16  right = (head, j, k)
    out_ptr,           # *fp32  output = (head, i, j)
    N,                 # i32   sequence length
    stride_lh, stride_li, stride_lk,   # i64 strides for left
    stride_rh, stride_ri, stride_rk,   # i64 strides for right
    stride_oh, stride_oi, stride_oj,   # i64 strides for out
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(0)          # which head (batch * hidden)
    pid_m = tl.program_id(1)          # block row (i)
    pid_n = tl.program_id(2)          # block col (j)

    # --------------------------------------------------------------
    # Offsets for the M and N dimensions and the corresponding masks
    # --------------------------------------------------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Pointers to the current head
    # --------------------------------------------------------------
    left_head_ptr = left_ptr + pid_h * stride_lh
    right_head_ptr = right_ptr + pid_h * stride_rh
    out_head_ptr = out_ptr + pid_h * stride_oh

    # --------------------------------------------------------------
    # Accumulator in FP32
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the K dimension in blocks
    # --------------------------------------------------------------
    num_blocks_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_blocks_k):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- left tile (M, K) -------------------------------------------------
        left_ptrs = (
            left_head_ptr
            + offs_i[:, None] * stride_li
            + offs_k[None, :] * stride_lk
        )
        left_mask = mask_i[:, None] & mask_k[None, :]
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0)   # fp16

        # ----- right tile (K, N) – note the transposed layout --------------------
        right_ptrs = (
            right_head_ptr
            + offs_k[:, None] * stride_rk          # K dimension first
            + offs_j[None, :] * stride_ri          # J dimension second
        )
        right_mask = mask_k[:, None] & mask_j[None, :]
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0)   # fp16

        # ----- dot product (uses tensor‑cores) ---------------------------------
        acc += tl.dot(left_tile, right_tile).to(tl.float32)

    # --------------------------------------------------------------
    # Store the result tile
    # --------------------------------------------------------------
    out_ptrs = out_head_ptr + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)

    # ------------------------------------------------------------------

    # 3) Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)          # fp16
    right_gate = torch.sigmoid(right_gate_raw)        # fp16
    out_gate = torch.sigmoid(out_gate_raw)            # fp16

    # ------------------------------------------------------------------
    # 4) Element‑wise gating (and optional mask)
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1]
        mask_f = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5) Prepare tensors for the batched GEMM (head = batch * hidden_dim)
    # ------------------------------------------------------------------
    total_heads = B * hidden_dim
    # (B, N, N, hidden) → (B, hidden, N, N) → (total_heads, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # Output buffer for the GEMM (FP32 accumulator)
    out_flat = torch.empty((total_heads, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 6) Launch the Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        total_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7) Reshape back to [B, N, N, hidden_dim] (float32)
    # ------------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # shape [B, N, N, hidden_dim] – already FP32

    # ------------------------------------------------------------------
    # 8) Second LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # still FP32

    # ------------------------------------------------------------------
    # 9) Apply the output gate (broadcasted over hidden_dim)
    # ------------------------------------------------------------------
    out = out * out_gate.to(out.dtype)   # out_gate cast to FP32 on‑the‑fly

    # ------------------------------------------------------------------
    # 10) Final projection back to ``dim`` (FP16 matmul → FP32 result)
    # ------------------------------------------------------------------
    out = F.linear(out.to(torch.float16), weights["to_out.weight"].to(torch.float16))
    out = out.to(torch.float32)

    return out
