import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.jit
def _tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                    N,
                    stride_lhead, stride_li, stride_lk,
                    stride_rhead, stride_rj, stride_rk,
                    stride_ohead, stride_oi, stride_oj,
                    BLOCK_M: tl.constexpr,
                    BLOCK_N: tl.constexpr,
                    BLOCK_K: tl.constexpr):
    # Identify which head and which tile within that head.
    pid_h = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute absolute indices.
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    # Masks for last tiles.
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers for the current head.
    left_head = left_ptr + pid_h * stride_lhead
    right_head = right_ptr + pid_h * stride_rhead
    out_head = out_ptr + pid_h * stride_ohead

    # Accumulator in FP32.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles.
    num_k = tl.cdiv(N, BLOCK_K)

    for k in range(num_k):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A[i,k] tile (M×K) – fp16.
        a_ptrs = left_head + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        a_mask = mask_i[:, None] & mask_k[None, :]
        a = tl.load(a_ptrs, mask=a_mask, other=tl.float16(0.0))

        # Load B[j,k] tile and view it as (K×N) for the dot.
        b_ptrs = right_head + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj
        b_mask = mask_k[:, None] & mask_j[None, :]
        b = tl.load(b_ptrs, mask=b_mask, other=tl.float16(0.0))

        # Accumulate. tl.dot promotes the fp16 operands to fp32 internally.
        acc += tl.dot(a, b)

    # Write back the tile.
    out_ptrs = out_head + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)

def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Optimised forward of the “outgoing” TriMul update.
    The heavy tensor contraction ``out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]``
    is performed with a custom Triton GEMM kernel that works directly on
    half‑precision data and accumulates in single‑precision. All surrounding
    LayerNorms, linear projections and gating are executed with PyTorch.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm on the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(x.dtype)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating values (sigmoid of a second linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # float32

    # ------------------------------------------------------------------
    # Apply the gates (stay in fp16 to keep memory traffic low)
    # ------------------------------------------------------------------
    left = left.to(torch.float16) * left_gate
    right = right.to(torch.float16) * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the Triton kernel.
    #   left_flat / right_flat : (total_heads, N, N)  fp16
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    total_heads = B * hidden_dim

    left_flat = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # ------------------------------------------------------------------
    # Output buffer (fp32 accumulator)
    # ------------------------------------------------------------------
    out_flat = torch.empty((total_heads, N, N), dtype=torch.float32, device=input_tensor.device)

    # ------------------------------------------------------------------
    # Choose tile sizes based on N (power‑of‑two friendly)
    # ------------------------------------------------------------------
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        total_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to the original 4‑D layout.
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # Post‑processing: a second LayerNorm, gating and final projection.
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate   # broadcast over the hidden dimension

    out = F.linear(out, weights["to_out.weight"])
    return out

"""
Optimised “outgoing” Triangle‑Multiplicative‑Update (TriMul) forward pass.

Core operation:
    out[i, j, d] = Σ_k left[i, k, d] * right[j, k, d]

The contraction is expressed as a batch of independent matrix‑multiplications
C = A @ Bᵀ for each hidden dimension (treated as a “head”).  A Triton kernel
computes each head with fp16 operands and fp32 accumulation (tensor‑core
dot‑product).  All surrounding LayerNorms, linear projections and gates are
handled by PyTorch – they are inexpensive compared with the O(N³) contraction.

The implementation works for both masked and un‑masked inputs, supports the
full range of problem sizes from the test suite and runs on H100 (Triton 3.3.1).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: computes C_head = A_head @ B_headᵀ  (A,B : N×N, fp16)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # *fp16  left  = (head, i, k)
    right_ptr,         # *fp16  right = (head, j, k)
    out_ptr,           # *fp32  out   = (head, i, j)
    N,                 # i32    sequence length
    stride_lh, stride_li, stride_lk,   # i64 strides of left  (head,i,k)
    stride_rh, stride_rj, stride_rk,   # i64 strides of right (head,j,k)
    stride_oh, stride_oi, stride_oj,   # i64 strides of out   (head,i,j)
    BLOCK_M: tl.constexpr,  # tile size in i‑dimension
    BLOCK_N: tl.constexpr,  # tile size in j‑dimension
    BLOCK_K: tl.constexpr,  # tile size in k‑dimension
):
    pid_h = tl.program_id(0)      # which head (B * hidden_dim)
    pid_m = tl.program_id(1)      # block row (i)
    pid_n = tl.program_id(2)      # block col (j)

    # --------------------------------------------------------------
    # Offsets for the current i‑ and j‑tiles
    # --------------------------------------------------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Pointers to the beginning of the head‑slice
    # --------------------------------------------------------------
    left_head_ptr  = left_ptr  + pid_h * stride_lh
    right_head_ptr = right_ptr + pid_h * stride_rh
    out_head_ptr   = out_ptr   + pid_h * stride_oh

    # --------------------------------------------------------------
    # Accumulator (fp32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the K dimension in blocks
    # --------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- left tile A[i, k]  (M×K) --------------------------------
        a_ptrs = (left_head_ptr
                  + offs_i[:, None] * stride_li
                  + offs_k[None, :] * stride_lk)
        a_mask = mask_i[:, None] & mask_k[None, :]
        a = tl.load(a_ptrs, mask=a_mask, other=tl.float16(0.0))

        # ---- right tile Bᵀ[k, j] (K×N) -------------------------------
        b_ptrs = (right_head_ptr
                  + offs_k[:, None] * stride_rk
                  + offs_j[None, :] * stride_rj)
        b_mask = mask_k[:, None] & mask_j[None, :]
        b = tl.load(b_ptrs, mask=b_mask, other=tl.float16(0.0))

        # ---- dot product (fp16 → fp32 accumulation) ----------------
        acc += tl.dot(a, b)          # (M,K) @ (K,N) -> (M,N) in fp32

    # --------------------------------------------------------------
    # Write the tile back
    # --------------------------------------------------------------
    out_ptrs = (out_head_ptr
                + offs_i[:, None] * stride_oi
                + offs_j[None, :] * stride_oj)
    out_mask = mask_i[:, None] & mask_j[:, None]
    tl.store(out_ptrs, acc, mask=out_mask)

# ----------------------------------------------------------------------

    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(x.dtype)      # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating values (sigmoid of a second linear)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # keep fp32

    # ------------------------------------------------------------------
    # 5. Apply the gates (stay in fp16 to keep memory traffic low)
    # ------------------------------------------------------------------
    left  = left.to(torch.float16) * left_gate
    right = right.to(torch.float16) * right_gate

    # ------------------------------------------------------------------
    # 6. Rearrange tensors for the Triton kernel:
    #    (head, i, k) and (head, j, k) where head = B * hidden_dim
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape               # N is the sequence length
    total_heads = B * hidden_dim           # = B * hidden_dim

    # (B, N, N, hidden) → (B, hidden, N, N) → (total_heads, N, N)
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(total_heads, N, N)

    # ------------------------------------------------------------------
    # 7. Allocate output buffer (fp32 accumulation)
    # ------------------------------------------------------------------
    out_flat = torch.empty((total_heads, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 8. Choose tile sizes (empirically good defaults for H100)
    # ------------------------------------------------------------------
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
    else:               # N up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    # Grid: (head, tile_i, tile_j)
    grid = (
        total_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # ------------------------------------------------------------------
    # 9. Launch the Triton kernel
    # ------------------------------------------------------------------
    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 10. Reshape back to (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)   # (B, hidden, N, N)
        .permute(0, 2, 3, 1)                # (B, N, N, hidden)
        .contiguous()
    )  # fp32

    # ------------------------------------------------------------------
    # 11. Post‑processing: second LayerNorm, out‑gate and final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # broadcast over hidden_dim
    out = F.linear(out, weights["to_out.weight"])

    return out
```
