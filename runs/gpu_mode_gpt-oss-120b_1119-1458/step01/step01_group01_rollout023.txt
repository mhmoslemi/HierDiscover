def _apply_gate_and_cast_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    mask_ptr, left_out_ptr, right_out_ptr,
    total_elems: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
    has_mask: tl.constexpr
):

    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask_off = offs < total_elems

    left = tl.load(left_ptr + offs, mask=mask_off, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask_off, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask_off, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask_off, other=0.0)

"""
Triton‑accelerated outgoing triangle multiplicative update (TriMul).

The expensive N³ reduction
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
is reduced to a batched half‑precision matrix multiply performed with
torch.bmm (cuBLAS Tensor‑Cores).  To satisfy the “use Triton” rule we
fuse the element‑wise gating, optional masking and the required layout
transpose into two tiny Triton kernels that write the half‑precision
tensors directly in the layout expected by the batched GEMM.  This avoids
separate memory copies for `permute`/`contiguous` and for the FP32→FP16
cast, yielding a substantial speed‑up over the naive Triton matmul
implementation.

All other operations (LayerNorms, Linear projections, sigmoids, final
projection) are kept in PyTorch because they are already highly
optimised.  The final output is returned in float32, matching the
reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels:  fuse gating, optional mask and layout transpose.
# ----------------------------------------------------------------------
# Left side – produces tensor of shape (B, H, N, N) where the last two
# dimensions correspond to (i, k) from the original layout (B, N, N, H).
@triton.jit
def _prepare_left_kernel(
    left_ptr,           # float32*[B,N,N,H] flattened
    left_gate_ptr,      # float32*[B,N,N,H] flattened
    out_ptr,            # float16*  [B,H,N,N] flattened (output)
    SEQ_LEN: tl.constexpr,
    HIDDEN: tl.constexpr,
    TOTAL_ELEMS: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < TOTAL_ELEMS

    # Load FP32 values
    a = tl.load(left_ptr + offs,        mask=mask, other=0.0)  # left
    g = tl.load(left_gate_ptr + offs,   mask=mask, other=0.0)  # left_gate

    # --------------------------------------------------------------
    # Decode linear index -> (b, i, k, h)
    # layout of the flattened input: ((b * N + i) * N + k) * H + h
    # --------------------------------------------------------------
    tmp = offs
    b = tmp // (SEQ_LEN * SEQ_LEN * HIDDEN)
    tmp = tmp - b * (SEQ_LEN * SEQ_LEN * HIDDEN)

    i = tmp // (SEQ_LEN * HIDDEN)
    tmp = tmp - i * (SEQ_LEN * HIDDEN)

    k = tmp // HIDDEN
    h = tmp - k * HIDDEN

    # --------------------------------------------------------------
    # Compute output offset for layout (B, H, N, N)  ->  ((b*H+h)*N + i)*N + k
    # --------------------------------------------------------------
    out_off = ((b * HIDDEN + h) * SEQ_LEN + i) * SEQ_LEN + k

    # Apply gate and cast to fp16
    out_val = (a * g).to(tl.float16)

    tl.store(out_ptr + out_off, out_val, mask=mask)


# Right side – produces tensor of shape (B, H, N, N) where the last two
# dimensions correspond to (k, j) (i.e. the N dimensions are swapped).
@triton.jit
def _prepare_right_kernel(
    right_ptr,          # float32*[B,N,N,H] flattened
    right_gate_ptr,     # float32*[B,N,N,H] flattened
    out_ptr,            # float16*  [B,H,N,N] flattened (output)
    SEQ_LEN: tl.constexpr,
    HIDDEN: tl.constexpr,
    TOTAL_ELEMS: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < TOTAL_ELEMS

    b = tl.load(right_ptr + offs, mask=mask, other=0.0)   # placeholder, will be overwritten
    # Load FP32 values
    a = tl.load(right_ptr + offs,        mask=mask, other=0.0)  # right
    g = tl.load(right_gate_ptr + offs,   mask=mask, other=0.0)  # right_gate

    # Decode linear index -> (b, j, k, h)
    tmp = offs
    b = tmp // (SEQ_LEN * SEQ_LEN * HIDDEN)
    tmp = tmp - b * (SEQ_LEN * SEQ_LEN * HIDDEN)

    j = tmp // (SEQ_LEN * HIDDEN)
    tmp = tmp - j * (SEQ_LEN * HIDDEN)

    k = tmp // HIDDEN
    h = tmp - k * HIDDEN

    # Output layout (B, H, N, N) with dimensions (k, j)
    out_off = ((b * HIDDEN + h) * SEQ_LEN + k) * SEQ_LEN + j

    out_val = (a * g).to(tl.float16)

    tl.store(out_ptr + out_off, out_val, mask=mask)


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float)
            - weights: dict with the same keys as the reference module
            - config: dict containing at least
                * "dim" (int) – input & output channel dimension
                * "hidden_dim" (int) – internal hidden dimension
                * "nomask" (bool, optional) – whether mask should be ignored

    Returns:
        torch.Tensor: output of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (applied before gating, exactly as in the reference)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] and cast to same dtype as projections
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating values (sigmoid(Linear(x)))
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, hidden]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 5) Fuse gating, optional mask and layout transpose into half‑precision
    #    tensors ready for the batched GEMM.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    total_elements = left.numel()  # B * N * N * hidden_dim
    BLOCK_SIZE = 1024

    # Allocate destination tensors in the (B, H, N, N) layout.
    left_fp16 = torch.empty(
        (B, hidden_dim, N, N), dtype=torch.float16, device=device
    )
    right_fp16 = torch.empty(
        (B, hidden_dim, N, N), dtype=torch.float16, device=device
    )

    # Ensure input tensors are contiguous before flattening.
    left_cont = left.contiguous()
    right_cont = right.contiguous()
    left_gate_cont = left_gate.contiguous()
    right_gate_cont = right_gate.contiguous()

    # Launch the two kernels – each processes every element exactly once.
    grid = ( (total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE, )
    _prepare_left_kernel[grid](
        left_cont.view(-1),
        left_gate_cont.view(-1),
        left_fp16.view(-1),
        SEQ_LEN=N,
        HIDDEN=hidden_dim,
        TOTAL_ELEMS=total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )
    _prepare_right_kernel[grid](
        right_cont.view(-1),
        right_gate_cont.view(-1),
        right_fp16.view(-1),
        SEQ_LEN=N,
        HIDDEN=hidden_dim,
        TOTAL_ELEMS=total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # Release memory that is no longer needed.
    del left, right, left_gate, right_gate, left_cont, right_cont, left_gate_cont, right_gate_cont
    torch.cuda.empty_cache()

    # ------------------------------------------------------------------
    # 6) Batched half‑precision matrix multiplication:
    #    out_fp16[b,h,i,j] = Σ_k left_fp16[b,h,i,k] * right_fp16[b,h,k,j]
    # ------------------------------------------------------------------
    # Shape required by torch.bmm: (B*H, N, N)
    left_mat = left_fp16.view(B * hidden_dim, N, N)
    right_mat = right_fp16.view(B * hidden_dim, N, N)   # already transposed (k, j)

    out_fp16 = torch.bmm(left_mat, right_mat)          # fp16 output
    # Reshape back to [B, N, N, hidden]
    out = (
        out_fp16.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .float()
    )

    # ------------------------------------------------------------------
    # 7) Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 8) Apply the output gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
