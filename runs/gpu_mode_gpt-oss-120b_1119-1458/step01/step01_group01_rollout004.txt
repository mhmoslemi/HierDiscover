"""
Triton‑accelerated outgoing Triangle Multiplicative Update (TriMul).

The heavy N³ reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed by a custom Triton kernel that batches the
(b, hidden) dimension (P = B * hidden_dim) and computes a
batched matrix‑multiplication for each hidden channel.  The
kernel works on fp16 data (to cut memory traffic) while
accumulating in fp32 for numerical stability.

All other operations (LayerNorm, Linear projections, sigmoids,
mask‑application and final output projection) are kept in
PyTorch – they are cheap compared to the N³ reduction.
The block sizes (128×128×64) are chosen for H100 tensor‑core
efficiency and to minimise the number of program launches.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel : batched matrix multiplication (fp16 → fp32 acc)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                     # pointers
    stride_ap, stride_am, stride_ak,          # strides for A (P, M, K)
    stride_bp, stride_bk, stride_bn,          # strides for B (P, K, N)
    stride_cp, stride_cm, stride_cn,          # strides for C (P, M, N)
    M, N, K,                                  # dimensions (all = seq_len)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B   where   A : (P, M, K) , B : (P, K, N) , C : (P, M, N)
    A, B, C are fp16 tensors, accumulation is fp32.
    P = batch * hidden_dim.
    """
    pid_m = tl.program_id(0)      # block index over M
    pid_n = tl.program_id(1)      # block index over N
    pid_p = tl.program_id(2)      # block index over P (batch * hidden)

    # -------------------- tile offsets ---------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this (p) slice
    a_base = A_ptr + pid_p * stride_ap
    b_base = B_ptr + pid_p * stride_bp
    c_base = C_ptr + pid_p * stride_cp

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- K loop ---------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        start_k = k * BLOCK_K
        offs_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile (M, K)
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)          # fp16

        # Load b tile (K, N)
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)          # fp16

        # Matrix multiply (fp16) → accumulate in fp32
        acc += tl.dot(a, b).to(tl.float32)

    # -------------------- store result ---------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def _triton_matmul(left: torch.Tensor,
                   right: torch.Tensor,
                   seq_len: int,
                   hidden_dim: int,
                   *,
                   BLOCK_M: int = 128,
                   BLOCK_N: int = 128,
                   BLOCK_K: int = 64) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel.
    left, right : [B, N, N, H] (fp16)
    Returns out : [B, N, N, H] (fp16)
    """
    B = left.shape[0]

    # (B, N, N, H) -> (B, H, N, N) and make contiguous
    left_t = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 2, 1).contiguous()   # (B, H, K, J)

    # Collapse batch*hidden -> P
    P = B * hidden_dim
    left_flat = left_t.view(P, seq_len, seq_len)
    right_flat = right_t.view(P, seq_len, seq_len)

    out = torch.empty_like(left_flat)

    # Strides (tensors are contiguous)
    stride_ap = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bp = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_cp = out.stride(0)
    stride_cm = out.stride(1)
    stride_cn = out.stride(2)

    # Grid: blocks over M, over N, over P
    grid = ((seq_len + BLOCK_M - 1) // BLOCK_M,
            (seq_len + BLOCK_N - 1) // BLOCK_N,
            P)

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out,
        stride_ap, stride_am, stride_ak,
        stride_bp, stride_bk, stride_bn,
        stride_cp, stride_cm, stride_cn,
        seq_len, seq_len, seq_len,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # tensor‑core friendly
        num_stages=3,
    )

    # Reshape back to (B, N, N, H)
    out = out.view(B, hidden_dim, seq_len, seq_len) \
             .permute(0, 2, 3, 1) \
             .contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator with a Triton‑accelerated
    N³ reduction.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor, shape [bs, N, N, dim] (float32)
            mask         : torch.Tensor, shape [bs, N, N] (bool/float)
            weights      : dict of model parameters
            config       : dict with "dim", "hidden_dim", "nomask" etc.

    Returns:
        torch.Tensor of shape [bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) – fp16 for speed / memory
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Optional mask (applied after projection)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask broadcasted over hidden dimension
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating: sigmoid(linear(x))  – keep same dtype as left/right
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(left.dtype)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(right.dtype)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # fp32, kept for later

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Heavy N³ reduction using Triton
    # ------------------------------------------------------------------
    seq_len = input_tensor.shape[1]  # N
    out_fp16 = _triton_matmul(
        left,
        right,
        seq_len=seq_len,
        hidden_dim=hidden_dim,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=64,
    )  # [B, N, N, hidden] fp16

    # back to fp32 for the remaining ops
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6) to_out_norm (LayerNorm over hidden dimension)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 7) Apply the output gate (broadcast over hidden dim)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to input dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, dim] float32

    return out
